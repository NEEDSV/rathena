//===== G2 isolated test harness (stack copy only, never in the repository load lists) =====
// @mbqainst <1|2> creates a solo (IM_CHAR) instance and enters it: 1 = LASTROOM 1@uns, 2 = CENTRALLAB 1@lab.
// @mbqaclear <1|2> calls need_marble_instance_clear() for the boss map of the caller's instance.
// @mbqaleave destroys the caller's instance and warps to prontera.
-	script	MBQA_Harness	-1,{
OnInit:
	setarray .key$[1], "LASTROOM", "CENTRALLAB";
	setarray .name$[1], "마지막 방", "중앙 실험실";
	setarray .map$[1], "1@uns", "1@lab";
	bindatcmd "mbqainst", strnpcinfo(3) + "::OnInst", 99, 100;
	bindatcmd "mbqaclear", strnpcinfo(3) + "::OnClear", 99, 100;
	bindatcmd "mbqaleave", strnpcinfo(3) + "::OnLeave", 99, 100;
	end;
OnInst:
	.@k = atoi(.@atcmd_parameters$[0]);
	if (.@k < 1 || .@k > 2) end;
	.@id = instance_create(.name$[.@k], IM_CHAR);
	if (.@id > 0) @MBQA_IID = .@id;
	.@e = instance_enter(.name$[.@k], -1, -1, getcharid(0), @MBQA_IID);
	dispbottom "[MBQA] inst k=" + .@k + " create=" + .@id + " iid=" + @MBQA_IID + " enter=" + .@e + " map=" + instance_mapname(.map$[.@k], @MBQA_IID);
	end;
OnClear:
	.@k = atoi(.@atcmd_parameters$[0]);
	if (.@k < 1 || .@k > 2) end;
	.@m$ = instance_mapname(.map$[.@k], @MBQA_IID);
	.@n = need_marble_instance_clear(.key$[.@k], .@m$);
	dispbottom "[MBQA] clear k=" + .@k + " map=" + .@m$ + " iid=" + @MBQA_IID + " credited=" + .@n;
	end;
OnLeave:
	if (@MBQA_IID > 0) instance_destroy @MBQA_IID;
	@MBQA_IID = 0;
	warp "prontera", 156, 191;
	end;
}

// G2-C: @mbqabgstart <mode 1-3> <token> <Guillaume names,> <Croix names,> (- = nobody), @mbqabgend <result 1|2|3>,
// @mbqabgauto (real auto reward function), @mbqabgnpc <team 1|2> (gid of the real reward NPC), @mbqabgstop (real admin force stop),
// @mbqabgreset (real abort + finish), @mbqabgcredit <type> <token> (second end call for the caller), @mbqabgwhere <name>, @mbqabgitems <name>
-	script	MBQA_BGHarness	-1,{
OnInit:
	setarray .map$[1], "bat_a01", "bat_b01", "bat_c01";
	setarray .type$[1], "Tierra", "Flavius", "KVM";
	setarray .gx[1], 50, 10, 52;
	setarray .gy[1], 374, 290, 129;
	setarray .cx[1], 42, 390, 147;
	setarray .cy[1], 16, 10, 55;
	setarray .npc$[1], "기욤 보상관#a01_a", "기욤 빈테나르#b01_a", "KVM 담당관#KVM01A";
	setarray .npc2$[1], "크루아 보상관#a01_b", "크루아 빈테나르#b01_b", "KVM 담당관#KVM01A";
	bindatcmd "mbqabgstart", strnpcinfo(3) + "::OnStart", 99, 100;
	bindatcmd "mbqabgend", strnpcinfo(3) + "::OnEnd", 99, 100;
	bindatcmd "mbqabgauto", strnpcinfo(3) + "::OnAuto", 99, 100;
	bindatcmd "mbqabgnpc", strnpcinfo(3) + "::OnNpc", 99, 100;
	bindatcmd "mbqabgstop", strnpcinfo(3) + "::OnStop", 99, 100;
	bindatcmd "mbqabgreset", strnpcinfo(3) + "::OnReset", 99, 100;
	bindatcmd "mbqabgcredit", strnpcinfo(3) + "::OnCredit", 99, 100;
	bindatcmd "mbqabgwhere", strnpcinfo(3) + "::OnWhere", 99, 100;
	bindatcmd "mbqabgitems", strnpcinfo(3) + "::OnItems", 99, 100;
	end;
OnStart:
	.@mode = atoi(.@atcmd_parameters$[0]);
	.@tok = atoi(.@atcmd_parameters$[1]);
	if (.@mode < 1 || .@mode > 3 || .@tok <= 0) end;
	.@gm = getcharid(3);
	.@m$ = .map$[.@mode];
	.@g = bg_create(.@m$, .gx[.@mode], .gy[.@mode], "", "");
	.@c = bg_create(.@m$, .cx[.@mode], .cy[.@mode], "", "");
	if (.@mode == 1) { $@TierraBG1_id1 = .@g; $@TierraBG1_id2 = .@c; $@TierraBG1 = 1; $@TierraBG1_Victory = 0; }
	else if (.@mode == 2) { $@FlaviusBG1_id1 = .@g; $@FlaviusBG1_id2 = .@c; $@FlaviusBG1 = 1; $@FlaviusBG1_Victory = 0; }
	else { $@KvM01BG_id1 = .@g; $@KvM01BG_id2 = .@c; $@KvM01BG = 1; $@KvM01BG_Victory = 0; }
	$@BG_RM_TOKEN = .@tok;
	$@BG_RM_MODE = .@mode;
	$@BG_RM_ACTIVE = 1;
	$@BG_RM_CYCLE_ACTIVE = 1;
	$@BG_RM_STATE = 3;
	callfunc "F_BG_RM_BeginRewardStage", .@mode;
	deletearray $@BG_RM_G_AID[0], 128; deletearray $@BG_RM_G_CID[0], 128; deletearray $@BG_RM_G_NAME$[0], 128;
	deletearray $@BG_RM_C_AID[0], 128; deletearray $@BG_RM_C_CID[0], 128; deletearray $@BG_RM_C_NAME$[0], 128;
	explode(.@gn$, .@atcmd_parameters$[2], ",");
	explode(.@cn$, .@atcmd_parameters$[3], ",");
	.@ng = getarraysize(.@gn$);
	.@nc = getarraysize(.@cn$);
	.@j = 0;
	for (.@i = 0; .@i < .@ng; .@i++) {
		.@cid = getcharid(0, .@gn$[.@i]);
		if (.@cid <= 0) continue;
		.@aid = getcharid(3, .@gn$[.@i]);
		.@joined += bg_join(.@g, .@m$, .gx[.@mode], .gy[.@mode], .@cid);
		$@BG_RM_G_AID[.@j] = .@aid; $@BG_RM_G_CID[.@j] = .@cid; $@BG_RM_G_NAME$[.@j] = .@gn$[.@i]; .@j++;
		if (attachrid(.@aid)) Bat_Team = 1;
		attachrid .@gm;
	}
	$@BG_RM_G_COUNT = .@j;
	.@j = 0;
	for (.@i = 0; .@i < .@nc; .@i++) {
		.@cid = getcharid(0, .@cn$[.@i]);
		if (.@cid <= 0) continue;
		.@aid = getcharid(3, .@cn$[.@i]);
		.@joined += bg_join(.@c, .@m$, .cx[.@mode], .cy[.@mode], .@cid);
		$@BG_RM_C_AID[.@j] = .@aid; $@BG_RM_C_CID[.@j] = .@cid; $@BG_RM_C_NAME$[.@j] = .@cn$[.@i]; .@j++;
		if (attachrid(.@aid)) Bat_Team = 2;
		attachrid .@gm;
	}
	$@BG_RM_C_COUNT = .@j;
	dispbottom "[MBQA] bgstart mode=" + .@mode + " token=" + .@tok + " g=" + .@g + " c=" + .@c + " gcount=" + $@BG_RM_G_COUNT + " ccount=" + $@BG_RM_C_COUNT + " joined=" + .@joined;
	end;
OnEnd:
	.@mode = $@BG_RM_MODE;
	.@r = atoi(.@atcmd_parameters$[0]);
	if (.@mode == 1) $@TierraBG1_Victory = .@r;
	else if (.@mode == 2) $@FlaviusBG1_Victory = .@r;
	else if (.@mode == 3) $@KvM01BG_Victory = .@r;
	$@MBQA_BG_RESULT = .@r;
	enablenpc .npc$[.@mode];
	if (.@mode < 3) enablenpc .npc2$[.@mode];
	dispbottom "[MBQA] bgend mode=" + .@mode + " result=" + .@r;
	end;
OnAuto:
	.@mode = $@BG_RM_MODE;
	if (.@mode == 3) .@n = callfunc("F_KvM_AutoRewardRandom");
	else .@n = callfunc("F_BG_RM_AutoRewardBadge", .@mode, .map$[.@mode], $@MBQA_BG_RESULT, .type$[.@mode]);
	dispbottom "[MBQA] bgauto mode=" + .@mode + " rewarded=" + .@n;
	end;
OnNpc:
	.@mode = $@BG_RM_MODE;
	.@t = atoi(.@atcmd_parameters$[0]);
	.@n$ = (.@t == 2) ? .npc2$[.@mode] : .npc$[.@mode];
	dispbottom "[MBQA] bgnpc mode=" + .@mode + " team=" + .@t + " gid=" + getnpcid(0, .@n$);
	end;
OnStop:
	donpcevent "NEED_BG_RM_Core::OnAdminForceStop";
	sleep2 500;
	dispbottom "[MBQA] bgstop cycle=" + $@BG_RM_CYCLE_ACTIVE + " token=" + $@BG_RM_TOKEN;
	end;
OnReset:
	.@mode = $@BG_RM_MODE;
	if (.@mode >= 1 && .@mode <= 3) {
		disablenpc .npc$[.@mode];
		if (.@mode < 3) disablenpc .npc2$[.@mode];
		callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
	}
	callfunc "F_BG_RM_Finish";
	$@BG_RM_CYCLE_ACTIVE = 0;
	$@BG_RM_ACTIVE = 0;
	dispbottom "[MBQA] bgreset mode=" + .@mode + " cycle=" + $@BG_RM_CYCLE_ACTIVE;
	end;
OnCredit:
	dispbottom "[MBQA] bgcredit rc=" + need_marble_bg_credit(.@atcmd_parameters$[0], atoi(.@atcmd_parameters$[1]));
	end;
// @mbqabgitems <name>: the existing BG rewards held by that character (badges, KVM badge, 6378, 399990, #BATTLEGROUND_POINT)
OnItems:
	.@aid = getcharid(3, .@atcmd_parameters$[0]);
	if (.@aid <= 0) end;
	.@gm = getcharid(3);
	if (attachrid(.@aid)) {
		.@s$ = "7828=" + countitem(7828) + " 7829=" + countitem(7829) + " 7773=" + countitem(7773) + " 6378=" + countitem(6378) + " 399990=" + countitem(399990) + " bgp=" + #BATTLEGROUND_POINT;
	}
	attachrid .@gm;
	dispbottom "[MBQA] bgitems name=" + .@atcmd_parameters$[0] + " " + .@s$;
	end;
OnWhere:
	.@aid = getcharid(3, .@atcmd_parameters$[0]);
	.@cid = getcharid(0, .@atcmd_parameters$[0]);
	.@bg = 0;
	if (.@aid > 0) {
		.@gm = getcharid(3);
		if (attachrid(.@aid)) { .@bg = getcharid(4); .@m$ = strcharinfo(3); getmapxy(.@mm$, .@x, .@y, BL_PC); .@team = Bat_Team; }
		attachrid .@gm;
	}
	dispbottom "[MBQA] bgwhere name=" + .@atcmd_parameters$[0] + " aid=" + .@aid + " cid=" + .@cid + " bg=" + .@bg + " map=" + .@m$ + " x=" + .@x + " y=" + .@y + " team=" + .@team;
	end;
}
