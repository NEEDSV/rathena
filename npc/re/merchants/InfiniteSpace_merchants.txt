lasa_in01,189,56,3	script	유물 강화사#pa0829_01	4_F_JOB_BLACKSMITH,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "아이템이 너무 많아서 대화를 진행할 수 없습니다.";
		mes "물품들을 정리하신 후 다시 시도해 주세요.";
		close;
	}
	disable_items;
	.@stone_id = 6905;
	.@stone_name$ = getitemname(.@stone_id);
	function equip_check;
	mes "[유물 강화사]";
	mes "무한의 공간에서 획득한 장비를 강화하고 싶다면 잘 찾아오셨습니다.";
	mes .@stone_name$ + "을 가지고 계신가요?";
	next;
	switch (select("인챈트 방법을 듣는다:장비를 인챈트한다:인챈트를 초기화한다:인챈트 리스트 확인")) {
		case 1:
			mes "[유물 강화사]";
			mes .@stone_name$ + "은 파로스 등대 아래의 무한의 공간을 탐험하면 얻을 수 있습니다.";
			next;
			mes "[유물 강화사]";
			mes "그 재료를 가져오시면 무한의 공간에서 얻은 장비에 특별한 힘을 부여해 드릴 수 있습니다.";
			next;
			mes "[유물 강화사]";
			mes "인챈트는 세 번째와 네 번째 슬롯에만 부여할 수 있습니다.";
			next;
			mes "[유물 강화사]";
			mes "인챈트할 때 장비가 파괴되지는 않지만, 인챈트를 초기화할 때는 장비가 파괴될 수 있습니다.";
			next;
			mes "[유물 강화사]";
			mes "인챈트가 필요할 때 언제든 다시 찾아오세요.";
			break;

		case 2:
			.@stone_id = 6905;
			.@fee = 20;
			mes "[유물 강화사]";
			mes "인챈트할 장비가 장착된 부위를 선택해 주세요.";
			next;
			switch (select("취소:무기:갑옷:신발:걸칠것:투구")) {
				case 1:
					mes "[유물 강화사]";
					mes "인챈트가 필요할 때 언제든 다시 찾아오세요.";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0;.@i < 4;.@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = getequipcardid(.@part,.@i);
			}
			if (.@card[2] > 0) {
				mes "[유물 강화사]";
				mes "이 장비에는 이미 인챈트가 두 개 부여되어 있어 더 이상 인챈트할 수 없습니다.";
				close;
			}
			switch (.@part) {
				case EQI_HAND_R:
					setarray .@enchant$,
					"4700,4701:4710,4711:4720,4721",
					"4811,4810,4809,4808,4820,4821,4822,4823:4815,4814,4813,4812,4826,4827,4828,4829:4832,4833,4834,4835,4836,4837,4838,4839";

				case EQI_SHOES:

				case EQI_HEAD_TOP:

				mes "[유물 강화사]";
				mes "인챈트가 필요할 때 언제든 다시 찾아오세요.";
				close;
			}
			.@type -= 2;
			mes "[유물 강화사]";
			if (.@slot == 3)

			else
				mes "그럼 두 번째 인챈트를 진행하겠습니다.";
			next;
			if (select("나중에 다시 온다:계속 진행한다") == 1) {
				mes "[유물 강화사]";
				mes "마음이 바뀌면 언제든 다시 찾아오세요.";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[유물 강화사]";
				mes .@stone_name$ + "이 부족합니다.";
				close;
			}
			explode(.@T$,.@enchant$[.@index],":");
			explode(.@TT$,.@T$[.@type],",");
			.@enchant = atoi(.@TT$[rand(getarraysize(.@TT$))]);
			.@card[.@slot] = .@enchant;
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
			specialeffect2 EF_REPAIRWEAPON;
			mes "[유물 강화사]";
			mes "완료되었습니다. 장비를 확인해 보세요.";
			break;
		
		case 3:
			.@stone_id = 6905;
			.@fee = 30;
			.@break_chance = 30;
			mes "[유물 강화사]";
			mes "인챈트를 초기화할 장비가 장착된 부위를 선택해 주세요.";
			next;
			switch (select("취소:무기:갑옷:신발:걸칠것:투구")) {

					mes "[유물 강화사]";

					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

					.@part = EQI_SHOES;
					break;

					.@part = EQI_GARMENT;
					break;

					.@part = EQI_HEAD_TOP;
					break;

			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0; .@i < 4; .@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = .@card[.@i];
			}
			if (.@card[3] == 0) {

				mes "이 장비에는 초기화할 인챈트가 없습니다.";
				close;
			}
			mes "[유물 강화사]";
			mes "초기화에는 " + .@stone_name$ + " " + .@fee + "개가 필요하며, ^FF000030% 확률로 장비가 파괴^000000됩니다.";
			mes "그래도 진행하시겠습니까?";
			next;
			if (select("취소:초기화한다") == 1) {
				mes "[유물 강화사]";
				mes "마음이 바뀌면 언제든 다시 찾아오세요.";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[유물 강화사]";
				mes .@stone_name$ + "이 부족합니다.";
				close;
			}
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			if (rand(1,100) > .@break_chance) {
				getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],0,0;
				specialeffect2 EF_REPAIRWEAPON;
				mes "[유물 강화사]";
				mes "완료되었습니다. 장비를 확인해 보세요.";
			} else {
				specialeffect2 EF_REFINEFAIL;
				mes "[유물 강화사]";
				mes "경고드렸던 대로 장비가 파괴되었습니다.";
			}
			break;

		case 4:
			callfunc "F_ShowInfiniteSpaceEnchantList";
			end;
	}
	close;

	function	equip_check	{
		setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128,15141,22075,20779,19033;
		if (!getarg(0)) {
			mes "[유물 강화사]";
			mes "선택한 부위에 장비가 착용되어 있지 않습니다.";
			close;
		}
		if (inarray(.@equip_id,getarg(0)) == -1) {
			mes "[유물 강화사]";
			mes "이 장비는 인챈트할 수 없습니다. 무한의 공간에서 획득하는 무한/균열 장비만 인챈트할 수 있습니다.";
			close;
		}
		return;
	}
Oninit:
	waitingroom "무한의 공간 인챈트",0;
	end;
}

function	script	F_ShowInfiniteSpaceEnchantGroup	{
	mes "^990000[" + getarg(0) + "]^000000";
