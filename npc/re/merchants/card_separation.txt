-	script	::CardSeparation_mal	-1,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes needtr("아이템 종류가 너무 많습니다. 아이템 종류를 줄인 뒤에 다시 시도해 주세요.", "You have too many kinds of objects. Let's try to continue after reducing those objects.");
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes needtr("무거운 물건을 너무 많이 들고 있어 진행할 수 없습니다. 무게를 줄인 뒤에 다시 시도해 주세요.", "Can't continue because you have too many heavy objects. Let's try to continue after reducing the weight.");
		close;
	}
	if (strnpcinfo(1) == "제레미") {
		set .@Jeremy,1;
		set .@n$, needtr("[제레미]", "[Jeremy]");
		setarray .@equip_name$[0],  needtr("갑옷", "Armor"),  needtr("신발", "Shoes"),  needtr("걸칠것", "Garment"), needtr("투구 상단", "Upper Hat"), needtr("투구 중단", "Middle Hat"), needtr("악세서리(오른쪽)", "Accessory (Right)"), needtr("악세서리(왼쪽)", "Accessory (Left)");
		setarray .@equip_slot[0], EQI_ARMOR, EQI_SHOES, EQI_GARMENT, EQI_HEAD_TOP, EQI_HEAD_MID, EQI_ACC_L, EQI_ACC_R;

		mes .@n$;
		mes needtr("오랜만이네요~", "Long time no see~");
		mes needtr("갑옷, 신발, 걸칠것, 투구에서 카드를 분리하는 새로운 기술을 배웠습니다. 한번 해보시겠어요?", "I have learned a new skill that separates cards from Armor, Shoes, Garment and Headgear. Do you want to try it?");
	} else {
		set .@Jeremy,0;
		set .@n$, needtr("[리처드]", "[Richard]");
		setarray .@equip_name$[0], needtr("왼손", "Left hand"), needtr("오른손", "Right hand");
		setarray .@equip_slot[0],   EQI_HAND_L,   EQI_HAND_R;

		mes .@n$;
		mes needtr("바보 같은 제레미는 카드가 손상될까 봐 무기나 방패는 건드리고 싶어 하지 않아요. 그래서 제가 무기와 방패를 위한 카드 분리 기술을 준비했죠...", "Silly Jeremy does not want to touch weapons and shields because of picking several cards that might be damaged. That is why I prepared a card separaion skill for weapons and shields...");
	}
	next;
	mes .@n$;
	mes needtr("카드 분리 수수료는 부위와 관계없이 ^ff00001,000,000 제니^000000입니다. 다른 재료는 필요하지 않아요.", "The card separation fee is ^ff00001,000,000 Zeny^000000 whatever the slot. No other materials are needed.");
	mes needtr("^0000ff분리 성공률은 100%^000000라서 실패로 수수료를 날릴 일은 없습니다.", "^0000ffThe success rate is 100%^000000, so you never lose the fee to a failure.");
	next;
	mes .@n$;
	mes needtr("분리한 카드는 그대로 돌려드리고, 장비의 ^0000ff제련도와 인챈트 효과도 유지^000000됩니다. 분리할 장비가 있으신가요?", "You get the separated cards back as they are, and the equipment ^0000ffkeeps its refine level and enchant effects^000000. Do you have equipment to separate?");
	next;

	for(set .@i,0; .@i<getarraysize(.@equip_slot); set .@i,.@i+1) {
		if (getequipisequiped(.@equip_slot[.@i]))
			set .@menu$, .@menu$+getequipname(.@equip_slot[.@i])+":";
		else
			set .@menu$, .@menu$+"^999999"+.@equip_name$[.@i]+needtr(" (비어 있음)^000000:", " (empty)^000000:");
	}

	set .@i, select(needtr("작업 중단:", "Stop the work:")+((.@Jeremy)?needtr("그게 어떻게 가능한가요?", "How is it possible?"):"")+":"+.@menu$);
	switch(.@i) {
	case 1:
		mes .@n$;
		mes needtr("작업이 필요하면 여기로 저를 찾아오세요.", "Whenever you need the work, visit me here.");
		close;
	case 2:
		mes .@n$;
		mes needtr("뭐가 그렇게 특별한지 궁금하시군요. 음, 그냥 안내만 하는 건 재미없으니 이야기를 하나 해드리죠...", "You wonder what is so special. Well, I hate to give only a guide, so let me tell you the story...");
		next;
		mes needtr("^000099제레미가 어깨와 손을 풀고 있습니다. 누군가 말을 걸어주기를 기다리는 것 같네요.^000000", "^000099Jeremy is stretching his shoulders and hands. He might be waiting for someone to talk with him.^000000");
		next;
		mes .@n$;
		mes needtr("말랑도의 특산물이 통조림이라는 걸 아시나요?", "Do you know that Malangdo's specialty is canned food?");
		next;
		select(needtr("잘 알고 있습니다. 그게 그렇게 한정적인가요?", "I knew that well. Is it that limited?"));
		mes .@n$;
		mes needtr("헤헤... 모두가 좋아하죠. 하지만 예전에는 몇 가지 문제가 있었답니다.", "Hehe... Everyone loves it. But there were some problems before.");
		next;
		select(needtr("문제요? 무슨 물고기에 문제라도 있었나요?", "Problems? Is there any faulty fish?"));
		mes .@n$;
		mes needtr("아뇨, 물고기 자체에는 문제가 없어요. 문제는 가공 후에 생기는 엄청난 양의 생선 기름이죠. 그건 그야말로 산업 폐기물이었으니까요.", "No, the fish does not have any problem. The problem is lots of fish oil produced after processing. That is such an industrial waste.");
		next;
		mes .@n$;
		mes needtr("하지만 재활성 정제 과정을 거친 후, 이 생선 기름은 고전 장비 관리와 산업용 윤활유로 사용될 만큼 가치가 높아졌죠.", "However, after a revitalizing refining process, this fish oil became valuable to use for old equipment care and industrial lubricant.");
		next;
		mes .@n$;
		mes needtr("게다가 이 오일은 장착된 장비에서 카드를 떼어내는 데 매우 유용했죠. 한동안은 윤활유를 구해 오셔야 작업이 가능했습니다.", "This oil used to be very useful for taking cards off equipped gear. For a while you had to bring lubricant before I could do the work.");
		next;
		mes .@n$;
		mes needtr("지금은 공정을 다듬어서 윤활유 없이도 확실하게 떼어낼 수 있게 됐어요. 그래서 수수료만 주시면 ^0000ff실패 없이^000000 분리해 드립니다.", "Now the process is refined, so I can take them off reliably without lubricant. Just pay the fee and I will separate them ^0000ffwithout fail^000000.");
		close;
	default:
		set .@equip_num, .@equip_slot[.@i-3];
		if (!getequipisequiped(.@equip_num)) {
			mes .@n$;
			if (.@Jeremy)
				mes needtr("이 부위에는 아무것도 없는데요?", "In this part, there is nothing?");
			else
				mes needtr("그 부위에는 아무것도 없나요?", "There is nothing on that part?");
			close;
		}
		break;
	}

	// NEED: 카드 제거는 1,000,000 제니 · 성공률 100% 단일 방식으로 통일한다.
	//       투구 중단 2배 비용, 고급/일반 윤활유, 실릿퐁, 봉인된 실릿퐁 경로는 모두 제거했다.
	set .@zeny_cost, 1000000;
	set .@zeny_text$, "1,000,000z";

	setarray .@check_equip_card[0], getequipcardid(.@equip_num,0),getequipcardid(.@equip_num,1),getequipcardid(.@equip_num,2),getequipcardid(.@equip_num,3);
	setarray .@equip_card[0], .@check_equip_card[0], .@check_equip_card[1], .@check_equip_card[2], .@check_equip_card[3];

	set .@has_card,0;
	for ( .@i = 0; .@i < MAX_SLOTS; .@i++ ) {
		// 인챈트 카드는 제외하고 실제 일반 카드만 분리 대상으로 판정합니다.
		if (.@equip_card[.@i] && getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_NORMAL)
			set .@has_card,1;
	}

	if (!.@has_card) {
		mes .@n$;
		mes needtr("분리할 일반 카드가 장착되어 있지 않습니다.", "There is no normal card fitted that could be separated.");
		close;
	}

	mes .@n$;
	mes needtr("장착된 ^0000ff모든 일반 카드^000000를 한 번에 분리합니다.", "This separates ^0000ffall the normal cards^000000 fitted to it, in one go.");
	mes needtr("MVP 카드와 봉인된 보스 카드도 같은 방식으로 분리됩니다.", "MVP cards and Sealed Boss Cards are separated the same way.");
	mes needtr("장비의 ^0000ff제련도와 인챈트 효과는 유지^000000됩니다.", "The equipment's ^0000ffrefine level and enchant effects are kept^000000.");
	mes needtr("수수료 ^ff0000", "Fee ^ff0000") + .@zeny_text$ + needtr("^000000 · 성공률 ^0000ff100%^000000", "^000000 - Success rate ^0000ff100%^000000");
	next;

	set .@menu$,
		"다음에...:" +
		((Zeny >= .@zeny_cost) ? (.@zeny_text$ + " 지불하고 분리") : ("^999999" + .@zeny_text$ + " (잔액 부족)^000000"));
	if (select(.@menu$) == 1) {
		mes .@n$;
		mes needtr("작업이 필요하면 여기로 저를 찾아오세요.", "Whenever you need the work, visit me here.");
		close;
	}
	if (Zeny < .@zeny_cost) {
		mes .@n$;
		mes needtr("제니가 부족합니다. 수수료를 충분히 챙겨서 다시 와주세요.", "You don't have enough zeny. Please come back with enough fees.");
		close;
	}

	// anti-hack
	if (callfunc("F_IsEquipCardHack", .@equip_num, .@check_equip_card[0], .@check_equip_card[1], .@check_equip_card[2], .@check_equip_card[3]))
		close;

	// 확정 제거이므로 수수료 차감 직후 바로 분리한다.
	set Zeny, Zeny - .@zeny_cost;

	// 장비를 착용한 상태에서 호출해야 합니다.
	// item_db.yml의 Slots 범위에 있는 일반 카드를 모두 분리합니다.
	successremovecards .@equip_num;

	specialeffect2 EF_MAXPOWER;

	mes needtr("-- 카드 분리 결과 --", "-- Result of Card Separation --");
	mes needtr("^0000ff카드 분리에 성공했습니다.^000000", "^0000ffCard separation has succeeded.^000000");
	mes needtr("^0000ff장착된 모든 일반 카드가 정상적으로 분리되었습니다.^000000", "^0000ffEvery normal card fitted to it was separated cleanly.^000000");
	mes needtr("^0000ff장비의 제련도와 인챈트 효과는 유지되었습니다.^000000", "^0000ffThe equipment's refine level and enchant effects were kept.^000000");
	next;
	mes .@n$;
	mes needtr("카드 분리 결과는 여기까지입니다. 다음에 또 이용해 주세요.", "That is all for the results of the card separation. Please come again.");
	close;
	
OnInit:
	if (strnpcinfo(1) == "제레미") {
		waitingroom "방어구/악세 카드분리",0;
	} else {
		waitingroom "무기/방패 카드분리",0;
	}
		end;
}
lasa_in01,161,54,6	duplicate(CardSeparation_mal)	제레미#pa0829	553	// Armors :: mal_yong
