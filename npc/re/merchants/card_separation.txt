-	script	::CardSeparation_mal	-1,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "아이템 종류가 너무 많습니다. 아이템 종류를 줄인 뒤에 다시 시도해 주세요.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "무거운 물건을 너무 많이 들고 있어 진행할 수 없습니다. 무게를 줄인 뒤에 다시 시도해 주세요.";
		close;
	}
	if (strnpcinfo(1) == "제레미") {
		set .@Jeremy,1;
		set .@n$, "[제레미]";
		setarray .@equip_name$[0],  "갑옷",  "신발",  "걸칠것", "투구 상단","악세서리(오른쪽)","악세서리(왼쪽)";
		setarray .@equip_slot[0], EQI_ARMOR,EQI_SHOES,EQI_GARMENT,EQI_HEAD_TOP, EQI_ACC_L, EQI_ACC_R;

		mes .@n$;
		mes "오랜만이네요~";
		mes "갑옷, 신발, 걸칠것, 투구에서 카드를 분리하는 새로운 기술을 배웠습니다. 한번 해보시겠어요?";
	} else {
		set .@Jeremy,0;
		set .@n$, "[리처드]";
		setarray .@equip_name$[0], "왼손", "오른손";
		setarray .@equip_slot[0],   EQI_HAND_L,   EQI_HAND_R;

		mes .@n$;
		mes "바보 같은 제레미는 카드가 손상될까 봐 무기나 방패는 건드리고 싶어 하지 않아요. 그래서 제가 무기와 방패를 위한 카드 분리 기술을 준비했죠...";
	}
	next;
	mes .@n$;
	mes "일반적으로 수수료는 1,000,000 제니입니다. 카드 분리 시 ^990000분리 성공률을 높여주는 특별한 아이템^000000을 사용할 수 있습니다. 이에 대한 추가 제니는 받지 않아요.";
	next;
	mes .@n$;
	mes "특수 아이템을 사용하더라도 실패할 가능성은 있습니다. 실패 시 ^ff0000사용한 재화만 소모^000000되며, 장비와 카드는 그대로 유지됩니다. 분리할 장비가 있으신가요?";
	next;

	for(set .@i,0; .@i<getarraysize(.@equip_slot); set .@i,.@i+1) {
		if (getequipisequiped(.@equip_slot[.@i]))
			set .@menu$, .@menu$+getequipname(.@equip_slot[.@i])+":";
		else
			set .@menu$, .@menu$+"^999999"+.@equip_name$[.@i]+" (비어 있음)^000000:";
	}

	set .@i, select("작업 중단:"+((.@Jeremy)?"그게 어떻게 가능한가요?":"")+":"+.@menu$);
	switch(.@i) {
	case 1:
		mes .@n$;
		mes "작업이 필요하면 여기로 저를 찾아오세요.";
		close;
	case 2:
		mes .@n$;
		mes "뭐가 그렇게 특별한지 궁금하시군요. 음, 그냥 안내만 하는 건 재미없으니 이야기를 하나 해드리죠...";
		next;
		mes "^000099제레미가 어깨와 손을 풀고 있습니다. 누군가 말을 걸어주기를 기다리는 것 같네요.^000000";
		next;
		mes .@n$;
		mes "말랑도의 특산물이 통조림이라는 걸 아시나요?";
		next;
		select("잘 알고 있습니다. 그게 그렇게 한정적인가요?");
		mes .@n$;
		mes "헤헤... 모두가 좋아하죠. 하지만 예전에는 몇 가지 문제가 있었답니다.";
		next;
		select("문제요? 무슨 물고기에 문제라도 있었나요?");
		mes .@n$;
		mes "아뇨, 물고기 자체에는 문제가 없어요. 문제는 가공 후에 생기는 엄청난 양의 생선 기름이죠. 그건 그야말로 산업 폐기물이었으니까요.";
		next;
		mes .@n$;
		mes "하지만 재활성 정제 과정을 거친 후, 이 생선 기름은 고전 장비 관리와 산업용 윤활유로 사용될 만큼 가치가 높아졌죠.";
		next;
		mes .@n$;
		mes "게다가 이 오일은 장착된 무기에서 유물을 분리하는 데 매우 유용해서, 고급 윤활유와 일반 윤활유는 제니로 살 수 없답니다.";
		next;
		mes .@n$;
		mes "음, 돈 걱정은 마세요. 확실히 고급 윤활유는 비싸지만요. 제니를 조금 내시면 더 저렴한 윤활유를 드릴 수 있습니다.";
		next;
		mes .@n$;
		mes "성공 품질은 확신할 수 없지만, 어쨌든 이건 아주 저렴하잖아요, 그렇죠?";
		close;
	default:
		set .@equip_num, .@equip_slot[.@i-3];
		if (!getequipisequiped(.@equip_num)) {
			mes .@n$;
			if (.@Jeremy)
				mes "이 부위에는 아무것도 없는데요?";
			else
				mes "그 부위에는 아무것도 없나요?";
			close;
		}
		break;
	}

	setarray .@check_equip_card[0], getequipcardid(.@equip_num,0),getequipcardid(.@equip_num,1),getequipcardid(.@equip_num,2),getequipcardid(.@equip_num,3);
	setarray .@equip_card[0], .@check_equip_card[0], .@check_equip_card[1], .@check_equip_card[2], .@check_equip_card[3];
	set .@mvp_list$,
		"|4121|4123|4128|4131|4132"+
		"|4134|4135|4137|4142|4143"+
		"|4144|4145|4146|4147|4148"+
		"|4168|4236|4263|4276|4302"+
		"|4305|4318|4324|4330|4342"+
		"|4352|4357|4359|4361|4363"+
		"|4365|4367|4372|4374|4376"+
		"|4386|4399|4403|4407|4408"+
		"|4419|4425|4430|4441|4451"+
		"|4456|4507|4509|4520|4525"+
		"|4526|4527|4528|4529|4534"+
		"|4556|4557|4610|4636|4647"+
		"|4648|4649|4650|4651|4662"+
		"|4674|4675|4676|4677|4678"+
		"|4679|27025|27081|27104|27106"+
		"|27113|27119|27151|27180|27182"+
		"|27265|27294|27298|27362|27363"+
		"|31026|300007|300145|300151|300217"+
		"|300227|300228|300239";

	if (.@Jeremy) {
		set .@has_card,0;
		for ( .@i = 0; .@i < MAX_SLOTS; .@i++ ) {
			if (getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_ENCHANT)
				.@equip_card[.@i] = 0;// Armor Enchant System
			if (.@equip_card[.@i])
				set .@has_card,1;
		}
		if (!.@has_card) {
			mes .@n$;
			mes "카드가 장착되어 있지 않습니다. 다시 확인하시겠습니까?";
			close;
		}
		if ((.@equip_card[0] && compare(.@mvp_list$,"|"+.@equip_card[0]+"|")) ||
			(.@equip_card[1] && compare(.@mvp_list$,"|"+.@equip_card[1]+"|")) ||
			(.@equip_card[2] && compare(.@mvp_list$,"|"+.@equip_card[2]+"|")) ||
			(.@equip_card[3] && compare(.@mvp_list$,"|"+.@equip_card[3]+"|")))
			set .@boss_chk,1;
	} else {
		// Official "Richard" script uses a hardcoded list including every possible item.
		//if (!getequipisequiped(.@equip_num)) {
		//	mes "[Richard]";
		//	mes "I'm sorry. We don't provide that equipment yet.";
		//	close;
		//}

		mes "[리처드]";
		mes "몇 번 소켓의 카드를 분리하시겠습니까? 왼쪽 소켓부터 순서대로 1, 2, 3, 4번입니다.";
		next;
		set .@menu$,"";
		for ( .@i = 0; .@i < MAX_SLOTS; .@i++ ) {
			if (.@equip_card[.@i] && getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_NORMAL) // Armor Enchant System
				.@menu$ = .@menu$ + "소켓 " + (.@i+1) + " - " + getitemname(.@equip_card[.@i])+":";
			else {
				.@menu$ = .@menu$ + "^777777소켓 " + (.@i+1) + " - 카드 없음^000000:";
			}
		}
		set .@i, select("작업 중단:"+.@menu$);
		switch(.@i) {
		case 1:
			mes .@n$;
			mes "작업이 필요할 때면 언제든 찾아주세요.";
			close;
		default:
			set .@slot, .@i-2;
			if (.@equip_card[.@slot] == 0 || getiteminfo(.@equip_card[.@slot], ITEMINFO_SUBTYPE) == CARD_ENCHANT) {
				mes .@n$;
				mes "이 소켓에는 장착된 카드가 없습니다. 다시 확인해 보시겠어요?";
				close;
			}
			break;
		}
		if (compare(.@mvp_list$,"|"+.@equip_card[.@slot]+"|"))
			set .@boss_chk,1;
	}
	if (.@boss_chk == 0) {
		mes .@n$;
		if (.@Jeremy)
			mes "카드를 제외한 ^ff0000모든 인챈트 효과는 사라집니다.^000000 이에 동의하신다면 작업 유형을 선택해 주세요.:";
		else
			mes "작업 수수료를 선택해 주세요.";
		next;
		set .@menu$,
			"다음에...:"+
			((Zeny >= 1000000)?"1,000,000z 사용 (특수 아이템 미사용):":"^9999991,000,000z 사용 (잔액 부족)^000000:")+
			((countitem(6441))?"고급 윤활유 사용:":"^999999고급 윤활유 (부족)^000000:")+
			((countitem(6440))?"일반 윤활유 사용":"^999999일반 윤활유 (부족)^000000");
		switch(select(.@menu$)) {
		case 1:
			mes .@n$;
			mes "작업이 필요하면 여기로 저를 찾아오세요.";
			close;
		case 2:
			if (Zeny < 1000000) {
				mes .@n$;
				mes "제니가 부족합니다. 수수료를 충분히 챙겨서 다시 와주세요.";
				close;
			}
			mes .@n$;
			mes "이건 상당히 오래된 장비군요. 작업 실패 확률이 높습니다. 실패해도 장비와 카드는 유지되지만, 수수료는 반환되지 않습니다. 정말 계속하시겠습니까?";
			next;
			if(select("다음에...:계속") == 1) {
				mes .@n$;
				mes "작업이 필요하면 여기로 저를 찾아오세요.";
				close;
			}
			set .@sf_c_num,150;
			set .@sf_r_num,150;
			set .@sf_w_num,150;
			set Zeny, Zeny - 1000000;
			break;
		case 3:
			if (countitem(6441) == 0) {
				mes .@n$;
				mes "고급 윤활유가 없습니다.";
				close;
			}
			mes .@n$;
			mes "고급 윤활유를 사용하면 성공 확률이 크게 올라가지만, 100% 보장할 수는 없습니다. 실패 시 재화만 소모됩니다. 정말 계속하시겠습니까?";
			next;
			if(select("다음에...:계속") == 1) {
				mes .@n$;
				mes "작업이 필요하면 여기로 저를 찾아오세요.";
				close;
			}
			set .@sf_c_num,75;
			set .@sf_r_num,75;
			set .@sf_w_num,75;
			delitem 6441,1; //High_RankLubricant
			break;
		case 4:
			if (countitem(6440) == 0) {
				mes .@n$;
				mes "일반 윤활유가 없습니다.";
				close;
			}
			mes .@n$;
			mes "일반 윤활유를 사용하면 성공 확률이 올라가지만, 100% 보장할 수는 없습니다. 실패 시 재화만 소모됩니다. 정말 계속하시겠습니까?";
			next;
			if(select("다음에...:계속") == 1) {
				mes .@n$;
				mes "작업이 필요하면 여기로 저를 찾아오세요.";
				close;
			}
			set .@sf_c_num,75;
			set .@sf_r_num,150;
			set .@sf_w_num,150;
			delitem 6440,1; //General_Lubricant
			break;
		}
	} else if (.@boss_chk == 1) {
		mes .@n$;
		mes "이 장비에는 귀중한 MVP 카드가 박혀 있군요. 이 카드는 윤활유만으로는 분리할 수 없습니다. 강력한 계면활성제인 ^0000ff실릿퐁^000000을 가져오시면 작업해 드릴 수 있어요.";
		next;
		switch(select("다음에...:실릿퐁을 가지고 있습니다.")) {
		case 1:
			mes .@n$;
			mes "작업이 필요하면 여기로 저를 찾아오세요.";
			close;
		case 2:
			if (countitem(6443) == 0) {
				mes .@n$;
				mes "실릿퐁이 없습니다.";
				close;
			}
			break;
		}
		mes .@n$;
		if (.@Jeremy) {
			mes "카드를 제외한 ^ff0000모든 인챈트 효과는 사라집니다.^000000 이에 동의하신다면 작업 유형을 선택해 주세요.:";
			set .@menu$,"좋아요, 해봅시다!";
		} else {
			mes "계속할까요?";
			set .@menu$,"알겠어요. 그냥 빨리 좀 해주세요!";
		}
		next;
		switch(select("다음에...:"+.@menu$)) {
		case 1:
			mes .@n$;
			mes "작업이 필요하면 여기로 저를 찾아오세요.";
			close;
		case 2:
			set .@sf_c_num,60;
			set .@sf_r_num,60;
			set .@sf_w_num,60;
			delitem 6443,1; //Sillit_Pong_Bottle
			break;
		}
	}

	set .@equip_id, getequipid(.@equip_num);
	set .@equip_refine, getequiprefinerycnt(.@equip_num);

	// anti-hack
	if (callfunc("F_IsEquipCardHack", .@equip_num, .@check_equip_card[0], .@check_equip_card[1], .@check_equip_card[2], .@check_equip_card[3]))
		close;

	// 기존 구조는 delequip 후 장비/카드를 각각 확률로 재지급했기 때문에
	// 실패 시 장비 파괴 또는 카드 소실이 발생했습니다.
	// 변경 후에는 먼저 성공 여부만 판정하고, 실패하면 장비는 건드리지 않습니다.
	if (rand(1,.@sf_c_num) >= 61) {
		specialeffect2 EF_SUI_EXPLOSION;

		mes "-- 카드 분리 결과 --";
		mes "카드 분리에 실패했습니다.";
		mes "^ff0000사용한 재화는 반환되지 않습니다.^000000";
		mes "^0000ff장비와 카드는 손상되지 않고 그대로 유지됩니다.^000000";
		next;
		mes .@n$;
		mes "아쉽지만 분리에 실패했습니다. 장비와 카드는 그대로이니 다시 준비해서 찾아와 주세요.";
		close;
	}

	delequip .@equip_num;

	if (.@Jeremy) {
		// 성공 시 장비는 카드가 제거된 상태로 반환됩니다.
		// 카드가 아닌 인챈트 슬롯은 기존 공식 스크립트 흐름처럼 제거됩니다.
		getitem2 .@equip_id,1,1,.@equip_refine,0,0,0,0,0;

		// 성공 시 장착되어 있던 일반 카드는 모두 반환됩니다.
		for(set .@i,0; .@i<4; set .@i,.@i+1) {
			if (.@equip_card[.@i])
				getitem .@equip_card[.@i],1;
		}
	} else {
		set .@card, .@equip_card[.@slot];
		set .@check_equip_card[.@slot],0;

		// 성공 시 선택한 카드만 분리하고, 나머지 카드/인챈트는 유지합니다.
		getitem2 .@equip_id,1,1,.@equip_refine,0,.@check_equip_card[0],.@check_equip_card[1],.@check_equip_card[2],.@check_equip_card[3];
		getitem .@card,1;
	}

	specialeffect2 EF_MAXPOWER;

	mes "-- 카드 분리 결과 --";
	mes "^0000ff카드 분리에 성공했습니다.^000000";
	mes "^0000ff장비와 카드는 정상적으로 분리되었습니다.^000000";
	next;
	mes .@n$;
	mes "카드 분리 결과는 여기까지입니다. 다음에 또 이용해 주세요.";
	close;
	
OnInit:
	if (strnpcinfo(1) == "제레미") {
		waitingroom "방어구/악세 카드분리",0;
	} else {
		waitingroom "무기 카드분리",0;
	}
		end;
}
