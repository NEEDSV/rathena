payon,139,236,4	script	구매 담당자#Buying	59,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2400) {
		mes "- 소지한 아이템이 너무 많아 대화하거나 퀘스트를 진행할 수 없습니다. -";
		close;
	}
	mes "[휴]";
	if (BaseClass == Job_Merchant && getskilllv("MC_VENDING") >= 1) {
		if (getskilllv("ALL_BUYING_STORE") == 1) {
			mes "구매 담당자 휴입니다.";
			mes "무엇을 도와드릴까요?";
			next;
			if(select("구매 노점 개설 허가증 구입:그만두기") == 2) {
				mes "[휴]";
				mes "구매 노점 개설 허가증이 필요하시면 언제든 말씀해 주세요.";
				mes "다음에 또 찾아주세요~";
				close;
			}
			mes "[휴]";
			mes "구매 노점 개설 허가증은 한 장당 200 제니이며,";
			mes "한 번에 최대 50장까지 구입할 수 있습니다.";
			mes "몇 장이 필요하신가요?";
			next;
			while(1) {
				input .@input;
				mes "[휴]";
				if (.@input == 0) {
					mes "거래를 취소하셨습니다.";
					mes "좋은 하루 보내세요.";
					close;
				} else if (.@input > 50) {
					mes "50 이하의 수량을 입력해 주세요.";
					next;
				} else {
					mes "허가증 "+.@input+"장의 가격은 총 "+(.@input*200)+" 제니입니다.";
					if (Zeny < (.@input*200)) {
						mes "하지만 소지하신 제니가 부족한 것 같습니다.";
					} else {
						mes "이용해 주셔서 감사합니다.";
						getitem 6377,.@input; //Buy_Market_Permit
						set Zeny,Zeny-(.@input*200);
					}
					close;
				}
			}
		} else {
			mes "알베르타 상인 길드 구매 담당자 휴입니다.";
			mes "당신은...";
			next;
			mes "[휴]";
			mes "우리 길드에서 개인이 다른 사람의 물건을 구매할 수 있도록";
			mes "새로운 허가 제도를 만들었다는 사실을 알고 계신가요?";
			next;
			select("물건을 사는 데 불편했던 적은 없는데...");
			mes "[휴]";
			mes "맞는 말씀이지만, 이렇게 생각해 보세요.";
			mes "물건을 대량으로 구매하려다 어려움을 겪은 적은 없으신가요?";
			mes "원하는 물건을 가진 사람을 일일이 찾아가 대화해야 하죠.";
			mes "상당히 불편하지 않습니까?";
			next;
			mes "[휴]";
			mes "대량 구매는 우리 상인들에게 매우 중요한 문제이기 때문에,";
			mes "저는 10년 동안 물건을 구매하며 쌓은 경험을 바탕으로";
			mes "길드에 혁신적인 계획을 제안했습니다.";
			next;
			mes "[휴]";
			mes "'누구나 자유롭게";
			mes "구매 노점을 열 수 있게 하자!'";
			mes "바로 이것이 제 제안서의 제목입니다.";
			mes "직접 읽어 보시면 아시겠지만...";
			next;
			mes "[휴]";
			mes "...";
			mes "...(휴가 세부 내용을 끝없이 설명하고 있다.)";
			next;
			select("그래서 결론이 뭔가요?");
			mes "[휴]";
			mes "아, 그렇군요. 간단히 정리하자면,";
			mes "노점 기능을 이용해 특정 아이템을 대량으로 구매할 수 있습니다.";
			next;
			mes "[휴]";
			mes "구매 노점을 열려면 상인 길드에서 발급하는 허가증이 필요합니다.";
			next;
			mes "[휴]";
			mes "노점을 개설할 때마다 허가증 한 장이 필요합니다.";
			mes "이 새로운 허가 제도를 통해 상당한 수익 증가를 기대하고 있죠.";
			next;
			mes "[휴]";
			mes "그게 바로 제 제안의 핵심입니다!";
			mes "큰돈을 벌 수 있다는 말에 길드장님도 무척 기뻐하셨습니다!";
			next;
			mes "["+strcharinfo(0)+"]";
			mes "그러니까 결론부터 말해 주세요!";
			next;
			mes "[휴]";
			mes "그렇게 서두르지 마세요.";
			mes "결론은, 노점 개설 스킬을 배운 분이라면";
			mes "구매 노점을 열 수 있도록 해드린다는 것입니다.";
			next;
			mes "[휴]";
			mes "물론 최초 등록 수수료로 10,000 제니를 받고 있습니다.";
			mes "구매 노점을 열 때마다 ^4A4AFF구매 노점 개설 허가증^000000도 필요합니다.";
			next;
			mes "[휴]";
			mes "지금 등록하시겠습니까?";
			mes "등록하시면 구매 노점을 여는 방법을 알려드리겠습니다.";
			next;
			if(select("구매 노점 개설 방법 배우기:취소") == 2) {
				mes "[휴]";
				mes "아, 아쉽군요!";
				mes "대량 구매 기능은 사업에 여러모로 도움이 될 텐데,";
				mes "결정은 본인의 몫이니까요.";
				mes "상담이 필요하시면 언제든 다시 찾아주세요!";
				close;
			}
			mes "[휴]";
			if (Zeny < 10000) {
				mes "등록 수수료는 10,000 제니입니다.";
				mes "먼저 수수료를 준비해 주세요.";
				close;
			} else {
				mes "좋은 결정을 하셨습니다.";
				mes "등록 수수료를 내시고 이곳에 서명해 주세요....";
				next;
				input .@name$;
				mes "[휴]";
				mes .@name$+"....";
				mes "글씨체가 마음에 드는군요.";
				mes "좋습니다. 이제 구매 노점을 열 수 있도록 승인되었습니다.";
				set Zeny,Zeny-10000;
				getitem 6377,5; //Buy_Stall_Permit
				skill "ALL_BUYING_STORE",1,SKILL_PERM_GRANT;
				next;
				mes "[휴]";
				mes "현재는 장비, 일부 포션, 제작 아이템을 ^8C2121제외^000000한";
				mes "일반 아이템만 대량 구매할 수 있습니다.";
				mes "그래도 활용 방법에 따라 사업에 큰 도움이 될 겁니다.";
				next;
				mes "[휴]";
				mes "그리고 구매하려는 아이템은 인벤토리에 최소 1개 이상 가지고 있어야 합니다.";
				mes "구매 노점에 견본으로 보여줄 아이템이 필요하기 때문입니다.";
				next;
				mes "[휴]";
				mes "스킬 창에 구매 노점 스킬이 등록되었을 겁니다.";
				mes "보이지 않는다면 스킬 목록을 접었다 펼친 뒤";
				mes "3차 직업 탭을 확인해 보세요.";
				next;
				mes "[휴]";
				mes "시험 삼아 사용할 수 있도록 구매 노점 개설 허가증 5장을 드렸습니다.";
				mes "즐거운 거래 되세요!";
				close;
			}
		}
	}
	mes "알베르타 상인 길드 구매 담당자 휴입니다.";
	mes "이야기를 나누고 싶지만 지금은 너무 바쁘군요.";
	next;
	mes "[휴]";
	mes "제 시간은 오직 상인 고객 여러분을 위해 사용하고 있습니다.";
	close;
}
