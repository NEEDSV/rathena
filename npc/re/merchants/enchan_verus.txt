	setarray .@list[0],
		6965, 4970, 1,0,3,	// Reactor_P_FIRE_		Reactor_P_FIRE
		6966, 4971, 1,0,3,	// Reactor_P_WATER_		Reactor_P_WATER
		6967, 4972, 1,0,3,	// Reactor_P_GROUND_	Reactor_P_GROUND
		6968, 4973, 1,0,3,	// Reactor_P_WIND_		Reactor_P_WIND
		6969, 4974, 0,3,0,	// Reactor_T_FIRE_		Reactor_T_FIRE
		6970, 4975, 0,3,0,	// Reactor_T_WATER_		Reactor_T_WATER
		6971, 4976, 0,3,0,	// Reactor_T_GROUND_	Reactor_T_GROUND
		6972, 4977, 0,3,0,	// Reactor_T_WIND_		Reactor_T_WIND
		6973, 4978, 3,3,0,	// Reactor_Cure_101_	Reactor_Cure_101
		6974, 4979, 3,3,0,	// Reactor_Cure_102_	Reactor_Cure_102
		6975, 4980, 3,3,0,	// Reactor_Cure_201_	Reactor_Cure_201
		6976, 4981, 3,3,0,	// Reactor_Cure_202_	Reactor_Cure_202
		6977, 4982, 1,0,0,	// Reactor_A_STR_		Reactor_A_STR
		6978, 4983, 1,0,0,	// Reactor_A_INT_		Reactor_A_INT
		6979, 4984, 3,3,0,	// Reactor_A_DEF_		Reactor_A_DEF
		6980, 4985, 1,1,0,	// Reactor_A_AVOI_		Reactor_A_AVOI
		6981, 4986, 3,3,0,	// Reactor_A_ATK_		Reactor_A_ATK
		6982, 4987, 3,3,0,	// Reactor_A_MATK_		Reactor_A_MATK
		6983, 4988, 3,3,0,	// Reactor_A_MHP_		Reactor_A_MHP
		6984, 4989, 3,3,0,	// Reactor_A_MSP_		Reactor_A_MSP
		6985, 4990, 1,0,0,	// Reactor_A_FROZ_		Reactor_A_FROZ
		6986, 4991, 1,1,0;	// Reactor_A_ASPD_		Reactor_A_ASPD
	.@size = getarraysize(.@list);

	.@menu$ = needtr("그만두기", "Quit.");

	for (.@i = 0; .@i < .@size; .@i += 5) {
		.@menu$ += ":";

		if (countitem(.@list[.@i]) < 1)
			.@menu$ += "^aaaaaa" + getitemname(.@list[.@i]) + needtr(" (미소지)^000000", " (not owned)^000000");
		else
			.@menu$ += getitemname(.@list[.@i]);
	}

	mes "[MARS_01]";
	mes strcharinfo(0) + needtr(" 님,", ",");
	mes needtr("사용할 인챈트 설계도를 선택해 주세요!", "Please select the enchant blueprint to use!");
	next;
	.@s = select(.@menu$) - 2;
	if (.@s < 0) {
		mes "[MARS_01]";
		mes needtr("다음에 다시 이용해 주세요.", "So far.");
		close;
	}
	.@s *= 5;
	.@item_req = .@list[.@s];
	.@item_enchant_id = .@list[.@s+1];
	.@first_slot_only = .@list[.@s+4];

	.@equip_id = getequipid(.@part);
	.@equip_refine = getequiprefinerycnt(.@part);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	if (countitem(.@item_req) < 1) {
		mes "[MARS_01]";
		mes needtr("다음에 다시 이용해 주세요.", "So far.");
		close;
	}
	if (.@part == EQI_ARMOR)
		.@max_num_enchant = .@list[.@s+2];
	else if (.@part == EQI_GARMENT)
		.@max_num_enchant = .@list[.@s+3];
	else {
		mes "[MARS_01]";
		mes needtr("관리자에게 문의해 주세요.", "Please contact the administrator.");
		close;
	}

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@equip_refine)) {
		mes "[MARS_01]";
		mes needtr("진행 도중 장비 상태가 변경되었습니다.", "Did you remove your gear?");
		close;
	}

	.@slot_sum = 0;
	.@enchant_total = 0;

	// 동일 인챈트 개수
	if (.@card[1] > 0 && .@item_enchant_id == .@card[1])
		.@slot_sum++;
	if (.@card[2] > 0 && .@item_enchant_id == .@card[2])
		.@slot_sum++;
	if (.@card[3] > 0 && .@item_enchant_id == .@card[3])
		.@slot_sum++;

	// 현재 장비에 부여된 전체 인챈트 개수
	if (.@card[1] > 0)
		.@enchant_total++;
	if (.@card[2] > 0)
		.@enchant_total++;
	if (.@card[3] > 0)
		.@enchant_total++;

	if (.@max_num_enchant < 1) {
		mes "[MARS_01]";
		mes needtr("선택한 설계도는 해당 장비에 사용할 수 없습니다.", "The design you choose is incompatible with the equipment.");
		close;
	}
	if (.@first_slot_only > 0 && .@card[3] != 0) {
		mes "[MARS_01]";
		mes needtr("이 설계도는 첫 번째 인챈트 슬롯에만 사용할 수 있습니다. 다른 설계도를 선택해 주세요.", "This design is only for the first time to strengthen the use, but the equipment has other performance, please use other design to strengthen it!");
		close;
	}
	if (.@slot_sum >= .@max_num_enchant) {
		mes "[MARS_01]";
		mes needtr("동일한 인챈트의 최대 부여 횟수는 ", "The maximum number of enchants is ") + .@max_num_enchant + needtr("회입니다. 이미 최대치에 도달했습니다.", ". The item has reached the upper limit.");
		close;
	}
	if (.@card[1] != 0) {
		mes "[MARS_01]";
		mes needtr("장비의 인챈트 슬롯이 모두 사용되었습니다.", "The number of equipment has reached the limit.");
		close;
	}

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@equip_refine)) {
		mes "[MARS_01]";
		mes needtr("진행 도중 장비 상태가 변경되었습니다.", "Please pay special attention if the equipment will be removed.");
		close;
	}

	switch( .@equip_id ) {
	case 20773:	// Excelion_Wing
	case 15128:	// Excelion_Suit
		break;
	default:
		mes "[MARS_01]";
		mes needtr("이 장비는 엑셀리온 장비가 아니므로 인챈트할 수 없습니다.", "The product does not have a serial number, does not meet the specifications of the product can not be strengthened.");
		close;
	}
	mes "[MARS_01]^0000ff";
	mes needtr("선택한 장비: ", "Selected equipment: ") + getequipname(.@part) + needtr(" / 설계도: ", " / Blueprint: ") + getitemname(.@item_req);
	mes "^000000------------------";
	mes needtr("현재 장비 인챈트: ^0000ff", "Current enchants on equipment: ^0000ff") + .@enchant_total + needtr("/3회^000000", "/3^000000");
	mes needtr("선택한 설계도 부여 횟수: ^0000ff", "Times the selected blueprint is applied: ^0000ff") + .@slot_sum + "/" + .@max_num_enchant + needtr("회^000000", "^000000");
	mes needtr("계속하시겠습니까?", "Do you wish to continue?");
	next;
	if (select( needtr("취소", "I'll think about it..."), needtr("계속 진행", "Go on.") ) == 1) {
		mes "[MARS_01]";
		mes needtr("필요하실 때 다시 찾아와 주세요~", "Come back next time you need me~");
		close;
	}

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@equip_refine))
		close;

	if (.@card[3] == 0) {
		.@enchant_count = 1;
		.@card[3] = .@item_enchant_id;
	}
	else if (.@card[2] == 0) {
		.@enchant_count = 2;
		.@card[2] = .@item_enchant_id;
	}
	else if (.@card[1] == 0) {
		.@enchant_count = 3;
		.@card[1] = .@item_enchant_id;
	}
	else {
		mes "[MARS_01]";
		mes needtr("장비의 인챈트 슬롯이 모두 사용되었습니다.", "The equipment has reached the upper limit of strengthening Hello!");
		close;
	}
	specialeffect2 EF_REPAIRWEAPON;
	mes "[MARS_01]";
	if (.@enchant_count < 5)
		mes needtr("장비에 ^990000", "The first equipment has been upgraded ^990000") + .@enchant_count + needtr("번째 인챈트가 부여되었습니다.^000000", " times^000000.");
	else
		mes needtr("장비 인챈트가 완료되었습니다.", "For additional performance equipment^990000 ^000000 upgrade.");	// never displayed

	delitem .@item_req, 1;
	delequip .@part;
	getitem2 .@equip_id,1,1,.@equip_refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
	close;
OnInit:
	waitingroom "엑셀리온 인챈트",0;
	end;
}

		// 하나의 문자열에 ':' 구분자를 사용하는 방식으로 통일합니다.

			.@first_start = 0;	// P 계열

			mes .@line$;
		}

		mes " ";
		mes needtr("^0000ff[Cure 계열]^000000", "^0000ff[Cure Series]^000000");

			.@max = .@list[.@i + .@max_index];

		next;
	}
