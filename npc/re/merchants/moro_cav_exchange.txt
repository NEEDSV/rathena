lasa_in01,187,49,5	script	인챈터 번즈::Expert Enchanter Byrnes	1_M_03,{
	disable_items;
	mes "[번즈]";
	mes "진정한 영웅만이 가질 자격이 있다는 " + getitemname(2981) + "에 대해 들어보셨습니까?";
	mes "저는 그 반지의 주인이 나타나기를 기다리고 있었습니다.";
	mes "무엇을 도와드릴까요?";
	next;
	switch(select(getitemname(2981) + " 인챈트", getitemname(2981) + " 인챈트 초기화", "인챈트 리스트 확인", "대화 종료")) {
	case 1:
		mes "[번즈]";
		mes "장착 중인 반지만 인챈트할 수 있습니다.";
		mes "어느 쪽에 장착한 반지를 인챈트하시겠습니까?";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select("왼쪽 액세서리", "오른쪽 액세서리") - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[번즈]";
			mes "인챈트할 " + getitemname(2981) + "을 먼저 장착해 주세요.";
			close;
		case 2981:
			break;
		default:
			mes "[번즈]";
			mes getitemname(2981) + "을 장착해 주세요.";
			mes "다른 장비에는 인챈트할 수 없습니다.";
			close;
		}
		// custom at this point
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[0] > 0) {
			mes "[번즈]";
			mes "모든 슬롯에 인챈트가 부여된 " + getitemname(2981) + "에는 더 이상 인챈트할 수 없습니다.";
			close;
		}
		if (.@card[3] == 0) {
			setarray .@enchant_list$[0], "DEF", "MDEF";
			.@slot = 4;
		}
		else if (.@card[2] == 0) {
			setarray .@enchant_list$[0], "최대 HP", "최대 SP";
			.@slot = 3;
		}
		else if (.@card[1] == 0) {
			setarray .@enchant_list$[0], "ATK", "MATK";
			.@slot = 2;
		}
		else {
			setarray .@enchant_list$[0], "STR", "INT", "AGI", "VIT", "DEX", "LUK";
			.@slot = 1;
		}
		mes "[번즈]";
		mes "슬롯 " + .@slot + "에 부여할 인챈트 계열을 선택해 주세요.";
		next;
		.@enchant_list_size = getarraysize(.@enchant_list$);
		for ( .@i = 0; .@i < .@enchant_list_size; ++.@i )
			.@menu$ += .@enchant_list$[.@i] + ":";
		.@type = select(.@menu$) - 1;

		switch(.@slot) {
		case 4:
			switch(.@type) {// enchant list index
			case 0:	// DEF
				setarray .@bonus[0], 4791, 4792, 4793, 4794, 4893;
				break;
			default:	// MDEF
				setarray .@bonus[0], 4890, 4786, 4891, 4787, 4892;
				break;
			}
			break;
		case 3:
			switch(.@type) {
			case 0:	// 최대 HP
				setarray .@bonus[0], 4861, 4862, 4867, 4868, 4900;
				break;
			default:	// 최대 SP
				setarray .@bonus[0], 4800, 4870, 4871, 4801, 4802;
				break;
			}
			break;
		case 2:
			switch(.@type) {
			case 0:	// ATK
				setarray .@bonus[0], 4882, 4766, 4767, 4894, 4895;
				break;
			default:	// MATK
				setarray .@bonus[0], 4883, 4896, 4897, 4898, 4899;
				break;
			}
			break;
		case 1:
			switch(.@type) {
			case 0:	// STR
				setarray .@bonus[0], 4700, 4701, 4702, 4703, 4704;
				break;
			case 1:	// INT
				setarray .@bonus[0], 4710, 4711, 4712, 4713, 4714; 
				break;
			case 2:	// AGI
				setarray .@bonus[0], 4730, 4731, 4732, 4733, 4734;
				break;
			case 3:	// VIT
				setarray .@bonus[0], 4740, 4741, 4742, 4743, 4744;
				break;
			case 4:	// DEX
				setarray .@bonus[0], 4720, 4721, 4722, 4723, 4724;
				break;
			default:	// LUK
				setarray .@bonus[0], 4750, 4751, 4752, 4753, 4754;
				break;
			}
			break;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		.@bonus_size = getarraysize(.@bonus);
		.@card[.@slot-1] = .@bonus[ rand(.@bonus_size) ];	// unknown rates
		specialeffect2 EF_ANGEL2;
		mes "[번즈]";
		mes "인챈트가 완료되었습니다!";
		delequip .@eqi_slot[.@s];
		getitem2 2981,1,1,0,0,.@card[0],.@card[1],.@card[2],.@card[3];
		close;
	case 2:
		mes "[번즈]";
		mes getitemname(2981) + "의 인챈트를 초기화하려면 " + getitemname(6684) + " 3개가 필요합니다.";
		next;
		if (countitem(6684) < 3) {
			mes "[번즈]";
			mes getitemname(6684) + "이 부족하여 인챈트를 초기화할 수 없습니다.";
			close;
		}
		mes "[번즈]";
		mes "장착 중인 반지만 초기화할 수 있습니다.";
		mes "어느 쪽에 장착한 반지를 초기화하시겠습니까?";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select("왼쪽 액세서리", "오른쪽 액세서리") - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[번즈]";
			mes "초기화할 " + getitemname(2981) + "을 먼저 장착해 주세요.";
			close;
		case 2981:
			break;
		default:
			mes "[번즈]";
			mes getitemname(2981) + "을 장착해 주세요.";
			mes "다른 장비는 초기화할 수 없습니다.";
			close;
		}
		// custom at this point
		if (countitem(6684) < 3) {
			mes "[번즈]";
			mes getitemname(6684) + "이 부족하여 인챈트를 초기화할 수 없습니다.";
			close;
		}
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[3] == 0) {
			mes "[번즈]";
			mes "현재 장착한 " + getitemname(2981) + "에는 초기화할 인챈트가 없습니다.";
			close;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		specialeffect2 EF_ANGEL2;
		mes "[번즈]";
		mes "인챈트 초기화가 완료되었습니다.";
		delitem 6684,3;
		delequip .@eqi_slot[.@s];
		getitem 2981,1;
		close;
	case 3:
		callfunc "F_ShowMoroEnchantList";
		end;
	case 4:
		close;
	}

OnInit:
	waitingroom "용사의 반지 인챈트",0;
	end;
}

// 공통 함수 F_ShowEnchantGroup은 다른 공통 스크립트에 선언되어 있어야 합니다.
// 이 파일에서는 중복 선언하지 않고 호출만 합니다.
function	script	F_ShowMoroEnchantList	{
	mes "[번즈]";
	mes getitemname(2981) + "에 부여할 수 있는 인챈트 목록입니다.";
	mes "인챈트는 ^FF00004번 → 3번 → 2번 → 1번 슬롯^000000 순서로 진행됩니다.";
	mes "선택한 계열의 목록 중 하나가 동일 확률로 부여됩니다.";
	next;

	while (true) {
		switch(select(
			"4번 슬롯 - DEF / MDEF",
			"3번 슬롯 - 최대 HP / 최대 SP",
			"2번 슬롯 - ATK / MATK",
			"1번 슬롯 - 기본 능력치",
			"닫기"
		)) {
		case 1:
			setdialogsize(220,300);
			callfunc "F_ShowEnchantGroup","DEF",4791,4792,4793,4794,4893;
			callfunc "F_ShowEnchantGroup","MDEF",4890,4786,4891,4787,4892;
			break;

		case 2:
			setdialogsize(220,300);
			callfunc "F_ShowEnchantGroup","최대 HP",4861,4862,4867,4868,4900;
			callfunc "F_ShowEnchantGroup","최대 SP",4800,4870,4871,4801,4802;
			break;

		case 3:
			setdialogsize(220,300);
			callfunc "F_ShowEnchantGroup","ATK",4882,4766,4767,4894,4895;
			callfunc "F_ShowEnchantGroup","MATK",4883,4896,4897,4898,4899;
			break;

		case 4:
			setdialogsize(220,300);
			callfunc "F_ShowEnchantGroup","STR",4700,4701,4702,4703,4704;
			callfunc "F_ShowEnchantGroup","INT",4710,4711,4712,4713,4714;
			callfunc "F_ShowEnchantGroup","AGI",4730,4731,4732,4733,4734;
			callfunc "F_ShowEnchantGroup","VIT",4740,4741,4742,4743,4744;
			callfunc "F_ShowEnchantGroup","DEX",4720,4721,4722,4723,4724;
			callfunc "F_ShowEnchantGroup","LUK",4750,4751,4752,4753,4754;
			break;

		case 5:
			close;
		}
	}
}
