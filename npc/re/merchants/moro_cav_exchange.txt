lasa_in01,187,49,5	script	인챈터 번즈::Expert Enchanter Byrnes	1_M_03,{
	disable_items;
	mes needtr("[번즈]", "[Byrnes]");
	mes needtr("진정한 영웅만이 가질 자격이 있다는 ", "Have you heard of the ") + getitemname(2981) + needtr("에 대해 들어보셨습니까?", ", which only a true hero deserves?");
	mes needtr("저는 그 반지의 주인이 나타나기를 기다리고 있었습니다.", "I have been waiting for the owner of that ring to appear.");
	mes needtr("무엇을 도와드릴까요?", "What can I do for you?");
	next;
	switch(select(getitemname(2981) + needtr(" 인챈트", " Enchant"), getitemname(2981) + needtr(" 인챈트 초기화", " Enchant Reset"), needtr("인챈트 리스트 확인", "Check the enchant list"), needtr("대화 종료", "End Conversation"))) {
	case 1:
		mes needtr("[번즈]", "[Byrnes]");
		mes needtr("장착 중인 반지만 인챈트할 수 있습니다.", "I can Enchant only equipped rings. Which one of the");
		mes needtr("어느 쪽에 장착한 반지를 인챈트하시겠습니까?", "equipped rings do you want to Enchant?");
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select(needtr("왼쪽 액세서리", "Left"), needtr("오른쪽 액세서리", "Right")) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes needtr("[번즈]", "[Byrnes]");
			mes needtr("인챈트할 ", "To enchant, please equip the ") + getitemname(2981) + needtr("을 먼저 장착해 주세요.", " first.");
			close;
		case 2981:
			break;
		default:
			mes needtr("[번즈]", "[Byrnes]");
			mes getitemname(2981) + needtr("을 장착해 주세요.", " must be equipped.");
			mes needtr("다른 장비에는 인챈트할 수 없습니다.", "Other equipment cannot be enchanted.");
			close;
		}
		// custom at this point
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[0] > 0) {
			mes needtr("[번즈]", "[Byrnes]");
			mes needtr("모든 슬롯에 인챈트가 부여된 ", "A ") + getitemname(2981) + needtr("에는 더 이상 인챈트할 수 없습니다.", " with every slot enchanted cannot be enchanted any further.");
			close;
		}
		if (.@card[3] == 0) {
			setarray .@enchant_list$[0], "DEF", "MDEF";
			.@slot = 4;
		}
		else if (.@card[2] == 0) {
			setarray .@enchant_list$[0], needtr("최대 HP", "Max HP"), needtr("최대 SP", "Max SP");
			.@slot = 3;
		}
		else if (.@card[1] == 0) {
			setarray .@enchant_list$[0], "ATK", "MATK";
			.@slot = 2;
		}
		else {
			setarray .@enchant_list$[0], "STR", "INT", "AGI", "VIT", "DEX", "LUK";
			.@slot = 1;
		}
		mes needtr("[번즈]", "[Byrnes]");
		mes needtr("슬롯 ", "Which ability do you want as an enchantment on slot ") + .@slot + needtr("에 부여할 인챈트 계열을 선택해 주세요.", "?");
		next;
		.@enchant_list_size = getarraysize(.@enchant_list$);
		for ( .@i = 0; .@i < .@enchant_list_size; ++.@i )
			.@menu$ += .@enchant_list$[.@i] + ":";
		.@type = select(.@menu$) - 1;

		switch(.@slot) {
		case 4:
			switch(.@type) {// enchant list index
			case 0:	// DEF
				setarray .@bonus[0], 4791, 4792, 4793, 4794, 4893;
				break;
			default:	// MDEF
				setarray .@bonus[0], 4890, 4786, 4891, 4787, 4892;
				break;
			}
			break;
		case 3:
			switch(.@type) {
			case 0:	// 최대 HP
				setarray .@bonus[0], 4861, 4862, 4867, 4868, 4900;
				break;
			default:	// 최대 SP
				setarray .@bonus[0], 4800, 4870, 4871, 4801, 4802;
				break;
			}
			break;
		case 2:
			switch(.@type) {
			case 0:	// ATK
				setarray .@bonus[0], 4882, 4766, 4767, 4894, 4895;
				break;
			default:	// MATK
				setarray .@bonus[0], 4883, 4896, 4897, 4898, 4899;
				break;
			}
			break;
		case 1:
			switch(.@type) {
			case 0:	// STR
				setarray .@bonus[0], 4700, 4701, 4702, 4703, 4704;
				break;
			case 1:	// INT
				setarray .@bonus[0], 4710, 4711, 4712, 4713, 4714; 
				break;
			case 2:	// AGI
				setarray .@bonus[0], 4730, 4731, 4732, 4733, 4734;
				break;
			case 3:	// VIT
				setarray .@bonus[0], 4740, 4741, 4742, 4743, 4744;
				break;
			case 4:	// DEX
				setarray .@bonus[0], 4720, 4721, 4722, 4723, 4724;
				break;
			default:	// LUK
				setarray .@bonus[0], 4750, 4751, 4752, 4753, 4754;
				break;
			}
			break;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		.@bonus_size = getarraysize(.@bonus);
		.@card[.@slot-1] = .@bonus[ rand(.@bonus_size) ];	// unknown rates
		specialeffect2 EF_ANGEL2;
		mes needtr("[번즈]", "[Byrnes]");
		mes needtr("인챈트가 완료되었습니다!", "Enchantment completed!");
		delequip .@eqi_slot[.@s];
		getitem2 2981,1,1,0,0,.@card[0],.@card[1],.@card[2],.@card[3];
		close;
	case 2:
		mes needtr("[번즈]", "[Byrnes]");
		mes getitemname(2981) + needtr("의 인챈트를 초기화하려면 ", " - resetting its enchants takes ") + getitemname(6684) + needtr(" 3개가 필요합니다.", " x3.");
		next;
		if (countitem(6684) < 3) {
			mes needtr("[번즈]", "[Byrnes]");
			mes getitemname(6684) + needtr("이 부족하여 인챈트를 초기화할 수 없습니다.", " - not enough to reset the enchants.");
			close;
		}
		mes needtr("[번즈]", "[Byrnes]");
		mes needtr("장착 중인 반지만 초기화할 수 있습니다.", "I can reset only equipped rings. Which one of the");
		mes needtr("어느 쪽에 장착한 반지를 초기화하시겠습니까?", "equipped rings do you want to reset?");
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select(needtr("왼쪽 액세서리", "Left"), needtr("오른쪽 액세서리", "Right")) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes needtr("[번즈]", "[Byrnes]");
			mes needtr("초기화할 ", "To reset, please equip the ") + getitemname(2981) + needtr("을 먼저 장착해 주세요.", " first.");
			close;
		case 2981:
			break;
		default:
			mes needtr("[번즈]", "[Byrnes]");
			mes getitemname(2981) + needtr("을 장착해 주세요.", " must be equipped.");
			mes needtr("다른 장비는 초기화할 수 없습니다.", "Other equipment cannot be reset.");
			close;
		}
		// custom at this point
		if (countitem(6684) < 3) {
			mes needtr("[번즈]", "[Byrnes]");
			mes getitemname(6684) + needtr("이 부족하여 인챈트를 초기화할 수 없습니다.", " - not enough to reset the enchants.");
			close;
		}
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[3] == 0) {
			mes needtr("[번즈]", "[Byrnes]");
			mes needtr("현재 장착한 ", "The ") + getitemname(2981) + needtr("에는 초기화할 인챈트가 없습니다.", " you are wearing has no enchant to reset.");
			close;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		specialeffect2 EF_ANGEL2;
		mes needtr("[번즈]", "[Byrnes]");
		mes needtr("인챈트 초기화가 완료되었습니다.", "Initialization completed.");
		delitem 6684,3;
		delequip .@eqi_slot[.@s];
		getitem 2981,1;
		close;
	case 3:
		callfunc "F_ShowMoroEnchantList";
		end;
	case 4:
		close;
	}

OnInit:
	waitingroom "용사의 반지 인챈트",0;
	end;
}

// 공통 함수 F_ShowEnchantGroup은 다른 공통 스크립트에 선언되어 있어야 합니다.
// 이 파일에서는 중복 선언하지 않고 호출만 합니다.
function	script	F_ShowMoroEnchantList	{
	mes needtr("[번즈]", "[Byrnes]");
	mes getitemname(2981) + needtr("에 부여할 수 있는 인챈트 목록입니다.", " - list of enchants it can receive.");
	mes needtr("인챈트는 ^FF00004번 → 3번 → 2번 → 1번 슬롯^000000 순서로 진행됩니다.", "Enchants are applied in the order ^FF0000slot 4 -> 3 -> 2 -> 1^000000.");
	mes needtr("선택한 계열의 목록 중 하나가 동일 확률로 부여됩니다.", "One option from the chosen series is granted with equal probability.");
	next;

	while (true) {
		switch(select(
			needtr("4번 슬롯 - DEF / MDEF", "Slot 4 - DEF / MDEF"),
			needtr("3번 슬롯 - 최대 HP / 최대 SP", "Slot 3 - Max HP / Max SP"),
			needtr("2번 슬롯 - ATK / MATK", "Slot 2 - ATK / MATK"),
			needtr("1번 슬롯 - 기본 능력치", "Slot 1 - Base stats"),
			needtr("닫기", "Close")
		)) {
		case 1:
			setdialogsize(220,300);
			callfunc "F_ShowEnchantGroup","DEF",4791,4792,4793,4794,4893;
			callfunc "F_ShowEnchantGroup","MDEF",4890,4786,4891,4787,4892;
			break;

		case 2:
			setdialogsize(220,300);
			callfunc "F_ShowEnchantGroup",needtr("최대 HP", "Max HP"),4861,4862,4867,4868,4900;
			callfunc "F_ShowEnchantGroup",needtr("최대 SP", "Max SP"),4800,4870,4871,4801,4802;
			break;

		case 3:
			setdialogsize(220,300);
			callfunc "F_ShowEnchantGroup","ATK",4882,4766,4767,4894,4895;
			callfunc "F_ShowEnchantGroup","MATK",4883,4896,4897,4898,4899;
			break;

		case 4:
			setdialogsize(220,300);
			callfunc "F_ShowEnchantGroup","STR",4700,4701,4702,4703,4704;
			callfunc "F_ShowEnchantGroup","INT",4710,4711,4712,4713,4714;
			callfunc "F_ShowEnchantGroup","AGI",4730,4731,4732,4733,4734;
			callfunc "F_ShowEnchantGroup","VIT",4740,4741,4742,4743,4744;
			callfunc "F_ShowEnchantGroup","DEX",4720,4721,4722,4723,4724;
			callfunc "F_ShowEnchantGroup","LUK",4750,4751,4752,4753,4754;
			break;

		case 5:
			close;
		}
	}
}
