	if (countitem(.@coin) < .@exchange_cost) {

		mes needtr("교환에 필요한 코인이 부족합니다.", "You don't have enough coins for the exchange.");
		close;
	}

	getitem .@exchange_id[.@i], .@exchange_total;
	mes needtr("[코인교환기 CX-1]", "[Coin Exchanger CX-1]");

	mes "^003399"+getitemname(.@exchange_id[.@i])+"^000000 "+.@exchange_total+needtr("개를 획득했습니다.", ".");
	mes needtr("총 수수료 차감량: ^ff3333", "Total fee deducted: ^ff3333")+.@exchange_loss_total+needtr("개^000000", "^000000");
	close;
	
OnInit:
	waitingroom "코인 교환 자판기",0;
	end;
}
