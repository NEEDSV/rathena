S_Exchange:
	.@size = getargcount();
	.@menu$ = "이전 메뉴로 돌아간다.:";
	for (.@i = 0; .@i < .@size; .@i++) {
		if (countitem(getarg(.@i)) < 1)
			.@menu$ += "^999999" + getitemname(getarg(.@i)) + " 없음^000000:";
		else
			.@menu$ += getitemname(getarg(.@i)) + " 보유 중:";
	}
	.@s = select(.@menu$) - 2;
	if (.@s == -1)
		return;

	.@item_id = getarg(.@s);
	.@batch_max = 100;	// 서버 부하/스크립트 과다 반복 방지를 위한 1회 처리 상한

	// 실제 인벤토리에서 미장착 상태의 대상 무기 수량만 계산한다.
	getinventorylist;
	.@available = 0;
	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
		if (@inventorylist_id[.@i] == .@item_id && @inventorylist_equip[.@i] == 0)
			.@available += @inventorylist_amount[.@i];
	}

	if (.@available < 1) {
		mes "[억울한 혼의 사념]";
		mes "^006400" + getitemname(.@item_id) + "^000000이... 보이지 않는다...";
		if (countitem(.@item_id) > 0)
			mes "^FF0000※ 장착 중인 무기는 분해할 수 없습니다. 장착을 해제한 뒤 다시 시도해 주세요.^000000";
		return;
	}

	mes "[억울한 혼의 사념]";
	mes "^006400" + getitemname(.@item_id) + "^000000...";
	mes "이 무기에 깃든 힘을... 에너지 파편으로... 되돌릴 수 있다...";
	mes "현재 분해 가능한 수량은 ^006400" + .@available + "개^000000다...";
	next;

	.@batch_count = .@available;
	if (.@batch_count > .@batch_max)
		.@batch_count = .@batch_max;

	.@choice = select(
		"그만둔다.",
		"1개만 분해한다.",
		"일괄 분해한다. (" + .@batch_count + "개" + (.@available > .@batch_max ? " / 1회 최대 " + .@batch_max + "개" : "") + ")"
	);

	if (.@choice == 1)
		return;

	if (.@choice == 2)
		.@target_count = 1;
	else
		.@target_count = .@batch_count;

	// 결과는 무기 1개당 최대 에너지 파편 3개.
	// 최악의 결과를 기준으로 무게를 미리 확인해 지급 실패를 방지한다.
	if (checkweight(6820, .@target_count * 3) == 0) {
		mes "[억울한 혼의 사념]";
		mes "에너지 파편을 받아 들이기엔... 가방이 너무 무겁다...";

		return;
	}

	if (.@choice == 3 && .@available > .@batch_max) {
		mes "[억울한 혼의 사념]";
		mes "한 번에 너무 많은 사념을 해방하면... 힘이 폭주할 수 있다...";

		next;
	}

	mes "[억울한 혼의 사념]";
	mes "^006400" + getitemname(.@item_id) + " " + .@target_count + "개^000000를 분해하겠다...";
	mes "정말... 괜찮겠나...?";
	next;
	if (select("그만둔다.", "에너지 파편으로 분해한다.") == 1)

	// 삭제 직전에 목록을 새로 받아 유효한 inventory index만 사용한다.
	getinventorylist;

	.@processed = 0;
	.@total_fragment = 0;

	for (.@i = 0; .@i < @inventorylist_count && .@processed < .@target_count; ++.@i) {
		if (@inventorylist_id[.@i] != .@item_id || @inventorylist_equip[.@i] != 0)
			continue;

		.@delete_amount = @inventorylist_amount[.@i];
		if (.@delete_amount > (.@target_count - .@processed))
			.@delete_amount = .@target_count - .@processed;

		// getinventorylist()에서 방금 얻은 유효 index만 사용한다.
		if (!delitemidx(@inventorylist_idx[.@i], .@delete_amount))
			continue;

		for (.@j = 0; .@j < .@delete_amount; ++.@j) {
			.@r = rand(100);
			if (.@r < 70)
				.@total_fragment += 1;
			else if (.@r < 90)
				.@total_fragment += 2;
			else
				.@total_fragment += 3;
		}

		.@processed += .@delete_amount;
	}

	if (.@processed < 1) {
		mes "[억울한 혼의 사념]";
		mes "분해할 무기를... 찾지 못했다...";
		mes "^FF0000※ 인벤토리 상태가 변경되어 분해가 취소되었습니다.^000000";
		return;
	}

	specialeffect EF_BLACKBODY;
	specialeffect EF_GUMGANG4;
	specialeffect EF_DARKCASTING2;
	specialeffect EF_RG_COIN7;
	percentheal -50,-50;
	specialeffect2 EF_POISONSMOKE;
	specialeffect2 EF_NPC_BURNT;

	getitem 6820, .@total_fragment; // Energy Fragment

	mes "[억울한 혼의 사념]";
	mes "^006400" + getitemname(.@item_id) + " " + .@processed + "개^000000에 깃든 힘을...";
	mes "에너지 파편 ^006400" + .@total_fragment + "개^000000로 되돌렸다...";

	if (.@processed < .@target_count) {
		mes "^FF0000※ 인벤토리 상태 변경으로 요청한 수량 중 " + .@processed + "개만 처리되었습니다.^000000";
	}
	else if (.@choice == 3 && .@available > .@processed) {
		mes "^777777※ 동일 무기가 더 남아 있다면 다시 말을 걸어 추가로 분해할 수 있습니다.^000000";
	}

	next;
	mes "- 강한 사념의 파동이 몸을 덮쳐 큰 피해를 입었다. -";
	next;
	mes "[억울한 혼의 사념]";
	mes "다른... 볼일이 있나...?";
	return;

OnBar:
	progressbar_npc "000000",5;
	end;
}

	}

	if (Zeny < .@zeny_cost) {
		mes "[서글픈 혼의 사념]";
		mes "이 원한을... 다른 모습으로 뒤섞으려면...";
		mes "그만한 대가가... 필요하다...";
		mes "^FF000010,000,000 Zeny가 필요합니다.^000000";
		return;
	}

	// 선택한 혼을 제외한 5종 후보 구성
	deletearray .@result_list;
	.@result_count = 0;
	for (.@i = 0; .@i < getarraysize(.@soul_list); ++.@i) {
		if (.@i == .@source_index)
			continue;
		.@result_list[.@result_count] = .@soul_list[.@i];
		.@result_count++;
	}

	.@result_id = .@result_list[rand(.@result_count)];

	if (checkweight(.@result_id,1) == 0) {
		mes "[서글픈 혼의 사념]";
		mes "새로운 혼을... 받아들이기엔...";
		mes "가방이... 너무 무겁다...";
		return;
	}

	mes "[서글픈 혼의 사념]";
	mes "^006400" + getitemname(.@source_id) + " 2개^000000와";
	mes "^00640010,000,000 Zeny^000000를 바쳐...";
	mes "다른 전사자의 혼 하나로... 뒤섞겠나...?";
	next;
	if (select("그만둔다.", "혼을 뒤섞는다.") == 1)
		return;

	progressbar "000000",2;

	// 진행 중 인벤토리/제니 상태가 바뀔 수 있으므로 차감 직전에 재검증
	if (countitem(.@source_id) < 2 || Zeny < .@zeny_cost) {
		mes "[서글픈 혼의 사념]";
		mes "방금 전과... 상태가 달라졌군...";
		mes "재료와 Zeny를... 다시 확인해라...";
		return;
	}

	if (checkweight(.@result_id,1) == 0) {
		mes "[서글픈 혼의 사념]";
		mes "가방이... 너무 무거워졌다...";
		return;
	}

	delitem .@source_id,2;
	Zeny -= .@zeny_cost;

	specialeffect EF_BLACKBODY;
	specialeffect EF_DARKCASTING2;
	specialeffect2 EF_POISONSMOKE;

	getitem .@result_id,1;

	mes "[서글픈 혼의 사념]";
	mes "뒤섞인 원한이... 새로운 모습을 드러냈다...";
	mes "^006400" + getitemname(.@result_id) + " 1개^000000를 얻었다...";
	next;
	return;

S_Make:
	.@reward_id = getarg(0);
	.@equipment_req = getarg(1);
	.@item_misc_req = getarg(2);
	.@is_cloak = getarg(3,0);

	if (.@is_cloak) {
		.@material_misc = 20;  // Ghost Chill
		.@material_energy = 100;
		.@material_name$ = "망자의 한기";
		.@equipment_name$ = "전사자의 망토";
	}
	else {
		.@material_misc = 1;   // Class Soul
		.@material_energy = 20;
		.@material_name$ = "혼";
		.@equipment_name$ = "의상 투구";
	}

	mes "[서글픈 혼의 사념]";
	mes "^006400" + getitemname(.@reward_id) + "^000000을 만들고 싶은가...?";
	next;
	mes "[서글픈 혼의 사념]";
	mes "필요한 것은...";
	mes "^006400" + getitemname(.@equipment_req) + " 1개^000000,";
	mes "^006400" + getitemname(.@item_misc_req) + " " + .@material_misc + "개^000000,";
	mes "그리고 ^006400" + getitemname(6820) + " " + .@material_energy + "개^000000다...";
	mes "혼과 에너지 파편을 5배 사용하면... 반드시 성공한다...";
	if (!.@is_cloak)
		mes "^FF0000※ 귀속된 의상 투구는 조합 재료로 사용할 수 없습니다.^000000";
	next;

	switch (select(
		"이전 메뉴로 돌아간다.",
		"일반 조합 - " + .@material_name$ + " " + .@material_misc + "개 / 에너지 파편 " + .@material_energy + "개 (20%)",
		"안전 조합 - " + .@material_name$ + " " + (.@material_misc * 5) + "개 / 에너지 파편 " + (.@material_energy * 5) + "개 (100%)"
	)) {
	case 1:
		mes "[서글픈 혼의 사념]";
		mes "다른 것을... 살펴보는 것도 좋겠지...";
		return;
	case 2:
		.@rate = 1;
		break;
	case 3:
		.@rate = 5;
		break;
	}

	// 낡은 투구 조합은 귀속되지 않은 의상 투구만 재료로 인정한다.
	// countitem()/delitem()은 동일 Item ID의 귀속 여부를 구분하지 않으므로,
	// 실제 인벤토리에서 bound == 0인 개체의 index를 찾아 직접 소비한다.
	.@equipment_idx = -1;

	if (!.@is_cloak) {
		getinventorylist;
		for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
			if (@inventorylist_id[.@i] == .@equipment_req && @inventorylist_bound[.@i] == 0) {
				.@equipment_idx = @inventorylist_idx[.@i];
				break;
			}
		}

		if (.@equipment_idx < 0) {
			mes "[서글픈 혼의 사념]";
			if (countitem(.@equipment_req) > 0) {
				mes "그 의상에는... 주인을 묶어 둔 힘이 느껴지는군...";
				mes "^FF0000※ 귀속된 의상 투구는 조합 재료로 사용할 수 없습니다.^000000";
				mes "^FF0000귀속되지 않은 " + getitemname(.@equipment_req) + " 1개가 필요합니다.^000000";
			}
			else {
				mes "^006400" + getitemname(.@reward_id) + "^000000을 만들기 위한 의상 투구가... 부족하다...";
			}
			return;
		}
	}
	else if (countitem(.@equipment_req) < 1) {
		mes "[서글픈 혼의 사념]";
		mes "^006400" + getitemname(.@reward_id) + "^000000을 만들기 위한 장비가... 부족하다...";
		return;
	}

	if (countitem(.@item_misc_req) < (.@material_misc * .@rate) || countitem(6820) < (.@material_energy * .@rate)) {
		mes "[서글픈 혼의 사념]";
		mes "^006400" + getitemname(.@reward_id) + "^000000을 만들기 위한 재료가... 부족하다...";
		return;
	}

	progressbar "000000",5;

	// 진행바 이후 재료 상태를 한 번 더 확인한다.
	// 특히 의상 투구는 다시 검색하여 비귀속 개체의 정확한 inventory index를 확보한다.
	if (!.@is_cloak) {
		.@equipment_idx = -1;
		getinventorylist;
		for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
			if (@inventorylist_id[.@i] == .@equipment_req && @inventorylist_bound[.@i] == 0) {
				.@equipment_idx = @inventorylist_idx[.@i];
				break;
			}
		}

		if (.@equipment_idx < 0) {
			mes "[서글픈 혼의 사념]";
			mes "^FF0000귀속되지 않은 " + getitemname(.@equipment_req) + "을 찾을 수 없다... 조합을 중단한다.^000000";
			return;
		}
	}
	else if (countitem(.@equipment_req) < 1) {
		mes "[서글픈 혼의 사념]";
		mes "조합에 필요한 장비가... 사라졌다...";
		return;
	}

	if (countitem(.@item_misc_req) < (.@material_misc * .@rate) || countitem(6820) < (.@material_energy * .@rate)) {
		mes "[서글픈 혼의 사념]";
		mes "조합에 필요한 재료가... 부족해졌다...";
		return;
	}

	percentheal -50,-50;
	specialeffect2 EF_POISONSMOKE;
	specialeffect2 EF_NPC_BURNT;

	if (.@is_cloak)
		delitem .@equipment_req,1;
	else
		delitemidx .@equipment_idx,1;

	delitem .@item_misc_req,(.@material_misc * .@rate);
	delitem 6820,(.@material_energy * .@rate); // Energy Fragment

	if ((20 * .@rate) > rand(100)) {
		specialeffect EF_BLACKBODY;
		specialeffect EF_GUMGANG4;
		specialeffect EF_DARKCASTING2;
		specialeffect EF_RG_COIN7;
		mes "[서글픈 혼의 사념]";
		mes "성공했다...";
		mes "^006400" + getitemname(.@reward_id) + "^000000을 받아라...";
		getitem .@reward_id,1;
	}
	else {
		specialeffect EF_POISONSMOKE;
		specialeffect EF_GUMGANG4;
		mes "[서글픈 혼의 사념]";
		mes "^006400" + getitemname(.@reward_id) + "^000000의 조합에... 실패했다...";
	}
	next;
	mes "- 강한 사념의 파동이 몸을 덮쳐 큰 피해를 입었다. -";
	next;
	mes "[서글픈 혼의 사념]";
	mes "다른... 도움이 필요한가...?";
	return;
}
