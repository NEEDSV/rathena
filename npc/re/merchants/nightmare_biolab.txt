				.@total_fragment += 1;
			else if (.@r < 90)
				.@total_fragment += 2;
			else
				.@total_fragment += 3;
		}

		.@processed += .@delete_amount;
	}

	if (.@processed < 1) {
		mes needtr("[억울한 혼의 사념]", "[Victimized Soul's Mind]");
		mes needtr("분해할 무기를... 찾지 못했다...", "I could not find... the weapon to dismantle...");

		mes needtr("^FF0000※ 인벤토리 상태 변경으로 요청한 수량 중 ", "^FF0000* Your inventory changed, so only ") + .@processed + needtr("개만 처리되었습니다.^000000", " of the requested amount were processed.^000000");

	}

	next;
	mes needtr("- 강한 사념의 파동이 몸을 덮쳐 큰 피해를 입었다. -", "- I got critical damage - - by the mind's powerful wavelength - - which seeped into my body. -");

	// 일괄 분해는 처리 완료 즉시 NPC 대화를 종료하여
	// 이동/다른 NPC 이용을 바로 가능하게 한다.
	if (.@choice == 3) {
		next;
		mes needtr("[억울한 혼의 사념]", "[Victimized Soul's Mind]");
		mes needtr("한꺼번에 뒤엉킨 사념은... 모두 흩어졌다...", "The minds tangled together... have all scattered...");
		mes needtr("필요하다면... 다시 찾아와라...", "Come back to me... if you need to...");
		close;
	}

	next;
	mes needtr("[억울한 혼의 사념]", "[Victimized Soul's Mind]");

			mes needtr("가방이... 너무 무겁다...", "Your bag... is too heavy...");
			close;
		}
		next;
		.@main_menu = select(needtr("장비를 조합한다.", "Combine equipment."), needtr("전사자의 혼을 다른 혼으로 바꾼다.", "Exchange a warrior soul for a different one."), needtr("대화를 끝낸다.", "End the conversation."));
		if (.@main_menu == 3) {
			mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
			mes needtr("...다시... 찾아오길...", "...come back... again...");
			close;
		}

		if (.@main_menu == 2) {
			callsub S_SoulExchange;
			continue;
		}

		.@menu$ = "이전 메뉴로 돌아간다.:";
		for (.@i = 0; .@i < getarraysize(.@reward); .@i++)
			.@menu$ += getitemname(.@reward[.@i]) + ":";
		.@s = select(.@menu$) - 2;
		if (.@s == -1)
			continue;

		if (.@s == 14)
			callsub S_Make,.@reward[.@s],.@equipment[.@s],.@misc[.@s],true;
		else
			callsub S_Make,.@reward[.@s],.@equipment[.@s],.@misc[.@s],false;
	}
	end;

S_SoulExchange:
	// 전사자의 혼 2개 + 10,000,000 Zeny
	// 선택한 혼을 제외한 나머지 5종 중 1종을 균등 랜덤 지급
	setarray .@soul_list[0],
		6814, // Swordman Soul
		6815, // Merchant Soul
		6816, // Thief Soul
		6817, // Mage Soul
		6818, // Archer Soul
		6819; // Acolyte Soul

	.@zeny_cost = 10000000;

	mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
	mes needtr("같은 혼이... 너무 많이 남았나...?", "Too many of the same soul... left over...?");
	mes needtr("서로 다른 원한도... 이곳에서는... 결국 뒤섞이게 마련이지...", "Even different grudges... end up mixing together... in this place...");
	next;
	mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
	mes needtr("같은 종류의 전사자의 혼 ^0064002개^000000와...", "^0064002 warrior souls of the same kind^000000 and...");
	mes needtr("^00640010,000,000 Zeny^000000를 내놓는다면...", "^00640010,000,000 Zeny^000000, if you offer them...");
	mes needtr("그 혼과는 다른... 또 하나의 혼으로... 뒤섞어 주겠다...", "and I will mix them... into another soul... different from that one...");
	next;
	mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
	mes needtr("^FF0000※ 선택한 혼은 결과에서 제외됩니다.^000000", "^FF0000* The soul you choose is excluded from the result.^000000");
	mes needtr("^FF0000※ 나머지 5종의 혼 중 1종이 동일한 확률로 등장합니다.^000000", "^FF0000* One of the other 5 kinds appears, each with the same chance.^000000");
	next;

	.@menu$ = "그만둔다.:";
	for (.@i = 0; .@i < getarraysize(.@soul_list); ++.@i) {
		if (countitem(.@soul_list[.@i]) < 2)
			.@menu$ += "^777777" + getitemname(.@soul_list[.@i]) + " - 2개 미만^000000:";
		else
			.@menu$ += getitemname(.@soul_list[.@i]) + " 2개를 건넨다.:";
	}

	.@choice = select(.@menu$) - 1;
	if (.@choice == 0)
		return;

	.@source_index = .@choice - 1;
	.@source_id = .@soul_list[.@source_index];

	if (countitem(.@source_id) < 2) {
		mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
		mes needtr("그 혼은... 아직 충분히 모이지 않았군...", "That soul... has not gathered enough yet...");
		mes "^FF0000" + getitemname(.@source_id) + needtr(" 2개가 필요합니다.^000000", " x2 is required.^000000");
		return;
	}

	if (Zeny < .@zeny_cost) {
		mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
		mes needtr("이 원한을... 다른 모습으로 뒤섞으려면...", "To mix this grudge... into another shape...");
		mes needtr("그만한 대가가... 필요하다...", "a price to match... is needed...");
		mes needtr("^FF000010,000,000 Zeny가 필요합니다.^000000", "^FF000010,000,000 Zeny is required.^000000");
		return;
	}

	// 선택한 혼을 제외한 5종 후보 구성
	deletearray .@result_list;
	.@result_count = 0;
	for (.@i = 0; .@i < getarraysize(.@soul_list); ++.@i) {
		if (.@i == .@source_index)
			continue;
		.@result_list[.@result_count] = .@soul_list[.@i];
		.@result_count++;
	}

	.@result_id = .@result_list[rand(.@result_count)];

	if (checkweight(.@result_id,1) == 0) {
		mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
		mes needtr("새로운 혼을... 받아들이기엔...", "To take in a new soul...");
		mes needtr("가방이... 너무 무겁다...", "Your bag... is too heavy...");
		return;
	}

	mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
	mes "^006400" + getitemname(.@source_id) + needtr(" 2개^000000와", " x2^000000 and");
	mes needtr("^00640010,000,000 Zeny^000000를 바쳐...", "^00640010,000,000 Zeny^000000 offered up...");
	mes needtr("다른 전사자의 혼 하나로... 뒤섞겠나...?", "shall I mix them into one other warrior soul...?");
	next;
	if (select(needtr("그만둔다.", "Quit."), needtr("혼을 뒤섞는다.", "Mix the souls.")) == 1)
		return;

	progressbar "000000",2;

	// 진행 중 인벤토리/제니 상태가 바뀔 수 있으므로 차감 직전에 재검증
	if (countitem(.@source_id) < 2 || Zeny < .@zeny_cost) {
		mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
		mes needtr("방금 전과... 상태가 달라졌군...", "Something has changed... since a moment ago...");
		mes needtr("재료와 Zeny를... 다시 확인해라...", "Check your materials and Zeny... again...");
		return;
	}

	if (checkweight(.@result_id,1) == 0) {
		mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
		mes needtr("가방이... 너무 무거워졌다...", "Your bag... has grown too heavy...");
		return;
	}

	delitem .@source_id,2;
	Zeny -= .@zeny_cost;

	specialeffect EF_BLACKBODY;
	specialeffect EF_DARKCASTING2;
	specialeffect2 EF_POISONSMOKE;

	getitem .@result_id,1;

	mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
	mes needtr("뒤섞인 원한이... 새로운 모습을 드러냈다...", "The mixed grudge... has taken a new shape...");
	mes "^006400" + getitemname(.@result_id) + needtr(" 1개^000000를 얻었다...", " x1^000000 obtained...");
	next;
	return;

S_Make:
	.@reward_id = getarg(0);
	.@equipment_req = getarg(1);
	.@item_misc_req = getarg(2);
	.@is_cloak = getarg(3,0);

	if (.@is_cloak) {
		.@material_misc = 20;  // Ghost Chill
		.@material_energy = 100;
		.@material_name$ = "망자의 한기";
		.@equipment_name$ = "전사자의 망토";
	}
	else {
		.@material_misc = 1;   // Class Soul
		.@material_energy = 20;
		.@material_name$ = "혼";
		.@equipment_name$ = "의상 투구";
	}

	mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
	mes "^006400" + getitemname(.@reward_id) + needtr("^000000을 만들고 싶은가...?", "^000000...");
	next;
	mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
	mes needtr("필요한 것은...", "What I need is...");
	mes "^006400" + getitemname(.@equipment_req) + needtr(" 1개^000000,", " x1^000000,");
	mes "^006400" + getitemname(.@item_misc_req) + " " + .@material_misc + needtr("개^000000,", "^000000,");
	mes needtr("그리고 ^006400", "and ^006400") + getitemname(6820) + " " + .@material_energy + needtr("개^000000다...", "^000000 of them right now...");
	mes needtr("혼과 에너지 파편을 5배 사용하면... 반드시 성공한다...", "Use five times the souls and Energy Fragments... and it cannot fail...");
	if (!.@is_cloak)
		mes needtr("^FF0000※ 귀속된 의상 투구는 조합 재료로 사용할 수 없습니다.^000000", "^FF0000* A bound costume headgear cannot be used as a combination material.^000000");
	next;

	switch (select(
		needtr("이전 메뉴로 돌아간다.", "Go back to previous menu."),
		needtr("일반 조합 - ", "Normal combination - ") + .@material_name$ + " " + .@material_misc + needtr("개 / 에너지 파편 ", " / Energy Fragment ") + .@material_energy + needtr("개 (20%)", " (20%)"),
		needtr("안전 조합 - ", "Safe combination - ") + .@material_name$ + " " + (.@material_misc * 5) + needtr("개 / 에너지 파편 ", " / Energy Fragment ") + (.@material_energy * 5) + needtr("개 (100%)", " (100%)")
	)) {
	case 1:
		mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
		mes needtr("다른 것을... 살펴보는 것도 좋겠지...", "I hoped that... you could... see... other things... as well...");
		return;
	case 2:
		.@rate = 1;
		break;
	case 3:
		.@rate = 5;
		break;
	}

	// 낡은 투구 조합은 귀속되지 않은 의상 투구만 재료로 인정한다.
	// countitem()/delitem()은 동일 Item ID의 귀속 여부를 구분하지 않으므로,
	// 실제 인벤토리에서 bound == 0인 개체의 index를 찾아 직접 소비한다.
	.@equipment_idx = -1;

	if (!.@is_cloak) {
		getinventorylist;
		for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
			if (@inventorylist_id[.@i] == .@equipment_req && @inventorylist_bound[.@i] == 0) {
				.@equipment_idx = @inventorylist_idx[.@i];
				break;
			}
		}

		if (.@equipment_idx < 0) {
			mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
			if (countitem(.@equipment_req) > 0) {
				mes needtr("그 의상에는... 주인을 묶어 둔 힘이 느껴지는군...", "I feel a power on that costume... binding it to its owner...");
				mes needtr("^FF0000※ 귀속된 의상 투구는 조합 재료로 사용할 수 없습니다.^000000", "^FF0000* A bound costume headgear cannot be used as a combination material.^000000");
				mes needtr("^FF0000귀속되지 않은 ", "^FF0000An unbound ") + getitemname(.@equipment_req) + needtr(" 1개가 필요합니다.^000000", " is required.^000000");
			}
			else {
				mes "^006400" + getitemname(.@reward_id) + needtr("^000000을 만들기 위한 의상 투구가... 부족하다...", "^000000... you do not have enough costume headgear to make it...");
			}
			return;
		}
	}
	else if (countitem(.@equipment_req) < 1) {
		mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
		mes "^006400" + getitemname(.@reward_id) + needtr("^000000을 만들기 위한 장비가... 부족하다...", "^000000... you do not have enough equipment to make it...");
		return;
	}

	if (countitem(.@item_misc_req) < (.@material_misc * .@rate) || countitem(6820) < (.@material_energy * .@rate)) {
		mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
		mes "^006400" + getitemname(.@reward_id) + needtr("^000000을 만들기 위한 재료가... 부족하다...", "^000000...");
		return;
	}

	progressbar "000000",5;

	// 진행바 이후 재료 상태를 한 번 더 확인한다.
	// 특히 의상 투구는 다시 검색하여 비귀속 개체의 정확한 inventory index를 확보한다.
	if (!.@is_cloak) {
		.@equipment_idx = -1;
		getinventorylist;
		for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
			if (@inventorylist_id[.@i] == .@equipment_req && @inventorylist_bound[.@i] == 0) {
				.@equipment_idx = @inventorylist_idx[.@i];
				break;
			}
		}

		if (.@equipment_idx < 0) {
			mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
			mes needtr("^FF0000귀속되지 않은 ", "^FF0000An unbound ") + getitemname(.@equipment_req) + needtr("을 찾을 수 없다... 조합을 중단한다.^000000", " cannot be found... the combination stops here.^000000");
			return;
		}
	}
	else if (countitem(.@equipment_req) < 1) {
		mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
		mes needtr("조합에 필요한 장비가... 사라졌다...", "The equipment needed for the combination... is gone...");
		return;
	}

	if (countitem(.@item_misc_req) < (.@material_misc * .@rate) || countitem(6820) < (.@material_energy * .@rate)) {
		mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
		mes needtr("조합에 필요한 재료가... 부족해졌다...", "The materials needed for the combination... have run short...");
		return;
	}

	percentheal -50,-50;
	specialeffect2 EF_POISONSMOKE;
	specialeffect2 EF_NPC_BURNT;

	if (.@is_cloak)
		delitem .@equipment_req,1;
	else
		delitemidx .@equipment_idx,1;

	delitem .@item_misc_req,(.@material_misc * .@rate);
	delitem 6820,(.@material_energy * .@rate); // Energy Fragment

	if ((20 * .@rate) > rand(100)) {
		specialeffect EF_BLACKBODY;
		specialeffect EF_GUMGANG4;
		specialeffect EF_DARKCASTING2;
		specialeffect EF_RG_COIN7;
		mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
		mes needtr("성공했다...", "It worked...");
		mes "^006400" + getitemname(.@reward_id) + needtr("^000000을 받아라...", "^000000...");
		getitem .@reward_id,1;
	}
	else {
		specialeffect EF_POISONSMOKE;
		specialeffect EF_GUMGANG4;
		mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
		mes "^006400" + getitemname(.@reward_id) + needtr("^000000의 조합에... 실패했다...", "^000000... failed...");
	}
	next;
	mes needtr("- 강한 사념의 파동이 몸을 덮쳐 큰 피해를 입었다. -", "- I got critical damage - - by the mind's powerful wavelength - - which seeped into my body. -");
	next;
	mes needtr("[서글픈 혼의 사념]", "[Sorrowful Soul's Mind]");
	mes needtr("다른... 도움이 필요한가...?", "Is there... anything... I can... help you with?...");
	return;
}
