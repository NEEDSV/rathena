	killmonster "lhz_dun_n", "lhz_dun_n_MVP::OnMyMVPDead";

	callsub S_SpawnMVP;
	end;

	.@mob_id = .MVPList[rand(getarraysize(.MVPList))];

	// 기존 rAthena 기준 12시 방향 MVP 위치

		.@mob_id,
		1,
		"lhz_dun_n_MVP::OnMyMVPDead";

	end;


OnMyMVPDead:
	// 2013 사양: MVP 사망 즉시 다음 MVP 등장
	callsub S_SpawnMVP;
	end;
