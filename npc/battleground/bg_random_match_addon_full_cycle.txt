// - 한 번의 매칭으로 티에라 -> 플라비우스 -> KVM 전체 사이클 진행
// - 최초 티에라는 자동 입장, 이후 전장은 양 팀 파티장의 입장 준비로 진행
// - 티에라/플라비우스 종료 후 파티 유지, KVM 종료 후에만 자동 해체
// - 명령어는 bat_room에서만 사용 가능
//   @전장매칭 : 설정된 시간 동안 모집 시작 (신청자는 자동 참가)
//   @전장참가 : 진행 중인 모집 참가
//   @전장취소 : 참가 신청 철회
//   @전장현황 : 모집/진행 상태 확인
// - 테스트본: 최소 4명 / 최대 4명 (2 vs 2)
// - 동일 직업(Class)을 가능한 한 양 팀으로 분산
// - 모집 중 전장 관리인은 유지하고 홍보 외 메뉴만 잠금
// - 자동 선정 파티장을 기존 입장 준비 시스템에 등록하여 다음 전장 진행
// - 자동 선정 파티장은 기존 수동 파티장 추가 보상 대상에서 제외
// - 모집 중 bat_room 이탈/로그아웃/파티 가입 시 참가 신청 자동 취소
//============================================================

function	script	F_BG_RM_ModeName$	{
	switch (getarg(0)) {
	case 1: return "티에라 협곡";
	case 2: return "플라비우스";
	case 3: return "KVM";
	}
	return "알 수 없음";
}

	.@count += getwaitingroomstate(0,"기욤 대기실#bat_a01");
	.@count += getwaitingroomstate(0,"크루아 대기실#bat_a01");
	.@count += getwaitingroomstate(0,"기욤 대기실#bat_b01");
	.@count += getwaitingroomstate(0,"크루아 대기실#bat_b01");
	.@count += getwaitingroomstate(0,"KvM01R_Guillaume");
	.@count += getwaitingroomstate(0,"KvM01R_Croix");
	return .@count;
}

function	script	F_BG_RM_SetWaitingRooms	{
	.@enable = getarg(0);

	// 현재 통합 스크립트는 waitingroom 대신 파티장 입장 준비 방식을 사용합니다.
	// 기존 대기실 NPC가 우연히 활성화되어 있지 않도록 모집 중에는 숨기고,
	// 종료 후에는 각 전장 OnRefresh 정책에 맡깁니다.
	if (!.@enable) {
		disablenpc "기욤 대기실#bat_a01";
		disablenpc "크루아 대기실#bat_a01";
		disablenpc "기욤 대기실#bat_b01";
		disablenpc "크루아 대기실#bat_b01";
		disablenpc "KvM01R_Guillaume";
		disablenpc "KvM01R_Croix";
	}
	else {
		donpcevent "티에라 대기실 갱신#bat_a01::OnRefresh";
		donpcevent "플라비우스 대기실 갱신#bat_b01::OnRefresh";
		donpcevent "KVM 대기실 갱신#bat_c01::OnRefresh";
	}
	return;
}

function	script	F_BG_RM_SetManager	{
	// 랜덤 매칭 모집 중에도 전장 관리인 NPC 자체는 유지합니다.
	// 대신 bg_common.txt에서 홍보 메뉴를 제외한 나머지 메뉴만 잠급니다.
	$@BG_RM_MANAGER_MENU_LOCK = !getarg(0);
	enablenpc "전장 관리인";
	return;
}

function	script	F_BG_RM_PrepareTeamMembers	{
	.@team = getarg(0);
	.@bg_id = getarg(1);
	.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;

	for (.@i = 0; .@i < .@count; .@i++) {
		.@aid = (.@team == 1) ? $@BG_RM_G_AID[.@i] : $@BG_RM_C_AID[.@i];
		.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];

		if (!isloggedin(.@aid,.@cid))
			continue;
		if (!attachrid(.@aid))
			continue;
		if (getcharid(0) != .@cid) {
			detachrid;
			continue;
		}

		if (getcharid(4) == .@bg_id) {
			Bat_Team = .@team;
			percentheal 100,100;
		}
		detachrid;
	}
	return;
}

function	script	F_BG_RM_ResetMembers	{
	for (.@team = 1; .@team <= 2; .@team++) {
		.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;

		for (.@i = 0; .@i < .@count; .@i++) {
			.@aid = (.@team == 1) ? $@BG_RM_G_AID[.@i] : $@BG_RM_C_AID[.@i];
			.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];

			if (!isloggedin(.@aid,.@cid))
				continue;
			if (!attachrid(.@aid))
				continue;
			if (getcharid(0) != .@cid) {
				detachrid;
				continue;
			}

			// BG 관련 상태는 항상 정리합니다.
			Bat_Team = 0;
			if (getcharid(4))
				bg_leave;

			// 실제 전장 내부에 남아 있는 경우에만 대기존으로 이동합니다.
			// 이미 bat_room 또는 다른 맵에 있는 참가자를 154,150으로 다시 끌어오지 않습니다.
			.@map$ = strcharinfo(3);
			if (
				.@map$ == "bat_a01" ||
				.@map$ == "bat_b01" ||
				.@map$ == "bat_c01"
			)
				warp "bat_room",154,150;

			detachrid;
		}
	}
	return;
}
function	script	F_BG_RM_ClearArrays	{
	cleararray $@BG_RM_AID[0], 0, MAX_PARTY * 2;
	cleararray $@BG_RM_CID[0], 0, MAX_PARTY * 2;
	cleararray $@BG_RM_CLASS[0], 0, MAX_PARTY * 2;
	cleararray $@BG_RM_NAME$[0], "", MAX_PARTY * 2;

	cleararray $@BG_RM_G_AID[0], 0, MAX_PARTY;
	cleararray $@BG_RM_G_CID[0], 0, MAX_PARTY;
	cleararray $@BG_RM_G_NAME$[0], "", MAX_PARTY;

	cleararray $@BG_RM_C_AID[0], 0, MAX_PARTY;
	cleararray $@BG_RM_C_CID[0], 0, MAX_PARTY;
	cleararray $@BG_RM_C_NAME$[0], "", MAX_PARTY;

	$@BG_RM_COUNT = 0;
	$@BG_RM_G_COUNT = 0;
	$@BG_RM_C_COUNT = 0;
	return;
}


// 모집 명단에서 지정된 슬롯을 제거하고 뒤의 데이터를 앞으로 당깁니다.
function	script	F_BG_RM_RemoveRecruitSlot	{
	.@slot = getarg(0);

	if (.@slot < 0 || .@slot >= $@BG_RM_COUNT)
		return 0;

	for (.@i = .@slot; .@i < $@BG_RM_COUNT - 1; .@i++) {
		$@BG_RM_AID[.@i] = $@BG_RM_AID[.@i + 1];
		$@BG_RM_CID[.@i] = $@BG_RM_CID[.@i + 1];
		$@BG_RM_CLASS[.@i] = $@BG_RM_CLASS[.@i + 1];
		$@BG_RM_NAME$[.@i] = $@BG_RM_NAME$[.@i + 1];
	}

	$@BG_RM_COUNT--;
	$@BG_RM_AID[$@BG_RM_COUNT] = 0;
	$@BG_RM_CID[$@BG_RM_COUNT] = 0;
	$@BG_RM_CLASS[$@BG_RM_COUNT] = 0;
	$@BG_RM_NAME$[$@BG_RM_COUNT] = "";
	return 1;
}

// 최초 티에라 입장 직전에 최종 팀원이 모두 대기실과 자동 생성 파티에 남아 있는지 확인합니다.
function	script	F_BG_RM_ValidateInitialReady	{
	for (.@team = 1; .@team <= 2; .@team++) {
		.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;
		.@party_id = (.@team == 1) ? $@BG_RM_PARTY_G : $@BG_RM_PARTY_C;

		for (.@i = 0; .@i < .@count; .@i++) {
			.@aid = (.@team == 1) ? $@BG_RM_G_AID[.@i] : $@BG_RM_C_AID[.@i];
			.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];
			.@name$ = (.@team == 1) ? $@BG_RM_G_NAME$[.@i] : $@BG_RM_C_NAME$[.@i];

			if (!isloggedin(.@aid,.@cid))
				return 0;

			if (getmapxy(.@map$,.@x,.@y,BL_PC,.@name$) != 0)
				return 0;

			if (.@map$ != "bat_room")
				return 0;

			if (getcharid(1,.@name$) != .@party_id)
				return 0;

			if (getcharid(4,.@name$))
				return 0;
		}
	}

	return 1;
}

function	script	F_BG_RM_CancelReason$	{
	switch(getarg(0)) {
	case 1:
		return "게임 종료";
	case 2:
		return "전장 대기존 이탈";
	case 3:
		return "파티 또는 전장 팀 가입";
	case 4:
		return "접속 위치 확인 실패";
	case 5:
		return "직접 참가 취소";
	case 6:
		return "랜덤 매칭 참가 레벨 미달";
	}
	return "알 수 없는 사유";
}

function	script	F_BG_RM_FindRecruitSlot	{
	.@aid = getarg(0);
	.@cid = getarg(1);

	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		if ($@BG_RM_AID[.@i] == .@aid && $@BG_RM_CID[.@i] == .@cid)
			return .@i;
	}

	return -1;
}

// 온라인 상태에서 자동 취소된 참가자에게 눈에 띄는 개인 알림을 표시하고
// 같은 모집에서 @전장현황을 확인할 때 취소 사유가 보이도록 기록합니다.
function	script	F_BG_RM_NotifyAutoCancel	{
	.@aid = getarg(0);
	.@cid = getarg(1);
	.@reason = getarg(2);
	.@reason$ = callfunc("F_BG_RM_CancelReason$", .@reason);

	if (!isloggedin(.@aid,.@cid))
		return 0;
	if (!attachrid(.@aid))
		return 0;
	if (getcharid(0) != .@cid) {
		detachrid;
		return 0;
	}

	BG_RM_LAST_CANCEL_REASON = .@reason;
	BG_RM_LAST_CANCEL_TOKEN = $@BG_RM_TOKEN;
	BG_RM_LAST_CANCEL_TICK = gettimetick(2);

	announce "[전장] 랜덤 매칭 참가 신청이 자동 취소되었습니다.", bc_self, "0xFF5555";
	dispbottom "[전장] 취소 사유: " + .@reason$;
	dispbottom "[전장] 다시 참가하려면 bat_room에서 @전장참가를 입력해 주세요.";
	detachrid;
	return 1;
}

// 모집 중인 참가자에게 개인 상태 유지 알림을 보냅니다.
function	script	F_BG_RM_NotifyApplicants	{
	.@remain = getarg(0);

	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		.@aid = $@BG_RM_AID[.@i];
		.@cid = $@BG_RM_CID[.@i];
		.@name$ = $@BG_RM_NAME$[.@i];

		if (!isloggedin(.@aid,.@cid))
			continue;

		message .@name$,
			"[전장] 참가 유지 중 | 남은 시간 " + .@remain
			+ " | 현재 " + $@BG_RM_COUNT + "명 | 상태 확인: @전장현황";
	}
	return;
}

//============================================================
// 전장 대기존 미신청 인원 안내
// - 모집 중 30초마다 전장 관리인이 NPC 채팅으로 안내
// - 이름이 많으면 6명씩 순환하여 표시
//============================================================
function	script	F_BG_RM_AnnounceMissingApplicants	{
	if ($@BG_RM_STATE != 1)
		return;

	.@unit_count = getmapunits(BL_PC,"bat_room",.@unit_aid[0]);
	.@missing_count = 0;

	for (.@i = 0; .@i < .@unit_count; .@i++) {
		.@aid = .@unit_aid[.@i];
		.@cid = convertpcinfo(.@aid,CPC_CHAR);
		.@name$ = convertpcinfo(.@aid,CPC_NAME);

		if (.@aid <= 0 || .@cid <= 0 || .@name$ == "")
			continue;

		// 운영자/테스트 캐릭터 미신청 명단 제외
		.@excluded = 0;
		for (.@j = 0; .@j < $@BG_RM_MISSING_EXCLUDE_COUNT; .@j++) {
			if (.@name$ == $@BG_RM_MISSING_EXCLUDE_NAME$[.@j]) {
				.@excluded = 1;
				break;
			}
		}
		if (.@excluded)
			continue;

		// 참가 레벨 미달 캐릭터에게는 미신청 홍보를 하지 않습니다.
		if ($@BG_RM_MIN_JOIN_LEVEL > 0 && readparam(BaseLevel,.@cid) < $@BG_RM_MIN_JOIN_LEVEL)
			continue;

		if (callfunc("F_BG_RM_FindRecruitSlot",.@aid,.@cid) >= 0)
			continue;

		.@missing_name$[.@missing_count] = .@name$;
		.@missing_count++;
	}

	if (.@missing_count <= 0) {
		$@BG_RM_MISSING_CURSOR = 0;
		return;
	}

	.@start = $@BG_RM_MISSING_CURSOR;
	if (.@start < 0 || .@start >= .@missing_count)
		.@start = 0;

	.@show_count = 6;
	if (.@missing_count < .@show_count)
		.@show_count = .@missing_count;

	.@list$ = "";
	for (.@i = 0; .@i < .@show_count; .@i++) {
		.@index = (.@start + .@i) % .@missing_count;

		if (.@list$ != "")
			.@list$ += ", ";

		.@list$ += .@missing_name$[.@index];
	}

	.@message$ = "랜덤 매칭 미신청 " + .@missing_count + "명: " + .@list$;

	if (.@missing_count > .@show_count)
		.@message$ += " 외 " + (.@missing_count - .@show_count) + "명";

	.@message$ += " / 참가: @전장참가";
	npctalk .@message$, "전장 관리인";

	$@BG_RM_MISSING_CURSOR = (.@start + .@show_count) % .@missing_count;
	return;
}

function	script	F_BG_RM_PerformerFamily	{
	.@class = getarg(0);

	if (
		.@class == JOB_BARD ||
		.@class == JOB_CLOWN ||
		.@class == JOB_MINSTREL ||
		.@class == JOB_MINSTREL_T
	)
		return 1;

	if (
		.@class == JOB_DANCER ||
		.@class == JOB_GYPSY ||
		.@class == JOB_WANDERER ||
		.@class == JOB_WANDERER_T
	)
		return 2;

	return 0;
}

//============================================================
// 티에라 입장 시 직업 구성 표시용 13개 직업 계열
//
// 저레벨 참가자도 보기 좋게 2차/전승 직업을 해당 3차 계열로 묶습니다.
// 알 수 없는/특수 직업은 "기타"로 표시합니다.
//============================================================
function	script	F_BG_RM_NoticeJobFamily	{
	.@class = getarg(0);

	// 룬나이트 계열
	if (
		.@class == JOB_KNIGHT ||
		.@class == JOB_KNIGHT2 ||
		.@class == JOB_LORD_KNIGHT ||
		.@class == JOB_LORD_KNIGHT2 ||
		.@class == JOB_RUNE_KNIGHT ||
		.@class == JOB_RUNE_KNIGHT_T ||
		.@class == JOB_RUNE_KNIGHT2 ||
		.@class == JOB_RUNE_KNIGHT_T2
	)
		return 1;

	// 로열가드 계열
	if (
		.@class == JOB_CRUSADER ||
		.@class == JOB_CRUSADER2 ||
		.@class == JOB_PALADIN ||
		.@class == JOB_PALADIN2 ||
		.@class == JOB_ROYAL_GUARD ||
		.@class == JOB_ROYAL_GUARD_T ||
		.@class == JOB_ROYAL_GUARD2 ||
		.@class == JOB_ROYAL_GUARD_T2
	)
		return 2;

	// 워록 계열
	if (
		.@class == JOB_WIZARD ||
		.@class == JOB_HIGH_WIZARD ||
		.@class == JOB_WARLOCK ||
		.@class == JOB_WARLOCK_T
	)
		return 3;

	// 소서러 계열
	if (
		.@class == JOB_SAGE ||
		.@class == JOB_PROFESSOR ||
		.@class == JOB_SORCERER ||
		.@class == JOB_SORCERER_T
	)
		return 4;

	// 레인저 계열
	if (
		.@class == JOB_HUNTER ||
		.@class == JOB_SNIPER ||
		.@class == JOB_RANGER ||
		.@class == JOB_RANGER_T ||
		.@class == JOB_RANGER2 ||
		.@class == JOB_RANGER_T2
	)
		return 5;

	// 민스트럴 계열
	if (
		.@class == JOB_BARD ||
		.@class == JOB_CLOWN ||
		.@class == JOB_MINSTREL ||
		.@class == JOB_MINSTREL_T
	)
		return 6;

	// 원더러 계열
	if (
		.@class == JOB_DANCER ||
		.@class == JOB_GYPSY ||
		.@class == JOB_WANDERER ||
		.@class == JOB_WANDERER_T
	)
		return 7;

	// 미케닉 계열
	if (
		.@class == JOB_BLACKSMITH ||
		.@class == JOB_WHITESMITH ||
		.@class == JOB_MECHANIC ||
		.@class == JOB_MECHANIC_T ||
		.@class == JOB_MECHANIC2 ||
		.@class == JOB_MECHANIC_T2
	)
		return 8;

	// 제네릭 계열
	if (
		.@class == JOB_ALCHEMIST ||
		.@class == JOB_CREATOR ||
		.@class == JOB_GENETIC ||
		.@class == JOB_GENETIC_T
	)
		return 9;

	// 길로틴크로스 계열
	if (
		.@class == JOB_ASSASSIN ||
		.@class == JOB_ASSASSIN_CROSS ||
		.@class == JOB_GUILLOTINE_CROSS ||
		.@class == JOB_GUILLOTINE_CROSS_T
	)
		return 10;

	// 쉐도우체이서 계열
	if (
		.@class == JOB_ROGUE ||
		.@class == JOB_STALKER ||
		.@class == JOB_SHADOW_CHASER ||
		.@class == JOB_SHADOW_CHASER_T
	)
		return 11;

	// 아크비숍 계열
	if (
		.@class == JOB_PRIEST ||
		.@class == JOB_HIGH_PRIEST ||
		.@class == JOB_ARCH_BISHOP ||
		.@class == JOB_ARCH_BISHOP_T
	)
		return 12;

//============================================================
// 티에라 내부 준비 구역 입장 직후 양 팀 멤버/직업 구성 공지
//
// 한 줄이 너무 길어지는 것을 피하기 위해 닉네임은 6명 단위로 분할합니다.
// 양 팀 모두 같은 정보를 볼 수 있도록 bat_a01 맵 전체에 출력합니다.
//============================================================
function	script	F_BG_RM_AnnounceTierraTeamInfo	{
	for (.@team = 1; .@team <= 2; .@team++) {
		.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;
		.@team_name$ = (.@team == 1) ? "기욤" : "크루아";
		.@color$ = (.@team == 1) ? "0x66CCFF" : "0xFF7777";

		cleararray .@job_count[0], 0, 15;
		.@other_count = 0;

		// 직업 구성 집계
		for (.@i = 0; .@i < .@count; .@i++) {
			.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];
			.@class = readparam(Class,.@cid);
			.@family = callfunc("F_BG_RM_NoticeJobFamily", .@class);

			if (.@family > 0 && .@family <= 14)
				.@job_count[.@family]++;
			else
				.@other_count++;
		}

		.@composition$ = "";
		for (.@family = 1; .@family <= 14; .@family++) {
			if (.@job_count[.@family] <= 0)
				continue;

			.@job_name$ = callfunc("F_BG_RM_NoticeJobName$", .@family);

			if (.@composition$ != "")
				.@composition$ += " / ";

			.@composition$ += .@job_name$ + "-" + .@job_count[.@family];
		}

		if (.@other_count > 0) {
			if (.@composition$ != "")
				.@composition$ += " / ";
			.@composition$ += "기타-" + .@other_count;
		}

		if (.@composition$ == "")
			.@composition$ = "구성 정보 없음";

		mapannounce "bat_a01",
			"[랜덤 매칭][" + .@team_name$ + " 직업 구성] " + .@composition$,
			bc_map, .@color$;

		// 닉네임은 6명씩 나눠서 출력합니다.
		.@page_count = (.@count + 5) / 6;

		for (.@page = 0; .@page < .@page_count; .@page++) {
			.@member_list$ = "";
			.@start = .@page * 6;
			.@end = .@start + 6;
			if (.@end > .@count)
				.@end = .@count;

			for (.@i = .@start; .@i < .@end; .@i++) {
				.@name$ = (.@team == 1) ? $@BG_RM_G_NAME$[.@i] : $@BG_RM_C_NAME$[.@i];

				if (.@member_list$ != "")
					.@member_list$ += ", ";

				.@member_list$ += .@name$;
			}

			mapannounce "bat_a01",
				"[랜덤 매칭][" + .@team_name$ + " 멤버 "
				+ (.@page + 1) + "/" + .@page_count + "] " + .@member_list$,
				bc_map, .@color$;
		}
	}

	return;
}


//============================================================
// 랜덤 매칭 단독 민스트럴 현행(버프형) 모드 강제
//
// @모드전환의 캐릭터 저장 키:
//   NEED_SOLO_PERFORMANCE_MODE
//   0 = 장판형(Classic)
//   1 = 현행 버프형(Modern)
//
// 랜덤 매칭 전체 참가자 중 실제 민스트럴 계열
// (JOB_MINSTREL / JOB_MINSTREL_T)이 정확히 1명일 때만 사용합니다.
// 원래 설정값은 NEED_BG_RM_PERFORMANCE_RESTORE에 보관하고
// 풀사이클 종료/취소 시 복구합니다.
//============================================================
function	script	F_BG_RM_EnsureForcedPerformanceMode	{
	if ($@BG_RM_FORCED_PERFORMANCE_AID <= 0 || $@BG_RM_FORCED_PERFORMANCE_CID <= 0)
		return 0;

	if (!isloggedin($@BG_RM_FORCED_PERFORMANCE_AID,$@BG_RM_FORCED_PERFORMANCE_CID))
		return 0;

	.@caller_aid = getattachedrid();

	if (!attachrid($@BG_RM_FORCED_PERFORMANCE_AID)) {
		if (.@caller_aid)
			attachrid .@caller_aid;
		return 0;
	}

	if (getcharid(0) != $@BG_RM_FORCED_PERFORMANCE_CID) {
		if (.@caller_aid)
			attachrid .@caller_aid;
		else
			detachrid;
		return 0;
	}

	// +1 형태로 저장하여 원래 값이 0이어도 복구 예약 여부를 구분합니다.
	if (!NEED_BG_RM_PERFORMANCE_RESTORE)
		NEED_BG_RM_PERFORMANCE_RESTORE = NEED_SOLO_PERFORMANCE_MODE + 1;

	NEED_SOLO_PERFORMANCE_MODE = 1;

	if (.@caller_aid)
		attachrid .@caller_aid;
	else
		detachrid;

	return 1;
}

function	script	F_BG_RM_RestoreForcedPerformanceMode	{
	.@aid = $@BG_RM_FORCED_PERFORMANCE_AID;
	.@cid = $@BG_RM_FORCED_PERFORMANCE_CID;
	$@BG_RM_FORCED_PERFORMANCE_AID = 0;
	$@BG_RM_FORCED_PERFORMANCE_CID = 0;

	if (.@aid <= 0 || .@cid <= 0)
		return;

	if (!isloggedin(.@aid,.@cid))
		return;

	.@caller_aid = getattachedrid();

	if (!attachrid(.@aid)) {
		if (.@caller_aid)
			attachrid .@caller_aid;
		return;
	}

	if (getcharid(0) == .@cid && NEED_BG_RM_PERFORMANCE_RESTORE) {
		NEED_SOLO_PERFORMANCE_MODE = NEED_BG_RM_PERFORMANCE_RESTORE - 1;
		NEED_BG_RM_PERFORMANCE_RESTORE = 0;
		dispbottom "[전장] 랜덤 매칭이 종료되어 @모드전환 설정이 이전 상태로 복구되었습니다.";
	}

	if (.@caller_aid)
		attachrid .@caller_aid;
	else
		detachrid;

	return;
}

function	script	F_BG_RM_StartBattle	{
	.@mode = getarg(0);

	// 단독 민스트럴이 있는 매칭은 각 전장 입장 직전에도 현행 모드를 다시 보장합니다.
	if ($@BG_RM_FORCED_PERFORMANCE_CID > 0)
		callfunc "F_BG_RM_EnsureForcedPerformanceMode";

	// 최초 티에라는 모집 확정 이후에도 참가자가 이탈하지 않았을 때만 시작합니다.
	if (.@mode == 1 && !callfunc("F_BG_RM_ValidateInitialReady"))
		return 0;

	// 티에라: 기존 30초 내부 준비 타이머를 그대로 사용합니다.
	if (.@mode == 1) {
		if ($@TierraBG1)
			return 0;

		$@TierraBG1_id1 = bg_create("bat_a01",50,374,"티에라 진행#bat_a01::OnGuillaumeQuit","");
		$@TierraBG1_id2 = bg_create("bat_a01",42,16,"티에라 진행#bat_a01::OnCroixQuit","");
		if (!$@TierraBG1_id1 || !$@TierraBG1_id2) {
			callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
			return 0;
		}

		for (.@i = 0; .@i < $@BG_RM_G_COUNT; .@i++) {
			if (!bg_join($@TierraBG1_id1,"bat_a01",50,374,$@BG_RM_G_CID[.@i])) {
				callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
				return 0;
			}
		}
		for (.@i = 0; .@i < $@BG_RM_C_COUNT; .@i++) {
			if (!bg_join($@TierraBG1_id2,"bat_a01",42,16,$@BG_RM_C_CID[.@i])) {
				callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
				return 0;
			}
		}

		callfunc "F_BG_RM_PrepareTeamMembers", 1, $@TierraBG1_id1;
		callfunc "F_BG_RM_PrepareTeamMembers", 2, $@TierraBG1_id2;
		$@TierraBG1 = 1;
		$@TierraBG1_Victory = 0;

		announce "[전장 안내] 랜덤 매칭 티에라 협곡이 기욤 " + $@BG_RM_G_COUNT + "명 vs 크루아 " + $@BG_RM_C_COUNT + "명으로 내부 준비를 시작합니다.", bc_all, "0x33CCFF";
		stopnpctimer "티에라 진행#bat_a01";
		initnpctimer "티에라 진행#bat_a01";

		// bg_join 직후에는 클라이언트가 아직 맵 로딩 중일 수 있으므로
		// 팀 구성 노티는 별도 타이머에서 5초 뒤 출력합니다.
		donpcevent "NEED_BG_RM_TierraTeamNotice::OnStart";
		debugmes "[BG RM] Tierra team notice scheduled delay=5000";
		return 1;
	}

	// 플라비우스: 오브젝트와 카운트다운은 30초 후 기존 엔진이 시작합니다.
	if (.@mode == 2) {
		if ($@FlaviusBG1)
			return 0;

		$@FlaviusBG1_id1 = bg_create("bat_b01",10,290,"start#bat_b01::OnGuillaumeQuit","");
		$@FlaviusBG1_id2 = bg_create("bat_b01",390,10,"start#bat_b01::OnCroixQuit","");
		if (!$@FlaviusBG1_id1 || !$@FlaviusBG1_id2) {
			callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
			return 0;
		}

		for (.@i = 0; .@i < $@BG_RM_G_COUNT; .@i++) {
			if (!bg_join($@FlaviusBG1_id1,"bat_b01",10,290,$@BG_RM_G_CID[.@i])) {
				callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
				return 0;
			}
		}
		for (.@i = 0; .@i < $@BG_RM_C_COUNT; .@i++) {
			if (!bg_join($@FlaviusBG1_id2,"bat_b01",390,10,$@BG_RM_C_CID[.@i])) {
				callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
				return 0;
			}
		}

		callfunc "F_BG_RM_PrepareTeamMembers", 1, $@FlaviusBG1_id1;
		callfunc "F_BG_RM_PrepareTeamMembers", 2, $@FlaviusBG1_id2;
		$@FlaviusBG1 = 1;
		$@FlaviusBG1_Victory = 0;
		$@Croix_ScoreBG1 = 0;
		$@Guill_ScoreBG1 = 0;
		bg_updatescore "bat_b01",0,0;

		announce "[전장 안내] 랜덤 매칭 플라비우스가 기욤 " + $@BG_RM_G_COUNT + "명 vs 크루아 " + $@BG_RM_C_COUNT + "명으로 내부 준비를 시작합니다.", bc_all;
		stopnpctimer "start#bat_b01";
		initnpctimer "start#bat_b01";
		return 1;
	}

	// KVM
	if (.@mode == 3) {
		if ($@KvM01BG)
			return 0;

		$@KvM01BG_id1 = bg_create("bat_c01",52,129,"KvM01_BG::OnGuillaumeQuit","KvM01_BG::OnGuillaumeDie");
		$@KvM01BG_id2 = bg_create("bat_c01",147,55,"KvM01_BG::OnCroixQuit","KvM01_BG::OnCroixDie");
		if (!$@KvM01BG_id1 || !$@KvM01BG_id2) {
			callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
			return 0;
		}

		for (.@i = 0; .@i < $@BG_RM_G_COUNT; .@i++) {
			if (!bg_join($@KvM01BG_id1,"bat_c01",52,129,$@BG_RM_G_CID[.@i])) {
				callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
				return 0;
			}
		}
		for (.@i = 0; .@i < $@BG_RM_C_COUNT; .@i++) {
			if (!bg_join($@KvM01BG_id2,"bat_c01",147,55,$@BG_RM_C_CID[.@i])) {
				callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
				return 0;
			}
		}

		callfunc "F_BG_RM_PrepareTeamMembers", 1, $@KvM01BG_id1;
		callfunc "F_BG_RM_PrepareTeamMembers", 2, $@KvM01BG_id2;
		$@KvM01BG = 1;
		$@KvM01BG_Victory = 0;
		bg_updatescore "bat_c01",0,0;
		bg_warp $@KvM01BG_id1,"bat_c01",53,128;
		bg_warp $@KvM01BG_id2,"bat_c01",146,55;

		announce "[전장 안내] 랜덤 매칭 KVM이 기욤 " + $@BG_RM_G_COUNT + "명 vs 크루아 " + $@BG_RM_C_COUNT + "명으로 내부 준비를 시작합니다.", bc_all;
		donpcevent "KvM01_BG::OnStart";
		return 1;
	}

	return 0;
}

// KVM 보상 수령 또는 자동 이동으로 모든 온라인 참가자가 bat_c01을 벗어났는지 확인합니다.
// 로그아웃한 참가자는 대기 대상으로 잡지 않습니다.
function	script	F_BG_RM_HasOnlineMemberInMap	{
	.@target_map$ = getarg(0);

	for (.@team = 1; .@team <= 2; .@team++) {
		.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;

		for (.@i = 0; .@i < .@count; .@i++) {
			.@aid = (.@team == 1) ? $@BG_RM_G_AID[.@i] : $@BG_RM_C_AID[.@i];
			.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];
			.@name$ = (.@team == 1) ? $@BG_RM_G_NAME$[.@i] : $@BG_RM_C_NAME$[.@i];

			if (!isloggedin(.@aid,.@cid))
				continue;

			if (getmapxy(.@map$,.@x,.@y,BL_PC,.@name$) != 0)
				continue;

			if (.@map$ == .@target_map$)
				return 1;
		}
	}

	return 0;
}

function	script	F_BG_RM_Finish	{
	.@reason$ = getarg(0, "티에라 → 플라비우스 → KVM 랜덤 매칭 사이클이 종료되었습니다.");

	if (!$@BG_RM_CYCLE_ACTIVE && !$@BG_RM_ACTIVE)
		return;

	stopnpctimer "NEED_BG_RM_Monitor";

	callfunc "F_BG_RM_DestroyParties";
	callfunc "F_BG_RM_RestoreCounts";
	callfunc "F_BG_RM_ClearLeaders";
	callfunc "F_BG_RM_RestoreForcedPerformanceMode";

	$@BG_RM_ACTIVE = 0;
	$@BG_RM_CYCLE_ACTIVE = 0;
	$@BG_RM_CYCLE_STAGE = 0;
	$@BG_RM_STATE = 0;
	$@BG_RM_MODE = 0;
	$@BG_RM_END_TICK = 0;
	$@BG_RM_KVM_PARTY_DESTROYED = 0;
	$@BG_RM_STAGE_WAIT_TICK = 0;

	callfunc "F_BG_RM_SetWaitingRooms", 1;

}

		return;

		$@BG_RM_MODE = 2;
		$@BG_RM_STATE = 4;

		mapannounce "bat_room", "[전장 랜덤 매칭] 기욤/크루아 파티장은 전장 관리인에서 입장 준비를 완료해 주세요.", bc_map, "0x99FF99";
		return;

	if ($@BG_RM_CYCLE_STAGE == 2) {

		$@BG_RM_STAGE_WAIT_TICK = gettimetick(2);

		mapannounce "bat_room", "[전장 랜덤 매칭] 기욤/크루아 파티장은 전장 관리인에서 입장 준비를 완료해 주세요.", bc_map, "0x99FF99";

	return;
}

		end;

		end;

		.@name$ = convertpcinfo(.@aid,CPC_NAME);

		if (.@aid <= 0 || .@cid <= 0 || .@name$ == "")
			continue;
		if (readparam(BaseLevel,.@cid) < .MinJoinLevel)
			continue;

			continue;

		dispbottom "[전장] BaseLevel " + .MinJoinLevel + " 이상 참가 가능 인원이 " + .MinPlayers + "명 미만입니다. (현재 " + .@eligible + "명)";

	BG_RM_LAST_CANCEL_REASON = 0;
	BG_RM_LAST_CANCEL_TOKEN = 0;
	BG_RM_LAST_CANCEL_TICK = 0;

	callfunc "F_BG_RM_SetWaitingRooms", 0;
	callfunc "F_BG_RM_SetManager", 0;

	end;

OnCommandJoin:
	if (strcharinfo(3) != "bat_room") {
		dispbottom "[전장] @전장참가 명령어는 전장 대기존에서만 사용할 수 있습니다.";
		end;
	}

	if ($@BG_RM_STATE != 1) {
		dispbottom "[전장] 현재 참가자를 모집 중인 랜덤 매칭이 없습니다.";
		end;

		end;

		end;
	}

	if (getcharid(1) || getcharid(4)) {
		dispbottom "[전장] 파티 또는 전장 팀에 가입된 상태에서는 참가할 수 없습니다.";
		end;
	}

	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		if ($@BG_RM_AID[.@i] == getcharid(3) && $@BG_RM_CID[.@i] == getcharid(0)) {
			.@remain = $@BG_RM_END_TICK - gettimetick(2);
			if (.@remain < 0)
				.@remain = 0;
			announce "[전장] 현재 랜덤 매칭 참가 신청이 유지 중입니다.", bc_self, "0x66FF66";

		dispbottom "[전장] 랜덤 매칭 참가 정원 " + .MaxPlayers + "명이 모두 찼습니다.";
		end;

	$@BG_RM_CID[.@slot] = getcharid(0);
	$@BG_RM_CLASS[.@slot] = Class;
	$@BG_RM_NAME$[.@slot] = strcharinfo(0);
	$@BG_RM_COUNT++;

		dispbottom "[전장] @전장취소 명령어는 전장 대기존에서만 사용할 수 있습니다.";

		}

		mes "현재 유지 중인 랜덤 매칭 파티와";
		mes "남은 전장 사이클을 모두 종료하시겠습니까?";
		next;
		if (select("사이클 종료:취소") == 2)

		end;
	}

	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		if ($@BG_RM_AID[.@i] == getcharid(3) && $@BG_RM_CID[.@i] == getcharid(0)) {
			.@slot = .@i;
			break;
		}
	}

	if (.@slot < 0) {
		dispbottom "[전장] 랜덤 매칭에 참가 신청한 상태가 아닙니다.";
		end;
	}

	callfunc "F_BG_RM_RemoveRecruitSlot", .@slot;

	BG_RM_LAST_CANCEL_REASON = 5;
	BG_RM_LAST_CANCEL_TOKEN = $@BG_RM_TOKEN;
	BG_RM_LAST_CANCEL_TICK = gettimetick(2);

	announce "[전장] 랜덤 매칭 참가 신청을 취소했습니다.", bc_self, "0xFFAA55";
	dispbottom "[전장] 다시 참가하려면 모집 종료 전 @전장참가를 입력해 주세요.";

	mapannounce "bat_room", "[전장 랜덤 매칭] " + strcharinfo(0) + "님이 참가를 취소했습니다. 현재 " + $@BG_RM_COUNT + "명", bc_map, "0xFFCC99";

	if ($@BG_RM_COUNT <= 0) {
		callfunc "F_BG_RM_CancelRecruit", "모든 참가자가 신청을 취소하여 랜덤 매칭 모집이 종료되었습니다.";
		end;
	}
	end;

OnCommandStatus:
	if (strcharinfo(3) != "bat_room") {
		dispbottom "[전장] @전장현황 명령어는 전장 대기존에서만 사용할 수 있습니다.";
		end;
	}

	if ($@BG_RM_STATE == 1) {
		.@remain = $@BG_RM_END_TICK - gettimetick(2);
		if (.@remain < 0)
			.@remain = 0;

		.@slot = callfunc("F_BG_RM_FindRecruitSlot", getcharid(3), getcharid(0));
		if (.@slot >= 0) {
			announce "[전장] 내 상태: 랜덤 매칭 참가 중", bc_self, "0x66FF66";
			dispbottom "[전장] 참가 유지 중 / 현재 " + $@BG_RM_COUNT + "명 / 남은 시간 " + .@remain + "초";
			dispbottom "[전장] bat_room 이탈·로그아웃·파티 가입 시 자동 취소됩니다.";
		}
		else if (BG_RM_LAST_CANCEL_TOKEN == $@BG_RM_TOKEN && BG_RM_LAST_CANCEL_REASON > 0) {
			announce "[전장] 내 상태: 참가 신청 취소됨", bc_self, "0xFF5555";
			dispbottom "[전장] 취소 사유: " + callfunc("F_BG_RM_CancelReason$", BG_RM_LAST_CANCEL_REASON);
			dispbottom "[전장] 현재 모집에 다시 참가하려면 @전장참가를 입력해 주세요.";
		}
		else {
			announce "[전장] 내 상태: 현재 모집 미참가", bc_self, "0xFFCC66";
			if (BaseLevel < .MinJoinLevel)
				dispbottom "[전장] 참가 조건: BaseLevel " + .MinJoinLevel + " 이상 / 현재 " + BaseLevel;
			else
				dispbottom "[전장] 참가하려면 @전장참가를 입력해 주세요.";
		}

		dispbottom "[전장] 모집 현황: " + callfunc("F_BG_RM_ModeName$",$@BG_RM_MODE)
			+ " / 참가 " + $@BG_RM_COUNT + "명 / 남은 시간 " + .@remain + "초";
		end;
	}

	if ($@BG_RM_CYCLE_ACTIVE) {
		if ($@BG_RM_STATE == 4)
			dispbottom "[전장] 사이클 유지 중 / 다음 전장: " + callfunc("F_BG_RM_ModeName$",$@BG_RM_MODE) + " / 양 팀 파티장 입장 준비 대기";
		else
			dispbottom "[전장] 진행 중: " + callfunc("F_BG_RM_ModeName$",$@BG_RM_MODE) + " / 기욤 " + $@BG_RM_G_COUNT + "명 vs 크루아 " + $@BG_RM_C_COUNT + "명 / 단계 " + $@BG_RM_CYCLE_STAGE + "/3";
		end;
	}

	dispbottom "[전장] 현재 랜덤 매칭 없음 / 시작 전장: 티에라 협곡 / 시작: @전장매칭";
	end;

OnTimer60000:
	if ($@BG_RM_STATE == 1) {
		mapannounce "bat_room", "[전장 랜덤 매칭] 모집 종료까지 4분. 현재 참가 " + $@BG_RM_COUNT + "명", bc_map, "0x66CCFF";
		callfunc "F_BG_RM_NotifyApplicants", "4분";
	}
	end;

OnTimer120000:
	if ($@BG_RM_STATE == 1) {
		mapannounce "bat_room", "[전장 랜덤 매칭] 모집 종료까지 3분. 현재 참가 " + $@BG_RM_COUNT + "명", bc_map, "0x66CCFF";
		callfunc "F_BG_RM_NotifyApplicants", "3분";
	}
	end;

OnTimer180000:
	if ($@BG_RM_STATE == 1) {
		mapannounce "bat_room", "[전장 랜덤 매칭] 모집 종료까지 2분. 현재 참가 " + $@BG_RM_COUNT + "명", bc_map, "0x66CCFF";
		callfunc "F_BG_RM_NotifyApplicants", "2분";
	}
	end;

OnTimer240000:
	if ($@BG_RM_STATE == 1) {
		mapannounce "bat_room", "[전장 랜덤 매칭] 모집 종료까지 1분. 현재 참가 " + $@BG_RM_COUNT + "명", bc_map, "0x66CCFF";
		callfunc "F_BG_RM_NotifyApplicants", "1분";
	}
	end;

OnTimer270000:
	if ($@BG_RM_STATE == 1)
		mapannounce "bat_room", "[전장 랜덤 매칭] 모집 종료까지 30초. 현재 참가 " + $@BG_RM_COUNT + "명", bc_map, "0x66CCFF";
	end;

OnTimer285000:
	if ($@BG_RM_STATE == 1)
		mapannounce "bat_room", "[전장 랜덤 매칭] 모집 종료까지 15초. 현재 참가 " + $@BG_RM_COUNT + "명", bc_map, "0x66CCFF";
	end;

OnTimer290000:
	if ($@BG_RM_STATE == 1)
		mapannounce "bat_room", "[전장 랜덤 매칭] 모집 종료까지 10초. 현재 참가 " + $@BG_RM_COUNT + "명", bc_map, "0x66CCFF";
	end;

OnTimer295000:
	if ($@BG_RM_STATE == 1)
		mapannounce "bat_room", "[전장 랜덤 매칭] 모집 종료까지 5초 남았습니다. 현재 참가 " + $@BG_RM_COUNT + "명", bc_map, "0xFFCC66";
	end;

OnTimer300000:
	stopnpctimer;
	if ($@BG_RM_STATE != 1)
		end;

	// 같은 NPC의 OnFinalize를 다시 이벤트로 예약하지 않고
	// 현재 타이머 실행 흐름에서 바로 마감 처리합니다.
	goto OnFinalize;

OnFinalize:
	if ($@BG_RM_STATE != 1)
		end;

	stopnpctimer "NEED_BG_RM_RecruitMonitor";
	$@BG_RM_MISSING_CURSOR = 0;
	$@BG_RM_STATE = 2;
	.@finalize_token = $@BG_RM_TOKEN;
	$@BG_RM_FINALIZE_WATCH_TOKEN = .@finalize_token;
	donpcevent "NEED_BG_RM_FinalizeWatch::OnStart";

	mapannounce "bat_room",
		"[전장 랜덤 매칭] 모집이 마감되었습니다. 참가자 확인 및 팀 편성을 시작합니다.",
		bc_map, "0xFFCC66";
	debugmes "[BG RM] finalize begin count=" + $@BG_RM_COUNT;

	// 모집 중 이미 열려 있던 관리인 대화 등으로 수동 준비 상태가 생겼는지 재확인
	if ($@TierraBG1 || $@FlaviusBG1 || $@KvM01BG || callfunc("F_BG_RM_ManualReadyActive") || $BG_GUILLAUME_PARTY_ID || $BG_CROIX_PARTY_ID) {
		callfunc "F_BG_RM_CancelRecruit", "수동 파티장 전장 준비와 충돌하여 매칭이 취소되었습니다.";
		end;
	}

	.@count = 0;
	.@max_level_count = 0;
	.@minstrel_total_count = 0;
	.@single_minstrel_aid = 0;
	.@single_minstrel_cid = 0;

	// 접속, 위치, 파티 상태를 최종 재검증
	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		.@aid = $@BG_RM_AID[.@i];
		.@cid = $@BG_RM_CID[.@i];
		.@name$ = $@BG_RM_NAME$[.@i];

		if (!isloggedin(.@aid,.@cid))
			continue;
		if (getmapxy(.@map$,.@x,.@y,BL_PC,.@name$) != 0)
			continue;
		if (.@map$ != "bat_room") {
			message .@name$, "[전장] 전장 대기존을 이탈하여 랜덤 매칭에서 제외되었습니다.";
			continue;
		}
		if (getcharid(1,.@name$) || getcharid(4,.@name$)) {
			message .@name$, "[전장] 파티 또는 전장 팀에 가입하여 랜덤 매칭에서 제외되었습니다.";
			continue;
		}

		.@base_level = readparam(BaseLevel,.@cid);
		if (.@base_level < .MinJoinLevel) {
			message .@name$, "[전장] BaseLevel " + .MinJoinLevel + " 미만이라 랜덤 매칭 최종 참가자에서 제외되었습니다.";
			continue;
		}

		.@aid_list[.@count] = .@aid;
		.@cid_list[.@count] = .@cid;
		.@name_list$[.@count] = .@name$;
		.@class_list[.@count] = readparam(Class,.@cid);
		.@level_list[.@count] = .@base_level;

		if (.@level_list[.@count] == .MaxBaseLevel)
			.@max_level_count++;

		// "민스트럴만" 적용: 바드/클로운은 자동 현행 모드 대상에서 제외합니다.
		if (
			.@class_list[.@count] == JOB_MINSTREL ||
			.@class_list[.@count] == JOB_MINSTREL_T
		) {
			.@minstrel_total_count++;
			.@single_minstrel_aid = .@aid;
			.@single_minstrel_cid = .@cid;
		}

		.@count++;
	}

	if (.@count < .MinPlayers) {
		callfunc "F_BG_RM_CancelRecruit", "최종 참가자가 " + .MinPlayers + "명 미만이어서 매칭이 취소되었습니다. (" + .@count + "명)";
		end;
	}

	// 양 팀 파티장은 반드시 현재 만렙(175)이어야 하므로 최소 2명이 필요합니다.
	if (.@max_level_count < 2) {
		callfunc "F_BG_RM_CancelRecruit",
			"양 팀 파티장 선정을 위해 175레벨 참가자가 최소 2명 필요합니다. (현재 " + .@max_level_count + "명)";
		end;
	}

	// 전체 참가자 중 실제 민스트럴이 정확히 1명인 경우에만
	// 해당 캐릭터의 @모드전환 저장값을 현행(Modern=1)으로 강제합니다.
	if (.@minstrel_total_count == 1) {
		$@BG_RM_FORCED_PERFORMANCE_AID = .@single_minstrel_aid;
		$@BG_RM_FORCED_PERFORMANCE_CID = .@single_minstrel_cid;

		if (!callfunc("F_BG_RM_EnsureForcedPerformanceMode")) {
			callfunc "F_BG_RM_CancelRecruit", "단독 민스트럴의 현행 모드 적용에 실패하여 매칭이 취소되었습니다.";
			end;
		}

		debugmes "[BG RM] single minstrel modern mode forced"
			+ " cid=" + .@single_minstrel_cid;
	}

	debugmes "[BG RM] finalize validated eligible=" + .@count
		+ " max=" + .MaxPlayers
		+ " party_max=" + .MaxPartyMembers;

	// 참가자가 많을 경우 아래의 다중 for/while 배정 로직이
	// rAthena의 1회 실행 GOTO/command 안전 한도에 먼저 도달할 수 있습니다.
	// 모든 반복문의 상한은 참가 인원(.MaxPlayers)으로 제한되어 있으므로
	// 팀 편성 구간에서만 loop guard를 일시 해제합니다.
	freeloop 1;
	debugmes "[BG RM] matching loop guard disabled count=" + .@count;

	// 전체 참가 순서 셔플
	for (.@i = .@count - 1; .@i > 0; .@i--) {
		.@j = rand(.@i + 1);

		.@tmp = .@aid_list[.@i];
		.@aid_list[.@i] = .@aid_list[.@j];
		.@aid_list[.@j] = .@tmp;

		.@tmp = .@cid_list[.@i];
		.@cid_list[.@i] = .@cid_list[.@j];
		.@cid_list[.@j] = .@tmp;

		.@tmp$ = .@name_list$[.@i];
		.@name_list$[.@i] = .@name_list$[.@j];
		.@name_list$[.@j] = .@tmp$;

		.@tmp = .@class_list[.@i];
		.@class_list[.@i] = .@class_list[.@j];
		.@class_list[.@j] = .@tmp;

		.@tmp = .@level_list[.@i];
		.@level_list[.@i] = .@level_list[.@j];
		.@level_list[.@j] = .@tmp;
	}

	// 홀수 인원일 때 한 명이 더 많은 팀은 무작위
	if ((.@count % 2) == 0) {
		.@cap_g = .@count / 2;
		.@cap_c = .@count / 2;
	}
	else if (rand(2) == 0) {
		.@cap_g = (.@count + 1) / 2;
		.@cap_c = .@count / 2;
	}
	else {
		.@cap_g = .@count / 2;
		.@cap_c = (.@count + 1) / 2;
	}

	cleararray .@used[0], 0, .@count;
	cleararray .@is_low[0], 0, .@count;
	.@count_g = 0;
	.@count_c = 0;
	.@single_count = 0;
	.@low_count = 0;

	// 175 미만 참가자는 일반 직업 배정에서 제외하고 마지막 단계에 따로 배정합니다.
	for (.@i = 0; .@i < .@count; .@i++) {
		if (.@level_list[.@i] < .MaxBaseLevel) {
			.@is_low[.@i] = 1;
			.@low_index[.@low_count] = .@i;
			.@low_count++;
		}
	}

	// 낮은 레벨 인원 자체가 양 팀에 최대한 균등하게 들어가도록
	// 최종 팀 정원에서 저레벨 전용 슬롯을 먼저 예약합니다.
	.@low_cap_g = .@low_count / 2;
	.@low_cap_c = .@low_count / 2;

	if ((.@low_count % 2) == 1) {
		if (.@cap_g > .@cap_c)
			.@low_cap_g++;
		else if (.@cap_c > .@cap_g)
			.@low_cap_c++;
		else if (rand(2) == 0)
			.@low_cap_g++;
		else
			.@low_cap_c++;
	}

	.@high_cap_g = .@cap_g - .@low_cap_g;
	.@high_cap_c = .@cap_c - .@low_cap_c;

	if (.@high_cap_g < 1 || .@high_cap_c < 1) {
		callfunc "F_BG_RM_CancelRecruit",
			"175레벨 파티장을 양 팀에 배치할 수 없는 인원 구성이라 매칭이 취소되었습니다.";
		end;
	}

	debugmes "[BG RM] level slots"
		+ " max175=" + .@max_level_count
		+ " low=" + .@low_count
		+ " high_cap_G=" + .@high_cap_g
		+ " high_cap_C=" + .@high_cap_c
		+ " low_cap_G=" + .@low_cap_g
		+ " low_cap_C=" + .@low_cap_c;

	//========================================================
	// 우선 분리 규칙 v2
	//
	// 적용 쌍:
	//  rule 0: 민스트럴(6) ↔ 원더러(7)
	//  rule 1: 로열가드(2) ↔ 소서러(4)
	//  rule 2: 아크비숍(12) ↔ 제네릭(9)
	//  rule 3: 소울링커(14) ↔ 워록(3)
	//
	// 핵심 원칙:
	// 1) 각 직업군 자체의 인원이 2명 이상이면 먼저 양 팀에 최대한 균등 분배
	//    - 2명 -> 1 / 1
	//    - 3명 -> 2 / 1
	//    - 4명 -> 2 / 2
	// 2) 짝을 이루는 두 직업군이 모두 홀수라면
	//    각 직업군의 "한 명 더 많은 팀"은 반드시 서로 반대로 배치
	// 3) 175 미만은 실제 팀 배열에 즉시 넣지 않고 팀만 예약한 뒤
	//    기존 정책대로 저레벨 마지막 단계에서 실제 배치
	//
	// 예: 아크비숍 2 + 제네릭 1
	//     -> 아크비숍 1/1 선분배
	//     -> 제네릭 1명은 정원에 맞는 한쪽 팀
	//
	// 예: 아크비숍 3 + 제네릭 1
	//     -> 아크비숍의 추가 1명과 제네릭 1명은 서로 반대 팀
	//========================================================
	cleararray .@forced_team[0], 0, .@count;
	.@forced_high_g = 0;
	.@forced_high_c = 0;
	.@forced_low_g = 0;
	.@forced_low_c = 0;

	.@split_rule_count = 4;

	.@split_family_a[0] = 6;
	.@split_family_b[0] = 7;

	.@split_family_a[1] = 2;
	.@split_family_b[1] = 4;

	.@split_family_a[2] = 12;
	.@split_family_b[2] = 9;

	.@split_family_a[3] = 14;
	.@split_family_b[3] = 3;

	// 수라 홀수 + 제네릭 단독 배치 규칙 계산용
	.@total_sura = 0;
	.@total_genetic = 0;
	.@single_genetic_index = -1;

	for (.@i = 0; .@i < .@count; .@i++) {
		.@family = callfunc("F_BG_RM_NoticeJobFamily", .@class_list[.@i]);

		if (.@family == 13)
			.@total_sura++;

		if (.@family == 9) {
			.@total_genetic++;
			.@single_genetic_index = .@i;
		}
	}

	for (.@rule = 0; .@rule < .@split_rule_count; .@rule++) {
		.@family_a = .@split_family_a[.@rule];
		.@family_b = .@split_family_b[.@rule];
		.@count_a = 0;
		.@count_b = 0;

		// 해당 우선 분리 직업군의 전체 참가자 인덱스를 모읍니다.
		for (.@i = 0; .@i < .@count; .@i++) {
			.@family = callfunc("F_BG_RM_NoticeJobFamily", .@class_list[.@i]);

			if (.@family == .@family_a) {
				.@rule_a_index[.@count_a] = .@i;
				.@count_a++;
			}
			else if (.@family == .@family_b) {
				.@rule_b_index[.@count_b] = .@i;
				.@count_b++;
			}
		}

		if (.@count_a <= 0 && .@count_b <= 0)
			continue;

		// 같은 직업군 안에서 특정 캐릭터가 항상 같은 팀을 받지 않도록 셔플합니다.
		for (.@i = .@count_a - 1; .@i > 0; .@i--) {
			.@j = rand(.@i + 1);
			.@tmp = .@rule_a_index[.@i];
			.@rule_a_index[.@i] = .@rule_a_index[.@j];
			.@rule_a_index[.@j] = .@tmp;
		}

		for (.@i = .@count_b - 1; .@i > 0; .@i--) {
			.@j = rand(.@i + 1);
			.@tmp = .@rule_b_index[.@i];
			.@rule_b_index[.@i] = .@rule_b_index[.@j];
			.@rule_b_index[.@j] = .@tmp;
		}

		// A/B 각각 어느 팀부터 교차 배정할지 4가지 조합을 검사합니다.
		// 두 직업군 모두 홀수면 추가 1명이 같은 팀에 몰리지 않도록
		// 시작 팀이 반드시 서로 반대인 조합만 허용합니다.
		.@selected = 0;
		.@candidate_offset = rand(4);

		for (.@try = 0; .@try < 4; .@try++) {
			.@candidate = (.@candidate_offset + .@try) % 4;
			.@start_a = .@candidate % 2;
			.@start_b = (.@candidate / 2) % 2;

			if (
				(.@count_a % 2) == 1 &&
				(.@count_b % 2) == 1 &&
				.@start_a == .@start_b
			)
				continue;

			.@need_hg = 0;
			.@need_hc = 0;
			.@need_lg = 0;
			.@need_lc = 0;

			for (.@i = 0; .@i < .@count_a; .@i++) {
				.@index = .@rule_a_index[.@i];
				.@team = (.@start_a + .@i) % 2;

				if (.@is_low[.@index]) {
					if (.@team == 0) .@need_lg++;
					else .@need_lc++;
				}
				else {
					if (.@team == 0) .@need_hg++;
					else .@need_hc++;
				}
			}

			for (.@i = 0; .@i < .@count_b; .@i++) {
				.@index = .@rule_b_index[.@i];
				.@team = (.@start_b + .@i) % 2;

				if (.@is_low[.@index]) {
					if (.@team == 0) .@need_lg++;
					else .@need_lc++;
				}
				else {
					if (.@team == 0) .@need_hg++;
					else .@need_hc++;
				}
			}

			if (
				.@forced_high_g + .@need_hg > .@high_cap_g ||
				.@forced_high_c + .@need_hc > .@high_cap_c ||
				.@forced_low_g + .@need_lg > .@low_cap_g ||
				.@forced_low_c + .@need_lc > .@low_cap_c
			)
				continue;

			.@selected = 1;
			break;
		}

		if (!.@selected) {
			callfunc "F_BG_RM_CancelRecruit",
				"직업 우선 분리 규칙을 만족할 수 없는 팀 정원이라 매칭이 취소되었습니다. (규칙 " + .@rule + ")";
			end;
		}

		// 선택된 방향대로 해당 직업군 전원을 교차 예약합니다.
		for (.@i = 0; .@i < .@count_a; .@i++) {
			.@index = .@rule_a_index[.@i];
			.@team = (.@start_a + .@i) % 2;
			.@forced_team[.@index] = .@team + 1;

			if (.@is_low[.@index]) {
				if (.@team == 0) .@forced_low_g++;
				else .@forced_low_c++;
			}
			else {
				if (.@team == 0) .@forced_high_g++;
				else .@forced_high_c++;
			}
		}

		for (.@i = 0; .@i < .@count_b; .@i++) {
			.@index = .@rule_b_index[.@i];
			.@team = (.@start_b + .@i) % 2;
			.@forced_team[.@index] = .@team + 1;

			if (.@is_low[.@index]) {
				if (.@team == 0) .@forced_low_g++;
				else .@forced_low_c++;
			}
			else {
				if (.@team == 0) .@forced_high_g++;
				else .@forced_high_c++;
			}
		}

		debugmes "[BG RM] priority family split"
			+ " rule=" + .@rule
			+ " familyA=" + .@family_a + " countA=" + .@count_a
			+ " familyB=" + .@family_b + " countB=" + .@count_b
			+ " startA=" + .@start_a
			+ " startB=" + .@start_b;
	}

	// -------------------------------------------------------
	// 수라 홀수 + 제네릭 1명 특수 규칙
	//
	// 제네릭이 정확히 한 명이고 수라 전체 인원이 홀수라면,
	// 수라의 "한 명 더 많은 팀"을 제네릭이 없는 반대 팀으로 지정합니다.
	//
	// 실제 수라 배정은 기존 동일직업 분산 단계에서 처리하며,
	// 여기서는 어느 팀이 수라 추가 1명을 받을지만 저장합니다.
	// -------------------------------------------------------
	.@sura_extra_team = -1;

	if (
		.@total_genetic == 1 &&
		(.@total_sura % 2) == 1 &&
		.@single_genetic_index >= 0 &&
		.@forced_team[.@single_genetic_index] > 0
	) {
		// forced_team: 1=G, 2=C
		// sura_extra_team: 0=G, 1=C
		if (.@forced_team[.@single_genetic_index] == 1)
			.@sura_extra_team = 1;
		else
			.@sura_extra_team = 0;

		debugmes "[BG RM] sura odd preference"
			+ " genetic_cid=" + .@cid_list[.@single_genetic_index]
			+ " genetic_team=" + ((.@forced_team[.@single_genetic_index] == 1) ? "G" : "C")
			+ " sura_extra_team=" + ((.@sura_extra_team == 0) ? "G" : "C")
			+ " sura_count=" + .@total_sura;
	}

	// 175레벨 우선 분리 대상은 지금 실제 팀에 넣습니다.
	// 175 미만은 forced_team 예약만 유지하고 마지막 단계에서 처리합니다.
	for (.@i = 0; .@i < .@count; .@i++) {
		if (!.@forced_team[.@i] || .@is_low[.@i])
			continue;

		if (.@forced_team[.@i] == 1) {
			.@aid_g[.@count_g] = .@aid_list[.@i];
			.@cid_g[.@count_g] = .@cid_list[.@i];
			.@name_g$[.@count_g] = .@name_list$[.@i];
			.@count_g++;
		}
		else {
			.@aid_c[.@count_c] = .@aid_list[.@i];
			.@cid_c[.@count_c] = .@cid_list[.@i];
			.@name_c$[.@count_c] = .@name_list$[.@i];
			.@count_c++;
		}

		.@used[.@i] = 1;
	}

	debugmes "[BG RM] priority high applied"
		+ " G=" + .@count_g
		+ " C=" + .@count_c
		+ " forced_low_G=" + .@forced_low_g
		+ " forced_low_C=" + .@forced_low_c;

	// 1단계: 동일 직업이 2명 이상인 그룹을 기욤/크루아에 교차 배정
	// 1명뿐인 직업은 즉시 배정하지 않고 단독 직업 목록에 보관합니다.
	for (.@i = 0; .@i < .@count; .@i++) {
		if (.@used[.@i] || .@is_low[.@i])
			continue;

		.@group_count = 0;
		.@target_class = .@class_list[.@i];

		for (.@j = 0; .@j < .@count; .@j++) {
			if (!.@used[.@j] && !.@is_low[.@j] && .@class_list[.@j] == .@target_class) {
				.@group_index[.@group_count] = .@j;
				.@used[.@j] = 1;
				.@group_count++;
			}
		}

		// 동일 직업 그룹 내부 순서 무작위화
		for (.@j = .@group_count - 1; .@j > 0; .@j--) {
			.@k = rand(.@j + 1);
			.@tmp = .@group_index[.@j];
			.@group_index[.@j] = .@group_index[.@k];
			.@group_index[.@k] = .@tmp;
		}

		// 한 명뿐인 직업은 2단계에서 서로 다른 팀으로 분산합니다.
		if (.@group_count == 1) {
			.@single_index[.@single_count] = .@group_index[0];
			.@single_count++;
			continue;
		}

		.@target_family = callfunc("F_BG_RM_NoticeJobFamily", .@target_class);

		// 수라가 홀수이고 제네릭이 한쪽에만 있을 때는
		// 수라의 추가 1명이 제네릭 반대 팀으로 가도록 시작 팀을 고정합니다.
		if (
			.@target_family == 13 &&
			(.@group_count % 2) == 1 &&
			.@sura_extra_team >= 0
		)
			.@start_team = .@sura_extra_team;
		else if (.@count_g < .@count_c)
			.@start_team = 0;
		else if (.@count_c < .@count_g)
			.@start_team = 1;
		else
			.@start_team = rand(2);

		for (.@j = 0; .@j < .@group_count; .@j++) {
			.@index = .@group_index[.@j];
			.@team = (.@start_team + .@j) % 2;

			if (.@team == 0 && .@count_g >= .@high_cap_g)
				.@team = 1;
			else if (.@team == 1 && .@count_c >= .@high_cap_c)
				.@team = 0;

			if (.@team == 0) {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
			}
			else {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
			}
		}
	}

	// 2단계: 1명뿐인 직업 목록을 무작위로 섞은 뒤 2명씩 양 팀에 한 명씩 배정
	for (.@i = .@single_count - 1; .@i > 0; .@i--) {
		.@j = rand(.@i + 1);
		.@tmp = .@single_index[.@i];
		.@single_index[.@i] = .@single_index[.@j];
		.@single_index[.@j] = .@tmp;
	}

	// 같은 팀부터 계속 시작하지 않도록 쌍마다 시작 팀을 교대합니다.
	.@pair_start_team = rand(2);

	for (.@i = 0; .@i < .@single_count; .@i += 2) {
		.@remain_g = .@high_cap_g - .@count_g;
		.@remain_c = .@high_cap_c - .@count_c;

		// 두 명이 남았고 양 팀 모두 자리가 있으면 반드시 한 명씩 나눕니다.
		if (.@i + 1 < .@single_count && .@remain_g > 0 && .@remain_c > 0) {
			if (.@remain_g > .@remain_c)
				.@first_team = 0;
			else if (.@remain_c > .@remain_g)
				.@first_team = 1;
			else
				.@first_team = .@pair_start_team;

			.@index = .@single_index[.@i];
			if (.@first_team == 0) {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
			}
			else {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
			}

			// 같은 쌍의 두 번째 단독 직업은 반대 팀으로 배정합니다.
			.@index = .@single_index[.@i + 1];
			if (.@first_team == 0) {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
			}
			else {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
			}

			.@pair_start_team = 1 - .@pair_start_team;
			continue;
		}

		// 한 명만 남았거나 한쪽 팀 정원이 찬 경우 남은 정원에 맞춰 배정합니다.
		for (.@j = .@i; .@j < .@single_count && .@j < .@i + 2; .@j++) {
			.@index = .@single_index[.@j];
			.@remain_g = .@high_cap_g - .@count_g;
			.@remain_c = .@high_cap_c - .@count_c;

			if (.@remain_g <= 0)
				.@team = 1;
			else if (.@remain_c <= 0)
				.@team = 0;
			else if (.@remain_g > .@remain_c)
				.@team = 0;
			else if (.@remain_c > .@remain_g)
				.@team = 1;
			else if (.@count_g < .@count_c)
				.@team = 0;
			else if (.@count_c < .@count_g)
				.@team = 1;
			else
				.@team = rand(2);

			if (.@team == 0) {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
			}
			else {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
			}
		}
	}

	// 175레벨 참가자 우선 배정이 예약된 정원만큼 정확히 끝났는지 확인합니다.
	if (.@count_g != .@high_cap_g || .@count_c != .@high_cap_c) {
		debugmes "[BG RM] high-level assignment invalid"
			+ " G=" + .@count_g + "/" + .@high_cap_g
			+ " C=" + .@count_c + "/" + .@high_cap_c;
		callfunc "F_BG_RM_CancelRecruit", "175레벨 우선 팀 배정 중 오류가 발생하여 매칭이 취소되었습니다.";
		end;
	}

	//========================================================
	// 마지막 단계: 175 미만 참가자 균형 배정
	//
	// 우선순위:
	// 1) 하드 분리 규칙으로 예약된 저레벨 팀 배치
	// 2) 동일 직업 저레벨을 가능한 한 서로 다른 팀에 분산
	// 3) 양 팀 저레벨 인원 수 균형
	// 4) 양 팀 저레벨 BaseLevel 합 균형
	//
	// 예:
	// 쉐도우체이서 170 / 쉐도우체이서 174가 있으면,
	// 양 팀 저레벨 슬롯이 남아 있는 한 1명씩 서로 다른 팀에 배정합니다.
	//========================================================
	cleararray .@low_assigned[0], 0, .@count;
	cleararray .@low_team[0], 0, .@count;
	cleararray .@low_group_done[0], 0, .@count;

	.@low_g = 0;
	.@low_c = 0;
	.@low_level_sum_g = 0;
	.@low_level_sum_c = 0;

	// -------------------------------------------------------
	// 1단계: 하드 분리 규칙으로 팀이 예약된 저레벨 먼저 확정
	// -------------------------------------------------------
	for (.@i = 0; .@i < .@low_count; .@i++) {
		.@index = .@low_index[.@i];

		if (!.@forced_team[.@index])
			continue;

		.@team = .@forced_team[.@index] - 1;
		.@remain_low_g = .@low_cap_g - .@low_g;
		.@remain_low_c = .@low_cap_c - .@low_c;

		if (
			(.@team == 0 && .@remain_low_g <= 0) ||
			(.@team == 1 && .@remain_low_c <= 0)
		) {
			callfunc "F_BG_RM_CancelRecruit",
				"저레벨 직업 우선 분리 슬롯이 부족하여 매칭이 취소되었습니다.";
			end;
		}

		if (.@team == 0) {
			.@aid_g[.@count_g] = .@aid_list[.@index];
			.@cid_g[.@count_g] = .@cid_list[.@index];
			.@name_g$[.@count_g] = .@name_list$[.@index];
			.@count_g++;
			.@low_g++;
			.@low_level_sum_g += .@level_list[.@index];
			.@low_team[.@index] = 1;
		}
		else {
			.@aid_c[.@count_c] = .@aid_list[.@index];
			.@cid_c[.@count_c] = .@cid_list[.@index];
			.@name_c$[.@count_c] = .@name_list$[.@index];
			.@count_c++;
			.@low_c++;
			.@low_level_sum_c += .@level_list[.@index];
			.@low_team[.@index] = 2;
		}

		.@low_assigned[.@index] = 1;
	}

	// -------------------------------------------------------
	// 2단계: 남은 저레벨은 동일 직업별 그룹으로 처리
	//
	// low_index 자체는 먼저 랜덤 셔플 후 레벨 내림차순 정렬합니다.
	// 같은 레벨끼리는 랜덤성이 유지되고, 높은 저레벨부터 그룹이 선택됩니다.
	// -------------------------------------------------------
	for (.@i = .@low_count - 1; .@i > 0; .@i--) {
		.@j = rand(.@i + 1);
		.@tmp = .@low_index[.@i];
		.@low_index[.@i] = .@low_index[.@j];
		.@low_index[.@j] = .@tmp;
	}

	for (.@i = 0; .@i < .@low_count - 1; .@i++) {
		for (.@j = 0; .@j < .@low_count - .@i - 1; .@j++) {
			if (
				.@level_list[.@low_index[.@j]] <
				.@level_list[.@low_index[.@j + 1]]
			) {
				.@tmp = .@low_index[.@j];
				.@low_index[.@j] = .@low_index[.@j + 1];
				.@low_index[.@j + 1] = .@tmp;
			}
		}
	}

	for (.@i = 0; .@i < .@low_count; .@i++) {
		.@first_index = .@low_index[.@i];

		if (.@low_assigned[.@first_index] || .@low_group_done[.@first_index])
			continue;

		.@target_class = .@class_list[.@first_index];
		.@group_count = 0;

		// 아직 미배정인 동일 직업 저레벨을 한 그룹으로 모읍니다.
		for (.@j = 0; .@j < .@low_count; .@j++) {
			.@index = .@low_index[.@j];

			if (
				.@low_assigned[.@index] ||
				.@low_group_done[.@index] ||
				.@class_list[.@index] != .@target_class
			)
				continue;

			.@low_group_index[.@group_count] = .@index;
			.@low_group_done[.@index] = 1;
			.@group_count++;
		}

		// 동일 직업 그룹 내부는 먼저 섞고 레벨 높은 순으로 정렬합니다.
		for (.@j = .@group_count - 1; .@j > 0; .@j--) {
			.@k = rand(.@j + 1);
			.@tmp = .@low_group_index[.@j];
			.@low_group_index[.@j] = .@low_group_index[.@k];
			.@low_group_index[.@k] = .@tmp;
		}

		for (.@j = 0; .@j < .@group_count - 1; .@j++) {
			for (.@k = 0; .@k < .@group_count - .@j - 1; .@k++) {
				if (
					.@level_list[.@low_group_index[.@k]] <
					.@level_list[.@low_group_index[.@k + 1]]
				) {
					.@tmp = .@low_group_index[.@k];
					.@low_group_index[.@k] = .@low_group_index[.@k + 1];
					.@low_group_index[.@k + 1] = .@tmp;
				}
			}
		}

		// 같은 직업의 하드 분리 예약 인원이 이미 팀에 들어간 경우까지 포함해서
		// 현재 해당 직업의 팀별 인원 수를 계산합니다.
		.@class_g = 0;
		.@class_c = 0;

		for (.@j = 0; .@j < .@low_count; .@j++) {
			.@index = .@low_index[.@j];

			if (
				.@class_list[.@index] != .@target_class ||
				!.@low_assigned[.@index]
			)
				continue;

			if (.@low_team[.@index] == 1)
				.@class_g++;
			else if (.@low_team[.@index] == 2)
				.@class_c++;
		}

		for (.@j = 0; .@j < .@group_count; .@j++) {
			.@index = .@low_group_index[.@j];
			.@remain_low_g = .@low_cap_g - .@low_g;
			.@remain_low_c = .@low_cap_c - .@low_c;

			if (.@remain_low_g <= 0)
				.@team = 1;
			else if (.@remain_low_c <= 0)
				.@team = 0;
			// 수라 홀수 특수 규칙:
			// 해당 저레벨 수라 그룹에서 양 팀 수라 수가 같아지는 순간에는
			// 추가 1명을 제네릭 반대 팀으로 우선 배치합니다.
			else if (
				callfunc("F_BG_RM_NoticeJobFamily", .@target_class) == 13 &&
				.@sura_extra_team >= 0 &&
				.@class_g == .@class_c &&
				((.@group_count - .@j) % 2) == 1
			)
				.@team = .@sura_extra_team;
			// 동일 직업이 적은 팀을 가장 먼저 선택합니다.
			else if (.@class_g < .@class_c)
				.@team = 0;
			else if (.@class_c < .@class_g)
				.@team = 1;
			// 동일 직업 수가 같으면 전체 저레벨 인원 수를 맞춥니다.
			else if (.@low_g < .@low_c)
				.@team = 0;
			else if (.@low_c < .@low_g)
				.@team = 1;
			// 인원까지 같으면 저레벨 레벨 합을 맞춥니다.
			else if (.@low_level_sum_g < .@low_level_sum_c)
				.@team = 0;
			else if (.@low_level_sum_c < .@low_level_sum_g)
				.@team = 1;
			else
				.@team = rand(2);

			if (.@team == 0) {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
				.@low_g++;
				.@low_level_sum_g += .@level_list[.@index];
				.@class_g++;
				.@low_team[.@index] = 1;
			}
			else {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
				.@low_c++;
				.@low_level_sum_c += .@level_list[.@index];
				.@class_c++;
				.@low_team[.@index] = 2;
			}

			.@low_assigned[.@index] = 1;
		}

		debugmes "[BG RM] low class split"
			+ " class=" + .@target_class
			+ " added=" + .@group_count
			+ " classG=" + .@class_g
			+ " classC=" + .@class_c;
	}


	debugmes "[BG RM] low-level assignment done"
		+ " G_low=" + .@low_g + " sum=" + .@low_level_sum_g
		+ " C_low=" + .@low_c + " sum=" + .@low_level_sum_c;

	if (
		.@count_g != .@cap_g ||
		.@count_c != .@cap_c ||
		.@low_g != .@low_cap_g ||
		.@low_c != .@low_cap_c ||
		(.@count_g > .@count_c + 1 || .@count_c > .@count_g + 1) ||
		.@count_g > .MaxPartyMembers ||
		.@count_c > .MaxPartyMembers
	) {
		debugmes "[BG RM] team assignment invalid"
			+ " G=" + .@count_g + "/" + .@cap_g
			+ " C=" + .@count_c + "/" + .@cap_c
			+ " G_low=" + .@low_g + "/" + .@low_cap_g
			+ " C_low=" + .@low_c + "/" + .@low_cap_c;
		callfunc "F_BG_RM_CancelRecruit", "최종 팀 배정 중 오류가 발생하여 매칭이 취소되었습니다.";
		end;
	}

	debugmes "[BG RM] team assignment done G=" + .@count_g + " C=" + .@count_c;

	// 팀 편성은 끝났으므로 기본 무한루프 보호를 즉시 복구합니다.
	freeloop 0;

	// run_script_main의 command/goto 카운터를 새 실행 구간으로 갱신합니다.
	// 1ms 뒤 같은 위치에서 이어지며 지역 배열/변수는 그대로 유지됩니다.
	sleep 1;
	debugmes "[BG RM] matching loop guard restored";

	//========================================================
	// 파티 생성 직전 최종 검증
	// 이름 기반 조회 대신 신청 당시 저장한 AID/CID로 직접 확인합니다.
	//========================================================
	.@invalid = 0;
	.@invalid_reason$ = "";
	.@invalid_name$ = "";
	.@invalid_cid = 0;

	for (.@team = 1; .@team <= 2 && !.@invalid; .@team++) {
		.@team_count = (.@team == 1) ? .@count_g : .@count_c;

		for (.@i = 0; .@i < .@team_count; .@i++) {
			.@check_aid = (.@team == 1) ? .@aid_g[.@i] : .@aid_c[.@i];
			.@check_cid = (.@team == 1) ? .@cid_g[.@i] : .@cid_c[.@i];
			.@check_name$ = (.@team == 1) ? .@name_g$[.@i] : .@name_c$[.@i];

			if (!isloggedin(.@check_aid,.@check_cid)) {
				.@invalid = 1;
				.@invalid_reason$ = "로그아웃";
			}
			else if (!attachrid(.@check_aid)) {
				.@invalid = 1;
				.@invalid_reason$ = "RID 연결 실패";
			}
			else {
				if (getcharid(0) != .@check_cid) {
					.@invalid = 1;
					.@invalid_reason$ = "캐릭터 ID 불일치";
				}
				else if (strcharinfo(3) != "bat_room") {
					.@invalid = 1;
					.@invalid_reason$ = "전장 대기존 이탈";
				}
				else if (getcharid(1)) {
					.@invalid = 1;
					.@invalid_reason$ = "기존 파티 가입";
				}
				else if (getcharid(4)) {
					.@invalid = 1;
					.@invalid_reason$ = "기존 BG 팀 가입";
				}

				detachrid;
			}

			if (.@invalid) {
				.@invalid_name$ = .@check_name$;
				.@invalid_cid = .@check_cid;
				debugmes "[BG RM] pre-party validation failed"
					+ " team=" + ((.@team == 1) ? "G" : "C")
					+ " index=" + .@i
					+ " cid=" + .@invalid_cid
					+ " reason=" + .@invalid_reason$;
				break;
			}
		}
	}

	if (.@invalid) {
		callfunc "F_BG_RM_CancelRecruit",
			"파티 생성 직전 참가자 상태가 변경되어 매칭이 취소되었습니다. 대상: "
			+ .@invalid_name$ + " / 사유: " + .@invalid_reason$;
		end;
	}

	debugmes "[BG RM] pre-party validation done G=" + .@count_g + " C=" + .@count_c;

	//========================================================
	// 파티장 선정
	// 현재 만렙 175 캐릭터만 후보에 포함합니다.
	// 175 미만 참가자는 어떤 경우에도 자동 파티장이 되지 않습니다.
	//========================================================
	.@leader_candidate_g = 0;
	for (.@i = 0; .@i < .@count_g; .@i++) {
		if (readparam(BaseLevel,.@cid_g[.@i]) == .MaxBaseLevel) {
			.@leader_index_g[.@leader_candidate_g] = .@i;
			.@leader_candidate_g++;
		}
	}

	.@leader_candidate_c = 0;
	for (.@i = 0; .@i < .@count_c; .@i++) {
		if (readparam(BaseLevel,.@cid_c[.@i]) == .MaxBaseLevel) {
			.@leader_index_c[.@leader_candidate_c] = .@i;
			.@leader_candidate_c++;
		}
	}

	if (.@leader_candidate_g <= 0 || .@leader_candidate_c <= 0) {
		callfunc "F_BG_RM_CancelRecruit",
			"175레벨 파티장 후보가 없는 진영이 발생하여 매칭이 취소되었습니다.";
		end;
	}

	.@leader_index = .@leader_index_g[rand(.@leader_candidate_g)];
	if (.@leader_index > 0) {
		.@tmp = .@aid_g[0];
		.@aid_g[0] = .@aid_g[.@leader_index];
		.@aid_g[.@leader_index] = .@tmp;

		.@tmp = .@cid_g[0];
		.@cid_g[0] = .@cid_g[.@leader_index];
		.@cid_g[.@leader_index] = .@tmp;

		.@tmp$ = .@name_g$[0];
		.@name_g$[0] = .@name_g$[.@leader_index];
		.@name_g$[.@leader_index] = .@tmp$;
	}

	.@leader_index = .@leader_index_c[rand(.@leader_candidate_c)];
	if (.@leader_index > 0) {
		.@tmp = .@aid_c[0];
		.@aid_c[0] = .@aid_c[.@leader_index];
		.@aid_c[.@leader_index] = .@tmp;

		.@tmp = .@cid_c[0];
		.@cid_c[0] = .@cid_c[.@leader_index];
		.@cid_c[.@leader_index] = .@tmp;

		.@tmp$ = .@name_c$[0];
		.@name_c$[0] = .@name_c$[.@leader_index];
		.@name_c$[.@leader_index] = .@tmp$;
	}

	debugmes "[BG RM] level-175 leaders selected"
		+ " G_cid=" + .@cid_g[0]
		+ " C_cid=" + .@cid_c[0]
		+ " G_candidates=" + .@leader_candidate_g
		+ " C_candidates=" + .@leader_candidate_c;

	$@BG_RM_PARTY_SERIAL++;
	.@serial = $@BG_RM_PARTY_SERIAL % 10000;
	.@party_suffix$ = (gettimetick(2) % 100000000) + "-" + .@serial;
	.@party_name_g$ = "[기욤]" + .@party_suffix$;
	.@party_name_c$ = "[크루아]" + .@party_suffix$;

	debugmes "[BG RM] leaders selected"
		+ " G_cid=" + .@cid_g[0]
		+ " C_cid=" + .@cid_c[0]
		+ " G_count=" + .@count_g
		+ " C_count=" + .@count_c;

	//========================================================
	// 기욤 파티 생성 및 전원 가입 확인
	//========================================================
	$@party_create_id = 0;
	debugmes "[BG RM] party_create G name=" + .@party_name_g$
		+ " cid=" + .@cid_g[0]
		+ " count=" + .@count_g
		+ " state=" + $@BG_RM_STATE;

	.@result = party_create(.@party_name_g$,.@cid_g[0],0,0);
	if (.@result != 1) {
		callfunc "F_BG_RM_CancelRecruit", "기욤 파티 생성 요청에 실패했습니다. 오류 코드: " + .@result;
		end;
	}

	.@party_id = 0;
	for (.@retry = 0; .@retry < 50; .@retry++) {
		sleep 100;
		if ($@BG_RM_STATE != 2 || .@finalize_token != $@BG_RM_TOKEN)
			end;
		.@party_id = getcharid(1,.@name_g$[0]);
		if (.@party_id > 0)
			break;
	}

	if (.@party_id <= 0) {
		debugmes "[BG RM] G party id timeout leader=" + .@name_g$[0]
			+ " cid=" + .@cid_g[0]
			+ " global_id=" + $@party_create_id;
		callfunc "F_BG_RM_CancelRecruit", "기욤 파티 생성 완료를 확인하지 못했습니다.";
		end;
	}

	$@BG_RM_PARTY_G = .@party_id;
	debugmes "[BG RM] G party created party_id=" + $@BG_RM_PARTY_G
		+ " leader=" + .@name_g$[0];
	sleep 500;
	if ($@BG_RM_STATE != 2 || .@finalize_token != $@BG_RM_TOKEN)
		end;

	for (.@i = 1; .@i < .@count_g; .@i++) {
		if (getcharid(1,.@name_g$[.@i]) > 0) {
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "기욤 파티원이 다른 파티에 가입하여 매칭이 취소되었습니다: " + .@name_g$[.@i];
			end;
		}

		.@result = 0;
		for (.@retry = 0; .@retry < 10; .@retry++) {
			.@result = party_addmember($@BG_RM_PARTY_G,.@cid_g[.@i]);
			if (.@result == 1)
				break;
			sleep 200;
		}

		debugmes "[BG RM] G addmember name=" + .@name_g$[.@i]
			+ " cid=" + .@cid_g[.@i]
			+ " party_id=" + $@BG_RM_PARTY_G
			+ " result=" + .@result;

		if (.@result != 1) {
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "기욤 파티원 추가 요청에 실패했습니다. 대상: " + .@name_g$[.@i] + " / 오류 코드: " + .@result;
			end;
		}

		.@joined = 0;
		for (.@retry = 0; .@retry < 50; .@retry++) {
			sleep 100;
			if ($@BG_RM_STATE != 2 || .@finalize_token != $@BG_RM_TOKEN)
				end;
			.@current_party = getcharid(1,.@name_g$[.@i]);
			if (.@current_party == $@BG_RM_PARTY_G) {
				.@joined = 1;
				break;
			}
			if (.@current_party > 0)
				break;
		}

		if (!.@joined) {
			debugmes "[BG RM] G join timeout name=" + .@name_g$[.@i]
				+ " expected=" + $@BG_RM_PARTY_G
				+ " actual=" + getcharid(1,.@name_g$[.@i]);
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "기욤 파티원의 실제 가입 완료를 확인하지 못했습니다: " + .@name_g$[.@i];
			end;
		}
	}

	//========================================================
	// 크루아 파티 생성 및 전원 가입 확인
	//========================================================
	$@party_create_id = 0;
	debugmes "[BG RM] party_create C name=" + .@party_name_c$
		+ " cid=" + .@cid_c[0]
		+ " count=" + .@count_c
		+ " state=" + $@BG_RM_STATE;

	.@result = party_create(.@party_name_c$,.@cid_c[0],0,0);
	if (.@result != 1) {
		callfunc "F_BG_RM_DestroyParties";
		callfunc "F_BG_RM_CancelRecruit", "크루아 파티 생성 요청에 실패했습니다. 오류 코드: " + .@result;
		end;
	}

	.@party_id = 0;
	for (.@retry = 0; .@retry < 50; .@retry++) {
		sleep 100;
		if ($@BG_RM_STATE != 2 || .@finalize_token != $@BG_RM_TOKEN)
			end;
		.@party_id = getcharid(1,.@name_c$[0]);
		if (.@party_id > 0)
			break;
	}

	if (.@party_id <= 0) {
		debugmes "[BG RM] C party id timeout leader=" + .@name_c$[0]
			+ " cid=" + .@cid_c[0]
			+ " global_id=" + $@party_create_id;
		callfunc "F_BG_RM_DestroyParties";
		callfunc "F_BG_RM_CancelRecruit", "크루아 파티 생성 완료를 확인하지 못했습니다.";
		end;
	}

	$@BG_RM_PARTY_C = .@party_id;
	debugmes "[BG RM] C party created party_id=" + $@BG_RM_PARTY_C
		+ " leader=" + .@name_c$[0];
	sleep 500;
	if ($@BG_RM_STATE != 2 || .@finalize_token != $@BG_RM_TOKEN)
		end;

	for (.@i = 1; .@i < .@count_c; .@i++) {
		if (getcharid(1,.@name_c$[.@i]) > 0) {
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "크루아 파티원이 다른 파티에 가입하여 매칭이 취소되었습니다: " + .@name_c$[.@i];
			end;
		}

		.@result = 0;
		for (.@retry = 0; .@retry < 10; .@retry++) {
			.@result = party_addmember($@BG_RM_PARTY_C,.@cid_c[.@i]);
			if (.@result == 1)
				break;
			sleep 200;
		}

		debugmes "[BG RM] C addmember name=" + .@name_c$[.@i]
			+ " cid=" + .@cid_c[.@i]
			+ " party_id=" + $@BG_RM_PARTY_C
			+ " result=" + .@result;

		if (.@result != 1) {
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "크루아 파티원 추가 요청에 실패했습니다. 대상: " + .@name_c$[.@i] + " / 오류 코드: " + .@result;
			end;
		}

		.@joined = 0;
		for (.@retry = 0; .@retry < 50; .@retry++) {
			sleep 100;
			if ($@BG_RM_STATE != 2 || .@finalize_token != $@BG_RM_TOKEN)
				end;
			.@current_party = getcharid(1,.@name_c$[.@i]);
			if (.@current_party == $@BG_RM_PARTY_C) {
				.@joined = 1;
				break;
			}
			if (.@current_party > 0)
				break;
		}

		if (!.@joined) {
			debugmes "[BG RM] C join timeout name=" + .@name_c$[.@i]
				+ " expected=" + $@BG_RM_PARTY_C
				+ " actual=" + getcharid(1,.@name_c$[.@i]);
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "크루아 파티원의 실제 가입 완료를 확인하지 못했습니다: " + .@name_c$[.@i];
			end;
		}
	}

	for (.@i = 0; .@i < .@count_g; .@i++) {
		if (
			!isloggedin(.@aid_g[.@i],.@cid_g[.@i]) ||
			getmapxy(.@map$,.@x,.@y,BL_PC,.@name_g$[.@i]) != 0 ||
			.@map$ != "bat_room" ||
			getcharid(1,.@name_g$[.@i]) != $@BG_RM_PARTY_G ||
			getcharid(4,.@name_g$[.@i])
		) {
			debugmes "[BG RM] final G validation failed name=" + .@name_g$[.@i]
				+ " expected_party=" + $@BG_RM_PARTY_G
				+ " actual_party=" + getcharid(1,.@name_g$[.@i]);
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "기욤 참가자가 대기실을 이탈하거나 파티 구성이 변경되어 전장 입장이 취소되었습니다.";
			end;
		}
	}

	for (.@i = 0; .@i < .@count_c; .@i++) {
		if (
			!isloggedin(.@aid_c[.@i],.@cid_c[.@i]) ||
			getmapxy(.@map$,.@x,.@y,BL_PC,.@name_c$[.@i]) != 0 ||
			.@map$ != "bat_room" ||
			getcharid(1,.@name_c$[.@i]) != $@BG_RM_PARTY_C ||
			getcharid(4,.@name_c$[.@i])
		) {
			debugmes "[BG RM] final C validation failed name=" + .@name_c$[.@i]
				+ " expected_party=" + $@BG_RM_PARTY_C
				+ " actual_party=" + getcharid(1,.@name_c$[.@i]);
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "크루아 참가자가 대기실을 이탈하거나 파티 구성이 변경되어 전장 입장이 취소되었습니다.";
			end;
		}
	}

	// 전장 생성 함수에서 사용할 최종 팀 배열 저장
	$@BG_RM_G_COUNT = .@count_g;
	$@BG_RM_C_COUNT = .@count_c;

	for (.@i = 0; .@i < .@count_g; .@i++) {
		$@BG_RM_G_AID[.@i] = .@aid_g[.@i];
		$@BG_RM_G_CID[.@i] = .@cid_g[.@i];
		$@BG_RM_G_NAME$[.@i] = .@name_g$[.@i];
	}
	for (.@i = 0; .@i < .@count_c; .@i++) {
		$@BG_RM_C_AID[.@i] = .@aid_c[.@i];
		$@BG_RM_C_CID[.@i] = .@cid_c[.@i];
		$@BG_RM_C_NAME$[.@i] = .@name_c$[.@i];
	}

	// 랜덤 매칭 동안 기존 인원 설정을 임시 보관
	$@BG_RM_COUNTS_SAVED = 1;
	$@BG_RM_SAVED_USER_COUNT = $BG_User_Count;
	$@BG_RM_SAVED_G_COUNT = $BG_Guillaume_Count;
	$@BG_RM_SAVED_C_COUNT = $BG_Croix_Count;

	// 보상 추가량은 인원이 적은 팀을 기준으로 계산
	$BG_User_Count = ((.@count_g < .@count_c) ? .@count_g : .@count_c);
	$BG_Guillaume_Count = .@count_g;
	$BG_Croix_Count = .@count_c;

	debugmes "[BG RM] parties ready G_party=" + $@BG_RM_PARTY_G
		+ " C_party=" + $@BG_RM_PARTY_C
		+ " mode=" + $@BG_RM_MODE;

	callfunc "F_BG_RM_BeginRewardStage", $@BG_RM_MODE;

	if (!callfunc("F_BG_RM_StartBattle",$@BG_RM_MODE)) {
		callfunc "F_BG_RM_AbortCreatedBattle", $@BG_RM_MODE;
		callfunc "F_BG_RM_DestroyParties";
		callfunc "F_BG_RM_RestoreCounts";
		callfunc "F_BG_RM_CancelRecruit", "전장 팀 생성 또는 입장 처리에 실패했습니다.";
		end;
	}

	if (!callfunc("F_BG_RM_RegisterLeaders")) {
		callfunc "F_BG_RM_AbortCreatedBattle", $@BG_RM_MODE;
		callfunc "F_BG_RM_DestroyParties";
		callfunc "F_BG_RM_RestoreCounts";
		callfunc "F_BG_RM_CancelRecruit", "자동 파티장 등록에 실패하여 매칭이 취소되었습니다.";
		end;
	}

	$@BG_RM_ACTIVE = 1;
	$@BG_RM_CYCLE_ACTIVE = 1;
	$@BG_RM_CYCLE_STAGE = 1;
	$@BG_RM_STATE = 3;
	$@BG_RM_FINALIZE_WATCH_TOKEN = 0;
	donpcevent "NEED_BG_RM_FinalizeWatch::OnStop";
	callfunc "F_BG_RM_SetManager", 1;

	for (.@i = 0; .@i < $@BG_RM_G_COUNT; .@i++)
		message $@BG_RM_G_NAME$[.@i], "[전장] 기욤 팀으로 배정되었습니다. 이 파티는 KVM 종료까지 유지됩니다.";

	for (.@i = 0; .@i < $@BG_RM_C_COUNT; .@i++)
		message $@BG_RM_C_NAME$[.@i], "[전장] 크루아 팀으로 배정되었습니다. 이 파티는 KVM 종료까지 유지됩니다.";

	message $@BG_RM_G_NAME$[0], "[전장] 랜덤 매칭 기욤 파티장입니다. 티에라 종료 후 다음 전장 입장 준비를 진행해 주세요.";
	message $@BG_RM_C_NAME$[0], "[전장] 랜덤 매칭 크루아 파티장입니다. 티에라 종료 후 다음 전장 입장 준비를 진행해 주세요.";

	donpcevent "NEED_BG_RM_Monitor::OnStart";
	end;

OnAdminForceStart:
	.@admin$ = $@BG_RM_ADMIN_NAME$;
	$@BG_RM_ADMIN_NAME$ = "";

	if ($@BG_RM_STATE != 1) {
		if (.@admin$ != "")
			message .@admin$, "[전장] 현재 즉시 시작할 랜덤 매칭 모집이 없습니다.";
		end;
	}

	if ($@BG_RM_COUNT < .MinPlayers) {
		if (.@admin$ != "")
			message .@admin$, "[전장] 최소 참가 인원이 부족합니다. 현재 " + $@BG_RM_COUNT + "명 / 최소 " + .MinPlayers + "명";
		end;
	}

	stopnpctimer;
	mapannounce "bat_room",
		"[전장 랜덤 매칭] 운영자 " + ((.@admin$ == "") ? "관리자" : .@admin$)
		+ "님이 모집을 즉시 마감했습니다. 팀 편성을 시작합니다.",
		bc_map, "0xFFCC66";

	goto OnFinalize;

OnAdminForceStop:
	.@admin$ = $@BG_RM_ADMIN_NAME$;
	$@BG_RM_ADMIN_NAME$ = "";

	if (
		!$@BG_RM_STATE &&
		!$@BG_RM_ACTIVE &&
		!$@BG_RM_CYCLE_ACTIVE &&
		!$@TierraBG1 &&
		!$@FlaviusBG1 &&
		!$@KvM01BG
	) {
		if (.@admin$ != "")
			message .@admin$, "[전장] 강제 종료할 랜덤 매칭이 없습니다.";
		end;
	}

	// OnFinalize가 sleep 중이라면 기존 실행이 재개되지 않도록 토큰을 변경합니다.
	$@BG_RM_TOKEN++;

	stopnpctimer;
	stopnpctimer "NEED_BG_RM_RecruitMonitor";
	stopnpctimer "NEED_BG_RM_Monitor";
	donpcevent "NEED_BG_RM_FinalizeWatch::OnStop";
	donpcevent "NEED_BG_RM_TierraTeamNotice::OnStop";
	$@BG_RM_FINALIZE_WATCH_TOKEN = 0;

	// 전장 내부 카운트다운/정리 타이머 중지
	stopnpctimer "티에라 진행#bat_a01";
	stopnpctimer "티에라 정리 타이머#bat_a01";
	stopnpctimer "티에라 카운트다운#bat_a01";
	stopnpctimer "티에라 종료 타이머#bat_a01";
	stopnpctimer "start#bat_b01";
	stopnpctimer "#bat_b01_timer";
	stopnpctimer "countdown#bat_b01";
	stopnpctimer "time#bat_b01";
	stopnpctimer "KvM01_BG";
	stopnpctimer "KvM01_BG_Out";

	// 현재 랜덤 매칭이 만든 BG 팀 정리 및 참가자 복귀
	if ($@BG_RM_MODE >= 1 && $@BG_RM_MODE <= 3)
		callfunc "F_BG_RM_AbortCreatedBattle", $@BG_RM_MODE;
	else
		callfunc "F_BG_RM_ResetMembers";

	callfunc "F_BG_ClearAllReadyState";
	callfunc "F_BG_RM_DestroyParties";
	callfunc "F_BG_RM_RestoreCounts";
	callfunc "F_BG_RM_ClearLeaders";
	callfunc "F_BG_RM_RestoreForcedPerformanceMode";

	$@BG_RM_STATE = 0;
	$@BG_RM_MANAGER_MENU_LOCK = 0;
	$@BG_RM_ACTIVE = 0;
	$@BG_RM_CYCLE_ACTIVE = 0;
	$@BG_RM_CYCLE_STAGE = 0;
	$@BG_RM_MODE = 0;
	$@BG_RM_END_TICK = 0;
	$@BG_RM_KVM_PARTY_DESTROYED = 0;
	$@BG_RM_MISSING_CURSOR = 0;
	$@BG_RM_STAGE_WAIT_TICK = 0;
	$@BG_RM_REWARD_KEY = 0;

	callfunc "F_BG_RM_ClearArrays";
	callfunc "F_BG_RM_SetWaitingRooms", 1;
	callfunc "F_BG_RM_SetManager", 1;

	mapannounce "bat_room",
		"[전장 랜덤 매칭] 운영자 " + ((.@admin$ == "") ? "관리자" : .@admin$)
		+ "님이 랜덤 매칭을 강제 종료했습니다.",
		bc_map, "0xFF7777";
	end;
}

//============================================================
// 랜덤 매칭 모집 참가자 이탈 감시
// - 모집 중 로그아웃, bat_room 이탈, 파티/BG 가입 시 신청 자동 취소
// - 맵 이탈은 최대 1초 이내 반영
//============================================================
-	script	NEED_BG_RM_RecruitMonitor	-1,{
	end;

OnInit:
	.NextMissingNoticeTick = 0;
	stopnpctimer;
	end;

OnStart:
	.NextMissingNoticeTick = gettimetick(2) + 30;
	stopnpctimer;
	initnpctimer;
	end;

OnStop:
	.NextMissingNoticeTick = 0;
	stopnpctimer;
	end;

OnPCLoginEvent:
	if (BG_RM_LAST_CANCEL_REASON != 1)
		end;

	announce "[전장] 이전 랜덤 매칭 참가 신청은 게임 종료로 자동 취소되었습니다.", bc_self, "0xFF5555";
	dispbottom "[전장] 모집이 아직 진행 중이라면 bat_room에서 @전장참가로 다시 신청해 주세요.";
	end;

OnPCLogoutEvent:
	if ($@BG_RM_STATE != 1)
		end;

	.@slot = -1;
	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		if ($@BG_RM_AID[.@i] == getcharid(3) && $@BG_RM_CID[.@i] == getcharid(0)) {
			.@slot = .@i;
			break;
		}
	}

	if (.@slot < 0)
		end;

	.@name$ = $@BG_RM_NAME$[.@slot];

	BG_RM_LAST_CANCEL_REASON = 1;
	BG_RM_LAST_CANCEL_TOKEN = $@BG_RM_TOKEN;
	BG_RM_LAST_CANCEL_TICK = gettimetick(2);

	callfunc "F_BG_RM_RemoveRecruitSlot", .@slot;

	mapannounce "bat_room",
		"[전장 랜덤 매칭] " + .@name$ + "님이 게임을 종료하여 참가 신청이 취소되었습니다. 현재 " + $@BG_RM_COUNT + "명",
		bc_map, "0xFFCC99";

	if ($@BG_RM_COUNT <= 0) {
		callfunc "F_BG_RM_CancelRecruit", "모든 참가자가 이탈하여 랜덤 매칭 모집이 취소되었습니다.";
		end;
	}
	end;

OnTimer1000:
	stopnpctimer;

	if ($@BG_RM_STATE != 1)
		end;

	if (.NextMissingNoticeTick > 0 && gettimetick(2) >= .NextMissingNoticeTick) {
		callfunc "F_BG_RM_AnnounceMissingApplicants";
		.NextMissingNoticeTick = gettimetick(2) + 30;
	}

	// 뒤에서부터 제거하여 배열 이동 이후의 인덱스가 꼬이지 않도록 합니다.
	for (.@i = $@BG_RM_COUNT - 1; .@i >= 0; .@i--) {
		.@aid = $@BG_RM_AID[.@i];
		.@cid = $@BG_RM_CID[.@i];
		.@name$ = $@BG_RM_NAME$[.@i];
		.@reason$ = "";
		.@reason_code = 0;

		if (!isloggedin(.@aid,.@cid)) {
			.@reason$ = "게임을 종료하여";
			.@reason_code = 1;
		}
		else if (getmapxy(.@map$,.@x,.@y,BL_PC,.@name$) != 0) {
			.@reason$ = "접속 위치를 확인할 수 없어";
			.@reason_code = 4;
		}
		else if (.@map$ != "bat_room") {
			.@reason$ = "전장 대기존을 이탈하여";
			.@reason_code = 2;
		}
		else if (getcharid(1,.@name$) || getcharid(4,.@name$)) {
			.@reason$ = "파티 또는 전장 팀에 가입하여";
			.@reason_code = 3;
		}
		else if ($@BG_RM_MIN_JOIN_LEVEL > 0 && readparam(BaseLevel,.@cid) < $@BG_RM_MIN_JOIN_LEVEL) {
			.@reason$ = "참가 레벨 조건을 충족하지 못해";
			.@reason_code = 6;
		}

		if (.@reason_code == 0)
			continue;

		if (isloggedin(.@aid,.@cid))
			callfunc "F_BG_RM_NotifyAutoCancel", .@aid, .@cid, .@reason_code;

		callfunc "F_BG_RM_RemoveRecruitSlot", .@i;

		mapannounce "bat_room",
			"[전장 랜덤 매칭] " + .@name$ + "님이 " + .@reason$ + " 참가 신청이 취소되었습니다. 현재 " + $@BG_RM_COUNT + "명",
			bc_map, "0xFFCC99";
	}

	if ($@BG_RM_COUNT <= 0) {
		callfunc "F_BG_RM_CancelRecruit", "모든 참가자가 이탈하여 랜덤 매칭 모집이 취소되었습니다.";
		end;
	}

	if ($@BG_RM_STATE == 1)
		initnpctimer;
	end;
}

//============================================================
// 랜덤 매칭 풀사이클 감시
// state 3: 현재 단계 전장 진행/정리 대기
// state 4: 다음 전장 파티장 입장 준비 대기
//============================================================
-	script	NEED_BG_RM_Monitor	-1,{
	end;

OnInit:
	stopnpctimer;
	end;

OnStart:
	stopnpctimer;
	initnpctimer;
	end;

OnTimer1000:
	stopnpctimer;

	if (!$@BG_RM_CYCLE_ACTIVE)
		end;

	.@running = 0;
	if ($@BG_RM_MODE == 1)
		.@running = $@TierraBG1;
	else if ($@BG_RM_MODE == 2)
		.@running = $@FlaviusBG1;
	else if ($@BG_RM_MODE == 3)
		.@running = $@KvM01BG;

	// 다음 전장 준비 대기 중에는 해당 전장이 시작되는 순간만 감지합니다.
	if ($@BG_RM_STATE == 4) {
		if (.@running) {
			$@BG_RM_STATE = 3;
			mapannounce "bat_room", "[전장 랜덤 매칭] " + callfunc("F_BG_RM_ModeName$",$@BG_RM_MODE) + " 입장이 완료되었습니다.", bc_map, "0x99FF99";
		}
		initnpctimer;
		end;
	}

	// KVM 보상 단계에서 온라인 랜덤 매칭 참가자가 모두 대기실로 복귀하면
	// KVM 자체의 60초 정리 타이머를 기다리지 않고 자동 생성 파티만 즉시 해체합니다.
	// 전체 랜덤 매칭 상태/인원 설정/BG 정리는 기존 KVM 종료 타이머가 끝난 뒤 처리됩니다.
	if (
		$@BG_RM_MODE == 3 &&
		$@BG_RM_CYCLE_STAGE == 3 &&
		$@KvM01BG == 3 &&
		$@KvM01BG_Victory > 0 &&
		!$@BG_RM_KVM_PARTY_DESTROYED &&
		!callfunc("F_BG_RM_HasOnlineMemberInMap","bat_c01")
	) {
		callfunc "F_BG_RM_DestroyParties";
		$@BG_RM_KVM_PARTY_DESTROYED = 1;

		mapannounce "bat_room",
			"[전장 랜덤 매칭] KVM 참가자가 모두 대기실로 복귀하여 자동 생성 파티를 해체했습니다.",
			bc_map, "0x99FF99";
	}

	if (.@running) {
		initnpctimer;
		end;
	}

	// 현재 단계의 BG 팀과 보상 처리가 모두 정리된 뒤 다음 단계로 전환합니다.
	callfunc "F_BG_RM_AdvanceCycle";

	if ($@BG_RM_CYCLE_ACTIVE)
		initnpctimer;
	end;
}
