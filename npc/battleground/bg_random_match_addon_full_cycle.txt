// NEED 전장 랜덤 매칭 시스템 - 티에라/플라비우스/KVM 풀사이클 v12.1
// - 한 번의 매칭으로 티에라 -> 플라비우스 -> KVM 전체 사이클 진행
// - 최초 티에라는 자동 입장, 이후 전장은 양 팀 파티장의 입장 준비로 진행
// - 티에라/플라비우스 종료 후 파티 유지, KVM 종료 후에만 자동 해체
// - 명령어는 bat_room에서만 사용 가능
//   @전장매칭 : 설정된 시간 동안 모집 시작 (신청자는 자동 참가)
//   @전장참가 : 진행 중인 모집 참가
//   @전장취소 : 참가 신청 철회
//   @전장현황 : 모집/진행 상태 확인
// - 테스트본: 최소 4명 / 최대 4명 (2 vs 2)
// - 동일 직업(Class)을 가능한 한 양 팀으로 분산
// - 모집 중 전장 관리인은 유지하고 홍보 외 메뉴만 잠금
// - 자동 선정 파티장을 기존 입장 준비 시스템에 등록하여 다음 전장 진행
// - 자동 선정 파티장은 기존 수동 파티장 추가 보상 대상에서 제외
// - 모집 중 bat_room 이탈/로그아웃/파티 가입 시 참가 신청 자동 취소
//============================================================

function	script	F_BG_RM_ModeName$	{
	switch (getarg(0)) {
	case 1: return "티에라 협곡";
	case 2: return "플라비우스";
	case 3: return "KVM";
	}
	return "알 수 없음";
}

function	script	F_BG_RM_ManualWaitingCount	{
	.@count = 0;
	.@count += getwaitingroomstate(0,"기욤 대기실#bat_a01");
	.@count += getwaitingroomstate(0,"크루아 대기실#bat_a01");
	.@count += getwaitingroomstate(0,"기욤 대기실#bat_b01");
	.@count += getwaitingroomstate(0,"크루아 대기실#bat_b01");
	.@count += getwaitingroomstate(0,"KvM01R_Guillaume");
	.@count += getwaitingroomstate(0,"KvM01R_Croix");
	return .@count;
}

function	script	F_BG_RM_SetWaitingRooms	{
	.@enable = getarg(0);

	// 현재 통합 스크립트는 waitingroom 대신 파티장 입장 준비 방식을 사용합니다.
	// 기존 대기실 NPC가 우연히 활성화되어 있지 않도록 모집 중에는 숨기고,
	// 종료 후에는 각 전장 OnRefresh 정책에 맡깁니다.
	if (!.@enable) {
		disablenpc "기욤 대기실#bat_a01";
		disablenpc "크루아 대기실#bat_a01";
		disablenpc "기욤 대기실#bat_b01";
		disablenpc "크루아 대기실#bat_b01";
		disablenpc "KvM01R_Guillaume";
		disablenpc "KvM01R_Croix";
	}
	else {
		donpcevent "티에라 대기실 갱신#bat_a01::OnRefresh";
		donpcevent "플라비우스 대기실 갱신#bat_b01::OnRefresh";
		donpcevent "KVM 대기실 갱신#bat_c01::OnRefresh";
	}
	return;
}

function	script	F_BG_RM_SetManager	{
	// 랜덤 매칭 모집 중에도 전장 관리인 NPC 자체는 유지합니다.
	// 대신 bg_common.txt에서 홍보 메뉴를 제외한 나머지 메뉴만 잠급니다.
	$@BG_RM_MANAGER_MENU_LOCK = !getarg(0);
	enablenpc "전장 관리인";
	return;
}

function	script	F_BG_RM_ManualReadyActive	{
	if ($@BG_READY_TYPE)
		return 1;

	if ($@TierraReady_Guillaume || $@TierraReady_Croix || $@TierraReady_Starting)
		return 1;
	if ($@FlaviusReady_Guillaume || $@FlaviusReady_Croix || $@FlaviusReady_Starting)
		return 1;
	if ($@KvMReady_Guillaume || $@KvMReady_Croix || $@KvMReady_Starting)

}

function	script	F_BG_RM_PrepareTeamMembers	{
	.@team = getarg(0);
	.@bg_id = getarg(1);
	.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;

	for (.@i = 0; .@i < .@count; .@i++) {
		.@aid = (.@team == 1) ? $@BG_RM_G_AID[.@i] : $@BG_RM_C_AID[.@i];
		.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];

		if (!isloggedin(.@aid,.@cid))
			continue;
		if (!attachrid(.@aid))
			continue;
		if (getcharid(0) != .@cid) {
			detachrid;
			continue;
		}

		if (getcharid(4) == .@bg_id) {
			Bat_Team = .@team;
			percentheal 100,100;
		}
		detachrid;
	}
	return;
}

function	script	F_BG_RM_ResetMembers	{
	for (.@team = 1; .@team <= 2; .@team++) {
		.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;

		for (.@i = 0; .@i < .@count; .@i++) {
			.@aid = (.@team == 1) ? $@BG_RM_G_AID[.@i] : $@BG_RM_C_AID[.@i];
			.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];

			if (!isloggedin(.@aid,.@cid))
				continue;
			if (!attachrid(.@aid))
				continue;
			if (getcharid(0) != .@cid) {
				detachrid;
				continue;
			}

			// BG 관련 상태는 항상 정리합니다.
			Bat_Team = 0;
			if (getcharid(4))
				bg_leave;

			// 실제 전장 내부에 남아 있는 경우에만 대기존으로 이동합니다.
			// 이미 bat_room 또는 다른 맵에 있는 참가자를 154,150으로 다시 끌어오지 않습니다.
			.@map$ = strcharinfo(3);
			if (
				.@map$ == "bat_a01" ||
				.@map$ == "bat_b01" ||
				.@map$ == "bat_c01"
			)
				warp "bat_room",154,150;

			detachrid;
		}
	}
	return;
}
function	script	F_BG_RM_ClearArrays	{
	cleararray $@BG_RM_AID[0], 0, 24;
	cleararray $@BG_RM_CID[0], 0, 24;
	cleararray $@BG_RM_CLASS[0], 0, 24;
	cleararray $@BG_RM_NAME$[0], "", 24;

	cleararray $@BG_RM_G_AID[0], 0, 12;
	cleararray $@BG_RM_G_CID[0], 0, 12;
	cleararray $@BG_RM_G_NAME$[0], "", 12;

	cleararray $@BG_RM_C_AID[0], 0, 12;
	cleararray $@BG_RM_C_CID[0], 0, 12;
	cleararray $@BG_RM_C_NAME$[0], "", 12;

	$@BG_RM_COUNT = 0;
	$@BG_RM_G_COUNT = 0;
	$@BG_RM_C_COUNT = 0;
	return;
}


// 모집 명단에서 지정된 슬롯을 제거하고 뒤의 데이터를 앞으로 당깁니다.
function	script	F_BG_RM_RemoveRecruitSlot	{
	.@slot = getarg(0);

	if (.@slot < 0 || .@slot >= $@BG_RM_COUNT)
		return 0;

	for (.@i = .@slot; .@i < $@BG_RM_COUNT - 1; .@i++) {
		$@BG_RM_AID[.@i] = $@BG_RM_AID[.@i + 1];
		$@BG_RM_CID[.@i] = $@BG_RM_CID[.@i + 1];
		$@BG_RM_CLASS[.@i] = $@BG_RM_CLASS[.@i + 1];
		$@BG_RM_NAME$[.@i] = $@BG_RM_NAME$[.@i + 1];
	}

	$@BG_RM_COUNT--;
	$@BG_RM_AID[$@BG_RM_COUNT] = 0;
	$@BG_RM_CID[$@BG_RM_COUNT] = 0;
	$@BG_RM_CLASS[$@BG_RM_COUNT] = 0;
	$@BG_RM_NAME$[$@BG_RM_COUNT] = "";
	return 1;
}

// 최초 티에라 입장 직전에 최종 팀원이 모두 대기실과 자동 생성 파티에 남아 있는지 확인합니다.
function	script	F_BG_RM_ValidateInitialReady	{
	for (.@team = 1; .@team <= 2; .@team++) {
		.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;
		.@party_id = (.@team == 1) ? $@BG_RM_PARTY_G : $@BG_RM_PARTY_C;

		for (.@i = 0; .@i < .@count; .@i++) {
			.@aid = (.@team == 1) ? $@BG_RM_G_AID[.@i] : $@BG_RM_C_AID[.@i];
			.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];
			.@name$ = (.@team == 1) ? $@BG_RM_G_NAME$[.@i] : $@BG_RM_C_NAME$[.@i];

			if (!isloggedin(.@aid,.@cid))
				return 0;

			if (getmapxy(.@map$,.@x,.@y,BL_PC,.@name$) != 0)
				return 0;

			if (.@map$ != "bat_room")
				return 0;

			if (getcharid(1,.@name$) != .@party_id)
				return 0;

			if (getcharid(4,.@name$))
				return 0;
		}
	}

	return 1;
}

function	script	F_BG_RM_CancelReason$	{
	switch(getarg(0)) {
	case 1:
		return "게임 종료";
	case 2:
		return "전장 대기존 이탈";
	case 3:
		return "파티 또는 전장 팀 가입";
	case 4:
		return "접속 위치 확인 실패";
	case 5:
		return "직접 참가 취소";
	}
	return "알 수 없는 사유";
}

function	script	F_BG_RM_FindRecruitSlot	{
	.@aid = getarg(0);
	.@cid = getarg(1);

	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		if ($@BG_RM_AID[.@i] == .@aid && $@BG_RM_CID[.@i] == .@cid)
			return .@i;
	}

	return -1;
}

// 온라인 상태에서 자동 취소된 참가자에게 눈에 띄는 개인 알림을 표시하고
// 같은 모집에서 @전장현황을 확인할 때 취소 사유가 보이도록 기록합니다.
function	script	F_BG_RM_NotifyAutoCancel	{
	.@aid = getarg(0);
	.@cid = getarg(1);
	.@reason = getarg(2);
	.@reason$ = callfunc("F_BG_RM_CancelReason$", .@reason);

	if (!isloggedin(.@aid,.@cid))
		return 0;
	if (!attachrid(.@aid))
		return 0;
	if (getcharid(0) != .@cid) {
		detachrid;
		return 0;
	}

	BG_RM_LAST_CANCEL_REASON = .@reason;
	BG_RM_LAST_CANCEL_TOKEN = $@BG_RM_TOKEN;
	BG_RM_LAST_CANCEL_TICK = gettimetick(2);

	announce "[전장] 랜덤 매칭 참가 신청이 자동 취소되었습니다.", bc_self, "0xFF5555";
	dispbottom "[전장] 취소 사유: " + .@reason$;
	dispbottom "[전장] 다시 참가하려면 bat_room에서 @전장참가를 입력해 주세요.";
	detachrid;
	return 1;
}

// 모집 중인 참가자에게 개인 상태 유지 알림을 보냅니다.
function	script	F_BG_RM_NotifyApplicants	{
	.@remain = getarg(0);

	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		.@aid = $@BG_RM_AID[.@i];
		.@cid = $@BG_RM_CID[.@i];
		.@name$ = $@BG_RM_NAME$[.@i];

		if (!isloggedin(.@aid,.@cid))
			continue;

		message .@name$,
			"[전장] 참가 유지 중 | 남은 시간 " + .@remain
			+ " | 현재 " + $@BG_RM_COUNT + "명 | 상태 확인: @전장현황";
	}
	return;
}

//============================================================
// 전장 대기존 미신청 인원 안내
// - 모집 중 30초마다 전장 관리인이 NPC 채팅으로 안내
// - 이름이 많으면 6명씩 순환하여 표시
//============================================================
function	script	F_BG_RM_AnnounceMissingApplicants	{
	if ($@BG_RM_STATE != 1)
		return;

	.@unit_count = getmapunits(BL_PC,"bat_room",.@unit_aid[0]);
	.@missing_count = 0;

	for (.@i = 0; .@i < .@unit_count; .@i++) {
		.@aid = .@unit_aid[.@i];
		.@cid = convertpcinfo(.@aid,CPC_CHAR);
		.@name$ = convertpcinfo(.@aid,CPC_NAME);

		if (.@aid <= 0 || .@cid <= 0 || .@name$ == "")
			continue;

		if (callfunc("F_BG_RM_FindRecruitSlot",.@aid,.@cid) >= 0)
			continue;

		.@missing_name$[.@missing_count] = .@name$;
		.@missing_count++;
	}

	if (.@missing_count <= 0) {
		$@BG_RM_MISSING_CURSOR = 0;
		return;
	}

	.@start = $@BG_RM_MISSING_CURSOR;
	if (.@start < 0 || .@start >= .@missing_count)
		.@start = 0;

	.@show_count = 6;
	if (.@missing_count < .@show_count)
		.@show_count = .@missing_count;

	.@list$ = "";
	for (.@i = 0; .@i < .@show_count; .@i++) {
		.@index = (.@start + .@i) % .@missing_count;

		if (.@list$ != "")
			.@list$ += ", ";

		.@list$ += .@missing_name$[.@index];
	}

	.@message$ = "랜덤 매칭 미신청 " + .@missing_count + "명: " + .@list$;

	if (.@missing_count > .@show_count)
		.@message$ += " 외 " + (.@missing_count - .@show_count) + "명";

	.@message$ += " / 참가: @전장참가";
	npctalk .@message$, "전장 관리인";

	$@BG_RM_MISSING_CURSOR = (.@start + .@show_count) % .@missing_count;
	return;
}

function	script	F_BG_RM_PerformerFamily	{
	.@class = getarg(0);

	if (
		.@class == JOB_BARD ||
		.@class == JOB_CLOWN ||
		.@class == JOB_MINSTREL ||
		.@class == JOB_MINSTREL_T
	)
		return 1;

	if (
		.@class == JOB_DANCER ||
		.@class == JOB_GYPSY ||
		.@class == JOB_WANDERER ||
		.@class == JOB_WANDERER_T
	)
		return 2;

	return 0;
}

	$@BG_RM_MANAGER_MENU_LOCK = 0;

function	script	F_BG_RM_AbortCreatedBattle	{
	.@mode = getarg(0);

	if (.@mode == 1) {
		if ($@TierraBG1_id1) {
			bg_destroy $@TierraBG1_id1;
			$@TierraBG1_id1 = 0;
		}
		if ($@TierraBG1_id2) {
			bg_destroy $@TierraBG1_id2;
			$@TierraBG1_id2 = 0;
		}
		$@TierraBG1 = 0;
		$@TierraBG1_Victory = 0;
	}
	else if (.@mode == 2) {
		if ($@FlaviusBG1_id1) {
			bg_destroy $@FlaviusBG1_id1;
			$@FlaviusBG1_id1 = 0;
		}
		if ($@FlaviusBG1_id2) {
			bg_destroy $@FlaviusBG1_id2;
			$@FlaviusBG1_id2 = 0;
		}
		$@FlaviusBG1 = 0;
		$@FlaviusBG1_Victory = 0;
		$@Croix_ScoreBG1 = 0;
		$@Guill_ScoreBG1 = 0;
	}
	else if (.@mode == 3) {
		if ($@KvM01BG_id1) {
			bg_destroy $@KvM01BG_id1;
			$@KvM01BG_id1 = 0;
		}
		if ($@KvM01BG_id2) {
			bg_destroy $@KvM01BG_id2;
			$@KvM01BG_id2 = 0;
		}
		$@KvM01BG = 0;
		$@KvM01BG_Victory = 0;
	}

	callfunc "F_BG_RM_ResetMembers";
	return;
}
function	script	F_BG_RM_StartBattle	{
	.@mode = getarg(0);

	// 최초 티에라는 모집 확정 이후에도 참가자가 이탈하지 않았을 때만 시작합니다.
	if (.@mode == 1 && !callfunc("F_BG_RM_ValidateInitialReady"))
		return 0;

	// 티에라: 기존 30초 내부 준비 타이머를 그대로 사용합니다.
	if (.@mode == 1) {
		if ($@TierraBG1)
			return 0;

		$@TierraBG1_id1 = bg_create("bat_a01",50,374,"티에라 진행#bat_a01::OnGuillaumeQuit","");
		$@TierraBG1_id2 = bg_create("bat_a01",42,16,"티에라 진행#bat_a01::OnCroixQuit","");
		if (!$@TierraBG1_id1 || !$@TierraBG1_id2) {
			callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
			return 0;
		}

		for (.@i = 0; .@i < $@BG_RM_G_COUNT; .@i++) {
			if (!bg_join($@TierraBG1_id1,"bat_a01",50,374,$@BG_RM_G_CID[.@i])) {
				callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
				return 0;
			}
		}
		for (.@i = 0; .@i < $@BG_RM_C_COUNT; .@i++) {
			if (!bg_join($@TierraBG1_id2,"bat_a01",42,16,$@BG_RM_C_CID[.@i])) {
				callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
				return 0;
			}
		}

		callfunc "F_BG_RM_PrepareTeamMembers", 1, $@TierraBG1_id1;
		callfunc "F_BG_RM_PrepareTeamMembers", 2, $@TierraBG1_id2;
		$@TierraBG1 = 1;
		$@TierraBG1_Victory = 0;

		announce "[전장 안내] 랜덤 매칭 티에라 협곡이 기욤 " + $@BG_RM_G_COUNT + "명 vs 크루아 " + $@BG_RM_C_COUNT + "명으로 내부 준비를 시작합니다.", bc_all, "0x33CCFF";
		stopnpctimer "티에라 진행#bat_a01";
		initnpctimer "티에라 진행#bat_a01";
		return 1;
	}

	// 플라비우스: 오브젝트와 카운트다운은 30초 후 기존 엔진이 시작합니다.
	if (.@mode == 2) {
		if ($@FlaviusBG1)
			return 0;

		$@FlaviusBG1_id1 = bg_create("bat_b01",10,290,"start#bat_b01::OnGuillaumeQuit","");
		$@FlaviusBG1_id2 = bg_create("bat_b01",390,10,"start#bat_b01::OnCroixQuit","");
		if (!$@FlaviusBG1_id1 || !$@FlaviusBG1_id2) {
			callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
			return 0;
		}

		for (.@i = 0; .@i < $@BG_RM_G_COUNT; .@i++) {
			if (!bg_join($@FlaviusBG1_id1,"bat_b01",10,290,$@BG_RM_G_CID[.@i])) {
				callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
				return 0;
			}
		}
		for (.@i = 0; .@i < $@BG_RM_C_COUNT; .@i++) {
			if (!bg_join($@FlaviusBG1_id2,"bat_b01",390,10,$@BG_RM_C_CID[.@i])) {
				callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
				return 0;
			}
		}

		callfunc "F_BG_RM_PrepareTeamMembers", 1, $@FlaviusBG1_id1;
		callfunc "F_BG_RM_PrepareTeamMembers", 2, $@FlaviusBG1_id2;
		$@FlaviusBG1 = 1;
		$@FlaviusBG1_Victory = 0;
		$@Croix_ScoreBG1 = 0;
		$@Guill_ScoreBG1 = 0;
		bg_updatescore "bat_b01",0,0;

		announce "[전장 안내] 랜덤 매칭 플라비우스가 기욤 " + $@BG_RM_G_COUNT + "명 vs 크루아 " + $@BG_RM_C_COUNT + "명으로 내부 준비를 시작합니다.", bc_all;
		stopnpctimer "start#bat_b01";
		initnpctimer "start#bat_b01";
		return 1;
	}

	// KVM
	if (.@mode == 3) {
		if ($@KvM01BG)
			return 0;

		$@KvM01BG_id1 = bg_create("bat_c01",52,129,"KvM01_BG::OnGuillaumeQuit","KvM01_BG::OnGuillaumeDie");
		$@KvM01BG_id2 = bg_create("bat_c01",147,55,"KvM01_BG::OnCroixQuit","KvM01_BG::OnCroixDie");
		if (!$@KvM01BG_id1 || !$@KvM01BG_id2) {
			callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
			return 0;
		}

		for (.@i = 0; .@i < $@BG_RM_G_COUNT; .@i++) {
			if (!bg_join($@KvM01BG_id1,"bat_c01",52,129,$@BG_RM_G_CID[.@i])) {
				callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
				return 0;
			}
		}
		for (.@i = 0; .@i < $@BG_RM_C_COUNT; .@i++) {
			if (!bg_join($@KvM01BG_id2,"bat_c01",147,55,$@BG_RM_C_CID[.@i])) {
				callfunc "F_BG_RM_AbortCreatedBattle", .@mode;
				return 0;
			}
		}

		callfunc "F_BG_RM_PrepareTeamMembers", 1, $@KvM01BG_id1;
		callfunc "F_BG_RM_PrepareTeamMembers", 2, $@KvM01BG_id2;
		$@KvM01BG = 1;
		$@KvM01BG_Victory = 0;
		bg_updatescore "bat_c01",0,0;
		bg_warp $@KvM01BG_id1,"bat_c01",53,128;
		bg_warp $@KvM01BG_id2,"bat_c01",146,55;

		announce "[전장 안내] 랜덤 매칭 KVM이 기욤 " + $@BG_RM_G_COUNT + "명 vs 크루아 " + $@BG_RM_C_COUNT + "명으로 내부 준비를 시작합니다.", bc_all;
		donpcevent "KvM01_BG::OnStart";
		return 1;
	}

	return 0;
}

// KVM 보상 수령 또는 자동 이동으로 모든 온라인 참가자가 bat_c01을 벗어났는지 확인합니다.
// 로그아웃한 참가자는 대기 대상으로 잡지 않습니다.
function	script	F_BG_RM_HasOnlineMemberInMap	{
	.@target_map$ = getarg(0);

	for (.@team = 1; .@team <= 2; .@team++) {
		.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;

		for (.@i = 0; .@i < .@count; .@i++) {
			.@aid = (.@team == 1) ? $@BG_RM_G_AID[.@i] : $@BG_RM_C_AID[.@i];
			.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];
			.@name$ = (.@team == 1) ? $@BG_RM_G_NAME$[.@i] : $@BG_RM_C_NAME$[.@i];

			if (!isloggedin(.@aid,.@cid))
				continue;

			if (getmapxy(.@map$,.@x,.@y,BL_PC,.@name$) != 0)
				continue;

			if (.@map$ == .@target_map$)
				return 1;
		}
	}

	return 0;
}

function	script	F_BG_RM_Finish	{
	.@reason$ = getarg(0, "티에라 → 플라비우스 → KVM 랜덤 매칭 사이클이 종료되었습니다.");

	if (!$@BG_RM_CYCLE_ACTIVE && !$@BG_RM_ACTIVE)
		return;

	stopnpctimer "NEED_BG_RM_Monitor";
	stopnpctimer "NEED_BG_RM_RecruitMonitor";
	callfunc "F_BG_ClearAllReadyState";
	callfunc "F_BG_RM_ResetMembers";
	callfunc "F_BG_RM_DestroyParties";
	callfunc "F_BG_RM_RestoreCounts";
	callfunc "F_BG_RM_ClearLeaders";

	$@BG_RM_ACTIVE = 0;
	$@BG_RM_CYCLE_ACTIVE = 0;
	$@BG_RM_CYCLE_STAGE = 0;
	$@BG_RM_STATE = 0;
	$@BG_RM_MODE = 0;
	$@BG_RM_END_TICK = 0;
	$@BG_RM_KVM_PARTY_DESTROYED = 0;

	callfunc "F_BG_RM_ClearArrays";
	callfunc "F_BG_RM_SetWaitingRooms", 1;
	callfunc "F_BG_RM_SetManager", 1;

	mapannounce "bat_room", "[전장 랜덤 매칭] " + .@reason$, bc_map, "0x99FF99";
	return;
}

// 현재 전장이 완전히 정리되면 같은 파티를 유지한 채 다음 단계로 전환합니다.
function	script	F_BG_RM_AdvanceCycle	{
	if (!$@BG_RM_CYCLE_ACTIVE)
		return;

	callfunc "F_BG_ClearAllReadyState";

	if ($@BG_RM_CYCLE_STAGE == 1) {
		$@BG_RM_CYCLE_STAGE = 2;
		$@BG_RM_MODE = 2;
		$@BG_RM_STATE = 4;
		mapannounce "bat_room", "[전장 랜덤 매칭] 티에라가 종료되었습니다. 파티를 유지하며 다음 전장은 플라비우스입니다.", bc_map, "0x99FF99";
		mapannounce "bat_room", "[전장 랜덤 매칭] 기욤/크루아 파티장은 전장 관리인에서 입장 준비를 완료해 주세요.", bc_map, "0x99FF99";
		return;
	}

	if ($@BG_RM_CYCLE_STAGE == 2) {
		$@BG_RM_CYCLE_STAGE = 3;
		$@BG_RM_MODE = 3;
		$@BG_RM_STATE = 4;
		mapannounce "bat_room", "[전장 랜덤 매칭] 플라비우스가 종료되었습니다. 파티를 유지하며 다음 전장은 KVM입니다.", bc_map, "0x99FF99";
		mapannounce "bat_room", "[전장 랜덤 매칭] 기욤/크루아 파티장은 전장 관리인에서 입장 준비를 완료해 주세요.", bc_map, "0x99FF99";
		return;
	}

	callfunc "F_BG_RM_Finish", "KVM까지 모든 전장이 종료되어 자동 생성 파티를 해체했습니다.";
	return;
}

// 랜덤 매칭 명령어 및 모집 타이머
//============================================================
-	script	NEED_BG_RM_Core	-1,{
	end;

OnInit:
	bindatcmd "전장매칭", strnpcinfo(3) + "::OnCommandStart", 0, 99;
	bindatcmd "전장참가", strnpcinfo(3) + "::OnCommandJoin", 0, 99;
	bindatcmd "전장취소", strnpcinfo(3) + "::OnCommandCancel", 0, 99;
	bindatcmd "전장현황", strnpcinfo(3) + "::OnCommandStatus", 0, 99;

	//========================================================
	// 내부 테스트 설정: 총 4명 모집 = 2 vs 2
	// 운영 전환 시 아래 두 값을 10 / (.MaxPartyMembers * 2)로 복구
	//========================================================
	.MinPlayers = 10;
	.MaxPartyMembers = MAX_PARTY;
	.MaxPlayers = 48;
	.RecruitSeconds = 300;

	stopnpctimer;
	stopnpctimer "NEED_BG_RM_RecruitMonitor";

	// 스크립트 리로드 시 진행 중인 사이클 정보를 강제로 지우지 않습니다.
	if ($@BG_RM_CYCLE_ACTIVE) {
		callfunc "F_BG_RM_SetWaitingRooms", 0;
		callfunc "F_BG_RM_SetManager", 1;
		donpcevent "NEED_BG_RM_Monitor::OnStart";
		end;
	}

	$@BG_RM_STATE = 0;
	$@BG_RM_ACTIVE = 0;
	$@BG_RM_CYCLE_ACTIVE = 0;
	$@BG_RM_CYCLE_STAGE = 0;
	$@BG_RM_MODE = 0;
	$@BG_RM_END_TICK = 0;
	$@BG_RM_KVM_PARTY_DESTROYED = 0;
	$@BG_RM_PARTY_G = 0;
	$@BG_RM_PARTY_C = 0;
	$@BG_RM_COUNTS_SAVED = 0;
	$@BG_RM_KVM_PARTY_DESTROYED = 0;
	$@BG_RM_MISSING_CURSOR = 0;
	callfunc "F_BG_RM_ClearArrays";
	end;

OnCommandStart:
	if (strcharinfo(3) != "bat_room") {
		dispbottom "[전장] @전장매칭 명령어는 전장 대기존에서만 사용할 수 있습니다.";
		end;
	}

	if (!#NEED_BG_RULE_AGREE) {
		dispbottom "[전장] 전장 이용 규칙 동의 후 사용할 수 있습니다.";
		end;
	}

	if ($@BG_RM_STATE || $@BG_RM_ACTIVE || $@BG_RM_CYCLE_ACTIVE) {
		dispbottom "[전장] 랜덤 매칭 모집 또는 티에라/플라비우스/KVM 사이클이 이미 진행 중입니다.";
		end;
	}

	if ($@TierraBG1 || $@FlaviusBG1 || $@KvM01BG) {
		dispbottom "[전장] 현재 진행 중인 전장이 있어 랜덤 매칭을 시작할 수 없습니다.";
		end;
	}

	if (callfunc("F_BG_RM_ManualReadyActive")) {
		dispbottom "[전장] 파티장 전장 입장 준비가 진행 중이어서 랜덤 매칭을 시작할 수 없습니다.";
		end;
	}

	if (getcharid(1) || getcharid(4)) {
		dispbottom "[전장] 파티 또는 전장 팀에 가입되지 않은 상태에서 신청해 주세요.";
		end;
	}

	if ($BG_GUILLAUME_PARTY_ID || $BG_CROIX_PARTY_ID) {
		dispbottom "[전장] 수동 전장 파티장 등록이 존재하여 랜덤 매칭을 시작할 수 없습니다.";
		end;
	}

	if (callfunc("F_BG_RM_ManualWaitingCount") > 0) {
		dispbottom "[전장] 기존 전장 대기실에 참가 중인 인원이 있어 랜덤 매칭을 시작할 수 없습니다.";
		end;
	}

	// 현재 bat_room에서 참가 가능한 인원이 최소 인원 이상인지 사전 확인
	.@unit_count = getmapunits(BL_PC,"bat_room",.@unit_aid[0]);
	.@eligible = 0;

	for (.@i = 0; .@i < .@unit_count; .@i++) {
		.@aid = .@unit_aid[.@i];
		.@cid = convertpcinfo(.@aid,CPC_CHAR);
		.@name$ = convertpcinfo(.@aid,CPC_NAME);

		if (.@aid <= 0 || .@cid <= 0 || .@name$ == "")
			continue;
		if (getcharid(1,.@name$) || getcharid(4,.@name$))
			continue;

		.@eligible++;
	}

	if (.@eligible < .MinPlayers) {
		dispbottom "[전장] 현재 참가 가능한 인원이 " + .MinPlayers + "명 미만입니다. (현재 " + .@eligible + "명)";
		end;
	}

	callfunc "F_BG_RM_ClearArrays";
	$@BG_RM_KVM_PARTY_DESTROYED = 0;
	$@BG_RM_MISSING_CURSOR = 0;

	$@BG_RM_TOKEN++;
	$@BG_RM_STATE = 1;
	$@BG_RM_MODE = 1;
	$@BG_RM_END_TICK = gettimetick(2) + .RecruitSeconds;

	// 신청자는 자동 참가
	$@BG_RM_AID[0] = getcharid(3);
	$@BG_RM_CID[0] = getcharid(0);
	$@BG_RM_CLASS[0] = Class;
	$@BG_RM_NAME$[0] = strcharinfo(0);
	$@BG_RM_COUNT = 1;

	BG_RM_LAST_CANCEL_REASON = 0;
	BG_RM_LAST_CANCEL_TOKEN = 0;
	BG_RM_LAST_CANCEL_TICK = 0;

	announce "[전장] 랜덤 매칭 참가 신청이 완료되었습니다.", bc_self, "0x66FF66";
	dispbottom "[전장] 모집 종료까지 bat_room에 머물러야 참가가 유지됩니다.";
	dispbottom "[전장] 내 참가 상태 확인: @전장현황";

	callfunc "F_BG_RM_SetWaitingRooms", 0;
	callfunc "F_BG_RM_SetManager", 0;

	stopnpctimer;
	initnpctimer;
	donpcevent "NEED_BG_RM_RecruitMonitor::OnStart";

	mapannounce "bat_room", "[전장 랜덤 매칭] " + strcharinfo(0) + "님이 티에라 → 플라비우스 → KVM 풀사이클 매칭을 신청했습니다.", bc_map, "0x66CCFF";
	mapannounce "bat_room", "[전장 랜덤 매칭] 참가자는 " + .RecruitSeconds + "초 이내에 @전장참가를 입력해 주세요. 신청 취소: @전장취소", bc_map, "0x66CCFF";
	mapannounce "bat_room", "[전장 랜덤 매칭] 현재 참가 1명 / 최소 " + .MinPlayers + "명", bc_map, "0x66CCFF";
	end;

OnCommandJoin:
	if (strcharinfo(3) != "bat_room") {
		dispbottom "[전장] @전장참가 명령어는 전장 대기존에서만 사용할 수 있습니다.";
		end;
	}

	if ($@BG_RM_STATE != 1) {
		dispbottom "[전장] 현재 참가자를 모집 중인 랜덤 매칭이 없습니다.";
		end;
	}

	if (!#NEED_BG_RULE_AGREE) {
		dispbottom "[전장] 전장 이용 규칙 동의 후 참가할 수 있습니다.";
		end;
	}

	if (getcharid(1) || getcharid(4)) {
		dispbottom "[전장] 파티 또는 전장 팀에 가입된 상태에서는 참가할 수 없습니다.";
		end;
	}

	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		if ($@BG_RM_AID[.@i] == getcharid(3) && $@BG_RM_CID[.@i] == getcharid(0)) {
			.@remain = $@BG_RM_END_TICK - gettimetick(2);
			if (.@remain < 0)
				.@remain = 0;
			announce "[전장] 현재 랜덤 매칭 참가 신청이 유지 중입니다.", bc_self, "0x66FF66";
			dispbottom "[전장] 내 상태: 참가 중 / 남은 시간 " + .@remain + "초 / 현재 " + $@BG_RM_COUNT + "명";
			end;
		}
	}

	if ($@BG_RM_COUNT >= .MaxPlayers) {
		dispbottom "[전장] 랜덤 매칭 참가 정원 " + .MaxPlayers + "명이 모두 찼습니다.";
		end;
	}

	.@slot = $@BG_RM_COUNT;
	$@BG_RM_AID[.@slot] = getcharid(3);
	$@BG_RM_CID[.@slot] = getcharid(0);
	$@BG_RM_CLASS[.@slot] = Class;
	$@BG_RM_NAME$[.@slot] = strcharinfo(0);
	$@BG_RM_COUNT++;

	BG_RM_LAST_CANCEL_REASON = 0;
	BG_RM_LAST_CANCEL_TOKEN = 0;
	BG_RM_LAST_CANCEL_TICK = 0;

	announce "[전장] 랜덤 매칭 참가 신청이 완료되었습니다.", bc_self, "0x66FF66";
	dispbottom "[전장] 모집 종료까지 bat_room에 머물러야 참가가 유지됩니다.";
	dispbottom "[전장] 내 참가 상태 확인: @전장현황";

	mapannounce "bat_room", "[전장 랜덤 매칭] " + strcharinfo(0) + "님 참가. 현재 " + $@BG_RM_COUNT + "명 / 최소 " + .MinPlayers + "명", bc_map, "0x99DDFF";
	end;

OnCommandCancel:
	if (strcharinfo(3) != "bat_room") {
		dispbottom "[전장] @전장취소 명령어는 전장 대기존에서만 사용할 수 있습니다.";
		end;
	}

	// 전장 사이의 준비 대기 단계에서는 자동 선정 파티장만 전체 사이클을 종료할 수 있습니다.
	if ($@BG_RM_CYCLE_ACTIVE && $@BG_RM_STATE == 4) {
		if (getcharid(0) != $BG_GUILLAUME_LEADER_CID && getcharid(0) != $BG_CROIX_LEADER_CID) {
			dispbottom "[전장] 랜덤 매칭 사이클 종료는 자동 선정된 양 팀 파티장만 가능합니다.";
			end;
		}

		mes "[전장 랜덤 매칭]";
		mes "현재 유지 중인 랜덤 매칭 파티와";
		mes "남은 전장 사이클을 모두 종료하시겠습니까?";
		next;
		if (select("사이클 종료:취소") == 2)
			close;

		close2;
		callfunc "F_BG_RM_Finish", strcharinfo(0) + " 파티장이 랜덤 매칭 사이클을 종료했습니다.";
		end;
	}

	if ($@BG_RM_STATE != 1) {
		dispbottom "[전장] 현재 취소할 랜덤 매칭 참가 신청이 없습니다.";
		end;
	}

	.@slot = -1;
	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		if ($@BG_RM_AID[.@i] == getcharid(3) && $@BG_RM_CID[.@i] == getcharid(0)) {
			.@slot = .@i;
			break;
		}
	}

	if (.@slot < 0) {
		dispbottom "[전장] 랜덤 매칭에 참가 신청한 상태가 아닙니다.";
		end;
	}

	callfunc "F_BG_RM_RemoveRecruitSlot", .@slot;

	BG_RM_LAST_CANCEL_REASON = 5;
	BG_RM_LAST_CANCEL_TOKEN = $@BG_RM_TOKEN;
	BG_RM_LAST_CANCEL_TICK = gettimetick(2);

	announce "[전장] 랜덤 매칭 참가 신청을 취소했습니다.", bc_self, "0xFFAA55";
	dispbottom "[전장] 다시 참가하려면 모집 종료 전 @전장참가를 입력해 주세요.";

	mapannounce "bat_room", "[전장 랜덤 매칭] " + strcharinfo(0) + "님이 참가를 취소했습니다. 현재 " + $@BG_RM_COUNT + "명", bc_map, "0xFFCC99";

	if ($@BG_RM_COUNT <= 0) {
		callfunc "F_BG_RM_CancelRecruit", "모든 참가자가 신청을 취소하여 랜덤 매칭 모집이 종료되었습니다.";
		end;
	}
	end;

OnCommandStatus:
	if (strcharinfo(3) != "bat_room") {
		dispbottom "[전장] @전장현황 명령어는 전장 대기존에서만 사용할 수 있습니다.";
		end;
	}

	if ($@BG_RM_STATE == 1) {
		.@remain = $@BG_RM_END_TICK - gettimetick(2);
		if (.@remain < 0)
			.@remain = 0;

		.@slot = callfunc("F_BG_RM_FindRecruitSlot", getcharid(3), getcharid(0));
		if (.@slot >= 0) {
			announce "[전장] 내 상태: 랜덤 매칭 참가 중", bc_self, "0x66FF66";
			dispbottom "[전장] 참가 유지 중 / 현재 " + $@BG_RM_COUNT + "명 / 남은 시간 " + .@remain + "초";
			dispbottom "[전장] bat_room 이탈·로그아웃·파티 가입 시 자동 취소됩니다.";
		}
		else if (BG_RM_LAST_CANCEL_TOKEN == $@BG_RM_TOKEN && BG_RM_LAST_CANCEL_REASON > 0) {
			announce "[전장] 내 상태: 참가 신청 취소됨", bc_self, "0xFF5555";
			dispbottom "[전장] 취소 사유: " + callfunc("F_BG_RM_CancelReason$", BG_RM_LAST_CANCEL_REASON);
			dispbottom "[전장] 현재 모집에 다시 참가하려면 @전장참가를 입력해 주세요.";
		}
		else {
			announce "[전장] 내 상태: 현재 모집 미참가", bc_self, "0xFFCC66";
			dispbottom "[전장] 참가하려면 @전장참가를 입력해 주세요.";
		}

		dispbottom "[전장] 모집 현황: " + callfunc("F_BG_RM_ModeName$",$@BG_RM_MODE)
			+ " / 참가 " + $@BG_RM_COUNT + "명 / 남은 시간 " + .@remain + "초";
		end;
	}

	if ($@BG_RM_CYCLE_ACTIVE) {
		if ($@BG_RM_STATE == 4)
			dispbottom "[전장] 사이클 유지 중 / 다음 전장: " + callfunc("F_BG_RM_ModeName$",$@BG_RM_MODE) + " / 양 팀 파티장 입장 준비 대기";
		else
			dispbottom "[전장] 진행 중: " + callfunc("F_BG_RM_ModeName$",$@BG_RM_MODE) + " / 기욤 " + $@BG_RM_G_COUNT + "명 vs 크루아 " + $@BG_RM_C_COUNT + "명 / 단계 " + $@BG_RM_CYCLE_STAGE + "/3";
		end;
	}

	dispbottom "[전장] 현재 랜덤 매칭 없음 / 시작 전장: 티에라 협곡 / 시작: @전장매칭";
	end;

OnTimer60000:
	if ($@BG_RM_STATE == 1) {
		mapannounce "bat_room", "[전장 랜덤 매칭] 모집 종료까지 4분. 현재 참가 " + $@BG_RM_COUNT + "명", bc_map, "0x66CCFF";
		callfunc "F_BG_RM_NotifyApplicants", "4분";

OnTimer120000:

		callfunc "F_BG_RM_NotifyApplicants", "3분";
	}
	end;

	if ($@BG_RM_STATE == 1) {
		mapannounce "bat_room", "[전장 랜덤 매칭] 모집 종료까지 2분. 현재 참가 " + $@BG_RM_COUNT + "명", bc_map, "0x66CCFF";
		callfunc "F_BG_RM_NotifyApplicants", "2분";
	}
	end;

		callfunc "F_BG_RM_NotifyApplicants", "1분";
	}
	end;

OnTimer270000:

	end;

	if ($@BG_RM_STATE == 1)
		mapannounce "bat_room", "[전장 랜덤 매칭] 모집 종료까지 5초 남았습니다. 현재 참가 " + $@BG_RM_COUNT + "명", bc_map, "0xFFCC66";
	end;

	stopnpctimer;
	if ($@BG_RM_STATE == 1)
		donpcevent strnpcinfo(3) + "::OnFinalize";
	end;

OnFinalize:
	if ($@BG_RM_STATE != 1)
		end;

	stopnpctimer "NEED_BG_RM_RecruitMonitor";
	$@BG_RM_MISSING_CURSOR = 0;
	$@BG_RM_STATE = 2;

	// 모집 중 이미 열려 있던 관리인 대화 등으로 수동 준비 상태가 생겼는지 재확인
	if ($@TierraBG1 || $@FlaviusBG1 || $@KvM01BG || callfunc("F_BG_RM_ManualReadyActive") || $BG_GUILLAUME_PARTY_ID || $BG_CROIX_PARTY_ID) {
		callfunc "F_BG_RM_CancelRecruit", "수동 파티장 전장 준비와 충돌하여 매칭이 취소되었습니다.";
		end;
	}

	.@count = 0;

	// 접속, 위치, 파티 상태를 최종 재검증
	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		.@aid = $@BG_RM_AID[.@i];
		.@cid = $@BG_RM_CID[.@i];
		.@name$ = $@BG_RM_NAME$[.@i];

		if (!isloggedin(.@aid,.@cid))
			continue;
		if (getmapxy(.@map$,.@x,.@y,BL_PC,.@name$) != 0)
			continue;
		if (.@map$ != "bat_room") {
			message .@name$, "[전장] 전장 대기존을 이탈하여 랜덤 매칭에서 제외되었습니다.";
			continue;
		}
		if (getcharid(1,.@name$) || getcharid(4,.@name$)) {
			message .@name$, "[전장] 파티 또는 전장 팀에 가입하여 랜덤 매칭에서 제외되었습니다.";
			continue;
		}

		.@aid_list[.@count] = .@aid;
		.@cid_list[.@count] = .@cid;
		.@name_list$[.@count] = .@name$;
		.@class_list[.@count] = readparam(Class,.@cid);
		.@count++;
	}

	if (.@count < .MinPlayers) {
		callfunc "F_BG_RM_CancelRecruit", "최종 참가자가 " + .MinPlayers + "명 미만이어서 매칭이 취소되었습니다. (" + .@count + "명)";
		end;
	}

	// 전체 참가 순서 셔플
	for (.@i = .@count - 1; .@i > 0; .@i--) {
		.@j = rand(.@i + 1);

		.@tmp = .@aid_list[.@i];
		.@aid_list[.@i] = .@aid_list[.@j];
		.@aid_list[.@j] = .@tmp;

		.@tmp = .@cid_list[.@i];
		.@cid_list[.@i] = .@cid_list[.@j];
		.@cid_list[.@j] = .@tmp;

		.@tmp$ = .@name_list$[.@i];
		.@name_list$[.@i] = .@name_list$[.@j];
		.@name_list$[.@j] = .@tmp$;

		.@tmp = .@class_list[.@i];
		.@class_list[.@i] = .@class_list[.@j];
		.@class_list[.@j] = .@tmp;
	}

	// 홀수 인원일 때 한 명이 더 많은 팀은 무작위
	if ((.@count % 2) == 0) {
		.@cap_g = .@count / 2;
		.@cap_c = .@count / 2;
	}
	else if (rand(2) == 0) {
		.@cap_g = (.@count + 1) / 2;
		.@cap_c = .@count / 2;
	}
	else {
		.@cap_g = .@count / 2;
		.@cap_c = (.@count + 1) / 2;
	}

	cleararray .@used[0], 0, .@count;
	.@count_g = 0;
	.@count_c = 0;
	.@single_count = 0;

	// 우선 분리 규칙:
	// 민스트럴 계열과 원더러 계열이 전체 참가자 중 각각 정확히 1명이라면
	// 일반 단독 직업 무작위 배정보다 먼저 서로 다른 팀에 고정 배정합니다.
	.@minstrel_count = 0;
	.@wanderer_count = 0;

	for (.@i = 0; .@i < .@count; .@i++) {
		.@family = callfunc("F_BG_RM_PerformerFamily", .@class_list[.@i]);

		if (.@family == 1) {
			.@minstrel_index = .@i;
			.@minstrel_count++;
		}
		else if (.@family == 2) {
			.@wanderer_index = .@i;
			.@wanderer_count++;
		}
	}

	if (.@minstrel_count == 1 && .@wanderer_count == 1) {
		// 어느 직업군이 기욤/크루아로 갈지는 무작위지만 두 직업군은 반드시 분리됩니다.
		.@minstrel_team = rand(2);

		if (.@minstrel_team == 0) {
			.@aid_g[.@count_g] = .@aid_list[.@minstrel_index];
			.@cid_g[.@count_g] = .@cid_list[.@minstrel_index];
			.@name_g$[.@count_g] = .@name_list$[.@minstrel_index];
			.@count_g++;

			.@aid_c[.@count_c] = .@aid_list[.@wanderer_index];
			.@cid_c[.@count_c] = .@cid_list[.@wanderer_index];
			.@name_c$[.@count_c] = .@name_list$[.@wanderer_index];
			.@count_c++;
		}
		else {
			.@aid_c[.@count_c] = .@aid_list[.@minstrel_index];
			.@cid_c[.@count_c] = .@cid_list[.@minstrel_index];
			.@name_c$[.@count_c] = .@name_list$[.@minstrel_index];
			.@count_c++;

			.@aid_g[.@count_g] = .@aid_list[.@wanderer_index];
			.@cid_g[.@count_g] = .@cid_list[.@wanderer_index];
			.@name_g$[.@count_g] = .@name_list$[.@wanderer_index];
			.@count_g++;
		}

		.@used[.@minstrel_index] = 1;
		.@used[.@wanderer_index] = 1;

		debugmes "[BG RM] priority split performer"
			+ " minstrel=" + .@name_list$[.@minstrel_index]
			+ " wanderer=" + .@name_list$[.@wanderer_index]
			+ " minstrel_team=" + ((.@minstrel_team == 0) ? "G" : "C");
	}

	// 1단계: 동일 직업이 2명 이상인 그룹을 기욤/크루아에 교차 배정
	// 1명뿐인 직업은 즉시 배정하지 않고 단독 직업 목록에 보관합니다.
	for (.@i = 0; .@i < .@count; .@i++) {
		if (.@used[.@i])
			continue;

		.@group_count = 0;
		.@target_class = .@class_list[.@i];

		for (.@j = 0; .@j < .@count; .@j++) {
			if (!.@used[.@j] && .@class_list[.@j] == .@target_class) {
				.@group_index[.@group_count] = .@j;
				.@used[.@j] = 1;
				.@group_count++;
			}
		}

		// 동일 직업 그룹 내부 순서 무작위화
		for (.@j = .@group_count - 1; .@j > 0; .@j--) {
			.@k = rand(.@j + 1);
			.@tmp = .@group_index[.@j];
			.@group_index[.@j] = .@group_index[.@k];
			.@group_index[.@k] = .@tmp;
		}

		// 한 명뿐인 직업은 2단계에서 서로 다른 팀으로 분산합니다.
		if (.@group_count == 1) {
			.@single_index[.@single_count] = .@group_index[0];
			.@single_count++;
			continue;
		}

		if (.@count_g < .@count_c)
			.@start_team = 0;
		else if (.@count_c < .@count_g)
			.@start_team = 1;
		else
			.@start_team = rand(2);

		for (.@j = 0; .@j < .@group_count; .@j++) {
			.@index = .@group_index[.@j];
			.@team = (.@start_team + .@j) % 2;

			if (.@team == 0 && .@count_g >= .@cap_g)
				.@team = 1;
			else if (.@team == 1 && .@count_c >= .@cap_c)
				.@team = 0;

			if (.@team == 0) {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
			}
			else {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
			}
		}
	}

	// 2단계: 1명뿐인 직업 목록을 무작위로 섞은 뒤 2명씩 양 팀에 한 명씩 배정
	for (.@i = .@single_count - 1; .@i > 0; .@i--) {
		.@j = rand(.@i + 1);
		.@tmp = .@single_index[.@i];
		.@single_index[.@i] = .@single_index[.@j];
		.@single_index[.@j] = .@tmp;
	}

	// 같은 팀부터 계속 시작하지 않도록 쌍마다 시작 팀을 교대합니다.
	.@pair_start_team = rand(2);

	for (.@i = 0; .@i < .@single_count; .@i += 2) {
		.@remain_g = .@cap_g - .@count_g;
		.@remain_c = .@cap_c - .@count_c;

		// 두 명이 남았고 양 팀 모두 자리가 있으면 반드시 한 명씩 나눕니다.
		if (.@i + 1 < .@single_count && .@remain_g > 0 && .@remain_c > 0) {
			if (.@remain_g > .@remain_c)
				.@first_team = 0;
			else if (.@remain_c > .@remain_g)
				.@first_team = 1;
			else
				.@first_team = .@pair_start_team;

			.@index = .@single_index[.@i];
			if (.@first_team == 0) {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
			}
			else {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
			}

			// 같은 쌍의 두 번째 단독 직업은 반대 팀으로 배정합니다.
			.@index = .@single_index[.@i + 1];
			if (.@first_team == 0) {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
			}
			else {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
			}

			.@pair_start_team = 1 - .@pair_start_team;
			continue;
		}

		// 한 명만 남았거나 한쪽 팀 정원이 찬 경우 남은 정원에 맞춰 배정합니다.
		for (.@j = .@i; .@j < .@single_count && .@j < .@i + 2; .@j++) {
			.@index = .@single_index[.@j];
			.@remain_g = .@cap_g - .@count_g;
			.@remain_c = .@cap_c - .@count_c;

			if (.@remain_g <= 0)
				.@team = 1;
			else if (.@remain_c <= 0)
				.@team = 0;
			else if (.@remain_g > .@remain_c)
				.@team = 0;
			else if (.@remain_c > .@remain_g)
				.@team = 1;
			else if (.@count_g < .@count_c)
				.@team = 0;
			else if (.@count_c < .@count_g)
				.@team = 1;
			else
				.@team = rand(2);

			if (.@team == 0) {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
			}
			else {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
			}
		}
	}

	if (.@count_g != .@cap_g || .@count_c != .@cap_c || (.@count_g > .@count_c + 1 || .@count_c > .@count_g + 1) || .@count_g > .MaxPartyMembers || .@count_c > .MaxPartyMembers) {
		callfunc "F_BG_RM_CancelRecruit", "팀 배정 중 오류가 발생하여 매칭이 취소되었습니다.";
		end;
	}

	// 파티 생성 직전 접속/위치/파티 상태를 한 번 더 확인
	for (.@i = 0; .@i < .@count_g; .@i++) {
		if (
			!isloggedin(.@aid_g[.@i],.@cid_g[.@i]) ||
			getmapxy(.@map$,.@x,.@y,BL_PC,.@name_g$[.@i]) != 0 ||
			.@map$ != "bat_room" ||
			getcharid(1,.@name_g$[.@i]) ||
			getcharid(4,.@name_g$[.@i])
		) {
			callfunc "F_BG_RM_CancelRecruit", "파티 생성 직전 참가자가 대기실을 이탈하거나 접속 상태가 변경되어 매칭이 취소되었습니다.";
			end;
		}
	}
	for (.@i = 0; .@i < .@count_c; .@i++) {
		if (
			!isloggedin(.@aid_c[.@i],.@cid_c[.@i]) ||
			getmapxy(.@map$,.@x,.@y,BL_PC,.@name_c$[.@i]) != 0 ||
			.@map$ != "bat_room" ||
			getcharid(1,.@name_c$[.@i]) ||
			getcharid(4,.@name_c$[.@i])
		) {
			callfunc "F_BG_RM_CancelRecruit", "파티 생성 직전 참가자가 대기실을 이탈하거나 접속 상태가 변경되어 매칭이 취소되었습니다.";
			end;
		}
	}

	//========================================================
	// 파티장 완전 무작위 선정
	// 팀 배열의 0번이 파티장이므로 무작위 대상과 교환합니다.
	//========================================================
	.@leader_index = rand(.@count_g);
	if (.@leader_index > 0) {
		.@tmp = .@aid_g[0];
		.@aid_g[0] = .@aid_g[.@leader_index];
		.@aid_g[.@leader_index] = .@tmp;

		.@tmp = .@cid_g[0];
		.@cid_g[0] = .@cid_g[.@leader_index];
		.@cid_g[.@leader_index] = .@tmp;

		.@tmp$ = .@name_g$[0];
		.@name_g$[0] = .@name_g$[.@leader_index];
		.@name_g$[.@leader_index] = .@tmp$;
	}

	.@leader_index = rand(.@count_c);
	if (.@leader_index > 0) {
		.@tmp = .@aid_c[0];
		.@aid_c[0] = .@aid_c[.@leader_index];
		.@aid_c[.@leader_index] = .@tmp;

		.@tmp = .@cid_c[0];
		.@cid_c[0] = .@cid_c[.@leader_index];
		.@cid_c[.@leader_index] = .@tmp;

		.@tmp$ = .@name_c$[0];
		.@name_c$[0] = .@name_c$[.@leader_index];
		.@name_c$[.@leader_index] = .@tmp$;
	}

	$@BG_RM_PARTY_SERIAL++;
	.@serial = $@BG_RM_PARTY_SERIAL % 10000;
	.@party_suffix$ = (gettimetick(2) % 100000000) + "-" + .@serial;
	.@party_name_g$ = "[기욤]" + .@party_suffix$;
	.@party_name_c$ = "[크루아]" + .@party_suffix$;

	//========================================================
	// 기욤 파티 생성 및 전원 가입 확인
	//========================================================
	$@party_create_id = 0;
	debugmes "[BG RM] party_create G name=" + .@party_name_g$
		+ " cid=" + .@cid_g[0]
		+ " count=" + .@count_g
		+ " state=" + $@BG_RM_STATE;

	.@result = party_create(.@party_name_g$,.@cid_g[0],0,0);
	if (.@result != 1) {
		callfunc "F_BG_RM_CancelRecruit", "기욤 파티 생성 요청에 실패했습니다. 오류 코드: " + .@result;
		end;
	}

	.@party_id = 0;
	for (.@retry = 0; .@retry < 50; .@retry++) {
		sleep 100;
		.@party_id = getcharid(1,.@name_g$[0]);
		if (.@party_id > 0)
			break;
	}

	if (.@party_id <= 0) {
		debugmes "[BG RM] G party id timeout leader=" + .@name_g$[0]
			+ " cid=" + .@cid_g[0]
			+ " global_id=" + $@party_create_id;
		callfunc "F_BG_RM_CancelRecruit", "기욤 파티 생성 완료를 확인하지 못했습니다.";
		end;
	}

	$@BG_RM_PARTY_G = .@party_id;
	debugmes "[BG RM] G party created party_id=" + $@BG_RM_PARTY_G
		+ " leader=" + .@name_g$[0];
	sleep 500;

	for (.@i = 1; .@i < .@count_g; .@i++) {
		if (getcharid(1,.@name_g$[.@i]) > 0) {
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "기욤 파티원이 다른 파티에 가입하여 매칭이 취소되었습니다: " + .@name_g$[.@i];
			end;
		}

		.@result = 0;
		for (.@retry = 0; .@retry < 10; .@retry++) {
			.@result = party_addmember($@BG_RM_PARTY_G,.@cid_g[.@i]);
			if (.@result == 1)
				break;
			sleep 200;
		}

		debugmes "[BG RM] G addmember name=" + .@name_g$[.@i]
			+ " cid=" + .@cid_g[.@i]
			+ " party_id=" + $@BG_RM_PARTY_G
			+ " result=" + .@result;

		if (.@result != 1) {
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "기욤 파티원 추가 요청에 실패했습니다. 대상: " + .@name_g$[.@i] + " / 오류 코드: " + .@result;
			end;
		}

		.@joined = 0;
		for (.@retry = 0; .@retry < 50; .@retry++) {
			sleep 100;
			.@current_party = getcharid(1,.@name_g$[.@i]);
			if (.@current_party == $@BG_RM_PARTY_G) {
				.@joined = 1;
				break;
			}
			if (.@current_party > 0)
				break;
		}

		if (!.@joined) {
			debugmes "[BG RM] G join timeout name=" + .@name_g$[.@i]
				+ " expected=" + $@BG_RM_PARTY_G
				+ " actual=" + getcharid(1,.@name_g$[.@i]);
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "기욤 파티원의 실제 가입 완료를 확인하지 못했습니다: " + .@name_g$[.@i];
			end;
		}
	}

	//========================================================
	// 크루아 파티 생성 및 전원 가입 확인
	//========================================================
	$@party_create_id = 0;
	debugmes "[BG RM] party_create C name=" + .@party_name_c$
		+ " cid=" + .@cid_c[0]
		+ " count=" + .@count_c
		+ " state=" + $@BG_RM_STATE;

	.@result = party_create(.@party_name_c$,.@cid_c[0],0,0);
	if (.@result != 1) {
		callfunc "F_BG_RM_DestroyParties";
		callfunc "F_BG_RM_CancelRecruit", "크루아 파티 생성 요청에 실패했습니다. 오류 코드: " + .@result;
		end;
	}

	.@party_id = 0;
	for (.@retry = 0; .@retry < 50; .@retry++) {
		sleep 100;
		.@party_id = getcharid(1,.@name_c$[0]);
		if (.@party_id > 0)
			break;
	}

	if (.@party_id <= 0) {
		debugmes "[BG RM] C party id timeout leader=" + .@name_c$[0]
			+ " cid=" + .@cid_c[0]
			+ " global_id=" + $@party_create_id;
		callfunc "F_BG_RM_DestroyParties";
		callfunc "F_BG_RM_CancelRecruit", "크루아 파티 생성 완료를 확인하지 못했습니다.";
		end;
	}

	$@BG_RM_PARTY_C = .@party_id;
	debugmes "[BG RM] C party created party_id=" + $@BG_RM_PARTY_C
		+ " leader=" + .@name_c$[0];
	sleep 500;

	for (.@i = 1; .@i < .@count_c; .@i++) {
		if (getcharid(1,.@name_c$[.@i]) > 0) {
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "크루아 파티원이 다른 파티에 가입하여 매칭이 취소되었습니다: " + .@name_c$[.@i];
			end;
		}

		.@result = 0;
		for (.@retry = 0; .@retry < 10; .@retry++) {
			.@result = party_addmember($@BG_RM_PARTY_C,.@cid_c[.@i]);
			if (.@result == 1)
				break;
			sleep 200;
		}

		debugmes "[BG RM] C addmember name=" + .@name_c$[.@i]
			+ " cid=" + .@cid_c[.@i]
			+ " party_id=" + $@BG_RM_PARTY_C
			+ " result=" + .@result;

		if (.@result != 1) {
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "크루아 파티원 추가 요청에 실패했습니다. 대상: " + .@name_c$[.@i] + " / 오류 코드: " + .@result;
			end;
		}

		.@joined = 0;
		for (.@retry = 0; .@retry < 50; .@retry++) {
			sleep 100;
			.@current_party = getcharid(1,.@name_c$[.@i]);
			if (.@current_party == $@BG_RM_PARTY_C) {
				.@joined = 1;
				break;
			}
			if (.@current_party > 0)
				break;
		}

		if (!.@joined) {
			debugmes "[BG RM] C join timeout name=" + .@name_c$[.@i]
				+ " expected=" + $@BG_RM_PARTY_C
				+ " actual=" + getcharid(1,.@name_c$[.@i]);
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "크루아 파티원의 실제 가입 완료를 확인하지 못했습니다: " + .@name_c$[.@i];
			end;
		}
	}

	for (.@i = 0; .@i < .@count_g; .@i++) {
		if (
			!isloggedin(.@aid_g[.@i],.@cid_g[.@i]) ||
			getmapxy(.@map$,.@x,.@y,BL_PC,.@name_g$[.@i]) != 0 ||
			.@map$ != "bat_room" ||
			getcharid(1,.@name_g$[.@i]) != $@BG_RM_PARTY_G ||
			getcharid(4,.@name_g$[.@i])
		) {
			debugmes "[BG RM] final G validation failed name=" + .@name_g$[.@i]
				+ " expected_party=" + $@BG_RM_PARTY_G
				+ " actual_party=" + getcharid(1,.@name_g$[.@i]);
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "기욤 참가자가 대기실을 이탈하거나 파티 구성이 변경되어 전장 입장이 취소되었습니다.";
			end;
		}
	}

	for (.@i = 0; .@i < .@count_c; .@i++) {
		if (
			!isloggedin(.@aid_c[.@i],.@cid_c[.@i]) ||
			getmapxy(.@map$,.@x,.@y,BL_PC,.@name_c$[.@i]) != 0 ||
			.@map$ != "bat_room" ||
			getcharid(1,.@name_c$[.@i]) != $@BG_RM_PARTY_C ||
			getcharid(4,.@name_c$[.@i])
		) {
			debugmes "[BG RM] final C validation failed name=" + .@name_c$[.@i]
				+ " expected_party=" + $@BG_RM_PARTY_C
				+ " actual_party=" + getcharid(1,.@name_c$[.@i]);
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "크루아 참가자가 대기실을 이탈하거나 파티 구성이 변경되어 전장 입장이 취소되었습니다.";
			end;
		}
	}

	// 전장 생성 함수에서 사용할 최종 팀 배열 저장
	$@BG_RM_G_COUNT = .@count_g;
	$@BG_RM_C_COUNT = .@count_c;

	for (.@i = 0; .@i < .@count_g; .@i++) {
		$@BG_RM_G_AID[.@i] = .@aid_g[.@i];
		$@BG_RM_G_CID[.@i] = .@cid_g[.@i];
		$@BG_RM_G_NAME$[.@i] = .@name_g$[.@i];
	}
	for (.@i = 0; .@i < .@count_c; .@i++) {
		$@BG_RM_C_AID[.@i] = .@aid_c[.@i];
		$@BG_RM_C_CID[.@i] = .@cid_c[.@i];
		$@BG_RM_C_NAME$[.@i] = .@name_c$[.@i];
	}

	// 랜덤 매칭 동안 기존 인원 설정을 임시 보관
	$@BG_RM_COUNTS_SAVED = 1;
	$@BG_RM_SAVED_USER_COUNT = $BG_User_Count;
	$@BG_RM_SAVED_G_COUNT = $BG_Guillaume_Count;
	$@BG_RM_SAVED_C_COUNT = $BG_Croix_Count;

	// 보상 추가량은 인원이 적은 팀을 기준으로 계산
	$BG_User_Count = ((.@count_g < .@count_c) ? .@count_g : .@count_c);
	$BG_Guillaume_Count = .@count_g;
	$BG_Croix_Count = .@count_c;

	if (!callfunc("F_BG_RM_StartBattle",$@BG_RM_MODE)) {
		callfunc "F_BG_RM_AbortCreatedBattle", $@BG_RM_MODE;
		callfunc "F_BG_RM_DestroyParties";
		callfunc "F_BG_RM_RestoreCounts";
		callfunc "F_BG_RM_CancelRecruit", "전장 팀 생성 또는 입장 처리에 실패했습니다.";
		end;
	}

	if (!callfunc("F_BG_RM_RegisterLeaders")) {
		callfunc "F_BG_RM_AbortCreatedBattle", $@BG_RM_MODE;
		callfunc "F_BG_RM_DestroyParties";
		callfunc "F_BG_RM_RestoreCounts";
		callfunc "F_BG_RM_CancelRecruit", "자동 파티장 등록에 실패하여 매칭이 취소되었습니다.";
		end;
	}

	$@BG_RM_ACTIVE = 1;
	$@BG_RM_CYCLE_ACTIVE = 1;
	$@BG_RM_CYCLE_STAGE = 1;
	$@BG_RM_STATE = 3;
	callfunc "F_BG_RM_SetManager", 1;

	for (.@i = 0; .@i < $@BG_RM_G_COUNT; .@i++)
		message $@BG_RM_G_NAME$[.@i], "[전장] 기욤 팀으로 배정되었습니다. 이 파티는 KVM 종료까지 유지됩니다.";

	for (.@i = 0; .@i < $@BG_RM_C_COUNT; .@i++)
		message $@BG_RM_C_NAME$[.@i], "[전장] 크루아 팀으로 배정되었습니다. 이 파티는 KVM 종료까지 유지됩니다.";

	message $@BG_RM_G_NAME$[0], "[전장] 랜덤 매칭 기욤 파티장입니다. 티에라 종료 후 다음 전장 입장 준비를 진행해 주세요.";
	message $@BG_RM_C_NAME$[0], "[전장] 랜덤 매칭 크루아 파티장입니다. 티에라 종료 후 다음 전장 입장 준비를 진행해 주세요.";

	donpcevent "NEED_BG_RM_Monitor::OnStart";
	end;
}



//============================================================
// 랜덤 매칭 모집 참가자 이탈 감시
// - 모집 중 로그아웃, bat_room 이탈, 파티/BG 가입 시 신청 자동 취소
// - 맵 이탈은 최대 1초 이내 반영
//============================================================
-	script	NEED_BG_RM_RecruitMonitor	-1,{
	end;

OnInit:
	.NextMissingNoticeTick = 0;
	stopnpctimer;
	end;

OnStart:
	.NextMissingNoticeTick = gettimetick(2) + 30;
	stopnpctimer;
	initnpctimer;
	end;

OnStop:
	.NextMissingNoticeTick = 0;
	stopnpctimer;
	end;

OnPCLoginEvent:
	if (BG_RM_LAST_CANCEL_REASON != 1)
		end;

	announce "[전장] 이전 랜덤 매칭 참가 신청은 게임 종료로 자동 취소되었습니다.", bc_self, "0xFF5555";
	dispbottom "[전장] 모집이 아직 진행 중이라면 bat_room에서 @전장참가로 다시 신청해 주세요.";
	end;

OnPCLogoutEvent:
	if ($@BG_RM_STATE != 1)
		end;

	.@slot = -1;
	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		if ($@BG_RM_AID[.@i] == getcharid(3) && $@BG_RM_CID[.@i] == getcharid(0)) {
			.@slot = .@i;
			break;
		}
	}

	if (.@slot < 0)
		end;

	.@name$ = $@BG_RM_NAME$[.@slot];

	BG_RM_LAST_CANCEL_REASON = 1;
	BG_RM_LAST_CANCEL_TOKEN = $@BG_RM_TOKEN;
	BG_RM_LAST_CANCEL_TICK = gettimetick(2);

	callfunc "F_BG_RM_RemoveRecruitSlot", .@slot;

	mapannounce "bat_room",
		"[전장 랜덤 매칭] " + .@name$ + "님이 게임을 종료하여 참가 신청이 취소되었습니다. 현재 " + $@BG_RM_COUNT + "명",
		bc_map, "0xFFCC99";

	if ($@BG_RM_COUNT <= 0) {
		callfunc "F_BG_RM_CancelRecruit", "모든 참가자가 이탈하여 랜덤 매칭 모집이 취소되었습니다.";
		end;
	}
	end;

OnTimer1000:
	stopnpctimer;

	if ($@BG_RM_STATE != 1)
		end;

	if (.NextMissingNoticeTick > 0 && gettimetick(2) >= .NextMissingNoticeTick) {
		callfunc "F_BG_RM_AnnounceMissingApplicants";
		.NextMissingNoticeTick = gettimetick(2) + 30;
	}

	// 뒤에서부터 제거하여 배열 이동 이후의 인덱스가 꼬이지 않도록 합니다.
	for (.@i = $@BG_RM_COUNT - 1; .@i >= 0; .@i--) {
		.@aid = $@BG_RM_AID[.@i];
		.@cid = $@BG_RM_CID[.@i];
		.@name$ = $@BG_RM_NAME$[.@i];
		.@reason$ = "";
		.@reason_code = 0;

		if (!isloggedin(.@aid,.@cid)) {
			.@reason$ = "게임을 종료하여";
			.@reason_code = 1;
		}
		else if (getmapxy(.@map$,.@x,.@y,BL_PC,.@name$) != 0) {
			.@reason$ = "접속 위치를 확인할 수 없어";
			.@reason_code = 4;
		}
		else if (.@map$ != "bat_room") {
			.@reason$ = "전장 대기존을 이탈하여";
			.@reason_code = 2;
		}
		else if (getcharid(1,.@name$) || getcharid(4,.@name$)) {
			.@reason$ = "파티 또는 전장 팀에 가입하여";
			.@reason_code = 3;
		}

		if (.@reason_code == 0)
			continue;

		if (isloggedin(.@aid,.@cid))
			callfunc "F_BG_RM_NotifyAutoCancel", .@aid, .@cid, .@reason_code;

		callfunc "F_BG_RM_RemoveRecruitSlot", .@i;

		mapannounce "bat_room",
			"[전장 랜덤 매칭] " + .@name$ + "님이 " + .@reason$ + " 참가 신청이 취소되었습니다. 현재 " + $@BG_RM_COUNT + "명",
			bc_map, "0xFFCC99";
	}

	if ($@BG_RM_COUNT <= 0) {
		callfunc "F_BG_RM_CancelRecruit", "모든 참가자가 이탈하여 랜덤 매칭 모집이 취소되었습니다.";
		end;
	}

	if ($@BG_RM_STATE == 1)
		initnpctimer;
	end;
}

//============================================================
// 랜덤 매칭 풀사이클 감시
// state 3: 현재 단계 전장 진행/정리 대기
// state 4: 다음 전장 파티장 입장 준비 대기
//============================================================
-	script	NEED_BG_RM_Monitor	-1,{
	end;

OnInit:
	stopnpctimer;
	end;

OnStart:
	stopnpctimer;
	initnpctimer;
	end;

OnTimer1000:
	stopnpctimer;

	if (!$@BG_RM_CYCLE_ACTIVE)
		end;

	.@running = 0;
	if ($@BG_RM_MODE == 1)
		.@running = $@TierraBG1;
	else if ($@BG_RM_MODE == 2)
		.@running = $@FlaviusBG1;
	else if ($@BG_RM_MODE == 3)
		.@running = $@KvM01BG;

	// 다음 전장 준비 대기 중에는 해당 전장이 시작되는 순간만 감지합니다.
	if ($@BG_RM_STATE == 4) {
		if (.@running) {
			$@BG_RM_STATE = 3;
			mapannounce "bat_room", "[전장 랜덤 매칭] " + callfunc("F_BG_RM_ModeName$",$@BG_RM_MODE) + " 입장이 완료되었습니다.", bc_map, "0x99FF99";
		}
		initnpctimer;
		end;
	}

	// KVM 보상 단계에서 온라인 랜덤 매칭 참가자가 모두 대기실로 복귀하면
	// KVM 자체의 60초 정리 타이머를 기다리지 않고 자동 생성 파티만 즉시 해체합니다.
	// 전체 랜덤 매칭 상태/인원 설정/BG 정리는 기존 KVM 종료 타이머가 끝난 뒤 처리됩니다.
	if (
		$@BG_RM_MODE == 3 &&
		$@BG_RM_CYCLE_STAGE == 3 &&
		$@KvM01BG == 3 &&
		$@KvM01BG_Victory > 0 &&
		!$@BG_RM_KVM_PARTY_DESTROYED &&
		!callfunc("F_BG_RM_HasOnlineMemberInMap","bat_c01")
	) {
		callfunc "F_BG_RM_DestroyParties";
		$@BG_RM_KVM_PARTY_DESTROYED = 1;

		mapannounce "bat_room",
			"[전장 랜덤 매칭] KVM 참가자가 모두 대기실로 복귀하여 자동 생성 파티를 해체했습니다.",
			bc_map, "0x99FF99";
	}

	if (.@running) {
		initnpctimer;
		end;
	}

	// 현재 단계의 BG 팀과 보상 처리가 모두 정리된 뒤 다음 단계로 전환합니다.
	callfunc "F_BG_RM_AdvanceCycle";

	if ($@BG_RM_CYCLE_ACTIVE)
		initnpctimer;
	end;
}
