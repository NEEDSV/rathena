// NEED 전장 랜덤 매칭 시스템 - 티에라/플라비우스/KVM 풀사이클 v12.3-global-queue
// - 한 번의 매칭으로 티에라 -> 플라비우스 -> KVM 전체 사이클 진행
// - 최초 티에라는 자동 입장, 이후 전장은 양 팀 파티장의 입장 준비로 진행
// - 티에라/플라비우스 종료 후 파티 유지, KVM 종료 후에만 자동 해체
// - 2026-11-12 통합 모집(NEED_BGUX): 참가 경로는 @전장대기 하나다. 어느 맵에서나 신청한다.
//   대기 8명 -> 이동 전 안내 20초($@BG_RM_PREWARP_SECONDS) -> 전장 대기실 소환 -> 추가 모집 120초(.RecruitSeconds)
//   -> 마감(신청 잠금) -> 팀 편성 -> 티에라 입장(원래 준비 45초). 추가 모집 중 @전장대기 = 같은 모집에 등록 + 즉시 소환.
//   @전장매칭 / @전장참가 = @전장대기 별칭, @전장현황 = @전장대기현황, @전장취소 = @전장대기취소(사이클 중 파티장 종료는 그대로).
//   예전 bat_room 전용 5분 모집(@전장매칭)은 없앴다.
// - 테스트본: 최소 4명 / 최대 4명 (2 vs 2)
// - 동일 직업(Class)을 가능한 한 양 팀으로 분산
// - 모집 중 전장 관리인은 유지하고 홍보 외 메뉴만 잠금
// - 자동 선정 파티장을 기존 입장 준비 시스템에 등록하여 다음 전장 진행
// - 자동 선정 파티장은 기존 수동 파티장 추가 보상 대상에서 제외
// - 모집 중 bat_room 이탈/로그아웃/파티 가입 시 참가 신청 자동 취소
//============================================================

function	script	F_BG_RM_ModeName$	{
	switch (getarg(0)) {
	case 1: return "티에라 협곡";
	case 2: return "플라비우스";
	case 3: return "KVM";
	}
	return "알 수 없음";
}

		switch (getarg(0)) {

		case 2: return "Flavius";
		case 3: return "KVM";
		}
		return "Unknown";
	}
	return callfunc("F_BG_RM_ModeName$", getarg(0));
}

function	script	F_BG_RM_SetWaitingRooms	{
	.@enable = getarg(0);

	// 현재 통합 스크립트는 waitingroom 대신 파티장 입장 준비 방식을 사용합니다.
	// 기존 대기실 NPC가 우연히 활성화되어 있지 않도록 모집 중에는 숨기고,
	// 종료 후에는 각 전장 OnRefresh 정책에 맡깁니다.
	if (!.@enable) {
		disablenpc "기욤 대기실#bat_a01";
		disablenpc "크루아 대기실#bat_a01";
		disablenpc "기욤 대기실#bat_b01";
		disablenpc "크루아 대기실#bat_b01";
		disablenpc "KvM01R_Guillaume";
		disablenpc "KvM01R_Croix";
	}
	else {
		donpcevent "티에라 대기실 갱신#bat_a01::OnRefresh";
		donpcevent "플라비우스 대기실 갱신#bat_b01::OnRefresh";
		donpcevent "KVM 대기실 갱신#bat_c01::OnRefresh";
	}
	return;
}

function	script	F_BG_RM_ManualReadyActive	{
	if ($@BG_READY_TYPE)
		return 1;

	if ($@TierraReady_Guillaume || $@TierraReady_Croix || $@TierraReady_Starting)
		return 1;
	if ($@FlaviusReady_Guillaume || $@FlaviusReady_Croix || $@FlaviusReady_Starting)

	if ($@KvMReady_Guillaume || $@KvMReady_Croix || $@KvMReady_Starting)

function	script	F_BG_RM_PrepareTeamMembers	{
	.@team = getarg(0);
	.@bg_id = getarg(1);
	.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;

	for (.@i = 0; .@i < .@count; .@i++) {
		.@aid = (.@team == 1) ? $@BG_RM_G_AID[.@i] : $@BG_RM_C_AID[.@i];
		.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];

		if (!isloggedin(.@aid,.@cid))
			continue;
		if (!attachrid(.@aid))
			continue;
		if (getcharid(0) != .@cid) {
			detachrid;
			continue;
		}

		if (getcharid(4) == .@bg_id) {
			Bat_Team = .@team;
			percentheal 100,100;
		}
		detachrid;
	}
	return;
}

function	script	F_BG_RM_ResetMembers	{
	for (.@team = 1; .@team <= 2; .@team++) {
		.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;

		for (.@i = 0; .@i < .@count; .@i++) {
			.@aid = (.@team == 1) ? $@BG_RM_G_AID[.@i] : $@BG_RM_C_AID[.@i];
			.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];

			if (!isloggedin(.@aid,.@cid))
				continue;
			if (!attachrid(.@aid))
				continue;
			if (getcharid(0) != .@cid) {
				detachrid;
				continue;
			}

			// BG 관련 상태는 항상 정리합니다.
			Bat_Team = 0;
			if (getcharid(4))
				bg_leave;

			// 실제 전장 내부에 남아 있는 경우에만 대기존으로 이동합니다.
			// 이미 bat_room 또는 다른 맵에 있는 참가자를 154,150으로 다시 끌어오지 않습니다.
			.@map$ = strcharinfo(3);
			if (
				.@map$ == "bat_a01" ||
				.@map$ == "bat_b01" ||
				.@map$ == "bat_c01"
			)
				warp "bat_room",154,150;

			detachrid;
		}
	}
	return;
}

	cleararray $@BG_RM_C_CID[0], 0, MAX_PARTY;

//============================================================
// 전장 글로벌 대기열
// - 실제 랜덤 매칭 참가 배열과 완전히 분리
// - 어느 맵에서든 등록 가능
// - 최소 인원이 모이면 기존 자동매칭 팀 편성으로 넘김
//============================================================
function	script	F_BG_RM_QueueFind	{
	.@aid = getarg(0);
	.@cid = getarg(1);

	for (.@i = 0; .@i < $@BG_RM_QUEUE_COUNT; .@i++) {
		if (
			$@BG_RM_QUEUE_AID[.@i] == .@aid &&
			$@BG_RM_QUEUE_CID[.@i] == .@cid
		)
			return .@i;
	}

	return -1;
}

function	script	F_BG_RM_QueueRemoveSlot	{
	.@slot = getarg(0);

	if (.@slot < 0 || .@slot >= $@BG_RM_QUEUE_COUNT)
		return 0;

	for (.@i = .@slot; .@i < $@BG_RM_QUEUE_COUNT - 1; .@i++) {
		$@BG_RM_QUEUE_AID[.@i] = $@BG_RM_QUEUE_AID[.@i + 1];
		$@BG_RM_QUEUE_CID[.@i] = $@BG_RM_QUEUE_CID[.@i + 1];
		$@BG_RM_QUEUE_CLASS[.@i] = $@BG_RM_QUEUE_CLASS[.@i + 1];
		$@BG_RM_QUEUE_NAME$[.@i] = $@BG_RM_QUEUE_NAME$[.@i + 1];
	}

	if ($@BG_RM_QUEUE_COUNT > 0) {
		.@last = $@BG_RM_QUEUE_COUNT - 1;
		$@BG_RM_QUEUE_AID[.@last] = 0;
		$@BG_RM_QUEUE_CID[.@last] = 0;
		$@BG_RM_QUEUE_CLASS[.@last] = 0;
		$@BG_RM_QUEUE_NAME$[.@last] = "";
		$@BG_RM_QUEUE_COUNT--;
	}

	return 1;
}

function	script	F_BG_RM_QueueClear	{
	cleararray $@BG_RM_QUEUE_AID[0], 0, 48;
	cleararray $@BG_RM_QUEUE_CID[0], 0, 48;
	cleararray $@BG_RM_QUEUE_CLASS[0], 0, 48;
	cleararray $@BG_RM_QUEUE_NAME$[0], "", 48;

	$@BG_RM_QUEUE_COUNT = 0;
	$@BG_RM_QUEUE_LOCK = 0;
	$@BG_RM_QUEUE_READY_TICK = 0;
	$@BG_RM_QUEUE_TOKEN++;
	return;
}

function	script	F_BG_RM_QueueNotifyAll	{
	.@msg$ = getarg(0);
	.@msg_en$ = getarg(1, "");	// 2026-11-12: 영어 문구(없으면 한국어 그대로)

	for (.@i = 0; .@i < $@BG_RM_QUEUE_COUNT; .@i++) {
		if ($@BG_RM_QUEUE_NAME$[.@i] == "")
			continue;
		if (.@msg_en$ != "")
			messagelang $@BG_RM_QUEUE_NAME$[.@i], .@msg$, .@msg_en$;
		else
			message $@BG_RM_QUEUE_NAME$[.@i], .@msg$;
	}

	return;
}



// 모집 명단에서 지정된 슬롯을 제거하고 뒤의 데이터를 앞으로 당깁니다.
function	script	F_BG_RM_RemoveRecruitSlot	{
	.@slot = getarg(0);

	if (.@slot < 0 || .@slot >= $@BG_RM_COUNT)
		return 0;

	for (.@i = .@slot; .@i < $@BG_RM_COUNT - 1; .@i++) {
		$@BG_RM_AID[.@i] = $@BG_RM_AID[.@i + 1];
		$@BG_RM_CID[.@i] = $@BG_RM_CID[.@i + 1];
		$@BG_RM_CLASS[.@i] = $@BG_RM_CLASS[.@i + 1];
		$@BG_RM_NAME$[.@i] = $@BG_RM_NAME$[.@i + 1];
	}

	$@BG_RM_COUNT--;
	$@BG_RM_AID[$@BG_RM_COUNT] = 0;
	$@BG_RM_CID[$@BG_RM_COUNT] = 0;
	$@BG_RM_CLASS[$@BG_RM_COUNT] = 0;
	$@BG_RM_NAME$[$@BG_RM_COUNT] = "";
	return 1;
}

// 최초 티에라 입장 직전에 최종 팀원이 모두 대기실과 자동 생성 파티에 남아 있는지 확인합니다.
function	script	F_BG_RM_ValidateInitialReady	{
	for (.@team = 1; .@team <= 2; .@team++) {
		.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;
		.@party_id = (.@team == 1) ? $@BG_RM_PARTY_G : $@BG_RM_PARTY_C;

		for (.@i = 0; .@i < .@count; .@i++) {
			.@aid = (.@team == 1) ? $@BG_RM_G_AID[.@i] : $@BG_RM_C_AID[.@i];
			.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];
			.@name$ = (.@team == 1) ? $@BG_RM_G_NAME$[.@i] : $@BG_RM_C_NAME$[.@i];

			if (!isloggedin(.@aid,.@cid))
				return 0;

			if (getmapxy(.@map$,.@x,.@y,BL_PC,.@name$) != 0)
				return 0;

			if (.@map$ != "bat_room")
				return 0;

			if (getcharid(1,.@name$) != .@party_id)
				return 0;

			if (getcharid(4,.@name$))
				return 0;
		}
	}

	return 1;
}

function	script	F_BG_RM_FindRecruitSlot	{
	.@aid = getarg(0);
	.@cid = getarg(1);

	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		if ($@BG_RM_AID[.@i] == .@aid && $@BG_RM_CID[.@i] == .@cid)
			return .@i;

}

// 온라인 상태에서 자동 취소된 참가자에게 눈에 띄는 개인 알림을 표시하고
// 같은 모집에서 @전장현황을 확인할 때 취소 사유가 보이도록 기록합니다.
function	script	F_BG_RM_NotifyAutoCancel	{
	.@aid = getarg(0);
	.@cid = getarg(1);
	.@reason = getarg(2);
	.@reason$ = callfunc("F_BG_RM_CancelReason$", .@reason);

	if (!isloggedin(.@aid,.@cid))
		return 0;
	if (!attachrid(.@aid))
		return 0;
	if (getcharid(0) != .@cid) {
		detachrid;
		return 0;
	}

	BG_RM_LAST_CANCEL_REASON = .@reason;
	BG_RM_LAST_CANCEL_TOKEN = $@BG_RM_TOKEN;
	BG_RM_LAST_CANCEL_TICK = gettimetick(2);

	announce needtr("[전장] 랜덤 매칭 참가 신청이 자동 취소되었습니다.", "[Battleground] Your random match application was cancelled automatically."), bc_self, "0xFF5555";
	dispbottom needtr("[전장] 취소 사유: ", "[Battleground] Cancellation reason: ") + callfunc("F_BG_RM_CancelReasonLang$", .@reason, getneedlang());
	dispbottom needtr("[전장] 모집이 아직 진행 중이면 어디서든 @전장대기로 다시 참가할 수 있습니다.","[Battleground] If recruitment is still open you can rejoin from anywhere with @bgqueue.");
	detachrid;
	return 1;
}

// 모집 중인 참가자에게 개인 상태 유지 알림을 보냅니다.
function	script	F_BG_RM_NotifyApplicants	{
	.@remain = getarg(0);

	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		.@aid = $@BG_RM_AID[.@i];
		.@cid = $@BG_RM_CID[.@i];
		.@name$ = $@BG_RM_NAME$[.@i];

		if (!isloggedin(.@aid,.@cid))
			continue;

		message .@name$,
			needtr("[전장] 참가 유지 중 | 남은 시간 ","[Battleground] In queue | Time remaining ") + .@remain
			+ needtr(" | 현재 "," | Currently ") + $@BG_RM_COUNT + needtr("명 | 상태 확인: @전장현황"," | Check status: @bgstatus");
	}
	return;
}

//============================================================
// 전장 대기존 미신청 인원 안내
// - 모집 중 30초마다 전장 관리인이 NPC 채팅으로 안내
// - 이름이 많으면 6명씩 순환하여 표시
//============================================================
function	script	F_BG_RM_AnnounceMissingApplicants	{
	if ($@BG_RM_STATE != 1)
		return;

	.@unit_count = getmapunits(BL_PC,"bat_room",.@unit_aid[0]);
	.@missing_count = 0;

	for (.@i = 0; .@i < .@unit_count; .@i++) {
		.@aid = .@unit_aid[.@i];
		.@cid = convertpcinfo(.@aid,CPC_CHAR);
		.@name$ = convertpcinfo(.@aid,CPC_NAME);

		if (.@aid <= 0 || .@cid <= 0 || .@name$ == "")
			continue;

		// 운영자/테스트 캐릭터 미신청 명단 제외
		.@excluded = 0;
		for (.@j = 0; .@j < $@BG_RM_MISSING_EXCLUDE_COUNT; .@j++) {
			if (.@name$ == $@BG_RM_MISSING_EXCLUDE_NAME$[.@j]) {
				.@excluded = 1;
				break;
			}
		}
		if (.@excluded)
			continue;

		// 참가 레벨 미달 캐릭터에게는 미신청 홍보를 하지 않습니다.
		if ($@BG_RM_MIN_JOIN_LEVEL > 0 && readparam(BaseLevel,.@cid) < $@BG_RM_MIN_JOIN_LEVEL)
			continue;

		if (callfunc("F_BG_RM_FindRecruitSlot",.@aid,.@cid) >= 0)
			continue;

		.@missing_name$[.@missing_count] = .@name$;
		.@missing_count++;
	}

	if (.@missing_count <= 0) {
		$@BG_RM_MISSING_CURSOR = 0;
		return;
	}

	.@start = $@BG_RM_MISSING_CURSOR;
	if (.@start < 0 || .@start >= .@missing_count)
		.@start = 0;

	.@show_count = 6;
	if (.@missing_count < .@show_count)
		.@show_count = .@missing_count;

	.@list$ = "";
	for (.@i = 0; .@i < .@show_count; .@i++) {
		.@index = (.@start + .@i) % .@missing_count;

		if (.@list$ != "")
			.@list$ += ", ";

		.@list$ += .@missing_name$[.@index];
	}

	.@message$ = "랜덤 매칭 미신청 " + .@missing_count + "명: " + .@list$;
	.@message_en$ = "Random match non-applicants " + .@missing_count + " players: " + .@list$;

	if (.@missing_count > .@show_count) {
		.@message$ += " 외 " + (.@missing_count - .@show_count) + "명";
		.@message_en$ += " and " + (.@missing_count - .@show_count) + " players";
	}

	.@message$ += " / 참가: @전장대기";

//============================================================
// 티에라 내부 준비 구역 입장 직후 양 팀 멤버/직업 구성 공지
//
// 한 줄이 너무 길어지는 것을 피하기 위해 닉네임은 6명 단위로 분할합니다.
// 양 팀 모두 같은 정보를 볼 수 있도록 bat_a01 맵 전체에 출력합니다.
//============================================================
function	script	F_BG_RM_AnnounceTierraTeamInfo	{
	for (.@team = 1; .@team <= 2; .@team++) {
		.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;
		.@team_name$ = (.@team == 1) ? "기욤" : "크루아";
		.@team_name_en$ = (.@team == 1) ? "Guillaume" : "Croix";
		.@color$ = (.@team == 1) ? "0x66CCFF" : "0xFF7777";

		cleararray .@job_count[0], 0, 15;
		.@other_count = 0;

		// 직업 구성 집계
		for (.@i = 0; .@i < .@count; .@i++) {
			.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];
			.@class = readparam(Class,.@cid);
			.@family = callfunc("F_BG_RM_NoticeJobFamily", .@class);

			if (.@family > 0 && .@family <= 14)
				.@job_count[.@family]++;
			else
				.@other_count++;
		}

		.@composition$ = "";
		.@composition_en$ = "";
		for (.@family = 1; .@family <= 14; .@family++) {
			if (.@job_count[.@family] <= 0)
				continue;

			.@job_name$ = callfunc("F_BG_RM_NoticeJobName$", .@family);
			.@job_name_en$ = callfunc("F_BG_RM_NoticeJobNameLang$", .@family, NEED_LANG_EN);

			if (.@composition$ != "")
				.@composition$ += " / ";

			.@composition$ += .@job_name$ + "-" + .@job_count[.@family];

			if (.@composition_en$ != "")
				.@composition_en$ += " / ";

			.@composition_en$ += .@job_name_en$ + "-" + .@job_count[.@family];
		}

		if (.@other_count > 0) {
			if (.@composition_en$ != "")
				.@composition_en$ += " / ";
			.@composition_en$ += "Other-" + .@other_count;
			if (.@composition$ != "")
				.@composition$ += " / ";
			.@composition$ += "기타-" + .@other_count;
		}

		if (.@composition_en$ == "")
			.@composition_en$ = "No composition data";

		if (.@composition$ == "")
			.@composition$ = "구성 정보 없음";

		mapannouncelang "bat_a01",
			"[랜덤 매칭][" + .@team_name$ + " 직업 구성] " + .@composition$, "[Random Match][" + .@team_name_en$ + " Job Composition] " + .@composition_en$,
			bc_map, .@color$;

		// 닉네임은 6명씩 나눠서 출력합니다.
		.@page_count = (.@count + 5) / 6;

		for (.@page = 0; .@page < .@page_count; .@page++) {
			.@member_list$ = "";
			.@start = .@page * 6;
			.@end = .@start + 6;
			if (.@end > .@count)
				.@end = .@count;

			for (.@i = .@start; .@i < .@end; .@i++) {
				.@name$ = (.@team == 1) ? $@BG_RM_G_NAME$[.@i] : $@BG_RM_C_NAME$[.@i];

				if (.@member_list$ != "")
					.@member_list$ += ", ";

				.@member_list$ += .@name$;
			}

			mapannouncelang "bat_a01",
				"[랜덤 매칭][" + .@team_name$ + " 멤버 "
				+ (.@page + 1) + "/" + .@page_count + "] " + .@member_list$,
				"[Random Match][" + .@team_name_en$ + " Members "
				+ (.@page + 1) + "/" + .@page_count + "] " + .@member_list$,
				bc_map, .@color$;
		}
	}

	return;
}


//============================================================
// 랜덤 매칭 단독 민스트럴 현행(버프형) 모드 강제
//
// @모드전환의 캐릭터 저장 키:
//   NEED_SOLO_PERFORMANCE_MODE
//   0 = 장판형(Classic)
//   1 = 현행 버프형(Modern)
//
// 랜덤 매칭 전체 참가자 중 실제 민스트럴 계열
// (JOB_MINSTREL / JOB_MINSTREL_T)이 정확히 1명일 때만 사용합니다.
// 원래 설정값은 NEED_BG_RM_PERFORMANCE_RESTORE에 보관하고
// 풀사이클 종료/취소 시 복구합니다.
//============================================================
function	script	F_BG_RM_EnsureForcedPerformanceMode	{
	if ($@BG_RM_FORCED_PERFORMANCE_AID <= 0 || $@BG_RM_FORCED_PERFORMANCE_CID <= 0)
		return 0;

	if (!isloggedin($@BG_RM_FORCED_PERFORMANCE_AID,$@BG_RM_FORCED_PERFORMANCE_CID))
		return 0;

	.@caller_aid = getattachedrid();

	if (!attachrid($@BG_RM_FORCED_PERFORMANCE_AID)) {
		if (.@caller_aid)
			attachrid .@caller_aid;
		return 0;
	}

	if (getcharid(0) != $@BG_RM_FORCED_PERFORMANCE_CID) {
		if (.@caller_aid)
			attachrid .@caller_aid;
		else
			detachrid;
		return 0;
	}

	// +1 형태로 저장하여 원래 값이 0이어도 복구 예약 여부를 구분합니다.
	if (!NEED_BG_RM_PERFORMANCE_RESTORE)
		NEED_BG_RM_PERFORMANCE_RESTORE = NEED_SOLO_PERFORMANCE_MODE + 1;

	NEED_SOLO_PERFORMANCE_MODE = 1;

	if (.@caller_aid)
		attachrid .@caller_aid;
	else
		detachrid;

	return 1;
}

function	script	F_BG_RM_RestoreForcedPerformanceMode	{
	.@aid = $@BG_RM_FORCED_PERFORMANCE_AID;
	.@cid = $@BG_RM_FORCED_PERFORMANCE_CID;
	$@BG_RM_FORCED_PERFORMANCE_AID = 0;
	$@BG_RM_FORCED_PERFORMANCE_CID = 0;

	if (.@aid <= 0 || .@cid <= 0)
		return;

	if (!isloggedin(.@aid,.@cid))
		return;

	.@caller_aid = getattachedrid();

	if (!attachrid(.@aid)) {
		if (.@caller_aid)
			attachrid .@caller_aid;
		return;
	}

	if (getcharid(0) == .@cid && NEED_BG_RM_PERFORMANCE_RESTORE) {
		NEED_SOLO_PERFORMANCE_MODE = NEED_BG_RM_PERFORMANCE_RESTORE - 1;
		NEED_BG_RM_PERFORMANCE_RESTORE = 0;
		dispbottom needtr("[전장] 랜덤 매칭이 종료되어 @모드전환 설정이 이전 상태로 복구되었습니다.","[Battleground] The random match has ended, so your @performancemode setting has been restored.");
	}

	if (.@caller_aid)
		attachrid .@caller_aid;
	else
		detachrid;

	return;
}

// KVM 보상 수령 또는 자동 이동으로 모든 온라인 참가자가 bat_c01을 벗어났는지 확인합니다.
// 로그아웃한 참가자는 대기 대상으로 잡지 않습니다.
function	script	F_BG_RM_HasOnlineMemberInMap	{
	.@target_map$ = getarg(0);

	for (.@team = 1; .@team <= 2; .@team++) {
		.@count = (.@team == 1) ? $@BG_RM_G_COUNT : $@BG_RM_C_COUNT;

		for (.@i = 0; .@i < .@count; .@i++) {
			.@aid = (.@team == 1) ? $@BG_RM_G_AID[.@i] : $@BG_RM_C_AID[.@i];
			.@cid = (.@team == 1) ? $@BG_RM_G_CID[.@i] : $@BG_RM_C_CID[.@i];
			.@name$ = (.@team == 1) ? $@BG_RM_G_NAME$[.@i] : $@BG_RM_C_NAME$[.@i];

			if (!isloggedin(.@aid,.@cid))
				continue;

			if (getmapxy(.@map$,.@x,.@y,BL_PC,.@name$) != 0)
				continue;

			if (.@map$ == .@target_map$)
				return 1;
		}
	}

	return 0;
}

OnQueueLaunch:
	if ($@BG_RM_QUEUE_COUNT < .MinPlayers) {

		end;

		$@BG_RM_ACTIVE ||
		$@BG_RM_CYCLE_ACTIVE ||
		$@TierraBG1 ||
		$@FlaviusBG1 ||
		$@KvM01BG ||

		$@BG_RM_QUEUE_LOCK = 0;
		$@BG_RM_QUEUE_READY_TICK = 0;
		end;

			detachrid;
			continue;

			detachrid;
			break;

		"[전장 대기열] 참가 취소: @전장대기취소 / 현황: @전장대기현황", "[BG Queue] Leave: @bgqueuecancel / Status: @bgqueuestatus",
		bc_map, "0x99DDFF";
	announcelang "[전장 대기열] 기존 전장 추가 모집 시작: 현재 " + $@BG_RM_COUNT + "/" + .MaxPlayers + "명, " + .RecruitSeconds + "초 후 마감. 마을에서도 @전장대기 로 바로 합류할 수 있습니다.",
		"[BG Queue] Classic BG extra recruitment is open: " + $@BG_RM_COUNT + "/" + .MaxPlayers + " players, closes in " + .RecruitSeconds + " seconds. Join from town with @bgqueue.",
		bc_all, "0x66FF99";
	end;

//------------------------------------------------------------
// 2026-11-12 NEED_BGUX 통합 모집: @전장매칭 / @전장참가 는 @전장대기 와 같은 참가 흐름이다(별칭).
// @전장현황 = @전장대기현황, @전장취소 = @전장대기취소. 예전 bat_room 전용 5분 모집(.RecruitSeconds 300,
// OnTimer60000~OnTimer300000 공지·마감)은 없앴다. 대화 중 doevent 는 엔진이 이 스크립트가 끝난 뒤 실행한다.
//------------------------------------------------------------
OnCommandStart:
OnCommandJoin:
	doevent "NEED_BG_RM_GlobalQueue::OnCommandQueue";
	end;

OnCommandStatus:
	doevent "NEED_BG_RM_GlobalQueue::OnCommandQueueStatus";
	end;

OnCommandCancel:
	// 전장 사이의 준비 대기 단계에서는 자동 선정 파티장만 전체 사이클을 종료할 수 있습니다(기존 그대로, bat_room 에서).
	if ($@BG_RM_CYCLE_ACTIVE && $@BG_RM_STATE == 4 && strcharinfo(3) == "bat_room" && (getcharid(0) == $BG_GUILLAUME_LEADER_CID || getcharid(0) == $BG_CROIX_LEADER_CID)) {
		goto L_LeaderEndCycle;
	}
	doevent "NEED_BG_RM_GlobalQueue::OnCommandQueueCancel";
	end;

L_LeaderEndCycle:
	if ($@BG_RM_CYCLE_ACTIVE && $@BG_RM_STATE == 4) {
		if (getcharid(0) != $BG_GUILLAUME_LEADER_CID && getcharid(0) != $BG_CROIX_LEADER_CID) {
			dispbottom needtr("[전장] 랜덤 매칭 사이클 종료는 자동 선정된 양 팀 파티장만 가능합니다.","[Battleground] Only the two automatically chosen party leaders can end the cycle.");
			end;
		}

		mes needtr("[전장 랜덤 매칭]","[Battleground Random Match]");
		mes needtr("현재 유지 중인 랜덤 매칭 파티와","Do you want to disband the random match party");
		mes needtr("남은 전장 사이클을 모두 종료하시겠습니까?","and end the rest of the battleground cycle?");
		next;
		if (select(needtr("사이클 종료:취소","End the cycle:Cancel")) == 2)
			close;

		close2;
		callfunc "F_BG_RM_Finish", strcharinfo(0) + " 파티장이 랜덤 매칭 사이클을 종료했습니다.",
			strcharinfo(0) + " (party leader) ended the random match cycle.";
		end;
	}

	end;

OnFinalize:
	if ($@BG_RM_STATE != 1)
		end;

	stopnpctimer "NEED_BG_RM_RecruitMonitor";
	$@BG_RM_MISSING_CURSOR = 0;
	$@BG_RM_STATE = 2;
	.@finalize_token = $@BG_RM_TOKEN;
	$@BG_RM_FINALIZE_WATCH_TOKEN = .@finalize_token;
	donpcevent "NEED_BG_RM_FinalizeWatch::OnStart";

	mapannouncelang "bat_room",
		"[전장 랜덤 매칭] 모집이 마감되었습니다. 참가자 확인 및 팀 편성을 시작합니다.", "[Battleground Random Match] Recruitment has closed. Checking participants and starting team formation.",
		bc_map, "0xFFCC66";
	debugmes "[BG RM] finalize begin count=" + $@BG_RM_COUNT;

	// 모집 중 이미 열려 있던 관리인 대화 등으로 수동 준비 상태가 생겼는지 재확인
	if ($@TierraBG1 || $@FlaviusBG1 || $@KvM01BG || callfunc("F_BG_RM_ManualReadyActive") || $BG_GUILLAUME_PARTY_ID || $BG_CROIX_PARTY_ID) {
		callfunc "F_BG_RM_CancelRecruit", "수동 파티장 전장 준비와 충돌하여 매칭이 취소되었습니다.",
			"The match was cancelled because it conflicted with manual party leader Battleground preparation.";
		end;
	}

	.@count = 0;
	.@max_level_count = 0;
	.@minstrel_total_count = 0;
	.@single_minstrel_aid = 0;
	.@single_minstrel_cid = 0;

	// 접속, 위치, 파티 상태를 최종 재검증
	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		.@aid = $@BG_RM_AID[.@i];
		.@cid = $@BG_RM_CID[.@i];
		.@name$ = $@BG_RM_NAME$[.@i];

		if (!isloggedin(.@aid,.@cid))
			continue;
		if (getmapxy(.@map$,.@x,.@y,BL_PC,.@name$) != 0)
			continue;
		if (.@map$ != "bat_room") {
			messagelang .@name$, "[전장] 전장 대기존을 이탈하여 랜덤 매칭에서 제외되었습니다.", "[Battleground] You left the Battleground waiting zone and were removed from the random match.";
			continue;
		}
		if (getcharid(1,.@name$) || getcharid(4,.@name$)) {
			messagelang .@name$, "[전장] 파티 또는 전장 팀에 가입하여 랜덤 매칭에서 제외되었습니다.", "[Battleground] You joined a party or a battleground team and were removed from the random match.";
			continue;
		}

		.@base_level = readparam(BaseLevel,.@cid);
		if (.@base_level < .MinJoinLevel) {
			messagelang .@name$, "[전장] BaseLevel " + .MinJoinLevel + " 미만이라 랜덤 매칭 최종 참가자에서 제외되었습니다.", "[Battleground] Your BaseLevel is below " + .MinJoinLevel + ", so you were removed from the final random match participants.";
			continue;
		}

		.@aid_list[.@count] = .@aid;
		.@cid_list[.@count] = .@cid;
		.@name_list$[.@count] = .@name$;
		.@class_list[.@count] = readparam(Class,.@cid);
		.@level_list[.@count] = .@base_level;

		if (.@level_list[.@count] == .MaxBaseLevel)
			.@max_level_count++;

		// "민스트럴만" 적용: 바드/클로운은 자동 현행 모드 대상에서 제외합니다.
		if (
			.@class_list[.@count] == JOB_MINSTREL ||
			.@class_list[.@count] == JOB_MINSTREL_T
		) {
			.@minstrel_total_count++;
			.@single_minstrel_aid = .@aid;
			.@single_minstrel_cid = .@cid;
		}

		.@count++;
	}

	if (.@count < .MinPlayers) {
		callfunc "F_BG_RM_CancelRecruit", "최종 참가자가 " + .MinPlayers + "명 미만이어서 매칭이 취소되었습니다. (" + .@count + "명)",
			"The final participant count was under " + .MinPlayers + ", so the match was cancelled. (" + .@count + ")";
		end;
	}

	// 양 팀 파티장은 반드시 현재 만렙(175)이어야 하므로 최소 2명이 필요합니다.
	if (.@max_level_count < 2) {
		callfunc "F_BG_RM_CancelRecruit",
			"양 팀 파티장 선정을 위해 175레벨 참가자가 최소 2명 필요합니다. (현재 " + .@max_level_count + "명)",
			"Selecting party leaders for both teams requires at least 2 Level 175 participants. (currently " + .@max_level_count + ")";
		end;
	}

	// 전체 참가자 중 실제 민스트럴이 정확히 1명인 경우에만
	// 해당 캐릭터의 @모드전환 저장값을 현행(Modern=1)으로 강제합니다.
	if (.@minstrel_total_count == 1) {
		$@BG_RM_FORCED_PERFORMANCE_AID = .@single_minstrel_aid;
		$@BG_RM_FORCED_PERFORMANCE_CID = .@single_minstrel_cid;

		if (!callfunc("F_BG_RM_EnsureForcedPerformanceMode")) {
			callfunc "F_BG_RM_CancelRecruit", "단독 민스트럴의 현행 모드 적용에 실패하여 매칭이 취소되었습니다.",
				"Applying the active mode for the solo Minstrel failed, so the match was cancelled.";
			end;
		}

		debugmes "[BG RM] single minstrel modern mode forced"
			+ " cid=" + .@single_minstrel_cid;
	}

	debugmes "[BG RM] finalize validated eligible=" + .@count
		+ " max=" + .MaxPlayers
		+ " party_max=" + .MaxPartyMembers;

	// 참가자가 많을 경우 아래의 다중 for/while 배정 로직이
	// rAthena의 1회 실행 GOTO/command 안전 한도에 먼저 도달할 수 있습니다.
	// 모든 반복문의 상한은 참가 인원(.MaxPlayers)으로 제한되어 있으므로
	// 팀 편성 구간에서만 loop guard를 일시 해제합니다.
	freeloop 1;
	debugmes "[BG RM] matching loop guard disabled count=" + .@count;

	// 전체 참가 순서 셔플
	for (.@i = .@count - 1; .@i > 0; .@i--) {
		.@j = rand(.@i + 1);

		.@tmp = .@aid_list[.@i];
		.@aid_list[.@i] = .@aid_list[.@j];
		.@aid_list[.@j] = .@tmp;

		.@tmp = .@cid_list[.@i];
		.@cid_list[.@i] = .@cid_list[.@j];
		.@cid_list[.@j] = .@tmp;

		.@tmp$ = .@name_list$[.@i];
		.@name_list$[.@i] = .@name_list$[.@j];
		.@name_list$[.@j] = .@tmp$;

		.@tmp = .@class_list[.@i];
		.@class_list[.@i] = .@class_list[.@j];
		.@class_list[.@j] = .@tmp;

		.@tmp = .@level_list[.@i];
		.@level_list[.@i] = .@level_list[.@j];
		.@level_list[.@j] = .@tmp;
	}

	// 홀수 인원일 때 한 명이 더 많은 팀은 무작위
	if ((.@count % 2) == 0) {
		.@cap_g = .@count / 2;
		.@cap_c = .@count / 2;
	}
	else if (rand(2) == 0) {
		.@cap_g = (.@count + 1) / 2;
		.@cap_c = .@count / 2;
	}
	else {
		.@cap_g = .@count / 2;
		.@cap_c = (.@count + 1) / 2;
	}

	cleararray .@used[0], 0, .@count;
	cleararray .@is_low[0], 0, .@count;
	.@count_g = 0;
	.@count_c = 0;
	.@single_count = 0;
	.@low_count = 0;

	// 175 미만 참가자는 일반 직업 배정에서 제외하고 마지막 단계에 따로 배정합니다.
	for (.@i = 0; .@i < .@count; .@i++) {
		if (.@level_list[.@i] < .MaxBaseLevel) {
			.@is_low[.@i] = 1;
			.@low_index[.@low_count] = .@i;
			.@low_count++;
		}
	}

	// 낮은 레벨 인원 자체가 양 팀에 최대한 균등하게 들어가도록
	// 최종 팀 정원에서 저레벨 전용 슬롯을 먼저 예약합니다.
	.@low_cap_g = .@low_count / 2;
	.@low_cap_c = .@low_count / 2;

	if ((.@low_count % 2) == 1) {
		if (.@cap_g > .@cap_c)
			.@low_cap_g++;
		else if (.@cap_c > .@cap_g)
			.@low_cap_c++;
		else if (rand(2) == 0)
			.@low_cap_g++;
		else
			.@low_cap_c++;
	}

	.@high_cap_g = .@cap_g - .@low_cap_g;
	.@high_cap_c = .@cap_c - .@low_cap_c;

	if (.@high_cap_g < 1 || .@high_cap_c < 1) {
		callfunc "F_BG_RM_CancelRecruit",
			"175레벨 파티장을 양 팀에 배치할 수 없는 인원 구성이라 매칭이 취소되었습니다.",
			"The roster could not place a Level 175 party leader on both teams, so the match was cancelled.";
		end;
	}

	debugmes "[BG RM] level slots"
		+ " max175=" + .@max_level_count
		+ " low=" + .@low_count
		+ " high_cap_G=" + .@high_cap_g
		+ " high_cap_C=" + .@high_cap_c
		+ " low_cap_G=" + .@low_cap_g
		+ " low_cap_C=" + .@low_cap_c;

	//========================================================
	// 우선 분리 규칙 v2
	//
	// 적용 쌍:
	//  rule 0: 민스트럴(6) ↔ 원더러(7)
	//  rule 1: 로열가드(2) ↔ 소서러(4)
	//  rule 2: 아크비숍(12) ↔ 제네릭(9)
	//  rule 3: 소울링커(14) ↔ 워록(3)
	//
	// 핵심 원칙:
	// 1) 각 직업군 자체의 인원이 2명 이상이면 먼저 양 팀에 최대한 균등 분배
	//    - 2명 -> 1 / 1
	//    - 3명 -> 2 / 1
	//    - 4명 -> 2 / 2
	// 2) 짝을 이루는 두 직업군이 모두 홀수라면
	//    각 직업군의 "한 명 더 많은 팀"은 반드시 서로 반대로 배치
	// 3) 175 미만은 실제 팀 배열에 즉시 넣지 않고 팀만 예약한 뒤
	//    기존 정책대로 저레벨 마지막 단계에서 실제 배치
	//
	// 예: 아크비숍 2 + 제네릭 1
	//     -> 아크비숍 1/1 선분배
	//     -> 제네릭 1명은 정원에 맞는 한쪽 팀
	//
	// 예: 아크비숍 3 + 제네릭 1
	//     -> 아크비숍의 추가 1명과 제네릭 1명은 서로 반대 팀
	//========================================================
	cleararray .@forced_team[0], 0, .@count;
	.@forced_high_g = 0;
	.@forced_high_c = 0;
	.@forced_low_g = 0;
	.@forced_low_c = 0;

	.@split_rule_count = 4;

	.@split_family_a[0] = 6;
	.@split_family_b[0] = 7;

	.@split_family_a[1] = 2;
	.@split_family_b[1] = 4;

	.@split_family_a[2] = 12;
	.@split_family_b[2] = 9;

	.@split_family_a[3] = 14;
	.@split_family_b[3] = 3;

	// 수라 홀수 + 제네릭 단독 배치 규칙 계산용
	.@total_sura = 0;
	.@total_genetic = 0;
	.@single_genetic_index = -1;

	for (.@i = 0; .@i < .@count; .@i++) {
		.@family = callfunc("F_BG_RM_NoticeJobFamily", .@class_list[.@i]);

		if (.@family == 13)
			.@total_sura++;

		if (.@family == 9) {
			.@total_genetic++;
			.@single_genetic_index = .@i;
		}
	}

	for (.@rule = 0; .@rule < .@split_rule_count; .@rule++) {
		.@family_a = .@split_family_a[.@rule];
		.@family_b = .@split_family_b[.@rule];
		.@count_a = 0;
		.@count_b = 0;

		// 해당 우선 분리 직업군의 전체 참가자 인덱스를 모읍니다.
		for (.@i = 0; .@i < .@count; .@i++) {
			.@family = callfunc("F_BG_RM_NoticeJobFamily", .@class_list[.@i]);

			if (.@family == .@family_a) {
				.@rule_a_index[.@count_a] = .@i;
				.@count_a++;
			}
			else if (.@family == .@family_b) {
				.@rule_b_index[.@count_b] = .@i;
				.@count_b++;
			}
		}

		if (.@count_a <= 0 && .@count_b <= 0)
			continue;

		// 같은 직업군 안에서 특정 캐릭터가 항상 같은 팀을 받지 않도록 셔플합니다.
		for (.@i = .@count_a - 1; .@i > 0; .@i--) {
			.@j = rand(.@i + 1);
			.@tmp = .@rule_a_index[.@i];
			.@rule_a_index[.@i] = .@rule_a_index[.@j];
			.@rule_a_index[.@j] = .@tmp;
		}

		for (.@i = .@count_b - 1; .@i > 0; .@i--) {
			.@j = rand(.@i + 1);
			.@tmp = .@rule_b_index[.@i];
			.@rule_b_index[.@i] = .@rule_b_index[.@j];
			.@rule_b_index[.@j] = .@tmp;
		}

		// A/B 각각 어느 팀부터 교차 배정할지 4가지 조합을 검사합니다.
		// 두 직업군 모두 홀수면 추가 1명이 같은 팀에 몰리지 않도록
		// 시작 팀이 반드시 서로 반대인 조합만 허용합니다.
		.@selected = 0;
		.@candidate_offset = rand(4);

		for (.@try = 0; .@try < 4; .@try++) {
			.@candidate = (.@candidate_offset + .@try) % 4;
			.@start_a = .@candidate % 2;
			.@start_b = (.@candidate / 2) % 2;

			if (
				(.@count_a % 2) == 1 &&
				(.@count_b % 2) == 1 &&
				.@start_a == .@start_b
			)
				continue;

			.@need_hg = 0;
			.@need_hc = 0;
			.@need_lg = 0;
			.@need_lc = 0;

			for (.@i = 0; .@i < .@count_a; .@i++) {
				.@index = .@rule_a_index[.@i];
				.@team = (.@start_a + .@i) % 2;

				if (.@is_low[.@index]) {
					if (.@team == 0) .@need_lg++;
					else .@need_lc++;
				}
				else {
					if (.@team == 0) .@need_hg++;
					else .@need_hc++;
				}
			}

			for (.@i = 0; .@i < .@count_b; .@i++) {
				.@index = .@rule_b_index[.@i];
				.@team = (.@start_b + .@i) % 2;

				if (.@is_low[.@index]) {
					if (.@team == 0) .@need_lg++;
					else .@need_lc++;
				}
				else {
					if (.@team == 0) .@need_hg++;
					else .@need_hc++;
				}
			}

			if (
				.@forced_high_g + .@need_hg > .@high_cap_g ||
				.@forced_high_c + .@need_hc > .@high_cap_c ||
				.@forced_low_g + .@need_lg > .@low_cap_g ||
				.@forced_low_c + .@need_lc > .@low_cap_c
			)
				continue;

			.@selected = 1;
			break;
		}

		if (!.@selected) {
			callfunc "F_BG_RM_CancelRecruit",
				"직업 우선 분리 규칙을 만족할 수 없는 팀 정원이라 매칭이 취소되었습니다. (규칙 " + .@rule + ")",
				"The team capacity could not satisfy the job priority split rule, so the match was cancelled. (rule " + .@rule + ")";
			end;
		}

		// 선택된 방향대로 해당 직업군 전원을 교차 예약합니다.
		for (.@i = 0; .@i < .@count_a; .@i++) {
			.@index = .@rule_a_index[.@i];
			.@team = (.@start_a + .@i) % 2;
			.@forced_team[.@index] = .@team + 1;

			if (.@is_low[.@index]) {
				if (.@team == 0) .@forced_low_g++;
				else .@forced_low_c++;
			}
			else {
				if (.@team == 0) .@forced_high_g++;
				else .@forced_high_c++;
			}
		}

		for (.@i = 0; .@i < .@count_b; .@i++) {
			.@index = .@rule_b_index[.@i];
			.@team = (.@start_b + .@i) % 2;
			.@forced_team[.@index] = .@team + 1;

			if (.@is_low[.@index]) {
				if (.@team == 0) .@forced_low_g++;
				else .@forced_low_c++;
			}
			else {
				if (.@team == 0) .@forced_high_g++;
				else .@forced_high_c++;
			}
		}

		debugmes "[BG RM] priority family split"
			+ " rule=" + .@rule
			+ " familyA=" + .@family_a + " countA=" + .@count_a
			+ " familyB=" + .@family_b + " countB=" + .@count_b
			+ " startA=" + .@start_a
			+ " startB=" + .@start_b;
	}

	// -------------------------------------------------------
	// 수라 홀수 + 제네릭 1명 특수 규칙
	//
	// 제네릭이 정확히 한 명이고 수라 전체 인원이 홀수라면,
	// 수라의 "한 명 더 많은 팀"을 제네릭이 없는 반대 팀으로 지정합니다.
	//
	// 실제 수라 배정은 기존 동일직업 분산 단계에서 처리하며,
	// 여기서는 어느 팀이 수라 추가 1명을 받을지만 저장합니다.
	// -------------------------------------------------------
	.@sura_extra_team = -1;

	// 구형/커스텀 포크에서는 && 조건이 C/C++처럼 단락 평가되지 않을 수 있습니다.
	// 따라서 제네릭이 없어서 index=-1인 상태에서
	// .@forced_team[-1]을 읽지 않도록 조건을 단계별로 분리합니다.
	if (.@total_genetic == 1) {
		if ((.@total_sura % 2) == 1) {
			if (.@single_genetic_index >= 0) {
				if (.@forced_team[.@single_genetic_index] > 0) {
					// forced_team: 1=G, 2=C
					// sura_extra_team: 0=G, 1=C
					if (.@forced_team[.@single_genetic_index] == 1)
						.@sura_extra_team = 1;
					else
						.@sura_extra_team = 0;

					debugmes "[BG RM] sura odd preference"
						+ " genetic_cid=" + .@cid_list[.@single_genetic_index]
						+ " genetic_team=" + ((.@forced_team[.@single_genetic_index] == 1) ? "G" : "C")
						+ " sura_extra_team=" + ((.@sura_extra_team == 0) ? "G" : "C")
						+ " sura_count=" + .@total_sura;
				}
			}
		}
	}

	// 175레벨 우선 분리 대상은 지금 실제 팀에 넣습니다.
	// 175 미만은 forced_team 예약만 유지하고 마지막 단계에서 처리합니다.
	for (.@i = 0; .@i < .@count; .@i++) {
		if (!.@forced_team[.@i] || .@is_low[.@i])
			continue;

		if (.@forced_team[.@i] == 1) {
			.@aid_g[.@count_g] = .@aid_list[.@i];
			.@cid_g[.@count_g] = .@cid_list[.@i];
			.@name_g$[.@count_g] = .@name_list$[.@i];
			.@count_g++;
		}
		else {
			.@aid_c[.@count_c] = .@aid_list[.@i];
			.@cid_c[.@count_c] = .@cid_list[.@i];
			.@name_c$[.@count_c] = .@name_list$[.@i];
			.@count_c++;
		}

		.@used[.@i] = 1;
	}

	debugmes "[BG RM] priority high applied"
		+ " G=" + .@count_g
		+ " C=" + .@count_c
		+ " forced_low_G=" + .@forced_low_g
		+ " forced_low_C=" + .@forced_low_c
		+ " genetic_count=" + .@total_genetic
		+ " sura_count=" + .@total_sura;

	// 1단계: 동일 직업이 2명 이상인 그룹을 기욤/크루아에 교차 배정
	// 1명뿐인 직업은 즉시 배정하지 않고 단독 직업 목록에 보관합니다.
	for (.@i = 0; .@i < .@count; .@i++) {
		if (.@used[.@i] || .@is_low[.@i])
			continue;

		.@group_count = 0;
		.@target_class = .@class_list[.@i];

		for (.@j = 0; .@j < .@count; .@j++) {
			if (!.@used[.@j] && !.@is_low[.@j] && .@class_list[.@j] == .@target_class) {
				.@group_index[.@group_count] = .@j;
				.@used[.@j] = 1;
				.@group_count++;
			}
		}

		// 동일 직업 그룹 내부 순서 무작위화
		for (.@j = .@group_count - 1; .@j > 0; .@j--) {
			.@k = rand(.@j + 1);
			.@tmp = .@group_index[.@j];
			.@group_index[.@j] = .@group_index[.@k];
			.@group_index[.@k] = .@tmp;
		}

		// 한 명뿐인 직업은 2단계에서 서로 다른 팀으로 분산합니다.
		if (.@group_count == 1) {
			.@single_index[.@single_count] = .@group_index[0];
			.@single_count++;
			continue;
		}

		.@target_family = callfunc("F_BG_RM_NoticeJobFamily", .@target_class);

		// 수라가 홀수이고 제네릭이 한쪽에만 있을 때는
		// 수라의 추가 1명이 제네릭 반대 팀으로 가도록 시작 팀을 고정합니다.
		if (
			.@target_family == 13 &&
			(.@group_count % 2) == 1 &&
			.@sura_extra_team >= 0
		)
			.@start_team = .@sura_extra_team;
		else if (.@count_g < .@count_c)
			.@start_team = 0;
		else if (.@count_c < .@count_g)
			.@start_team = 1;
		else
			.@start_team = rand(2);

		for (.@j = 0; .@j < .@group_count; .@j++) {
			.@index = .@group_index[.@j];
			.@team = (.@start_team + .@j) % 2;

			if (.@team == 0 && .@count_g >= .@high_cap_g)
				.@team = 1;
			else if (.@team == 1 && .@count_c >= .@high_cap_c)
				.@team = 0;

			if (.@team == 0) {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
			}
			else {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
			}
		}
	}

	// 2단계: 1명뿐인 직업 목록을 무작위로 섞은 뒤 2명씩 양 팀에 한 명씩 배정
	for (.@i = .@single_count - 1; .@i > 0; .@i--) {
		.@j = rand(.@i + 1);
		.@tmp = .@single_index[.@i];
		.@single_index[.@i] = .@single_index[.@j];
		.@single_index[.@j] = .@tmp;
	}

	// 같은 팀부터 계속 시작하지 않도록 쌍마다 시작 팀을 교대합니다.
	.@pair_start_team = rand(2);

	for (.@i = 0; .@i < .@single_count; .@i += 2) {
		.@remain_g = .@high_cap_g - .@count_g;
		.@remain_c = .@high_cap_c - .@count_c;

		// 두 명이 남았고 양 팀 모두 자리가 있으면 반드시 한 명씩 나눕니다.
		if (.@i + 1 < .@single_count && .@remain_g > 0 && .@remain_c > 0) {
			if (.@remain_g > .@remain_c)
				.@first_team = 0;
			else if (.@remain_c > .@remain_g)
				.@first_team = 1;
			else
				.@first_team = .@pair_start_team;

			.@index = .@single_index[.@i];
			if (.@first_team == 0) {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
			}
			else {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
			}

			// 같은 쌍의 두 번째 단독 직업은 반대 팀으로 배정합니다.
			.@index = .@single_index[.@i + 1];
			if (.@first_team == 0) {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
			}
			else {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
			}

			.@pair_start_team = 1 - .@pair_start_team;
			continue;
		}

		// 한 명만 남았거나 한쪽 팀 정원이 찬 경우 남은 정원에 맞춰 배정합니다.
		for (.@j = .@i; .@j < .@single_count && .@j < .@i + 2; .@j++) {
			.@index = .@single_index[.@j];
			.@remain_g = .@high_cap_g - .@count_g;
			.@remain_c = .@high_cap_c - .@count_c;

			if (.@remain_g <= 0)
				.@team = 1;
			else if (.@remain_c <= 0)
				.@team = 0;
			else if (.@remain_g > .@remain_c)
				.@team = 0;
			else if (.@remain_c > .@remain_g)
				.@team = 1;
			else if (.@count_g < .@count_c)
				.@team = 0;
			else if (.@count_c < .@count_g)
				.@team = 1;
			else
				.@team = rand(2);

			if (.@team == 0) {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
			}
			else {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
			}
		}
	}

	// 175레벨 참가자 우선 배정이 예약된 정원만큼 정확히 끝났는지 확인합니다.
	if (.@count_g != .@high_cap_g || .@count_c != .@high_cap_c) {
		debugmes "[BG RM] high-level assignment invalid"
			+ " G=" + .@count_g + "/" + .@high_cap_g
			+ " C=" + .@count_c + "/" + .@high_cap_c;
		callfunc "F_BG_RM_CancelRecruit", "175레벨 우선 팀 배정 중 오류가 발생하여 매칭이 취소되었습니다.",
			"An error occurred during Level 175 priority team assignment, so the match was cancelled.";
		end;
	}

	//========================================================
	// 마지막 단계: 175 미만 참가자 균형 배정
	//
	// 우선순위:
	// 1) 하드 분리 규칙으로 예약된 저레벨 팀 배치
	// 2) 동일 직업 저레벨을 가능한 한 서로 다른 팀에 분산
	// 3) 양 팀 저레벨 인원 수 균형
	// 4) 양 팀 저레벨 BaseLevel 합 균형
	//
	// 예:
	// 쉐도우체이서 170 / 쉐도우체이서 174가 있으면,
	// 양 팀 저레벨 슬롯이 남아 있는 한 1명씩 서로 다른 팀에 배정합니다.
	//========================================================
	cleararray .@low_assigned[0], 0, .@count;
	cleararray .@low_team[0], 0, .@count;
	cleararray .@low_group_done[0], 0, .@count;

	.@low_g = 0;
	.@low_c = 0;
	.@low_level_sum_g = 0;
	.@low_level_sum_c = 0;

	// -------------------------------------------------------
	// 1단계: 하드 분리 규칙으로 팀이 예약된 저레벨 먼저 확정
	// -------------------------------------------------------
	for (.@i = 0; .@i < .@low_count; .@i++) {
		.@index = .@low_index[.@i];

		if (!.@forced_team[.@index])
			continue;

		.@team = .@forced_team[.@index] - 1;
		.@remain_low_g = .@low_cap_g - .@low_g;
		.@remain_low_c = .@low_cap_c - .@low_c;

		if (
			(.@team == 0 && .@remain_low_g <= 0) ||
			(.@team == 1 && .@remain_low_c <= 0)
		) {
			callfunc "F_BG_RM_CancelRecruit",
				"저레벨 직업 우선 분리 슬롯이 부족하여 매칭이 취소되었습니다.",
				"There were not enough low-level job priority split slots, so the match was cancelled.";
			end;
		}

		if (.@team == 0) {
			.@aid_g[.@count_g] = .@aid_list[.@index];
			.@cid_g[.@count_g] = .@cid_list[.@index];
			.@name_g$[.@count_g] = .@name_list$[.@index];
			.@count_g++;
			.@low_g++;
			.@low_level_sum_g += .@level_list[.@index];
			.@low_team[.@index] = 1;
		}
		else {
			.@aid_c[.@count_c] = .@aid_list[.@index];
			.@cid_c[.@count_c] = .@cid_list[.@index];
			.@name_c$[.@count_c] = .@name_list$[.@index];
			.@count_c++;
			.@low_c++;
			.@low_level_sum_c += .@level_list[.@index];
			.@low_team[.@index] = 2;
		}

		.@low_assigned[.@index] = 1;
	}

	// -------------------------------------------------------
	// 2단계: 남은 저레벨은 동일 직업별 그룹으로 처리
	//
	// low_index 자체는 먼저 랜덤 셔플 후 레벨 내림차순 정렬합니다.
	// 같은 레벨끼리는 랜덤성이 유지되고, 높은 저레벨부터 그룹이 선택됩니다.
	// -------------------------------------------------------
	for (.@i = .@low_count - 1; .@i > 0; .@i--) {
		.@j = rand(.@i + 1);
		.@tmp = .@low_index[.@i];
		.@low_index[.@i] = .@low_index[.@j];
		.@low_index[.@j] = .@tmp;
	}

	for (.@i = 0; .@i < .@low_count - 1; .@i++) {
		for (.@j = 0; .@j < .@low_count - .@i - 1; .@j++) {
			if (
				.@level_list[.@low_index[.@j]] <
				.@level_list[.@low_index[.@j + 1]]
			) {
				.@tmp = .@low_index[.@j];
				.@low_index[.@j] = .@low_index[.@j + 1];
				.@low_index[.@j + 1] = .@tmp;
			}
		}
	}

	for (.@i = 0; .@i < .@low_count; .@i++) {
		.@first_index = .@low_index[.@i];

		if (.@low_assigned[.@first_index] || .@low_group_done[.@first_index])
			continue;

		.@target_class = .@class_list[.@first_index];
		.@group_count = 0;

		// 아직 미배정인 동일 직업 저레벨을 한 그룹으로 모읍니다.
		for (.@j = 0; .@j < .@low_count; .@j++) {
			.@index = .@low_index[.@j];

			if (
				.@low_assigned[.@index] ||
				.@low_group_done[.@index] ||
				.@class_list[.@index] != .@target_class
			)
				continue;

			.@low_group_index[.@group_count] = .@index;
			.@low_group_done[.@index] = 1;
			.@group_count++;
		}

		// 동일 직업 그룹 내부는 먼저 섞고 레벨 높은 순으로 정렬합니다.
		for (.@j = .@group_count - 1; .@j > 0; .@j--) {
			.@k = rand(.@j + 1);
			.@tmp = .@low_group_index[.@j];
			.@low_group_index[.@j] = .@low_group_index[.@k];
			.@low_group_index[.@k] = .@tmp;
		}

		for (.@j = 0; .@j < .@group_count - 1; .@j++) {
			for (.@k = 0; .@k < .@group_count - .@j - 1; .@k++) {
				if (
					.@level_list[.@low_group_index[.@k]] <
					.@level_list[.@low_group_index[.@k + 1]]
				) {
					.@tmp = .@low_group_index[.@k];
					.@low_group_index[.@k] = .@low_group_index[.@k + 1];
					.@low_group_index[.@k + 1] = .@tmp;
				}
			}
		}

		// 같은 직업의 하드 분리 예약 인원이 이미 팀에 들어간 경우까지 포함해서
		// 현재 해당 직업의 팀별 인원 수를 계산합니다.
		.@class_g = 0;
		.@class_c = 0;

		for (.@j = 0; .@j < .@low_count; .@j++) {
			.@index = .@low_index[.@j];

			if (
				.@class_list[.@index] != .@target_class ||
				!.@low_assigned[.@index]
			)
				continue;

			if (.@low_team[.@index] == 1)
				.@class_g++;
			else if (.@low_team[.@index] == 2)
				.@class_c++;
		}

		for (.@j = 0; .@j < .@group_count; .@j++) {
			.@index = .@low_group_index[.@j];
			.@remain_low_g = .@low_cap_g - .@low_g;
			.@remain_low_c = .@low_cap_c - .@low_c;

			if (.@remain_low_g <= 0)
				.@team = 1;
			else if (.@remain_low_c <= 0)
				.@team = 0;
			// 수라 홀수 특수 규칙:
			// 해당 저레벨 수라 그룹에서 양 팀 수라 수가 같아지는 순간에는
			// 추가 1명을 제네릭 반대 팀으로 우선 배치합니다.
			else if (
				callfunc("F_BG_RM_NoticeJobFamily", .@target_class) == 13 &&
				.@sura_extra_team >= 0 &&
				.@class_g == .@class_c &&
				((.@group_count - .@j) % 2) == 1
			)
				.@team = .@sura_extra_team;
			// 동일 직업이 적은 팀을 가장 먼저 선택합니다.
			else if (.@class_g < .@class_c)
				.@team = 0;
			else if (.@class_c < .@class_g)
				.@team = 1;
			// 동일 직업 수가 같으면 전체 저레벨 인원 수를 맞춥니다.
			else if (.@low_g < .@low_c)
				.@team = 0;
			else if (.@low_c < .@low_g)
				.@team = 1;
			// 인원까지 같으면 저레벨 레벨 합을 맞춥니다.
			else if (.@low_level_sum_g < .@low_level_sum_c)
				.@team = 0;
			else if (.@low_level_sum_c < .@low_level_sum_g)
				.@team = 1;
			else
				.@team = rand(2);

			if (.@team == 0) {
				.@aid_g[.@count_g] = .@aid_list[.@index];
				.@cid_g[.@count_g] = .@cid_list[.@index];
				.@name_g$[.@count_g] = .@name_list$[.@index];
				.@count_g++;
				.@low_g++;
				.@low_level_sum_g += .@level_list[.@index];
				.@class_g++;
				.@low_team[.@index] = 1;
			}
			else {
				.@aid_c[.@count_c] = .@aid_list[.@index];
				.@cid_c[.@count_c] = .@cid_list[.@index];
				.@name_c$[.@count_c] = .@name_list$[.@index];
				.@count_c++;
				.@low_c++;
				.@low_level_sum_c += .@level_list[.@index];
				.@class_c++;
				.@low_team[.@index] = 2;
			}

			.@low_assigned[.@index] = 1;
		}

		debugmes "[BG RM] low class split"
			+ " class=" + .@target_class
			+ " added=" + .@group_count
			+ " classG=" + .@class_g
			+ " classC=" + .@class_c;
	}


	debugmes "[BG RM] low-level assignment done"
		+ " G_low=" + .@low_g + " sum=" + .@low_level_sum_g
		+ " C_low=" + .@low_c + " sum=" + .@low_level_sum_c;

	if (
		.@count_g != .@cap_g ||
		.@count_c != .@cap_c ||
		.@low_g != .@low_cap_g ||
		.@low_c != .@low_cap_c ||
		(.@count_g > .@count_c + 1 || .@count_c > .@count_g + 1) ||
		.@count_g > .MaxPartyMembers ||
		.@count_c > .MaxPartyMembers
	) {
		debugmes "[BG RM] team assignment invalid"
			+ " G=" + .@count_g + "/" + .@cap_g
			+ " C=" + .@count_c + "/" + .@cap_c
			+ " G_low=" + .@low_g + "/" + .@low_cap_g
			+ " C_low=" + .@low_c + "/" + .@low_cap_c;
		callfunc "F_BG_RM_CancelRecruit", "최종 팀 배정 중 오류가 발생하여 매칭이 취소되었습니다.",
			"An error occurred during final team assignment, so the match was cancelled.";
		end;
	}

	debugmes "[BG RM] team assignment done G=" + .@count_g + " C=" + .@count_c;

	// 팀 편성은 끝났으므로 기본 무한루프 보호를 즉시 복구합니다.
	freeloop 0;

	// run_script_main의 command/goto 카운터를 새 실행 구간으로 갱신합니다.
	// 1ms 뒤 같은 위치에서 이어지며 지역 배열/변수는 그대로 유지됩니다.
	sleep 1;
	debugmes "[BG RM] matching loop guard restored";

	//========================================================
	// 파티 생성 직전 최종 검증
	// 이름 기반 조회 대신 신청 당시 저장한 AID/CID로 직접 확인합니다.
	//========================================================
	.@invalid = 0;
	.@invalid_reason$ = "";
	.@invalid_reason_en$ = "";
	.@invalid_name$ = "";
	.@invalid_cid = 0;

	for (.@team = 1; .@team <= 2 && !.@invalid; .@team++) {
		.@team_count = (.@team == 1) ? .@count_g : .@count_c;

		for (.@i = 0; .@i < .@team_count; .@i++) {
			.@check_aid = (.@team == 1) ? .@aid_g[.@i] : .@aid_c[.@i];
			.@check_cid = (.@team == 1) ? .@cid_g[.@i] : .@cid_c[.@i];
			.@check_name$ = (.@team == 1) ? .@name_g$[.@i] : .@name_c$[.@i];

			if (!isloggedin(.@check_aid,.@check_cid)) {
				.@invalid = 1;
				.@invalid_reason$ = "로그아웃";
				.@invalid_reason_en$ = "logged out";
			}
			else if (!attachrid(.@check_aid)) {
				.@invalid = 1;
				.@invalid_reason$ = "RID 연결 실패";
				.@invalid_reason_en$ = "RID attach failed";
			}
			else {
				if (getcharid(0) != .@check_cid) {
					.@invalid = 1;
					.@invalid_reason$ = "캐릭터 ID 불일치";
					.@invalid_reason_en$ = "character ID mismatch";
				}
				else if (strcharinfo(3) != "bat_room") {
					.@invalid = 1;
					.@invalid_reason$ = "전장 대기존 이탈";
					.@invalid_reason_en$ = "left the Battleground waiting zone";
				}
				else if (getcharid(1)) {
					.@invalid = 1;
					.@invalid_reason$ = "기존 파티 가입";
					.@invalid_reason_en$ = "already in an existing party";
				}
				else if (getcharid(4)) {
					.@invalid = 1;
					.@invalid_reason$ = "기존 BG 팀 가입";
					.@invalid_reason_en$ = "already on an existing BG team";
				}

				detachrid;
			}

			if (.@invalid) {
				.@invalid_name$ = .@check_name$;
				.@invalid_cid = .@check_cid;
				debugmes "[BG RM] pre-party validation failed"
					+ " team=" + ((.@team == 1) ? "G" : "C")
					+ " index=" + .@i
					+ " cid=" + .@invalid_cid
					+ " reason=" + .@invalid_reason$;
				break;
			}
		}
	}

	if (.@invalid) {
		callfunc "F_BG_RM_CancelRecruit",
			"파티 생성 직전 참가자 상태가 변경되어 매칭이 취소되었습니다. 대상: "
			+ .@invalid_name$ + " / 사유: " + .@invalid_reason$,
			"A participant's state changed just before party creation, so the match was cancelled. Target: " + .@invalid_name$ + " / Reason: " + .@invalid_reason_en$;
		end;
	}

	debugmes "[BG RM] pre-party validation done G=" + .@count_g + " C=" + .@count_c;

	//========================================================
	// 파티장 선정
	// 현재 만렙 175 캐릭터만 후보에 포함합니다.
	// 175 미만 참가자는 어떤 경우에도 자동 파티장이 되지 않습니다.
	//========================================================
	.@leader_candidate_g = 0;
	for (.@i = 0; .@i < .@count_g; .@i++) {
		if (readparam(BaseLevel,.@cid_g[.@i]) == .MaxBaseLevel) {
			.@leader_index_g[.@leader_candidate_g] = .@i;
			.@leader_candidate_g++;
		}
	}

	.@leader_candidate_c = 0;
	for (.@i = 0; .@i < .@count_c; .@i++) {
		if (readparam(BaseLevel,.@cid_c[.@i]) == .MaxBaseLevel) {
			.@leader_index_c[.@leader_candidate_c] = .@i;
			.@leader_candidate_c++;
		}
	}

	if (.@leader_candidate_g <= 0 || .@leader_candidate_c <= 0) {
		callfunc "F_BG_RM_CancelRecruit",
			"175레벨 파티장 후보가 없는 진영이 발생하여 매칭이 취소되었습니다.",
			"One side had no Level 175 party leader candidate, so the match was cancelled.";
		end;
	}

	.@leader_index = .@leader_index_g[rand(.@leader_candidate_g)];
	if (.@leader_index > 0) {
		.@tmp = .@aid_g[0];
		.@aid_g[0] = .@aid_g[.@leader_index];
		.@aid_g[.@leader_index] = .@tmp;

		.@tmp = .@cid_g[0];
		.@cid_g[0] = .@cid_g[.@leader_index];
		.@cid_g[.@leader_index] = .@tmp;

		.@tmp$ = .@name_g$[0];
		.@name_g$[0] = .@name_g$[.@leader_index];
		.@name_g$[.@leader_index] = .@tmp$;
	}

	.@leader_index = .@leader_index_c[rand(.@leader_candidate_c)];
	if (.@leader_index > 0) {
		.@tmp = .@aid_c[0];
		.@aid_c[0] = .@aid_c[.@leader_index];
		.@aid_c[.@leader_index] = .@tmp;

		.@tmp = .@cid_c[0];
		.@cid_c[0] = .@cid_c[.@leader_index];
		.@cid_c[.@leader_index] = .@tmp;

		.@tmp$ = .@name_c$[0];
		.@name_c$[0] = .@name_c$[.@leader_index];
		.@name_c$[.@leader_index] = .@tmp$;
	}

	debugmes "[BG RM] level-175 leaders selected"
		+ " G_cid=" + .@cid_g[0]
		+ " C_cid=" + .@cid_c[0]
		+ " G_candidates=" + .@leader_candidate_g
		+ " C_candidates=" + .@leader_candidate_c;

	$@BG_RM_PARTY_SERIAL++;
	.@serial = $@BG_RM_PARTY_SERIAL % 10000;
	.@party_suffix$ = (gettimetick(2) % 100000000) + "-" + .@serial;
	.@party_name_g$ = "[기욤]" + .@party_suffix$;
	.@party_name_c$ = "[크루아]" + .@party_suffix$;

	debugmes "[BG RM] leaders selected"
		+ " G_cid=" + .@cid_g[0]
		+ " C_cid=" + .@cid_c[0]
		+ " G_count=" + .@count_g
		+ " C_count=" + .@count_c;

	//========================================================
	// 기욤 파티 생성 및 전원 가입 확인
	//========================================================
	$@party_create_id = 0;
	debugmes "[BG RM] party_create G name=" + .@party_name_g$
		+ " cid=" + .@cid_g[0]
		+ " count=" + .@count_g
		+ " state=" + $@BG_RM_STATE;

	.@result = party_create(.@party_name_g$,.@cid_g[0],0,0);
	if (.@result != 1) {
		callfunc "F_BG_RM_CancelRecruit", "기욤 파티 생성 요청에 실패했습니다. 오류 코드: " + .@result,
			"The Guillaume party creation request failed. Error code: " + .@result;
		end;
	}

	.@party_id = 0;
	for (.@retry = 0; .@retry < 50; .@retry++) {
		sleep 100;
		if ($@BG_RM_STATE != 2 || .@finalize_token != $@BG_RM_TOKEN)
			end;
		.@party_id = getcharid(1,.@name_g$[0]);
		if (.@party_id > 0)
			break;
	}

	if (.@party_id <= 0) {
		debugmes "[BG RM] G party id timeout leader=" + .@name_g$[0]
			+ " cid=" + .@cid_g[0]
			+ " global_id=" + $@party_create_id;
		callfunc "F_BG_RM_CancelRecruit", "기욤 파티 생성 완료를 확인하지 못했습니다.",
			"Could not confirm that the Guillaume party was created.";
		end;
	}

	$@BG_RM_PARTY_G = .@party_id;
	debugmes "[BG RM] G party created party_id=" + $@BG_RM_PARTY_G
		+ " leader=" + .@name_g$[0];
	sleep 500;
	if ($@BG_RM_STATE != 2 || .@finalize_token != $@BG_RM_TOKEN)
		end;

	for (.@i = 1; .@i < .@count_g; .@i++) {
		if (getcharid(1,.@name_g$[.@i]) > 0) {
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "기욤 파티원이 다른 파티에 가입하여 매칭이 취소되었습니다: " + .@name_g$[.@i],
				"A Guillaume party member joined another party, so the match was cancelled: " + .@name_g$[.@i];
			end;
		}

		.@result = 0;
		for (.@retry = 0; .@retry < 10; .@retry++) {
			.@result = party_addmember($@BG_RM_PARTY_G,.@cid_g[.@i]);
			if (.@result == 1)
				break;
			sleep 200;
		}

		debugmes "[BG RM] G addmember name=" + .@name_g$[.@i]
			+ " cid=" + .@cid_g[.@i]
			+ " party_id=" + $@BG_RM_PARTY_G
			+ " result=" + .@result;

		if (.@result != 1) {
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "기욤 파티원 추가 요청에 실패했습니다. 대상: " + .@name_g$[.@i] + " / 오류 코드: " + .@result,
				"The request to add a Guillaume party member failed. Target: " + .@name_g$[.@i] + " / Error code: " + .@result;
			end;
		}

		.@joined = 0;
		for (.@retry = 0; .@retry < 50; .@retry++) {
			sleep 100;
			if ($@BG_RM_STATE != 2 || .@finalize_token != $@BG_RM_TOKEN)
				end;
			.@current_party = getcharid(1,.@name_g$[.@i]);
			if (.@current_party == $@BG_RM_PARTY_G) {
				.@joined = 1;
				break;
			}
			if (.@current_party > 0)
				break;
		}

		if (!.@joined) {
			debugmes "[BG RM] G join timeout name=" + .@name_g$[.@i]
				+ " expected=" + $@BG_RM_PARTY_G
				+ " actual=" + getcharid(1,.@name_g$[.@i]);
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "기욤 파티원의 실제 가입 완료를 확인하지 못했습니다: " + .@name_g$[.@i],
				"Could not confirm that the Guillaume party member actually joined: " + .@name_g$[.@i];
			end;
		}
	}

	//========================================================
	// 크루아 파티 생성 및 전원 가입 확인
	//========================================================
	$@party_create_id = 0;
	debugmes "[BG RM] party_create C name=" + .@party_name_c$
		+ " cid=" + .@cid_c[0]
		+ " count=" + .@count_c
		+ " state=" + $@BG_RM_STATE;

	.@result = party_create(.@party_name_c$,.@cid_c[0],0,0);
	if (.@result != 1) {
		callfunc "F_BG_RM_DestroyParties";
		callfunc "F_BG_RM_CancelRecruit", "크루아 파티 생성 요청에 실패했습니다. 오류 코드: " + .@result,
			"The Croix party creation request failed. Error code: " + .@result;
		end;
	}

	.@party_id = 0;
	for (.@retry = 0; .@retry < 50; .@retry++) {
		sleep 100;
		if ($@BG_RM_STATE != 2 || .@finalize_token != $@BG_RM_TOKEN)
			end;
		.@party_id = getcharid(1,.@name_c$[0]);
		if (.@party_id > 0)
			break;
	}

	if (.@party_id <= 0) {
		debugmes "[BG RM] C party id timeout leader=" + .@name_c$[0]
			+ " cid=" + .@cid_c[0]
			+ " global_id=" + $@party_create_id;
		callfunc "F_BG_RM_DestroyParties";
		callfunc "F_BG_RM_CancelRecruit", "크루아 파티 생성 완료를 확인하지 못했습니다.",
			"Could not confirm that the Croix party was created.";
		end;
	}

	$@BG_RM_PARTY_C = .@party_id;
	debugmes "[BG RM] C party created party_id=" + $@BG_RM_PARTY_C
		+ " leader=" + .@name_c$[0];
	sleep 500;
	if ($@BG_RM_STATE != 2 || .@finalize_token != $@BG_RM_TOKEN)
		end;

	for (.@i = 1; .@i < .@count_c; .@i++) {
		if (getcharid(1,.@name_c$[.@i]) > 0) {
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "크루아 파티원이 다른 파티에 가입하여 매칭이 취소되었습니다: " + .@name_c$[.@i],
				"A Croix party member joined another party, so the match was cancelled: " + .@name_c$[.@i];
			end;
		}

		.@result = 0;
		for (.@retry = 0; .@retry < 10; .@retry++) {
			.@result = party_addmember($@BG_RM_PARTY_C,.@cid_c[.@i]);
			if (.@result == 1)
				break;
			sleep 200;
		}

		debugmes "[BG RM] C addmember name=" + .@name_c$[.@i]
			+ " cid=" + .@cid_c[.@i]
			+ " party_id=" + $@BG_RM_PARTY_C
			+ " result=" + .@result;

		if (.@result != 1) {
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "크루아 파티원 추가 요청에 실패했습니다. 대상: " + .@name_c$[.@i] + " / 오류 코드: " + .@result,
				"The request to add a Croix party member failed. Target: " + .@name_c$[.@i] + " / Error code: " + .@result;
			end;
		}

		.@joined = 0;
		for (.@retry = 0; .@retry < 50; .@retry++) {
			sleep 100;
			if ($@BG_RM_STATE != 2 || .@finalize_token != $@BG_RM_TOKEN)
				end;
			.@current_party = getcharid(1,.@name_c$[.@i]);
			if (.@current_party == $@BG_RM_PARTY_C) {
				.@joined = 1;
				break;
			}
			if (.@current_party > 0)
				break;
		}

		if (!.@joined) {
			debugmes "[BG RM] C join timeout name=" + .@name_c$[.@i]
				+ " expected=" + $@BG_RM_PARTY_C
				+ " actual=" + getcharid(1,.@name_c$[.@i]);
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "크루아 파티원의 실제 가입 완료를 확인하지 못했습니다: " + .@name_c$[.@i],
				"Could not confirm that the Croix party member actually joined: " + .@name_c$[.@i];
			end;
		}
	}

	for (.@i = 0; .@i < .@count_g; .@i++) {
		if (
			!isloggedin(.@aid_g[.@i],.@cid_g[.@i]) ||
			getmapxy(.@map$,.@x,.@y,BL_PC,.@name_g$[.@i]) != 0 ||
			.@map$ != "bat_room" ||
			getcharid(1,.@name_g$[.@i]) != $@BG_RM_PARTY_G ||
			getcharid(4,.@name_g$[.@i])
		) {
			debugmes "[BG RM] final G validation failed name=" + .@name_g$[.@i]
				+ " expected_party=" + $@BG_RM_PARTY_G
				+ " actual_party=" + getcharid(1,.@name_g$[.@i]);
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "기욤 참가자가 대기실을 이탈하거나 파티 구성이 변경되어 전장 입장이 취소되었습니다.",
				"A Guillaume participant left the waiting room or the party composition changed, so Battleground entry was cancelled.";
			end;
		}
	}

	for (.@i = 0; .@i < .@count_c; .@i++) {
		if (
			!isloggedin(.@aid_c[.@i],.@cid_c[.@i]) ||
			getmapxy(.@map$,.@x,.@y,BL_PC,.@name_c$[.@i]) != 0 ||
			.@map$ != "bat_room" ||
			getcharid(1,.@name_c$[.@i]) != $@BG_RM_PARTY_C ||
			getcharid(4,.@name_c$[.@i])
		) {
			debugmes "[BG RM] final C validation failed name=" + .@name_c$[.@i]
				+ " expected_party=" + $@BG_RM_PARTY_C
				+ " actual_party=" + getcharid(1,.@name_c$[.@i]);
			callfunc "F_BG_RM_DestroyParties";
			callfunc "F_BG_RM_CancelRecruit", "크루아 참가자가 대기실을 이탈하거나 파티 구성이 변경되어 전장 입장이 취소되었습니다.",
				"A Croix participant left the waiting room or the party composition changed, so Battleground entry was cancelled.";
			end;
		}
	}

	// 전장 생성 함수에서 사용할 최종 팀 배열 저장
	$@BG_RM_G_COUNT = .@count_g;
	$@BG_RM_C_COUNT = .@count_c;

	for (.@i = 0; .@i < .@count_g; .@i++) {
		$@BG_RM_G_AID[.@i] = .@aid_g[.@i];
		$@BG_RM_G_CID[.@i] = .@cid_g[.@i];
		$@BG_RM_G_NAME$[.@i] = .@name_g$[.@i];
	}
	for (.@i = 0; .@i < .@count_c; .@i++) {
		$@BG_RM_C_AID[.@i] = .@aid_c[.@i];
		$@BG_RM_C_CID[.@i] = .@cid_c[.@i];
		$@BG_RM_C_NAME$[.@i] = .@name_c$[.@i];
	}

	// 랜덤 매칭 동안 기존 인원 설정을 임시 보관
	$@BG_RM_COUNTS_SAVED = 1;

			announce needtr("[전장] 추가 모집 참가를 취소했습니다.", "[Battleground] You left the extra recruitment."), bc_self, "0xFFAA55";
			mapannouncelang "bat_room", "[전장 대기열] " + strcharinfo(0) + "님이 참가를 취소했습니다. 현재 " + $@BG_RM_COUNT + "명",
				"[Battleground Queue] " + strcharinfo(0) + " left. Currently " + $@BG_RM_COUNT + " players", bc_map, "0xFFCC99";
			callfunc "F_NEED_BGUX_ReturnSelf", "cancel";
			if ($@BG_RM_COUNT <= 0)
				callfunc "F_BG_RM_CancelRecruit", "모든 참가자가 신청을 취소하여 모집이 종료되었습니다.", "All participants cancelled, so recruitment has ended.";
			end;
		}
	}

	.@slot = callfunc("F_BG_RM_QueueFind", getcharid(3), getcharid(0));

	if (.@slot < 0) {
		dispbottom needtr("[전장] 현재 전장 대기열이나 추가 모집에 등록되어 있지 않습니다.", "[Battleground] You are not in the Battleground queue or the extra recruitment.");
		end;
	}

	callfunc "F_BG_RM_QueueRemoveSlot", .@slot;
	callfunc "F_NEED_BGUX_QueueLockCheck";	// 2026-11-12: 출발 준비는 최소 인원 미만이 될 때만 취소

	announce needtr("[전장] 자동매칭 대기 신청을 취소했습니다.", "[Battleground] You left the auto-match queue."), bc_self, "0xFFAA55";
	announcelang "[전장 대기열] 현재 대기 " + $@BG_RM_QUEUE_COUNT + "/" + $@BG_RM_MIN_PLAYERS + "명", "[BG Queue] Now queued: " + $@BG_RM_QUEUE_COUNT + "/" + $@BG_RM_MIN_PLAYERS,
		bc_all, "0xFFCC99";
	end;

OnCommandQueueStatus:
	.@slot = callfunc("F_BG_RM_QueueFind", getcharid(3), getcharid(0));
	.@rslot = ($@BG_RM_STATE == 1 || $@BG_RM_STATE == 2) ? callfunc("F_BG_RM_FindRecruitSlot", getcharid(3), getcharid(0)) : -1;

	if (.@rslot >= 0)
		announce needtr("[전장] 내 상태: 추가 모집 참가 중", "[Battleground] My status: in the extra recruitment"), bc_self, "0x66FF66";
	else if (.@slot >= 0)
		announce needtr("[전장] 내 상태: 자동매칭 대기 중", "[Battleground] My status: queued for auto-match"), bc_self, "0x66FF66";
	else
		announce needtr("[전장] 내 상태: 전장 대기 미신청", "[Battleground] My status: not queued"), bc_self, "0xFFCC66";

	// 2026-11-12 통합 모집: 대기 / 출발 준비 / 추가 모집 / 마감 / 경기 를 한 줄로(안내 NPC 와 같은 함수)
	dispbottom "[" + needtr("전장", "Battleground") + "] " + callfunc("F_NEED_BGUX_RmStateLine$");
	if ($@BG_RM_STATE == 1 && BG_RM_LAST_CANCEL_TOKEN == $@BG_RM_TOKEN && BG_RM_LAST_CANCEL_REASON > 0 && .@rslot < 0)
		dispbottom needtr("[전장] 최근 취소 사유: ", "[Battleground] Last cancellation: ") + callfunc("F_BG_RM_CancelReasonLang$", BG_RM_LAST_CANCEL_REASON, getneedlang());
	end;

OnPCLogoutEvent:
	.@slot = callfunc("F_BG_RM_QueueFind", getcharid(3), getcharid(0));

	if (.@slot < 0)
		end;

	.@name$ = strcharinfo(0);
	callfunc "F_BG_RM_QueueRemoveSlot", .@slot;
	callfunc "F_NEED_BGUX_QueueLockCheck";	// 2026-11-12: 출발 준비는 최소 인원 미만이 될 때만 취소

	announcelang "[전장 대기열] " + .@name$ + "님이 로그아웃하여 대기에서 제외되었습니다. 현재 "
		+ $@BG_RM_QUEUE_COUNT + "/" + $@BG_RM_MIN_PLAYERS + "명", "[BG Queue] " + .@name$ + " logged out and was removed from the queue. Now " + $@BG_RM_QUEUE_COUNT + "/" + $@BG_RM_MIN_PLAYERS,
		bc_all, "0xFFCC99";
	end;

OnTimer1000:
	stopnpctimer;

	if ($@BG_RM_MIN_JOIN_LEVEL <= 0)
		$@BG_RM_MIN_JOIN_LEVEL = 150;
	if ($@BG_RM_MIN_PLAYERS <= 0)
		$@BG_RM_MIN_PLAYERS = 8;
	if ($@BG_RM_MAX_PLAYERS <= 0)
		$@BG_RM_MAX_PLAYERS = 48;

	for (.@i = $@BG_RM_QUEUE_COUNT - 1; .@i >= 0; .@i--) {
		.@aid = $@BG_RM_QUEUE_AID[.@i];
		.@cid = $@BG_RM_QUEUE_CID[.@i];
		.@name$ = $@BG_RM_QUEUE_NAME$[.@i];
		.@remove = 0;
		.@reason$ = ""; .@reason_en$ = "";

		if (!isloggedin(.@aid,.@cid)) {
			.@remove = 1;
			.@reason$ = "로그아웃"; .@reason_en$ = "logged out";
		}
		else if (!attachrid(.@aid)) {
			.@remove = 1;
			.@reason$ = "접속 상태 확인 실패"; .@reason_en$ = "connection check failed";
		}
		else {
			if (getcharid(0) != .@cid) {
				.@remove = 1;
				.@reason$ = "캐릭터 변경"; .@reason_en$ = "changed character";
			}
			else if (BaseLevel < $@BG_RM_MIN_JOIN_LEVEL) {
				.@remove = 1;
				.@reason$ = "참가 레벨 미달"; .@reason_en$ = "below the level requirement";
			}
			else if (getcharid(1) || getcharid(4)) {
				.@remove = 1;
				.@reason$ = "파티 또는 전장 팀 가입"; .@reason_en$ = "joined a party or Battleground team";
			}
			detachrid;
		}

		if (!.@remove)
			continue;

		callfunc "F_BG_RM_QueueRemoveSlot", .@i;

		if (isloggedin(.@aid,.@cid))
			messagelang .@name$, "[전장] " + .@reason$ + " 상태로 자동매칭 대기열에서 제외되었습니다.", "[Battleground] You were removed from the auto-match queue (" + .@reason_en$ + ").";

		callfunc "F_NEED_BGUX_QueueLockCheck";	// 2026-11-12: 출발 준비는 최소 인원 미만이 될 때만 취소
	}

	if ($@BG_RM_STATE == 1 || $@BG_RM_STATE == 2) {
		initnpctimer;
		end;
	}

	.@busy = 0;
	if ($@BG_RM_ACTIVE || $@BG_RM_CYCLE_ACTIVE)
		.@busy = 1;
	if ($@TierraBG1 || $@FlaviusBG1 || $@KvM01BG)
		.@busy = 1;
	if ($BG_GUILLAUME_PARTY_ID || $BG_CROIX_PARTY_ID)
		.@busy = 1;
	if (callfunc("F_BG_RM_ManualReadyActive"))
		.@busy = 1;

	if (.@busy) {
		$@BG_RM_QUEUE_LOCK = 0;
		$@BG_RM_QUEUE_READY_TICK = 0;
		initnpctimer;
		end;
	}

	if (!$@BG_RM_QUEUE_LOCK && $@BG_RM_QUEUE_COUNT >= $@BG_RM_MIN_PLAYERS) {
		$@BG_RM_QUEUE_LOCK = 1;
		.@pre = ($@BG_RM_PREWARP_SECONDS > 0) ? $@BG_RM_PREWARP_SECONDS : 20;	// 2026-11-12: 출발 전 안내 10 -> 20초 (운영자 결정). 값은 NEED_BG_RM_Core OnInit 한 곳
		$@BG_RM_QUEUE_READY_TICK = gettimetick(2) + .@pre;
		$@BG_RM_QUEUE_TOKEN++;

		callfunc "F_BG_RM_QueueNotifyAll",
			"[전장] 전장 최소 인원이 충족되었습니다. " + .@pre + "초 후 전장 대기실로 자동 이동합니다.",
			"[Battleground] The minimum number of players was reached. You will be moved to the Battleground waiting room in " + .@pre + " seconds.";

		announcelang "[전장 대기열] 최소 인원 " + $@BG_RM_MIN_PLAYERS
			+ "명이 충족되었습니다. " + .@pre + "초 후 전장 대기실로 이동해 추가 모집을 시작합니다. 지금 @전장대기 하면 함께 이동합니다.",
			"[BG Queue] The minimum of " + $@BG_RM_MIN_PLAYERS + " players was reached. In " + .@pre + " seconds everyone moves to the waiting room and extra recruitment starts. Use @bgqueue now to go along.",
			bc_all, "0x66FF99";
	}

	if ($@BG_RM_QUEUE_LOCK && $@BG_RM_QUEUE_READY_TICK > 0) {
		if (gettimetick(2) >= $@BG_RM_QUEUE_READY_TICK) {
			if ($@BG_RM_QUEUE_COUNT < $@BG_RM_MIN_PLAYERS) {
				$@BG_RM_QUEUE_LOCK = 0;
				$@BG_RM_QUEUE_READY_TICK = 0;
				callfunc "F_BG_RM_QueueNotifyAll",
					"[전장] 출발 전 대기 인원이 부족해져 자동 이동이 취소되었습니다. 대기는 그대로 유지됩니다.",
					"[Battleground] Too few players were left before departure, so the automatic move was cancelled. You are still in the queue.";
			}
			else
				donpcevent "NEED_BG_RM_Core::OnQueueLaunch";
		}
	}

	initnpctimer;
	end;
}


//============================================================
// 랜덤 매칭 모집 참가자 이탈 감시
// - 모집 중 로그아웃, bat_room 이탈, 파티/BG 가입 시 신청 자동 취소
// - 맵 이탈은 최대 1초 이내 반영
//============================================================
-	script	NEED_BG_RM_RecruitMonitor	-1,{
	end;

OnInit:
	.NextMissingNoticeTick = 0;
	stopnpctimer;
	end;

OnStart:
	.NextMissingNoticeTick = gettimetick(2) + 30;
	.AnnMask = 0;	// 2026-11-12: 마감 60/30/10초 전 안내를 한 번씩만
	stopnpctimer;
	initnpctimer;
	end;

OnStop:
	.NextMissingNoticeTick = 0;
	stopnpctimer;
	end;

OnPCLoginEvent:
	if (BG_RM_LAST_CANCEL_REASON != 1)
		end;

	announce needtr("[전장] 이전 랜덤 매칭 참가 신청은 게임 종료로 자동 취소되었습니다.", "[Battleground] Your previous random match application was cancelled automatically because you logged out."), bc_self, "0xFF5555";
	dispbottom needtr("[전장] 모집이 아직 진행 중이라면 어디서든 @전장대기로 다시 신청해 주세요.","[Battleground] If recruitment is still open, apply again from anywhere with @bgqueue.");
	end;

OnPCLogoutEvent:
	if ($@BG_RM_STATE != 1)
		end;

	.@slot = -1;
	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		if ($@BG_RM_AID[.@i] == getcharid(3) && $@BG_RM_CID[.@i] == getcharid(0)) {
			.@slot = .@i;
			break;
		}
	}

	if (.@slot < 0)
		end;

	.@name$ = $@BG_RM_NAME$[.@slot];

	BG_RM_LAST_CANCEL_REASON = 1;
	BG_RM_LAST_CANCEL_TOKEN = $@BG_RM_TOKEN;
	BG_RM_LAST_CANCEL_TICK = gettimetick(2);

	callfunc "F_BG_RM_RemoveRecruitSlot", .@slot;

	mapannouncelang "bat_room",
		"[전장 랜덤 매칭] " + .@name$ + "님이 게임을 종료하여 참가 신청이 취소되었습니다. 현재 " + $@BG_RM_COUNT + "명", "[Battleground Random Match] " + .@name$ + " logged out, so their application was cancelled. Currently " + $@BG_RM_COUNT + " players",
		bc_map, "0xFFCC99";

	if ($@BG_RM_COUNT <= 0) {
		callfunc "F_BG_RM_CancelRecruit", "모든 참가자가 이탈하여 랜덤 매칭 모집이 취소되었습니다.",
			"All participants left, so random match recruitment was cancelled.";
		end;
	}
	end;

OnTimer1000:
	stopnpctimer;

	if ($@BG_RM_STATE != 1)
		end;

	if (.NextMissingNoticeTick > 0 && gettimetick(2) >= .NextMissingNoticeTick) {
		callfunc "F_BG_RM_AnnounceMissingApplicants";
		.NextMissingNoticeTick = gettimetick(2) + 30;
	}

	// 뒤에서부터 제거하여 배열 이동 이후의 인덱스가 꼬이지 않도록 합니다.
	for (.@i = $@BG_RM_COUNT - 1; .@i >= 0; .@i--) {
		.@aid = $@BG_RM_AID[.@i];
		.@cid = $@BG_RM_CID[.@i];
		.@name$ = $@BG_RM_NAME$[.@i];
		.@reason$ = "";
		.@reason_en$ = "";
		.@reason_code = 0;

		if (!isloggedin(.@aid,.@cid)) {
			.@reason$ = "게임을 종료하여";
			.@reason_en$ = "logged out of the game";
			.@reason_code = 1;
		}
		else if (getmapxy(.@map$,.@x,.@y,BL_PC,.@name$) != 0) {
			.@reason$ = "접속 위치를 확인할 수 없어";
			.@reason_en$ = "their location could not be confirmed";
			.@reason_code = 4;
		}
		else if (.@map$ != "bat_room") {
			.@reason$ = "전장 대기존을 이탈하여";
			.@reason_en$ = "left the Battleground waiting zone";
			.@reason_code = 2;
		}
		else if (getcharid(1,.@name$) || getcharid(4,.@name$)) {
			.@reason$ = "파티 또는 전장 팀에 가입하여";
			.@reason_en$ = "joined a party or a Battleground team";
			.@reason_code = 3;
		}
		else if ($@BG_RM_MIN_JOIN_LEVEL > 0 && readparam(BaseLevel,.@cid) < $@BG_RM_MIN_JOIN_LEVEL) {
			.@reason$ = "참가 레벨 조건을 충족하지 못해";
			.@reason_en$ = "did not meet the level requirement";
			.@reason_code = 6;
		}

		if (.@reason_code == 0)
			continue;

		if (isloggedin(.@aid,.@cid))
			callfunc "F_BG_RM_NotifyAutoCancel", .@aid, .@cid, .@reason_code;

		callfunc "F_BG_RM_RemoveRecruitSlot", .@i;

		mapannouncelang "bat_room",
			"[전장 랜덤 매칭] " + .@name$ + "님이 " + .@reason$ + " 참가 신청이 취소되었습니다. 현재 " + $@BG_RM_COUNT + "명",
			"[Battleground Random Match] " + .@name$ + " " + .@reason_en$ + ", so their application was cancelled. Currently " + $@BG_RM_COUNT + " players",
			bc_map, "0xFFCC99";
	}

	if ($@BG_RM_COUNT <= 0) {
		callfunc "F_BG_RM_CancelRecruit", "모든 참가자가 이탈하여 랜덤 매칭 모집이 취소되었습니다.",
			"All participants left, so random match recruitment was cancelled.";
		end;
	}

	// 2026-11-12 통합 모집: 마감은 여기 한 곳에서만 판정한다(서버 시각 기준, 늦은 신청으로 늘지 않음).
	// OnFinalize 는 첫 줄에서 STATE 를 2 로 바꿔 신청을 잠그고, 스크립트는 한 번에 하나씩 실행되므로 같은 틱의 신청과 겹치지 않는다.
	if ($@BG_RM_STATE == 1 && $@BG_RM_END_TICK > 0) {
		.@left = $@BG_RM_END_TICK - gettimetick(2);
		if (.@left <= 0) {
			stopnpctimer;
			debugmes "[NEED_BGUX] RECRUIT_CLOSE token=" + $@BG_RM_TOKEN + " count=" + $@BG_RM_COUNT;
			mapannouncelang "bat_room", "[전장 대기열] 추가 모집이 마감되었습니다. 참가자 확인 후 팀을 편성합니다. (" + $@BG_RM_COUNT + "명)",
				"[BG Queue] Extra recruitment has closed. Checking participants and forming teams. (" + $@BG_RM_COUNT + " players)", bc_map, "0xFFCC66";
			donpcevent "NEED_BG_RM_Core::OnFinalize";
			end;
		}
		.@max = ($@BG_RM_MAX_PLAYERS > 0) ? $@BG_RM_MAX_PLAYERS : 48;
		if (.@left <= 10 && !(.AnnMask & 4)) {
			.AnnMask = .AnnMask | 7;
			mapannouncelang "bat_room", "[전장 대기열] 추가 모집 마감 10초 전. 현재 " + $@BG_RM_COUNT + "/" + .@max + "명",
				"[BG Queue] Extra recruitment closes in 10 seconds. " + $@BG_RM_COUNT + "/" + .@max + " players", bc_map, "0xFFCC66";
		}
		else if (.@left <= 30 && !(.AnnMask & 2)) {
			.AnnMask = .AnnMask | 3;
			mapannouncelang "bat_room", "[전장 대기열] 추가 모집 마감 30초 전. 현재 " + $@BG_RM_COUNT + "/" + .@max + "명",
				"[BG Queue] Extra recruitment closes in 30 seconds. " + $@BG_RM_COUNT + "/" + .@max + " players", bc_map, "0x66CCFF";
		}
		else if (.@left <= 60 && !(.AnnMask & 1)) {
			.AnnMask = .AnnMask | 1;
			announcelang "[전장 대기열] 기존 전장 추가 모집 마감 60초 전: 현재 " + $@BG_RM_COUNT + "/" + .@max + "명. 마을에서도 @전장대기 로 바로 합류할 수 있습니다.",
				"[BG Queue] Classic BG extra recruitment closes in 60 seconds: " + $@BG_RM_COUNT + "/" + .@max + " players. Join from town with @bgqueue.", bc_all, "0x66CCFF";
		}
	}

	if ($@BG_RM_STATE == 1)
		initnpctimer;
	end;
}

//============================================================
// 랜덤 매칭 풀사이클 감시
// state 3: 현재 단계 전장 진행/정리 대기
// state 4: 다음 전장 파티장 입장 준비 대기
//============================================================
-	script	NEED_BG_RM_Monitor	-1,{
	end;

OnInit:
	stopnpctimer;
	end;

OnStart:
	stopnpctimer;
	initnpctimer;
	end;

OnTimer1000:
	stopnpctimer;

	if (!$@BG_RM_CYCLE_ACTIVE)
		end;

	.@running = 0;
	if ($@BG_RM_MODE == 1)
		.@running = $@TierraBG1;
	else if ($@BG_RM_MODE == 2)
		.@running = $@FlaviusBG1;
	else if ($@BG_RM_MODE == 3)
		.@running = $@KvM01BG;

	// 다음 전장 준비 대기 중에는 해당 전장이 시작되는 순간만 감지합니다.
	if ($@BG_RM_STATE == 4) {
		if (.@running) {
			$@BG_RM_STATE = 3;
			mapannouncelang "bat_room", "[전장 랜덤 매칭] " + callfunc("F_BG_RM_ModeName$",$@BG_RM_MODE) + " 입장이 완료되었습니다.", "[Battleground Random Match] " + callfunc("F_BG_RM_ModeNameLang$",$@BG_RM_MODE,NEED_LANG_EN) + " entry is complete.", bc_map, "0x99FF99";
		}
		initnpctimer;
		end;
	}

	// KVM 보상 단계에서 온라인 랜덤 매칭 참가자가 모두 대기실로 복귀하면
	// KVM 자체의 60초 정리 타이머를 기다리지 않고 자동 생성 파티만 즉시 해체합니다.
	// 전체 랜덤 매칭 상태/인원 설정/BG 정리는 기존 KVM 종료 타이머가 끝난 뒤 처리됩니다.
	if (
		$@BG_RM_MODE == 3 &&
		$@BG_RM_CYCLE_STAGE == 3 &&
		$@KvM01BG == 3 &&
		$@KvM01BG_Victory > 0 &&
		!$@BG_RM_KVM_PARTY_DESTROYED &&
		!callfunc("F_BG_RM_HasOnlineMemberInMap","bat_c01")
	) {
		callfunc "F_BG_RM_DestroyParties";
		$@BG_RM_KVM_PARTY_DESTROYED = 1;

		mapannouncelang "bat_room",
			"[전장 랜덤 매칭] KVM 참가자가 모두 대기실로 복귀하여 자동 생성 파티를 해체했습니다.", "[Battleground Random Match] All KVM participants have returned to the waiting room, so the auto-created party was disbanded.",
			bc_map, "0x99FF99";
	}

	if (.@running) {
		initnpctimer;
		end;
	}

	// 현재 단계의 BG 팀과 보상 처리가 모두 정리된 뒤 다음 단계로 전환합니다.
	callfunc "F_BG_RM_AdvanceCycle";

	if ($@BG_RM_CYCLE_ACTIVE)
		initnpctimer;
	end;
}

// 2026-11-12 NEED_BGUX: 맵서버 재시작/비정상 종료 뒤 남은 "고아" 랜덤 매칭 상태 복구

// 문제: 랜덤 매칭은 자동 파티장 등록을 영구 변수($BG_GUILLAUME_* / $BG_CROIX_*)에 쓰고, 기존 인원 설정
//       ($BG_User_Count / $BG_Guillaume_Count / $BG_Croix_Count)을 덮어쓴 뒤 원래 값을 임시 변수($@)에만 둔다.

//       판단해 영원히 출발하지 못하고(2026-11-12 격리 재현), 인원 설정도 랜덤 매칭 값으로 남는다.
// 소유 표식: 랜덤 매칭이 인원 설정을 덮어쓰는 순간(파티 2개 생성·검증 직후, 파티장 등록과 같은 실행 흐름)에
//       $NEED_BGUX_RM_PG / PC (자동 생성 파티 ID), $NEED_BGUX_RM_SAVED + _U/_G/_C (원래 인원 설정) 를 남기고,
//       정상 종료·취소 경로(F_BG_RM_RestoreCounts / F_BG_RM_ClearLeaders)에서 지운다.
// 복구: 진행 중인 랜덤 매칭·전장·입장 준비가 하나도 없고(재시작 직후가 이 상태) 표식이 남아 있을 때만.
//       - 파티장 등록은 등록된 파티 ID 가 표식과 같은 팀만 지운다(수동 등록은 표식과 다르므로 그대로 둔다).
//       - 인원 설정은 표식에 보관한 원래 값으로 되돌린다.
//       - 자동 생성 파티는 재시작 직후에는 맵서버에 파티 정보가 없어 지울 수 없으므로 ID 를 보관했다가,
//         그 파티 구성원이 접속해 파티 정보가 올라오면 이름이 자동 파티 형식("[기욤]" / "[크루아]")일 때만 해체한다.
//       보상·전투 결과·BG 팀은 건드리지 않는다(재시작으로 이미 사라졌고 보상 지급 경로를 부르지 않는다).
// 진입: OnInit(맵서버 시작·@reloadscript) + GM @전장복구 / @bgrecover (GM 60, status / run / destroy).
//       주의: 이 엔진의 @reloadscript 는 mapreg_reload() 로 $@ 임시 변수를 모두 지운다. 그래서 사이클 도중 리로드도
//       재시작과 같은 고아 상태를 만들고(NEED_BG_RM_Core OnInit 의 "리로드 시 사이클 유지" 분기는 실제로 타지 않는다),
//       이 복구가 그때도 실행된다. 리로드·재시작은 기존 전장이 비어 있을 때 한다.
// 로그: debugmes "[NEED_BGUX] RM_ORPHAN_* ..." -> map-msg_log.log
//============================================================

// 랜덤 매칭 소유 표식 기록 (OnFinalize 에서 인원 설정을 덮어쓰기 직전에 부른다)
function	script	F_NEED_BGUX_RmMarkSet	{
	$NEED_BGUX_RM_PG = $@BG_RM_PARTY_G;
	$NEED_BGUX_RM_PC = $@BG_RM_PARTY_C;
	$NEED_BGUX_RM_SAVED = 1;
	$NEED_BGUX_RM_SAVED_U = $@BG_RM_SAVED_USER_COUNT;
	$NEED_BGUX_RM_SAVED_G = $@BG_RM_SAVED_G_COUNT;
	$NEED_BGUX_RM_SAVED_C = $@BG_RM_SAVED_C_COUNT;
	$NEED_BGUX_RM_MARK_AT = gettimetick(2);
	debugmes "[NEED_BGUX] RM_MARK_SET pg=" + $NEED_BGUX_RM_PG + " pc=" + $NEED_BGUX_RM_PC + " saved_u=" + $NEED_BGUX_RM_SAVED_U + " saved_g=" + $NEED_BGUX_RM_SAVED_G + " saved_c=" + $NEED_BGUX_RM_SAVED_C;
	return;
}

// 인원 설정 3개를 DB 에 즉시 남긴다. mapreg 는 0 -> 즉시 DELETE, 새 값 -> 즉시 INSERT 이지만 기존 값 덮어쓰기는
// 300초 자동저장까지 미뤄진다(src/map/mapreg.cpp). 표식(즉시 지워짐)보다 복원값이 늦게 저장되면, 그 사이 강제 종료 때
// 표식 없이 랜덤 매칭 값이 남는다. 그래서 0 을 거쳐 새 값으로 쓴다.
function	script	F_NEED_BGUX_PersistCounts	{
	.@u = $BG_User_Count; .@g = $BG_Guillaume_Count; .@c = $BG_Croix_Count;
	$BG_User_Count = 0; $BG_Guillaume_Count = 0; $BG_Croix_Count = 0;
	$BG_User_Count = .@u; $BG_Guillaume_Count = .@g; $BG_Croix_Count = .@c;
	return;
}

// 정상 종료·취소 경로에서 표식 제거
function	script	F_NEED_BGUX_RmMarkClear	{
	if (!$NEED_BGUX_RM_PG && !$NEED_BGUX_RM_PC && !$NEED_BGUX_RM_SAVED)
		return;
	if ($NEED_BGUX_RM_SAVED)
		callfunc "F_NEED_BGUX_PersistCounts";	// 되돌린 인원 설정을 표식보다 먼저 DB 에 남긴다
	debugmes "[NEED_BGUX] RM_MARK_CLEAR why=" + getarg(0, "normal") + " pg=" + $NEED_BGUX_RM_PG + " pc=" + $NEED_BGUX_RM_PC;
	$NEED_BGUX_RM_PG = 0;
	$NEED_BGUX_RM_PC = 0;
	$NEED_BGUX_RM_SAVED = 0;
	$NEED_BGUX_RM_SAVED_U = 0;
	$NEED_BGUX_RM_SAVED_G = 0;
	$NEED_BGUX_RM_SAVED_C = 0;
	$NEED_BGUX_RM_MARK_AT = 0;
	return;
}

// 지금 랜덤 매칭·전장·입장 준비 중 하나라도 진행 중이면 1 (이때는 절대 복구하지 않는다)
function	script	F_NEED_BGUX_RmLive	{
	if ($@BG_RM_CYCLE_ACTIVE || $@BG_RM_ACTIVE || $@BG_RM_STATE)
		return 1;
	if ($@TierraBG1 || $@FlaviusBG1 || $@KvM01BG)
		return 1;
	if (callfunc("F_BG_RM_ManualReadyActive"))
		return 1;
	return 0;
}

// 자동 생성 파티 이름 형식인지("[기욤]..." / "[크루아]...")
function	script	F_NEED_BGUX_RmAutoPartyName	{
	.@n$ = getarg(0);
	if (.@n$ == "" || .@n$ == "null")
		return 0;
	if (strpos(.@n$, "[기욤]") == 0 || strpos(.@n$, "[크루아]") == 0)
		return 1;
	return 0;
}

// 해체 대기 파티 목록에 추가(중복 없음, 최대 8건; 넘치면 가장 오래된 것부터 버린다)
function	script	F_NEED_BGUX_RmOrphanPush	{
	.@pid = getarg(0);
	if (.@pid <= 0)
		return;
	.@n = getarraysize($NEED_BGUX_ORPHAN_PID);
	for (.@i = 0; .@i < .@n; .@i++)
		if ($NEED_BGUX_ORPHAN_PID[.@i] == .@pid) return;
	if (.@n >= 8) {
		debugmes "[NEED_BGUX] RM_ORPHAN_PARTY_DROP pid=" + $NEED_BGUX_ORPHAN_PID[0] + " why=list_full";
		deletearray $NEED_BGUX_ORPHAN_PID[0], 1;
		deletearray $NEED_BGUX_ORPHAN_T[0], 1;
		.@n--;
	}
	$NEED_BGUX_ORPHAN_PID[.@n] = .@pid;
	$NEED_BGUX_ORPHAN_T[.@n] = gettimetick(2);
	return;
}

function	script	F_NEED_BGUX_RmOrphanRemove	{
	.@n = getarraysize($NEED_BGUX_ORPHAN_PID);
	for (.@i = 0; .@i < .@n; .@i++) {
		if ($NEED_BGUX_ORPHAN_PID[.@i] == getarg(0)) {
			deletearray $NEED_BGUX_ORPHAN_PID[.@i], 1;
			deletearray $NEED_BGUX_ORPHAN_T[.@i], 1;
			return 1;
		}
	}
	return 0;
}

// 해체 대기 파티 하나를 지금 해체해 본다. 1 = 해체함 / 0 = 파티 정보 없음(나중에 다시) / -1 = 자동 파티 아님(목록에서만 뺌)
function	script	F_NEED_BGUX_RmOrphanTryDestroy	{
	.@pid = getarg(0);
	.@name$ = getpartyname(.@pid);
	if (.@name$ == "null")
		return 0;
	if (!callfunc("F_NEED_BGUX_RmAutoPartyName", .@name$)) {
		debugmes "[NEED_BGUX] RM_ORPHAN_PARTY_KEEP pid=" + .@pid + " why=not_auto_name";
		callfunc "F_NEED_BGUX_RmOrphanRemove", .@pid;
		return -1;
	}
	.@r = party_destroy(.@pid);
	debugmes "[NEED_BGUX] RM_ORPHAN_PARTY_DESTROY pid=" + .@pid + " result=" + .@r + " via=" + getarg(1, "");
	callfunc "F_NEED_BGUX_RmOrphanRemove", .@pid;
	return 1;
}

// 고아 상태 복구. getarg(0) = 호출 경로(init / gm). 반환: 0 = 할 일 없음 / 1 = 복구함 / -1 = 진행 중이라 건너뜀
function	script	F_NEED_BGUX_RmOrphanRecover	{
	.@src$ = getarg(0, "init");
	if (!$NEED_BGUX_RM_PG && !$NEED_BGUX_RM_PC && !$NEED_BGUX_RM_SAVED)
		return 0;
	if (callfunc("F_NEED_BGUX_RmLive")) {
		debugmes "[NEED_BGUX] RM_ORPHAN_SKIP src=" + .@src$ + " why=live cycle=" + $@BG_RM_CYCLE_ACTIVE + " state=" + $@BG_RM_STATE + " tierra=" + $@TierraBG1 + " flavius=" + $@FlaviusBG1 + " kvm=" + $@KvM01BG;
		return -1;
	}
	.@pg = $NEED_BGUX_RM_PG;
	.@pc = $NEED_BGUX_RM_PC;
	debugmes "[NEED_BGUX] RM_ORPHAN_FOUND src=" + .@src$ + " mark_pg=" + .@pg + " mark_pc=" + .@pc + " reg_g=" + $BG_GUILLAUME_PARTY_ID + "/" + $BG_GUILLAUME_LEADER_CID
		+ " reg_c=" + $BG_CROIX_PARTY_ID + "/" + $BG_CROIX_LEADER_CID + " counts=" + $BG_User_Count + "/" + $BG_Guillaume_Count + "/" + $BG_Croix_Count + " marked_at=" + $NEED_BGUX_RM_MARK_AT;

	.@cleared = 0;
	.@cg = 0;
	.@cc = 0;
	if (.@pg > 0 && $BG_GUILLAUME_PARTY_ID == .@pg) {
		debugmes "[NEED_BGUX] RM_ORPHAN_CLEAR team=G party=" + $BG_GUILLAUME_PARTY_ID + " leader_cid=" + $BG_GUILLAUME_LEADER_CID + " leader_aid=" + $BG_GUILLAUME_LEADER_AID;
		$BG_GUILLAUME_LEADER_CID = 0;
		$BG_GUILLAUME_LEADER_AID = 0;
		$BG_GUILLAUME_PARTY_ID = 0;
		$BG_GUILLAUME_LEADER_NAME$ = "";
		.@cleared++;
		.@cg = 1;
	}
	else if ($BG_GUILLAUME_PARTY_ID)
		debugmes "[NEED_BGUX] RM_ORPHAN_KEEP team=G party=" + $BG_GUILLAUME_PARTY_ID + " why=not_marked_auto";
	if (.@pc > 0 && $BG_CROIX_PARTY_ID == .@pc) {
		debugmes "[NEED_BGUX] RM_ORPHAN_CLEAR team=C party=" + $BG_CROIX_PARTY_ID + " leader_cid=" + $BG_CROIX_LEADER_CID + " leader_aid=" + $BG_CROIX_LEADER_AID;
		$BG_CROIX_LEADER_CID = 0;
		$BG_CROIX_LEADER_AID = 0;
		$BG_CROIX_PARTY_ID = 0;
		$BG_CROIX_LEADER_NAME$ = "";
		.@cleared++;
		.@cc = 1;
	}
	else if ($BG_CROIX_PARTY_ID)
		debugmes "[NEED_BGUX] RM_ORPHAN_KEEP team=C party=" + $BG_CROIX_PARTY_ID + " why=not_marked_auto";
	// 인원 부족 완화 신청은 파티장 등록에 딸린 상태라, 남은 등록이 하나도 없을 때만 지운다
	if (.@cleared && !$BG_GUILLAUME_PARTY_ID && !$BG_CROIX_PARTY_ID) {
		$BG_SHORT_PENDING = 0;
		$BG_SHORT_REQ_TEAM$ = "";
		$BG_SHORT_TARGET_TEAM$ = "";
		$BG_SHORT_REQ_NAME$ = "";
		$BG_SHORT_REQ_TIME = 0;
	}

	if ($NEED_BGUX_RM_SAVED) {
		debugmes "[NEED_BGUX] RM_ORPHAN_COUNTS restore " + $BG_User_Count + "/" + $BG_Guillaume_Count + "/" + $BG_Croix_Count
			+ " -> " + $NEED_BGUX_RM_SAVED_U + "/" + $NEED_BGUX_RM_SAVED_G + "/" + $NEED_BGUX_RM_SAVED_C;
		$BG_User_Count = $NEED_BGUX_RM_SAVED_U;
		$BG_Guillaume_Count = $NEED_BGUX_RM_SAVED_G;
		$BG_Croix_Count = $NEED_BGUX_RM_SAVED_C;
	}

	// 자동 생성 파티: 등록을 실제로 지운 팀의 파티만. 지금 해체할 수 있으면 해체, 아니면(재시작 직후) 구성원 접속 때 해체하도록 보관.
	// 등록과 맞지 않는 표식(이미 끝났거나 다른 등록)은 파티를 건드리지 않는다.
	if (.@cg) {
		callfunc "F_NEED_BGUX_RmOrphanPush", .@pg;
		callfunc "F_NEED_BGUX_RmOrphanTryDestroy", .@pg, .@src$;
	}
	if (.@cc) {
		callfunc "F_NEED_BGUX_RmOrphanPush", .@pc;
		callfunc "F_NEED_BGUX_RmOrphanTryDestroy", .@pc, .@src$;
	}

	callfunc "F_NEED_BGUX_RmMarkClear", "orphan_" + .@src$;
	debugmes "[NEED_BGUX] RM_ORPHAN_DONE src=" + .@src$ + " cleared_teams=" + .@cleared + " pending_parties=" + getarraysize($NEED_BGUX_ORPHAN_PID);
	return 1;
}

OnInit:
	bindatcmd "전장복구", strnpcinfo(3) + "::OnCmd", 60, 60;
	bindatcmd "bgrecover", strnpcinfo(3) + "::OnCmd", 60, 60;	// NEED EN: English name of @전장복구
	// 14일 지난 해체 대기 파티는 목록에서만 뺀다(구성원이 오래 접속하지 않은 경우)
	for (.@i = getarraysize($NEED_BGUX_ORPHAN_PID) - 1; .@i >= 0; .@i--) {
		if ($NEED_BGUX_ORPHAN_T[.@i] + 1209600 < gettimetick(2)) {
			debugmes "[NEED_BGUX] RM_ORPHAN_PARTY_EXPIRE pid=" + $NEED_BGUX_ORPHAN_PID[.@i];
			deletearray $NEED_BGUX_ORPHAN_PID[.@i], 1;
			deletearray $NEED_BGUX_ORPHAN_T[.@i], 1;
		}
	}
	callfunc "F_NEED_BGUX_RmOrphanRecover", "init";
	end;

// 해체 대기 파티 구성원이 접속하면, 파티 정보가 올라온 뒤(3초) 해체한다
OnPCLoginEvent:
	if (!getarraysize($NEED_BGUX_ORPHAN_PID) || !getcharid(1))
		end;
	addtimer 3000, strnpcinfo(3) + "::OnLoginCheck";
	end;

OnLoginCheck:
	.@pid = getcharid(1);
	if (!.@pid)
		end;
	.@n = getarraysize($NEED_BGUX_ORPHAN_PID);
	for (.@i = 0; .@i < .@n; .@i++) {
		if ($NEED_BGUX_ORPHAN_PID[.@i] != .@pid)
			continue;
		if (callfunc("F_NEED_BGUX_RmLive")) {	// 새 사이클이 같은 ID 를 쓸 일은 없지만, 진행 중에는 손대지 않는다
			debugmes "[NEED_BGUX] RM_ORPHAN_PARTY_WAIT pid=" + .@pid + " why=live";
			end;
		}
		if (callfunc("F_NEED_BGUX_RmOrphanTryDestroy", .@pid, "login aid=" + getcharid(3)) == 1)
			dispbottom needtr("[전장] 서버 점검 전 랜덤 매칭에서 자동으로 만들어진 파티를 정리했습니다. 이제 @전장대기를 다시 이용할 수 있습니다.",
				"[Battleground] The party created automatically by a random match before the server restart was disbanded. You can use @bgqueue again."), 0x99DDFF;
		end;
	}
	end;

OnCmd:
	if (getgmlevel() < 60)
		end;
	.@sub$ = (.@atcmd_numparameters > 0) ? .@atcmd_parameters$[0] : "status";
	if (.@sub$ == "run") {
		.@r = callfunc("F_NEED_BGUX_RmOrphanRecover", "gm aid=" + getcharid(3));
		dispbottom "[BG recover] run result=" + .@r + " (1 = recovered, 0 = nothing marked, -1 = a match/cycle is live, skipped)";
	}
	else if (.@sub$ == "destroy") {	// 해체 대기 파티를 지금 다시 시도(구성원 접속 중일 때만 가능)
		for (.@i = getarraysize($NEED_BGUX_ORPHAN_PID) - 1; .@i >= 0; .@i--)
			dispbottom "[BG recover] destroy pid=" + $NEED_BGUX_ORPHAN_PID[.@i] + " result=" + callfunc("F_NEED_BGUX_RmOrphanTryDestroy", $NEED_BGUX_ORPHAN_PID[.@i], "gm");
	}
	else if (.@sub$ != "status") {
		dispbottom "[BG recover] usage: @bgrecover [status|run|destroy]";
		end;
	}
	dispbottom "[BG recover] live=" + callfunc("F_NEED_BGUX_RmLive") + " cycle=" + $@BG_RM_CYCLE_ACTIVE + " state=" + $@BG_RM_STATE + " stage=" + $@BG_RM_CYCLE_STAGE
		+ " tierra=" + $@TierraBG1 + " flavius=" + $@FlaviusBG1 + " kvm=" + $@KvM01BG + " queue=" + $@BG_RM_QUEUE_COUNT + " lock=" + $@BG_RM_QUEUE_LOCK;
	dispbottom "[BG recover] leader G party=" + $BG_GUILLAUME_PARTY_ID + " cid=" + $BG_GUILLAUME_LEADER_CID + " name=" + $BG_GUILLAUME_LEADER_NAME$
		+ " / C party=" + $BG_CROIX_PARTY_ID + " cid=" + $BG_CROIX_LEADER_CID + " name=" + $BG_CROIX_LEADER_NAME$;
	dispbottom "[BG recover] mark pg=" + $NEED_BGUX_RM_PG + " pc=" + $NEED_BGUX_RM_PC + " saved=" + $NEED_BGUX_RM_SAVED + " (" + $NEED_BGUX_RM_SAVED_U + "/" + $NEED_BGUX_RM_SAVED_G + "/" + $NEED_BGUX_RM_SAVED_C + ")"
		+ " counts=" + $BG_User_Count + "/" + $BG_Guillaume_Count + "/" + $BG_Croix_Count + " pending_parties=" + getarraysize($NEED_BGUX_ORPHAN_PID);
	if (!$NEED_BGUX_RM_PG && !$NEED_BGUX_RM_PC && ($BG_GUILLAUME_PARTY_ID || $BG_CROIX_PARTY_ID) && !callfunc("F_NEED_BGUX_RmLive"))
		dispbottom "[BG recover] leaders are registered without an auto-match mark: a manual registration, or one left from before this fix. Reset it at the bat_room Battleground Manager > Reset leader registrations if it is stale.";
	end;

// 2026-11-12 NEED_BGUX 기존 전장 통합 모집 보조 함수
//
// 상태(기존 변수 재사용, 새 큐·타이머 없음):
//   WAITING    $@BG_RM_STATE 0, 대기열 $@BG_RM_QUEUE_* < 최소 인원
//   PRE_WARP   $@BG_RM_QUEUE_LOCK 1, $@BG_RM_QUEUE_READY_TICK (이동 전 안내 $@BG_RM_PREWARP_SECONDS)
//   RECRUITING $@BG_RM_STATE 1, 모집 명단 $@BG_RM_AID/CID/..., 마감 $@BG_RM_END_TICK = 소환 완료 + .RecruitSeconds
//   LOCKED → TEAM_ASSIGN  $@BG_RM_STATE 2 (OnFinalize 첫 줄), 이후 원래 팀 편성·경기·정리 그대로
// 원격 소환된 참가자: 세션 변수 @NEED_BGUX_SUM_TOKEN(= 그 모집의 $@BG_RM_TOKEN) / _MAP$ / _X / _Y.
// 모집이 취소되면 그 사람만 원래 위치로(안 되면 저장 지점) 돌려보낸다. 걸어 들어온 사람은 그대로 둔다.
// 영구 변수는 쓰지 않는다: 재시작·리로드 때 모집은 사라지고(P0 복구 대상 아님), 소환된 사람은 bat_room 에 남는다.
//============================================================

// 첨부된 플레이어의 현재 위치를 이번 모집의 "돌아갈 곳"으로 기록
function	script	F_NEED_BGUX_MarkSummon	{
	getmapxy(.@m$, .@x, .@y, BL_PC);
	@NEED_BGUX_SUM_MAP$ = .@m$;
	@NEED_BGUX_SUM_X = .@x;
	@NEED_BGUX_SUM_Y = .@y;
	@NEED_BGUX_SUM_TOKEN = $@BG_RM_TOKEN;
	return;
}

// 첨부된 플레이어가 이번 모집으로 원격 소환된 사람이고 아직 bat_room 에 있으면 원래 위치(안 되면 저장 지점)로 보낸다.
// 반환 1 = 보냄 / 0 = 해당 없음
function	script	F_NEED_BGUX_ReturnSelf	{
	if (@NEED_BGUX_SUM_TOKEN == 0 || @NEED_BGUX_SUM_TOKEN != $@BG_RM_TOKEN || strcharinfo(3) != "bat_room") {
		@NEED_BGUX_SUM_TOKEN = 0;
		return 0;
	}
	.@m$ = @NEED_BGUX_SUM_MAP$;
	.@x = @NEED_BGUX_SUM_X;
	.@y = @NEED_BGUX_SUM_Y;
	@NEED_BGUX_SUM_TOKEN = 0;
	@NEED_BGUX_SUM_MAP$ = "";
	.@ok = 0;
	if (.@m$ != "" && .@m$ != "bat_room" && .@m$ != "bat_a01" && .@m$ != "bat_b01" && .@m$ != "bat_c01" && !compare(.@m$, "@")) {
		if (getmapusers(.@m$) >= 0) {
			if (!getmapflag(.@m$, mf_battleground) && !getmapflag(.@m$, mf_pvp) && !getmapflag(.@m$, mf_gvg)) {
				if (checkcell(.@m$, .@x, .@y, cell_chkpass))
					.@ok = 1;
			}
		}
	}
	debugmes "[NEED_BGUX] RECRUIT_RETURN aid=" + getcharid(3) + " cid=" + getcharid(0) + " why=" + getarg(0, "") + " to=" + (.@ok ? .@m$ + "," + .@x + "," + .@y : "SavePoint");
	if (.@ok)
		warp .@m$, .@x, .@y;
	else
		warp "SavePoint", 0, 0;
	return 1;
}

// 모집 취소 때(F_BG_RM_CancelRecruit 시작): 명단의 접속자 전원에게 재신청 안내, 원격 소환된 사람은 원래 위치로.
function	script	F_NEED_BGUX_RecruitReturnAll	{
	.@caller = playerattached();
	.@back = 0;
	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		.@aid = $@BG_RM_AID[.@i];
		.@cid = $@BG_RM_CID[.@i];
		if (.@aid <= 0 || !isloggedin(.@aid, .@cid))
			continue;
		if (!attachrid(.@aid))
			continue;
		if (getcharid(0) == .@cid) {
			announce needtr("[전장] 이번 모집이 취소되었습니다: ", "[Battleground] This recruitment was cancelled: ") + needtr(getarg(0, ""), getarg(1, "")), bc_self, "0xFF7777";
			dispbottom needtr("[전장] 다시 참가하려면 @전장대기 를 신청해 주세요. (자동으로 다시 등록되지 않습니다)", "[Battleground] To take part again, apply with @bgqueue. (You are not re-registered automatically.)");
			if (callfunc("F_NEED_BGUX_ReturnSelf", "recruit_cancel")) {
				dispbottom needtr("[전장] 전장 대기실로 이동하기 전 위치(또는 저장 지점)로 돌려보냈습니다.", "[Battleground] You were sent back to where you were before the move (or your save point).");
				.@back++;
			}
		}
		detachrid;
	}
	if (.@caller)
		attachrid .@caller;
	debugmes "[NEED_BGUX] RECRUIT_CANCELLED token=" + $@BG_RM_TOKEN + " count=" + $@BG_RM_COUNT + " returned=" + .@back;
	return;
}

// 출발 준비(이동 전 안내) 중 대기열이 최소 인원 미만이 되면 출발을 취소하고 대기로 돌린다(대기는 유지).
// 최소 인원 이상이면 그대로 둔다(예전에는 한 명만 빠져도 20초가 처음부터 다시 시작됐다).
function	script	F_NEED_BGUX_QueueLockCheck	{
	.@min = ($@BG_RM_MIN_PLAYERS > 0) ? $@BG_RM_MIN_PLAYERS : 8;
	if ($@BG_RM_QUEUE_LOCK && $@BG_RM_QUEUE_COUNT < .@min) {
		$@BG_RM_QUEUE_LOCK = 0;
		$@BG_RM_QUEUE_READY_TICK = 0;
		debugmes "[NEED_BGUX] PREWARP_CANCEL count=" + $@BG_RM_QUEUE_COUNT + " min=" + .@min;
		callfunc "F_BG_RM_QueueNotifyAll",
			"[전장] 출발 전 대기 인원이 최소 인원보다 적어져 자동 이동이 취소되었습니다. 대기는 그대로 유지됩니다.",
			"[Battleground] The queue fell below the minimum before departure, so the automatic move was cancelled. You are still in the queue.";
	}
	return;
}

// 추가 모집(RECRUITING) 중 첨부된 플레이어의 신청: 같은 모집 명단에 넣고, bat_room 밖이면 바로 소환한다.
// 자격 검사(동의·레벨·파티/전장 팀·2:2/3:3 신청 중)는 부른 쪽(OnCommandQueue)이 원래대로 먼저 한다.
// 반환 1 = 등록 / 2 = 이미 참가 중 / 0 = 거절
function	script	F_NEED_BGUX_RecruitAdd	{
	.@aid = getcharid(3);
	.@cid = getcharid(0);
	.@max = ($@BG_RM_MAX_PLAYERS > 0) ? $@BG_RM_MAX_PLAYERS : 48;
	.@left = $@BG_RM_END_TICK - gettimetick(2);
	if (.@left < 0)
		.@left = 0;
	if ($@BG_RM_STATE != 1) {
		dispbottom needtr("[전장] 지금은 추가 모집 중이 아닙니다.", "[Battleground] Extra recruitment is not open right now.");
		return 0;
	}
	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++) {
		if ($@BG_RM_AID[.@i] == .@aid) {
			announce needtr("[전장] 이미 이번 추가 모집에 참가 중입니다.", "[Battleground] You are already in this extra recruitment."), bc_self, "0x66FF66";
			dispbottom needtr("[전장] 현재 " + $@BG_RM_COUNT + "/" + .@max + "명 / 마감까지 " + .@left + "초", "[Battleground] " + $@BG_RM_COUNT + "/" + .@max + " players / closes in " + .@left + "s");
			return 2;
		}
	}
	if ($@BG_RM_COUNT >= .@max) {
		dispbottom needtr("[전장] 이번 모집은 정원 " + .@max + "명이 모두 찼습니다. 경기가 시작된 뒤 @전장대기 하면 다음 모집 대기열에 등록됩니다.",
			"[Battleground] This recruitment is full (" + .@max + " players). Use @bgqueue after the match starts to queue for the next one.");
		return 0;
	}
	.@slot = $@BG_RM_COUNT;
	$@BG_RM_AID[.@slot] = .@aid;
	$@BG_RM_CID[.@slot] = .@cid;
	$@BG_RM_CLASS[.@slot] = Class;
	$@BG_RM_NAME$[.@slot] = strcharinfo(0);
	$@BG_RM_COUNT++;
	BG_RM_LAST_CANCEL_REASON = 0;
	BG_RM_LAST_CANCEL_TOKEN = 0;
	BG_RM_LAST_CANCEL_TICK = 0;
	.@moved = 0;
	if (strcharinfo(3) != "bat_room") {
		callfunc "F_NEED_BGUX_MarkSummon";
		.@moved = 1;
	}
	debugmes "[NEED_BGUX] RECRUIT_JOIN aid=" + .@aid + " cid=" + .@cid + " count=" + $@BG_RM_COUNT + " left=" + .@left + " summoned=" + .@moved;
	announce needtr("[전장] 기존 전장 추가 모집에 참가했습니다.", "[Battleground] You joined the classic BG extra recruitment."), bc_self, "0x66FF66";
	dispbottom needtr("[전장] 현재 " + $@BG_RM_COUNT + "/" + .@max + "명 / 마감까지 " + .@left + "초. 마감 후 팀을 편성합니다. 취소: @전장대기취소",
		"[Battleground] " + $@BG_RM_COUNT + "/" + .@max + " players / closes in " + .@left + "s. Teams are formed after it closes. Leave: @bgqueuecancel");
	mapannouncelang "bat_room", "[전장 대기열] " + strcharinfo(0) + "님 참가. 현재 " + $@BG_RM_COUNT + "/" + .@max + "명, 마감까지 " + .@left + "초",
		"[BG Queue] " + strcharinfo(0) + " joined. " + $@BG_RM_COUNT + "/" + .@max + " players, closes in " + .@left + "s", bc_map, "0x99DDFF";
	if (.@moved)
		warp "bat_room", 154, 150;
	return 1;
}

// 기존 전장 통합 상태 한 줄(@전장대기현황 / @전장현황 / 전장 안내 NPC 공용). 첨부된 플레이어의 언어로.
function	script	F_NEED_BGUX_RmStateLine$	{
	.@min = ($@BG_RM_MIN_PLAYERS > 0) ? $@BG_RM_MIN_PLAYERS : 8;
	.@max = ($@BG_RM_MAX_PLAYERS > 0) ? $@BG_RM_MAX_PLAYERS : 48;
	.@now = gettimetick(2);
	if ($@BG_RM_CYCLE_ACTIVE)
		return needtr("경기 진행 중: ", "Match running: ") + callfunc("F_BG_RM_ModeNameLang$", $@BG_RM_MODE, getneedlang())
			+ needtr(" (" + $@BG_RM_CYCLE_STAGE + "/3 단계) / 다음 모집 대기 " + $@BG_RM_QUEUE_COUNT + "명", " (stage " + $@BG_RM_CYCLE_STAGE + "/3) / queued for the next: " + $@BG_RM_QUEUE_COUNT);
	if ($@BG_RM_STATE == 2)
		return needtr("추가 모집 마감 - 팀 편성 중 (" + $@BG_RM_COUNT + "명)", "Extra recruitment closed - forming teams (" + $@BG_RM_COUNT + " players)");
	if ($@BG_RM_STATE == 1) {
		.@left = $@BG_RM_END_TICK - .@now;
		if (.@left < 0)
			.@left = 0;
		return needtr("추가 모집 중 " + $@BG_RM_COUNT + "/" + .@max + "명, 남은 " + .@left + "초 - 마을에서도 @전장대기 로 바로 합류",
			"Extra recruitment " + $@BG_RM_COUNT + "/" + .@max + " players, " + .@left + "s left - join from town with @bgqueue");
	}
	if ($@BG_RM_QUEUE_LOCK && $@BG_RM_QUEUE_READY_TICK > 0) {
		.@left = $@BG_RM_QUEUE_READY_TICK - .@now;
		if (.@left < 0)
			.@left = 0;
		return needtr(.@left + "초 후 전장 대기실 이동 (대기 " + $@BG_RM_QUEUE_COUNT + "명) - 지금 신청하면 함께 이동",
			"Moving to the waiting room in " + .@left + "s (" + $@BG_RM_QUEUE_COUNT + " queued) - apply now to go along");
	}
	if ($@TierraBG1 || $@FlaviusBG1 || $@KvM01BG)
		return needtr("수동(파티장) 전장 진행 중 / 대기 " + $@BG_RM_QUEUE_COUNT + "/" + .@min + "명", "A party-leader battleground is running / queued " + $@BG_RM_QUEUE_COUNT + "/" + .@min);
	if (callfunc("F_BG_RM_ManualReadyActive"))
		return needtr("수동 전장 입장 준비 중 - 자동매칭 출발 보류 / 대기 " + $@BG_RM_QUEUE_COUNT + "/" + .@min + "명", "Manual BG entry preparation is active - auto-match departure is on hold / queued " + $@BG_RM_QUEUE_COUNT + "/" + .@min);
	if ($BG_GUILLAUME_PARTY_ID || $BG_CROIX_PARTY_ID)
		return needtr("수동 전장 파티장 등록으로 자동매칭 출발 보류 / 대기 " + $@BG_RM_QUEUE_COUNT + "/" + .@min + "명. 수동 준비가 아니라면 전장 관리인에서 등록을 해제하거나 '파티장 등록 초기화'를 이용해 주세요.", "Manual BG team leaders are registered, so auto-match departure is on hold / queued " + $@BG_RM_QUEUE_COUNT + "/" + .@min + ". If no manual match is planned, release it or use 'Reset leader registrations' at the Battleground Manager.");
	return needtr("기존 전장 대기 " + $@BG_RM_QUEUE_COUNT + "/" + .@min + "명", "Classic BG queue " + $@BG_RM_QUEUE_COUNT + "/" + .@min);
}
