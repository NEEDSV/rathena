function	script	F_KvM_NormalizeCount	{
	if ($BG_User_Count <= 0)
		$BG_User_Count = 5;

	if ($BG_Guillaume_Count <= 0)
		$BG_Guillaume_Count = $BG_User_Count;

	if ($BG_Croix_Count <= 0)
		$BG_Croix_Count = $BG_User_Count;

	if ($BG_Guillaume_Count < 1)
		$BG_Guillaume_Count = 1;

	if ($BG_Croix_Count < 1)
		$BG_Croix_Count = 1;

	return;
}

function	script	F_KvM_GetGuillaumeNeed	{
	callfunc "F_KvM_NormalizeCount";
	return $BG_Guillaume_Count;
}

function	script	F_KvM_GetCroixNeed	{
	callfunc "F_KvM_NormalizeCount";
	return $BG_Croix_Count;
}

function	script	F_KvM_AnnounceWaitingRoom	{
	callfunc "F_KvM_NormalizeCount";

	.@g = getwaitingroomstate(0,"KvM01R_Guillaume");
	.@c = getwaitingroomstate(0,"KvM01R_Croix");

	areaannounce "bat_room",159,219,179,233,

		bc_area,"0x99CCFF";

	areaannounce "bat_room",159,198,179,212,
		"[KVM] 현재 대기 인원 - 기욤 " + .@g + "/" + $BG_Guillaume_Count + "명, 크루아 " + .@c + "/" + $BG_Croix_Count + "명",
		bc_area,"0x99CCFF";

	return;
}

function	script	F_KvM_SetPartyTeam	{
	.@party_id = getarg(0);
	.@bg_id = getarg(1);
	.@team = getarg(2);

	getpartymember .@party_id, 1;
	.@count = $@partymembercount;
	copyarray .@cid[0], $@partymembercid[0], .@count;
	getpartymember .@party_id, 2;
	copyarray .@aid[0], $@partymemberaid[0], .@count;

	for (.@i = 0; .@i < .@count; .@i++) {
		if (!isloggedin(.@aid[.@i], .@cid[.@i]))
			continue;
		if (!attachrid(.@aid[.@i]))
			continue;
		if (getcharid(0) != .@cid[.@i]) {
			detachrid;
			continue;
		}
		if (getcharid(4) == .@bg_id) {
			Bat_Team = .@team;
			percentheal 100,100;
		}
		detachrid;
	}
	return;
}

function	script	F_KvM_RollbackPartyBG	{
	.@party_id = getarg(0);
	.@bg_id = getarg(1);

	if (.@party_id <= 0)
		return;

	getpartymember .@party_id, 1;
	.@count = $@partymembercount;
	copyarray .@cid[0], $@partymembercid[0], .@count;
	getpartymember .@party_id, 2;
	copyarray .@aid[0], $@partymemberaid[0], .@count;

	for (.@i = 0; .@i < .@count; .@i++) {
		setpcblock PCBLOCK_MOVE|PCBLOCK_USEITEM|PCBLOCK_COMMANDS|PCBLOCK_NPCCLICK, false, .@aid[.@i];
		if (!isloggedin(.@aid[.@i], .@cid[.@i]))
			continue;
		if (!attachrid(.@aid[.@i]))
			continue;
		if (getcharid(0) != .@cid[.@i]) {
			detachrid;
			continue;
		}
		if (.@bg_id && getcharid(4) == .@bg_id)
			bg_leave;
		Bat_Team = 0;
		warp "bat_room",154,150;
		detachrid;
	}
	return;
}

function	script	F_BG_ClearAllRegisterAfterKVM	{
	// 랜덤 매칭 풀사이클은 애드온 종료 감시에서 파티/등록/인원 설정을 함께 정리합니다.
	if ($@BG_RM_CYCLE_ACTIVE)
		return;

	// 파티장 등록 초기화
	$BG_GUILLAUME_LEADER_CID = 0;
	$BG_GUILLAUME_LEADER_AID = 0;
	$BG_GUILLAUME_PARTY_ID = 0;
	$BG_GUILLAUME_LEADER_NAME$ = "";

	$BG_CROIX_LEADER_CID = 0;
	$BG_CROIX_LEADER_AID = 0;
	$BG_CROIX_PARTY_ID = 0;
	$BG_CROIX_LEADER_NAME$ = "";

	// 부족 인원 완화 신청 상태 초기화
	$BG_SHORT_PENDING = 0;
	$BG_SHORT_REQ_TEAM$ = "";
	$BG_SHORT_TARGET_TEAM$ = "";
	$BG_SHORT_REQ_NAME$ = "";
	$BG_SHORT_REQ_TIME = 0;

	// 팀별 완화 인원 초기화
	// 기본 인원은 유지하고, 팀별 필요 인원만 기본값으로 복구
	if ($BG_User_Count > 0) {
		$BG_Guillaume_Count = $BG_User_Count;
		$BG_Croix_Count = $BG_User_Count;
	} else {
		$BG_User_Count = 10;
		$BG_Guillaume_Count = 10;
		$BG_Croix_Count = 10;
	}

	return;
}
