function	script	F_Flavius_NormalizeCount	{
	if ($BG_User_Count <= 0)
		$BG_User_Count = 10;

	if ($BG_Guillaume_Count <= 0)
		$BG_Guillaume_Count = $BG_User_Count;

	if ($BG_Croix_Count <= 0)
		$BG_Croix_Count = $BG_User_Count;

	if ($BG_Guillaume_Count < 1)
		$BG_Guillaume_Count = 1;

	if ($BG_Croix_Count < 1)
		$BG_Croix_Count = 1;

	return;
}

function	script	F_Flavius_GetGuillaumeNeed	{
	callfunc "F_Flavius_NormalizeCount";
	return $BG_Guillaume_Count;
}

function	script	F_Flavius_GetCroixNeed	{
	callfunc "F_Flavius_NormalizeCount";
	return $BG_Croix_Count;
}

function	script	F_Flavius_AnnounceRooms	{
	callfunc "F_Flavius_NormalizeCount";

	.@guillaume = getwaitingroomstate(0,"기욤 대기실#bat_b01");
	.@croix = getwaitingroomstate(0,"크루아 대기실#bat_b01");

	.@msg$ = "[플라비우스] 현재 대기 인원 - 기욤 " + .@guillaume + "/" + $BG_Guillaume_Count + "명, 크루아 " + .@croix + "/" + $BG_Croix_Count + "명";

	// 기욤 대기방 내부 안내
	areaannounce "bat_room",75,220,96,234,.@msg$,bc_map,"0x66CCFF";

	// 크루아 대기방 내부 안내
	areaannounce "bat_room",75,197,96,211,.@msg$,bc_map,"0xFF99CC";
	return;
}

//============================================================
// 플라비우스 자동 입장용 BG 직접 가입 함수
//============================================================
function	script	F_Flavius_JoinPartyToBG	{
	.@party_id = getarg(0);
	.@bg_id = getarg(1);
	.@team = getarg(2);
	.@map$ = getarg(3);
	.@x = getarg(4);
	.@y = getarg(5);

	getpartymember .@party_id, 0;
	.@count = $@partymembercount;
	copyarray .@name$[0], $@partymembername$[0], .@count;
	getpartymember .@party_id, 1;
	copyarray .@cid[0], $@partymembercid[0], .@count;
	getpartymember .@party_id, 2;
	copyarray .@aid[0], $@partymemberaid[0], .@count;

	for (.@i = 0; .@i < .@count; .@i++) {
		// 준비 잠금은 자동 입장 직전에 해제합니다.
		setpcblock PCBLOCK_MOVE|PCBLOCK_USEITEM|PCBLOCK_COMMANDS|PCBLOCK_NPCCLICK, false, .@aid[.@i];

		if (!isloggedin(.@aid[.@i], .@cid[.@i])) {
			$@FlaviusJoin_FailName$ = .@name$[.@i];
			$@FlaviusJoin_FailReason$ = "입장 직전에 접속이 종료되었습니다.";
			return 0;
		}

		if (!bg_join(.@bg_id, .@map$, .@x, .@y, .@cid[.@i])) {
			$@FlaviusJoin_FailName$ = .@name$[.@i];
			$@FlaviusJoin_FailReason$ = "BG 팀 가입 명령이 실패했습니다.";
			return 0;
		}

		if (!attachrid(.@aid[.@i]) || getcharid(0) != .@cid[.@i]) {
			$@FlaviusJoin_FailName$ = .@name$[.@i];
			$@FlaviusJoin_FailReason$ = "가입 후 캐릭터 정보를 확인할 수 없습니다.";
			return 0;
		}

		if (getcharid(4) != .@bg_id) {
			$@FlaviusJoin_FailName$ = strcharinfo(0);
			$@FlaviusJoin_FailReason$ = "가입된 BG 팀 ID가 일치하지 않습니다.";
			detachrid;
			return 0;
		}

		Bat_Team = .@team;
		percentheal 100,100;
		detachrid;
	}

	return .@count;
}

function	script	F_Flavius_RollbackPartyBG	{
	.@party_id = getarg(0);
	.@bg_id = getarg(1);

	if (.@party_id <= 0)
		return;

	getpartymember .@party_id, 1;
	.@count = $@partymembercount;
	copyarray .@cid[0], $@partymembercid[0], .@count;
	getpartymember .@party_id, 2;
	copyarray .@aid[0], $@partymemberaid[0], .@count;

	for (.@i = 0; .@i < .@count; .@i++) {
		setpcblock PCBLOCK_MOVE|PCBLOCK_USEITEM|PCBLOCK_COMMANDS|PCBLOCK_NPCCLICK, false, .@aid[.@i];
		if (!isloggedin(.@aid[.@i], .@cid[.@i]))
			continue;
		if (!attachrid(.@aid[.@i]) || getcharid(0) != .@cid[.@i])
			continue;
		if (.@bg_id && getcharid(4) == .@bg_id)
			bg_leave;
		Bat_Team = 0;
		warp "bat_room",154,150;
		detachrid;
	}
	return;
}
