//============================================================
// 플라비우스 공통 함수
//============================================================
function	script	F_Flavius_NormalizeCount	{
	if ($BG_User_Count <= 0)
		$BG_User_Count = 10;

	if ($BG_Guillaume_Count <= 0)
		$BG_Guillaume_Count = $BG_User_Count;

	if ($BG_Croix_Count <= 0)
		$BG_Croix_Count = $BG_User_Count;

	if ($BG_Guillaume_Count < 1)
		$BG_Guillaume_Count = 1;

	if ($BG_Croix_Count < 1)
		$BG_Croix_Count = 1;

	return;
}

function	script	F_Flavius_GetGuillaumeNeed	{
	callfunc "F_Flavius_NormalizeCount";
	return $BG_Guillaume_Count;
}

function	script	F_Flavius_GetCroixNeed	{
	callfunc "F_Flavius_NormalizeCount";
	return $BG_Croix_Count;
}

bat_room,2,151,3	script	#bat_b01_timer	844,{
	end;

OnEnable:
	initnpctimer;
	end;

OnStop:
	stopnpctimer;
	end;

OnTimer1000:
	stopnpctimer;

	if ($@FlaviusBG1 && bg_get_data($@FlaviusBG1_id1,0) == 0 && bg_get_data($@FlaviusBG1_id2,0) == 0) {
		.@completed = ($@FlaviusBG1_Victory > 0);

		donpcevent "start#bat_b01::OnClean";

		$@FlaviusBG1 = 0;
		$@FlaviusBG1_Victory = 0;
		$@Croix_ScoreBG1 = 0;
		$@Guill_ScoreBG1 = 0;

		if ($@FlaviusBG1_id1) {
			bg_destroy $@FlaviusBG1_id1;
			$@FlaviusBG1_id1 = 0;
		}

		if ($@FlaviusBG1_id2) {
			bg_destroy $@FlaviusBG1_id2;
			$@FlaviusBG1_id2 = 0;
		}

		if (.@completed)
			callfunc "F_BG_AdvanceRotation",2;
		end;
	}

	initnpctimer;
	end;
}
