// 전장 관리 메뉴 - 주사위 굴리기
// 기욤 / 크루아에 등록된 현재 파티장만 가능
// 결과는 bat_room 맵 전체에 알림

function	script	F_BG_Menu_DiceRoll	{
	.@my_team$ = callfunc("F_BG_GetMyRegisteredTeam$");

	if (.@my_team$ == "") {
		mes "[전장 관리인]";
		mes "주사위는 등록된 기욤 또는 크루아 팀 파티장만 굴릴 수 있습니다.";
		mes "현재 파티의 파티장 상태와 등록 정보를 확인해주세요.";
		close;
	}

	.@dice = rand(1, 6);

	mapannounce "bat_room",
		"[전장 주사위] " + .@my_team$ + " 팀 파티장 " + strcharinfo(0) + "님 주사위 [" + .@dice + "]",
		bc_map, 0x00FFFF;

	mes "[전장 관리인]";
	mes "주사위를 굴렸습니다.";
	mes "결과: ^FF0000" + .@dice + "^000000";
	close;
}

//============================================================
// 전장 파티장 등록
// 파티명 중복 문제 방지를 위해
// 파티ID + 캐릭터ID + 계정ID 기준으로 등록/검증
//============================================================
function	script	F_BG_Menu_RegisterLeader	{
	.@party_id = getcharid(1);

	if ($@BG_READY_TYPE) {
		mes "[전장 관리인]";
		mes "전장 입장 준비가 진행 중이라 등록 정보를 변경할 수 없습니다.";
		close;
	}

	if ($@TierraBG1 || $@FlaviusBG1 || $@KvM01BG || $@BG_READY_TYPE)
	{
		mes "[전장 관리인]";
		mes "진행 중엔 등록할 수 없습니다.";
		close;
	}
	
	if (.@party_id <= 0) {
		mes "[전장 관리인]";
		mes "파티에 가입된 상태에서만 등록할 수 있습니다.";
		close;
	}

	if (getpartyleader(.@party_id, 0) != strcharinfo(0)) {
		mes "[전장 관리인]";
		mes "전장 파티장 등록은 현재 파티의 파티장만 가능합니다.";
		close;
	}

	mes "[전장 관리인]";
	mes "전장 팀 파티장을 등록합니다.";
	mes "파티명은 중복될 수 있으므로,";
	mes "등록된 파티ID와 캐릭터ID를 기준으로 검증합니다.";
	next;

	mes "[현재 등록 정보]";
	mes "기욤 파티장: " + (($BG_GUILLAUME_LEADER_NAME$ == "") ? "^777777미등록^000000" : "^0055FF" + $BG_GUILLAUME_LEADER_NAME$ + "^000000");
	mes "크루아 파티장: " + (($BG_CROIX_LEADER_NAME$ == "") ? "^777777미등록^000000" : "^0055FF" + $BG_CROIX_LEADER_NAME$ + "^000000");
	next;

	switch(select("기욤 파티장으로 등록:크루아 파티장으로 등록:내 등록 해제:관리자 초기화:취소")) {
	case 1:
		callfunc "F_BG_RegisterLeader", "기욤";
		close;

	case 2:
		callfunc "F_BG_RegisterLeader", "크루아";
		close;

	case 3:
		callfunc "F_BG_UnregisterMyLeader";
		close;

	case 4:
		if (getgmlevel() < 60) {
			mes "[전장 관리인]";
			mes "관리자만 사용할 수 있습니다.";
			close;
		}

		mes "[전장 관리인]";
		mes "전장 파티장 등록 정보를 모두 초기화하시겠습니까?";
		next;

		if (select("초기화한다:취소한다") == 2) {
			mes "[전장 관리인]";
			mes "취소했습니다.";
			close;
		}

		callfunc "F_BG_ClearLeaderRegister";

		mes "[전장 관리인]";
		mes "전장 파티장 등록 정보가 초기화되었습니다.";
		close;

	case 5:
		mes "[전장 관리인]";
		mes "취소했습니다.";
		close;
	}
}

//============================================================
// 전장 파티장 등록 처리
// getarg(0) = "기욤" or "크루아"
//============================================================
function	script	F_BG_RegisterLeader	{
	.@team$ = getarg(0);
	.@party_id = getcharid(1);
	.@char_id = getcharid(0);
	.@account_id = getcharid(3);
	.@name$ = strcharinfo(0);

	if (.@party_id <= 0) {
		mes "[전장 관리인]";
		mes "파티에 가입된 상태에서만 등록할 수 있습니다.";
		close;
	}

	if (getpartyleader(.@party_id, 0) != .@name$) {
		mes "[전장 관리인]";
		mes "현재 파티의 파티장만 등록할 수 있습니다.";
		close;
	}

	.@my_team$ = callfunc("F_BG_GetMyRegisteredTeam$");

	if (.@my_team$ != "" && .@my_team$ != .@team$) {
		mes "[전장 관리인]";
		mes "이미 ^0055FF" + .@my_team$ + "^000000 팀 파티장으로 등록되어 있습니다.";
		mes "먼저 기존 등록을 해제해주세요.";
		close;
	}

	if (.@team$ == "기욤") {
		if ($BG_GUILLAUME_LEADER_CID > 0 && $BG_GUILLAUME_LEADER_CID != .@char_id) {
			mes "[전장 관리인]";
			mes "기욤 팀 파티장이 이미 등록되어 있습니다.";
			mes "등록자: ^0055FF" + $BG_GUILLAUME_LEADER_NAME$ + "^000000";
			mes "초기화가 필요하면 관리자에게 문의해주세요.";
			close;
		}

		$BG_GUILLAUME_LEADER_CID = .@char_id;
		$BG_GUILLAUME_LEADER_AID = .@account_id;
		$BG_GUILLAUME_PARTY_ID = .@party_id;
		$BG_GUILLAUME_LEADER_NAME$ = .@name$;
	}
	else if (.@team$ == "크루아") {
		if ($BG_CROIX_LEADER_CID > 0 && $BG_CROIX_LEADER_CID != .@char_id) {
			mes "[전장 관리인]";
			mes "크루아 팀 파티장이 이미 등록되어 있습니다.";
			mes "등록자: ^0055FF" + $BG_CROIX_LEADER_NAME$ + "^000000";
			mes "초기화가 필요하면 관리자에게 문의해주세요.";
			close;
		}

		$BG_CROIX_LEADER_CID = .@char_id;
		$BG_CROIX_LEADER_AID = .@account_id;
		$BG_CROIX_PARTY_ID = .@party_id;
		$BG_CROIX_LEADER_NAME$ = .@name$;
	}
	else {
		mes "[전장 관리인]";
		mes "알 수 없는 팀입니다.";
		close;
	}

	announce "[전장 안내] " + .@name$ + "님이 " + .@team$ + " 팀 파티장으로 등록되었습니다.", bc_map;

	mes "[전장 관리인]";
	mes "^0055FF" + .@team$ + "^000000 팀 파티장으로 등록되었습니다.";
	mes "이제 전장 인원 조정 및 인원 부족 완화 신청을 사용할 수 있습니다.";
	close;
}

//============================================================
function	script	F_BG_UnregisterMyLeader	{
	callfunc "F_BG_ClearAllReadyState";

	.@char_id = getcharid(0);
	.@account_id = getcharid(3);
	.@removed = 0;

	if ($BG_GUILLAUME_LEADER_CID == .@char_id && $BG_GUILLAUME_LEADER_AID == .@account_id) {
		$BG_GUILLAUME_LEADER_CID = 0;
		$BG_GUILLAUME_LEADER_AID = 0;
		$BG_GUILLAUME_PARTY_ID = 0;
		$BG_GUILLAUME_LEADER_NAME$ = "";
		.@removed = 1;
	}

	if ($BG_CROIX_LEADER_CID == .@char_id && $BG_CROIX_LEADER_AID == .@account_id) {
		$BG_CROIX_LEADER_CID = 0;
		$BG_CROIX_LEADER_AID = 0;
		$BG_CROIX_PARTY_ID = 0;
		$BG_CROIX_LEADER_NAME$ = "";
		.@removed = 1;
	}

	if (!.@removed) {
		mes "[전장 관리인]";
		mes "현재 등록된 전장 파티장 정보가 없습니다.";
		close;
	}

	announce "[전장 안내] " + strcharinfo(0) + "님의 전장 파티장 등록이 해제되었습니다.", bc_map;

	mes "[전장 관리인]";
	mes "전장 파티장 등록이 해제되었습니다.";
	close;
}

//============================================================
// 현재 유저가 등록된 전장 파티장인지 확인
// return: "기욤", "크루아", ""
//============================================================
function	script	F_BG_GetMyRegisteredTeam$	{
	.@party_id = getcharid(1);
	.@char_id = getcharid(0);
	.@account_id = getcharid(3);
	.@name$ = strcharinfo(0);

	if (.@party_id <= 0)
		return "";

	// 현재 파티장인지도 다시 확인
	if (getpartyleader(.@party_id, 0) != .@name$)
		return "";

	if (
		$BG_GUILLAUME_LEADER_CID == .@char_id &&
		$BG_GUILLAUME_LEADER_AID == .@account_id &&
		$BG_GUILLAUME_PARTY_ID == .@party_id
	) {
		return "기욤";
	}

	if (
		$BG_CROIX_LEADER_CID == .@char_id &&
		$BG_CROIX_LEADER_AID == .@account_id &&
		$BG_CROIX_PARTY_ID == .@party_id
	) {
		return "크루아";
	}

	return "";
}

//============================================================
// 전장 파티장 준비 시스템 - 티에라
//============================================================

// 파티 검사 결과는 다음 전역 임시 변수에 저장됩니다.
// $@TierraCheck_Reason$ : 실패 사유
// $@TierraCheck_Name$   : 실패한 파티원 이름
// $@TierraCheck_Count   : 실제 파티원 수
function	script	F_Tierra_CheckRegisteredParty	{
	.@party_id = getarg(0);
	.@need = getarg(1);
	.@leader_cid = getarg(2);
	.@leader_name$ = getarg(3);
	.@caller_aid = getattachedrid();

	$@TierraCheck_Reason$ = "";
	$@TierraCheck_Name$ = "";
	$@TierraCheck_Count = 0;

	if (.@party_id <= 0) {
		$@TierraCheck_Reason$ = "등록된 파티 정보가 없습니다.";
		return 0;
	}

	if (getpartyleader(.@party_id, 0) != .@leader_name$ || getpartyleader(.@party_id, 2) != .@leader_cid) {
		$@TierraCheck_Reason$ = "등록 이후 파티장이 변경되었습니다.";
		return 0;
	}

	getpartymember .@party_id, 0;
	.@count = $@partymembercount;
	copyarray .@name$[0], $@partymembername$[0], .@count;

	getpartymember .@party_id, 1;
	if ($@partymembercount != .@count) {
		$@TierraCheck_Reason$ = "파티원 정보를 불러오는 중 변경이 감지되었습니다.";
		return 0;
	}
	copyarray .@cid[0], $@partymembercid[0], .@count;

	getpartymember .@party_id, 2;
	if ($@partymembercount != .@count) {
		$@TierraCheck_Reason$ = "파티원 정보를 불러오는 중 변경이 감지되었습니다.";
		return 0;
	}
	copyarray .@aid[0], $@partymemberaid[0], .@count;

	$@TierraCheck_Count = .@count;

	// 자동 입장은 파티 전체를 대상으로 하므로 설정 인원과 정확히 일치해야 합니다.
	if (.@count != .@need) {
		$@TierraCheck_Reason$ = "파티원 수가 티에라 필요 인원과 일치하지 않습니다. 현재 " + .@count + "명 / 필요 " + .@need + "명";
		return 0;
	}

	for (.@i = 0; .@i < .@count; .@i++) {
		if (!isloggedin(.@aid[.@i], .@cid[.@i])) {
			$@TierraCheck_Name$ = .@name$[.@i];
			$@TierraCheck_Reason$ = "로그인하지 않은 파티원이 있습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (!attachrid(.@aid[.@i]) || getcharid(0) != .@cid[.@i]) {
			$@TierraCheck_Name$ = .@name$[.@i];
			$@TierraCheck_Reason$ = "파티원 접속 정보를 확인할 수 없습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (getcharid(1) != .@party_id) {
			$@TierraCheck_Name$ = strcharinfo(0);
			$@TierraCheck_Reason$ = "등록된 파티에서 이탈한 파티원이 있습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (strcharinfo(3) != "bat_room") {
			$@TierraCheck_Name$ = strcharinfo(0);
			$@TierraCheck_Reason$ = "전장 대기맵 bat_room에 없는 파티원이 있습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (HP <= 0) {
			$@TierraCheck_Name$ = strcharinfo(0);

// 준비 완료 파티의 개인 행동을 잠급니다.
// mapflag는 개인 단위 적용이 불가능하므로 setpcblock을 사용합니다.
// 파티장은 관리인을 다시 눌러 취소할 수 있도록 NPC 클릭만 허용합니다.
function	script	F_Tierra_SetPartyReadyLock	{
	.@party_id = getarg(0);
	.@leader_aid = getarg(1);
	.@state = getarg(2);

	if (.@party_id <= 0)
		return;

	getpartymember .@party_id, 2;
	.@count = $@partymembercount;
	copyarray .@aid[0], $@partymemberaid[0], .@count;

	getpartymember .@party_id, 1;
	copyarray .@cid[0], $@partymembercid[0], $@partymembercount;

	for (.@i = 0; .@i < .@count; .@i++) {
		if (!isloggedin(.@aid[.@i], .@cid[.@i]))
			continue;

		// 해제할 때는 두 마스크를 모두 내려 잔여 잠금을 방지합니다.
		if (!.@state) {
			setpcblock PCBLOCK_MOVE|PCBLOCK_USEITEM|PCBLOCK_COMMANDS|PCBLOCK_NPCCLICK, false, .@aid[.@i];
			continue;
		}

		if (.@aid[.@i] == .@leader_aid)
			setpcblock PCBLOCK_MOVE|PCBLOCK_USEITEM|PCBLOCK_COMMANDS, true, .@aid[.@i];
		else
			setpcblock PCBLOCK_MOVE|PCBLOCK_USEITEM|PCBLOCK_COMMANDS|PCBLOCK_NPCCLICK, true, .@aid[.@i];
	}
	return;
}

// 전장 파티장 준비 시스템 - 플라비우스

// 파티 검사 결과는 다음 전역 임시 변수에 저장됩니다.
// $@FlaviusCheck_Reason$ : 실패 사유
// $@FlaviusCheck_Name$   : 실패한 파티원 이름
// $@FlaviusCheck_Count   : 실제 파티원 수
function	script	F_Flavius_CheckRegisteredParty	{
	.@party_id = getarg(0);
	.@need = getarg(1);
	.@leader_cid = getarg(2);
	.@leader_name$ = getarg(3);
	.@caller_aid = getattachedrid();

	$@FlaviusCheck_Reason$ = "";
	$@FlaviusCheck_Name$ = "";
	$@FlaviusCheck_Count = 0;

	if (.@party_id <= 0) {
		$@FlaviusCheck_Reason$ = "등록된 파티 정보가 없습니다.";
		return 0;
	}

	if (getpartyleader(.@party_id, 0) != .@leader_name$ || getpartyleader(.@party_id, 2) != .@leader_cid) {
		$@FlaviusCheck_Reason$ = "등록 이후 파티장이 변경되었습니다.";
		return 0;
	}

	getpartymember .@party_id, 0;
	.@count = $@partymembercount;
	copyarray .@name$[0], $@partymembername$[0], .@count;

	getpartymember .@party_id, 1;
	if ($@partymembercount != .@count) {
		$@FlaviusCheck_Reason$ = "파티원 정보를 불러오는 중 변경이 감지되었습니다.";
		return 0;
	}
	copyarray .@cid[0], $@partymembercid[0], .@count;

	getpartymember .@party_id, 2;
	if ($@partymembercount != .@count) {
		$@FlaviusCheck_Reason$ = "파티원 정보를 불러오는 중 변경이 감지되었습니다.";
		return 0;
	}
	copyarray .@aid[0], $@partymemberaid[0], .@count;

	$@FlaviusCheck_Count = .@count;

	// 자동 입장은 파티 전체를 대상으로 하므로 설정 인원과 정확히 일치해야 합니다.
	if (.@count != .@need) {
		$@FlaviusCheck_Reason$ = "파티원 수가 플라비우스 필요 인원과 일치하지 않습니다. 현재 " + .@count + "명 / 필요 " + .@need + "명";
		return 0;
	}

	for (.@i = 0; .@i < .@count; .@i++) {
		if (!isloggedin(.@aid[.@i], .@cid[.@i])) {
			$@FlaviusCheck_Name$ = .@name$[.@i];
			$@FlaviusCheck_Reason$ = "로그인하지 않은 파티원이 있습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (!attachrid(.@aid[.@i]) || getcharid(0) != .@cid[.@i]) {
			$@FlaviusCheck_Name$ = .@name$[.@i];
			$@FlaviusCheck_Reason$ = "파티원 접속 정보를 확인할 수 없습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (getcharid(1) != .@party_id) {
			$@FlaviusCheck_Name$ = strcharinfo(0);
			$@FlaviusCheck_Reason$ = "등록된 파티에서 이탈한 파티원이 있습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (strcharinfo(3) != "bat_room") {
			$@FlaviusCheck_Name$ = strcharinfo(0);
			$@FlaviusCheck_Reason$ = "전장 대기맵 bat_room에 없는 파티원이 있습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (HP <= 0) {
			$@FlaviusCheck_Name$ = strcharinfo(0);
			$@FlaviusCheck_Reason$ = "사망 상태인 파티원이 있습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (getcharid(4)) {
			$@FlaviusCheck_Name$ = strcharinfo(0);
			$@FlaviusCheck_Reason$ = "이미 다른 전장 팀에 참가 중인 파티원이 있습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}
	}

	if (.@caller_aid)
		attachrid .@caller_aid;
	else
		detachrid;

	return 1;
}

// 파티 검사 결과는 다음 전역 임시 변수에 저장됩니다.
// $@KvMCheck_Reason$ : 실패 사유
// $@KvMCheck_Name$   : 실패한 파티원 이름
// $@KvMCheck_Count   : 실제 파티원 수
function	script	F_KvM_CheckRegisteredParty	{
	.@party_id = getarg(0);
	.@need = getarg(1);
	.@leader_cid = getarg(2);
	.@leader_name$ = getarg(3);
	.@caller_aid = getattachedrid();

	$@KvMCheck_Reason$ = "";
	$@KvMCheck_Name$ = "";
	$@KvMCheck_Count = 0;

	if (.@party_id <= 0) {
		$@KvMCheck_Reason$ = "등록된 파티 정보가 없습니다.";
		return 0;
	}

	if (getpartyleader(.@party_id, 0) != .@leader_name$ || getpartyleader(.@party_id, 2) != .@leader_cid) {
		$@KvMCheck_Reason$ = "등록 이후 파티장이 변경되었습니다.";
		return 0;
	}

	getpartymember .@party_id, 0;
	.@count = $@partymembercount;
	copyarray .@name$[0], $@partymembername$[0], .@count;

	getpartymember .@party_id, 1;
	if ($@partymembercount != .@count) {
		$@KvMCheck_Reason$ = "파티원 정보를 불러오는 중 변경이 감지되었습니다.";
		return 0;
	}
	copyarray .@cid[0], $@partymembercid[0], .@count;

	getpartymember .@party_id, 2;
	if ($@partymembercount != .@count) {
		$@KvMCheck_Reason$ = "파티원 정보를 불러오는 중 변경이 감지되었습니다.";
		return 0;
	}
	copyarray .@aid[0], $@partymemberaid[0], .@count;

	$@KvMCheck_Count = .@count;

	// 자동 입장은 파티 전체를 대상으로 하므로 설정 인원과 정확히 일치해야 합니다.
	if (.@count != .@need) {
		$@KvMCheck_Reason$ = "파티원 수가 KVM 필요 인원과 일치하지 않습니다. 현재 " + .@count + "명 / 필요 " + .@need + "명";
		return 0;
	}

	for (.@i = 0; .@i < .@count; .@i++) {
		if (!isloggedin(.@aid[.@i], .@cid[.@i])) {
			$@KvMCheck_Name$ = .@name$[.@i];
			$@KvMCheck_Reason$ = "로그인하지 않은 파티원이 있습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (!attachrid(.@aid[.@i]) || getcharid(0) != .@cid[.@i]) {
			$@KvMCheck_Name$ = .@name$[.@i];
			$@KvMCheck_Reason$ = "파티원 접속 정보를 확인할 수 없습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (getcharid(1) != .@party_id) {
			$@KvMCheck_Name$ = strcharinfo(0);
			$@KvMCheck_Reason$ = "등록된 파티에서 이탈한 파티원이 있습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (strcharinfo(3) != "bat_room") {
			$@KvMCheck_Name$ = strcharinfo(0);
			$@KvMCheck_Reason$ = "전장 대기맵 bat_room에 없는 파티원이 있습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (HP <= 0) {
			$@KvMCheck_Name$ = strcharinfo(0);
			$@KvMCheck_Reason$ = "사망 상태인 파티원이 있습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}

		if (getcharid(4)) {
			$@KvMCheck_Name$ = strcharinfo(0);
			$@KvMCheck_Reason$ = "이미 다른 전장 팀에 참가 중인 파티원이 있습니다.";
			if (.@caller_aid)
				attachrid .@caller_aid;
			else
				detachrid;
			return 0;
		}
	}

	if (.@caller_aid)
		attachrid .@caller_aid;
	else
		detachrid;

	return 1;
}

// 준비 완료 파티의 개인 행동을 잠급니다.
// mapflag는 개인 단위 적용이 불가능하므로 setpcblock을 사용합니다.
// 파티장은 관리인을 다시 눌러 취소할 수 있도록 NPC 클릭만 허용합니다.
function	script	F_KvM_SetPartyReadyLock	{
	.@party_id = getarg(0);
	.@leader_aid = getarg(1);
	.@state = getarg(2);

	if (.@party_id <= 0)
		return;

	getpartymember .@party_id, 2;
	.@count = $@partymembercount;
	copyarray .@aid[0], $@partymemberaid[0], .@count;

	getpartymember .@party_id, 1;
	copyarray .@cid[0], $@partymembercid[0], $@partymembercount;

	for (.@i = 0; .@i < .@count; .@i++) {
		if (!isloggedin(.@aid[.@i], .@cid[.@i]))
			continue;

		// 해제할 때는 두 마스크를 모두 내려 잔여 잠금을 방지합니다.
		if (!.@state) {
			setpcblock PCBLOCK_MOVE|PCBLOCK_USEITEM|PCBLOCK_COMMANDS|PCBLOCK_NPCCLICK, false, .@aid[.@i];
			continue;
		}

		if (.@aid[.@i] == .@leader_aid)
			setpcblock PCBLOCK_MOVE|PCBLOCK_USEITEM|PCBLOCK_COMMANDS, true, .@aid[.@i];
		else
			setpcblock PCBLOCK_MOVE|PCBLOCK_USEITEM|PCBLOCK_COMMANDS|PCBLOCK_NPCCLICK, true, .@aid[.@i];
	}
	return;
}

function	script	F_KvM_ClearReadyState	{
	callfunc "F_KvM_SetPartyReadyLock", $BG_GUILLAUME_PARTY_ID, $BG_GUILLAUME_LEADER_AID, 0;
	callfunc "F_KvM_SetPartyReadyLock", $BG_CROIX_PARTY_ID, $BG_CROIX_LEADER_AID, 0;

	$@KvMReady_Guillaume = 0;
	$@KvMReady_Croix = 0;
	$@KvMReady_Time = 0;
	$@KvMReady_Starting = 0;
	if ($@BG_READY_TYPE == 3)
		$@BG_READY_TYPE = 0;

	donpcevent "KVM 준비 감시#bat_room::OnStop";
	return;
}

// 전장 관리 메뉴 - 부족 인원 동의 시작

//============================================================
function	script	F_BG_Menu_ShortStart	{
	if ($@BG_READY_TYPE) {
		mes "[전장 관리인]";
		mes "전장 입장 준비 중에는 인원 부족 완화 신청을 변경할 수 없습니다.";
		close;
	}

	if (!callfunc("F_BG_IsRegisteredLeader")) {
		mes "[전장 관리인]";
		mes "인원 부족 완화 신청은 파티장만 가능합니다.";
		close;
	}
	
	if ($@TierraBG1 || $@FlaviusBG1 || $@KvM01BG || $@BG_READY_TYPE)
	{
		mes "[전장 관리인]";
		mes "진행 중엔 등록할 수 없습니다.";
		close;
	}

	if ($BG_User_Count <= 0) {
		mes "[전장 관리인]";
		mes "전장 기본 인원이 설정되어 있지 않습니다.";
		mes "먼저 '전장 인원 조정' 메뉴에서 인원을 설정해주세요.";
		close;
	}

	.@my_team$ = callfunc("F_BG_GetMyRegisteredTeam$");

	if (.@my_team$ == "") {
		mes "[전장 관리인]";
		mes "등록된 전장 파티장만 사용할 수 있습니다.";
		mes "먼저 '전장 파티장 등록' 메뉴에서";
		mes "기욤 또는 크루아 팀 파티장으로 등록해주세요.";
		close;
	}

	.@now = gettimetick(2);

	// 신청 만료 처리: 120초
	if ($BG_SHORT_PENDING && $BG_SHORT_REQ_TIME + 120 < .@now) {
		$BG_SHORT_PENDING = 0;
		$BG_SHORT_REQ_TEAM$ = "";
		$BG_SHORT_TARGET_TEAM$ = "";
		$BG_SHORT_REQ_NAME$ = "";
		$BG_SHORT_REQ_TIME = 0;
	}

	// 이미 상대 팀에서 신청한 상태라면 승낙/거절 처리
	if ($BG_SHORT_PENDING && $BG_SHORT_TARGET_TEAM$ == .@my_team$) {
		mes "[전장 관리인]";
		mes "^0055FF" + $BG_SHORT_REQ_TEAM$ + "^000000 팀에서";
		mes "인원 부족 완화 시작을 신청했습니다.";
		mes "신청자: ^0055FF" + $BG_SHORT_REQ_NAME$ + "^000000";
		next;

		switch(select("승낙한다:거절한다")) {
		case 1:
			if ($BG_SHORT_REQ_TEAM$ == "기욤") {
				$BG_Guillaume_Count = $BG_User_Count - 1;
				$BG_Croix_Count = $BG_User_Count;
			}
			else if ($BG_SHORT_REQ_TEAM$ == "크루아") {
				$BG_Croix_Count = $BG_User_Count - 1;
				$BG_Guillaume_Count = $BG_User_Count;
			}
			else {
				mes "[전장 관리인]";
				mes "신청 팀 정보가 올바르지 않습니다.";
				close;
			}

			announce "[전장 안내] " + $BG_SHORT_TARGET_TEAM$ + " 팀 파티장이 인원 부족 완화 시작을 승인했습니다.", bc_map;
			announce "[전장 안내] " + $BG_SHORT_REQ_TEAM$ + " 팀은 필요 인원이 1명 완화됩니다.", bc_map;

			$BG_SHORT_PENDING = 0;
			$BG_SHORT_REQ_TEAM$ = "";
			$BG_SHORT_TARGET_TEAM$ = "";
			$BG_SHORT_REQ_NAME$ = "";
			$BG_SHORT_REQ_TIME = 0;
			
			donpcevent "티에라 대기실 갱신#bat_a01::OnRefresh";
			donpcevent "플라비우스 대기실 갱신#bat_b01::OnRefresh";
			donpcevent "KVM 대기실 갱신#bat_c01::OnRefresh";

			mes "[전장 관리인]";
			mes "인원 부족 완화 신청을 승인했습니다.";
			mes "현재 필요 인원";
			mes "기욤: ^0055FF" + $BG_Guillaume_Count + "^000000명";
			mes "크루아: ^0055FF" + $BG_Croix_Count + "^000000명";
			close;

		case 2:
			announce "[전장 안내] " + .@my_team$ + " 팀 파티장이 인원 부족 완화 신청을 거절했습니다.", bc_map;

			$BG_SHORT_PENDING = 0;
			$BG_SHORT_REQ_TEAM$ = "";
			$BG_SHORT_TARGET_TEAM$ = "";
			$BG_SHORT_REQ_NAME$ = "";
			$BG_SHORT_REQ_TIME = 0;

			mes "[전장 관리인]";
			mes "인원 부족 완화 신청을 거절했습니다.";
			close;
		}
	}

	// 본인이 이미 신청한 상태
	if ($BG_SHORT_PENDING && $BG_SHORT_REQ_TEAM$ == .@my_team$) {
		mes "[전장 관리인]";
		mes "이미 상대 팀의 승낙을 기다리고 있습니다.";
		mes "신청 팀: ^0055FF" + $BG_SHORT_REQ_TEAM$ + "^000000";
		mes "승낙 대기 팀: ^0055FF" + $BG_SHORT_TARGET_TEAM$ + "^000000";
		close;
	}

	// 다른 신청이 이미 존재
	if ($BG_SHORT_PENDING) {
		mes "[전장 관리인]";
		mes "이미 처리 대기 중인 인원 부족 완화 신청이 있습니다.";
		close;
	}

	.@target_team$ = "";
	if (.@my_team$ == "기욤")
		.@target_team$ = "크루아";
	else if (.@my_team$ == "크루아")
		.@target_team$ = "기욤";

	mes "[전장 관리인]";
	mes "현재 전장 기본 인원은";
	mes "^0055FF" + $BG_User_Count + " vs " + $BG_User_Count + "^000000 입니다.";
	next;

	mes "[전장 관리인]";
	mes "인원 부족 완화 신청을 진행하면,";
	mes "상대 팀 파티장의 승낙을 받아";
	mes "내 팀의 필요 인원을 1명 낮출 수 있습니다.";
	next;

	mes "[전장 관리인]";
	mes "신청 팀: ^0055FF" + .@my_team$ + "^000000";
	mes "승낙 필요 팀: ^0055FF" + .@target_team$ + "^000000";
	mes "진행하시겠습니까?";
	next;

	if (select("신청한다:취소한다") == 2) {
		mes "[전장 관리인]";
		mes "신청을 취소했습니다.";
		close;
	}

	$BG_SHORT_PENDING = 1;
	$BG_SHORT_REQ_TEAM$ = .@my_team$;
	$BG_SHORT_TARGET_TEAM$ = .@target_team$;
	$BG_SHORT_REQ_NAME$ = strcharinfo(0);
	$BG_SHORT_REQ_TIME = .@now;

	announce "[전장 안내] " + .@my_team$ + " 팀에서 인원 부족 완화 시작을 신청했습니다.", bc_map;
	announce "[전장 안내] " + .@target_team$ + " 팀 파티장은 전장 관리인을 통해 승인 또는 거절할 수 있습니다.", bc_map;

	mes "[전장 관리인]";
	mes "인원 부족 완화 신청이 완료되었습니다.";
	mes "^0055FF" + .@target_team$ + "^000000 팀 파티장의 승낙을 기다려주세요.";
	close;
}

//============================================================
// 현재 유저가 파티장인지 확인
//============================================================
function	script	F_BG_IsPartyLeader	{
	.@party_id = getcharid(1);

	if (.@party_id <= 0)
		return 0;

	// getpartyleader(party_id, 0) = 파티장 캐릭터명 기준
	if (getpartyleader(.@party_id, 0) == strcharinfo(0))
		return 1;

	return 0;
}

//============================================================
// 현재 유저의 전장 팀명 확인
// 기본은 파티명으로 판정
// 기존 전장 스크립트에 팀 변수/파티ID가 따로 있으면 여기만 수정하면 됨
//============================================================
function	script	F_BG_GetMyTeamName$	{
	.@party_id = getcharid(1);

	if (.@party_id <= 0)
		return "";

	.@party_name$ = getpartyname(.@party_id);

	// 서버에서 실제 파티명이 한글이면 이대로 사용
	if (.@party_name$ == "기욤")
		return "기욤";

	if (.@party_name$ == "크루아")
		return "크루아";

	// 혹시 영문 파티명을 쓰는 구조면 같이 대응
	if (.@party_name$ == "Guillaume")
		return "기욤";

	if (.@party_name$ == "Croix")
		return "크루아";

	return "";
}

// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@char_id = getcharid(0);
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);
	if ($BG_CROIX_LEADER_CID == .@char_id || $BG_GUILLAUME_LEADER_CID == .@char_id)
		.@partyleaderbonus = 3;
		
	if (.@bg_type$ == "Tierra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win, 2;
		set .@amount_lose,1;
		set .@amount_win_cash, 10;
		set .@amount_lose_cash, 7;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,3;
		set .@amount_lose,2;
		set .@amount_win_cash, 20;
		set .@amount_lose_cash, 15;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		set .@cashamount, .@amount_win_cash;
		if (.@team$ == "Guillaume") {
			mes "[액슬 로즈]";
			mes "복되신 기욤이시여!";
		} else if (.@team$ == "Croix") {
			mes "[스완데리]";
			mes "축복받은 크루아!";
		}
		mes "우리의 영광스러운 승리를 만끽합시다!";
		mes strcharinfo(0)+"그것은 승리를 나타내는 표시입니다.";
	} else {
		set .@amount, .@amount_lose;
		set .@cashamount, .@amount_lose_cash;
		set BATTLEGROUND_POINT, BATTLEGROUND_POINT + 3;
		if (.@team$ == "Guillaume") {
			mes "[액슬 로즈]";
			mes "당신은 졌지만, 이 싸움에 대한 당신의 헌신은 훌륭합니다.";
			mes "이것은 기욤 마롤로가 당신의 헌신적인 노력에 대한 보상으로 주는 것입니다!";
			mes "이번 패배를 교훈 삼아 다음번엔 반드시 이길 수 있을 거야.";
		} else if (.@team$ == "Croix") {
			mes "[스완데리]";
			mes "Oh, "+strcharinfo(0)+" 슬퍼하지 마세요.";
			mes "비록 이기지는 못했지만, 우리는 최선을 다했습니다.";
			mes "이것은 크루아에서 보낸 왕실 선물입니다. 이번 전투를 잊지 마십시오. 다음 전투에서는 반드시 승리할 것입니다.";
		}
	}
	close2;

	.@bonusmount = ($BG_User_Count - 5) * 2;
	getitem .@badge,.@amount + .@partyleaderbonus;

	.@gold_amount = .@cashamount + .@bonusmount;
	.@gold_amount = callfunc("F_NEED_GoldEvent_BGAmount", .@gold_amount);
	getitem 399990, .@gold_amount;

	return;
}
