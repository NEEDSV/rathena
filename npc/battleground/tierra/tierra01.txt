function	script	F_Tierra_NormalizeCount	{
	if ($BG_User_Count <= 0)
		$BG_User_Count = 10;

	if ($BG_Guillaume_Count <= 0)
		$BG_Guillaume_Count = $BG_User_Count;

	if ($BG_Croix_Count <= 0)
		$BG_Croix_Count = $BG_User_Count;

	if ($BG_Guillaume_Count < 1)
		$BG_Guillaume_Count = 1;

	if ($BG_Croix_Count < 1)
		$BG_Croix_Count = 1;

	return;
}

function	script	F_Tierra_GetGuillaumeNeed	{
	callfunc "F_Tierra_NormalizeCount";
	return $BG_Guillaume_Count;
}

function	script	F_Tierra_GetCroixNeed	{
	callfunc "F_Tierra_NormalizeCount";
	return $BG_Croix_Count;
}

function	script	F_Tierra_ResetTeamNeed	{
	callfunc "F_Tierra_NormalizeCount";
	$BG_Guillaume_Count = $BG_User_Count;
	$BG_Croix_Count = $BG_User_Count;
	return;
}

function	script	F_Tierra_AnnounceWaitingRoom	{
	.@msg$ = "[티에라 협곡] 현재 파티장 준비 상태 - 기욤 " + ($@TierraReady_Guillaume ? "완료" : "대기") + ", 크루아 " + ($@TierraReady_Croix ? "완료" : "대기");
	mapannounce "bat_room", .@msg$, bc_map, "0x99CCFF";
	return;
}

// 티에라 자동 입장용 BG 직접 가입 함수
//============================================================
function	script	F_Tierra_JoinPartyToBG	{
	.@party_id = getarg(0);
	.@bg_id = getarg(1);
	.@team = getarg(2);
	.@map$ = getarg(3);
	.@x = getarg(4);
	.@y = getarg(5);

	getpartymember .@party_id, 0;
	.@count = $@partymembercount;
	copyarray .@name$[0], $@partymembername$[0], .@count;
	getpartymember .@party_id, 1;
	copyarray .@cid[0], $@partymembercid[0], .@count;
	getpartymember .@party_id, 2;
	copyarray .@aid[0], $@partymemberaid[0], .@count;

	for (.@i = 0; .@i < .@count; .@i++) {
		// 준비 잠금은 자동 입장 직전에 해제합니다.
		setpcblock PCBLOCK_MOVE|PCBLOCK_USEITEM|PCBLOCK_COMMANDS|PCBLOCK_NPCCLICK, false, .@aid[.@i];

		if (!isloggedin(.@aid[.@i], .@cid[.@i])) {
			$@TierraJoin_FailName$ = .@name$[.@i];
			$@TierraJoin_FailReason$ = "입장 직전에 접속이 종료되었습니다.";
			return 0;
		}

		if (!bg_join(.@bg_id, .@map$, .@x, .@y, .@cid[.@i])) {
			$@TierraJoin_FailName$ = .@name$[.@i];
			$@TierraJoin_FailReason$ = "BG 팀 가입 명령이 실패했습니다.";
			return 0;
		}

		if (!attachrid(.@aid[.@i]) || getcharid(0) != .@cid[.@i]) {
			$@TierraJoin_FailName$ = .@name$[.@i];
			$@TierraJoin_FailReason$ = "가입 후 캐릭터 정보를 확인할 수 없습니다.";
			return 0;
		}

		if (getcharid(4) != .@bg_id) {
			$@TierraJoin_FailName$ = strcharinfo(0);
			$@TierraJoin_FailReason$ = "가입된 BG 팀 ID가 일치하지 않습니다.";
			detachrid;
			return 0;
		}

		Bat_Team = .@team;
		percentheal 100,100;
		detachrid;
	}

	return .@count;
}

function	script	F_Tierra_RollbackPartyBG	{
	.@party_id = getarg(0);
	.@bg_id = getarg(1);

	if (.@party_id <= 0)
		return;

	getpartymember .@party_id, 1;
	.@count = $@partymembercount;
	copyarray .@cid[0], $@partymembercid[0], .@count;
	getpartymember .@party_id, 2;
	copyarray .@aid[0], $@partymemberaid[0], .@count;

	for (.@i = 0; .@i < .@count; .@i++) {
		setpcblock PCBLOCK_MOVE|PCBLOCK_USEITEM|PCBLOCK_COMMANDS|PCBLOCK_NPCCLICK, false, .@aid[.@i];
		if (!isloggedin(.@aid[.@i], .@cid[.@i]))
			continue;
		if (!attachrid(.@aid[.@i]) || getcharid(0) != .@cid[.@i])
			continue;
		if (.@bg_id && getcharid(4) == .@bg_id)
			bg_leave;
		Bat_Team = 0;
		warp "bat_room",154,150;
		detachrid;
	}
	return;
}
