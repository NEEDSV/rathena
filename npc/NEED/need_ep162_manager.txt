//===== Description: =========================================

//= NEED 정책

// NEED: EP16.2 콘텐츠 이용 최소 레벨 (시작 / NEED 경로 공통).
function	script	F_NEED_EP162_MinLv	{
	return 100;
}

// NEED: EP16.2 를 시작할 수 있는지 판정합니다.
// 원본 조건(ep16_royal >= 17) 을 만족하거나, NEED 정책상 Base Lv.100 이상이면 시작 가능합니다.
// EP16.1 스토리 변수(ep16_royal)는 읽기만 하고 절대 변경하지 않습니다.
function	script	F_NEED_EP162_CanStart	{
	if (ep16_royal >= 17)
		return 1;
	if (BaseLevel >= callfunc("F_NEED_EP162_MinLv"))
		return 1;
	return 0;
}

// NEED: 스토리 진행과 무관하게 리벨리온 기지 / 연구소 구역을 이용할 수 있는지 판정합니다.
function	script	F_NEED_EP162_CanUseArea	{
	return (BaseLevel >= callfunc("F_NEED_EP162_MinLv"));
}

// NEED: 원본 조건으로 해당 반복 일일퀘스트가 열려 있는지 (스토리 진행 캐릭터).
//   getarg(0) : 반복 일퀘 대기(쿨타임) 퀘스트 ID
function	script	F_NEED_EP162_DailyStory	{
	switch (getarg(0)) {
	case 7743:	// 잔당 소탕
	case 7745:	// 숨어있는 연구원
		return (terra_gloria_main >= 25);
	case 7762:	// 키메라 진압
		return (terra_gloria_main >= 20);
	case 16010:	// 최상의 컨디션으로 대결 (엘레나)
		return (terra_gloria_main >= 16 && isbegin_quest(14602) == 2);
	case 11466:	// 디엔의 고급 차
	case 11468:	// 리드쉬의 오래된 편지
	case 11470:	// 콘테스의 빛나는 반지
		return (terra_gloria_lady_part >= 13);
	case 16049:	// 추모의 시간
		return (terra_gloria_main >= 17 && isbegin_quest(16048) == 2);
	case 16058:	// 차별화 광고 전략
		return (terra_gloria_main >= 17 && isbegin_quest(16054) == 2);
	case 16034:	// 어르신 다과
	case 16042:	// 채리티 데려오기
	case 14612:	// 총기 재활용
	case 14609:	// 신병의 부탁
		return (terra_gloria_main >= 17);
	case 16037:	// 하루 한 번 재료 수집
		return (terra_gloria_main >= 17 && rachel_quest_16_2 >= 3);
	case 16020:	// 오늘의 아이 돌보기
		return (lebedev_quest_16_2 == 11);
	}
	return 0;
}

// NEED: 스토리를 거치지 않는 "NEED 경로"로 보낼지 판정합니다.
// 스토리가 아직 해당 NPC 의 스토리 단계에 도달하지 않은 Lv.100 이상 캐릭터만 1 을 돌려줍니다.
// 스토리 진행 중(해당 NPC 의 스토리 구간)에는 0 이라 원본 스토리 분기가 우선합니다.
//   getarg(0) : 반복 일퀘 대기(쿨타임) 퀘스트 ID
function	script	F_NEED_EP162_DailyRoute	{
	if (BaseLevel < callfunc("F_NEED_EP162_MinLv"))
		return 0;
	switch (getarg(0)) {
	case 7743:	// 신병(que_swat) 은 메인 22~24 에 스토리 분기가 있음
		return (terra_gloria_main < 22);
	case 7745:	// 테스(que_swat) 는 메인 24 에 스토리 분기가 있음
		return (terra_gloria_main < 24);
	case 7762:
		return (terra_gloria_main < 20);
	case 16010:	// 엘레나는 메인 16 에 첫 결투(스토리)가 있음
		return (terra_gloria_main < 16);
	case 11466:
	case 11468:
	case 11470:	// 부인 파트는 메인 16 부터 시작
		return (terra_gloria_main < 16);
	case 14609:	// 신병(기지) 은 메인 16 에 안내 대사가 있음
		return (terra_gloria_main < 16);
	case 16049:
	case 16058:
	case 16034:
	case 16042:
	case 16037:
	case 14612:
	case 16020:
		return (terra_gloria_main < 17);
	}
	return 0;
}

// NEED: 현재 이용 가능한 반복 일일퀘스트인지 (스토리 조건 또는 NEED 경로).
function	script	F_NEED_EP162_DailyOpen	{
	if (callfunc("F_NEED_EP162_DailyStory", getarg(0)))
		return 1;
	if (callfunc("F_NEED_EP162_DailyRoute", getarg(0)))
		return 1;
	return 0;
}

// NEED: 최초 진행 퀘스트가 진행 중이라 즉시 완료에서 제외해야 하는지.
// 해당 NPC 가 대기 퀘스트를 최초 분기보다 먼저 검사하므로,
// 최초 진행 중에 대기 퀘스트가 생기면 그날 최초 보고가 막힐 수 있습니다.
function	script	F_NEED_EP162_FirstBusy	{
	switch (getarg(0)) {
	case 16042:	// 채리티 최초 16041
		return (isbegin_quest(16041) == 1);
	case 14612:	// 총기 재활용 최초 14610
		return (isbegin_quest(14610) == 1);
	case 14609:	// 신병 최초 14603~14605
		return (isbegin_quest(14603) == 1 || isbegin_quest(14604) == 1 || isbegin_quest(14605) == 1);
	case 16010:	// 투기장 입장 중 (대기실/결투)
		return (isbegin_quest(16011) == 1 && checkquest(16061,HUNTING) != 2 && strcharinfo(3) == "rebel_in" && getmapxy(.@m$, .@x, .@y) == 0 && .@x >= 170 && .@y <= 40);
	}
	return 0;
}

// NEED: 오늘의 EP16.2 반복 일일퀘스트 진행 상황을 계산합니다.
//   반환값             : 오늘 아직 받지 못한 슈발츠 명예의 증표 수 (개방된 일퀘 기준)
//   $@ep162_done       : 오늘 완료한 반복 일퀘 수
//   $@ep162_open       : 개방된 반복 일퀘 수
//   $@ep162_total      : 개방된 반복 일퀘의 하루 증표 합계
//   $@ep162_pend[]     : 즉시 완료 대상 쿨타임 퀘스트 ID
//   $@ep162_pend_amt[] : 각 항목의 증표 수
// 판정 기준은 각 반복 일퀘의 쿨타임 퀘스트 checkquest(id,PLAYTIME) 입니다.
function	script	F_NEED_EP162_DailyStatus	{
	callfunc "F_NEED_EP162_DailyList";
	.@size = getarraysize($@ep162_cd);
	deletearray $@ep162_pend[0], getarraysize($@ep162_pend);
	deletearray $@ep162_pend_amt[0], getarraysize($@ep162_pend_amt);
	$@ep162_done = 0;
	$@ep162_open = 0;
	$@ep162_total = 0;
	.@remain = 0;
	.@n = 0;
	for (.@i = 0; .@i < .@size; ++.@i) {
		.@cd = $@ep162_cd[.@i];
		if (!callfunc("F_NEED_EP162_DailyOpen", .@cd))
			continue;
		$@ep162_open++;
		$@ep162_total += $@ep162_amt[.@i];
		.@state = checkquest(.@cd,PLAYTIME);
		if (.@state == 0 || .@state == 1) {	// 오늘 이미 완료
			$@ep162_done++;
			continue;
		}
		if (callfunc("F_NEED_EP162_FirstBusy", .@cd))
			continue;
		$@ep162_pend[.@n] = .@cd;
		$@ep162_pend_amt[.@n] = $@ep162_amt[.@i];
		.@remain += $@ep162_amt[.@i];
		.@n++;
	}
	return .@remain;
}

function	script	F_NEED_EP162_InstantApply	{
	.@size = getarraysize($@ep162_pend);
	for (.@i = 0; .@i < .@size; ++.@i) {
		.@cd = $@ep162_pend[.@i];
		deletearray .@q[0], getarraysize(.@q);
		deletearray .@it[0], getarraysize(.@it);
		switch (.@cd) {
		case 7743:
			setarray .@q[0], 7742;
			break;
		case 7745:
			setarray .@q[0], 7744, 7760, 7752, 7753, 7754, 7755, 7756, 7757, 7758, 7759;
			break;
		case 7762:
			setarray .@q[0], 7761;
			break;
		case 16010:	// 16060(첫 결투)은 제외
			setarray .@q[0], 16009, 16011, 16061;
			break;
		case 11466:
			setarray .@q[0], 11465;
			setarray .@it[0], 25165;	// High_Class_Tea
			break;
		case 11468:
			setarray .@q[0], 11467;
			setarray .@it[0], 25167;	// Old_Letter
			break;
		case 11470:
			setarray .@q[0], 11469;
			setarray .@it[0], 25166;	// Very_Shining_Ring
			break;
		case 16049:	// 16045~16048(최초 추모)은 제외
			setarray .@q[0], 16050, 16051, 16052, 16053;
			break;
		case 16058:	// 16054(최초 광고)는 제외
			setarray .@q[0], 16055, 16056, 16057, 16059;
			setarray .@it[0], 25163;	// Arms_Shop_AD
			break;
		case 16034:	// 첫 회 표식은 정상 보고와 같이 완료 처리
			if (isbegin_quest(16032) == 1)
				completequest 16032;
			break;
		case 16042:	// 16041(최초)은 제외
			setarray .@q[0], 16043, 16044;
			setarray .@it[0], 25162;	// Anchovy_cookie
			break;
		case 16037:
			setarray .@q[0], 16038, 16039, 16040;
			break;
		case 14612:	// 14610(최초)은 제외
			setarray .@q[0], 14611;
			setarray .@it[0], 25153;	// Broken_Gun_Wreck
			break;
		case 14609:	// 14603~14605(최초)는 제외
			setarray .@q[0], 14606, 14607, 14608;
			break;
		case 16020:	// 16012~16019(쌍둥이 서브)는 제외
			setarray .@q[0], 16021, 16022, 16023, 16024, 16025, 16026, 16027, 16028, 16029, 16030, 16031;
			setarray .@it[0], 25160, 25161;	// Borrow_Book, Handmade_cookie2
			break;
		}
		for (.@j = 0; .@j < getarraysize(.@q); ++.@j) {
			if (isbegin_quest(.@q[.@j]) > 0)
				erasequest .@q[.@j];
		}
		for (.@j = 0; .@j < getarraysize(.@it); ++.@j)
			callfunc "F_NEED_EP162_DelItemAll", .@it[.@j];

		// 만료되어 남아 있는 쿨타임 기록을 정리한 뒤 오늘 쿨타임을 새로 설정합니다.
		if (isbegin_quest(.@cd) > 0)
			erasequest .@cd;
		setquest .@cd;
	}
	return;
}

//= NEED: EP16.2 메모리얼 던전 공용 (최초 클리어 기록 / 간소화 / 파티원 진행)
//=   클리어 기록은 기존 NEED 메모리얼 던전과 같이 계정 변수입니다.
//=     #NEED_MD_HHWB_CLEAR   : 하트헌터 기지 외곽 (스토리 또는 반복) 클리어
//=     #NEED_MD_WERNER_CLEAR : 베르너 연구소 중추실 (스토리 또는 반복) 클리어
//=   getarg(0) 종류 : 1 = 하트헌터 기지 외곽, 2 = 베르너 연구소 중추실
//============================================================
function	script	F_NEED_EP162_MDClear	{
	if (getarg(0) == 1)
		return #NEED_MD_HHWB_CLEAR;
	return #NEED_MD_WERNER_CLEAR;
}

// NEED: 파티 내 온라인 구성원 모두가 최초 클리어를 완료했는지 (마지막 방 F_LastRoomCanEnableSimple 과 같은 방식).
function	script	F_NEED_EP162_PartyAllClear	{
	.@type = getarg(0);
	.@caller_aid = getcharid(3);
	.@party_id = getcharid(1);
	if (.@party_id < 1)
		return 0;
	getpartymember .@party_id,1;
	.@member_count = $@partymembercount;
	copyarray .@member_cid[0],$@partymembercid[0],.@member_count;
	getpartymember .@party_id,2;
	copyarray .@member_aid[0],$@partymemberaid[0],.@member_count;
	.@allowed = 1;
	for (.@i = 0; .@i < .@member_count; .@i++) {
		if (!isloggedin(.@member_aid[.@i], .@member_cid[.@i]))
			continue;
		if (attachrid(.@member_aid[.@i]) == 0)
			continue;
		if (callfunc("F_NEED_EP162_MDClear", .@type) == 0) {
			.@allowed = 0;
			break;
		}
	}
	attachrid(.@caller_aid);
	return .@allowed;
}

// NEED: 파티장이 일반 / 간소화 진행을 고릅니다. 반환값 1 = 간소화.
// 파티장 본인과 온라인 파티원 모두에게 클리어 기록이 있을 때만 간소화를 고를 수 있습니다.
function	script	F_NEED_EP162_MDChooseMode	{
	.@type = getarg(0);
	if (callfunc("F_NEED_EP162_MDClear", .@type) == 0)
		return 0;
	mes needtr("[NEED 메모리얼 던전]", "[NEED Memorial Dungeon]");
	mes needtr("이미 이 메모리얼 던전을 클리어한 기록이 있습니다.", "You have already cleared this memorial dungeon once.");
	mes needtr("^4d4dff간소화 진행^000000은 대화/대기/퍼즐을 줄이고 전투와 보스는 그대로 진행합니다.", "The ^4d4dffsimplified run^000000 cuts dialogue, waiting and puzzles but keeps all fights and the boss.");
	next;
	if (select(needtr("일반 진행", "Normal run"), needtr("간소화 진행", "Simplified run")) == 1)
		return 0;
	if (callfunc("F_NEED_EP162_PartyAllClear", .@type) == 0) {
		mes needtr("[NEED 메모리얼 던전]", "[NEED Memorial Dungeon]");
		mes needtr("온라인 파티원 중 최초 클리어를 하지 않은 계정이 있어 일반 진행으로 시작합니다.", "Some online party members have not cleared it yet, so this run starts as a normal run.");
		next;
		return 0;
	}
	return 1;

// NEED: 인스턴스 소유 파티의 온라인 구성원 중 해당 맵에 있는 계정에 클리어 기록을 남깁니다.
// 몬스터 처치 이벤트처럼 플레이어가 붙어 있지 않은 상황에서도 호출할 수 있습니다.
//   getarg(0) 종류, getarg(1) 인스턴스 맵 이름
function	script	F_NEED_EP162_MDSetClear	{
	.@type = getarg(0);
	.@map$ = getarg(1);
	.@party_id = instance_live_info(ILI_OWNER);
	if (instance_live_info(ILI_MODE) != IM_PARTY || .@party_id < 1)
		return;
	.@orig = playerattached();
	getpartymember .@party_id,1;
	.@member_count = $@partymembercount;
	copyarray .@member_cid[0],$@partymembercid[0],.@member_count;
	getpartymember .@party_id,2;
	copyarray .@member_aid[0],$@partymemberaid[0],.@member_count;
	for (.@i = 0; .@i < .@member_count; .@i++) {
		if (!isloggedin(.@member_aid[.@i], .@member_cid[.@i]))
			continue;
		if (attachrid(.@member_aid[.@i]) == 0)
			continue;
		if (strcharinfo(3) != .@map$)
			continue;
		if (.@type == 1)
			#NEED_MD_HHWB_CLEAR = 1;
		else
			#NEED_MD_WERNER_CLEAR = 1;
	}
	if (.@orig)
		attachrid(.@orig);
	else
		detachrid;
	return;
}
