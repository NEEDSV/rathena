//===== rAthena Script ========================================
//= NEED 창고 보급 상점
//===== 설명 ==================================================
//= 대량으로 소비하는 보급품을 인벤토리를 거치지 않고
//= 개인 창고로 바로 지급하는 전용 상점.
//=
//= - 판매 품목 / 가격 / 1회 최대 구매 수량은 모두 서버측
//=   src/map/need_storage_shop.cpp 의 설정 테이블에서 관리한다.
//= - 이 상점은 상인 디스카운트(MC_DISCOUNT 등)가 적용되지 않고
//=   항상 item_db 의 기본 구매가(또는 Override 가격)로 판매한다.
//= - 지급 창고는 창고 1 / 2 / 3 중에서 고를 수 있다. 창고 2/3 은
//=   계정이 해금한 경우에만 메뉴에 표시되며, 개방 여부 판정과
//=   목적 창고 검증은 전부 서버에서 다시 수행한다.
//= - 실제 구매/검증/지급/로그는 storageshopbuy() 안에서 처리된다.
//=============================================================

lasagna,161,141,6	script	창고 보급 상점	558,{
	.@m_total = storageshopcategories();

	if (.@m_total < 1 || .@m_total > getarraysize(.cat_name$)) {
		mes needtr("[창고 보급 상점]","[Storage Supply Shop]");
		mes needtr("상점 설정에 문제가 있어 지금은 이용할 수 없습니다.","There is a problem with the shop setup, so it cannot be used right now.");
		mes needtr("운영자에게 문의해주세요.","Please contact an operator.");
		close;
	}

	mes needtr("[창고 보급 상점]","[Storage Supply Shop]");
	mes needtr("이곳에서 구매한 물품은 인벤토리를 거치지 않고","Anything you buy here skips your inventory and goes");
	mes needtr("^0000FF선택한 개인 창고^000000로 바로 들어갑니다.","straight into ^0000FFthe personal Storage you chose^000000.");
	next;

	mes needtr("[창고 보급 상점]","[Storage Supply Shop]");
	mes needtr("대신 창고 직행 편의를 제공하는 대가로","In exchange for delivering straight to Storage, however,");
	mes needtr("^FF0000상인 할인(디스카운트)이 적용되지 않습니다.^000000","^FF0000the Merchant discount does not apply.^000000");
	mes needtr("누구나 동일하게 할인 전 판매가로 구매합니다.","Everyone pays the same undiscounted price.");
	next;

	while (1) {
		// 저장된 지급 창고를 현재 계정 권한으로 다시 검증한다.
		// 이용할 수 없는 창고였다면 서버가 창고 1로 되돌리고 알려준다.
		.@m_target = storageshoptarget();

		if (@storageshop_target_reset) {
			@storageshop_target_reset = 0;

			mes needtr("[창고 보급 상점]","[Storage Supply Shop]");
			mes needtr("기존에 선택한 지급 창고를 현재 이용할 수 없어","The delivery Storage you chose is unavailable at the moment, so");
			mes needtr("지급 위치가 ^0000FF","The delivery location has been changed to ^0000FF") + needtr(.stor_name$[0], .stor_name_en$[0]) + needtr("^000000 (으)로 변경되었습니다.","^000000.");
			next;
		}

		mes needtr("[창고 보급 상점]","[Storage Supply Shop]");
		mes needtr("현재 지급 창고 : ^0000FF","Current delivery Storage : ^0000FF") + needtr(.stor_name$[.@m_target], .stor_name_en$[.@m_target]) + "^000000";
		next;

		switch (select(needtr("물품 구매","Buy supplies"), needtr("지급 창고 변경","Change delivery Storage"), needtr("종료","Exit"))) {
		case 1:
			callsub S_Categories;
			break;

		case 2:
			callsub S_ChangeTarget;
			break;

		default:
			mes needtr("[창고 보급 상점]","[Storage Supply Shop]");
			mes needtr("또 이용해주세요.","Please come again.");
			close;
		}
	}

	end;

// =========================
// 지급 창고 변경
// 개방된 창고만 메뉴에 표시한다. (판정은 서버의 storageshopunlocked)
// =========================
S_ChangeTarget:
	.@t_menu$ = "";
	.@t_menu_en$ = "";
	.@t_count = 0;

	for (.@t_i = 0; .@t_i <= .StorageMax; .@t_i++) {
		if (!storageshopunlocked(.@t_i))
			continue;

		.@t_ids[.@t_count] = .@t_i;
		.@t_menu$ += .stor_name$[.@t_i] + ":";
		.@t_menu_en$ += .stor_name_en$[.@t_i] + ":";
		.@t_count++;
	}

	if (.@t_count < 1) {
		mes needtr("[지급 창고 변경]","[Change Delivery Storage]");
		mes needtr("이용할 수 있는 창고가 없습니다.","There is no Storage you can use.");
		next;
		return;
	}

	.@t_menu$ += "이전으로";
	.@t_menu_en$ += "Back";

	mes needtr("[지급 창고 변경]","[Change Delivery Storage]");
	mes needtr("구매한 물품을 받을 창고를 선택해주세요.","Choose the Storage that will receive what you buy.");
	mes needtr("창고 2 / 창고 3 은 해금한 계정에만 표시됩니다.","Storage 2 and Storage 3 only appear on accounts that have unlocked them.");
	next;

	.@t_sel = select(needtr(.@t_menu$, .@t_menu_en$));

	if (.@t_sel < 1 || .@t_sel > .@t_count)
		return;

	.@t_pick = .@t_ids[.@t_sel - 1];

	// 서버에서 다시 개방 여부를 확인하고 저장한다.
	if (!storageshopsettarget(.@t_pick)) {
		mes needtr("[지급 창고 변경]","[Change Delivery Storage]");
		mes needtr("해당 창고는 아직 이용할 수 없습니다.","That Storage is not available yet.");
		mes needtr("창고를 개방한 후 다시 이용해주세요.","Please open the Storage first and come back.");
		next;
		return;
	}

	mes needtr("[지급 창고 변경]","[Change Delivery Storage]");
	mes needtr("지급 창고가 ^0000FF","Your delivery Storage has been changed to ^0000FF") + needtr(.stor_name$[.@t_pick], .stor_name_en$[.@t_pick]) + needtr("^000000 (으)로 변경되었습니다.","^000000.");
	next;
	return;

// =========================
// 목적 창고 준비 (창고 2/3 은 비동기 로드가 필요)
// 결과는 .@p_ret 에 담긴다. 0 준비완료 / 그 외 실패
// =========================
S_Prepare:
	.@p_target = getarg(0);
	.@p_ret = storageshopprepare(.@p_target);
	.@p_wait = 0;

	while (.@p_ret == 1 && .@p_wait < 25) {
		sleep2 200;

		.@p_ret = storageshopprepare(.@p_target);
		.@p_wait++;
	}

	return;

// 준비 실패 사유 안내
S_PrepareError:
	mes needtr("[창고 보급 상점]","[Storage Supply Shop]");

	if (.@p_ret == -1) {
		mes needtr("해당 창고는 아직 이용할 수 없습니다.","That Storage is not available yet.");
		mes needtr("창고를 개방한 후 다시 이용해주세요.","Please open the Storage first and come back.");
	} else if (.@p_ret == -2) {
		mes needtr("지금은 창고를 사용할 수 없습니다.","Storage cannot be used right now.");
		mes needtr("창고 창을 닫고 거래나 노점을 종료한 뒤","Close the Storage window, end any trade or vending, and");
		mes needtr("다시 시도해주세요.","try again.");
	} else {
		mes needtr(.stor_name$[.@p_target], .stor_name_en$[.@p_target]) + needtr(" 정보를 불러오지 못했습니다."," information could not be loaded.");
		mes needtr("잠시 후 다시 시도해주세요.","Please try again in a moment.");
	}

	next;
	return;

// =========================
// 카테고리 선택
// =========================
S_Categories:
	// 품목을 고르는 동안 목적 창고를 미리 로드해 둔다.
	.@g_target = storageshoptarget();

	callsub S_Prepare, .@g_target;

	if (.@p_ret != 0) {
		callsub S_PrepareError;
		return;
	}

	.@g_total = storageshopcategories();

	while (1) {
		.@g_menu$ = "";
		.@g_menu_en$ = "";

		for (.@g_i = 0; .@g_i < .@g_total; .@g_i++)
			.@g_menu$ += .cat_name$[.@g_i] + ":";
		for (.@g_i = 0; .@g_i < .@g_total; .@g_i++) .@g_menu_en$ += .cat_name_en$[.@g_i] + ":";

		.@g_menu$ += "이전으로";
		.@g_menu_en$ += "Back";

		mes needtr("[창고 보급 상점]","[Storage Supply Shop]");
		mes needtr("지급 창고 : ^0000FF","Delivery Storage : ^0000FF") + needtr(.stor_name$[.@g_target], .stor_name_en$[.@g_target]) + "^000000";
		mes needtr("어떤 품목을 보시겠습니까?","Which items would you like to see?");
		next;

		.@g_sel = select(needtr(.@g_menu$, .@g_menu_en$));

		if (.@g_sel < 1 || .@g_sel > .@g_total)
			return;

		.@g_cat = .@g_sel - 1;

		callsub S_Category, .@g_cat;
	}

	return;

// =========================
// 카테고리 품목 목록 (페이지 단위)
// =========================
S_Category:
	.@c_cat = getarg(0);
	.@c_count = storageshopitemcount(.@c_cat);

	if (.@c_count < 1) {
		mes "[" + needtr(.cat_name$[.@c_cat], .cat_name_en$[.@c_cat]) + "]";
		mes needtr("현재 판매 중인 품목이 없습니다.","There are no items on sale right now.");
		next;
		return;
	}

	.@c_page = 0;
	.@c_last = (.@c_count - 1) / .PageSize;

	while (1) {
		.@c_start = .@c_page * .PageSize;
		.@c_end = .@c_start + .PageSize;

		if (.@c_end > .@c_count)
			.@c_end = .@c_count;

		.@c_shown = 0;
		.@c_menu$ = "";

		for (.@c_i = .@c_start; .@c_i < .@c_end; .@c_i++) {
			.@c_id = storageshopitem(.@c_cat, .@c_i);

			if (.@c_id <= 0)
				continue;

			.@c_name$ = getitemname(.@c_id);

			.@c_ids[.@c_shown] = .@c_id;
			.@c_menu$ += .@c_name$ + " (" + callfunc("F_InsertComma", storageshopprice(.@c_id)) + " Zeny):";
			.@c_shown++;
		}

		if (.@c_shown < 1) {
			mes "[" + needtr(.cat_name$[.@c_cat], .cat_name_en$[.@c_cat]) + "]";
			mes needtr("현재 판매 중인 품목이 없습니다.","There are no items on sale right now.");
			next;
			return;
		}

		.@c_menu$ += needtr("이전 페이지:다음 페이지:이전 메뉴로", "Prev Page:Next Page:Back to Menu");

		mes "[" + needtr(.cat_name$[.@c_cat], .cat_name_en$[.@c_cat]) + "]";
		mes needtr("구매할 품목을 선택해주세요.","Choose the item you want to buy.");
		mes needtr("페이지 : ","Page : ") + (.@c_page + 1) + " / " + (.@c_last + 1);
		next;

		.@c_sel = select(.@c_menu$);

		if (.@c_sel >= 1 && .@c_sel <= .@c_shown) {
			.@c_pick = .@c_ids[.@c_sel - 1];

			callsub S_Buy, .@c_pick;
			continue;
		}

		if (.@c_sel == .@c_shown + 1) {
			if (.@c_page > 0)
				.@c_page--;

			continue;
		}

		if (.@c_sel == .@c_shown + 2) {
			if (.@c_page < .@c_last)
				.@c_page++;

			continue;
		}

		return;
	}

	return;

// =========================
// 수량 선택 및 구매
// =========================
S_Buy:
	.@b_id = getarg(0);
	.@b_price = storageshopprice(.@b_id);
	.@b_max = storageshopmaxqty(.@b_id);

	if (.@b_id <= 0 || .@b_price <= 0 || .@b_max <= 0) {
		mes needtr("[창고 보급 상점]","[Storage Supply Shop]");
		mes needtr("지금은 판매하지 않는 품목입니다.","That item is not on sale right now.");
		next;
		return;
	}

	while (1) {
		.@b_target = storageshoptarget();
		.@b_afford = Zeny / .@b_price;

		if (.@b_afford > .@b_max)
			.@b_afford = .@b_max;

		mes needtr("[창고 보급 상점]","[Storage Supply Shop]");
		mes needtr("품목 : ^0000FF","Item : ^0000FF") + getitemname(.@b_id) + "^000000";
		mes needtr("지급 위치 : ^0000FF","Delivery location : ^0000FF") + needtr(.stor_name$[.@b_target], .stor_name_en$[.@b_target]) + "^000000";
		mes needtr("개당 가격 : ","Price each : ") + callfunc("F_InsertComma", .@b_price) + " Zeny";
		mes needtr("1회 최대 구매 : ","Max per purchase : ") + callfunc("F_InsertComma", .@b_max) + needtr(" 개",".");
		mes needtr("보유 제니 : ","Your Zeny : ") + callfunc("F_InsertComma", Zeny) + " Zeny";
		next;

		.@b_sel = select(
			needtr("100 개","100"),
			needtr("1,000 개","1,000"),
			needtr("10,000 개","10,000"),
			needtr("최대 구매 가능 수량 (","Max you can afford (") + callfunc("F_InsertComma", .@b_afford) + needtr(" 개)",")"),
			needtr("직접 입력","Enter manually"),
			needtr("이전으로","Back")
		);

		.@b_amount = 0;

		switch (.@b_sel) {
		case 1:
			.@b_amount = 100;
			break;

		case 2:
			.@b_amount = 1000;
			break;

		case 3:
			.@b_amount = 10000;
			break;

		case 4:
			.@b_amount = .@b_afford;
			break;

		case 5:
			mes needtr("[창고 보급 상점]","[Storage Supply Shop]");
			mes needtr("구매할 수량을 입력해주세요.","Enter the quantity you want to buy.");
			mes needtr("입력 가능 범위 : 1 ~ ","Valid range : 1 ~ ") + callfunc("F_InsertComma", .@b_max);
			next;

			if (input(.@b_amount, 1, .@b_max) != 0) {
				mes needtr("[창고 보급 상점]","[Storage Supply Shop]");
				mes needtr("1 이상 ","You can only enter a quantity from 1 to ") + callfunc("F_InsertComma", .@b_max) + needtr(" 이하의 수량만 입력할 수 있습니다.",".");
				next;
				continue;
			}

			break;

		default:
			return;
		}

		// 입력값은 서버(storageshopbuy)에서 다시 검증하지만,
		// 안내를 위해 스크립트에서도 범위를 확인한다.
		if (.@b_amount < 1 || .@b_amount > .@b_max) {
			mes needtr("[창고 보급 상점]","[Storage Supply Shop]");

			if (.@b_afford < 1 && .@b_sel == 4)
				mes needtr("제니가 부족하여 구매할 수 없습니다.","You do not have enough Zeny to buy that.");
			else
				mes needtr("구매 수량이 올바르지 않습니다.","That purchase quantity is not valid.");

			next;
			continue;
		}

		.@b_total = .@b_amount * .@b_price;

		mes needtr("[창고 보급 상점]","[Storage Supply Shop]");
		mes "^0000FF" + getitemname(.@b_id) + "^000000";
		mes " ";
		mes needtr("구매 수량 : ","Quantity : ") + callfunc("F_InsertComma", .@b_amount) + needtr(" 개",".");
		mes needtr("개당 가격 : ","Price each : ") + callfunc("F_InsertComma", .@b_price) + " Zeny";
		mes needtr("총 가격 : ^FF0000","Total price : ^FF0000") + callfunc("F_InsertComma", .@b_total) + "^000000 Zeny";
		mes needtr("지급 위치 : ^0000FF","Delivery location : ^0000FF") + needtr(.stor_name$[.@b_target], .stor_name_en$[.@b_target]) + "^000000";
		mes " ";
		mes needtr("구매하시겠습니까?", "Would you like to purchase");
		next;

		if (select(needtr("구매","Buy"), needtr("취소","Cancel")) == 2)
			continue;

		// 결제 직전에 목적 창고가 아직 준비 상태인지 다시 확인한다.
		callsub S_Prepare, .@b_target;

		if (.@p_ret != 0) {
			callsub S_PrepareError;
			continue;
		}

		.@b_result = storageshopbuy(.@b_id, .@b_amount, .@b_target);

		mes needtr("[창고 보급 상점]","[Storage Supply Shop]");

		switch (.@b_result) {
		case 1:
			// 실제로 창고에 들어간 수량은 서버가 @storageshop_amount 로 알려준다.
			mes getitemname(.@b_id) + " " + callfunc("F_InsertComma", @storageshop_amount) + needtr("개를"," stored in");
			mes "^0000FF" + needtr(.stor_name$[.@b_target], .stor_name_en$[.@b_target]) + needtr("^000000에 보관했습니다.","^000000.");
			mes callfunc("F_InsertComma", @storageshop_total) + needtr(" Zeny를 사용했습니다."," Zeny spent.");
			break;

		case -1:
			mes needtr("지금은 판매하지 않는 품목입니다.","That item is not on sale right now.");
			break;

		case -2:
			mes needtr("구매 수량이 올바르지 않습니다.","That purchase quantity is not valid.");
			break;

		case -3:
			mes needtr("제니가 부족하여 구매할 수 없습니다.","You do not have enough Zeny to buy that.");
			break;

		case -4:
			mes needtr(.stor_name$[.@b_target], .stor_name_en$[.@b_target]) + needtr("의 공간이 부족하여 구매할 수 없습니다."," does not have enough space, so you cannot buy that.");
			mes needtr("필요한 공간을 확보한 후 다시 시도해주세요.","Please make the space you need and try again.");
			break;

		case -5:
			mes needtr("지금은 창고를 사용할 수 없습니다.","Storage cannot be used right now.");
			mes needtr("창고 창을 닫고 거래나 노점을 종료한 뒤","Close the Storage window, end any trade or vending, and");
			mes needtr("다시 시도해주세요.","try again.");
			break;

		case -6:
			mes needtr("창고에 보관할 수 없는 아이템입니다.","That item cannot be kept in Storage.");
			break;

		case -7:
			mes needtr("총 구매 금액이 너무 커서 구매할 수 없습니다.","The total price is too high to buy.");
			mes needtr("수량을 줄여서 다시 시도해주세요.","Please lower the quantity and try again.");
			break;

		case -8:
			mes needtr("해당 창고는 아직 이용할 수 없습니다.","That Storage is not available yet.");
			mes needtr("창고를 개방한 후 다시 이용해주세요.","Please open the Storage first and come back.");
			break;

		case -9:
			mes needtr(.stor_name$[.@b_target], .stor_name_en$[.@b_target]) + needtr(" 정보를 불러오지 못했습니다."," information could not be loaded.");
			mes needtr("잠시 후 다시 시도해주세요.","Please try again in a moment.");
			break;

		default:
			mes needtr("구매에 실패했습니다.","The purchase failed.");
			break;
		}

		next;
	}

	return;

OnInit:
	// 한 페이지에 표시할 품목 수
	.PageSize = 8;

	// 지급 가능한 최대 창고 index (need_storage_shop.hpp 의
