//============================================================
// 전장 맵 공용 판정
//   아이템 스크립트는 status_calc_pc 안에서 매우 자주 실행되므로
//   callfunc 오버헤드를 피하기 위해 item_db 쪽에서는 같은 판정을
//   인라인(getmapflag ... mf_battleground)으로 사용한다.
//   NPC / 스크립트에서 전장 여부가 필요할 때는 이 함수를 쓴다.
//============================================================
function	script	F_NEED_IsBattlegroundMap	{
	.@map$ = getarg(0, strcharinfo(3));
	if (.@map$ == "")
		return 0;
	return (getmapflag(.@map$, mf_battleground) > 0);
}

//============================================================
// 초보자 전장 장비 이용 자격
//   신규지원 대상 계정 + 성장지원 졸업 조건(Base 175 / Job 60).
//   기존 175 보상 수령 여부(NEED_BEGINNER_REWARD_175)와 무관하다.
//============================================================
function	script	F_NEED_NBG_CanUse	{
	if (!callfunc("F_NEED_IsBeginnerAccount"))
		return 0;
	if (BaseLevel < 175 || JobLevel < 60)
		return 0;
	return 1;
}

//============================================================
// 안전 교환 공용 함수
//   getarg(0)      지급할 아이템 ID
//   getarg(1)      지급 수량
//   getarg(2)      해제할 장착 부위 1 (없으면 -1)
//   getarg(3)      해제할 장착 부위 2 (없으면 -1)
//   getarg(4...)   같은 카테고리에서 회수할 아이템 ID 목록
//
//   반환 0 = 성공 / 1 = 소지 무게·공간 부족
//
//   처리 순서: 공간 확인 -> 장착 해제 -> 회수 -> 지급
//   공간 확인을 먼저 하므로 "회수만 되고 지급이 실패" 하는 경우가 없다.
//============================================================
function	script	F_NEED_NBG_Swap	{
	.@give = getarg(0);
	.@amount = getarg(1);
	.@slot1 = getarg(2);
	.@slot2 = getarg(3);
	.@argc = getargcount();

	// 1) 지급 공간 확인 (회수 전에 먼저 확인한다)
	if (checkweight(.@give, .@amount) == 0)
		return 1;

	// 2) 회수 대상이 장착 중이면 먼저 해제한다
	for (.@s = 2; .@s <= 3; ++.@s) {
		.@slot = getarg(.@s);
		if (.@slot < 0)
			continue;
		if (!getequipisequiped(.@slot))
			continue;
		.@worn = getequipid(.@slot);
		for (.@i = 4; .@i < .@argc; ++.@i) {
			if (.@worn == getarg(.@i)) {
				unequip .@slot;
				break;
			}
		}
	}

	// 3) 같은 카테고리 기존 장비 회수
	for (.@i = 4; .@i < .@argc; ++.@i) {
		.@id = getarg(.@i);
		.@have = countitem(.@id);
		if (.@have > 0)
			delitem .@id, .@have;
	}

	// 4) 신규 지급
	getitembound .@give, .@amount, Bound_Char;
	return 0;
}

//============================================================
// 보유 현황 문자열
//============================================================
function	script	F_NEED_NBG_Owned	{
	.@argc = getargcount();
	for (.@i = 0; .@i < .@argc; ++.@i) {
		if (countitem(getarg(.@i)) > 0)
			return getarg(.@i);
	}
	return 0;
}
