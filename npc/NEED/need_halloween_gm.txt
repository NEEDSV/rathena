//============================================================
// NEED 2026 할로윈 이벤트 - 운영 스위치 관리
//============================================================
//= GM(60+) 전용 @halloweengm. 추석 @chuseokgm 과 같은 구성.
//=
//= script 전역 (영구 변수, 서버 재시작 후에도 유지)
//=   $NEED_HALLOWEEN_EVENT_ON        할로윈 이벤트 전체 (NPC 표시 포함)
//=   $NEED_HALLOWEEN_HUNT_ON         사냥 월드 드롭 (battle flag 로 반영)
//=   $NEED_HALLOWEEN_BG_ON           전장 코인 지급
//=   $NEED_HALLOWEEN_JACK_ENTRY_ON   잭 잡기 입장
//=   $NEED_HALLOWEEN_JACK_REWARD_ON  잭 잡기 보상
//=   $NEED_HALLOWEEN_SHOP_ON         할로윈 교환 NPC
//=   $NEED_HALLOWEEN_LEGACY_NPC_ON   Event Hub 가 켜져 있어도 payo2 할로윈 NPC 표시 (운영/복구용)
//=
//= Event Hub (need_event_hub.txt) - 공통 Event 버튼
//=   [Event Hub 버튼 관리] 에서 할로윈 모듈(F_NEED_HW26_Hub, 202610)로 연결하고 버튼을 켠다.
//=   전체 ON 은 Hub 를 켜지 않는다: Event 버튼 exe 가 배포되기 전에 켜면 payo2 NPC 만 숨어 버린다.
//=   전체 OFF / 종료 처리 때 Hub 가 할로윈에 연결돼 있으면 버튼을 끈다.
//=
//= battle conf (conf/battle/need_world_drop.conf)
//=   need_halloween_hunt_enable = 실제 드롭 판정. 이 스크립트가 로드될 때(OnInit)와
//=   스위치를 바꿀 때마다 (전체 && 사냥) 값으로 다시 맞춘다. 그래서 conf 값(no)과
//=   상관없이 재시작 후에도 GM 이 정한 상태가 유지된다.
//=   단, @reloadbattleconf 를 쓰면 conf 값으로 돌아가므로 이 화면에서 [다시 맞추기] 를 누른다.
//============================================================

-	script	need_halloween_gm	-1,{
	if (getgmlevel() < 60) {
		mes "[NEED 할로윈 이벤트 관리]";
		mes "GM(60+) 전용 기능입니다.";
		close;
	}
	callsub S_Main;
	end;

OnCommand:
	if (getgmlevel() < 60) end;
	callsub S_Main;
	end;

OnInit:
	bindatcmd "halloweengm", "need_halloween_gm::OnCommand", 60, 60;
	callfunc "F_NEED_Halloween_Sync";
	end;

S_Main:
	while (1) {
		.@sel = select(
			"[전체 상태 보기]",
			"[할로윈 이벤트 전체 ON/OFF]",
			"[사냥 월드 드롭 관리]",
			"[전장 코인 관리]",
			"[잭 잡기 관리]",
			"[교환 NPC 관리]",
			"[Event Hub 버튼 관리]",
			"[이벤트 종료 처리]",
			"[닫기]" );
		if (.@sel == 1) callsub S_Board;
		else if (.@sel == 2) callsub S_Master;
		else if (.@sel == 3) callsub S_Hunt;
		else if (.@sel == 4) callsub S_BG;
		else if (.@sel == 5) callsub S_Jack;
		else if (.@sel == 6) callsub S_Shop;
		else if (.@sel == 7) callsub S_Hub;
		else if (.@sel == 8) callsub S_End;
		else break;
	}
	return;

S_Board:
	.@flag = getbattleflag("need_halloween_hunt_enable");
	.@want = ($NEED_HALLOWEEN_EVENT_ON && $NEED_HALLOWEEN_HUNT_ON) ? 1 : 0;
	.@day$ = gettimestr("%Y-%m-%d", 11, gettimetick(2) - 14400);
	.@rows = query_sql("SELECT COUNT(*) FROM `need_chuseok_honey_claim` WHERE `event_id`=202610 AND `logical_date`='" + .@day$ + "' AND `status`=1", .@jack_today);
	if (.@rows < 1) .@jack_today = -1;

	.@wc = 0;
	if (.@flag != .@want) { .@warn$[.@wc] = "사냥 battle flag(" + .@flag + ")가 스위치(" + .@want + ")와 다릅니다. [다시 맞추기] 필요"; .@wc++; }
	if ($NEED_HALLOWEEN_EVENT_ON) {
		if ($NEED_CHUSEOK_EVENT_ON) { .@warn$[.@wc] = "추석 이벤트가 아직 ON 입니다. 같은 칸의 추석 NPC 를 숨기지 않았습니다."; .@wc++; }
		if (!$NEED_HALLOWEEN_HUNT_ON) { .@warn$[.@wc] = "이벤트 진행 중이지만 사냥 월드 드롭이 OFF 입니다."; .@wc++; }
		if (!$NEED_HALLOWEEN_BG_ON) { .@warn$[.@wc] = "이벤트 진행 중이지만 전장 코인이 OFF 입니다."; .@wc++; }
		if (!$NEED_HALLOWEEN_JACK_ENTRY_ON) { .@warn$[.@wc] = "이벤트 진행 중이지만 잭 잡기 입장이 OFF 입니다."; .@wc++; }
		if (!$NEED_HALLOWEEN_JACK_REWARD_ON) { .@warn$[.@wc] = "이벤트 진행 중이지만 잭 잡기 보상이 OFF 입니다."; .@wc++; }
		if (!$NEED_HALLOWEEN_SHOP_ON) { .@warn$[.@wc] = "이벤트 진행 중이지만 교환 NPC 가 OFF 입니다."; .@wc++; }
	}

	mes "[NEED 할로윈 이벤트 관리]";
	mes "서버 시각 : " + gettimestr("%Y-%m-%d %H:%M:%S", 21);
	mes "할로윈 이벤트 전체 : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_EVENT_ON);
	if (.@wc > 0) mes "현재 경고 : ^FF9900" + .@wc + "건^000000";
	else mes "현재 경고 : 0건";
	mes "^999999----------------------------^000000";
	mes "^FFCC00[사냥]^000000";
	mes " [전역] 사냥 월드 드롭 : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_HUNT_ON);
	mes " [CONFIG] 실제 판정   : " + callfunc("F_NHGM_OO", .@flag);
	mes "   ^999999할로윈 코인 4.00% x1, 플레이어 레벨 ±30^000000";
	mes "^FFCC00[전장]^000000";
	mes " [전역] 전장 코인 : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_BG_ON);
	mes "   ^999999티에라 2 / 플라비우스 2 / KVM 3 (승패 무관)^000000";
	mes "^FFCC00[잭 잡기]^000000";
	mes " [전역] 입장 : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_JACK_ENTRY_ON) + " / 보상 : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_JACK_REWARD_ON);
	mes "   ^9999993분 / 15점 / 코인 10개, 계정+IP 하루 1회 (04:00)^000000";
	mes "   ^999999오늘(" + .@day$ + ") 지급 : " + ((.@jack_today < 0) ? "조회 불가" : .@jack_today + "건") + "^000000";
	mes "^FFCC00[교환]^000000";
	mes " [전역] 교환 NPC : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_SHOP_ON);
	mes "   ^999999망치 주 5 (현재 주기 " + callfunc("F_NEED_Chuseok_WeekKey") + ", 초기화 " + callfunc("F_NEED_Chuseok_WeekResetStr$") + ")^000000";
	mes "   ^999999아리아 1 / +9 보장권 1 (계정 기준, event_id 202610)^000000";
	mes "^FFCC00[Event Hub]^000000";
	mes " 버튼 : " + callfunc("F_NHGM_OO", $NEED_EVHUB_ON) + " / 연결 : " + (($NEED_EVHUB_MOD$ == "") ? "(없음)" : $NEED_EVHUB_MOD$) + " " + $NEED_EVHUB_ID;
	mes " payo2 NPC 강제 표시 : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_LEGACY_NPC_ON);
	if (.@wc > 0) {
		mes "^999999----------------------------^000000";
		mes "^FF9900[경고 목록]^000000";
		for (.@i = 0; .@i < .@wc; .@i++)
			mes " - ^FF9900" + .@warn$[.@i] + "^000000";
	}
	next;
	if (.@flag != .@want) {
		if (select("다시 맞추기:돌아가기") == 1) {
			callfunc "F_NEED_Halloween_Sync";
			mes "[할로윈 이벤트]";
			mes "사냥 battle flag 를 " + getbattleflag("need_halloween_hunt_enable") + " 로 맞췄습니다.";
			next;
		}
	}
	return;

S_Master:
	mes "[할로윈 이벤트 전체]";
	mes "현재 상태 : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_EVENT_ON);
	mes "^999999(OFF 로 바꾸면 할로윈에 연결된 Event 버튼도 숨깁니다)^000000";
	mes " ";
	mes "전체 스위치를 바꾸면 아래 항목이 한 번에 같은 값으로 설정됩니다.";
	mes "- 사냥 월드 드롭 / 전장 코인";
	mes "- 잭 잡기 입장 / 잭 잡기 보상 / 교환 NPC";
	mes "- 할로윈 NPC 표시 (추석이 끝났으면 같은 칸 추석 NPC 숨김)";
	mes " ";
	mes "^FF9900개별 조정은 각 하위 메뉴에서 다시 할 수 있습니다.^000000";
	next;
	if (select("상태 변경:돌아가기") == 1) {
		.@on = !$NEED_HALLOWEEN_EVENT_ON;
		$NEED_HALLOWEEN_EVENT_ON = .@on;
		$NEED_HALLOWEEN_HUNT_ON = .@on;
		$NEED_HALLOWEEN_BG_ON = .@on;
		$NEED_HALLOWEEN_JACK_ENTRY_ON = .@on;
		$NEED_HALLOWEEN_JACK_REWARD_ON = .@on;
		$NEED_HALLOWEEN_SHOP_ON = .@on;
		callfunc "F_NEED_Halloween_Sync";
		.@hub = callsub(S_HubOff);
		mes "[할로윈 이벤트 전체]";
		mes "변경됨 : " + callfunc("F_NHGM_OO", .@on);
		mes "사냥 battle flag : " + getbattleflag("need_halloween_hunt_enable");
		mes "Event Hub 버튼 : " + callfunc("F_NHGM_OO", $NEED_EVHUB_ON) + ((.@hub >= 0) ? " (접속자 " + .@hub + "명에게 전송)" : "");
		if (.@on && !$NEED_EVHUB_ON)
			mes "^999999Event 버튼은 [Event Hub 버튼 관리] 에서 따로 켭니다.^000000";
		next;
	}
	return;

S_Hunt:
	while (1) {
		mes "[할로윈 사냥 월드 드롭]";
		mes "스위치      : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_HUNT_ON);
		mes "실제 판정   : " + callfunc("F_NHGM_OO", getbattleflag("need_halloween_hunt_enable"));
		mes " ";
		mes "^999999이벤트 전체가 ON 일 때만 실제로 켜집니다.^000000";
		next;
		if (select("스위치 변경:돌아가기") == 2)
			break;
		$NEED_HALLOWEEN_HUNT_ON = !$NEED_HALLOWEEN_HUNT_ON;
		callfunc "F_NEED_Halloween_Sync";
	}
	return;

S_BG:
	while (1) {
		mes "[할로윈 전장 코인]";
		mes "전장 코인 : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_BG_ON);
		mes " ";
		mes "^999999티에라 2 / 플라비우스 2 / KVM 3, 기존 전장 보상 대상자에게만^000000";
		mes "^999999이벤트 전체가 ON 일 때만 지급됩니다.^000000";
		next;
		if (select("스위치 변경:돌아가기") == 2)
			break;
		$NEED_HALLOWEEN_BG_ON = !$NEED_HALLOWEEN_BG_ON;
	}
	return;

S_Jack:
	while (1) {
		mes "[잭 잡기]";
		mes "입장 : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_JACK_ENTRY_ON);
		mes "보상 : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_JACK_REWARD_ON);
		mes " ";
		mes "^999999입장 OFF : 새 입장만 막습니다 (진행 중인 판은 끝까지).^000000";
		mes "^999999보상 OFF : 성공해도 지급/기록하지 않습니다.^000000";
		next;
		.@sel = select("입장 변경:보상 변경:돌아가기");
		if (.@sel == 3)
			break;
		if (.@sel == 1) $NEED_HALLOWEEN_JACK_ENTRY_ON = !$NEED_HALLOWEEN_JACK_ENTRY_ON;
		else $NEED_HALLOWEEN_JACK_REWARD_ON = !$NEED_HALLOWEEN_JACK_REWARD_ON;
	}
	return;

S_Shop:
	while (1) {
		mes "[할로윈 교환 NPC]";
		mes "교환 NPC : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_SHOP_ON);
		mes " ";
		mes "^999999이벤트 전체가 ON 일 때만 동작합니다.^000000";
		next;
		if (select("스위치 변경:돌아가기") == 2)
			break;
		$NEED_HALLOWEEN_SHOP_ON = !$NEED_HALLOWEEN_SHOP_ON;
	}
	return;

// 할로윈이 꺼지면 할로윈에 연결된 Event 버튼도 끈다. 반환 = 전송한 접속자 수 (-1 = Hub 를 건드리지 않음)
S_HubOff:
	if (!$NEED_HALLOWEEN_EVENT_ON && callfunc("F_NEED_EventHub_IsLive", "F_NEED_HW26_Hub"))
		return callfunc("F_NEED_EventHub_Set", 0, "", 0);
	return -1;

S_Hub:
	while (1) {
		mes "[Event Hub 버튼]";
		mes "버튼 표시 : " + callfunc("F_NHGM_OO", $NEED_EVHUB_ON);
		mes "연결 : " + (($NEED_EVHUB_MOD$ == "") ? "(없음)" : $NEED_EVHUB_MOD$) + " / 번호 " + $NEED_EVHUB_ID;
		mes "payo2 NPC 강제 표시 : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_LEGACY_NPC_ON);
		mes " ";
		mes "^999999할로윈으로 켜면 [!] 왼쪽 Event 버튼이 접속자 전원에게 바로 보이고,^000000";
		mes "^999999payo2 의 꼬마 호박 잭 / 할로윈 상인 잭은 숨깁니다.^000000";
		mes "^999999강제 표시를 켜면 Hub 와 상관없이 두 NPC 가 다시 보입니다.^000000";
		mes "^999999다른 이벤트 연결은 @eventhub 에서 합니다.^000000";
		next;
		.@sel = select("할로윈으로 연결 + 버튼 ON:버튼 OFF:payo2 NPC 강제 표시 바꾸기:돌아가기");
		if (.@sel == 4)
			break;
		if (.@sel == 1)
			.@n = callfunc("F_NEED_EventHub_Set", 1, "F_NEED_HW26_Hub", 202610);
		else if (.@sel == 2)
			.@n = callfunc("F_NEED_EventHub_Set", 0, "", 0);
		else {
			$NEED_HALLOWEEN_LEGACY_NPC_ON = !$NEED_HALLOWEEN_LEGACY_NPC_ON;
			callfunc "F_NEED_Halloween_Sync";
			.@n = -2;
		}
		mes "[Event Hub 버튼]";
		if (.@n == -1)
			mes "^FF0000바꾸지 못했습니다. (할로윈 모듈이 등록되지 않음)^000000";
		else if (.@n >= 0)
			mes "버튼 " + callfunc("F_NHGM_OO", $NEED_EVHUB_ON) + ", 접속자 " + .@n + "명에게 전송했습니다.";
		else
			mes "payo2 NPC 강제 표시 : " + callfunc("F_NHGM_OO", $NEED_HALLOWEEN_LEGACY_NPC_ON);
		next;
	}
	return;

S_End:
	mes "[이벤트 종료 처리]";
	mes "모든 할로윈 스위치를 OFF 로 하고 할로윈 NPC 를 숨깁니다.";
	mes "Event Hub 가 할로윈에 연결돼 있으면 Event 버튼도 숨깁니다.";
	mes "(추석 NPC 표시는 원래대로 돌아갑니다.)";
	mes " ";
	mes "^FF0000잔여 할로윈 코인 삭제는 이 메뉴가 하지 않습니다.^000000";
	mes "서버를 모두 내린 뒤 sql-files/need_halloween_2026_end.sql 을";
	mes "운영 문서(md/NEED_Halloween_2026_Operations.md) 순서대로 실행하세요.";
	next;
	if (select("종료 처리:돌아가기") == 1) {
		$NEED_HALLOWEEN_EVENT_ON = 0;
		$NEED_HALLOWEEN_HUNT_ON = 0;
		$NEED_HALLOWEEN_BG_ON = 0;
		$NEED_HALLOWEEN_JACK_ENTRY_ON = 0;
		$NEED_HALLOWEEN_JACK_REWARD_ON = 0;
		$NEED_HALLOWEEN_SHOP_ON = 0;
		callfunc "F_NEED_Halloween_Sync";
		callsub S_HubOff;
		mes "[이벤트 종료 처리]";
		mes "모든 스위치 OFF, 사냥 battle flag : " + getbattleflag("need_halloween_hunt_enable");
		next;
	}
	return;
}

function	script	F_NHGM_OO	{
	return (getarg(0)) ? "^009900ON^000000" : "^FF0000OFF^000000";
}
