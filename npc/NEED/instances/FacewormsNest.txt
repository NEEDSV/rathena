function	script	F_NEED_FaceFastOK	{
	.@party_id = getcharid(1);
	.@original_aid = getcharid(3);

	if (.@party_id < 1 || !#NEED_MD_FACEWORM_CLEAR)
		return 0;

	getpartymember .@party_id, 2;
	.@member_count = $@partymembercount;
	copyarray .@member_aid[0], $@partymemberaid[0], .@member_count;

	.@eligible = 1;
	for (.@i = 0; .@i < .@member_count; ++.@i) {
		if (!attachrid(.@member_aid[.@i])) {
			.@eligible = 0;
			break;
		}

		if (!#NEED_MD_FACEWORM_CLEAR)
			.@eligible = 0;

		detachrid;

		if (!.@eligible)
			break;
	}

	if (.@original_aid)
		attachrid(.@original_aid);

	return .@eligible;
}

function	script	F_NEED_FaceUnlock	{
	.@party_id = getarg(0);
	.@map_name$ = getarg(1);
	.@original_aid = playerattached() ? getcharid(3) : 0;

	if (.@party_id < 1) {
		if (.@original_aid)
			#NEED_MD_FACEWORM_CLEAR = 1;
		return 0;
	}

	getpartymember .@party_id, 2;
	.@member_count = $@partymembercount;
	copyarray .@member_aid[0], $@partymemberaid[0], .@member_count;

	for (.@i = 0; .@i < .@member_count; ++.@i) {
		if (!attachrid(.@member_aid[.@i]))
			continue;

		if (strcharinfo(3) == .@map_name$)
			#NEED_MD_FACEWORM_CLEAR = 1;
		detachrid;
	}

	if (.@original_aid)
		attachrid(.@original_aid);

	return .@member_count;
}
