// NEED: 마신의 탑 빠른 진행 파티 처리
// mode 0: 대기 플래그 초기화
// mode 1: 전원 클리어/온라인 확인 후 빠른 진행 대기 플래그 설정
// mode 2: 온라인 파티원 전원에게 최초 클리어 기록 저장
function	script	F_NEED_DevilTowerPartyMode	{
	.@party_id = getarg(0);
	.@mode = getarg(1);
	.@origin_aid = getcharid(3);

	if (.@party_id < 1)
		return 0;

	getpartymember .@party_id,1;
	.@count = $@partymembercount;
	if (.@count < 1)
		return 0;
	copyarray .@aid[0],$@partymemberaid[0],.@count;

	if (.@mode == 1) {
		for (.@i = 0; .@i < .@count; .@i++) {
			if (!attachrid(.@aid[.@i])) {
				attachrid(.@origin_aid);
				return 0;
			}
			if (!#NEED_MD_DEVIL_TOWER_CLEAR) {
				attachrid(.@origin_aid);
				return 0;
			}
		}
	}

	for (.@i = 0; .@i < .@count; .@i++) {
		if (!attachrid(.@aid[.@i]))
			continue;
		switch (.@mode) {
		case 0:
			@NEED_DT_FAST_PARTY = 0;
			break;
		case 1:
			@NEED_DT_FAST_PARTY = .@party_id;
			break;
		case 2:
			#NEED_MD_DEVIL_TOWER_CLEAR = 1;
			break;
		}
	}
	attachrid(.@origin_aid);
	return 1;
}
