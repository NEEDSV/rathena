// NEED Horror Toy Factory repeat-clear helpers
//============================================================
function	script	F_NEED_HorrorFastOK	{
	.@party_id = getcharid(1);
	.@original_aid = getcharid(3);

	if (.@party_id < 1 || !#NEED_MD_HORROR_TOY_CLEAR)
		return 0;

	getpartymember .@party_id, 2;
	.@member_count = $@partymembercount;
	copyarray .@member_aid[0], $@partymemberaid[0], .@member_count;

	.@eligible = 1;
	for (.@i = 0; .@i < .@member_count; ++.@i) {
		if (!attachrid(.@member_aid[.@i])) {
			.@eligible = 0;
			break;
		}

		if (!#NEED_MD_HORROR_TOY_CLEAR)
			.@eligible = 0;

		detachrid;

		if (!.@eligible)
			break;
	}

	if (.@original_aid)
		attachrid(.@original_aid);

	return .@eligible;
}

function	script	F_NEED_HorrorUnlock	{
	.@party_id = getarg(0);
	.@map_name$ = getarg(1);
	.@original_aid = 0;
	if (playerattached())
		.@original_aid = getcharid(3);

	if (.@party_id < 1) {
		if (.@original_aid)
			#NEED_MD_HORROR_TOY_CLEAR = 1;
		return 0;
	}

	getpartymember .@party_id, 2;
	.@member_count = $@partymembercount;
	copyarray .@member_aid[0], $@partymemberaid[0], .@member_count;

	for (.@i = 0; .@i < .@member_count; ++.@i) {
		if (!attachrid(.@member_aid[.@i]))
			continue;

		if (strcharinfo(3) == .@map_name$)
			#NEED_MD_HORROR_TOY_CLEAR = 1;
		detachrid;
	}

	if (.@original_aid)
		attachrid(.@original_aid);

	return .@member_count;
}
