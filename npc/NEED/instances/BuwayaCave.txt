		mes "이곳에 들어올 수 없습니다.";
		mes "이곳은 위험합니다. 돌아가 주세요.";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$,"부여워 동굴";

	if (!.@party_id) {
		mes "[경비병]";
		mes "부여워는 위험합니다.";
		mes "다른 사람들과 함께";
		mes "팀을 꾸린 뒤 다시 와 주세요.";
		close;
	}
	if (is_party_leader() == true) {
		mes "[경비병]";
		mes "이곳은 위험합니다. 돌아가 주세요.";
		mes "볼일이 없다면 돌아가 주세요.";
		next;
		while(1) {
			switch(select("왜 그렇게 위험한가?:부여워를 처치하러 왔다.:대화를 마친다.")) {
			case 1:
				mes "[경비병]";
				mes "최근";
				mes "마을 사람들이 사라지는 사건이 발생했습니다.";
				mes "흔적을 조사한 결과,";
				mes "거대한 괴물 부여워가";
				mes "그 원인이라는 사실을 알아냈습니다.";
				next;
				mes "[경비병]";
				mes "부여워를 토벌하기 위해";
				mes "병사들과 용병들이 파견되었지만,";
				mes "완전한 성공을 거두지는 못했습니다.";
				next;
				mes "[경비병]";
				mes "부여워는 지금도";
				mes "마을 사람들을 납치하고 있습니다.";
				mes "부여워를 처치할 사람들을 모아 주세요.";
				next;
				break;
			case 2:
				set .@playtime, checkquest(4229,PLAYTIME);
				if (.@playtime == -1) {
					// fall through
				} else if (.@playtime == 0 || .@playtime == 1) {
					mes "[경비병]";
					mes "부여워는 아직 숨어 있습니다.";
					mes "지금 들어가도 부여워는 나오지 않을 것입니다.";
					mes "나중에 다시 와 주세요.";
					close;
				} else {
					erasequest 4229;
					// fall through
				}
				if (instance_create(.@md_name$) < 0) {
					mes "[경비병]";
					mes "파티 이름은..."+getpartyname(.@party_id)+".";
					mes "파티장은..."+strcharinfo(0);
					mes "^0000ff"+.@md_name$+"^000000 cannot be opened now.";
					mes "잠시 후 다시 시도해 주세요.";
					close;
				}
				mes "[경비병]";
				mes "^0000ff로 가는 통로를 열겠습니다."+.@md_name$+"^000000.";
				mes "행운을 빕니다.";
				close;
			case 3:
				mes "[경비병]";
				mes "이곳은 위험합니다. 들어갈 수 없습니다.";
				close;
			}
		}
	} else {
		mes "[경비병]";
		mes "파티장과 이야기하게 해 주세요.";
		mes "파티장을 불러와 주세요.";
		close;
	}
}
