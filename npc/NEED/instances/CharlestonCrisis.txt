// 빠른 진행용 공통 대화 생략 선택지
function	script	F_CharlestonSkipMenu	{
	if ('need_charleston_fast != 1 || is_party_leader() == false)
		return 0;

	mes "[빠른 진행]";
	mes "이미 확인한 대화와 연출을 생략할 수 있습니다.";
	next;
	return select( "대화와 연출을 확인한다.", "대화와 연출을 생략한다." ) == 2;
}

// 파티 내 온라인 구성원 모두가 최초 클리어를 완료했는지 확인합니다.
function	script	F_CharlestonCanEnableFast	{
	.@caller_aid = getcharid(3);
	.@party_id = getcharid(1);
	if (.@party_id < 1)
		return 0;

	getpartymember .@party_id,1;
	.@member_count = $@partymembercount;
	copyarray .@member_aid[0],$@partymemberaid[0],.@member_count;
	.@allowed = 1;

	for (.@i = 0; .@i < .@member_count; .@i++) {
		if (attachrid(.@member_aid[.@i]) == 0)
			continue;

		if (#NEED_MD_CHARLESTON_CLEAR == 0 && isbegin_quest(13186) == 2)
			#NEED_MD_CHARLESTON_CLEAR = 1;

		if (#NEED_MD_CHARLESTON_CLEAR == 0) {
			.@allowed = 0;
			break;
		}
	}

	attachrid(.@caller_aid);
	return .@allowed;
}

// 빠른 진행 모드를 활성화하고 이동 포탈 및 초반 게이트를 개방합니다.
function	script	F_CharlestonEnableFast	{
	'need_charleston_fast = 1;
	for (.@i = 1; .@i <= 13; .@i++)
		enablenpc instance_npcname("Charleston Fast Travel#" + .@i);

	enablenpc instance_npcname("1gate#1");
	enablenpc instance_npcname("2gate#1");
	enablenpc instance_npcname("3gate#1");
	enablenpc instance_npcname("4gate#1");

	mapannounce 'map_name$, "빠른 진행 모드가 활성화되었습니다. 빠른 이동 포탈과 대화 생략 메뉴를 이용할 수 있습니다.", bc_map, 0x33CCFF;
	return 1;
}
