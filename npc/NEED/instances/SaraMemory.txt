	// 20 hour cooldown
	set .@ins_sara_time,checkquest(15002,PLAYTIME);
	if (.@ins_sara_time == -1) {

		set .@party_id,getcharid(1);
		set .@p_name$,getpartyname(.@party_id);
		set .@md_name$,"사라의 기억";

		if (!.@party_id) {
			mes "[모험가 레온]";

			close;
		}

		if (is_party_leader() == true)
			set .@menu$, "차원 이동 장치를 가동한다.:차원 이동 장치를 사용한다.:취소.";
		else
			set .@menu$, ":차원 이동 장치를 사용한다.:취소.";

		switch(select(.@menu$)) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "파티명: "+.@p_name$;
				mes "파티장: "+strcharinfo(0);
				mes "^0000ff"+.@md_name$+" ^000000- 생성 예약 실패!";
				close;
			}
			mes "^FF0000차원 이동 장치가 정상적으로 가동되었습니다. 장치를 사용하여 시공간의 균열로 입장하십시오.^000000";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "알 수 없는 오류가 발생했습니다.";
				close;
			case IE_NOINSTANCE:
				mes "해당 메모리얼 던전이 "+.@md_name$+" 존재하지 않습니다.";
				mes "파티장이 아직 던전을 생성하지 않았습니다.";
				close;
			case IE_NOMEMBER:
				mes "등록된 파티원만 인스턴스 던전에 입장할 수 있습니다. "+.@md_name$+".";
				close;
			case IE_OK:
				mapannounce "dali",strcharinfo(0)+" 파티원이 "+.@p_name$+" 인스턴스에 입장했습니다. "+.@md_name$+".",bc_map,"0x00ff99";
				setquest 15002;
				end;
			}
		case 3:
			mes "^FF0000파티장이 아직 차원 이동 장치를 가동하지 않았습니다.^000000";
			close;
		}
	} else if (.@ins_sara_time == 0 || .@ins_sara_time == 1) {
		mes "[모험가 레온]";
		mes "다음 탐험을 위해서는 균열이 안정될 때까지 마지막 입장 후 약 20시간 정도의 시간이 필요합니다.";
		close;
	}
	else {
		erasequest 15002;
		mes "^0000FF[ 사라의 기억 ] 대기 시간이 해제되었습니다. 이제 차원 이동 장치를 다시 사용할 수 있습니다.^000000";
