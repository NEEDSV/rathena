L_EraseQuest:
	for(set .@j,0; .@j<getargcount(); set .@j,.@j+1) {
		if (checkquest(getarg(.@j)) > -1)

	close;

L_CheckQuest:
	if (checkquest(getarg(0),HUNTING) == 2) {
		mes "[마데카]";
		mes "^0000ff 녀석들을 아주 잘"+getarg(1)+"^000000!";
		mes "손봐줬구나! 자, 여기 약속한 보상이야!";
		erasequest getarg(0);
		specialeffect2 EF_STEAL;
		getitem getarg(2),getarg(3);
		close;
	}
	return;
}
