		close;
	}

		mes "^ff0000먼저 방황하는 노인을 만나야 이 던전에 입장할 수 있습니다.^000000";
		close;
	}
	.@party_id = getcharid(1);
	if (.@party_id < 1) {
		mes "^ff0000이 던전에 입장하려면 1명 이상의 파티를 생성하거나 가입해야 합니다.^000000";
		close;
	}
	switch( checkquest(15005,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		if (checkquest(15007,PLAYTIME) == 2) {// 재입장 유예 시간 5분
			mes "^ff0000마지막 입장 후 23시간 동안 이 메모리얼 던전에 입장할 수 없습니다.^000000";
			close;
		}
		switch( checkquest(15008,PLAYTIME) ) {
		case -1:
			break;
		case 0:
		case 1:
			mes "^0000ff마지막 입장 후 23시간 동안 이 메모리얼 던전에 입장할 수 없습니다.^000000";
			close;
		case 2:
			break;
		}
		break;
	case 2:
		erasequest 15005;// 메모리얼 던전: 비오스의 섬
		erasequest 15007;// 메모리얼 던전: 비오스의 섬
		if (isbegin_quest(15008) > 0)
			erasequest 15008;// 메모리얼 던전: 비오스의 섬
		break;
	}
	if (is_party_leader() == true)
		.@create$ = "메모리얼 던전 생성";
	switch( select( .@create$, "비오스의 섬 입장", "취소" ) ) {
	case 1:
		instance_create("비오스의 섬");
		mes "^ff0000메모리얼 던전을 생성했습니다.^000000";
		mes "^ff0000다시 말을 걸어 입장해 주세요.^000000";
		close;
	case 2:
		switch( instance_enter("비오스의 섬") ) {
		case IE_NOMEMBER:
		case IE_NOINSTANCE:
		case IE_OTHER:
			if (is_party_leader(.@party_id) == false) {
				mes "^ff0000마지막 입장 후 23시간 동안 이 메모리얼 던전에 입장할 수 없습니다.^000000";
				next;
			}
			mes "^ff0000파티장이 아직 차원의 균열을 생성하지 않았습니다.^000000";
			close;
		case IE_OK:
			mapannounce "moro_cav", getpartyname( getcharid(1) ) + " 파티의 " + strcharinfo(0) + " 님이 비오스의 섬에 입장합니다.",bc_map,"0x00FF99";
			if (isbegin_quest(15005) == 0) {
				setquest 15005;// 메모리얼 던전: 비오스의 섬
				setquest 15007;// 메모리얼 던전: 비오스의 섬
			}
			end;
		}
	case 3:
		end;
	}
	end;

OnInit:
	while(true) {
		sleep 500;
		specialeffect EF_ENHANCE;
	}
	end;
}
