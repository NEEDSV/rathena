		mes "보상 이용 횟수를 확인하지 못했습니다.";

		if (.@remaining <= 0) {

		mes "베이스 레벨 120 이상이어야 합니다.";
		close3;

		case 0:

				mes "옆의 루나인에게 차원의 균열을 열어 달라고 해주세요.";

			}
			close3;

			mes "[로엘]";
			mes "입장 준비가 되었다면 루나인에게 말해주세요.";
			close3;
		case 2:
			cutin "roel03",0;
			mes "[로엘]";
			mes "첫 원정을 무사히 마치셨군요.";

moc_para01,29,95,5	script	루나인#sara	4_F_RUNAIN,{
	cutin "lunain01",0;
	if (BaseLevel < 120) {
		mes "[루나인]";
		mes "^ff0000베이스 레벨 120 이상만 이용할 수 있습니다.^000000";
		close3;
	}
	if (isbegin_quest(13181) != 1) {
		mes "[루나인]";
		mes "먼저 로엘에게 원정 참가 신청을 해주세요.";
		close3;
	}

	.@party_id = getcharid(1);
	.@md_name$ = "악몽의 지터벅";

	if (.@party_id < 1) {
		mes "[루나인]";
		mes "파티를 구성한 뒤 다시 찾아와 주세요.";
		close3;
	}

	switch (checkquest(13182,PLAYTIME)) {
	case -1:
		break;
	case 0:
	case 1:
		mes "[루나인]";
		mes "^ff0000마지막 입장 후 20시간 동안 다시 입장할 수 없습니다.^000000";
		close3;
	case 2:
		erasequest 13182;
		mes "[루나인]";
		mes "입장 제한 시간이 종료되었습니다.";
		close3;
	}

	if (is_party_leader())
		.@choice = select("차원의 균열을 연다.:악몽의 지터벅에 입장한다.:취소한다.");
	else
		.@choice = select("악몽의 지터벅에 입장한다.:취소한다.") + 1;

	switch (.@choice) {
	case 1:
		if (!is_party_leader()) {
			mes "[루나인]";
			mes "차원의 균열은 파티장만 생성할 수 있습니다.";
			close3;
		}
		if (instance_create(.@md_name$) < 0) {
			mes "[루나인]";
			mes "메모리얼 던전 생성에 실패했습니다.";
			close3;
		}
		mes "[루나인]";
		mes "차원의 균열을 열었습니다.";
		mes "다시 말을 걸어 입장해 주세요.";
		close3;

	case 2:
		switch (instance_enter(.@md_name$)) {
		case IE_NOMEMBER:
			mes "[루나인]";
			mes "해당 파티의 구성원만 입장할 수 있습니다.";
			close3;
		case IE_NOINSTANCE:
			mes "[루나인]";
			mes "파티장이 아직 차원의 균열을 생성하지 않았습니다.";
			close3;
		case IE_OTHER:
			mes "[루나인]";
			mes "알 수 없는 오류가 발생했습니다.";
			close3;
		case IE_OK:
			setquest 13182;
			mapannounce "moc_para01",
				getpartyname(getcharid(1)) + " 파티의 " + strcharinfo(0) + " 님이 악몽의 지터벅에 입장합니다.",
				bc_map,0xFF99;
			specialeffect2 EF_READYPORTAL2;
			specialeffect2 EF_PORTAL2;
			end;
		}
	case 3:
		cutin "",255;
		end;
	}
}

1@jtb,28,30,5	script	뉴오즈#01	4_M_NEWOZ,{
	if ('room != 1) {
		mes "[뉴오즈]";
		mes "이미 모험이 시작되었거나 다른 파티원이 진행 중이야.";
		close;
	}

	// 선택 결과는 로컬 변수에 보관하고, 전환 잠금을 획득한 한 명의 선택만 적용합니다.
	.@fast_mode = 0;
	.@options = 0;

	// 반복 클리어 계정은 시작 시 빠른 진행 방식을 선택할 수 있습니다.
	if (isbegin_quest(13183) == 2) {
		mes "[뉴오즈]";
		mes "이 공간은 같은 사건을 반복하고 있어.";
		mes "이번에는 어떻게 진행할까?";
		next;

		switch (select("^3355FF빠른 진행 - 추가 지터벅 없음^000000:^FF5500빠른 진행 - 추가 지터벅 모두 등장^000000:일반 진행 - 대화 축약")) {
		case 1:
			.@fast_mode = 1;
			.@options = 0;
			break;
		case 2:
			.@fast_mode = 1;
			.@options = 31;
			break;
		case 3:
			.@fast_mode = 0;
			.@options = 0;
			break;
		}
	}

	if (.@fast_mode) {
		mes "[뉴오즈]";
		mes "이야기 구간을 생략합니다.";
		if (.@options == 31)
			mes "추가 지터벅 전투는 모두 활성화됩니다.";
		else
			mes "추가 지터벅 전투는 발생하지 않습니다.";
		next;

		if ('room != 1 || 'jitter_transition) {
			mes "[뉴오즈]";
			mes "다른 파티원이 먼저 모험을 시작했습니다.";
			close;
		}
		'jitter_transition = true;
		'NEED_JITTER_FAST = .@fast_mode;
		'jitterbug_options = .@options;

		disablenpc instance_npcname("로엘#01");
		disablenpc instance_npcname("멜로디 잭#01");
		disablenpc instance_npcname("아랑#01");
		disablenpc instance_npcname("라기#01");
		disablenpc instance_npcname("아이구#01");
		disablenpc instance_npcname("밍민#01");
		disablenpc instance_npcname("루나인#01");
		disablenpc instance_npcname("겔카#01");
		disablenpc instance_npcname("로터#01");
		disablenpc instance_npcname("뉴오즈#01");

		donpcevent 'jitterbug_main$ + "::OnNextIs2";
		close;
	}

	mes "[뉴오즈]";
	mes "드디어 모두 모였군!";
	mes "이곳은 길과 풍경이 계속 뒤틀리는 수상한 공간이야.";
	next;
	mes "[로터]";
	mes "뉴오즈를 찾는 데 시간을 너무 많이 썼다.";
	mes "우선 출구를 찾으며 이 공간의 정체를 조사하자.";
	next;
	mes "[밍민]";
	mes "공기와 땅에서 비정상적인 기운이 느껴져.";
	mes "몬스터가 나타날 수 있으니 조심해.";
	next;

	if ('room == 1 && !'jitter_transition) {
		'jitter_transition = true;
		'NEED_JITTER_FAST = .@fast_mode;
		'jitterbug_options = .@options;
		donpcevent 'jitterbug_main$ + "::OnNextIs2";
	}

	mes "[뉴오즈]";
	mes "좋아, 새로운 모험을 시작하자!";
	close;
}
