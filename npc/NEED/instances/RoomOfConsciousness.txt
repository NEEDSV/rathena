//=       계정 단위 #NEED_RC_CLEAR 로 최초 클리어 기록.
//=       파티 내 온라인 구성원 전원이 클리어한 경우 재방문 간소화.

// NEED: 파티 내 온라인 구성원 전원이 의식의 방을 1회 이상 완료했는지 확인합니다.
function	script	F_RoomConsciousnessCanEnableSimple	{
	.@caller_aid = getcharid(3);
	.@party_id = getcharid(1);
	if (.@party_id < 1)
		return 0;

	getpartymember .@party_id,1;
	.@member_count = $@partymembercount;
	copyarray .@member_cid[0],$@partymembercid[0],.@member_count;

	getpartymember .@party_id,2;
	copyarray .@member_aid[0],$@partymemberaid[0],.@member_count;

	.@allowed = 1;
	for (.@i = 0; .@i < .@member_count; ++.@i) {
		if (!isloggedin(.@member_aid[.@i], .@member_cid[.@i]))
			continue;

		if (attachrid(.@member_aid[.@i]) == 0)
			continue;

		// NEED: EP16.1 메인 스토리로 의식의 방을 진행 중인 파티원이 한 명이라도 있으면
		// 간소화 모드를 사용하지 않고 원본 스토리 연출을 그대로 진행한다.
		// 간소화 종료 분기는 7700 -> 7701 / ep16_royal = 20 갱신 단계를 건너뛰므로,
		// 그대로 두면 메인 스토리가 ep16_royal = 19 에서 멈춘다.
		if (ep16_royal == 19 && isbegin_quest(7700) == 1) {
			.@allowed = 0;
			break;
		}

		// 기존 EP16.1 스토리 완료 캐릭터는 최초 1회 자동 이관.
		if (#NEED_RC_CLEAR == 0 && ep16_royal >= 20)
			#NEED_RC_CLEAR = 1;

		if (#NEED_RC_CLEAR == 0) {
			.@allowed = 0;
			break;
		}
	}

	attachrid(.@caller_aid);
	return .@allowed;
}

// NEED: 클리어 시 파티 내 온라인 구성원 모두에게 계정 단위 클리어 기록을 저장합니다.
function	script	F_RoomConsciousnessMarkClear	{
	.@caller_aid = getcharid(3);
	.@party_id = getcharid(1);

	if (.@party_id < 1) {
		#NEED_RC_CLEAR = 1;
		return;
	}

	getpartymember .@party_id,1;
	.@member_count = $@partymembercount;
	copyarray .@member_cid[0],$@partymembercid[0],.@member_count;

	getpartymember .@party_id,2;
	copyarray .@member_aid[0],$@partymemberaid[0],.@member_count;

	for (.@i = 0; .@i < .@member_count; ++.@i) {
		if (!isloggedin(.@member_aid[.@i], .@member_cid[.@i]))
			continue;

		if (attachrid(.@member_aid[.@i]) == 0)
			continue;

		#NEED_RC_CLEAR = 1;
	}

	attachrid(.@caller_aid);
	return;
}

// NEED: 간소화 모드로 종료되더라도 EP16.1 메인 스토리 진행 중인 파티원의
// 진행도(7700 -> 7701 / ep16_royal = 20)가 유실되지 않도록 보정합니다.
// 원본 스토리 모드는 펜릴#6mir 마무리 대화에서 같은 처리를 수행하므로
// 이 함수는 간소화 종료 분기 전용 안전장치입니다.
function	script	F_RoomConsciousnessFinishStory	{
	if (playerattached() == 0)
		return;

	.@caller_aid = getcharid(3);
	.@party_id = getcharid(1);

	if (.@party_id < 1) {
		if (ep16_royal == 19 && isbegin_quest(7700) == 1) {
			erasequest 7700;// 다시 한 번!
			setquest 7701;// 잃어버린 이미르의 심장
			ep16_royal = 20;
		}
		return;
	}

	getpartymember .@party_id,1;
	.@member_count = $@partymembercount;
	copyarray .@member_cid[0],$@partymembercid[0],.@member_count;

	getpartymember .@party_id,2;
	copyarray .@member_aid[0],$@partymemberaid[0],.@member_count;

	for (.@i = 0; .@i < .@member_count; ++.@i) {
		if (!isloggedin(.@member_aid[.@i], .@member_cid[.@i]))
			continue;

		if (attachrid(.@member_aid[.@i]) == 0)
			continue;

		if (ep16_royal == 19 && isbegin_quest(7700) == 1) {
			erasequest 7700;// 다시 한 번!
			setquest 7701;// 잃어버린 이미르의 심장
			ep16_royal = 20;
		}
	}

	attachrid(.@caller_aid);
	return;
}
