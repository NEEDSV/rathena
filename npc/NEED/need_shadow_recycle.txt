//============================================================
// NEED 쉐도우 재활용 시스템
//
// 기능:
// - 장착하지 않은 일반 쉐도우 장비를 전부 일괄 변환
// - 쉐도우 장비 1개당 쉐도우 재활용 코인 1개 지급
// - 장착 중인 장비 및 기간제 장비 제외
// - 코인 사용처 안내
//
// 코인 아이템 ID: 399920
//============================================================

bat_room,150,150,4	script	쉐도우 재활용 관리인#need_shadow	4_M_REPAIR,{

	mes "[쉐도우 재활용 관리인]";
	mes "불필요한 쉐도우 장비를";
	mes "^0055FF" + getitemname(.CoinItemId) + "^000000로";
	mes "변환하여 보관해 드리고 있습니다.";
	next;

	switch (select(
		"쉐도우 장비 일괄 변환:" +
		"쉐도우 재활용 코인 사용처 안내:" +
		"대화를 종료한다"
	)) {
		case 1:
			goto L_Convert;

		case 2:
			goto L_Guide;
	}

	close;


//------------------------------------------------------------
// 쉐도우 장비 일괄 변환
//------------------------------------------------------------
L_Convert:

	.@convertable = 0;
	.@equipped = 0;
	.@rental = 0;

	getinventorylist;

	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {

		// 쉐도우 장비가 아니면 제외
		if (getiteminfo(@inventorylist_id[.@i], ITEMINFO_TYPE) != IT_SHADOWGEAR)
			continue;

		// 장착 중인 쉐도우 장비 제외
		if (@inventorylist_equip[.@i] != 0) {
			.@equipped += @inventorylist_amount[.@i];
			continue;
		}

		// 기간제 쉐도우 장비 제외
		if (@inventorylist_expire[.@i] != 0) {
			.@rental += @inventorylist_amount[.@i];
			continue;
		}

		.@convertable += @inventorylist_amount[.@i];
	}

	mes "[쉐도우 재활용 관리인]";
	mes "장착하지 않은 쉐도우 장비를";
	mes "일괄적으로 변환합니다.";
	mes " ";
	mes "^FF0000제련도, 카드, 랜덤 옵션 및";
	mes "귀속 정보는 보존되지 않습니다.^000000";
	mes " ";
	mes "변환 가능: ^0055FF" + .@convertable + "개^000000";
	mes "획득 예정: ^0055FF" + getitemname(.CoinItemId) + " " + .@convertable + "개^000000";
	mes " ";
	mes "제외된 장비";
	mes "장착 중: " + .@equipped + "개";
	mes "기간제: " + .@rental + "개";

	if (.@convertable <= 0) {
		mes " ";
		mes "현재 변환할 수 있는 쉐도우 장비가 없습니다.";
		close;
	}

	.@preview_count = .@convertable;

	next;

	mes "[쉐도우 재활용 관리인]";
	mes "쉐도우 장비 ^0055FF" + .@preview_count + "개^000000를 소모하여";
	mes "^0055FF" + getitemname(.CoinItemId) + " " + .@preview_count + "개^000000를 획득합니다.";
	mes " ";
	mes "^FF0000변환된 쉐도우 장비는 복구할 수 없습니다.^000000";
	mes "필요한 장비는 착용하거나";
	mes "창고에 보관한 뒤 진행해 주세요.";

	if (select("일괄 변환한다:취소한다") != 1)
		close;


	//--------------------------------------------------------
	// 최종 확인 후 인벤토리를 다시 검사
	//--------------------------------------------------------
	.@current_count = 0;

	getinventorylist;

	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {

		if (getiteminfo(@inventorylist_id[.@i], ITEMINFO_TYPE) != IT_SHADOWGEAR)
			continue;

		if (@inventorylist_equip[.@i] != 0)
			continue;

		if (@inventorylist_expire[.@i] != 0)
			continue;

		.@current_count += @inventorylist_amount[.@i];
	}

	// 확인 도중 쉐도우 수량이 변경된 경우 취소
	if (.@current_count != .@preview_count) {
		mes "[쉐도우 재활용 관리인]";
		mes "확인 도중 인벤토리의";
		mes "쉐도우 장비 수량이 변경되었습니다.";
		mes "처음부터 다시 진행해 주세요.";
		close;
	}

	if (!checkweight(.CoinItemId, .@current_count)) {
		mes "[쉐도우 재활용 관리인]";
		mes "보상 아이템을 받을 수 있는";
		mes "무게 또는 인벤토리 공간이 부족합니다.";
		close;
	}


	//--------------------------------------------------------
	// 인벤토리 슬롯을 기준으로 장비 삭제
	//--------------------------------------------------------
	.@converted = 0;

	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {

		if (getiteminfo(@inventorylist_id[.@i], ITEMINFO_TYPE) != IT_SHADOWGEAR)
			continue;

		if (@inventorylist_equip[.@i] != 0)
			continue;

		if (@inventorylist_expire[.@i] != 0)
			continue;

		.@amount = @inventorylist_amount[.@i];

		if (delitemidx(@inventorylist_idx[.@i], .@amount))
			.@converted += .@amount;
	}

	if (.@converted <= 0) {
		mes "[쉐도우 재활용 관리인]";
		mes "변환할 수 있는 쉐도우 장비가 없어";
		mes "작업을 취소했습니다.";
		close;
	}

	// 계정 귀속 코인 지급
	getitembound .CoinItemId, .@converted, BOUND_ACCOUNT;

	// 계정별 누적 변환량 기록
	#NEED_ShadowRecycleTotal += .@converted;

	// 운영 로그
	logmes "[ShadowRecycle] AID:" + getcharid(3)
		+ " CID:" + getcharid(0)
		+ " converted:" + .@converted;

	mes "[쉐도우 재활용 관리인]";
	mes "쉐도우 장비 ^0055FF" + .@converted + "개^000000를";
	mes "^0055FF" + getitemname(.CoinItemId) + " " + .@converted + "개^000000로";
	mes "변환했습니다.";
	close;


//------------------------------------------------------------
// 쉐도우 재활용 코인 사용처 안내
//------------------------------------------------------------
L_Guide:

	mes "[쉐도우 재활용 관리인]";
	mes "^0055FF" + getitemname(.CoinItemId) + "^000000는";
	mes "현재 사용할 수 없는 임시 보관 재화입니다.";
	mes " ";
	mes "쉐도우 장비 1개는";
	mes "^0055FF" + getitemname(.CoinItemId) + " 1개^000000로 환산됩니다.";
	mes " ";
	mes "추후 쉐도우 뚝딱상자가 도입되면";
	mes "^0055FF코인 3개를 쉐도우 장비 3개와";
	mes "동일한 재료로 사용할 수 있도록 연동됩니다.^000000";
	mes " ";
	mes "기간제 쉐도우 장비는";
	mes "영구 재화로 변환될 수 있어 제외됩니다.";
	close;


//------------------------------------------------------------
// 초기 설정
//------------------------------------------------------------
OnInit:

	.CoinItemId = 399920;
	end;
}
