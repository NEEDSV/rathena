//============================================================
// NEED 2026-10 이펙트 스톤 회수 보상 발송 (일회성)
//============================================================
//= 의상 인챈트 스톤 상자9 에서 나오던 이펙트 스톤 3종
//= (25138 슈링크 중단 / 25136 일렉트릭 중단 / 25137 그린 플로어 하단) 회수분을
//= 의상 인챈트 스톤 상자9 로 돌려주는 우편 발송 도구입니다.
//=
//= 대상 : sql-files/need_effectstone_recall.sql 을 점검 중(서버 OFF)에 실행해 시드
//= 지급 : 회수 1개(옷장 부여분 1건 포함)당 의상 인챈트 스톤 상자9 1개, 계정당 우편 1통
//=
//= 한 번에 .Batch 건씩 처리하고, 중단되어도 다시 실행하면 이어서 발송합니다.
//= 이미 발송한 계정은 status=1 이라 중복 발송되지 않습니다.
//=
//= 발송 완료 후 이 파일과 scripts_need.conf 등록 줄을 삭제하세요.
//============================================================

-	script	need_effectstone_recall	-1,{
	if (getgmlevel() < 99) end;
	callsub S_Main;
	end;

OnCommand:
	if (getgmlevel() < 99) end;
	callsub S_Main;
	end;

OnInit:
	.BoxId = 23058;		// 의상 인챈트 스톤 상자9
	.Batch = 100;		// 1회 실행당 발송 건수
	.Sender$ = "NEED";
	.Title$ = "이펙트 스톤 회수 보상 안내";
	bindatcmd "effectrecall", "need_effectstone_recall::OnCommand", 99, 99;
	end;

S_Main:
	while (1) {
		callsub S_Counts;

		mes "[이펙트 스톤 회수 보상]";
		mes "지급품 : " + getitemname(.BoxId) + " (회수 1개당 1개, 우편)";
		mes "^999999----------------------------^000000";
		mes "발송 대기 : ^FF9900" + @cnt_wait + "^000000 계정 / " + @box_wait + "개";
		mes "발송 완료 : ^009900" + @cnt_done + "^000000 계정 / " + @box_done + "개";
		mes "제외 : " + @cnt_skip + " 계정";
		if (@cnt_done > 0)
			mes "우편함 확인 : ^0055FF" + @cnt_inbox + "^000000 / " + @cnt_done + " 건";
		next;

		.@sel = select(
			"발송 실행 (최대 " + .Batch + "건)",
			"대상 목록 미리보기",
			"닫기");

		if (.@sel == 1) callsub S_Send;
		else if (.@sel == 2) callsub S_Preview;
		else break;
	}
	close;

// 현황 집계. 캐릭터 임시변수로 돌려준다 (callsub 는 .@ 를 공유하지 않는다).
S_Counts:
	query_sql("SELECT COUNT(*), COALESCE(SUM(`box_amount`),0) FROM `need_effectstone_recall` WHERE `status`=0", @cnt_wait, @box_wait);
	query_sql("SELECT COUNT(*), COALESCE(SUM(`box_amount`),0) FROM `need_effectstone_recall` WHERE `status`=1", @cnt_done, @box_done);
	query_sql("SELECT COUNT(*) FROM `need_effectstone_recall` WHERE `status`=2", @cnt_skip);
	query_sql("SELECT COUNT(DISTINCT nr.`account_id`) FROM `need_effectstone_recall` nr JOIN `mail` m ON m.`dest_id`=nr.`char_id` JOIN `mail_attachments` a ON a.`id`=m.`id` AND a.`nameid`=" + .BoxId + " WHERE nr.`status`=1", @cnt_inbox);
	return;

S_Preview:
	.@n = query_sql("SELECT `account_id`,`char_name`,`box_amount`,`removed_25136`,`removed_25137`,`removed_25138`,`kept_25136`,`removed_closet` FROM `need_effectstone_recall` WHERE `status`=0 ORDER BY `account_id` LIMIT 10", .@aid, .@cn$, .@box, .@r36, .@r37, .@r38, .@k36, .@rcl);

	mes "[이펙트 스톤 회수 보상] 대기 목록 (최대 10건)";
	if (.@n < 1)
		mes "발송 대기 중인 계정이 없습니다.";
	for (.@i = 0; .@i < .@n; .@i++)
		mes " " + .@aid[.@i] + " / " + .@cn$[.@i] + " / 상자 " + .@box[.@i] + " (일렉" + .@r36[.@i] + " 그린" + .@r37[.@i] + " 슈링" + .@r38[.@i] + ", 옷장" + .@rcl[.@i] + ", 남김" + .@k36[.@i] + ")";
	next;
	return;

S_Send:
	.@n = query_sql("SELECT `account_id`,`char_id`,`box_amount` FROM `need_effectstone_recall` WHERE `status`=0 ORDER BY `account_id` LIMIT " + .Batch, .@aid, .@cid, .@box);

	if (.@n < 1) {
		mes "[이펙트 스톤 회수 보상]";
		mes "발송할 대상이 없습니다.";
		next;
		return;
	}

	mes "[이펙트 스톤 회수 보상]";
	mes "^FF0000" + .@n + "계정^000000 에게 " + getitemname(.BoxId) + " 를 우편 발송합니다.";
	mes " ";
	mes "진행하시겠습니까?";
	next;
	if (select("발송한다:취소") == 2)
		return;

	freeloop(1);
	.@sent = 0;
	.@boxes = 0;

	for (.@i = 0; .@i < .@n; .@i++) {
		if (.@cid[.@i] <= 0 || .@box[.@i] <= 0) {
			query_sql("UPDATE `need_effectstone_recall` SET `status`=2,`note`='NO_CHAR_OR_ZERO' WHERE `account_id`=" + .@aid[.@i] + " AND `status`=0");
			continue;
		}

		// 상태를 먼저 기록한다 (query_sql 은 UPDATE 의 영향 행 수를 돌려주지 않는다)
		query_sql("UPDATE `need_effectstone_recall` SET `status`=1,`sent_at`=NOW() WHERE `account_id`=" + .@aid[.@i] + " AND `status`=0");

		deletearray .@mitem;
		deletearray .@mamt;
		setarray .@mitem[0], .BoxId;
		setarray .@mamt[0], .@box[.@i];
		mail .@cid[.@i], .Sender$, .Title$,
			"안녕하세요. NEED 운영팀입니다.\n\n의상 인챈트 스톤 상자9 에서 나오던 이펙트 스톤(옷장에 부여된 이펙트 포함)이 회수되어, 회수 수량만큼 의상 인챈트 스톤 상자9 를 보내드립니다.\n\n우편함이 가득 차 있으면 받을 수 없으니 정리 후 수령해 주세요.\n\n감사합니다.",
			0, .@mitem, .@mamt;

		.@sent++;
		.@boxes += .@box[.@i];
	}

	freeloop(0);

	logmes "[EffectStoneRecall] sent=" + .@sent + " boxes=" + .@boxes;
	debugmes "[EffectStoneRecall] 발송 " + .@sent + "건, 상자 " + .@boxes + "개";

	mes "[이펙트 스톤 회수 보상]";
	mes "^009900" + .@sent + "건^000000 발송했습니다. (상자 " + .@boxes + "개)";
	next;
	return;
}
