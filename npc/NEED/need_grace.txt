function	script	NeedGrace_GoldPCMenu	{
	set .@unit_seconds, $@NeedGraceGoldPCUnitSeconds;
	set .@unit_cost, $@NeedGraceGoldPCUnitCost;
	set .@max_seconds, $@NeedGraceGoldPCMaxSeconds;
	set .@remain_ms, getstatus(SC_NEED_GOLDPC, 5);

	if (.@remain_ms < 0)
		set .@remain_ms, 0;

	set .@remain_seconds, (.@remain_ms + 999) / 1000;
	if (.@remain_seconds > .@max_seconds)
		set .@remain_seconds, .@max_seconds;

	set .@available_units, (.@max_seconds - .@remain_seconds) / .@unit_seconds;

	mes needtr("[골드 PC방 은총]","[Gold PC Room Grace]");
	mes needtr("일부 장비의 골드 PC방 전용 옵션을 활성화합니다.","Activates the Gold PC Room-only options on some equipment.");
	mes needtr("^FF000030분당 ", "^FF0000") + callfunc("F_InsertComma", .@unit_cost) + needtr(" 제니^000000", " Zeny per 30 minutes^000000");
	mes needtr("최대 ^FF000024시간^000000까지 누적할 수 있습니다.","You can stack up to ^FF000024 hours^000000.");
	mes needtr("로그아웃 중에는 이용 시간이 감소하지 않습니다.","The time does not tick down while you are logged out.");
	next;

	mes needtr("[골드 PC방 은총]","[Gold PC Room Grace]");
	callfunc "NeedGrace_ShowStatusRemainLine", needtr("현재 남은 시간","Current time remaining"), .@remain_ms;

	if (.@available_units <= 0) {
		mes needtr("이미 최대 이용 시간인 24시간이 적용되어 있습니다.","You already have the maximum of 24 hours.");
		return;
	}

	mes needtr("현재 추가 구매 가능: ^FF0000","You can still buy: ^FF0000") + (.@available_units * 30) + needtr("분^000000","m^000000");
	next;

	switch(select(needtr("이용 시간 구매:뒤로가기","Buy time:Back"))) {
	case 1:
		mes needtr("[골드 PC방 은총]","[Gold PC Room Grace]");
		mes needtr("구매할 30분 단위 횟수를 입력해주세요.","Enter how many 30-minute units to buy.");
		mes needtr("1회 = 30분 / 5,000,000 제니","1 unit = 30 min / 5,000,000 Zeny");
		mes needtr("입력 가능 범위: 1 ~ ","Valid range: 1 ~ ") + .@available_units + needtr("회"," units");
		next;

		input .@amount;

		if (.@amount <= 0 || .@amount > 48) {
			mes needtr("[골드 PC방 은총]","[Gold PC Room Grace]");
			mes needtr("1회 이상, 최대 48회까지만 입력할 수 있습니다.","Enter between 1 and 48 units.");
			close;
		}

		if (.@amount > .@available_units) {
			mes needtr("[골드 PC방 은총]","[Gold PC Room Grace]");
			mes needtr("현재 남은 시간을 포함하여 최대 24시간까지만 구매할 수 있습니다.","Including the time you already have, you can hold at most 24 hours.");
			mes needtr("현재 추가 구매 가능: ^FF0000","You can still buy: ^FF0000") + .@available_units + needtr("회 ("," units (") + (.@available_units * 30) + needtr("분)^000000","m)^000000");
			close;
		}

		set .@add_seconds, .@amount * .@unit_seconds;
		set .@add_ms, .@add_seconds * 1000;
		set .@cost, .@amount * .@unit_cost;

		mes needtr("[골드 PC방 은총]","[Gold PC Room Grace]");
		mes needtr("구매 시간: ^FF0000","Time purchased: ^FF0000") + (.@amount * 30) + needtr("분^000000","m^000000");
		mes needtr("필요 제니: ^FF0000", "Zeny required: ^FF0000") + callfunc("F_InsertComma", .@cost) + " z^000000";
		mes needtr("구매하시겠습니까?", "Would you like to purchase");
		next;

		if (select(needtr("구매한다:취소","Buy:Cancel")) == 2)
			close;

		if (Zeny < .@cost) {
			mes needtr("[골드 PC방 은총]","[Gold PC Room Grace]");
			mes needtr("제니가 부족합니다.","You do not have enough Zeny.");
			mes needtr("필요 제니: ^FF0000", "Zeny required: ^FF0000") + callfunc("F_InsertComma", .@cost) + " z^000000";
			close;
		}

		// 대화 도중 경과한 시간을 반영해 현재 남은 시간을 다시 조회한다.
		set .@remain_ms, getstatus(SC_NEED_GOLDPC, 5);
		if (.@remain_ms < 0)
			set .@remain_ms, 0;

		set .@new_ms, .@remain_ms + .@add_ms;
		set .@max_ms, .@max_seconds * 1000;

		if (.@new_ms > .@max_ms) {
			mes needtr("[골드 PC방 은총]","[Gold PC Room Grace]");
			mes needtr("현재 남은 시간을 포함하여 최대 24시간까지만 구매할 수 있습니다.","Including the time you already have, you can hold at most 24 hours.");
			close;
		}

		Zeny -= .@cost;
		sc_start SC_NEED_GOLDPC, .@new_ms, 1;

		mes needtr("[골드 PC방 은총]","[Gold PC Room Grace]");
		mes needtr("골드 PC방 은총이 적용되었습니다.","Gold PC Room Grace has been applied.");
		callfunc "NeedGrace_ShowStatusRemainLine", needtr("남은 시간","Time remaining"), .@new_ms;
		close;
	default:
		return;
	}
}

function	script	NeedGrace_ApplyBuff	{
	set .@amount, getarg(0);
	set .@coin, $@NeedGraceCoin;
	set .@now, gettimetick(2);

	if (.@amount <= 0) {
		mes needtr("[NEED의 은총]","[NEED Grace]");
		mes needtr("1개 이상 입력해주세요.","Enter at least 1.");
		close;
	}

	if (countitem(.@coin) < .@amount) {
		mes needtr("[NEED의 은총]","[NEED Grace]");
		mes needtr("골드코인이 부족합니다.","You do not have enough Gold Coins.");
		mes needtr("보유 골드코인: ^FF0000", "Gold Coins held: ^FF0000") + countitem(.@coin) + needtr("개^000000", "^000000");
		close;
	}

	set .@total_hours, .@amount * 6;
	set .@days, .@total_hours / 24;
	set .@hours, .@total_hours % 24;

	mes needtr("[NEED의 은총]","[NEED Grace]");
	mes needtr("골드코인 ^FF0000","Use ^FF0000") + .@amount + needtr("개^000000를 사용하시겠습니까?"," Gold Coin(s)^000000?");
	mes needtr("연장 시간: ^FF0000","Extension: ^FF0000") + .@total_hours + needtr("시간^000000","h^000000");

	if (.@days > 0)
		mes needtr("환산 시간: ","That is: ") + .@days + needtr("일 ","d ") + .@hours + needtr("시간","h");

	mes needtr("진행하시겠습니까?","Do you want to proceed?");
	next;

	if (select(needtr("연장한다:취소","Extend:Cancel")) == 2)
		close;

	delitem .@coin, .@amount;

	if (NeedGraceUntil > .@now)
		NeedGraceUntil += .@amount * 21600;
	else
		NeedGraceUntil = .@now + (.@amount * 21600);

	sc_start SC_NEED_GRACE, (NeedGraceUntil - .@now) * 1000, 1;

	mes needtr("[NEED의 은총]","[NEED Grace]");
	mes needtr("NEED의 은총이 적용되었습니다.","NEED Grace has been applied.");
	callfunc "NeedGrace_ShowRemainLine", needtr("남은 시간","Time remaining"), NeedGraceUntil;
	close;
}

function	script	NeedGrace_BuyTicket	{
	set .@type, getarg(0);
	set .@now, gettimetick(2);
	set .@seconds, $@NeedGraceWeeklySeconds;

	switch(.@type) {
	case 1:
		set .@item, $@NeedGraceTradeTicket;
		set .@cost, $@NeedGraceTradeCost;
		set .@label$, needtr("거래 은총 이용권","Grace of Trade Pass");
		set .@effect$, needtr("오버챠지 / 디스카운트 편의 기능", "Overcharge / Discount convenience");
		set .@old_until, NeedGraceTradeUntil;
		break;
	case 2:
		set .@item, $@NeedGraceElementTicket;
		set .@cost, $@NeedGraceElementCost;
		set .@label$, needtr("속성 은총 이용권","Grace of Attribute Pass");
		set .@effect$, needtr("속성 부여 기능", "Attribute endowment");
		set .@old_until, NeedGraceElementUntil;
		break;
	case 3:
		set .@item, $@NeedGraceStorageTicket;
		set .@cost, $@NeedGraceStorageCost;
		set .@label$, needtr("보관 은총 이용권","Grace of Storage Pass");
		set .@effect$, needtr("@창고보관 기능", "@창고보관 storage access");
		set .@old_until, NeedGraceStorageUntil;
		break;
	case 4:
		set .@item, $@NeedGracePackageTicket;
		set .@cost, $@NeedGracePackageCost;
		set .@label$, needtr("종합 은총 이용권","Full Grace Pass");
		set .@effect$, needtr("거래 / 속성 / 보관 은총 전체", "All of Trade / Attribute / Storage Grace");
		set .@old_until, NeedGracePackageUntil;
		break;
	default:
		mes needtr("[주간 은총 이용권]","[Weekly Grace Pass]");
		mes needtr("잘못된 이용권입니다.","That is not a valid pass.");
		close;
	}

	if (.@item <= 0) {
		mes needtr("[주간 은총 이용권]","[Weekly Grace Pass]");
		mes needtr("이용권 아이템 ID가 설정되지 않았습니다.","The pass item ID has not been set.");
		mes needtr("스크립트 OnInit의 티켓 아이템 ID를 확인해주세요.","Please check the ticket item IDs in the script OnInit.");
		close;
	}

	mes needtr("[주간 은총 이용권]","[Weekly Grace Pass]");
	mes .@label$ + needtr("을 구매하시겠습니까?", " - would you like to purchase?");
	mes needtr("효과: ", "Effect: ") + .@effect$;
	mes needtr("기간: 7일","Duration: 7 days");
	mes needtr("비용: 	^FF0000", "Cost: 	^FF0000") + callfunc("F_InsertComma", .@cost) + "^000000 z";
	next;

	if (select(needtr("구매한다:취소","Buy:Cancel")) == 2)
		close;

	if (Zeny < .@cost) {
		mes needtr("[주간 은총 이용권]","[Weekly Grace Pass]");
		mes needtr("제니가 부족합니다.","You do not have enough Zeny.");
		close;
	}

	Zeny -= .@cost;

	if (.@old_until > .@now)
		set .@new_until, .@old_until + .@seconds;
	else
		set .@new_until, .@now + .@seconds;

	switch(.@type) {
	case 1:
		NeedGraceTradeUntil = .@new_until;
		break;
	case 2:
		NeedGraceElementUntil = .@new_until;
		break;
	case 3:
		NeedGraceStorageUntil = .@new_until;
		break;
	case 4:
		NeedGracePackageUntil = .@new_until;
		break;
	}

	callfunc "NeedGrace_RefreshRentalTicket", .@item, .@new_until;

	mes needtr("[주간 은총 이용권]","[Weekly Grace Pass]");
	mes .@label$ + needtr("이 지급되었습니다.", " has been granted.");
	mes needtr("기존 이용 기간이 남아있다면 남은 기간에 7일이 추가됩니다.","If the pass is still running, 7 days are added to the time it has left.");
	callfunc "NeedGrace_ShowRemainLine", needtr("남은 시간","Time remaining"), .@new_until;
	close;
}

function	script	NeedGrace_RefreshRentalTicket	{
	set .@item, getarg(0);
	set .@until, getarg(1);
	set .@now, gettimetick(2);
	set .@left, .@until - .@now;

	// 기존 동일 티켓을 제거하고, 새 남은 기간으로 임대 티켓을 다시 지급한다.
	// 이렇게 해야 우클릭 시 표시되는 만료 시간이 현재 변수와 맞는다.
	while (rentalcountitem(.@item) > 0)
		delitem .@item, 1;

	if (.@left > 0)
		rentitem .@item, .@left;

	return;
}

// ============================================================
// 외부(성장지원 등)에서 종합 은총을 지급할 때 쓰는 공용 함수 (15차)
//   getarg(0) = 지급 기간(초). 생략하면 주간 이용권과 같은 7일.
//   기존 종합 은총 기간이 남아 있으면 날리지 않고 뒤에 이어 붙인다.
//   변수(NeedGracePackageUntil)와 임대 티켓(399904)의 만료 시각을
//   NeedGrace_RefreshRentalTicket 으로 항상 동기화한다.
//   반환값 = 새 만료 시각. 실패하면 기존 값을 그대로 돌려준다.
// ============================================================
function	script	NeedGrace_GrantPackage	{
	set .@seconds, getarg(0, $@NeedGraceWeeklySeconds);
	set .@item, $@NeedGracePackageTicket;
	set .@now, gettimetick(2);

	if (.@seconds <= 0 || .@item <= 0)
		return NeedGracePackageUntil;

	// 기존 은총 구매와 같은 연장 규칙을 사용한다.
	if (NeedGracePackageUntil > .@now)
		set .@new_until, NeedGracePackageUntil + .@seconds;
	else
		set .@new_until, .@now + .@seconds;

	NeedGracePackageUntil = .@new_until;
	callfunc "NeedGrace_RefreshRentalTicket", .@item, .@new_until;

	return .@new_until;
}

function	script	NeedGrace_GetEffectiveUntil	{
	set .@type, getarg(0);

	switch(.@type) {
	case 1:
		set .@until, NeedGraceTradeUntil;
		break;
	case 2:
		set .@until, NeedGraceElementUntil;
		break;
	case 3:
		set .@until, NeedGraceStorageUntil;
		break;
	default:
		set .@until, 0;
		break;
	}

	if (NeedGracePackageUntil > .@until)
		set .@until, NeedGracePackageUntil;

	return .@until;
}

function	script	NeedGrace_HasTrade	{
	set .@now, gettimetick(2);

	if (NeedGraceTradeUntil > .@now)
		return 1;

	if (NeedGracePackageUntil > .@now)
		return 1;

	return 0;
}

function	script	NeedGrace_HasElement	{
	set .@now, gettimetick(2);

	if (NeedGraceElementUntil > .@now)
		return 1;

	if (NeedGracePackageUntil > .@now)
		return 1;

	return 0;
}

function	script	NeedGrace_HasStorage	{
	set .@now, gettimetick(2);

	if (NeedGraceStorageUntil > .@now)
		return 1;

	if (NeedGracePackageUntil > .@now)
		return 1;

	return 0;
}

function	script	NeedGrace_ShowStatusRemainLine	{
	set .@label$, getarg(0);
	set .@remain_ms, getarg(1);

	if (.@remain_ms <= 0) {
		mes .@label$ + needtr(": 이용 중 아님",": Not active");
		return;
	}

	set .@left, (.@remain_ms + 999) / 1000;
	set .@days, .@left / 86400;
	set .@hours, (.@left % 86400) / 3600;
	set .@minutes, (.@left % 3600) / 60;

	if (.@days > 0)
		mes .@label$ + ": " + .@days + needtr("일 ","d ") + .@hours + needtr("시간 ","h ") + .@minutes + needtr("분","m");
	else if (.@hours > 0)
		mes .@label$ + ": " + .@hours + needtr("시간 ","h ") + .@minutes + needtr("분","m");
	else if (.@minutes > 0)
		mes .@label$ + ": " + .@minutes + needtr("분","m");
	else
		mes .@label$ + needtr(": 1분 미만",": Under 1m");

	return;
}

function	script	NeedGrace_ShowRemainLine	{
	set .@label$, getarg(0);
	set .@until, getarg(1);
	set .@now, gettimetick(2);

	if (.@until <= .@now) {
		mes .@label$ + needtr(": 이용 중 아님",": Not active");
		return;
	}

	set .@left, .@until - .@now;
	set .@days, .@left / 86400;
	set .@hours, (.@left % 86400) / 3600;
	set .@minutes, (.@left % 3600) / 60;

	mes .@label$ + ": " + .@days + needtr("일 ","d ") + .@hours + needtr("시간 ","h ") + .@minutes + needtr("분","m");
	return;
}
