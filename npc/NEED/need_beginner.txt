//============================================================
// 신규 성장지원 대상 계정 여부
//============================================================
function	script	F_NEED_IsBeginnerAccount	{
	// GM / 운영진 계정은 테스트를 위해 account_id 컷만 우회한다.
	// 보상 / 레벨 조건 / 1회 지급 변수는 일반 신규 유저와 완전히 동일하다.
	if (getarraysize($NEED_BEGINNER_GM_GROUP) > 0 &&
		inarray($NEED_BEGINNER_GM_GROUP[0], getgroupid()) >= 0)
		return 1;

	return (getcharid(3) >= $NEED_BEGINNER_ACCOUNT_MIN);
}

//============================================================
// 초반 성장 단계 판정 (공용)
//   성장 지원관 / 초보존 안내원이 같은 목표를 말하도록
//   초반 단계 판정을 이 함수 하나로 통일한다.
//
//   1 : Job Lv.10 달성      (노비스 계열 / Job Lv.10 미만)
//   2 : 1차 전직 진행       (노비스 계열 / Job Lv.10 이상)
//   3 : Base Lv.40 달성     (1차 전직 완료 / Base Lv.40 미만)
//   4 : 초보자 지역 졸업     (1차 전직 완료 / Base Lv.40 이상)
//
//   전직 완료 여부는 별도 변수 없이 현재 Class 로 판정한다.
//   노비스 계열은 Base 레벨과 무관하게 전직 안내를 우선한다.
//============================================================
function	script	F_NEED_GetBeginnerStage	{
	// 아직 1차 전직을 하지 않았다면 레벨보다 전직 안내가 우선한다.
	if (Class == Job_Novice || Class == Job_Novice_High || Class == Job_Baby)
		return ((JobLevel < 10) ? 1 : 2);

	if (BaseLevel >= 40)
		return 4;

	return 3;
}

//============================================================
// 성장 학습(메모리얼 입장) 기록
//   호출 위치 : 각 메모리얼 던전의 실제 입장 성공(IE_OK) 직후
//   기존 계정에는 적용되지 않으며, 보상/입장 조건과 무관한
//   선택 학습 지표로만 사용한다.
//============================================================
function	script	F_NEED_GuideEnter	{
	.@key$ = getarg(0);

	// 기존 계정은 성장 학습 대상이 아님
	if (!callfunc("F_NEED_IsBeginnerAccount"))
		return 0;

	setarray .@keys$[0],
		"SARA", "OGH", "INFINITE", "CHARLESTON", "TOY", "OGH_HARD";
	setarray .@vars$[0],
		"NEED_GUIDE_SARA", "NEED_GUIDE_OGH", "NEED_GUIDE_INFINITE",
		"NEED_GUIDE_CHARLESTON", "NEED_GUIDE_TOY", "NEED_GUIDE_OGH_HARD";
	setarray .@names$[0],
		"사라의 기억", "옛 글래스트헤임", "무한의 공간",
		"위기의 찰스턴", "호러 장난감 공장", "옛 글래스트헤임 상급";
	setarray .@minlv[0],
		100, 150, 150, 150, 150, 150;

	.@idx = inarray(.@keys$, .@key$);
	if (.@idx < 0)
		return 0;

	if (BaseLevel < .@minlv[.@idx])
		return 0;

	// 이미 완료한 학습은 재처리하지 않음
	if (getd(.@vars$[.@idx]))
		return 0;

	setd .@vars$[.@idx], 1;
	dispbottom "[NEED 성장지원] " + .@names$[.@idx] + " 입장 학습을 완료했습니다.";
	return 1;
}


function	script	F_NEED_BeginnerSupply	{
	.@show = getarg(0, 0);

	// 기존 계정은 신규 성장지원 대상이 아님
	if (!callfunc("F_NEED_IsBeginnerAccount"))
		return 0;

	if (!#NEED_BEGINNER || #NEED_BEGINNER_START <= 0)
		return 0;

	.@now = gettimetick(2);
	.@elapsed = .@now - #NEED_BEGINNER_START;

	if (.@elapsed < 0 || .@elapsed >= 604800) {
		if (.@show) {
			setdialogsize(440,260);
			mes "^00AAFF[7일 초보자 지원]^000000";
			mes "";
			mes "^777777초보자 소모품 지원 기간이 종료되었습니다.^000000";
			mes "";
			mes "이후 필요한 소모품은";
			mes "^FF9900금화 / 캐시샵^000000을 통해 구매할 수 있습니다.";
		}
		return 0;
	}

	.@day = (.@elapsed / 86400) + 1;

	if (#NEED_BEGINNER_SUPPLY_DAY >= .@day) {
		if (.@show) {
			setdialogsize(440,260);
			mes "^00AAFF[7일 초보자 지원]^000000";
			mes "";
			mes "오늘의 소모품 보충은 ^00AA00완료^000000되었습니다.";
			mes "";
			mes "^777777지원 " + .@day + "일차 / 총 7일^000000";
		}
		return 0;
	}

	#NEED_BEGINNER_SUPPLY_DAY = .@day;

	.@small = 100 - countitem(14534);
	.@medium = 100 - countitem(14535);
	.@mysterious = 100 - countitem(12832);
	.@almighty = 10 - countitem(12883);

	if (.@small > 0)
		getitembound 14534, .@small, BOUND_ACCOUNT;
	if (.@medium > 0)
		getitembound 14535, .@medium, BOUND_ACCOUNT;
	if (.@mysterious > 0)
		getitembound 12832, .@mysterious, BOUND_ACCOUNT;
	if (.@almighty > 0)
		getitembound 12883, .@almighty, BOUND_ACCOUNT;

	if (.@show) {
		setdialogsize(440,260);
		mes "^00AAFF[7일 초보자 지원]^000000";
		mes "";
		mes "^00AA00오늘의 소모품 보충이 완료되었습니다.^000000";
		mes "";
		mes "소형 생명수    ^FF9900최대 100개^000000";
		mes "중형 생명수    ^FF9900최대 100개^000000";
		mes "신묘한 생명수  ^FF9900최대 100개^000000";
		mes "올마이티        ^FF9900최대 10개^000000";
		mes "";
		mes "^777777지원 " + .@day + "일차 / 총 7일^000000";
	}
	return 1;
}

S_GoalLine:
	.@pre$ = getarg(0);

		if (.@stage == 1) {
			mes .@pre$ + " : ^FF9900Job Lv.10 달성^000000";
			return 1;
		}
		if (.@stage == 2) {
			mes .@pre$ + " : ^FF99001차 전직^000000";
			return 2;
		}
		mes .@pre$ + " : ^FF9900Base Lv.40 달성^000000";
		return 3;
	}

	// Base Lv.40 ~ 99
	if (BaseLevel < 100) {
		mes .@pre$ + " : ^FF9900Base Lv.100 달성^000000";
		return 4;
	}

	// Base Lv.100 ~ 119 : Lv.100 보상 -> 사라의 기억 -> Base Lv.120
	if (BaseLevel < 120) {
		if (!NEED_BEGINNER_REWARD_100) {
			mes .@pre$ + " : ^FF9900Lv.100 성장 지원 보상 받기^000000";
			return 5;

		if (!NEED_GUIDE_SARA) {
			mes .@pre$ + " : ^FF9900사라의 기억 입장^000000";
			return 6;
		}
		mes .@pre$ + " : ^FF9900Base Lv.120 달성^000000";
		return 7;
	}

	// Base Lv.120 ~ 139 : 카드 학습 -> Base Lv.140
	if (BaseLevel < 140) {
		if (!NEED_GUIDE_CARD) {
			mes .@pre$ + " : ^FF9900카드 시스템 학습^000000";
			return 8;
		}
		mes .@pre$ + " : ^FF9900Base Lv.140 달성^000000";
		return 9;
	}

	// Base Lv.140 ~ 148 : 제련 학습 -> Base Lv.149
	if (BaseLevel < 149) {
		if (!NEED_GUIDE_REFINE) {
			mes .@pre$ + " : ^FF9900제련 시스템 학습^000000";
			return 10;

		mes .@pre$ + " : ^FF9900Base Lv.149 달성^000000";
		return 11;

	if (BaseLevel < 150) {

		return 12;
	}

	// Base Lv.150 ~ 174 : Lv.150 보상 -> 미완료 학습 -> Base Lv.175 / Job Lv.60
	if (BaseLevel < 175) {

		}

		// 미완료 학습이 있으면 그중 하나만 대표 목표로 보여준다.
		// 전부 필수처럼 보이지 않도록 한 번에 한 항목만 노출한다.

		else if (!NEED_GUIDE_OGH_HARD)
			.@rep$ = "옛 글래스트헤임 상급 입장";

		if (.@rep$ != "") {
			mes .@pre$ + " : ^FF9900" + .@rep$ + "^000000";
			return 14;
		}

		mes .@pre$ + " : ^FF9900Base Lv.175 / Job Lv.60 달성^000000";
		return 15;
	}

	// Base Lv.175 이상
	if (JobLevel < 60) {
		mes .@pre$ + " : ^FF9900Job Lv.60 달성^000000";
		return 16;
	}

		return 17;
