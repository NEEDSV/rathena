OnInit:
	$NEED_BEGINNER_ACCOUNT_MIN = 2000817;

	// GM / 운영진 테스트 계정 예외 그룹 (conf/groups.yml 기준)
	//    2  Support          (Level 1)
	//    3  Script Manager   (Level 1)
	//    4  Event Manager    (Level 1)
	//   10  Law Enforcement  (Level 2)
	//   99  Admin            (Level 99)
	// 0 Player / 1 Super Player / 5 VIP 는 Level 0 유저 그룹이므로 제외한다.

//============================================================
// 신규 성장지원 대상 계정 여부
//============================================================
function	script	F_NEED_IsBeginnerAccount	{
	// GM / 운영진 계정은 테스트를 위해 account_id 컷만 우회한다.
	// 보상 / 레벨 조건 / 1회 지급 변수는 일반 신규 유저와 완전히 동일하다.
	if (getarraysize($NEED_BEGINNER_GM_GROUP) > 0 &&
		inarray($NEED_BEGINNER_GM_GROUP[0], getgroupid()) >= 0)
		return 1;

	return (getcharid(3) >= $NEED_BEGINNER_ACCOUNT_MIN);
}

//============================================================
// 초반 성장 단계 판정 (공용)
//   성장 지원관 / 초보존 안내원이 같은 목표를 말하도록
//   초반 단계 판정을 이 함수 하나로 통일한다.
//
//   1 : Job Lv.10 달성      (노비스 계열 / Job Lv.10 미만)
//   2 : 1차 전직 진행       (노비스 계열 / Job Lv.10 이상)
//   3 : Base Lv.40 달성     (1차 전직 완료 / Base Lv.40 미만)
//   4 : 초보자 지역 졸업     (1차 전직 완료 / Base Lv.40 이상)
//
//   전직 완료 여부는 별도 변수 없이 현재 Class 로 판정한다.
//   노비스 계열은 Base 레벨과 무관하게 전직 안내를 우선한다.
//============================================================
function	script	F_NEED_GetBeginnerStage	{
	// 아직 1차 전직을 하지 않았다면 레벨보다 전직 안내가 우선한다.
	if (Class == Job_Novice || Class == Job_Novice_High || Class == Job_Baby)
		return ((JobLevel < 10) ? 1 : 2);

	if (BaseLevel >= 40)
		return 4;

	return 3;
}

//============================================================
// 성장 학습(메모리얼 입장) 기록
//   호출 위치 : 각 메모리얼 던전의 실제 입장 성공(IE_OK) 직후
//   기존 계정에는 적용되지 않으며, 보상/입장 조건과 무관한
//   선택 학습 지표로만 사용한다.
//============================================================
function	script	F_NEED_GuideEnter	{
	.@key$ = getarg(0);

	// 기존 계정은 성장 학습 대상이 아님
	if (!callfunc("F_NEED_IsBeginnerAccount"))
		return 0;

	setarray .@keys$[0],
		"SARA", "OGH", "INFINITE", "CHARLESTON", "TOY", "OGH_HARD";
	setarray .@vars$[0],
		"NEED_GUIDE_SARA", "NEED_GUIDE_OGH", "NEED_GUIDE_INFINITE",
		"NEED_GUIDE_CHARLESTON", "NEED_GUIDE_TOY", "NEED_GUIDE_OGH_HARD";
	setarray .@names$[0],
		needtr("사라의 기억","Sara's Memory"), needtr("옛 글래스트헤임","Old Glast Heim"), needtr("무한의 공간","Infinite Space"),
		needtr("위기의 찰스턴","Charleston Crisis"), needtr("호러 장난감 공장","Horror Toy Factory"), needtr("옛 글래스트헤임 상급","Old Glast Heim Hard");
	setarray .@minlv[0],
		100, 150, 150, 150, 150, 150;

	.@idx = inarray(.@keys$, .@key$);
	if (.@idx < 0)
		return 0;

	if (BaseLevel < .@minlv[.@idx])
		return 0;

	// 이미 완료한 학습은 재처리하지 않음
	if (getd(.@vars$[.@idx]))
		return 0;

	setd .@vars$[.@idx], 1;
	dispbottom needtr("[NEED 성장지원] ","[NEED Growth Support] ") + .@names$[.@idx] + needtr(" 입장 학습을 완료했습니다."," entry study complete.");
	return 1;
}

		if (.@show) {
			setdialogsize(440,260);

	if (.@medium > 0)
		getitembound 14535, .@medium, BOUND_ACCOUNT;
	if (.@mysterious > 0)
		getitembound 12832, .@mysterious, BOUND_ACCOUNT;

	if (.@show) {
		setdialogsize(440,260);
		mes needtr("^00AAFF[7일 초보자 지원]^000000","^00AAFF[7-Day Beginner Support]^000000");

		mes "";
		mes needtr("소형 생명수    ^FF9900최대 100개^000000","Small Life Potion      ^FF9900up to 100^000000");
		mes needtr("중형 생명수    ^FF9900최대 100개^000000","Medium Life Potion     ^FF9900up to 100^000000");
		mes needtr("신묘한 생명수  ^FF9900최대 100개^000000","Mysterious Life Potion ^FF9900up to 100^000000");

// Base Lv.100 이상으로 올라갈 수 있는 최종 직업인가
//   3차 직업 + 리벨리온 / 카게로우 / 오보로 (서버 분류상 2차) + 도람(99 이상)
function	script	F_NEED_CourseIsFinalClass	{
	if (BaseLevel >= 100)
		return 1;
	if (eaclass() & EAJL_THIRD)
		return 1;
	switch (Class) {
	case JOB_REBELLION: case JOB_KAGEROU: case JOB_OBORO:
		return 1;
	case JOB_SUMMONER: case JOB_BABY_SUMMONER:
		return (BaseLevel >= 99);
	}
	return 0;
}

// 직업 안내용 키 : 구분 * 100 + 계열 (-1 = 안내 없음 : 확장 직업 / 슈퍼노비스 / 4차)
// 구분 0 노비스 / 1 1차 / 2 2차 / 3 전승 2차 / 4 3차
// 1차 계열 : 1 검사 2 상인 3 도둑 4 궁수 5 마법사 6 복사
// 2차 이후 계열 : 1 나이트 2 크루세이더 3 블랙스미스 4 알케미스트 5 어쌔신 6 로그
//                 7 헌터 8 바드 9 댄서 10 위자드 11 세이지 12 프리스트 13 몽크
function	script	F_NEED_CourseJobKey	{
	.@ea = eaclass();
	.@base = .@ea & EAJ_BASEMASK;
	if (.@ea & EAJL_FOURTH)
		return -1;
	if (.@base == EAJ_NOVICE)
		return (.@ea & EAJL_2) ? -1 : 0;
	switch (.@base) {
	case EAJ_SWORDMAN: .@k = 1; break;
	case EAJ_MERCHANT: .@k = 2; break;
	case EAJ_THIEF:    .@k = 3; break;
	case EAJ_ARCHER:   .@k = 4; break;
	case EAJ_MAGE:     .@k = 5; break;
	case EAJ_ACOLYTE:  .@k = 6; break;
	default:
		return -1;
	}
	if (!(.@ea & EAJL_2))
		return 100 + .@k;

	// 계열마다 2-1 / 2-2 순서. 궁수 2-2 는 성별로 바드 / 댄서를 나눈다.
	setarray .@first[1], 1, 3, 5, 7, 10, 12;
	.@k2 = .@first[.@k] + ((.@ea & EAJL_2_2) ? 1 : 0);
	if (.@k == 4 && (.@ea & EAJL_2_2) && Sex == SEX_FEMALE)
		.@k2 = 9;
	if (.@ea & EAJL_THIRD)
		return 400 + .@k2;
	if (.@ea & EAJL_UPPER)
		return 300 + .@k2;
	return 200 + .@k2;
}

// 현재 코스 단계
//   1~9 = 단계 / 60 = 장비 착용 관문 / 70 = 무기 +7 관문 / 80 = 쉐도우 +4 관문
//   직업과 레벨만으로 판정하므로 전승 후에는 자연스럽게 1단계부터 다시 시작한다.
function	script	F_NEED_CoursePhase	{
	.@ea = eaclass();
	.@base = .@ea & EAJ_BASEMASK;

	if (callfunc("F_NEED_CourseIsFinalClass")) {
		if (BaseLevel < 100)
			return 5;
		if (!NEED_COURSE_GEAR)
			return 60;
		if (BaseLevel < 125)
			return 6;
		if (!NEED_COURSE_W7)
			return 70;
		if (BaseLevel < 140)
			return 7;
		if (!NEED_COURSE_S4)
			return 80;
		if (BaseLevel < $NEED_COURSE_GRAD_LV || !NEED_COURSE_SARA)
			return 8;
		return 9;
	}

	// 노비스 (슈퍼노비스는 EAJL_2 가 있어 여기 걸리지 않는다)
	if (.@base == EAJ_NOVICE && !(.@ea & EAJL_2))
		return 1;

	// 1차 직업
	if (!(.@ea & EAJL_2)) {
		if (BaseLevel < 35)
			return 2;
		// 2차 직업이 없는 1차 (건슬링거 / 닌자 / 도람) 는 레벨만 본다
		if (.@base == EAJ_GUNSLINGER || .@base == EAJ_NINJA || .@base == EAJ_SUMMONER) {
			if (BaseLevel < 65)
				return 3;
			if (BaseLevel < 99)
				return 4;
			return 5;
		}
		return 3;
	}

	// 2차 직업
	if (BaseLevel < 65)
		return 3;
	if (BaseLevel < 99)
		return 4;
	return 5;
}

// 관문 확인 : 60 = 초심자 장비 착용 / 70 = 무기 +7 / 80 = 쉐도우 6부위 +4
//   맞는 초심자 무기가 없는 직업(성제 / 소울리퍼 등)은 무기 조건을 건너뛴다.
function	script	F_NEED_CourseGateOK	{
	.@gate = getarg(0);
	.@cw = callfunc("F_NEED_CourseClassWeapon");
	.@wok = 1;
	if (.@cw > 0)
		.@wok = callfunc("F_NEED_CourseIsBeginnerWeapon", getequipid(EQI_HAND_R));

	if (.@gate == 60) {
		if (!.@wok)
			return 0;
		if (!isequipped(19310,15250,20906,22173,28566))
			return 0;
		return isequipped(24387,24388,24389,24390,24391,24392);
	}
	if (.@gate == 70) {
		if (.@cw <= 0)
			return 1;
		if (!.@wok)
			return 0;
		return (getequiprefinerycnt(EQI_HAND_R) >= 7);
	}
	if (.@gate == 80) {
		setarray .@slot[0], EQI_SHADOW_ARMOR, EQI_SHADOW_WEAPON, EQI_SHADOW_SHIELD,
			EQI_SHADOW_SHOES, EQI_SHADOW_ACC_R, EQI_SHADOW_ACC_L;
		for (.@i = 0; .@i < 6; .@i++) {
			.@id = getequipid(.@slot[.@i]);
			if (.@id < 24387 || .@id > 24392)
				return 0;
			if (getequiprefinerycnt(.@slot[.@i]) < 4)
				return 0;
		}
		return 1;
	}
	return 0;
}

// Lv.150 올마이티 지원 : 7일 동안 하루 1회, 10개까지 보충 (캐릭터 단위, 계정 귀속 지급)
function	script	F_NEED_AlmightySupply	{
	if (NEED_COURSE != 1 || BaseLevel < 150)
		return 0;

	.@now = gettimetick(2);
	if (NEED_ALMIGHTY_START <= 0) {
		NEED_ALMIGHTY_START = .@now;
		NEED_ALMIGHTY_DAY = 0;
		dispbottom needtr("[낙원단] Lv.150부터 버프 도우미의 버프가 줄어듭니다. 대신 올마이티를 드세요! (7일 동안 매일 10개까지 보충)","[Eden] From Lv.150 the Buff Helper gives fewer buffs. Eat Almighty instead! (topped up to 10 daily for 7 days)");
	}

	.@elapsed = .@now - NEED_ALMIGHTY_START;
	if (.@elapsed < 0 || .@elapsed >= 604800)
		return 0;

	.@day = (.@elapsed / 86400) + 1;
	if (NEED_ALMIGHTY_DAY >= .@day)
		return 0;

	.@need = 10 - countitem(12883);
	if (.@need > 0 && !checkweight(12883,.@need))
		return 0;

	NEED_ALMIGHTY_DAY = .@day;
	if (.@need > 0)
		getitembound 12883, .@need, BOUND_ACCOUNT;
	dispbottom needtr("[낙원단] 올마이티를 10개까지 보충했습니다. (","[Eden] Almighty topped up to 10. (Day ") + .@day + needtr("일차 / 7일)"," of 7)");
	return 1;
}

L_CourseJob:
	// 직업 안내 (CM 가이드라인 기준, 확장 직업은 안내하지 않는다)
	.@jk = callfunc("F_NEED_CourseJobKey");
	if (.@jk < 0)
		goto L_Course;
	.@t = .@jk / 100;
	.@k = .@jk % 100;
	if (.@t == 0)
		goto L_CourseJobPick;
	mes needtr("^00AAFF[직업 안내]^000000","^00AAFF[Job Guide]^000000");
	mes "";
	switch (.@t) {
	case 1:
		callsub S_CourseJob1, .@k;
		break;
	case 2:
		callsub S_CourseJob2, .@k;
		break;
	case 3:
		callsub S_CourseJob2, .@k;
		next;
		mes needtr("^00AAFF[직업 안내]^000000 전승 스킬","^00AAFF[Job Guide]^000000 Transcendent skills");
		mes "";
		callsub S_CourseJobTrans, .@k;
		break;
	case 4:
		callsub S_CourseJob3, .@k;
		break;
	}
	next;
	goto L_Course;

	if (!callfunc("F_NEED_IsBeginnerAccount"))
		end;

	if (!#NEED_BEGINNER || #NEED_BEGINNER_START <= 0)
		end;

	// 15차: Lv.100 종합 은총 지급 버그 1회 보정
	//   수정 전에는 종합 은총을 일반 getitem 399904 로 지급해
	//   NeedGracePackageUntil 이 설정되지 않고 임대 티켓도 없었다.
	//   정상 경로는 전부 rentitem 이므로 일반(비임대) 399904 보유가
	//   그 버그의 유일한 증거다. countitem 은 비임대만,
	//   rentalcountitem 은 임대만 세므로 둘은 확실히 구분된다.
	if (NEED_BEGINNER_REWARD_100 && !NEED_BEGINNER_GRACE_FIX) {
		.@now = gettimetick(2);

		if (countitem(399904) > 0 && NeedGracePackageUntil <= .@now) {

			// 임대 티켓이 하나도 없을 때만 잘못 지급된 일반 티켓을 회수한다.
			if (rentalcountitem(399904) <= 0)
