//============================================================
// 신규 성장지원 대상 계정 여부
//============================================================
function	script	F_NEED_IsBeginnerAccount	{
	// GM / 운영진 계정은 테스트를 위해 account_id 컷만 우회한다.
	// 보상 / 레벨 조건 / 1회 지급 변수는 일반 신규 유저와 완전히 동일하다.
	if (getarraysize($NEED_BEGINNER_GM_GROUP) > 0 &&
		inarray($NEED_BEGINNER_GM_GROUP[0], getgroupid()) >= 0)
		return 1;

	return (getcharid(3) >= $NEED_BEGINNER_ACCOUNT_MIN);
}

//============================================================
// 초반 성장 단계 판정 (공용)
//   성장 지원관 / 초보존 안내원이 같은 목표를 말하도록
//   초반 단계 판정을 이 함수 하나로 통일한다.
//
//   1 : Job Lv.10 달성      (노비스 계열 / Job Lv.10 미만)
//   2 : 1차 전직 진행       (노비스 계열 / Job Lv.10 이상)
//   3 : Base Lv.40 달성     (1차 전직 완료 / Base Lv.40 미만)
//   4 : 초보자 지역 졸업     (1차 전직 완료 / Base Lv.40 이상)
//
//   전직 완료 여부는 별도 변수 없이 현재 Class 로 판정한다.
//   노비스 계열은 Base 레벨과 무관하게 전직 안내를 우선한다.
//============================================================
function	script	F_NEED_GetBeginnerStage	{
	// 아직 1차 전직을 하지 않았다면 레벨보다 전직 안내가 우선한다.
	if (Class == Job_Novice || Class == Job_Novice_High || Class == Job_Baby)
		return ((JobLevel < 10) ? 1 : 2);

	if (BaseLevel >= 40)
		return 4;

	return 3;
}

//============================================================
// 성장 학습(메모리얼 입장) 기록
//   호출 위치 : 각 메모리얼 던전의 실제 입장 성공(IE_OK) 직후
//   기존 계정에는 적용되지 않으며, 보상/입장 조건과 무관한
//   선택 학습 지표로만 사용한다.
//============================================================
function	script	F_NEED_GuideEnter	{
	.@key$ = getarg(0);

	// 기존 계정은 성장 학습 대상이 아님
	if (!callfunc("F_NEED_IsBeginnerAccount"))
		return 0;

	setarray .@keys$[0],
		"SARA", "OGH", "INFINITE", "CHARLESTON", "TOY", "OGH_HARD";
	setarray .@vars$[0],
		"NEED_GUIDE_SARA", "NEED_GUIDE_OGH", "NEED_GUIDE_INFINITE",
		"NEED_GUIDE_CHARLESTON", "NEED_GUIDE_TOY", "NEED_GUIDE_OGH_HARD";
	setarray .@names$[0],
		needtr("사라의 기억","Sara's Memory"), needtr("옛 글래스트헤임","Old Glast Heim"), needtr("무한의 공간","Infinite Space"),
		needtr("위기의 찰스턴","Charleston Crisis"), needtr("호러 장난감 공장","Horror Toy Factory"), needtr("옛 글래스트헤임 상급","Old Glast Heim Hard");
	setarray .@minlv[0],
		100, 150, 150, 150, 150, 150;

	.@idx = inarray(.@keys$, .@key$);
	if (.@idx < 0)
		return 0;

	if (BaseLevel < .@minlv[.@idx])
		return 0;

	// 이미 완료한 학습은 재처리하지 않음
	if (getd(.@vars$[.@idx]))
		return 0;

	setd .@vars$[.@idx], 1;
	dispbottom needtr("[NEED 성장지원] ","[NEED Growth Support] ") + .@names$[.@idx] + needtr(" 입장 학습을 완료했습니다."," entry study complete.");
	return 1;
}


function	script	F_NEED_BeginnerSupply	{

		return 0;

	.@elapsed = .@now - #NEED_BEGINNER_START;

		if (.@show) {
			setdialogsize(440,260);

			mes needtr("^FF9900금화 / 캐시샵^000000을 통해 구매할 수 있습니다.","from the ^FF9900Gold Coin / Cash Shop^000000.");
		}
		return 0;
	}

	.@day = (.@elapsed / 86400) + 1;

	if (#NEED_BEGINNER_SUPPLY_DAY >= .@day) {
		if (.@show) {
			setdialogsize(440,260);
			mes needtr("^00AAFF[7일 초보자 지원]^000000","^00AAFF[7-Day Beginner Support]^000000");

			mes "";
			mes needtr("^777777지원 ","^777777Support day ") + .@day + needtr("일차 / 총 7일^000000"," of 7 total^000000");
		}
		return 0;

	.@medium = 100 - countitem(14535);
	.@mysterious = 100 - countitem(12832);
	.@almighty = 10 - countitem(12883);

		getitembound 14534, .@small, BOUND_ACCOUNT;
	if (.@medium > 0)
		getitembound 14535, .@medium, BOUND_ACCOUNT;
	if (.@mysterious > 0)
		getitembound 12832, .@mysterious, BOUND_ACCOUNT;
	if (.@almighty > 0)
		getitembound 12883, .@almighty, BOUND_ACCOUNT;

	if (.@show) {
		setdialogsize(440,260);
		mes needtr("^00AAFF[7일 초보자 지원]^000000","^00AAFF[7-Day Beginner Support]^000000");
		mes "";
		mes needtr("^00AA00오늘의 소모품 보충이 완료되었습니다.^000000","^00AA00Today's supply refill is complete.^000000");
		mes "";
		mes needtr("소형 생명수    ^FF9900최대 100개^000000","Small Life Potion      ^FF9900up to 100^000000");
		mes needtr("중형 생명수    ^FF9900최대 100개^000000","Medium Life Potion     ^FF9900up to 100^000000");
		mes needtr("신묘한 생명수  ^FF9900최대 100개^000000","Mysterious Life Potion ^FF9900up to 100^000000");
		mes needtr("올마이티        ^FF9900최대 10개^000000","Almighty               ^FF9900up to 10^000000");

		mes needtr("^777777지원 ","^777777Support day ") + .@day + needtr("일차 / 총 7일^000000"," of 7 total^000000");
	}
	return 1;

//   getarg(0) = 표시 접두 ("다음 목표" / "현재 목표")
//   반환값    = 목표 코드 (S_ShowGoal 의 상세 안내 분기용)
//
//   기존 레벨 / 보상 변수 / 학습 변수만 사용하며
//   목표 안내를 위한 영구 변수를 새로 만들지 않는다.
//   학습 완료는 어떤 보상 / 레벨업 / 졸업 조건도 막지 않는다.

S_GoalLine:
	.@pre$ = getarg(0);
	.@stage = callfunc("F_NEED_GetBeginnerStage");

	// stage 4 = 1차 전직 완료 + Base Lv.40 이상 (초보 구간 졸업)
	if (.@stage != 4) {
		if (.@stage == 1) {
			mes .@pre$ + needtr(" : ^FF9900Job Lv.10 달성^000000"," : ^FF9900Reach Job Lv.10^000000");
			return 1;
		}
		if (.@stage == 2) {
			mes .@pre$ + needtr(" : ^FF99001차 전직^000000"," : ^FF99001st Job Change^000000");
			return 2;
		}
		mes .@pre$ + needtr(" : ^FF9900Base Lv.40 달성^000000"," : ^FF9900Reach Base Lv.40^000000");
		return 3;
	}

	// Base Lv.40 ~ 99
	if (BaseLevel < 100) {
		mes .@pre$ + needtr(" : ^FF9900Base Lv.100 달성^000000"," : ^FF9900Reach Base Lv.100^000000");
		return 4;
	}

	// Base Lv.100 ~ 119 : Lv.100 보상 -> 사라의 기억 -> Base Lv.120
	if (BaseLevel < 120) {
		if (!NEED_BEGINNER_REWARD_100) {
			mes .@pre$ + needtr(" : ^FF9900Lv.100 성장 지원 보상 받기^000000"," : ^FF9900Claim the Lv.100 Growth Support Rewards^000000");
			return 5;
		}
		if (!NEED_GUIDE_SARA) {
			mes .@pre$ + needtr(" : ^FF9900사라의 기억 입장^000000"," : ^FF9900Enter Sara's Memory^000000");
			return 6;
		}
		mes .@pre$ + needtr(" : ^FF9900Base Lv.120 달성^000000"," : ^FF9900Reach Base Lv.120^000000");
		return 7;
	}

	// Base Lv.120 ~ 139 : 카드 학습 -> Base Lv.140
	if (BaseLevel < 140) {
		if (!NEED_GUIDE_CARD) {
			mes .@pre$ + needtr(" : ^FF9900카드 시스템 학습^000000"," : ^FF9900Card System Study^000000");
			return 8;
		}
		mes .@pre$ + needtr(" : ^FF9900Base Lv.140 달성^000000"," : ^FF9900Reach Base Lv.140^000000");
		return 9;
	}

	// Base Lv.140 ~ 148 : 제련 학습 -> Base Lv.149
	if (BaseLevel < 149) {
		if (!NEED_GUIDE_REFINE) {
			mes .@pre$ + needtr(" : ^FF9900제련 시스템 학습^000000"," : ^FF9900Refine System Study^000000");
			return 10;
		}
		mes .@pre$ + needtr(" : ^FF9900Base Lv.149 달성^000000"," : ^FF9900Reach Base Lv.149^000000");
		return 11;
	}

	// Base Lv.149 : 무료 초기화 마지막 구간이지만 목표는 멈추지 않는다
	if (BaseLevel < 150) {
		mes .@pre$ + needtr(" : ^FF9900Base Lv.150 달성^000000"," : ^FF9900Reach Base Lv.150^000000");
		return 12;
	}

	// Base Lv.150 ~ 174 : Lv.150 보상 -> 미완료 학습 -> Base Lv.175 / Job Lv.60
	if (BaseLevel < 175) {
		if (!NEED_BEGINNER_REWARD_150) {
			mes .@pre$ + needtr(" : ^FF9900Lv.150 성장 지원 보상 받기^000000"," : ^FF9900Claim the Lv.150 Growth Support Rewards^000000");
			return 13;
		}

		// 미완료 학습이 있으면 그중 하나만 대표 목표로 보여준다.
		// 전부 필수처럼 보이지 않도록 한 번에 한 항목만 노출한다.
		if (!NEED_GUIDE_EXCELION)
			.@rep$ = needtr("엑셀리온 인챈트 학습","Excellion Enchant Study");
		else if (!NEED_GUIDE_CASH)
			.@rep$ = needtr("금화 / 캐시 시스템 학습","Gold Coin / Cash System Study");
		else if (!NEED_GUIDE_OGH)
			.@rep$ = needtr("옛 글래스트헤임 입장","Enter Old Glast Heim");
		else if (!NEED_GUIDE_INFINITE)
			.@rep$ = needtr("무한의 공간 입장","Enter Infinite Space");
		else if (!NEED_GUIDE_CHARLESTON)
			.@rep$ = needtr("위기의 찰스턴 입장","Enter Charleston Crisis");
		else if (!NEED_GUIDE_TOY)
			.@rep$ = needtr("호러 장난감 공장 입장","Enter Horror Toy Factory");

			.@rep$ = needtr("옛 글래스트헤임 상급 입장","Enter Old Glast Heim Hard");

			mes .@pre$ + " : ^FF9900" + .@rep$ + "^000000";

		mes .@pre$ + needtr(" : ^FF9900Base Lv.175 / Job Lv.60 달성^000000"," : ^FF9900Reach Base Lv.175 / Job Lv.60^000000");

		mes .@pre$ + needtr(" : ^FF9900성장지원 졸업 보상 받기^000000"," : ^FF9900Claim the Growth Support graduation reward^000000");

	return 18;

//   한쪽만 수정되어 어긋나는 일이 없다.
//   성장 조건 / 보상 조건 / 목표 판정(S_GoalLine)에 관여하지 않으며
//   완료 변수도 만들지 않는다. 이동은 강제하지 않는다.
//   Base Lv.175 이상은 175 이후 콘텐츠 안내가 따로 있으므로 표시하지 않는다.
//
//   목적지 좌표는 npc/custom/warper.txt 의 @c[] 값을 그대로 재사용한다. (14차 검증)
//     anthell01  (35,262)  D4  @c[2],@c[3]
//     orcsdun01  (32,170)  D36 @c[2],@c[3]
//     gld_dun02  (39,161)  G2  @c[2],@c[3]
//     mag_dun02  (47,30)   D30 @c[4],@c[5]
//     lasa_dun02 (22,171)  D28 @c[4],@c[5]
//     lasa_dun03 (190,18)  D28 @c[6],@c[7]
//     c_tower3_  (60,142)  D11 @c[20],@c[21]
//   초보존은 L_NewbieZone 이 사용하는 좌표를 그대로 쓴다. (100,133)
//============================================================
S_HuntingGuide:
	// Base Lv.175 이상은 일반 레벨업 사냥터 안내 대상이 아니다.
	if (BaseLevel >= 175)
		return;

	setarray .@hg_name$[0],
		needtr("초보존","Newbie Zone"),

		needtr("루이나 지하 던전 1층","Luina Guild Dungeon F1"),
		needtr("노그로드 2층","Nogg Road Dungeon F2"),
		needtr("용의 둥지 2층","Dragon Nest F2"),
		needtr("용의 둥지 3층","Dragon Nest F3"),
		needtr("뒤틀린 시간의 시계탑 3층","Twisted Clock Tower F3");
	setarray .@hg_min[0],   1,  20,  40,  60, 100, 110, 130, 160;
	setarray .@hg_max[0],  19,  39,  59,  99, 109, 129, 159, 174;
	setarray .@hg_map$[0],
		"pvp_n_2-5",
		"anthell01",
		"orcsdun01",
		"gld_dun02",
		"mag_dun02",
		"lasa_dun02",
		"lasa_dun03",
		"c_tower3_";
	setarray .@hg_x[0], 100,  35,  32,  39,  47,  22, 190,  60;
	setarray .@hg_y[0], 133, 262, 170, 161,  30, 171,  18, 142;
	.@hg_size = getarraysize(.@hg_name$);

	// 현재 레벨이 속한 추천 구간을 찾는다.
	.@cur = 0;
	while (.@cur < .@hg_size - 1 && BaseLevel > .@hg_max[.@cur])
		.@cur += 1;

	mes "";
	mes needtr("^0055FF[추천 레벨업 사냥터]^000000 ^777777(참고용)^000000","^0055FF[Recommended Leveling Spots]^000000 ^777777(for reference)^000000");
	mes needtr("^777777현재 추천^000000","^777777Current recommendation^000000");
	mes "^00AAFF" + .@hg_name$[.@cur] + "^000000";
	mes needtr("^777777권장 구간 : Base Lv.","^777777Recommended range : Base Lv.") + .@hg_min[.@cur] + " ~ " + .@hg_max[.@cur] + "^000000";

	if (select(needtr("^00AAFF추천 사냥터 목록 / 이동^000000:돌아가기","^00AAFFRecommended hunting grounds / Go^000000:Back")) == 2)
		return;

	// 메뉴 선택으로 새 화면에 진입하므로 이전 대사를 지운다.
	clear;
	mes needtr("^00AAFF[추천 사냥터 이동]^000000","^00AAFF[Go to a Recommended Hunting Ground]^000000");
	mes "";
	mes needtr("원하는 사냥터를 선택해주세요.","Choose the hunting ground you want.");
	mes needtr("^777777괄호 안의 레벨은 권장 구간입니다.^000000","^777777The levels in brackets are the recommended range.^000000");

	// 현재 추천 구간만 색상과 별표로 구분한다. 선택 제한은 두지 않는다.
	.@menu$ = "";
	for (.@i = 0; .@i < .@hg_size; .@i += 1) {
		.@line$ = .@hg_name$[.@i] + " (Lv." + .@hg_min[.@i] + "~" + .@hg_max[.@i] + ")";

			.@menu$ += "^FF9900★ " + .@line$ + "^000000:";

			.@menu$ += .@line$ + ":";

	.@menu$ += needtr("돌아가기","Back");

	if (.@sel < 0 || .@sel >= .@hg_size)
		return;

	close2;

	end;

OnInit:

-	script	NEED_Beginner_Auto	-1,{

OnPCLoginEvent:
	if (!callfunc("F_NEED_IsBeginnerAccount"))
		end;

	if (!#NEED_BEGINNER || #NEED_BEGINNER_START <= 0)
		end;

	// 15차: Lv.100 종합 은총 지급 버그 1회 보정
	//   수정 전에는 종합 은총을 일반 getitem 399904 로 지급해
	//   NeedGracePackageUntil 이 설정되지 않고 임대 티켓도 없었다.
	//   정상 경로는 전부 rentitem 이므로 일반(비임대) 399904 보유가
	//   그 버그의 유일한 증거다. countitem 은 비임대만,
	//   rentalcountitem 은 임대만 세므로 둘은 확실히 구분된다.
	if (NEED_BEGINNER_REWARD_100 && !NEED_BEGINNER_GRACE_FIX) {
		.@now = gettimetick(2);

		if (countitem(399904) > 0 && NeedGracePackageUntil <= .@now) {
			// delitem 은 임대/일반을 구분하지 않으므로
			// 임대 티켓이 하나도 없을 때만 잘못 지급된 일반 티켓을 회수한다.
			if (rentalcountitem(399904) <= 0)
				delitem 399904, 1;

			callfunc("NeedGrace_GrantPackage", 604800);
			NEED_BEGINNER_GRACE_FIX = 1;
			dispbottom needtr("[NEED 성장지원] Lv.100 종합 은총을 7일 기간제로 보정했습니다.","[NEED Growth Support] Your Lv.100 Full Grace has been corrected to a 7-day rental.");
		}
		else if (NeedGracePackageUntil > .@now && rentalcountitem(399904) <= 0) {
			// 기간은 남아 있는데 임대 티켓만 없는 경우 동기화만 복구한다.
			// (기간을 추가로 늘리지 않는다)
			callfunc "NeedGrace_RefreshRentalTicket", 399904, NeedGracePackageUntil;
			NEED_BEGINNER_GRACE_FIX = 1;
		}
	}

	.@result = callfunc("F_NEED_BeginnerSupply", 0);
	if (.@result)
		dispbottom needtr("[NEED 성장지원] 오늘의 초보자 소모품이 보충되었습니다.","[NEED Growth Support] Today's beginner supplies have been topped up.");
	end;

OnPCBaseLvUpEvent:
	if (!callfunc("F_NEED_IsBeginnerAccount"))
		end;

	if (!#NEED_BEGINNER)
		end;

	switch(BaseLevel) {
	case 40:
		dispbottom needtr("[NEED 성장지원] 초보존 졸업 레벨에 도달했습니다. 라자냐로 이동해보세요.","[NEED Growth Support] You have reached the Newbie Zone Graduation level. Go to Lasagna.");
		break;
	case 100:
		dispbottom needtr("[NEED 성장지원] Lv.100 성장 보상을 받을 수 있습니다. 낙원단 성장 지원관을 찾아가세요.","[NEED Growth Support] Your Lv.100 Growth Reward is ready. Visit the Eden Growth Supporter.");
		break;
	case 149:
		dispbottom needtr("[중요] 카프라 무료 스탯/스킬 초기화는 Base Lv.149까지 이용할 수 있습니다.","[Important] The Kafra free Stat/Skill Reset is available only up to Base Lv.149.");
		break;
	case 150:
		dispbottom needtr("[NEED 성장지원] 초심자 무기 +7 제련 이용권을 받을 수 있습니다.","[NEED Growth Support] Your Beginner's Safe to 7 Weapon Certificate is ready.");
		break;
	case 175:
		dispbottom needtr("[NEED 성장지원] Job Lv.60까지 달성하면 성장지원 졸업 보상을 받을 수 있습니다.","[NEED Growth Support] Reach Job Lv.60 to claim the Growth Support graduation reward.");
		break;
	}
	end;

OnPCJobLvUpEvent:
	if (!callfunc("F_NEED_IsBeginnerAccount"))
		end;

	if (!#NEED_BEGINNER)
		end;

	// 노비스 계열이 Job Lv.10 에 도달한 순간 1회만 안내한다.
	// 레벨업 이벤트는 수치가 오를 때만 호출되므로 별도 변수가 필요 없다.
	// 단계 판정은 F_NEED_GetBeginnerStage 로 통일한다. (2 = 1차 전직 진행)
	if (JobLevel == 10 && callfunc("F_NEED_GetBeginnerStage") == 2) {
		dispbottom needtr("[NEED 성장지원] Job Lv.10을 달성했습니다! 낙원단 전직 도우미를 통해 1차 전직을 진행해보세요.","[NEED Growth Support] You have reached Job Lv.10! Use the Eden Group Job Master to take your 1st Job Change.");
		end;
	}

	if (BaseLevel >= 175 && JobLevel >= 60)
		dispbottom needtr("[NEED 성장지원] 졸업 조건을 달성했습니다. 낙원단 성장 지원관을 찾아가세요.","[NEED Growth Support] You have met the graduation requirements. Visit the Eden Growth Supporter.");
	end;
}
