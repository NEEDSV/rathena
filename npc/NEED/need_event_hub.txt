//============================================================
//= NEED Event Hub (공통 이벤트 버튼 -> @needevent -> 이 NPC)
//============================================================
//= 클라이언트(NEEDClient + WARP NeedEventButton)는 [!] 버튼 왼쪽 Event 버튼을 누르면
//= 항상 채팅 명령 "@needevent" 하나만 보낸다. 어떤 이벤트 화면을 열지는 서버가 정한다.
//= 다음 이벤트는 서버 스크립트(모듈)와 버튼 BMP 만 바꾸면 되고 exe 는 다시 패치하지 않는다.
//=
//= 버튼 표시 / 강조도 서버가 정한다.
//=   $NEED_EVHUB_ON     1 = 버튼 표시, 0 = 버튼 숨김
//=   $NEED_EVHUB_ID     현재 이벤트 번호 (예: 202610)
//=   $NEED_EVHUB_MOD$   현재 이벤트 모듈 함수 이름 (예: F_NEED_HW26_Hub)
//=   #NEED_EVHUB_SEEN   (계정) 마지막으로 연 이벤트 번호. 현재 번호와 다르면 버튼을 밝게(강조) 표시
//=   상태는 맵 로딩이 끝날 때마다 C++ 가 보내고(clif_need_eventhub_state),
//=   값을 바꾼 스크립트는 needeventhub_refresh(1) 로 접속자 전원에게 바로 다시 보낸다.
//=
//= 이벤트 모듈 규약 : callfunc(<모듈>, "<구역>", ...)
//=   "TITLE"    제목 문자열              "PERIOD"  기간 문자열
//=   "ACTIVE"   1 = 지금 열 수 있음 (0 이면 '진행 중인 이벤트 없음')
//=   "HEAD"     첫 화면 안내 / 핵심 상태 (mes 줄)
//=   "MENU"     메뉴 문자열 "가:나:다" (닫기는 공통이 붙인다. 항목 안에 ':' 금지)
//=   "RUN", n   메뉴 n 실행. 1 = 첫 화면으로 돌아감, 0 = 대화가 이미 끝남
//=   "SYNC"     허브 상태가 바뀐 뒤 모듈 쪽 NPC 표시 등을 맞춘다
//= 모듈은 자기 NPC 의 OnInit 에서 F_NEED_EventHub_Register 로 등록한다.
//= 등록된 이름만 연결할 수 있다 (없는 함수 callfunc 로 스크립트가 멈추는 것 방지).
//=
//= GM(60+) : @eventhub          상태 / 연결 / ON·OFF 화면
//=           @eventhub on|off   현재 연결된 모듈 그대로 버튼 표시 / 숨김
//=           @eventhub unseen   내 계정 '열어 봄' 기록 지우기 (강조 표시 확인용)
//============================================================

-	script	NEED_EVENT_HUB	-1,{
	end;

OnInit:
	bindatcmd "needevent", "NEED_EVENT_HUB::OnBtn", 0, 99;
	bindatcmd "eventhub", "NEED_EVENT_HUB::OnGM", 60, 60;
	end;

OnBtn:
	// 연타 방지 (1초)
	if (@needevent_tick && gettimetick(0) - @needevent_tick < 1000)
		end;
	@needevent_tick = gettimetick(0);
	callfunc "F_NEED_EventHub_Open";
	end;

OnGM:
	if (getgmlevel() < 60)
		end;
	.@a$ = (.@atcmd_numparameters >= 1) ? .@atcmd_parameters$[0] : "";
	if (.@a$ == "on" || .@a$ == "off") {
		.@n = callfunc("F_NEED_EventHub_Set", (.@a$ == "on"), $NEED_EVHUB_MOD$, $NEED_EVHUB_ID);
		if (.@n < 0)
			dispbottom "[Event Hub] 연결된 모듈/번호가 없어 켜지 않았습니다. @eventhub 화면에서 먼저 연결하세요.";
		else
			dispbottom "[Event Hub] 버튼 " + ((.@a$ == "on") ? "표시" : "숨김") + " (" + $NEED_EVHUB_MOD$ + " / " + $NEED_EVHUB_ID + "), 접속자 " + .@n + "명에게 전송";
		end;
	}
	if (.@a$ == "unseen") {
		#NEED_EVHUB_SEEN = 0;
		needeventhub_refresh();
		dispbottom "[Event Hub] 이 계정의 '열어 봄' 기록을 지웠습니다. (버튼 강조 확인용)";
		end;
	}
	callfunc "F_NEED_EventHub_GM";
	end;
}

//------------------------------------------------------------
// 모듈 등록 (모듈 NPC 의 OnInit 에서 호출)
//------------------------------------------------------------
function	script	F_NEED_EventHub_Register	{
	if (inarray($@NEED_EVHUB_MODS$, getarg(0)) < 0)
		$@NEED_EVHUB_MODS$[getarraysize($@NEED_EVHUB_MODS$)] = getarg(0);
	return;
}

// 등록된 모듈 이름인가
function	script	F_NEED_EventHub_Known	{
	return (getarg(0) != "" && inarray($@NEED_EVHUB_MODS$, getarg(0)) >= 0);
}

//------------------------------------------------------------
// 허브 상태 바꾸기 + 모든 모듈 SYNC + 접속자 전원에게 버튼 상태 재전송
//
// getarg(0) 1 = 버튼 표시 / 0 = 숨김
// getarg(1) 모듈 함수 이름 (""면 그대로)
// getarg(2) 이벤트 번호 (0 이하면 그대로)
// 반환 : 상태를 보낸 접속자 수, -1 = 켜려는데 모듈/번호가 없음 (아무것도 바꾸지 않음)
//------------------------------------------------------------
function	script	F_NEED_EventHub_Set	{
	.@on = (getarg(0) != 0);
	.@mod$ = getarg(1);
	.@id = getarg(2);

	if (.@mod$ != "" && !callfunc("F_NEED_EventHub_Known", .@mod$))
		return -1;
	if (.@on && ((.@mod$ == "" && !callfunc("F_NEED_EventHub_Known", $NEED_EVHUB_MOD$)) || (.@id <= 0 && $NEED_EVHUB_ID <= 0)))
		return -1;

	$NEED_EVHUB_ON = .@on;
	if (.@mod$ != "")
		$NEED_EVHUB_MOD$ = .@mod$;
	if (.@id > 0)
		$NEED_EVHUB_ID = .@id;

	if (playerattached())
		logmes "[EVHUB] on=" + $NEED_EVHUB_ON + " mod=" + $NEED_EVHUB_MOD$ + " id=" + $NEED_EVHUB_ID;
	debugmes "[EVHUB] on=" + $NEED_EVHUB_ON + " mod=" + $NEED_EVHUB_MOD$ + " id=" + $NEED_EVHUB_ID;

	callfunc "F_NEED_EventHub_SyncAll";
	return needeventhub_refresh(1);
}

// 등록된 모든 모듈의 "SYNC"
function	script	F_NEED_EventHub_SyncAll	{
	.@n = getarraysize($@NEED_EVHUB_MODS$);
	for (.@i = 0; .@i < .@n; .@i++)
		callfunc $@NEED_EVHUB_MODS$[.@i], "SYNC";
	return;
}

// 지금 허브가 이 모듈로 켜져 있나 (모듈 쪽 NPC 숨김 판단용)
function	script	F_NEED_EventHub_IsLive	{
	return ($NEED_EVHUB_ON != 0 && $NEED_EVHUB_MOD$ == getarg(0));
}

//------------------------------------------------------------
// 버튼을 눌렀을 때 여는 공통 화면
//------------------------------------------------------------
function	script	F_NEED_EventHub_Open	{
	.@mod$ = $NEED_EVHUB_MOD$;

	// 스크립트의 || 는 단락 평가를 하지 않으므로, 등록 확인이 끝난 뒤에만 모듈을 부른다.
	.@open = 0;
	if ($NEED_EVHUB_ON && callfunc("F_NEED_EventHub_Known", .@mod$))
		.@open = callfunc(.@mod$, "ACTIVE");
	if (!.@open) {
		dispbottom needtr("[이벤트] 지금 진행 중인 이벤트가 없습니다.", "[Event] There is no event running right now."), 0xFF6600;
		// 버튼이 남아 있었다면 숨김 상태로 다시 맞춘다
		needeventhub_refresh();
		end;
	}

	// 이번 이벤트 허브를 열었으므로 버튼 강조를 끈다 (계정 기준)
	if (#NEED_EVHUB_SEEN != $NEED_EVHUB_ID) {
		#NEED_EVHUB_SEEN = $NEED_EVHUB_ID;
		needeventhub_refresh();
	}

	setdialogsize 400, 440;
	while (1) {
		mes "^C05800" + callfunc(.@mod$, "TITLE") + "^000000";
		mes needtr("기간 : ", "Period : ") + "^0055FF" + callfunc(.@mod$, "PERIOD") + "^000000";
		callfunc .@mod$, "HEAD";

		.@menu$ = callfunc(.@mod$, "MENU");
		deletearray .@items$;
		explode .@items$, .@menu$, ":";
		.@cnt = getarraysize(.@items$);
		.@sel = select(.@menu$ + ":" + needtr("닫기", "Close"));
		if (.@sel < 1 || .@sel > .@cnt)
			end;

		clear;
		if (!callfunc(.@mod$, "RUN", .@sel))
			end;

		// 대화 중에 이벤트가 꺼졌거나 다른 이벤트로 바뀌었으면 여기서 끝낸다
		if (!$NEED_EVHUB_ON || $NEED_EVHUB_MOD$ != .@mod$ || !callfunc(.@mod$, "ACTIVE")) {
			clear;
			mes needtr("방금 이벤트가 종료되었습니다.", "The event has just ended.");
			close;
		}
		clear;
	}
	end;
}

//------------------------------------------------------------
// GM 화면 (@eventhub)
//------------------------------------------------------------
function	script	F_NEED_EventHub_GM	{
	while (1) {
		mes "[NEED Event Hub 관리]";
		mes "버튼 표시 : " + (($NEED_EVHUB_ON) ? "^009900ON^000000" : "^FF0000OFF^000000");
		mes "연결 모듈 : " + (($NEED_EVHUB_MOD$ == "") ? "(없음)" : $NEED_EVHUB_MOD$) + ((callfunc("F_NEED_EventHub_Known", $NEED_EVHUB_MOD$) || $NEED_EVHUB_MOD$ == "") ? "" : " ^FF0000(등록 안 됨)^000000");
		mes "이벤트 번호 : " + $NEED_EVHUB_ID;
		mes "등록된 모듈 : " + ((getarraysize($@NEED_EVHUB_MODS$)) ? implode($@NEED_EVHUB_MODS$, ", ") : "(없음)");
		mes "내 계정 열어 봄 : " + #NEED_EVHUB_SEEN;
		mes "^999999----------------------------^000000";
		mes "^999999ON/OFF 는 접속 중인 모든 클라이언트에 바로 반영되고,^000000";
		mes "^999999새로 접속하는 사람은 맵 로딩 때 받습니다.^000000";
		next;
		.@sel = select("버튼 표시 ON/OFF 바꾸기:다른 모듈 연결:이벤트 번호 바꾸기:내 열어 봄 기록 지우기:돌아가기");
		if (.@sel == 5)
			break;
		if (.@sel == 1) {
			.@n = callfunc("F_NEED_EventHub_Set", !$NEED_EVHUB_ON, "", 0);
			mes "[NEED Event Hub 관리]";
			if (.@n < 0)
				mes "^FF0000먼저 모듈과 이벤트 번호를 연결해야 켤 수 있습니다.^000000";
			else
				mes "버튼 " + (($NEED_EVHUB_ON) ? "표시" : "숨김") + ", 접속자 " + .@n + "명에게 전송했습니다.";
			next;
		}
		else if (.@sel == 2) {
			if (!getarraysize($@NEED_EVHUB_MODS$)) {
				mes "[NEED Event Hub 관리]";
				mes "등록된 모듈이 없습니다.";
				next;
				continue;
			}
			.@m = select(implode($@NEED_EVHUB_MODS$, ":") + ":취소") - 1;
			if (.@m >= getarraysize($@NEED_EVHUB_MODS$))
				continue;
			.@n = callfunc("F_NEED_EventHub_Set", $NEED_EVHUB_ON, $@NEED_EVHUB_MODS$[.@m], 0);
			mes "[NEED Event Hub 관리]";
			mes "연결 모듈 : " + $NEED_EVHUB_MOD$;
			next;
		}
		else if (.@sel == 3) {
			mes "[NEED Event Hub 관리]";
			mes "새 이벤트 번호 (예: 202610). 바꾸면 모든 계정에 버튼 강조가 다시 켜집니다.";
			mes "0 입력 시 취소";
			input .@id, 0, 999999999;
			if (.@id > 0) {
				callfunc "F_NEED_EventHub_Set", $NEED_EVHUB_ON, "", .@id;
				mes "이벤트 번호 : " + $NEED_EVHUB_ID;
			}
			next;
		}
		else {
			#NEED_EVHUB_SEEN = 0;
			needeventhub_refresh();
			mes "[NEED Event Hub 관리]";
			mes "이 계정의 열어 봄 기록을 지웠습니다.";
			next;
		}
	}
	close;
}
