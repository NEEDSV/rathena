//============================================================
//= NEED Event Hub (공통 이벤트 버튼 -> @needevent -> 이 NPC)
//============================================================
//= 클라이언트(NEEDClient + WARP NeedEventButton)의 Event 버튼은 [!] 버튼 왼쪽에 항상 보이고,
//= 누르면 채팅 명령 "@needevent" 하나만 보낸다. 어떤 이벤트 화면을 열지는 서버가 정한다.
//= 다음 이벤트는 서버 스크립트(모듈)와 버튼 BMP 만 바꾸면 되고 exe 는 다시 패치하지 않는다.
//=
//= 버튼 밝기 = 이벤트 진행 여부 (마우스 / 열어 봤는지와는 상관없음)
//=   $NEED_EVHUB_ON     1 = 진행 중(밝은 이미지), 0 = 진행 중인 이벤트 없음(흐린 이미지)
//=                      직접 바꾸지 않는다. F_NEED_EventHub_Update 가 계산해서 바뀔 때만 전원에게 보낸다.
//=   $NEED_EVHUB_MOD$   연결된 이벤트 모듈 함수 이름 (예: F_NEED_HW26_Hub)
//=   $NEED_EVHUB_ID     연결된 이벤트 번호 (예: 202610)
//=   $NEED_EVHUB_FORCE  0 = 자동(기본), 1 = 강제 ON, -1 = 강제 OFF  (GM 테스트 / 긴급 확인용)
//=   자동이면 ON = 연결된 모듈이 등록돼 있고 그 모듈의 "ACTIVE" 가 1.
//=   상태는 맵 로딩이 끝날 때마다 C++ 가 보낸다(clif_need_eventhub_state).
//=
//= 이벤트 모듈 규약 : callfunc(<모듈>, "<구역>", ...)
//=   "TITLE"    제목 문자열              "PERIOD"  기간 문자열
//=   "ACTIVE"   1 = 지금 진행 중 (버튼 밝음, 화면 열림)
//=   "HEAD"     첫 화면 안내 / 핵심 상태 (mes 줄)
//=   "MENU"     메뉴 문자열 "가:나:다" (닫기는 공통이 붙인다. 항목 안에 ':' 금지)
//=   "RUN", n   메뉴 n 실행. 1 = 첫 화면으로 돌아감, 0 = 대화가 이미 끝남
//=   "SYNC"     허브 연결이 바뀐 뒤 모듈 쪽 NPC 표시 등을 맞춘다
//=   모듈의 각 화면은 글을 다 쓴 뒤 next / select 전에 F_NEED_EventHub_Size 를 부른다 (공통 크기).
//= 모듈은 자기 NPC 의 OnInit 에서 F_NEED_EventHub_Register 로 등록하고,
//= 자기 이벤트가 켜지면 F_NEED_EventHub_Claim + F_NEED_EventHub_Update 를 부른다.
//= 등록된 이름만 callfunc 한다 (없는 함수 callfunc 로 스크립트가 멈추는 것 방지).
//= 스크립트의 || / && 는 단락 평가를 하지 않으므로, 등록 확인과 모듈 호출을 한 조건식에 쓰지 않는다.
//=
//= GM(60+) : @eventhub          상태 / 연결 / 자동·강제 화면
//=           @eventhub on|off   강제 ON / 강제 OFF (테스트·긴급 확인용)
//=           @eventhub auto     자동 연동으로 되돌리기 (정상 운영)
//============================================================

-	script	NEED_EVENT_HUB	-1,{
	end;

OnInit:
	bindatcmd "needevent", "NEED_EVENT_HUB::OnBtn", 0, 99;
	bindatcmd "eventhub", "NEED_EVENT_HUB::OnGM", 60, 60;
	end;

OnBtn:
	// 연타 방지 (1초)
	if (@needevent_tick && gettimetick(0) - @needevent_tick < 1000)
		end;
	@needevent_tick = gettimetick(0);
	callfunc "F_NEED_EventHub_Open";
	end;

OnGM:
	if (getgmlevel() < 60)
		end;
	.@a$ = (.@atcmd_numparameters >= 1) ? .@atcmd_parameters$[0] : "";
	if (.@a$ == "on" || .@a$ == "off" || .@a$ == "auto") {
		.@f = 0;
		if (.@a$ == "on")
			.@f = 1;
		else if (.@a$ == "off")
			.@f = -1;
		.@on = callfunc("F_NEED_EventHub_Force", .@f);
		dispbottom "[Event Hub] " + callfunc("F_NEED_EventHub_Mode$") + " / 버튼 " + ((.@on) ? "ON(밝음)" : "OFF(흐림)") + " (" + (($NEED_EVHUB_MOD$ == "") ? "연결 없음" : $NEED_EVHUB_MOD$ + " / " + $NEED_EVHUB_ID) + ")";
		end;
	}
	callfunc "F_NEED_EventHub_GM";
	end;
}

//------------------------------------------------------------
// 모듈 등록 (모듈 NPC 의 OnInit 에서 호출)
//------------------------------------------------------------
function	script	F_NEED_EventHub_Register	{
	if (inarray($@NEED_EVHUB_MODS$, getarg(0)) < 0)
		$@NEED_EVHUB_MODS$[getarraysize($@NEED_EVHUB_MODS$)] = getarg(0);
	return;
}

// 등록된 모듈 이름인가
function	script	F_NEED_EventHub_Known	{
	return (getarg(0) != "" && inarray($@NEED_EVHUB_MODS$, getarg(0)) >= 0);
}

// 연결된 모듈이 지금 진행 중인가 (등록 확인 뒤에만 모듈을 부른다)
function	script	F_NEED_EventHub_Running	{
	.@mod$ = $NEED_EVHUB_MOD$;
	if (!callfunc("F_NEED_EventHub_Known", .@mod$))
		return 0;
	return (callfunc(.@mod$, "ACTIVE") != 0);
}

//------------------------------------------------------------
// 버튼 상태 계산. 값이 바뀌었을 때만 접속자 전원에게 다시 보낸다.
// 반환 : 1 = ON(이벤트 진행 중) / 0 = OFF
//------------------------------------------------------------
function	script	F_NEED_EventHub_Update	{
	.@run = callfunc("F_NEED_EventHub_Running");
	.@on = .@run;
	if ($NEED_EVHUB_FORCE > 0)
		.@on = 1;
	else if ($NEED_EVHUB_FORCE < 0)
		.@on = 0;
	if (.@on != $NEED_EVHUB_ON) {
		$NEED_EVHUB_ON = .@on;
		.@n = needeventhub_refresh(1);
		debugmes "[EVHUB] state=" + .@on + " (running=" + .@run + " force=" + $NEED_EVHUB_FORCE + " mod=" + $NEED_EVHUB_MOD$ + " id=" + $NEED_EVHUB_ID + ") sent to " + .@n;
		if (playerattached())
			logmes "[EVHUB] state=" + .@on + " force=" + $NEED_EVHUB_FORCE + " mod=" + $NEED_EVHUB_MOD$ + " id=" + $NEED_EVHUB_ID;
	}
	return .@on;
}

//------------------------------------------------------------
// 이벤트 모듈이 자기 이벤트가 켜졌을 때 Hub 연결을 가져간다.
// 지금 연결된 모듈이 없거나, 같은 모듈이거나, 등록되지 않았거나, 진행 중이 아니면 바꾼다.
// (다른 이벤트가 진행 중이면 그 연결을 빼앗지 않는다.)
// getarg(0) 모듈 함수 이름 / getarg(1) 이벤트 번호
// 반환 : 1 = 연결됨 / 0 = 다른 이벤트가 진행 중이라 그대로 둠
//------------------------------------------------------------
function	script	F_NEED_EventHub_Claim	{
	.@mod$ = getarg(0);
	.@cur$ = $NEED_EVHUB_MOD$;
	.@take = 1;
	if (.@cur$ != "" && .@cur$ != .@mod$ && callfunc("F_NEED_EventHub_Known", .@cur$)) {
		if (callfunc(.@cur$, "ACTIVE"))
			.@take = 0;
	}
	if (!.@take)
		return 0;
	if (.@cur$ != .@mod$ || $NEED_EVHUB_ID != getarg(1)) {
		$NEED_EVHUB_MOD$ = .@mod$;
		$NEED_EVHUB_ID = getarg(1);
		debugmes "[EVHUB] bound to " + .@mod$ + " / " + getarg(1);
	}
	return 1;
}

//------------------------------------------------------------
// GM : 다른 모듈로 직접 연결 (등록된 모듈만). 모든 모듈 SYNC 후 상태 계산.
//------------------------------------------------------------
function	script	F_NEED_EventHub_Bind	{
	if (!callfunc("F_NEED_EventHub_Known", getarg(0)))
		return -1;
	$NEED_EVHUB_MOD$ = getarg(0);
	if (getarg(1) > 0)
		$NEED_EVHUB_ID = getarg(1);
	callfunc "F_NEED_EventHub_SyncAll";
	return callfunc("F_NEED_EventHub_Update");
}

// GM : 0 자동 / 1 강제 ON / -1 강제 OFF. 반환 = 결과 상태(0/1)
function	script	F_NEED_EventHub_Force	{
	$NEED_EVHUB_FORCE = getarg(0);
	.@on = callfunc("F_NEED_EventHub_Update");
	callfunc "F_NEED_EventHub_SyncAll";
	return .@on;
}

function	script	F_NEED_EventHub_Mode$	{
	if ($NEED_EVHUB_FORCE > 0)
		return "^FF6600강제 ON(테스트)^000000";
	if ($NEED_EVHUB_FORCE < 0)
		return "^FF6600강제 OFF(테스트)^000000";
	return "자동";
}

// 등록된 모든 모듈의 "SYNC"
function	script	F_NEED_EventHub_SyncAll	{
	.@n = getarraysize($@NEED_EVHUB_MODS$);
	for (.@i = 0; .@i < .@n; .@i++)
		callfunc $@NEED_EVHUB_MODS$[.@i], "SYNC";
	return;
}

// 지금 버튼이 이 모듈로 ON 인가 (모듈 쪽 NPC 숨김 판단용)
function	script	F_NEED_EventHub_IsLive	{
	return ($NEED_EVHUB_ON != 0 && $NEED_EVHUB_MOD$ == getarg(0));
}

//------------------------------------------------------------
// Event Hub 대화창 크기 (모든 이벤트 모듈 공통)
//
// 그 화면의 글(mes)을 다 쓴 뒤, next / select / close 직전에만 부른다.
// 대화창이 열리기 전(첫 mes 전)에 부르면 클라이언트가 무시하고, 글을 쓰는 중간에 부르면
// 줄이 겹쳐 그려진다 (need_daily_quest.txt 2026-10-21 실측).
// 높이 = 여백 65px + 줄 수 x 18px. 330 이면 14줄까지 스크롤 없음 (가장 긴 일반 화면 = 내 이벤트 현황 14줄).
// 너비 440 은 EN 안내 한 줄(약 60자)이 줄바꿈되지 않는 폭.
//------------------------------------------------------------
function	script	F_NEED_EventHub_Size	{
	setdialogsize(440, 330);
	return;
}

//------------------------------------------------------------
// 버튼을 눌렀을 때 여는 공통 화면
//------------------------------------------------------------
function	script	F_NEED_EventHub_Open	{
	.@mod$ = $NEED_EVHUB_MOD$;

	// 버튼이 ON 이고 연결된 모듈이 실제로 진행 중일 때만 연다.
	.@open = 0;
	if ($NEED_EVHUB_ON && callfunc("F_NEED_EventHub_Known", .@mod$))
		.@open = callfunc(.@mod$, "ACTIVE");
	if (!.@open) {
		dispbottom needtr("현재 진행 중인 이벤트가 없습니다.", "There is no event currently in progress."), 0xFF6600;
		end;
	}

	while (1) {
		mes "^C05800" + callfunc(.@mod$, "TITLE") + "^000000";
		mes needtr("기간 : ", "Period : ") + "^0055FF" + callfunc(.@mod$, "PERIOD") + "^000000";
		callfunc .@mod$, "HEAD";
		callfunc "F_NEED_EventHub_Size";

		.@menu$ = callfunc(.@mod$, "MENU");
		deletearray .@items$;
		explode .@items$, .@menu$, ":";
		.@cnt = getarraysize(.@items$);
		.@sel = select(.@menu$ + ":" + needtr("닫기", "Close"));
		if (.@sel < 1 || .@sel > .@cnt)
			end;

		clear;
		if (!callfunc(.@mod$, "RUN", .@sel))
			end;

		// 대화 중에 이벤트가 꺼졌거나 다른 이벤트로 바뀌었으면 여기서 끝낸다
		if (!$NEED_EVHUB_ON || $NEED_EVHUB_MOD$ != .@mod$ || !callfunc(.@mod$, "ACTIVE")) {
			clear;
			mes needtr("방금 이벤트가 종료되었습니다.", "The event has just ended.");
			close;
		}
		clear;
	}
	end;
}

//------------------------------------------------------------
// GM 화면 (@eventhub)
//------------------------------------------------------------
function	script	F_NEED_EventHub_GM	{
	while (1) {
		.@run = callfunc("F_NEED_EventHub_Running");
		mes "[NEED Event Hub 관리]";
		mes "버튼 : " + (($NEED_EVHUB_ON) ? "^009900ON (밝음)^000000" : "^999999OFF (흐림)^000000") + "  /  " + callfunc("F_NEED_EventHub_Mode$");
		mes "연결 모듈 : " + (($NEED_EVHUB_MOD$ == "") ? "(없음)" : $NEED_EVHUB_MOD$) + ((callfunc("F_NEED_EventHub_Known", $NEED_EVHUB_MOD$) || $NEED_EVHUB_MOD$ == "") ? "" : " ^FF0000(등록 안 됨)^000000");
		mes "이벤트 번호 : " + $NEED_EVHUB_ID + "  /  모듈 진행 중 : " + ((.@run) ? "예" : "아니오");
		mes "등록된 모듈 : " + ((getarraysize($@NEED_EVHUB_MODS$)) ? implode($@NEED_EVHUB_MODS$, ", ") : "(없음)");
		mes "^999999----------------------------^000000";
		mes "^999999정상 운영은 '자동' 입니다: 이벤트를 켜면 그 모듈이 연결되고^000000";
		mes "^999999버튼이 밝아지며, 이벤트를 끄면 흐려집니다.^000000";
		mes "^999999강제 ON/OFF 는 테스트·긴급 확인용입니다.^000000";
		next;
		.@sel = select("자동 연동 (정상 운영):강제 ON (테스트):강제 OFF (테스트):다른 모듈 연결:이벤트 번호 바꾸기:돌아가기");
		if (.@sel == 6)
			break;
		if (.@sel <= 3) {
			.@f = 0;
			if (.@sel == 2)
				.@f = 1;
			else if (.@sel == 3)
				.@f = -1;
			.@on = callfunc("F_NEED_EventHub_Force", .@f);
			mes "[NEED Event Hub 관리]";
			mes callfunc("F_NEED_EventHub_Mode$") + " / 버튼 " + ((.@on) ? "ON(밝음)" : "OFF(흐림)");
			next;
		}
		else if (.@sel == 4) {
			if (!getarraysize($@NEED_EVHUB_MODS$)) {
				mes "[NEED Event Hub 관리]";
				mes "등록된 모듈이 없습니다.";
				next;
				continue;
			}
			.@m = select(implode($@NEED_EVHUB_MODS$, ":") + ":취소") - 1;
			if (.@m >= getarraysize($@NEED_EVHUB_MODS$))
				continue;
			callfunc "F_NEED_EventHub_Bind", $@NEED_EVHUB_MODS$[.@m], 0;
			mes "[NEED Event Hub 관리]";
			mes "연결 모듈 : " + $NEED_EVHUB_MOD$;
			next;
		}
		else {
			mes "[NEED Event Hub 관리]";
			mes "새 이벤트 번호 (예: 202610). 0 입력 시 취소";
			input .@id, 0, 999999999;
			if (.@id > 0) {
				$NEED_EVHUB_ID = .@id;
				mes "이벤트 번호 : " + $NEED_EVHUB_ID;
			}
			next;
		}
	}
	close;
}
