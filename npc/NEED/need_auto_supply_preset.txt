-	script	SupplyPresetSystem	FAKE_NPC,{
	end;

OnInit:
	bindatcmd "자동보충", strnpcinfo(3)+"::OnSupplySetting", 0, 99;
	bindatcmd "보충", strnpcinfo(3)+"::OnSupplyRecent", 0, 99;
	bindatcmd "보충1", strnpcinfo(3)+"::OnSupply1", 0, 99;
	bindatcmd "보충2", strnpcinfo(3)+"::OnSupply2", 0, 99;
	bindatcmd "보충3", strnpcinfo(3)+"::OnSupply3", 0, 99;

	.MaxPreset = 3;
	.MaxSlot = 10;

	// 프리셋 2: 캐시 해금
	.UnlockCash2 = 50000;

	// 프리셋 3: 특정 아이템 해금
	// 아래 아이템 ID와 필요 수량을 서버 설정에 맞게 변경하세요.
	.UnlockItem3 = 399907;
	.UnlockItem3Amount = 1;

	.Cooldown = 10;
	end;

OnSupplySetting:
	mes needtr("[자동 보충 세팅]","[Auto Supply Setup]");
	mes needtr("설정할 프리셋을 선택하세요.","Choose a preset to set up.");
	next;

	.@name1$ = Supply_Name_1$;
	if (.@name1$ == "")
		.@name1$ = needtr("프리셋 1", "Preset 1");
	.@menu1$ = .@name1$ + needtr(" [무료]", " [Free]");

	.@name2$ = Supply_Name_2$;
	if (.@name2$ == "")
		.@name2$ = needtr("프리셋 2", "Preset 2");
	if (Supply_Unlocked_2)
		.@menu2$ = .@name2$;
	else
		.@menu2$ = .@name2$ + needtr(" [캐시 해금]", " [Unlock with Cash]");

	.@name3$ = Supply_Name_3$;
	if (.@name3$ == "")
		.@name3$ = needtr("프리셋 3", "Preset 3");
	if (Supply_Unlocked_3)
		.@menu3$ = .@name3$;
	else
		.@menu3$ = .@name3$ + needtr(" [아이템 해금]", " [Unlock with an item]");

	.@sel = select(
		.@menu1$,
		.@menu2$,
		.@menu3$,
		needtr("자동 보충 도움말","Auto Supply help")
	);

	if (.@sel == 4) {
		callsub S_Help;
		end;
	}

	.@preset = .@sel;

	if (!callsub(S_CheckPresetAccess, .@preset, 1))
		end;

	callsub S_EditPreset, .@preset;
	end;

OnSupplyRecent:
	.@preset = Supply_LastPreset;

	if (.@preset < 1 || .@preset > .MaxPreset) {
		dispbottom needtr("[자동 보충] 최근 사용한 프리셋이 없습니다.","[Auto Supply] You have not used a preset yet.");
		dispbottom needtr("[자동 보충] 먼저 @보충1, @보충2, @보충3 중 하나를 사용해주세요.","[Auto Supply] Use @보충1, @보충2 or @보충3 first.");
		end;
	}

	if (!callsub(S_CheckPresetAccess, .@preset, 0))
		end;

	callsub S_Supply, .@preset;
	end;

OnSupply1:
	callsub S_Supply, 1;
	end;

OnSupply2:
	if (!callsub(S_CheckPresetAccess, 2, 0))
		end;

	callsub S_Supply, 2;
	end;

OnSupply3:
	if (!callsub(S_CheckPresetAccess, 3, 0))
		end;

	callsub S_Supply, 3;
	end;

// 도움말
// =========================
S_Help:
	mes needtr("[자동 보충 도움말]","[Auto Supply Help]");
	mes needtr("자동 보충은 설정한 목표 수량만큼","Auto Supply tops your inventory up to the target");
	mes needtr("부족한 아이템을 창고에서 자동으로 꺼내주는 편의 기능입니다.","quantity you set, pulling what is missing from Storage.");
	next;

	mes needtr("[사용 방법]","[How to use]");
	mes needtr("1. @자동보충 입력","1. Enter @자동보충");
	mes needtr("2. 프리셋 선택","2. Choose a preset");
	mes needtr("3. 아이템 등록 및 목표 수량 설정","3. Register items and set target quantities");
	mes needtr("4. @보충1, @보충2, @보충3 사용","4. Use @보충1, @보충2 or @보충3");
	mes needtr("5. 이후 @보충으로 최근 프리셋 재사용","5. Later, @보충 reuses your most recent preset");
	next;

	mes needtr("[예시]","[Example]");
	mes needtr("파리의 날개 목표 수량 100개","Fly Wing target quantity 100");
	mes needtr("현재 인벤토리 35개 보유","Currently 35 in your inventory");
	mes needtr("@보충1 입력 시 창고에서 65개를 자동 보충합니다.","@보충1 auto-supplies 65 from Storage.");
	next;

	mes needtr("[프리셋 안내]","[Presets]");
	mes needtr("@보충1 : 기본 무료 프리셋","@보충1 : the free default preset");
	mes needtr("@보충2 : 캐시로 해금","@보충2 : unlocked with Cash");
	mes needtr("@보충3 : 지정 아이템으로 해금","@보충3 : unlocked with a specific item");
	mes needtr("@보충 : 최근 사용 프리셋 실행","@보충 : run your most recent preset");
	next;

	mes needtr("[주의 사항]","[Notes]");
	mes needtr("자동 보충은 마을에서만 사용할 수 있습니다.","Auto Supply only works in a town.");
	mes needtr("창고 수량, 무게 또는 인벤토리 공간이 부족하면","If Storage stock, weight or inventory space runs short,");
	mes needtr("일부 아이템 보충에 실패할 수 있습니다.","some items may fail to be supplied.");
	mes needtr("실패한 아이템은 채팅창에 안내됩니다.","Failed items are listed in your chat window.");
	return;

// =========================

S_EditPreset:
	.@preset = getarg(0);

	while (1) {
		.@name$ = getd("Supply_Name_" + .@preset + "$");

		if (.@name$ == "")
			.@name$ = needtr("프리셋 ", "Preset ") + .@preset;

		mes needtr("[자동 보충 세팅]","[Auto Supply Setup]");
		mes needtr("현재 프리셋 : ^0000FF","Current preset : ^0000FF") + .@name$ + "^000000";
		next;

		.@sel = select(
			needtr("프리셋 이름 수정","Rename preset"),
			needtr("아이템 등록","Register items"),
			needtr("등록 목록 보기","View registered items"),
			needtr("목표 수량 변경 / 삭제","Change or delete a target quantity"),
			needtr("다른 프리셋에서 복사","Copy from another preset"),
			needtr("프리셋 초기화","Reset preset"),
			needtr("종료","Exit")
		);

		switch (.@sel) {
		case 1:
			callsub S_ChangeName, .@preset;
			break;

		case 2:
			callsub S_RegisterItem, .@preset;
			break;

		case 3:
			callsub S_ShowPreset, .@preset;
			break;

		case 4:
			callsub S_EditAmount, .@preset;
			break;

		case 5:
			callsub S_CopyPreset, .@preset;
			break;

		case 6:
			callsub S_ClearPreset, .@preset;
			break;

		default:
			close;
		}
	}
	end;

// =========================
// 프리셋 이름 수정
// =========================
S_ChangeName:
	.@preset = getarg(0);

	mes needtr("[프리셋 이름 수정]","[Rename Preset]");
	mes needtr("새 이름을 입력하세요.","Enter a new name.");
	mes needtr("예: 사냥 세팅, 전장 세팅, MVP 세팅","Example: Hunting, Battleground, MVP");
	mes needtr("':' 및 '^' 문자는 사용할 수 없습니다.","The ':' and '^' characters cannot be used.");
	next;

	.@input_ret = input(.@name$, 1, 12);

	if (.@input_ret != 0) {
		mes needtr("[프리셋 이름 수정]","[Rename Preset]");
		mes needtr("프리셋 이름은 1~12자로 입력해주세요.","A preset name must be 1 to 12 characters.");
		next;
		return;
	}

	if (.@name$ == "") {
		mes needtr("이름을 입력하지 않았습니다.","You did not enter a name.");
		next;
		return;
	}

	if (strpos(.@name$, ":") != -1 || strpos(.@name$, "^") != -1) {
		mes needtr("[프리셋 이름 수정]","[Rename Preset]");
		mes needtr("사용할 수 없는 문자가 포함되어 있습니다.","That name contains a character you cannot use.");
		mes needtr("':' 및 '^' 문자를 제외하고 입력해주세요.","Enter it without the ':' and '^' characters.");
		next;
		return;
	}

	setd "Supply_Name_" + .@preset + "$", .@name$;

	mes needtr("프리셋 이름이 변경되었습니다.","The preset has been renamed.");
	mes needtr("새 이름 : ^0000FF","New name : ^0000FF") + .@name$ + "^000000";
	next;
	return;

// =========================
// 아이템 등록
// =========================
S_RegisterItem:
	.@preset = getarg(0);

	.@free_slot = -1;
	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		if (getd("Supply_Item_" + .@preset + "_" + .@i) == 0) {
			.@free_slot = .@i;
			break;
		}
	}

	if (.@free_slot < 0) {
		mes needtr("[아이템 등록]","[Register Items]");
		mes needtr("이 프리셋은 최대 ","This preset holds up to ") + .MaxSlot + needtr("종까지 등록할 수 있습니다."," item types.");
		next;
		return;
	}

	getinventorylist();

	// 이미 등록된 아이템 목록을 한 번만 구성
	// inarray()로 확인하여 인벤토리 전체에 대해 중첩 루프가 발생하지 않도록 한다.
	setarray .@registered[0], -1;
	.@registered_count = 0;

	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		.@registered_id = getd("Supply_Item_" + .@preset + "_" + .@i);

		if (.@registered_id <= 0)
			continue;

		.@registered[.@registered_count] = .@registered_id;
		.@registered_count++;
	}

	// 등록 가능한 아이템만 후보 배열에 구성
	// 동일 아이템 중복 여부도 inarray()로 처리하여 O(N^2) 스크립트 루프를 제거
	setarray .@item[0], -1;
	.@count = 0;

	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		.@id = @inventorylist_id[.@i];

		if (.@id <= 0)
			continue;

		.@type = getiteminfo(.@id, ITEMINFO_TYPE);

		// 소비템 / 기타 / 탄약 허용
		if (.@type != IT_HEALING &&
			.@type != IT_USABLE &&
			.@type != IT_ETC &&
			.@type != IT_AMMO)
			continue;

		// 이미 프리셋에 등록된 아이템 제외
		if (.@registered_count > 0 && inarray(.@registered, .@id) != -1)
			continue;

		// 인벤토리 내 동일 아이템 ID 중복 제외
		if (.@count > 0 && inarray(.@item, .@id) != -1)
			continue;

		.@item[.@count] = .@id;
		.@count++;
	}

	if (.@count == 0) {
		mes needtr("[아이템 등록]","[Register Items]");
		mes needtr("등록 가능한 아이템이 없습니다.","There is no item you can register.");
		next;
		return;
	}

	.@page_size = 8;
	.@page = 0;
	.@max_page = (.@count - 1) / .@page_size;

S_RegisterItem_Page:
	.@start = .@page * .@page_size;
	.@end = .@start + .@page_size;

	if (.@end > .@count)
		.@end = .@count;

	.@menu$ = "";
	.@menu_count = 0;

	// 현재 페이지의 아이템만 select() 메뉴에 구성
	for (.@i = .@start; .@i < .@end; .@i++) {
		.@menu$ += getitemname(.@item[.@i]) + " [" + countitem(.@item[.@i]) + needtr("개 보유]:", " held]:");

		// action 0 = 아이템 선택
		.@menu_action[.@menu_count] = 0;
		.@menu_value[.@menu_count] = .@i;
		.@menu_count++;
	}

	if (.@page > 0) {
		.@menu$ += needtr("<< 이전 페이지:", "<< Prev Page:");
		.@menu_action[.@menu_count] = 1;
		.@menu_value[.@menu_count] = 0;
		.@menu_count++;
	}

	if (.@page < .@max_page) {
		.@menu$ += needtr("다음 페이지 >>:", "Next Page >>:");
		.@menu_action[.@menu_count] = 2;
		.@menu_value[.@menu_count] = 0;
		.@menu_count++;
	}

	.@menu$ += needtr("취소:", "Cancel:");
	.@menu_action[.@menu_count] = 3;
	.@menu_value[.@menu_count] = 0;
	.@menu_count++;

	mes needtr("[아이템 등록]","[Register Items]");
	mes needtr("등록할 아이템을 선택하세요.","Choose an item to register.");
	mes needtr("페이지 : ^0000FF","Page : ^0000FF") + (.@page + 1) + " / " + (.@max_page + 1) + "^000000";
	next;

	.@sel = select(.@menu$) - 1;
	.@action = .@menu_action[.@sel];

	if (.@action == 1) {
		.@page--;
		goto S_RegisterItem_Page;
	}

	if (.@action == 2) {
		.@page++;
		goto S_RegisterItem_Page;
	}

	if (.@action == 3)
		return;

	.@idx = .@menu_value[.@sel];
	.@item_id = .@item[.@idx];

	mes "[" + getitemname(.@item_id) + "]";
	mes needtr("몇 개까지 자동 보충할까요?","How many should Auto Supply keep?");
	next;

	input .@amount, 1, 30000;

	if (.@amount <= 0) {
		mes needtr("잘못된 수량입니다.","That quantity is not valid.");
		next;
		return;
	}

	setd "Supply_Item_" + .@preset + "_" + .@free_slot, .@item_id;
	setd "Supply_Amount_" + .@preset + "_" + .@free_slot, .@amount;

	mes needtr("등록 완료!","Registered!");
	mes getitemname(.@item_id) + " : " + .@amount + needtr("개",".");
	next;
	return;

// =========================
// 등록 목록 보기
// =========================
S_ShowPreset:
	.@preset = getarg(0);
	.@name$ = getd("Supply_Name_" + .@preset + "$");

	if (.@name$ == "")
		.@name$ = needtr("프리셋 ", "Preset ") + .@preset;

	mes needtr("[등록 목록]","[Registered Items]");
	mes "^0000FF" + .@name$ + "^000000";

	.@has = 0;

	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		.@item_id = getd("Supply_Item_" + .@preset + "_" + .@i);
		.@amount = getd("Supply_Amount_" + .@preset + "_" + .@i);

		if (.@item_id <= 0)
			continue;

		mes "- " + getitemname(.@item_id) + " : " + .@amount + needtr("개",".");
		.@has = 1;
	}

	if (!.@has)
		mes needtr("등록된 아이템이 없습니다.","No items are registered.");

	next;
	return;

// =========================
// 목표 수량 변경 / 0 입력 시 삭제
// =========================
S_EditAmount:
	.@preset = getarg(0);

	.@menu$ = "";
	.@count = 0;

	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		.@item_id = getd("Supply_Item_" + .@preset + "_" + .@i);
		.@amount = getd("Supply_Amount_" + .@preset + "_" + .@i);

		if (.@item_id <= 0)
			continue;

		.@menu$ += getitemname(.@item_id) + " [" + .@amount + needtr("개]:", "]:");
		.@slot[.@count] = .@i;
		.@count++;
	}

	if (.@count == 0) {
		mes needtr("[목표 수량 변경]","[Change Target Quantity]");
		mes needtr("변경할 아이템이 없습니다.","There is no item to change.");
		next;
		return;
	}

	mes needtr("[목표 수량 변경]","[Change Target Quantity]");
	mes needtr("변경할 아이템을 선택하세요.","Choose an item to change.");
	next;

	.@sel = select(.@menu$);
	.@idx = .@slot[.@sel - 1];
	.@item_id = getd("Supply_Item_" + .@preset + "_" + .@idx);
	.@old_amount = getd("Supply_Amount_" + .@preset + "_" + .@idx);

	mes "[" + getitemname(.@item_id) + "]";
	mes needtr("현재 목표 수량 : ","Current target quantity : ") + .@old_amount + needtr("개",".");
	mes needtr("새 목표 수량을 입력하세요.","Enter the new target quantity.");
	mes needtr("0을 입력하면 등록 목록에서 삭제됩니다.","Enter 0 to remove it from the list.");
	next;

	input .@amount, 0, 30000;

	if (.@amount == 0) {
		setd "Supply_Item_" + .@preset + "_" + .@idx, 0;
		setd "Supply_Amount_" + .@preset + "_" + .@idx, 0;

		mes needtr("[목표 수량 변경]","[Change Target Quantity]");
		mes getitemname(.@item_id) + needtr(" 아이템이 삭제되었습니다."," has been removed.");
		next;
		return;
	}

	setd "Supply_Amount_" + .@preset + "_" + .@idx, .@amount;

	mes needtr("[목표 수량 변경]","[Change Target Quantity]");
	mes getitemname(.@item_id) + needtr("의 목표 수량이 변경되었습니다."," has a new target quantity.");
	mes .@old_amount + needtr("개 → "," → ") + .@amount + needtr("개",".");
	next;
	return;

// =========================
// 다른 프리셋의 아이템/수량 복사
// 대상 프리셋의 이름은 유지
// =========================
S_CopyPreset:
	.@target = getarg(0);

	mes needtr("[프리셋 복사]","[Copy Preset]");
	mes needtr("복사할 원본 프리셋을 선택하세요.","Choose the preset to copy from.");
	mes needtr("현재 프리셋의 이름은 유지되며,","The current preset keeps its name;");
	mes needtr("등록 아이템과 목표 수량만 덮어씁니다.","only its items and target quantities are overwritten.");
	next;

	.@sel = select(needtr("프리셋 1","Preset 1"), needtr("프리셋 2","Preset 2"), needtr("프리셋 3","Preset 3"), needtr("취소","Cancel"));

	if (.@sel == 4)
		return;

	.@source = .@sel;

	if (.@source == .@target) {
		mes needtr("[프리셋 복사]","[Copy Preset]");
		mes needtr("현재 프리셋과 같은 프리셋은 선택할 수 없습니다.","You cannot copy a preset onto itself.");
		next;
		return;
	}

	if (!callsub(S_CheckPresetAccess, .@source, 0)) {
		mes needtr("[프리셋 복사]","[Copy Preset]");
		mes needtr("해금되지 않은 프리셋은 복사할 수 없습니다.","A preset you have not unlocked cannot be copied.");
		next;
		return;
	}

	mes needtr("[프리셋 복사]","[Copy Preset]");
	mes needtr("현재 프리셋의 등록 내용이 모두 교체됩니다.","Everything registered in the current preset will be replaced.");
	mes needtr("계속하시겠습니까?","Continue?");
	next;

	if (select(needtr("취소","Cancel"), needtr("복사하기","Copy")) == 1)
		return;

	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		.@item_id = getd("Supply_Item_" + .@source + "_" + .@i);
		.@amount = getd("Supply_Amount_" + .@source + "_" + .@i);

		setd "Supply_Item_" + .@target + "_" + .@i, .@item_id;
		setd "Supply_Amount_" + .@target + "_" + .@i, .@amount;
	}

	mes needtr("[프리셋 복사]","[Copy Preset]");
	mes needtr("프리셋 ","Copied everything from preset ") + .@source + needtr("의 등록 내용을 복사했습니다.",".");
	next;
	return;

// =========================
// 프리셋 초기화
// =========================
S_ClearPreset:
	.@preset = getarg(0);

	mes needtr("[프리셋 초기화]","[Reset Preset]");
	mes needtr("정말 초기화하시겠습니까?","Really reset it?");
	next;

	if (select(needtr("취소","Cancel"), needtr("초기화","Reset")) == 1)
		return;

	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		setd "Supply_Item_" + .@preset + "_" + .@i, 0;
		setd "Supply_Amount_" + .@preset + "_" + .@i, 0;
	}

	setd "Supply_Name_" + .@preset + "$", "";

	if (Supply_LastPreset == .@preset)
		Supply_LastPreset = 0;

	mes needtr("초기화되었습니다.","The preset has been reset.");
	next;
	return;

// =========================
// 프리셋 접근 체크
// getarg(0): preset
// getarg(1): setting mode, 1이면 해금 안내창 출력
// 해금 여부는 캐릭터 단위로 저장
// =========================
S_CheckPresetAccess:
	.@preset = getarg(0);
	.@setting = getarg(1);

	if (.@preset == 1)
		return 1;

	if (getd("Supply_Unlocked_" + .@preset))
		return 1;

	if (!.@setting) {
		dispbottom needtr("[자동 보충] 프리셋 ","[Auto Supply] Preset ") + .@preset + needtr("은 아직 해금되지 않았습니다."," is not unlocked yet.");
		dispbottom needtr("[자동 보충] @자동보충에서 먼저 해금해주세요.","[Auto Supply] Unlock it in @자동보충 first.");
		return 0;
	}

	// 프리셋 2: 캐시 해금
	if (.@preset == 2) {
		mes needtr("[프리셋 해금]","[Unlock Preset]");
		mes needtr("프리셋 2는 캐시 해금이 필요합니다.","Preset 2 is unlocked with Cash.");
		mes needtr("해금 비용 : ^FF0000","Unlock cost : ^FF0000") + .UnlockCash2 + needtr(" 캐시^000000"," Cash^000000");
		next;

		if (select(needtr("취소","Cancel"), needtr("해금하기","Unlock")) == 1)
			return 0;

		if (#CASHPOINTS < .UnlockCash2) {
			mes needtr("[프리셋 해금]","[Unlock Preset]");
			mes needtr("캐시가 부족합니다.","You do not have enough Cash.");
			mes needtr("필요 캐시 : ","Cash required : ") + .UnlockCash2;
			mes needtr("보유 캐시 : ","Your Cash : ") + #CASHPOINTS;
			next;
			return 0;
		}

		#CASHPOINTS -= .UnlockCash2;
		Supply_Unlocked_2 = 1;

		mes needtr("[프리셋 해금]","[Unlock Preset]");
		mes needtr("프리셋 2가 해금되었습니다.","Preset 2 is now unlocked.");
		next;
		return 1;
	}

	// 프리셋 3: 특정 아이템 해금
	if (.@preset == 3) {
		if (.UnlockItem3 <= 0 || .UnlockItem3Amount <= 0) {
			mes needtr("[프리셋 해금]","[Unlock Preset]");
			mes needtr("프리셋 3 해금 아이템이 설정되지 않았습니다.","No unlock item is configured for preset 3.");
			mes needtr("관리자에게 문의해주세요.","Please contact an administrator.");
			next;
			return 0;
		}

		.@item_name$ = getitemname(.UnlockItem3);
		.@owned = countitem(.UnlockItem3);

		mes needtr("[프리셋 해금]","[Unlock Preset]");
		mes needtr("프리셋 3는 지정 아이템 해금이 필요합니다.","Preset 3 is unlocked with a specific item.");
		mes needtr("필요 아이템 : ^0000FF","Required item : ^0000FF") + .@item_name$ + "^000000";
		mes needtr("필요 수량 : ^FF0000","Required : ^FF0000") + .UnlockItem3Amount + needtr("개^000000","^000000");
		mes needtr("보유 수량 : ","In your inventory : ") + .@owned + needtr("개",".");
		mes needtr("^FF0000해당 아이템은 전장에서 전장포인트로 구매하실 수 있습니다.^000000","^FF0000You can buy that item with Battleground Points in the Battleground.^000000");
		next;

		if (select(needtr("취소","Cancel"), needtr("아이템 사용","Use the item")) == 1)
			return 0;

		// 확인창 이후 수량을 다시 검사하여 중복 처리나 수량 변동을 방지
		if (countitem(.UnlockItem3) < .UnlockItem3Amount) {
			mes needtr("[프리셋 해금]","[Unlock Preset]");
			mes needtr("해금 아이템이 부족합니다.","You do not have enough of the unlock item.");
			mes needtr("필요 아이템 : ","Required item : ") + .@item_name$ + " " + .UnlockItem3Amount + needtr("개",".");
			mes needtr("보유 수량 : ","In your inventory : ") + countitem(.UnlockItem3) + needtr("개",".");
			next;
			return 0;
		}

		delitem .UnlockItem3, .UnlockItem3Amount;
		Supply_Unlocked_3 = 1;

		mes needtr("[프리셋 해금]","[Unlock Preset]");
		mes needtr("프리셋 3가 해금되었습니다.","Preset 3 is now unlocked.");
		mes .@item_name$ + " " + .UnlockItem3Amount + needtr("개를 사용했습니다."," used.");
		next;
		return 1;
	}

	return 0;

// =========================
// 자동 보충 실행
// =========================
S_Supply:
	.@preset = getarg(0);

	// 마을에서만 사용 가능
	.@map$ = strcharinfo(3);

	if (!getmapflag(.@map$, mf_town) && .@map$ != "bat_room" && .@map$ != "turbo_room") {
		dispbottom needtr("[자동 보충] 자동 보충은 마을 또는 전장 대기실에서만 사용할 수 있습니다.","[Auto Supply] Auto Supply only works in a town or a Battleground waiting room.");
		end;
	}

	.@has_item = 0;
	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		if (getd("Supply_Item_" + .@preset + "_" + .@i) > 0 &&
			getd("Supply_Amount_" + .@preset + "_" + .@i) > 0) {
			.@has_item = 1;
			break;
		}
	}

	.@name$ = getd("Supply_Name_" + .@preset + "$");
	if (.@name$ == "")
		.@name$ = needtr("프리셋 ", "Preset ") + .@preset;

	if (!.@has_item) {
		dispbottom needtr("[자동 보충] ","[Auto Supply] ") + .@name$ + needtr("에 등록된 아이템이 없습니다."," has no registered items.");
		end;
	}

	// 짧은 재사용 대기시간
	.@now_tick = gettimetick(2);
	if (Supply_LastUseTick > 0 && .@now_tick < Supply_LastUseTick + .Cooldown) {
		.@remain = (Supply_LastUseTick + .Cooldown) - .@now_tick;
		dispbottom needtr("[자동 보충] ","[Auto Supply] You can use it again in ") + .@remain + needtr("초 후 다시 사용할 수 있습니다."," seconds.");
		end;
	}
	Supply_LastUseTick = .@now_tick;

	.@success = 0;
	.@fail = 0;
	.@skip = 0;

	dispbottom needtr("[자동 보충] ","[Auto Supply] ") + .@name$ + needtr(" 보충을 시작합니다."," is now being supplied.");

	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		.@item_id = getd("Supply_Item_" + .@preset + "_" + .@i);
		.@target = getd("Supply_Amount_" + .@preset + "_" + .@i);

		if (.@item_id <= 0 || .@target <= 0)
			continue;


		.@now = countitem(.@item_id);
		.@need = .@target - .@now;

		if (.@need <= 0) {
			.@skip++;
			continue;
		}

		.@ret = storagegetitem(.@item_id, .@need);

		if (.@ret == 1) {
			dispbottom needtr("[보충 완료] ","[Supply Complete] ") + getitemname(.@item_id) + " " + .@need + needtr("개",".");
			.@success++;
		}
		else if (.@ret == -1) {
			dispbottom needtr("[자동 보충] 현재 창고에서 아이템을 꺼낼 수 없는 상태입니다.","[Auto Supply] Storage cannot be accessed right now.");
			dispbottom needtr("[자동 보충] 잠시 후 다시 시도해주세요.","[Auto Supply] Please try again shortly.");
			end;
		}
		else if (.@ret == -2) {
			dispbottom needtr("[자동 보충] 창고 창을 닫은 뒤 다시 사용해주세요.","[Auto Supply] Close the Storage window and try again.");
			end;
		}
		else {
			dispbottom needtr("[보충 실패] ","[Supply Failed] ") + getitemname(.@item_id) + " " + .@need + needtr("개",".");
			dispbottom needtr("[보충 실패] 창고 수량, 무게 또는 인벤토리 공간을 확인해주세요.","[Supply Failed] Check Storage stock, your weight and your inventory space.");
			.@fail++;
		}
	}

	Supply_LastPreset = .@preset;

	dispbottom needtr("[자동 보충] 완료: ","[Auto Supply] Supplied: ") + .@success + needtr("종 / 실패: "," / Failed: ") + .@fail + needtr("종 / 이미 충분: "," / Already enough: ") + .@skip + needtr("종",".");
	end;
}
