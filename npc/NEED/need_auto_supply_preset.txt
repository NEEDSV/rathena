-	script	SupplyPresetSystem	FAKE_NPC,{
	end;

OnInit:
	bindatcmd "자동보충", strnpcinfo(3)+"::OnSupplySetting", 0, 99;
	bindatcmd "보충", strnpcinfo(3)+"::OnSupplyRecent", 0, 99;
	bindatcmd "보충1", strnpcinfo(3)+"::OnSupply1", 0, 99;
	bindatcmd "보충2", strnpcinfo(3)+"::OnSupply2", 0, 99;
	bindatcmd "보충3", strnpcinfo(3)+"::OnSupply3", 0, 99;

	.MaxPreset = 3;
	.MaxSlot = 10;

	// 프리셋 2: 캐시 해금
	.UnlockCash2 = 50000;

	// 프리셋 3: 특정 아이템 해금
	// 아래 아이템 ID와 필요 수량을 서버 설정에 맞게 변경하세요.
	.UnlockItem3 = 399907;
	.UnlockItem3Amount = 1;

	.Cooldown = 10;
	end;

OnSupplySetting:
	mes "[자동 보충 세팅]";
	mes "설정할 프리셋을 선택하세요.";
	next;

	.@name1$ = Supply_Name_1$;
	if (.@name1$ == "")
		.@name1$ = "프리셋 1";
	.@menu1$ = .@name1$ + " [무료]";

	.@name2$ = Supply_Name_2$;
	if (.@name2$ == "")
		.@name2$ = "프리셋 2";
	if (Supply_Unlocked_2)
		.@menu2$ = .@name2$;
	else
		.@menu2$ = .@name2$ + " [캐시 해금]";

	.@name3$ = Supply_Name_3$;
	if (.@name3$ == "")
		.@name3$ = "프리셋 3";
	if (Supply_Unlocked_3)
		.@menu3$ = .@name3$;
	else
		.@menu3$ = .@name3$ + " [아이템 해금]";

	.@sel = select(
		.@menu1$,
		.@menu2$,
		.@menu3$,
		"자동 보충 도움말"
	);

	if (.@sel == 4) {
		callsub S_Help;
		end;
	}

	.@preset = .@sel;

	if (!callsub(S_CheckPresetAccess, .@preset, 1))
		end;

	callsub S_EditPreset, .@preset;
	end;

OnSupplyRecent:
	.@preset = Supply_LastPreset;

	if (.@preset < 1 || .@preset > .MaxPreset) {
		dispbottom "[자동 보충] 최근 사용한 프리셋이 없습니다.";
		dispbottom "[자동 보충] 먼저 @보충1, @보충2, @보충3 중 하나를 사용해주세요.";
		end;
	}

	if (!callsub(S_CheckPresetAccess, .@preset, 0))
		end;

	callsub S_Supply, .@preset;
	end;

OnSupply1:
	callsub S_Supply, 1;
	end;

OnSupply2:
	if (!callsub(S_CheckPresetAccess, 2, 0))
		end;

	callsub S_Supply, 2;
	end;

OnSupply3:
	if (!callsub(S_CheckPresetAccess, 3, 0))
		end;

	callsub S_Supply, 3;
	end;

// =========================
// 도움말
// =========================
S_Help:
	mes "[자동 보충 도움말]";
	mes "자동 보충은 설정한 목표 수량만큼";
	mes "부족한 아이템을 창고에서 자동으로 꺼내주는 편의 기능입니다.";
	next;

	mes "[사용 방법]";
	mes "1. @자동보충 입력";
	mes "2. 프리셋 선택";
	mes "3. 아이템 등록 및 목표 수량 설정";
	mes "4. @보충1, @보충2, @보충3 사용";
	mes "5. 이후 @보충으로 최근 프리셋 재사용";
	next;

	mes "[예시]";
	mes "파리의 날개 목표 수량 100개";
	mes "현재 인벤토리 35개 보유";
	mes "@보충1 입력 시 창고에서 65개를 자동 보충합니다.";
	next;

	mes "[프리셋 안내]";
	mes "@보충1 : 기본 무료 프리셋";
	mes "@보충2 : 캐시로 해금";
	mes "@보충3 : 지정 아이템으로 해금";
	mes "@보충 : 최근 사용 프리셋 실행";
	next;

	mes "[주의 사항]";
	mes "자동 보충은 마을에서만 사용할 수 있습니다.";
	mes "창고 수량, 무게 또는 인벤토리 공간이 부족하면";
	mes "일부 아이템 보충에 실패할 수 있습니다.";
	mes "실패한 아이템은 채팅창에 안내됩니다.";
	return;

// =========================
// 프리셋 편집
// =========================
S_EditPreset:
	.@preset = getarg(0);

	while (1) {
		.@name$ = getd("Supply_Name_" + .@preset + "$");

		if (.@name$ == "")
			.@name$ = "프리셋 " + .@preset;

		mes "[자동 보충 세팅]";
		mes "현재 프리셋 : ^0000FF" + .@name$ + "^000000";
		next;

		.@sel = select(
			"프리셋 이름 수정",
			"아이템 등록",
			"등록 목록 보기",
			"목표 수량 변경 / 삭제",
			"다른 프리셋에서 복사",
			"프리셋 초기화",
			"종료"
		);

		switch (.@sel) {
		case 1:
			callsub S_ChangeName, .@preset;
			break;

		case 2:
			callsub S_RegisterItem, .@preset;
			break;

		case 3:
			callsub S_ShowPreset, .@preset;
			break;

		case 4:
			callsub S_EditAmount, .@preset;
			break;

		case 5:
			callsub S_CopyPreset, .@preset;
			break;

		case 6:
			callsub S_ClearPreset, .@preset;
			break;

		default:
			close;
		}
	}
	end;

// =========================
// 프리셋 이름 수정
// =========================
S_ChangeName:
	.@preset = getarg(0);

	mes "[프리셋 이름 수정]";
	mes "새 이름을 입력하세요.";
	mes "예: 사냥 세팅, 전장 세팅, MVP 세팅";
	mes "':' 및 '^' 문자는 사용할 수 없습니다.";
	next;

	.@input_ret = input(.@name$, 1, 12);

	if (.@input_ret != 0) {
		mes "[프리셋 이름 수정]";
		mes "프리셋 이름은 1~12자로 입력해주세요.";
		next;
		return;
	}

	if (.@name$ == "") {
		mes "이름을 입력하지 않았습니다.";
		next;
		return;
	}

	if (strpos(.@name$, ":") != -1 || strpos(.@name$, "^") != -1) {
		mes "[프리셋 이름 수정]";
		mes "사용할 수 없는 문자가 포함되어 있습니다.";
		mes "':' 및 '^' 문자를 제외하고 입력해주세요.";
		next;
		return;
	}

	setd "Supply_Name_" + .@preset + "$", .@name$;

	mes "프리셋 이름이 변경되었습니다.";
	mes "새 이름 : ^0000FF" + .@name$ + "^000000";
	next;
	return;

// =========================
// 아이템 등록
// =========================
S_RegisterItem:
	.@preset = getarg(0);

	.@free_slot = -1;
	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		if (getd("Supply_Item_" + .@preset + "_" + .@i) == 0) {
			.@free_slot = .@i;
			break;
		}
	}

	if (.@free_slot < 0) {
		mes "[아이템 등록]";
		mes "이 프리셋은 최대 " + .MaxSlot + "종까지 등록할 수 있습니다.";
		next;
		return;
	}

	getinventorylist();

	.@menu$ = "";
	.@count = 0;

	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		.@id = @inventorylist_id[.@i];

		if (.@id <= 0)
			continue;

		.@type = getiteminfo(.@id, ITEMINFO_TYPE);

		// 소비템 / 기타 / 탄약 허용
		if (.@type != IT_HEALING &&
			.@type != IT_USABLE &&
			.@type != IT_ETC &&
			.@type != IT_AMMO)
			continue;

		// 이미 프리셋에 등록된 아이템 제외
		.@exists = 0;
		for (.@j = 0; .@j < .MaxSlot; .@j++) {
			if (getd("Supply_Item_" + .@preset + "_" + .@j) == .@id) {
				.@exists = 1;
				break;
			}
		}

		if (.@exists)
			continue;

		// 인벤토리 내 동일 아이템 중복 표시 방지
		.@listed = 0;
		for (.@j = 0; .@j < .@count; .@j++) {
			if (.@item[.@j] == .@id) {
				.@listed = 1;
				break;
			}
		}

		if (.@listed)
			continue;

		.@menu$ += getitemname(.@id) + " [" + countitem(.@id) + "개 보유]:";
		.@item[.@count] = .@id;
		.@count++;
	}

	if (.@count == 0) {
		mes "[아이템 등록]";
		mes "등록 가능한 아이템이 없습니다.";
		next;
		return;
	}

	mes "[아이템 등록]";
	mes "등록할 아이템을 선택하세요.";
	next;

	.@sel = select(.@menu$);
	.@item_id = .@item[.@sel - 1];

	mes "[" + getitemname(.@item_id) + "]";
	mes "몇 개까지 자동 보충할까요?";
	next;

	input .@amount, 1, 30000;

	if (.@amount <= 0) {
		mes "잘못된 수량입니다.";
		next;
		return;
	}

	setd "Supply_Item_" + .@preset + "_" + .@free_slot, .@item_id;
	setd "Supply_Amount_" + .@preset + "_" + .@free_slot, .@amount;

	mes "등록 완료!";
	mes getitemname(.@item_id) + " : " + .@amount + "개";
	next;
	return;

// =========================
// 등록 목록 보기
// =========================
S_ShowPreset:
	.@preset = getarg(0);
	.@name$ = getd("Supply_Name_" + .@preset + "$");

	if (.@name$ == "")
		.@name$ = "프리셋 " + .@preset;

	mes "[등록 목록]";
	mes "^0000FF" + .@name$ + "^000000";

	.@has = 0;

	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		.@item_id = getd("Supply_Item_" + .@preset + "_" + .@i);
		.@amount = getd("Supply_Amount_" + .@preset + "_" + .@i);

		if (.@item_id <= 0)
			continue;

		mes "- " + getitemname(.@item_id) + " : " + .@amount + "개";
		.@has = 1;
	}

	if (!.@has)
		mes "등록된 아이템이 없습니다.";

	next;
	return;

// =========================
// 목표 수량 변경 / 0 입력 시 삭제
// =========================
S_EditAmount:
	.@preset = getarg(0);

	.@menu$ = "";
	.@count = 0;

	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		.@item_id = getd("Supply_Item_" + .@preset + "_" + .@i);
		.@amount = getd("Supply_Amount_" + .@preset + "_" + .@i);

		if (.@item_id <= 0)
			continue;

		.@menu$ += getitemname(.@item_id) + " [" + .@amount + "개]:";
		.@slot[.@count] = .@i;
		.@count++;
	}

	if (.@count == 0) {
		mes "[목표 수량 변경]";
		mes "변경할 아이템이 없습니다.";
		next;
		return;
	}

	mes "[목표 수량 변경]";
	mes "변경할 아이템을 선택하세요.";
	next;

	.@sel = select(.@menu$);
	.@idx = .@slot[.@sel - 1];
	.@item_id = getd("Supply_Item_" + .@preset + "_" + .@idx);
	.@old_amount = getd("Supply_Amount_" + .@preset + "_" + .@idx);

	mes "[" + getitemname(.@item_id) + "]";
	mes "현재 목표 수량 : " + .@old_amount + "개";
	mes "새 목표 수량을 입력하세요.";
	mes "0을 입력하면 등록 목록에서 삭제됩니다.";
	next;

	input .@amount, 0, 30000;

	if (.@amount == 0) {
		setd "Supply_Item_" + .@preset + "_" + .@idx, 0;
		setd "Supply_Amount_" + .@preset + "_" + .@idx, 0;

		mes "[목표 수량 변경]";
		mes getitemname(.@item_id) + " 아이템이 삭제되었습니다.";
		next;
		return;
	}

	setd "Supply_Amount_" + .@preset + "_" + .@idx, .@amount;

	mes "[목표 수량 변경]";
	mes getitemname(.@item_id) + "의 목표 수량이 변경되었습니다.";
	mes .@old_amount + "개 → " + .@amount + "개";
	next;
	return;

// =========================
// 다른 프리셋의 아이템/수량 복사
// 대상 프리셋의 이름은 유지
// =========================
S_CopyPreset:
	.@target = getarg(0);

	mes "[프리셋 복사]";
	mes "복사할 원본 프리셋을 선택하세요.";
	mes "현재 프리셋의 이름은 유지되며,";
	mes "등록 아이템과 목표 수량만 덮어씁니다.";
	next;

	.@sel = select("프리셋 1", "프리셋 2", "프리셋 3", "취소");

	if (.@sel == 4)
		return;

	.@source = .@sel;

	if (.@source == .@target) {
		mes "[프리셋 복사]";
		mes "현재 프리셋과 같은 프리셋은 선택할 수 없습니다.";
		next;
		return;
	}

	if (!callsub(S_CheckPresetAccess, .@source, 0)) {
		mes "[프리셋 복사]";
		mes "해금되지 않은 프리셋은 복사할 수 없습니다.";
		next;
		return;
	}

	mes "[프리셋 복사]";
	mes "현재 프리셋의 등록 내용이 모두 교체됩니다.";
	mes "계속하시겠습니까?";
	next;

	if (select("취소", "복사하기") == 1)
		return;

	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		.@item_id = getd("Supply_Item_" + .@source + "_" + .@i);
		.@amount = getd("Supply_Amount_" + .@source + "_" + .@i);

		setd "Supply_Item_" + .@target + "_" + .@i, .@item_id;
		setd "Supply_Amount_" + .@target + "_" + .@i, .@amount;
	}

	mes "[프리셋 복사]";
	mes "프리셋 " + .@source + "의 등록 내용을 복사했습니다.";
	next;
	return;

// =========================
// 프리셋 초기화
// =========================
S_ClearPreset:
	.@preset = getarg(0);

	mes "[프리셋 초기화]";
	mes "정말 초기화하시겠습니까?";
	next;

	if (select("취소", "초기화") == 1)
		return;

	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		setd "Supply_Item_" + .@preset + "_" + .@i, 0;
		setd "Supply_Amount_" + .@preset + "_" + .@i, 0;
	}

	setd "Supply_Name_" + .@preset + "$", "";

	if (Supply_LastPreset == .@preset)
		Supply_LastPreset = 0;

	mes "초기화되었습니다.";
	next;
	return;

// =========================
// 프리셋 접근 체크
// getarg(0): preset
// getarg(1): setting mode, 1이면 해금 안내창 출력
// 해금 여부는 캐릭터 단위로 저장
// =========================
S_CheckPresetAccess:
	.@preset = getarg(0);
	.@setting = getarg(1);

	if (.@preset == 1)
		return 1;

	if (getd("Supply_Unlocked_" + .@preset))
		return 1;

	if (!.@setting) {
		dispbottom "[자동 보충] 프리셋 " + .@preset + "은 아직 해금되지 않았습니다.";
		dispbottom "[자동 보충] @자동보충에서 먼저 해금해주세요.";
		return 0;
	}

	// 프리셋 2: 캐시 해금
	if (.@preset == 2) {
		mes "[프리셋 해금]";
		mes "프리셋 2는 캐시 해금이 필요합니다.";
		mes "해금 비용 : ^FF0000" + .UnlockCash2 + " 캐시^000000";
		next;

		if (select("취소", "해금하기") == 1)
			return 0;

		if (#CASHPOINTS < .UnlockCash2) {
			mes "[프리셋 해금]";
			mes "캐시가 부족합니다.";
			mes "필요 캐시 : " + .UnlockCash2;
			mes "보유 캐시 : " + #CASHPOINTS;
			next;
			return 0;
		}

		#CASHPOINTS -= .UnlockCash2;
		Supply_Unlocked_2 = 1;

		mes "[프리셋 해금]";
		mes "프리셋 2가 해금되었습니다.";
		next;
		return 1;
	}

	// 프리셋 3: 특정 아이템 해금
	if (.@preset == 3) {
		if (.UnlockItem3 <= 0 || .UnlockItem3Amount <= 0) {
			mes "[프리셋 해금]";
			mes "프리셋 3 해금 아이템이 설정되지 않았습니다.";
			mes "관리자에게 문의해주세요.";
			next;
			return 0;
		}

		.@item_name$ = getitemname(.UnlockItem3);
		.@owned = countitem(.UnlockItem3);

		mes "[프리셋 해금]";
		mes "프리셋 3는 지정 아이템 해금이 필요합니다.";
		mes "필요 아이템 : ^0000FF" + .@item_name$ + "^000000";
		mes "필요 수량 : ^FF0000" + .UnlockItem3Amount + "개^000000";
		mes "보유 수량 : " + .@owned + "개";
		mes "^FF0000해당 아이템은 전장에서 전장포인트로 구매하실 수 있습니다.^000000";
		next;

		if (select("취소", "아이템 사용") == 1)
			return 0;

		// 확인창 이후 수량을 다시 검사하여 중복 처리나 수량 변동을 방지
		if (countitem(.UnlockItem3) < .UnlockItem3Amount) {
			mes "[프리셋 해금]";
			mes "해금 아이템이 부족합니다.";
			mes "필요 아이템 : " + .@item_name$ + " " + .UnlockItem3Amount + "개";
			mes "보유 수량 : " + countitem(.UnlockItem3) + "개";
			next;
			return 0;
		}

		delitem .UnlockItem3, .UnlockItem3Amount;
		Supply_Unlocked_3 = 1;

		mes "[프리셋 해금]";
		mes "프리셋 3가 해금되었습니다.";
		mes .@item_name$ + " " + .UnlockItem3Amount + "개를 사용했습니다.";
		next;
		return 1;
	}

	return 0;

// =========================
// 자동 보충 실행
// =========================
S_Supply:
	.@preset = getarg(0);

	// 마을에서만 사용 가능
	.@map$ = strcharinfo(3);

	if (!getmapflag(.@map$, mf_town) && .@map$ != "bat_room") {
		dispbottom "[자동 보충] 자동 보충은 마을 또는 전장 대기실에서만 사용할 수 있습니다.";
		end;
	}

	.@has_item = 0;
	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		if (getd("Supply_Item_" + .@preset + "_" + .@i) > 0 &&
			getd("Supply_Amount_" + .@preset + "_" + .@i) > 0) {
			.@has_item = 1;
			break;
		}
	}

	.@name$ = getd("Supply_Name_" + .@preset + "$");
	if (.@name$ == "")
		.@name$ = "프리셋 " + .@preset;

	if (!.@has_item) {
		dispbottom "[자동 보충] " + .@name$ + "에 등록된 아이템이 없습니다.";
		end;
	}

	// 짧은 재사용 대기시간
	.@now_tick = gettimetick(2);
	if (Supply_LastUseTick > 0 && .@now_tick < Supply_LastUseTick + .Cooldown) {
		.@remain = (Supply_LastUseTick + .Cooldown) - .@now_tick;
		dispbottom "[자동 보충] " + .@remain + "초 후 다시 사용할 수 있습니다.";
		end;
	}
	Supply_LastUseTick = .@now_tick;

	.@success = 0;
	.@fail = 0;
	.@skip = 0;

	dispbottom "[자동 보충] " + .@name$ + " 보충을 시작합니다.";

	for (.@i = 0; .@i < .MaxSlot; .@i++) {
		.@item_id = getd("Supply_Item_" + .@preset + "_" + .@i);
		.@target = getd("Supply_Amount_" + .@preset + "_" + .@i);

		if (.@item_id <= 0 || .@target <= 0)
			continue;


		.@now = countitem(.@item_id);
		.@need = .@target - .@now;

		if (.@need <= 0) {
			.@skip++;
			continue;
		}

		.@ret = storagegetitem(.@item_id, .@need);

		if (.@ret == 1) {
			dispbottom "[보충 완료] " + getitemname(.@item_id) + " " + .@need + "개";
			.@success++;
		}
		else if (.@ret == -1) {
			dispbottom "[자동 보충] 현재 창고에서 아이템을 꺼낼 수 없는 상태입니다.";
			dispbottom "[자동 보충] 잠시 후 다시 시도해주세요.";
			end;
		}
		else if (.@ret == -2) {
			dispbottom "[자동 보충] 창고 창을 닫은 뒤 다시 사용해주세요.";
			end;
		}
		else {
			dispbottom "[보충 실패] " + getitemname(.@item_id) + " " + .@need + "개";
			dispbottom "[보충 실패] 창고 수량, 무게 또는 인벤토리 공간을 확인해주세요.";
			.@fail++;
		}
	}

	Supply_LastPreset = .@preset;

	dispbottom "[자동 보충] 완료: " + .@success + "종 / 실패: " + .@fail + "종 / 이미 충분: " + .@skip + "종";
	end;
}
