// F_NeedFishDrawReelRounds() : 릴 감기 횟수 독립 추첨(어종/희귀도/등급/풍어 무관). 반환 0~3.
//  릴0 40% / 릴1 35% / 릴2 20% / 릴3 5% (합계 100). 최종 스크립트에서 별도 참조.
function	script	F_NeedFishDrawReelRounds	{
	.@r = rand(10000);
	if ( .@r < 4000 ) return 0;
	if ( .@r < 7500 ) return 1;
	if ( .@r < 9500 ) return 2;
	return 3;
}

// F_NeedFishDraw(<spot_id>) : 희귀도/어종/길이/무게 추첨. 반환: 물고기 ID.
//  부가 결과는 $@nff_out_fish / $@nff_out_name$ / $@nff_out_rarity / $@nff_out_len / $@nff_out_wt.
//  릴 횟수/반응 등급 등은 결과에 영향을 주지 않는다.
function	script	F_NeedFishDraw	{
	.@spot = getarg(0);
	// 희귀도 기본 가중치 (합계 10000)
	.@rw1 = 8000; .@rw2 = 1700; .@rw3 = 280; .@rw4 = 20;
	// 풍어 배율은 희귀/전설에만 적용 (일반/고급 가중치 유지)
	.@fever = $NEED_FISHING_FEVER_RATE; if ( .@fever <= 0 ) .@fever = 10000;
	.@rw3 = .@rw3 * .@fever / 10000;
	.@rw4 = .@rw4 * .@fever / 10000;

	.@tot = .@rw1 + .@rw2 + .@rw3 + .@rw4;
	.@r = rand(.@tot);
	if ( .@r < .@rw1 ) .@rar = 1;
	else if ( .@r < .@rw1 + .@rw2 ) .@rar = 2;
	else if ( .@r < .@rw1 + .@rw2 + .@rw3 ) .@rar = 3;
	else .@rar = 4;

	// 해당 희귀도 + 낚시터 출현 후보 수집
	.@cn = 0; .@sum = 0;
	for ( .@i = 0; .@i < $@nff_count; .@i++ ) {
		if ( !$@nff_active[.@i] ) continue;
		.@sw = callfunc( "F_NeedFishSpotWeight", .@i, .@spot );
		if ( .@sw <= 0 ) continue;
		if ( $@nff_rarity[.@i] != .@rar ) continue;
		.@cand[.@cn] = .@i; .@candw[.@cn] = .@sw; .@sum += .@sw; .@cn++;
	}
	if ( .@cn == 0 ) {
		.@sum = 0;
		for ( .@i = 0; .@i < $@nff_count; .@i++ ) {
			if ( !$@nff_active[.@i] ) continue;
			.@sw = callfunc( "F_NeedFishSpotWeight", .@i, .@spot );
			if ( .@sw <= 0 ) continue;
			.@cand[.@cn] = .@i; .@candw[.@cn] = .@sw; .@sum += .@sw; .@cn++;
		}
	}
	if ( .@cn == 0 || .@sum <= 0 ) { $@nff_out_fish = 0; return 0; }

	.@pick = rand(.@sum); .@acc = 0; .@sel = .@cand[.@cn-1];
	for ( .@j = 0; .@j < .@cn; .@j++ ) { .@acc += .@candw[.@j]; if ( .@pick < .@acc ) { .@sel = .@cand[.@j]; break; } }

	// 크기 롤링 (0~10000) - 길이 기반 무게 편차
	.@roll = rand(0,10000);
	.@minl = $@nff_minlen[.@sel]; .@maxl = $@nff_maxlen[.@sel];
	.@len = .@minl + ( .@maxl - .@minl ) * .@roll / 10000;
	.@curve = .@roll * .@roll / 10000;
	.@minw = $@nff_minwt[.@sel]; .@maxw = $@nff_maxwt[.@sel];
	.@wt = .@minw + ( .@maxw - .@minw ) * .@curve / 10000;
	.@body = rand(9500,10500);
	.@wt = .@wt * .@body / 10000;
	if ( .@len < .@minl ) .@len = .@minl; if ( .@len > .@maxl ) .@len = .@maxl;
	if ( .@wt < .@minw ) .@wt = .@minw;   if ( .@wt > .@maxw ) .@wt = .@maxw;

	$@nff_out_fish   = $@nff_id[.@sel];
	$@nff_out_name$  = $@nff_name$[.@sel];
	$@nff_out_rarity = $@nff_rarity[.@sel];
	$@nff_out_len    = .@len;
	$@nff_out_wt     = .@wt;
	return $@nff_id[.@sel];
}

// F_NeedFishSpotWeight(<fish_index>, <spot_id>) : 낚시터별 출현 가중치(0=미출현). 현재 낚시터 1종 공용.
function	script	F_NeedFishSpotWeight	{
	.@i = getarg(0);
	return $@nff_spot1[.@i];
}

// F_NeedFishById(<fish_id>) : 특정 어종 강제 지정(강제/테스트용). 반환: 물고기 ID(없으면 0). 등급 보정 없음.
function	script	F_NeedFishById	{
	.@fid = getarg(0);
	.@sel = -1;
	for ( .@i = 0; .@i < $@nff_count; .@i++ ) { if ( $@nff_id[.@i] == .@fid ) { .@sel = .@i; break; } }
	if ( .@sel < 0 ) { $@nff_out_fish = 0; return 0; }
	.@roll = rand(0,10000);
	.@minl = $@nff_minlen[.@sel]; .@maxl = $@nff_maxlen[.@sel];
	.@len = .@minl + ( .@maxl - .@minl ) * .@roll / 10000;
	.@curve = .@roll * .@roll / 10000;
	.@minw = $@nff_minwt[.@sel]; .@maxw = $@nff_maxwt[.@sel];
	.@wt = .@minw + ( .@maxw - .@minw ) * .@curve / 10000;
	.@body = rand(9500,10500);
	.@wt = .@wt * .@body / 10000;
	if ( .@len < .@minl ) .@len = .@minl; if ( .@len > .@maxl ) .@len = .@maxl;
	if ( .@wt < .@minw ) .@wt = .@minw;   if ( .@wt > .@maxw ) .@wt = .@maxw;
	$@nff_out_fish   = $@nff_id[.@sel];
	$@nff_out_name$  = $@nff_name$[.@sel];
	$@nff_out_rarity = $@nff_rarity[.@sel];
	$@nff_out_len    = .@len;
	$@nff_out_wt     = .@wt;
	return $@nff_id[.@sel];
}

}

// ===== 어종 조회/도감 지원 함수 (fish_id 기반, 향후 도감 UI 재사용) =====
// F_NeedFishIndexById(<fish_id>) : 반환 배열 인덱스(없으면 -1)
function	script	F_NeedFishIndexById	{
	.@fid = getarg(0);
	for ( .@i = 0; .@i < $@nff_count; .@i++ ) { if ( $@nff_id[.@i] == .@fid ) return .@i; }
	return -1;
}
// F_NeedFishNameById(<fish_id>) : 어종 이름(없으면 "?")
function	script	F_NeedFishNameById	{
	.@i = callfunc( "F_NeedFishIndexById", getarg(0) );
	if ( .@i < 0 ) return "?";
	return $@nff_name$[.@i];
}
// F_NeedFishRarityById(<fish_id>) : 희귀도(1~4, 없으면 0)
function	script	F_NeedFishRarityById	{
	.@i = callfunc( "F_NeedFishIndexById", getarg(0) );
	if ( .@i < 0 ) return 0;
	return $@nff_rarity[.@i];
}
// F_NeedFishCutinRes(<fish_id>) : cutin 리소스 키(없으면 "")
function	script	F_NeedFishCutinRes	{
	.@i = callfunc( "F_NeedFishIndexById", getarg(0) );
	if ( .@i < 0 ) return "";
	return $@nff_cutin$[.@i];
}
// F_NeedFishShowCutin(<fish_id>) : 어종 cutin 표시(1=표시, 0=실패). 향후 도감 UI에서도 그대로 호출.
//  제거는 호출자가 cutin "",255; 로 처리한다(여기서는 표시만 담당).
function	script	F_NeedFishShowCutin	{
	.@res$ = callfunc( "F_NeedFishCutinRes", getarg(0) );
	if ( .@res$ == "" ) return 0;
	cutin .@res$, 2;
	return 1;
}
