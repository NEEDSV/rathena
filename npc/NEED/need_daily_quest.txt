// 구간 끝 레벨 경고 (129 / 149 / 174)
function	script	F_NEED_DQ_EdgeWarning	{
	if (BaseLevel != 129 && BaseLevel != 149 && BaseLevel != 174)
		return;
	callfunc "F_NEED_DQ_M", needtr("^FF0000레벨업으로 구간이 바뀌면", "^FF0000If you level into a new range,");
	callfunc "F_NEED_DQ_M", needtr("납품 안 한 재료는 점수가 안 됩니다.", "undelivered materials will not score.");
	callfunc "F_NEED_DQ_M", needtr("단계마다 먼저 납품하세요.^000000", "Deliver at each goal first.^000000");
	return;
}

// 요약판 한 화면: 날짜·트랙·점수 막대·보상 구간·활동별 점수
// 활동 상한: 사냥 50(Lv175 40), 납품 30(Lv175 20), 전장 50(Lv150 이상, 선택), 인던 40(Lv175, 선택). 합계 80.
// 1단계에는 사냥·전장·인던 연동이 없어 "준비 중"으로 보인다.
function	script	F_NEED_DQ_Board	{
	.@n$ = getarg(0);
	.@wday$ = getarg(1);
	.@t = @dq_locked_track ? @dq_locked_track : @dq_band;
	.@tot = @dq_acc_total;

	callfunc "F_NEED_DQ_M", .@n$;
	if (@dq_locked_track)
		callfunc "F_NEED_DQ_M", @dq_day$ + " (" + .@wday$ + ") " + callfunc("F_NEED_DQ_Range", @dq_locked_min, @dq_locked_max) + needtr(" 고정", " locked");
	else
		callfunc "F_NEED_DQ_M", @dq_day$ + " (" + .@wday$ + ") " + callfunc("F_NEED_DQ_Range", @dq_band_min, @dq_band_max) + needtr(" 미고정", " open");
	callfunc "F_NEED_DQ_M", needtr("오늘 점수 ^0055FF", "Today ^0055FF") + .@tot + "^000000 / 80  (" + (.@tot * 100 / 80) + "%)";
	.@fill = .@tot * 12 / 80;
	.@bar$ = "^00AA00";
	for (.@i = 0; .@i < 12; .@i++) {
		if (.@i == .@fill)
			.@bar$ += "^BBBBBB";
		.@bar$ += needtr("■", "#");
	}
	callfunc "F_NEED_DQ_M", .@bar$ + "^000000";
	.@line$ = needtr("보상 ", "Rewards ");
	for (.@k = 20; .@k <= 80; .@k += 20)
		.@line$ += (.@tot >= .@k ? "^00AA00" + needtr("★", "*") : "^BBBBBB" + needtr("○", "o")) + .@k + "^000000 ";
	callfunc "F_NEED_DQ_M", .@line$;
	callfunc "F_NEED_DQ_M", needtr("────────────────", "--------------------------------");

	.@c = 0;
	.@L$[.@c] = needtr("몬스터 사냥  ", "Hunting     ") + @dq_acc_hunt + " / " + (.@t == 175 ? 40 : 50) + needtr("점", " pts"); .@c++;
	if (@dq_hunt_track)
		{ .@L$[.@c] = needtr("^999999 └ 내 사냥 ", "^999999  mine ") + @dq_hunt_count + " / " + @dq_hunt_goal + needtr("마리^000000", "^000000"); .@c++; }
	else
		{ .@L$[.@c] = needtr("^999999 └ 아직 수락하지 않음^000000", "^999999  not accepted yet^000000"); .@c++; }
	.@L$[.@c] = needtr("보급품 납품  ", "Delivery    ") + @dq_acc_delivery + " / " + (.@t == 175 ? 20 : 30) + needtr("점", " pts"); .@c++;
	if (@dq_track)
		{ .@L$[.@c] = needtr("^999999 └ 내 납품 ", "^999999  mine ") + @dq_submitted + " / " + @dq_goal + (@dq_collected ? " (+" + @dq_collected + ")" : "") + "^000000"; .@c++; }
	else
		{ .@L$[.@c] = needtr("^999999 └ 아직 수락하지 않음^000000", "^999999  not accepted yet^000000"); .@c++; }
	if (.@t >= 150) {
		{ .@L$[.@c] = needtr("전장 (선택)  ", "Battleground ") + @dq_acc_bg + needtr(" / 50점", " / 50 pts"); .@c++; }
		if (@dq_track)
			{ .@L$[.@c] = needtr("^999999 └ 계정 ", "^999999  account ") + @dq_bg_acc_games + " / " + @dq_bg_goal + needtr("판, 이 캐릭터 ", " games, this char ") + @dq_bg_games + needtr("판^000000", "^000000"); .@c++; }
	}
	if (@dq_inst_cap) {
		{ .@L$[.@c] = needtr("인던 (선택)  ", "Instances   ") + @dq_acc_inst + " / " + @dq_inst_cap + needtr("점", " pts"); .@c++; }
		if (@dq_track)
			{ .@L$[.@c] = needtr("^999999 └ 이 캐릭터 클리어 ", "^999999  this char cleared ") + @dq_inst_pts + " / " + @dq_inst_cap + needtr("점^000000", " pts^000000"); .@c++; }
	}
	if (getgmlevel())
		{ .@L$[.@c] = needtr("^777777GM 처치는 집계되지 않음^000000", "^777777GM kills do not count^000000"); .@c++; }
	if (@dq_reset_soon)
		{ .@L$[.@c] = needtr("^FF0000곧 04:00 초기화: 납품·보상은 지금!^000000", "^FF0000Reset at 04:00 soon: deliver and claim now!^000000"); .@c++; }
		{ .@L$[.@c] = needtr("^FF0000(04:00 뒤 끝난 전장·인던은 새날 수락)^000000", "^FF0000(BG/instance ends after 04:00: accept again)^000000"); .@c++; }
	for (.@i = 0; .@i < .@c; .@i++)
		callfunc "F_NEED_DQ_M", .@L$[.@i];
	return;
}

// 일일 보급 납품 대화. 보급관 NPC 와 [!] 버튼(NEED_NOTICE_BTN)이 같은 대화를 연다.
function	script	F_NEED_DQ_Talk	{
	@dq_ln = 0;
	.@n$ = needtr("[일일 보급관]", "[Supply Officer]");
	setarray .@wd$[0], needtr("일", "Sun"), needtr("월", "Mon"), needtr("화", "Tue"), needtr("수", "Wed"), needtr("목", "Thu"), needtr("금", "Fri"), needtr("토", "Sat");

	.@s = needdq_status();
	if (.@s == -8 || .@s == -7) {
		callfunc "F_NEED_DQ_M", .@n$;
		callfunc "F_NEED_DQ_M", needtr("지금은 일일 퀘스트를", "The daily quest");
		callfunc "F_NEED_DQ_M", needtr("받지 않습니다.", "is closed right now.");
		callfunc "F_NEED_DQ_Fit";
		close;
	}
	if (.@s == 5) {
		callfunc "F_NEED_DQ_M", .@n$;
		callfunc "F_NEED_DQ_M", needtr("Lv100 이상만", "The daily quest requires");
		callfunc "F_NEED_DQ_M", needtr("일일 퀘스트에 참여할 수 있습니다.", "Lv100 or higher.");
		callfunc "F_NEED_DQ_Fit";
		close;
	}

	// 첫 화면: 요약판 (2026-10-21 운영자 요청, UI 콘셉트의 텍스트판)
	callfunc "F_NEED_DQ_Board", .@n$, .@wd$[@dq_wday];

	switch (.@s) {
	case 0:
		goto L_Offer;
	case 3:
		goto L_LockedOut;
	case 4:
		goto L_LevelChanged;
	}
	goto L_Progress;

L_Offer:
	callfunc "F_NEED_DQ_Fit";
	next;
	if (@dq_locked_track && @dq_locked_track != @dq_band) {
		callfunc "F_NEED_DQ_M", .@n$;
		callfunc "F_NEED_DQ_M", needtr("오늘 이 계정은 ", "Account locked today to ") + callfunc("F_NEED_DQ_Range", @dq_locked_min, @dq_locked_max);
		callfunc "F_NEED_DQ_M", needtr("구간으로 고정되어", "and this character is in");
		callfunc "F_NEED_DQ_M", needtr("이 캐릭터는 수락할 수 없습니다.", "another range: cannot accept.");
		callfunc "F_NEED_DQ_M", needtr("(04:00 초기화)", "(Resets at 04:00)");
		callfunc "F_NEED_DQ_Fit";
		close;
	}
	callfunc "F_NEED_DQ_M", .@n$;
	callfunc "F_NEED_DQ_M", needtr("납품: ^0055FF", "Delivery: ^0055FF") + callfunc("F_NEED_DQ_MapName", @dq_today_map$) + "^000000";
	callfunc "F_NEED_DQ_M", needtr("재료: ", "Material: ") + callfunc("F_NEED_DQ_Material", @dq_band) + needtr(" (가상)", " (virtual)");
	callfunc "F_NEED_DQ_ShowTargets", @dq_today_map$;
	.@g$ = "";
	for (.@i = 0; .@i < @dq_stage_count; .@i++)
		.@g$ += (.@i ? " / " : "- ") + @dq_stage_amount[.@i] + needtr("개 ", "=") + @dq_stage_points[.@i] + needtr("점", "");
	callfunc "F_NEED_DQ_M", .@g$;
	callfunc "F_NEED_DQ_M", needtr("사냥: ^0055FF", "Hunting: ^0055FF") + callfunc("F_NEED_DQ_HuntArea", @dq_band) + "^000000";
	if (@dq_inst_kind_count) {
		callfunc "F_NEED_DQ_M", needtr("인던(선택): 합계 최대 ", "Instances (optional): up to ") + @dq_inst_cap + needtr("점", " pts");
		callfunc "F_NEED_DQ_M", "- " + callfunc("F_NEED_DQ_InstList");
	}
	if (@dq_bg_stage_count)
		callfunc "F_NEED_DQ_M", needtr("전장(선택): 유효 경기 ", "BG (optional): ") + @dq_bg_stage_amount[0] + needtr("판 ", " game = ") + @dq_bg_stage_points[0] + needtr("점 / ", " / ") + @dq_bg_stage_amount[@dq_bg_stage_count - 1] + needtr("판 ", " games = ") + @dq_bg_stage_points[@dq_bg_stage_count - 1] + needtr("점", "");
	callfunc "F_NEED_DQ_M", "- " + @dq_hunt_stage_amount[0] + needtr("마리 ", " = ") + @dq_hunt_stage_points[0] + needtr("점 ~ ", " pts ... ") + @dq_hunt_stage_amount[@dq_hunt_stage_count - 1] + needtr("마리 ", " = ") + @dq_hunt_stage_points[@dq_hunt_stage_count - 1] + needtr("점", " pts");
	if (!@dq_locked_track) {
		callfunc "F_NEED_DQ_M", needtr("^777777가장 먼저 점수를 얻은 캐릭터의", "^777777The first character to score locks");

			callfunc "F_NEED_DQ_M", needtr("고정되어 다시 수락할 수 없습니다.", "previous range today.");
			callfunc "F_NEED_DQ_M", needtr("(04:00 초기화)", "(Resets at 04:00)");
		} else {
			callfunc "F_NEED_DQ_M", needtr("지금은 다시 수락할 수 없습니다.", "Cannot re-accept right now.");
			callfunc "F_NEED_DQ_M", "(" + .@r + ")";
		}
		callfunc "F_NEED_DQ_Fit";
		close;
	}
	callfunc "F_NEED_DQ_Fit";
	close;

}

// 인던 이름 (점수 표시용 짧은 이름). kind 1 상급 고성, 2 일반 고성, 3 공중요새.

	case 1: return needtr("상급 고성", "OGH Hard");
	case 2: return needtr("고성", "OGH");
	case 3: return needtr("공중요새", "Sky Fortress");
	}
	return "?";
}

// "상급 고성 40 / 고성 15 / 공중요새 15"
function	script	F_NEED_DQ_InstList	{
	.@s$ = "";
	for (.@i = 0; .@i < @dq_inst_kind_count; .@i++)
		.@s$ += (.@i ? " / " : "") + callfunc("F_NEED_DQ_InstName", @dq_inst_kind_id[.@i]) + " " + @dq_inst_kind_points[.@i];
	return .@s$;
}

// 오늘 이 캐릭터가 클리어한 인던: "상급 고성, 공중요새" 또는 ""
function	script	F_NEED_DQ_InstDone	{
	.@s$ = "";
	for (.@i = 0; .@i < @dq_inst_kind_count; .@i++) {
		if (@dq_inst_kind_done[.@i])
			.@s$ += (.@s$ != "" ? ", " : "") + callfunc("F_NEED_DQ_InstName", @dq_inst_kind_id[.@i]);
	}
	return .@s$;
}

// 인던 최종 보스 처치 지점에서 호출: 보스 맵의 플레이어 각자를 attach 해서 needdq_inst_clear.
//   callfunc "F_NEED_DQ_InstClear", "<OGH_HARD|OGH|SKY_FORTRESS>", <보스 맵>;
// 같은 판 중복은 호출하는 쪽의 'NEED_DQ_INST_DONE, 같은 날 중복은 C++(캐릭터·종류당 1행)이 막는다.
function	script	F_NEED_DQ_InstClear	{
	.@kind$ = getarg(0);
	.@map$ = getarg(1);
	.@inst = instance_id();
	.@orig = playerattached();
	.@n = getmapunits(BL_PC, .@map$, .@aids);
	for (.@i = 0; .@i < .@n; .@i++) {
		if (.@aids[.@i] > 0 && attachrid(.@aids[.@i]))
			needdq_inst_clear .@kind$, .@inst;
	}
	detachrid;
	if (.@orig)
		attachrid .@orig;
	return .@n;
}
