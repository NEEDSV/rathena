lasagna,182,146,4	script	도작 도우미	648,{

	if( Class != Job_Rogue &&
		Class != Job_Stalker &&
		Class != Job_Shadow_Chaser_T &&
		Class != Job_Shadow_Chaser &&
		Class != Job_Abyss_Chaser ) {
		mes(.npc_name$);
		mes(sprintf("안녕하세요 ^0055FF%s^000000 님.", strcharinfo(PC_NAME)));
		mes("저는 ^FF0000도작 마스터^000000 입니다.");
		mes("\n");
		mes("죄송하지만 제 서비스는 <b>^0000FF로그, 스토커, 쉐도우 체이서, 어비스 체이서^000000</b>만 이용할 수 있습니다.");
		close();
	}

	while(true) {

		mes(.npc_name$);
		mes(sprintf("안녕하세요 ^0055FF%s^000000 님.", strcharinfo(PC_NAME)));
		mes("저는 <b>^FF0000도작 마스터^000000</b> 입니다.");
		mes("\n");
		.@zeny_text$ = callfunc("F_InsertComma", .skill_zeny);
		if (.@zeny_text$ == "")
			.@zeny_text$ = sprintf("%d", .skill_zeny);

		mes(.@zeny_text$ + "z를 지불하면 원하는 스킬을 복사할 수 있게 도와드립니다.");
		next();

		.@main = select(
			"스킬 복사",
			"현재 복사한 스킬 확인",
			"스킬 복사 초기화",
			"종료"
		);

		//--------------------------------
		// 슬롯 확인
		//--------------------------------
		if(.@main == 2) {
		
			mes("[현재 복사한 스킬]");
			mes(" ");

			//--------------------------------
			// 도작 (Plagiarism)
			//--------------------------------
			if( getskilllv(getskillid("RG_PLAGIARISM")) > 0 ) {

				.@id1 = getplagiarizedskill(1);
				.@lv1 = getplagiarizedskilllv(1);

				if( .@id1 > 0 ) {
					mes("^0000FF[도작 슬롯]^000000");

					.@skill_name$ = getskilldesc(.@id1);
					if( .@skill_name$ == "" )
						.@skill_name$ = "알 수 없는 스킬 ID " + .@id1;

					mes("<b>" + .@skill_name$ + "</b> ^AA0000Lv." + .@lv1 + "^000000");
				}
				else {
					mes("^0000FF[도작 슬롯]^000000");
					mes("비어 있습니다.");
				}

			} else {
				mes("^0000FF[도작 슬롯]^000000");
				mes("도작 스킬을 배우지 않았습니다.");
			}

			//--------------------------------
			// 리프로듀스 (Reproduce)
			//--------------------------------
			if( getskilllv(getskillid("SC_REPRODUCE")) > 0 ) {

				.@id2 = getplagiarizedskill(2);
				.@lv2 = getplagiarizedskilllv(2);

				if( .@id2 > 0 ) {
					mes("^00AA00[리프로듀스 슬롯]^000000");

					.@skill_name$ = getskilldesc(.@id2);
					if( .@skill_name$ == "" )
						.@skill_name$ = "알 수 없는 스킬 ID " + .@id2;

					mes("<b>" + .@skill_name$ + "</b> ^AA0000Lv." + .@lv2 + "^000000");
				}
				else {
					mes("^00AA00[리프로듀스 슬롯]^000000");
					mes("비어 있습니다.");
				}

			} else {
				mes("^00AA00[리프로듀스 슬롯]^000000");
				mes("리프로듀스 스킬을 배우지 않았습니다.");
			}

			next;
			continue;			
		}

		//--------------------------------
		// 초기화
		//--------------------------------
		if(.@main == 3) {

			mes(.npc_name$);
			if( getstatus(SC_PRESERVE) ) {
				mes("^FF0000프리저브(Preserve)^000000 상태가 활성화되어 있습니다.");
				mes("스킬을 초기화하기 전에 해당 상태를 제거합니다.");
				next();
				sc_end(SC_PRESERVE);
			}
			plagiarizeskillreset(1);
			plagiarizeskillreset(2);
			mes("복사한 스킬이 초기화되었습니다.");
			specialeffect2(EF_REPAIRWEAPON);

			next;
			continue;		
		}

		if(.@main == 4) close;

		//--------------------------------
		// 복사 방식
		//--------------------------------
		mes(.npc_name$);
		mes(" ");
		mes "어떤 스킬로 복사 하시겠습니까?";

		.@mode = select("도작:리프로듀스");

		if(.@mode == 1)
			.@filter$ = "plagiarism = 1";
		else
			.@filter$ = "reproduce = 1";

		//--------------------------------
		// 직업군 선택
		//--------------------------------
		mes("검색이나 복사할 스킬의 직업군을 선택 하세요.");
		next;

		.@menu = select(
			"스킬 검색","검사 계열","성직자 계열","궁수 계열","마법사 계열","상인 계열","도둑 계열","태권 계열","닌자 계열","뒤로가기"
		);

		if(.@menu == 10) continue;
		
		//--------------------------------
		// 검색 입력
		//--------------------------------
		if(.@menu == 1) {

			mes("검색어 입력 (0:취소)");
			input .@search$;

			if(.@search$ == "0")
				continue;
		}

		//--------------------------------
		// 직업군 설정 (검색 아닐 때만)
		//--------------------------------
		if(.@menu >= 2 && .@menu <= 9) {
			switch(.@menu) {
				case 2: .@group$="Swordman"; break;
				case 3: .@group$="Acolyte"; break;
				case 4: .@group$="Archer"; break;
				case 5: .@group$="Mage"; break;
				case 6: .@group$="Merchant"; break;
				case 7: .@group$="Thief"; break;
				case 8: .@group$="Taekwon"; break;
				case 9: .@group$="Ninja"; break;
			}
		}

		if(.@menu == 1) {
			// ?? 검색 (공백 제거 포함)
			.@count = query_sql(
			 "SELECT skill_id FROM skill_copyable WHERE "+.@filter$+" AND REPLACE(skill_name,' ','') LIKE REPLACE('%"+escape_sql(.@search$)+"%',' ','') LIMIT 30",
			 .@tmp_ids
			);
		}
		else {
			// ?? 직업군
			.@count = query_sql(
			 "SELECT skill_id FROM skill_copyable WHERE job_group='"+.@group$+"' AND "+.@filter$+" LIMIT 35",
			 .@tmp_ids
			);
		}

		//--------------------------------
		// 현재 캐릭 스킬 제거 필터
		//--------------------------------
		.@valid = 0;

		for(.@i = 0; .@i < .@count; .@i++) {

			// 잘못된 ID 방지
			if(.@tmp_ids[.@i] <= 0)
				continue;

			// 이미 배운 스킬 제외
			if(getskilllv(.@tmp_ids[.@i]) > 0)
				continue;

			.@skill_ids[.@valid] = .@tmp_ids[.@i];
			.@valid++;
		}

		.@count = .@valid;

		//--------------------------------
		// 결과 없음
		//--------------------------------
		if(.@count <= 0) {
			mes("사용 가능한 스킬이 없습니다.");
			close;
		}

		if(.@count <= 0) {
			mes("도작 가능한 스킬이 검색되지 않았습니다.");
			next;
			continue;
		}

		mes(.npc_name$);
		mes(" ");
		mes("복사할 스킬을 선택하세요.");
		if(.@search$ != "") 
			mes("검색어 : "+.@search$);
		next;

		//--------------------------------
		// skillselectmenu
		//--------------------------------
		
		.@selected = skillselectmenuarray(.@skill_ids);

		//--------------------------------
		// 선택 처리
		//--------------------------------
		if(.@selected > 0) {

			if( Zeny < .skill_zeny ) {
				mes(.npc_name$);
				mes("^FF0000제니가 부족합니다.^000000");
				mes(" ");
				mes("필요 제니 : " + callfunc("F_InsertComma", .skill_zeny) + " z");
				mes("보유 제니 : " + callfunc("F_InsertComma", Zeny) + " z");
				close;
			}

			.@selected_name$ = getskilldesc(.@selected);
			if( .@selected_name$ == "" )
				.@selected_name$ = "알 수 없는 스킬 ID " + .@selected;

			if( getstatus(SC_PRESERVE) ) {
				mes("^FF0000프리저브(Preserve)^000000 상태가 활성화되어 있습니다.");
				mes("스킬을 복사하기 전에 해당 상태를 제거합니다.");
				next();
				sc_end(SC_PRESERVE);
			}

			if(.@mode == 1)
				sc_end SC__REPRODUCE;
			else
				sc_start SC__REPRODUCE,600000,1;

			if(plagiarizeskill(.@selected,10)) {

				Zeny -= .skill_zeny;

				specialeffect2(EF_BASH3D);
				specialeffect2(EF_STEAL);

				mes(.npc_name$);
				mes "스킬 복사 성공!";

				// 복사 방식 확인
				if(getplagiarizedskill(2) == .@selected)
					mes(.@selected_name$ + " 를 리프로듀스 스킬로 복사 했습니다.");

				else if(getplagiarizedskill(1) == .@selected)
					mes(.@selected_name$ + " 를 도작 스킬로 복사 했습니다.");
				else
					mes(.@selected_name$ + " 스킬을 복사 했습니다.");

				close();
			}
			else {
				mes(.npc_name$);
				mes("^FF0000스킬 복사에 실패했습니다.^000000");
				mes("잠시 후 다시 시도해 주세요.");
				close;
			}

		}
	}
	
OnTimer2500:
	showscript strnpcinfo(1), getnpcid(0);
	initnpctimer;
	end;
	
OnInit:
	showscript strnpcinfo(1), getnpcid(0);
	initnpctimer;
		.npc_name$ = "<b>[ ^800080도작 도우미^000000 ]</b>";
		.skill_zeny = 15000;
		end;
	
}
