//===== NEED: @commands - player command guide in the chat window ===========
//= @commands (@명령어)      : the commands a player can use, with a short
//=                             description and an example (KR / EN by session)
//=
//= bindatcmd runs before the built-in @commands (atcommand.cpp), so this
//= replaces it for everyone (the built-in list is no longer reachable).
//= Every line names a command in the session's language (KR aliases for KR,
//= English names for EN). Korean-only binds (전장*, 결혼 votes, 노점검색) and
//= test / internal commands (needlang, wikicode) are not listed.
//============================================================================
-	script	NEED Commands	-1,{
	end;

	bindatcmd "commands", strnpcinfo(3) + "::OnCommands", 0, 99;
	bindatcmd "명령어", strnpcinfo(3) + "::OnCommands", 0, 99;
	end;

OnCommands:

	.@c = 0xA0E0FF;	// command

	dispbottom needtr("@선택루팅 +<아이템ID> / -<아이템ID> / reset - 지정 아이템 자동 획득 / 예: @선택루팅 +7444", "@autolootitem +<item ID> / -<item ID> / reset - loot only these items / e.g. @autolootitem +7444"), .@c;
