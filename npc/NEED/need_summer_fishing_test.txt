// ---- 추첨 분포 테스트: 등급별로 희귀도 분포를 각각 뽑아본다(등급 무관 확인). 등급 보정 없음 ----
// 낚시터가 하나이므로 기본 가중치(8000/1700/280/20)를 사용. 신규 4종도 희귀도 그룹에 포함.
S_DistTest:
	.@n = getarg(0);
	.@fever = $NEED_FISHING_FEVER_RATE; if ( .@fever <= 0 ) .@fever = 10000;
	.@rw1 = 8000;
	.@rw2 = 1700;
	.@rw3 = 280 * .@fever / 10000;
	.@rw4 = 20 * .@fever / 10000;
	.@t1 = .@rw1;
	.@t2 = .@rw1 + .@rw2;
	.@t3 = .@rw1 + .@rw2 + .@rw3;
	.@tot = .@rw1 + .@rw2 + .@rw3 + .@rw4;
	for ( .@g = 1; .@g <= 4; .@g++ ) {
		.@c1 = 0; .@c2 = 0; .@c3 = 0; .@c4 = 0;
		for ( .@k = 0; .@k < .@n; .@k++ ) {
			.@r = rand(.@tot);
			.@rar = 1 + (.@r >= .@t1) + (.@r >= .@t2) + (.@r >= .@t3);
			.@c1 = .@c1 + (.@rar == 1);
			.@c2 = .@c2 + (.@rar == 2);
			.@c3 = .@c3 + (.@rar == 3);
			.@c4 = .@c4 + (.@rar == 4);
		}
		mes "^0000FF[" + callfunc( "F_NeedFishGradeName", .@g ) + " " + .@n + "회]^000000 (등급 무관)";
		mes " 일반 " + .@c1 + " / 고급 " + .@c2 + " / 희귀 " + .@c3 + " / 전설 " + .@c4;
		debugmes "[FishDist] grade=" + .@g + " n=" + .@n + " C=" + .@c1 + " U=" + .@c2 + " R=" + .@c3 + " L=" + .@c4;
		next;
	}
	mes "분포 테스트 완료. 4개 등급 각각의 희귀도 분포가 유사하면 공정 플레이(등급 분리) 유지.";
	return;

// ---- 12종 어종 분포: 희귀도 추첨 후 같은 희귀도 내 어종을 균등 선택(가중치 동일)하여 집계 ----
S_SpeciesDist:
	.@n = getarg(0);
	.@fever = $NEED_FISHING_FEVER_RATE; if ( .@fever <= 0 ) .@fever = 10000;
	.@rw1 = 8000;
	.@rw2 = 1700;
	.@rw3 = 280 * .@fever / 10000;
	.@rw4 = 20 * .@fever / 10000;
	.@t1 = .@rw1;
	.@t2 = .@rw1 + .@rw2;
	.@t3 = .@rw1 + .@rw2 + .@rw3;
	.@tot = .@rw1 + .@rw2 + .@rw3 + .@rw4;
	setarray .@grp1[0], 1, 2, 3, 4;
	setarray .@grp2[0], 5, 6, 7, 8;
	setarray .@grp3[0], 9, 10, 11;
	deletearray .@cnt[0], 13;
	for ( .@k = 0; .@k < .@n; .@k++ ) {
		.@r = rand(.@tot);
		.@rar = 1 + (.@r >= .@t1) + (.@r >= .@t2) + (.@r >= .@t3);
		.@fid = (.@rar == 1) * .@grp1[rand(4)] + (.@rar == 2) * .@grp2[rand(4)] + (.@rar == 3) * .@grp3[rand(3)] + (.@rar == 4) * 12;
		.@cnt[.@fid] = .@cnt[.@fid] + 1;
	}
	mes "^0000FF[12종 어종 분포 " + .@n + "회]^000000 (등급 무관, 동일 희귀도 내 균등)";
	for ( .@i = 0; .@i < $@nff_count; .@i++ ) {
		.@fid = $@nff_id[.@i];
		mes " No." + ( .@fid < 10 ? "0" + .@fid : "" + .@fid ) + " " + $@nff_name$[.@i] + " [" + callfunc( "F_NeedFishRarityName", $@nff_rarity[.@i] ) + "] : " + .@cnt[.@fid];
		if ( .@i == 5 ) next;
	}
	debugmes "[FishSpeciesDist] n=" + .@n + " done";
	return;

S_GiveRod:
	if ( countitem(399936) < 1 ) { getitem 399936, 1; dispbottom "한여름 낚싯대(399936)를 지급했습니다. F1~F9 단축키로 사용하세요."; }
	return;
