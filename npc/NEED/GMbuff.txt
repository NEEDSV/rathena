new_zone01,134,113,4	script	BUFF	4_F_KAFRA1,{
	if (getgroupid() < 99) {
		end;
	}

	mes "[축복 관리자]";
	mes "현재 접속 중인 모든 유저에게";
	mes "30분 버프를 지급하시겠습니까?";
	next;
	if (select("버프 지급한다:취소한다") == 2)
		close;

	.@gm_rid = getcharid(3);

	announce "[NEED 안내] 모든 모험가에게 NEED의 축복이 지급되었습니다.", bc_all, 0xFFD700;

	.@count = getunits(BL_PC, .@aid);
	.@give_time = 1800000; // 30분
	
	for (.@i = 0; .@i < .@count; .@i++) {
		if (!attachrid(.@aid[.@i]))
			continue;

		percentheal 100,100;

		// 기본 버프 30분
		sc_start SC_BLESSING,1800000,10;
		sc_start SC_INCREASEAGI,1800000,10;
		sc_start SC_ANGELUS,1800000,10;
		sc_start SC_IMPOSITIO,1800000,5;
		sc_start SC_MAGNIFICAT,1800000,5;
		sc_start SC_GLORIA,1800000,5;
		sc_start SC_ASSUMPTIO,1800000,5;
		sc_start SC_KYRIE,1800000,10;
		sc_start SC_SECRAMENT,1800000,5;
		
		// attachrid 이후 현재 유저의 남은 시간을 개별 확인합니다.
		// 기존 시간이 30분 이상이면 유지하고, 30분 미만일 때만 갱신합니다.
		.@remain_time = getstatus(SC_NEED_GRACE, 5);
		if (.@remain_time < .@give_time)
			sc_start SC_NEED_GRACE,.@give_time,1;

		// 올마이티(12883) 효과
		sc_start SC_ALMIGHTY,1800000,30;
		sc_start SC_FOOD_STR_CASH,1800000,10;
		sc_start SC_FOOD_AGI_CASH,1800000,10;
		sc_start SC_FOOD_INT_CASH,1800000,10;
		sc_start SC_FOOD_DEX_CASH,1800000,10;
		sc_start SC_FOOD_LUK_CASH,1800000,10;
		sc_start SC_FOOD_VIT_CASH,1800000,10;

		// 인피니티 드링크(23475) 효과
		sc_start SC_INFINITY_DRINK,1800000,0;

		specialeffect2 EF_BLESSING;
		detachrid;
	}

	attachrid(.@gm_rid);
	close2;
	end;
}
