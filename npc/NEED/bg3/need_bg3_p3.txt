//============================================================
// NEED 3:3 룬 거점전 - B단계 (경기 원장 mode=3 / 주간 점수·단계 보상 / 승패 보상 / 주간 랭킹 / 탈주 기록 / 검토)
//============================================================
//= 로드 순서: need_duo2v2_p3.txt(2:2 정산, 공용 함수) → need_bg3.txt(3:3 경기) → 이 파일.
//= need_bg3.txt 는 $@NBG3P3_LOADED 일 때만 이 파일의 함수를 부른다(이 파일을 빼도 3:3 경기는 그대로 돈다).
//= 스위치 (서버 영구, 기본 모두 0 = 꺼짐. GM 60+ 가 @bg3p3 set ... 으로 바꾼다):
//=   $NEED_BG3_P3_ON    원장/점수/단계 보상권 생성/탈주 기록. 경기 시작(Launch) 때 켜져 있던 경기만 정산한다.
//=   $NEED_BG3_P3_RANK  수요일 04:00 KST 지난 주 3:3 랭킹 자동 확정(+ GM 수동 확정).
//=   $NEED_BG3_P3_MCASH 승패 직접 캐시 보상권 생성(경기 시작 때 켜져 있던 경기만).
//=   $NEED_BG3_P3_MITEM 승패 아이템 보상권 생성(경기 시작 때 켜져 있던 경기만).
//=   수령(지급)은 2:2 와 같은 함수·같은 순번(#NEED_BG2_PAIDSEQ / NEED_BG2_IPAIDSEQ)·같은 수령 스위치($NEED_BG2_P3_MCASH/MITEM/WITEM)를 쓴다.
//= 규칙  : need_bg3_p3_rules.txt (점수·승패 캐시/아이템·주간 아이템). IP 하루 직접 캐시 한도는 2:2 규칙의 값 하나를 함께 쓴다.
//= 원장  : 2:2 와 같은 테이블(need_bg2_*)에 `mode` = 3. 경기 PK(match_pk)는 두 모드가 같은 순번을 쓴다.
//=         3:3 경기는 need_bg2_match_done 표식이 있어야(= 6명 기록과 보상권까지 한 트랜잭션으로 끝남) 점수·랭킹·보상에 들어간다.
//= 보상권: 정산 트랜잭션 안에서는 상태 6(정산 중)으로 넣고, 완료 표식 직전에 0(받을 수 있음) / 7(검토 대기)로 바꾼다.
//=         2:2 의 수령·목록·대조는 상태를 0/1/2/7/8/9 로만 비교하므로, 중간에 멈춘 3:3 정산의 보상권(6)은 어디에도 보이지 않는다.
//= 실패  : DB 실패는 $NEED_BG3_PEND_H$/P$ 에 보관하고 1분마다 이 파일의 타이머가 다시 넣는다. 반쪽·불일치·형식 오류는
//=         $NEED_BG3_PART_H$/P$/R$ (GM 확인 후 requeue). 2:2 의 보관 큐·재시도 함수는 쓰지 않는다.
//= 트랜잭션: query_sql 연결은 모든 스크립트가 함께 쓴다. 시작은 2:2 의 F_NEED_BG2P3_Begin(남은 트랜잭션은 COMMIT, ROLLBACK 금지)을 그대로 쓴다.
//= 탈주  : need_bg2_desertion(2:2 와 같은 표, 계정 24시간 합산). 3:3 의 확정 탈주(재접속 유예 만료)마다 1행(경기·계정당 1행).
//= 명령  : @bg3p3 (@룬정산) GM 60+ / @bg3points (@룬점수)
//============================================================

function	script	F_NEED_BG3P3_Init	{
	$@NBG3P3_MODE = 3;
	setarray $@NBG3P3_TIER_PT[0], 10, 30, 60;
	$@NBG3P3_MINGAMES = 20;		// 랭킹 후보: 그 주 유효 경기 20 이상
	$@NBG3P3_MINACC = 8;		// 순위 보상 개설: 그 주 유효 경기 참가 계정 8 이상
	$@NBG3P3_WEEK = 604800;
	$@NBG3P3_FINAL_AT = 777600;	// 주차 시작 + 9일 = 다음 주 수요일 04:00 KST
	$@NBG3P3_PEND_MAX = 64;
	$@NBG3P3_PEND_BATCH = 2;	// 재시도 한 번의 단위(보상 없는 정산 1, 승패 보상 정산 2). 엔진 한도 check_gotocount 2048
	$@NBG3P3_LOADED = 1;
	callfunc "F_NEED_BG3P3_RulesLoad";
	return;
}

function	script	F_NEED_BG3P3_Log	{
	debugmes "[NEED_BG3] " + getarg(0);
	return;
}

//------------------------------------------------------------
// 보상 규칙 (need_bg3_p3_rules.txt). 검사 방식은 2:2 와 같다.
//------------------------------------------------------------

function	script	F_NEED_BG3P3_RuleSet	{
	.@v = getarg(0);
	if (.@v != $@NBG3R_N) {
		$@NBG3R_ERR$ += "version " + .@v + " out of order; ";
		return;
	}
	$@NBG3R_EFF[.@v] = getarg(1);
	$@NBG3R_PW[.@v] = getarg(2);
	$@NBG3R_PL[.@v] = getarg(3);
	$@NBG3R_PD[.@v] = getarg(4);
	$@NBG3R_CW[.@v] = getarg(5);
	$@NBG3R_CL[.@v] = getarg(6);
	$@NBG3R_CD[.@v] = getarg(7);
	$@NBG3R_IW$[.@v] = getarg(8);
	$@NBG3R_IL$[.@v] = getarg(9);
	$@NBG3R_ID$[.@v] = getarg(10);
	$@NBG3R_T10$[.@v] = getarg(11);
	$@NBG3R_T30$[.@v] = getarg(12);
	$@NBG3R_T60$[.@v] = getarg(13);
	$@NBG3R_R1$[.@v] = getarg(14);
	$@NBG3R_R2$[.@v] = getarg(15);
	$@NBG3R_R3$[.@v] = getarg(16);
	$@NBG3R_N++;
	return;
}

function	script	F_NEED_BG3P3_RuleSig$	{
	.@v = getarg(0);
	return $@NBG3R_EFF[.@v] + "|" + $@NBG3R_PW[.@v] + "/" + $@NBG3R_PL[.@v] + "/" + $@NBG3R_PD[.@v] + "|" + $@NBG3R_CW[.@v] + "/" + $@NBG3R_CL[.@v] + "/" + $@NBG3R_CD[.@v]
		+ "|" + $@NBG3R_IW$[.@v] + "/" + $@NBG3R_IL$[.@v] + "/" + $@NBG3R_ID$[.@v]
		+ "|" + $@NBG3R_T10$[.@v] + "/" + $@NBG3R_T30$[.@v] + "/" + $@NBG3R_T60$[.@v] + "|" + $@NBG3R_R1$[.@v] + "/" + $@NBG3R_R2$[.@v] + "/" + $@NBG3R_R3$[.@v];
}

// 규칙 읽기·검사(Init / 리로드). 실패하면 $@NBG3P3_RULES_OK = 0: 새 정산·보상권 생성을 멈춘다(경기 결과는 보관 정산).
function	script	F_NEED_BG3P3_RulesLoad	{
	deletearray $@NBG3R_EFF; deletearray $@NBG3R_PW; deletearray $@NBG3R_PL; deletearray $@NBG3R_PD;
	deletearray $@NBG3R_CW; deletearray $@NBG3R_CL; deletearray $@NBG3R_CD;
	deletearray $@NBG3R_IW$; deletearray $@NBG3R_IL$; deletearray $@NBG3R_ID$;
	deletearray $@NBG3R_T10$; deletearray $@NBG3R_T30$; deletearray $@NBG3R_T60$;
	deletearray $@NBG3R_R1$; deletearray $@NBG3R_R2$; deletearray $@NBG3R_R3$;
	deletearray $@NBG3R_F$;
	$@NBG3R_N = 0;
	$@NBG3R_ERR$ = "";
	callfunc "F_NEED_BG3P3_RulesDef";
	.@now = gettimetick(2);
	.@next = callfunc("F_NEED_BG2P3_WeekStart", .@now) + $@NBG3P3_WEEK;
	if ($@NBG3R_N < 1 || $@NBG3R_EFF[0] != 0)
		$@NBG3R_ERR$ += "version 0 (effective 0) missing; ";
	for (.@v = 0; .@v < $@NBG3R_N; .@v++) {
		if (.@v > 0) {	// && 는 뒤도 계산하므로 [.@v - 1] 은 블록 안에서만 읽는다
			if ($@NBG3R_EFF[.@v] != callfunc("F_NEED_BG2P3_WeekStart", $@NBG3R_EFF[.@v]) || $@NBG3R_EFF[.@v] <= $@NBG3R_EFF[.@v - 1])
				$@NBG3R_ERR$ += "v" + .@v + " start is not a later Monday 04:00 KST; ";
		}
		if ($@NBG3R_PW[.@v] < 0 || $@NBG3R_PW[.@v] > 10 || $@NBG3R_PL[.@v] < 0 || $@NBG3R_PL[.@v] > 10 || $@NBG3R_PD[.@v] < 0 || $@NBG3R_PD[.@v] > 10)
			$@NBG3R_ERR$ += "v" + .@v + " points not 0..10; ";
		if ($@NBG3R_CW[.@v] < 0 || $@NBG3R_CW[.@v] > 1000000 || $@NBG3R_CL[.@v] < 0 || $@NBG3R_CL[.@v] > 1000000 || $@NBG3R_CD[.@v] < 0 || $@NBG3R_CD[.@v] > 1000000)
			$@NBG3R_ERR$ += "v" + .@v + " match cash not 0..1000000; ";
		setarray .@l$[0], $@NBG3R_IW$[.@v], $@NBG3R_IL$[.@v], $@NBG3R_ID$[.@v], $@NBG3R_T10$[.@v], $@NBG3R_T30$[.@v], $@NBG3R_T60$[.@v], $@NBG3R_R1$[.@v], $@NBG3R_R2$[.@v], $@NBG3R_R3$[.@v];
		setarray .@ln$[0], "win", "lose", "draw", "tier10", "tier30", "tier60", "rank1", "rank2", "rank3";
		for (.@j = 0; .@j < 9; .@j++) {
			.@c = callfunc("F_NEED_BG2P3_ParseItems", .@l$[.@j]);
			if (.@c < 0) {
				$@NBG3R_ERR$ += "v" + .@v + " " + .@ln$[.@j] + ": " + $@NBG2P3_IT_ERR$ + "; ";
				continue;
			}
			.@f$ = "";
			for (.@i = 0; .@i < .@c; .@i++)
				.@f$ += (.@i ? ", " : "") + "(%R%, " + .@i + ", " + $@NBG2P3_IT_ID[.@i] + ", " + $@NBG2P3_IT_N[.@i] + ")";
			$@NBG3R_F$[.@v * 9 + .@j] = .@f$;
		}
		.@sig$ = callfunc("F_NEED_BG3P3_RuleSig$", .@v);
		.@old$ = $NEED_BG3_RVSIG$[.@v];
		if (.@old$ != "" && .@old$ != .@sig$ && $@NBG3R_EFF[.@v] <= .@now)
			$@NBG3R_ERR$ += "v" + .@v + " already started and was changed; ";
		if (.@old$ == "" && .@v > 0 && $@NBG3R_EFF[.@v] < .@next)
			$@NBG3R_ERR$ += "v" + .@v + " is new but starts before next Monday 04:00 KST; ";
	}
	.@n = getarraysize($NEED_BG3_RVSIG$);
	for (.@v = $@NBG3R_N; .@v < .@n; .@v++) {
		if ($NEED_BG3_RVSIG$[.@v] == "")
			continue;
		explode(.@sf$, $NEED_BG3_RVSIG$[.@v], "|");
		if (atoi(.@sf$[0]) <= .@now)
			$@NBG3R_ERR$ += "v" + .@v + " already started and was removed; ";
	}
	if ($@NBG3R_ERR$ != "") {
		$@NBG3P3_RULES_OK = 0;
		callfunc "F_NEED_BG3P3_Log", "P3_RULES_INVALID " + $@NBG3R_ERR$ + "(new 3v3 settlements and reward tickets are held until the rules file is fixed and reloaded)";
		return 0;
	}
	for (.@v = 0; .@v < $@NBG3R_N; .@v++)
		if ($NEED_BG3_RVSIG$[.@v] != callfunc("F_NEED_BG3P3_RuleSig$", .@v))
			$NEED_BG3_RVSIG$[.@v] = callfunc("F_NEED_BG3P3_RuleSig$", .@v);
	for (.@v = $@NBG3R_N; .@v < .@n; .@v++)
		$NEED_BG3_RVSIG$[.@v] = "";
	$@NBG3P3_RULES_OK = 1;
	callfunc "F_NEED_BG3P3_Log", "P3_RULES_OK versions=" + $@NBG3R_N;
	return 1;
}

// 시각 t 가 속한 주차의 3:3 규칙 버전.
function	script	F_NEED_BG3P3_RuleVer	{
	.@ws = callfunc("F_NEED_BG2P3_WeekStart", getarg(0));
	for (.@v = $@NBG3R_N - 1; .@v > 0; .@v--)
		if ($@NBG3R_EFF[.@v] <= .@ws)
			return .@v;
	return 0;
}

// 3:3 아이템 보상권 하나(트랜잭션 안): INSERT IGNORE + 아이템 스냅샷. 새로 만들면 1, 이미 있으면 0, 실패 -1.
// getarg: ws, aid, kind$, key$, rank_slot(0=없음), cid, status, frag$
function	script	F_NEED_BG3P3_ItemTicket	{
	if (getarg(7) == "")
		return 0;
	if (query_sql("INSERT IGNORE INTO need_bg2_reward (`mode`, week_start, account_id, reward_kind, reward_key, rank_slot, char_id, amount, status, created_at) VALUES ("
		+ $@NBG3P3_MODE + ", " + getarg(0) + ", " + getarg(1) + ", '" + getarg(2) + "', '" + getarg(3) + "', " + (getarg(4) ? "" + getarg(4) : "NULL") + ", " + getarg(5) + ", 0, " + getarg(6) + ", " + gettimetick(2) + ")") < 0)
		return -1;
	.@rc = 0;
	query_sql "SELECT ROW_COUNT()", .@rc;
	if (.@rc < 1)
		return 0;
	.@id = 0;
	query_sql "SELECT LAST_INSERT_ID()", .@id;
	if (.@id <= 0 || callfunc("F_NEED_BG2P3_PutItems", .@id, getarg(7)) < 0)
		return -1;
	return 1;
}

//------------------------------------------------------------
// 점수 집계 (3:3 = mode 3, 완료 표식이 있는 경기만)
//------------------------------------------------------------

// 3:3 유효 경기 조건(JOIN 뒤에 붙이는 WHERE 조각). m = need_bg2_match, p = need_bg2_match_player.
function	script	F_NEED_BG3P3_Valid$	{
	return " JOIN need_bg2_match_done d ON d.match_pk = m.match_pk WHERE m.`mode` = " + $@NBG3P3_MODE + " AND m.week_start = " + getarg(0) + " AND p.valid = 1";
}

// 계정 주간 3:3 점수. held = 1 이면 검토 보류 중인 점수만.
function	script	F_NEED_BG3P3_AccPoints	{
	.@pts = 0;
	query_sql "SELECT COALESCE(SUM(p.points), 0) FROM need_bg2_match_player p JOIN need_bg2_match m ON m.match_pk = p.match_pk" + callfunc("F_NEED_BG3P3_Valid$", getarg(1))
		+ " AND p.account_id = " + getarg(0) + " AND m.review " + (getarg(2, 0) ? "= 1" : "IN (0, 2)"), .@pts;
	return .@pts;
}

// 계정 단계 보상권(아이템). 트랜잭션 안. curpts = 아직 완료 표식이 없는 이번 경기의 점수(정산 중일 때만), status = 새 보상권 상태.
// 새로 만든 수, 실패 -1. $@NBG3P3_LASTPTS = 계정 주간 점수, $@NBG3P3_TKEYS$ = 도달한 단계 키 목록("'10','30'").
function	script	F_NEED_BG3P3_TierSync	{
	.@aid = getarg(0);
	.@ws = getarg(1);
	.@pts = 0;
	if (query_sql("SELECT COALESCE(SUM(p.points), 0) FROM need_bg2_match_player p JOIN need_bg2_match m ON m.match_pk = p.match_pk" + callfunc("F_NEED_BG3P3_Valid$", .@ws)
		+ " AND p.account_id = " + .@aid + " AND m.review IN (0, 2)", .@pts) < 0)
		return -1;
	.@pts += getarg(2, 0);
	$@NBG3P3_LASTPTS = .@pts;
	$@NBG3P3_TKEYS$ = "";
	if (!$@NBG3P3_RULES_OK)
		return -1;
	.@v = callfunc("F_NEED_BG3P3_RuleVer", .@ws);
	.@new = 0;
	for (.@t = 0; .@t < 3; .@t++) {
		if (.@pts < $@NBG3P3_TIER_PT[.@t])
			break;
		$@NBG3P3_TKEYS$ += ($@NBG3P3_TKEYS$ == "" ? "" : ",") + "'" + $@NBG3P3_TIER_PT[.@t] + "'";
		.@r = callfunc("F_NEED_BG3P3_ItemTicket", .@ws, .@aid, "TIERI", "" + $@NBG3P3_TIER_PT[.@t], 0, 0, getarg(3, 0), $@NBG3R_F$[.@v * 9 + 3 + .@t]);
		if (.@r < 0)
			return -1;
		.@new += .@r;
	}
	return .@new;
}

// 경기 무효 뒤: 계정 점수가 단계 아래로 내려가면 미수령 단계 보상권은 회수(8), 지급된 것은 수동 정산(9) 표시.
function	script	F_NEED_BG3P3_TierRevoke	{
	.@aid = getarg(0);
	.@ws = getarg(1);
	.@pts = 0;
	if (query_sql("SELECT COALESCE(SUM(p.points), 0) FROM need_bg2_match_player p JOIN need_bg2_match m ON m.match_pk = p.match_pk" + callfunc("F_NEED_BG3P3_Valid$", .@ws)
		+ " AND p.account_id = " + .@aid + " AND m.review IN (0, 2)", .@pts) < 0)
		return -1;
	.@cnt = 0;
	for (.@t = 0; .@t < 3; .@t++) {
		if (.@pts >= $@NBG3P3_TIER_PT[.@t])
			continue;
		.@base$ = " WHERE `mode` = " + $@NBG3P3_MODE + " AND week_start = " + .@ws + " AND account_id = " + .@aid + " AND reward_kind = 'TIERI' AND reward_key = '" + $@NBG3P3_TIER_PT[.@t] + "'";
		if (query_sql("UPDATE need_bg2_reward SET status = 8, note = 'revoked: match voided, never paid'" + .@base$ + " AND status = 0") < 0)
			return -1;
		.@rc = 0;
		query_sql "SELECT ROW_COUNT()", .@rc;
		.@cnt += .@rc;
		if (query_sql("UPDATE need_bg2_reward SET note = CONCAT('MANUAL: paid or reserved (status ', status, '), match voided'), status = 9" + .@base$ + " AND status IN (1, 2)") < 0)
			return -1;
		.@rc = 0;
		query_sql "SELECT ROW_COUNT()", .@rc;
		.@cnt += .@rc;
	}
	return .@cnt;
}

//------------------------------------------------------------
// 승패 보상 (트랜잭션 안, Apply 가 부른다)
//------------------------------------------------------------
// 참가자: $@NBG3P3_MR_AID/CID/OUT/VAL[0..5], $@NBG3P3_MR_IP$[]. flags 1 직접 캐시 / 2 아이템(경기 시작 때 고정).
// 직접 캐시 금액은 3:3 규칙(v3), IP 하루 한도는 2:2 규칙(v2, 같은 시작 주차)의 값을 쓴다. 할당 방식은 2:2 와 같다
// (같은 경기·같은 IP 무리는 요청액 비율, 남는 금액은 계정 ID 오름차순 1씩, 할당은 need_bg2_ipcash_day/alloc 공용).
// 보상권은 상태 6 으로 넣는다(Apply 끝에서 0/7 로 바꾼다). IP 할당 상태는 검토 경기 1, 그 밖 0 (2:2 와 같은 의미).
function	script	F_NEED_BG3P3_MatchRewards	{
	.@pk = getarg(0);
	.@v3 = getarg(1);
	.@v2 = getarg(2);
	.@flags = getarg(3);
	.@ws = getarg(4);
	.@te = getarg(5);
	.@held = getarg(6);
	setarray .@o[0], 0, 1, 2, 3, 4, 5;
	for (.@a = 0; .@a < 5; .@a++)
		for (.@b = 0; .@b < 5 - .@a; .@b++)
			if ($@NBG3P3_MR_AID[.@o[.@b]] > $@NBG3P3_MR_AID[.@o[.@b + 1]]) {
				.@t = .@o[.@b];
				.@o[.@b] = .@o[.@b + 1];
				.@o[.@b + 1] = .@t;
			}
	if (.@flags & 2) {
		for (.@k = 0; .@k < 6; .@k++) {
			if (!$@NBG3P3_MR_VAL[.@k])
				continue;
			.@out = $@NBG3P3_MR_OUT[.@k];
			.@j = (.@out == 1) ? 0 : ((.@out == 2) ? 1 : 2);
			if (callfunc("F_NEED_BG3P3_ItemTicket", .@ws, $@NBG3P3_MR_AID[.@k], "MITEM", "" + .@pk, 0, $@NBG3P3_MR_CID[.@k], 6, $@NBG3R_F$[.@v3 * 9 + .@j]) < 0)
				return -1;
		}
	}
	.@cap = $@NBG2R_CAP[.@v2];
	if (!(.@flags & 1) || .@cap <= 0)
		return 0;
	.@day = (.@te + 18000) / 86400;
	for (.@k = 0; .@k < 6; .@k++) {
		.@out = $@NBG3P3_MR_OUT[.@k];
		.@req[.@k] = !$@NBG3P3_MR_VAL[.@k] ? 0 : ((.@out == 1) ? $@NBG3R_CW[.@v3] : ((.@out == 2) ? $@NBG3R_CL[.@v3] : $@NBG3R_CD[.@v3]));
		.@alloc[.@k] = 0;
		.@done[.@k] = 0;
		.@key$[.@k] = $@NBG3P3_MR_IP$[.@k];
		if (.@req[.@k] > 0)
			.@key$[.@k] = callfunc("F_NEED_BG2P3_CashKey$", $@NBG3P3_MR_AID[.@k], .@key$[.@k]);
	}
	for (.@x = 0; .@x < 6; .@x++) {
		.@k = .@o[.@x];
		if (.@done[.@k] || .@req[.@k] <= 0)
			continue;
		.@ip$ = $@NBG3P3_MR_IP$[.@k];
		.@gk$ = .@key$[.@k];
		if (.@ip$ == "") {
			.@done[.@k] = 1;
			if (query_sql("INSERT IGNORE INTO need_bg2_ipcash_alloc (match_pk, account_id, char_id, ip, day_key, requested, allocated, state, created_at) VALUES ("
				+ .@pk + ", " + $@NBG3P3_MR_AID[.@k] + ", " + $@NBG3P3_MR_CID[.@k] + ", '', " + .@day + ", " + .@req[.@k] + ", 0, 4, " + gettimetick(2) + ")") < 0)
				return -1;
			continue;
		}
		deletearray .@g;
		.@gn = 0;
		.@tot = 0;
		for (.@y = .@x; .@y < 6; .@y++) {
			.@j = .@o[.@y];
			if (.@done[.@j] || .@req[.@j] <= 0 || .@key$[.@j] != .@gk$)
				continue;
			.@g[.@gn] = .@j;
			.@gn++;
			.@tot += .@req[.@j];
			.@done[.@j] = 1;
		}
		.@used = 0;
		// 공용 가족 명단 계정: IP 한도 검사 대신 같은 한도를 계정별로 적용(그날 이 계정의 할당 합, 상태 0/1). IP 사용량에는 그대로 더한다.
		if (.@gk$ != .@ip$) {
			if (query_sql("SELECT COALESCE(SUM(allocated), 0) FROM need_bg2_ipcash_alloc WHERE account_id = " + $@NBG3P3_MR_AID[.@k] + " AND day_key = " + .@day + " AND state IN (0, 1)", .@used) < 0)
				return -1;
		} else if (query_sql("SELECT used FROM need_bg2_ipcash_day WHERE ip = '" + escape_sql(.@ip$) + "' AND day_key = " + .@day + " FOR UPDATE", .@used) < 0)
			return -1;
		.@rem = .@cap - .@used;
		if (.@rem < 0)
			.@rem = 0;
		.@sum = 0;
		for (.@y = 0; .@y < .@gn; .@y++) {
			.@j = .@g[.@y];
			.@alloc[.@j] = (.@tot <= .@rem) ? .@req[.@j] : .@req[.@j] * .@rem / .@tot;
			.@sum += .@alloc[.@j];
		}
		.@left = ((.@tot <= .@rem) ? .@tot : .@rem) - .@sum;
		for (.@y = 0; .@y < .@gn && .@left > 0; .@y++) {
			.@j = .@g[.@y];
			if (.@alloc[.@j] < .@req[.@j]) {
				.@alloc[.@j]++;
				.@sum++;
				.@left--;
			}
		}
		if (.@sum > 0) {
			if (query_sql("INSERT INTO need_bg2_ipcash_day (ip, day_key, used, updated_at) VALUES ('" + escape_sql(.@ip$) + "', " + .@day + ", " + .@sum + ", " + gettimetick(2)
				+ ") ON DUPLICATE KEY UPDATE used = used + " + .@sum + ", updated_at = VALUES(updated_at)") < 0)
				return -1;
		}
		for (.@y = 0; .@y < .@gn; .@y++) {
			.@j = .@g[.@y];
			if (query_sql("INSERT INTO need_bg2_ipcash_alloc (match_pk, account_id, char_id, ip, day_key, requested, allocated, state, created_at) VALUES ("
				+ .@pk + ", " + $@NBG3P3_MR_AID[.@j] + ", " + $@NBG3P3_MR_CID[.@j] + ", '" + escape_sql(.@ip$) + "', " + .@day + ", " + .@req[.@j] + ", " + .@alloc[.@j] + ", "
				+ (.@held ? 1 : 0) + ", " + gettimetick(2) + ")") < 0)
				return -1;
			if (.@alloc[.@j] > 0) {
				if (query_sql("INSERT INTO need_bg2_reward (`mode`, week_start, account_id, reward_kind, reward_key, char_id, amount, status, created_at) VALUES ("
					+ $@NBG3P3_MODE + ", " + .@ws + ", " + $@NBG3P3_MR_AID[.@j] + ", 'MCASH', '" + .@pk + "', " + $@NBG3P3_MR_CID[.@j] + ", " + .@alloc[.@j] + ", 6, " + gettimetick(2) + ")") < 0)
					return -1;
			}
		}
		callfunc "F_NEED_BG3P3_Log", "P3_MCASH_ALLOC match_pk=" + .@pk + " day=" + .@day + ((.@gk$ != .@ip$) ? " family_aid=" + $@NBG3P3_MR_AID[.@k] + " acct_used_before=" : " ip_used_before=") + .@used + " cap=" + .@cap + " requested=" + .@tot + " granted=" + .@sum
			+ " accounts=" + .@gn + (.@held ? " held" : "");
	}
	return 0;
}

//------------------------------------------------------------
// 3:3 경기 흐름 연결 (need_bg3.txt 가 $@NBG3P3_LOADED 일 때만 부른다)
//------------------------------------------------------------

// Launch: 이 경기의 정산 여부와 시작 때 값을 고정한다(경기 중 스위치를 바꿔도 그 경기는 시작 때 값).
function	script	F_NEED_BG3P3_Arm	{
	$@NBG3_P3ARMED = $NEED_BG3_P3_ON ? 1 : 0;
	$@NBG3_P3FLAGS = ($NEED_BG3_P3_MCASH ? 1 : 0) | ($NEED_BG3_P3_MITEM ? 2 : 0);
	$@NBG3_P3VER = callfunc("F_NEED_BG3P3_RuleVer", gettimetick(2));
	$@NBG3_P3SRC$ = (getarg(0) == 2) ? "queue" : "gm";
	$@NBG3_T0 = gettimetick(2);
	$@NBG3_TICK0 = gettimetick(0);
	cleararray $@NBG3_CLS[0], 0, 6;
	cleararray $@NBG3_IP$[0], "", 6;
	cleararray $@NBG3_SUP[0], 0, 6;
	return;
}

// Launch(첨부된 참가자 k): 직업/IP(경기 중 메모리에만), 경기 전부터 있던 엔진 탈주자 상태 보존.
function	script	F_NEED_BG3P3_LaunchSelf	{
	.@k = getarg(0);
	$@NBG3_CLS[.@k] = Class;
	$@NBG3_IP$[.@k] = getcharip();
	if (!$@NBG3_P3ARMED)
		return;
	NEED_BG3_P3 = 1;
	NEED_BG3_PREDES = 0;
	if (getstatus(SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT))
		NEED_BG3_PREDES = gettimetick(2) + getstatus(SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT, 5) / 1000;
	return;
}

// RestoreSelf(첨부): 정산 경기의 탈주는 원장 제재로만. 엔진 탈주자 상태는 지우고, 경기 전부터 있던 것은 남은 시간만큼 되돌린다.
function	script	F_NEED_BG3P3_EngineDeserter	{
	.@now = gettimetick(2);
	.@had = getstatus(SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT);
	if (.@had)
		sc_end SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT;
	.@back = 0;
	if (NEED_BG3_PREDES > .@now) {
		.@back = NEED_BG3_PREDES - .@now;
		sc_start SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT, .@back * 1000, 1, 10000, SCSTART_NOAVOID | SCSTART_NOTICKDEF;
	}
	NEED_BG3_P3 = 0;
	NEED_BG3_PREDES = 0;
	return;
}

// Finish 직후(상태 6). 경기 결과를 고정해 원장에 넣는다. 경기당 한 번.
// head = key|ext_mid|source|started|ended|kind|why|active_ms|score_b|score_r|sig_ip|flags|ver
// players = aid,cid,team,class,gone,hostile,hit,support,ip ; x6
function	script	F_NEED_BG3P3_Settle	{
	.@kind = getarg(0);
	.@why$ = getarg(1);
	if (!$@NBG3_P3ARMED)
		return 0;
	$@NBG3_P3ARMED = 0;	// 같은 경기를 두 번 정산하지 않는다(원장 match_key 유니크가 2차 방어)
	.@sigip = 0;
	for (.@b = 0; .@b < 3; .@b++)
		for (.@r = 3; .@r < 6; .@r++)
			if ($@NBG3_IP$[.@b] != "" && $@NBG3_IP$[.@b] == $@NBG3_IP$[.@r])
				.@sigip = 1;
	.@act = 0;
	if ($@NBG3_START_AT) {
		.@end = gettimetick(0);
		if (.@end > $@NBG3_START_AT + $@NBG3_C_MATCH)
			.@end = $@NBG3_START_AT + $@NBG3_C_MATCH;
		.@act = .@end - $@NBG3_START_AT;
	}
	.@key$ = "B3-" + $@NBG3_MATCH + "-" + $@NBG3_TICK0;
	.@h$ = .@key$ + "|" + $@NBG3_MATCH + "|" + $@NBG3_P3SRC$ + "|" + $@NBG3_T0 + "|" + gettimetick(2) + "|" + .@kind + "|" + .@why$
		+ "|" + .@act + "|" + $@NBG3_SCORE[1] + "|" + $@NBG3_SCORE[2] + "|" + .@sigip + "|" + $@NBG3_P3FLAGS + "|" + $@NBG3_P3VER;
	.@p$ = "";
	for (.@i = 0; .@i < 6; .@i++)
		.@p$ += (.@i ? ";" : "") + $@NBG3_AID[.@i] + "," + $@NBG3_CID[.@i] + "," + $@NBG3_TEAM[.@i] + "," + $@NBG3_CLS[.@i] + "," + $@NBG3_GONE[.@i]
			+ "," + $@NBG3_HOST[.@i] + "," + $@NBG3_HIT[.@i] + "," + $@NBG3_SUP[.@i] + "," + $@NBG3_IP$[.@i];
	.@r = callfunc("F_NEED_BG3P3_Apply", .@h$, .@p$, 1);
	callfunc "F_NEED_BG3P3_Log", "P3_SETTLE match=" + $@NBG3_MATCH + " key=" + .@key$ + " result=" + .@r + " head=" + .@h$ + " players=" + .@p$;
	if (.@r == -1)
		callfunc "F_NEED_BG3P3_PendPush", .@h$, .@p$;
	else if (.@r < 0)
		callfunc "F_NEED_BG3P3_PartPush", .@h$, .@p$, .@r;
	return .@r;
}

function	script	F_NEED_BG3P3_PendPush	{
	.@n = getarraysize($NEED_BG3_PEND_H$);
	if (.@n >= $@NBG3P3_PEND_MAX) {
		callfunc "F_NEED_BG3P3_Log", "P3_PENDING_OVERFLOW head=" + getarg(0) + " players=" + getarg(1) + " (manual replay from this log line)";
		return 0;
	}
	$NEED_BG3_PEND_H$[.@n] = getarg(0);
	$NEED_BG3_PEND_P$[.@n] = getarg(1);
	callfunc "F_NEED_BG3P3_Log", "P3_PENDING_RECONCILIATION slot=" + .@n + " head=" + getarg(0) + " players=" + getarg(1);
	return 1;
}

// 3:3 보관 정산을 앞에서부터 묶음 단위만큼 다시 넣는다(보상 정산 2단위). DB 오류면 그 건에서 멈추고 순서를 지킨다.
function	script	F_NEED_BG3P3_PendRetry	{
	.@ok = 0; .@dup = 0; .@part = 0; .@err = 0; .@units = 0;
	for (.@i = 0; .@i < $@NBG3P3_PEND_BATCH; .@i++) {
		if (getarraysize($NEED_BG3_PEND_H$) < 1)
			break;
		.@h$ = $NEED_BG3_PEND_H$[0];
		.@p$ = $NEED_BG3_PEND_P$[0];
		deletearray .@hh$;
		explode(.@hh$, .@h$, "|");
		.@w = (getarraysize(.@hh$) == 13 && (atoi(.@hh$[11]) & 3)) ? 2 : 1;
		if (.@units + .@w > $@NBG3P3_PEND_BATCH)
			break;
		.@units += .@w;
		.@r = callfunc("F_NEED_BG3P3_Apply", .@h$, .@p$, 1);
		if (.@r == -1) {
			.@err = 1;
			break;
		}
		if (.@r < 0) {
			callfunc "F_NEED_BG3P3_PartPush", .@h$, .@p$, .@r;
			.@part++;
		} else if (.@r == 0)
			.@dup++;
		else
			.@ok++;
		deletearray $NEED_BG3_PEND_H$[0], 1;
		deletearray $NEED_BG3_PEND_P$[0], 1;
		callfunc "F_NEED_BG3P3_Log", "P3_PENDING_DONE result=" + .@r + " head=" + .@h$;
	}
	$@NBG3P3_RT_OK = .@ok;
	$@NBG3P3_RT_DUP = .@dup;
	$@NBG3P3_RT_PART = .@part;
	$@NBG3P3_RT_ERR = .@err;
	callfunc "F_NEED_BG3P3_Log", "P3_PENDING_BATCH applied=" + .@ok + " duplicate_ok=" + .@dup + " partial=" + .@part + " db_error=" + .@err
		+ " left=" + getarraysize($NEED_BG3_PEND_H$) + " review_box=" + callfunc("F_NEED_BG3P3_PartCount");
	return .@ok + .@dup + .@part;
}

function	script	F_NEED_BG3P3_PartPush	{
	.@n = getarraysize($NEED_BG3_PART_H$);
	if (.@n >= $@NBG3P3_PEND_MAX) {
		callfunc "F_NEED_BG3P3_Log", "P3_PARTIAL_OVERFLOW code=" + getarg(2) + " head=" + getarg(0) + " players=" + getarg(1) + " (manual replay from this log line)";
		return 0;
	}
	$NEED_BG3_PART_H$[.@n] = getarg(0);
	$NEED_BG3_PART_P$[.@n] = getarg(1);
	$NEED_BG3_PART_R$[.@n] = (getarg(2) == -2 ? "bad_payload" : $@NBG3P3_PART_WHY$);
	callfunc "F_NEED_BG3P3_Log", "P3_PARTIAL_KEPT slot=" + .@n + " why=" + $NEED_BG3_PART_R$[.@n] + " head=" + getarg(0) + " players=" + getarg(1);
	return 1;
}

// 검토 보관함 건수(처리한 칸은 당기지 않고 비워 둔다: 2:2 와 같은 이유).
function	script	F_NEED_BG3P3_PartCount	{
	.@c = 0;
	.@n = getarraysize($NEED_BG3_PART_H$);
	for (.@i = 0; .@i < .@n; .@i++)
		if ($NEED_BG3_PART_H$[.@i] != "")
			.@c++;
	return .@c;
}

// 경기 결과 → 원장. 1 기록, 0 이미 기록됨(정상 중복), -1 DB 실패(롤백), -2 형식 오류, -3 반쪽·불일치(검토 보관).
function	script	F_NEED_BG3P3_Apply	{
	.@notify = getarg(2, 0);
	explode(.@h$, getarg(0), "|");
	if (getarraysize(.@h$) != 13)
		return -2;
	.@key$ = .@h$[0];
	.@mid = atoi(.@h$[1]);
	.@src$ = .@h$[2];
	.@t0 = atoi(.@h$[3]);
	.@te = atoi(.@h$[4]);
	.@kind = atoi(.@h$[5]);
	.@why$ = .@h$[6];
	.@act = atoi(.@h$[7]);
	.@sb = atoi(.@h$[8]);
	.@sr = atoi(.@h$[9]);
	.@sigip = atoi(.@h$[10]);
	.@flags = atoi(.@h$[11]);
	if (!$@NBG3P3_RULES_OK || !$@NBG2P3_RULES_OK) {
		callfunc "F_NEED_BG3P3_Log", "P3_RULES_BLOCKED key=" + .@key$ + " (2v2 or 3v3 rules file invalid: kept for retry)";
		return -1;
	}
	.@v3 = callfunc("F_NEED_BG3P3_RuleVer", .@t0);
	.@v2 = callfunc("F_NEED_BG2P3_RuleVer", .@t0);	// IP 하루 한도(2:2 규칙, 같은 시작 주차)
	explode(.@pl$, getarg(1), ";");
	if (getarraysize(.@pl$) != 6 || .@key$ == "" || .@te <= 0)
		return -2;
	.@hostile = 0;
	for (.@k = 0; .@k < 6; .@k++) {
		deletearray .@f$;
		explode(.@f$, .@pl$[.@k], ",");
		if (getarraysize(.@f$) != 9)
			return -2;
		.@aid[.@k] = atoi(.@f$[0]);
		.@cid[.@k] = atoi(.@f$[1]);
		.@team[.@k] = atoi(.@f$[2]);
		.@cls[.@k] = atoi(.@f$[3]);
		.@gone[.@k] = atoi(.@f$[4]);
		.@hs[.@k] = atoi(.@f$[5]);
		.@ht[.@k] = atoi(.@f$[6]);
		.@sp[.@k] = atoi(.@f$[7]);
		.@ip$[.@k] = .@f$[8];
		.@hostile += .@hs[.@k];
	}
	.@ws = callfunc("F_NEED_BG2P3_WeekStart", .@te);
	.@queue = (.@src$ == "queue");
	.@engaged = (.@hostile > 0);
	// 개인 정산: 공개 대기열 + 실교전 + 정상 결과(무/블루/레드)만 점수가 있다. 확정 탈주자(gone 1)는 0점·불인정.
	.@valid = 0;
	for (.@k = 0; .@k < 6; .@k++) {
		.@out[.@k] = 0;
		if (.@kind == 0)
			.@out[.@k] = 3;
		else if (.@kind == 1 || .@kind == 2)
			.@out[.@k] = (.@team[.@k] == .@kind) ? 1 : 2;
		.@des[.@k] = (.@gone[.@k] == 1) ? 1 : 0;
		.@pts[.@k] = 0;
		.@val[.@k] = 0;
		.@win[.@k] = 0;
		if (!.@queue || !.@engaged || .@kind > 2 || .@des[.@k])
			continue;
		.@val[.@k] = 1;
		.@valid = 1;
		if (.@out[.@k] == 1) {
			.@pts[.@k] = $@NBG3R_PW[.@v3];
			.@win[.@k] = 1;
		} else
			.@pts[.@k] = (.@out[.@k] == 2) ? $@NBG3R_PL[.@v3] : $@NBG3R_PD[.@v3];
	}
	// 탈주 기록: 공개 대기열 경기의 확정 탈주(재접속 유예 만료)마다. 준비시간 이탈(경기 취소)은 제재 없음.
	.@desert = (.@queue && .@why$ != "depart_in_preparation");
	if (.@why$ == "both_teams_departed")
		.@dkind$ = "both_left";
	else if (.@why$ == "forfeit_blue" || .@why$ == "forfeit_red")
		.@dkind$ = .@engaged ? "forfeit" : "no_combat";
	else
		.@dkind$ = "left";
	// 6계정 정렬 키(반복 매칭 신호).
	copyarray .@sa[0], .@aid[0], 6;
	for (.@a = 0; .@a < 5; .@a++)
		for (.@b = 0; .@b < 5 - .@a; .@b++)
			if (.@sa[.@b] > .@sa[.@b + 1]) {
				.@tmp = .@sa[.@b];
				.@sa[.@b] = .@sa[.@b + 1];
				.@sa[.@b + 1] = .@tmp;
			}
	.@akey$ = .@sa[0] + "-" + .@sa[1] + "-" + .@sa[2] + "-" + .@sa[3] + "-" + .@sa[4] + "-" + .@sa[5];

	if (callfunc("F_NEED_BG2P3_Begin", "bg3_apply_begin") < 0)
		return -1;
	deletearray .@dup;
	.@n = query_sql("SELECT match_pk FROM need_bg2_match WHERE match_key = '" + escape_sql(.@key$) + "'", .@dup);
	if (.@n < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_dupcheck");
	if (.@n > 0) {
		query_sql "ROLLBACK";
		// 정상 중복 = 참가자 정확히 6행(payload 와 같은 aid/cid/team) + 완료 표식. 그 밖에는 반쪽(다른 스크립트가 커밋한 중단 트랜잭션 포함).
		deletearray .@ea; deletearray .@ec; deletearray .@et;
		.@c = query_sql("SELECT account_id, char_id, team FROM need_bg2_match_player WHERE match_pk = " + .@dup, .@ea, .@ec, .@et);
		if (.@c < 0)
			return -1;
		.@same = (.@c == 6);
		for (.@k = 0; .@k < 6; .@k++) {
			if (!.@same)
				break;
			.@hit = 0;
			for (.@j = 0; .@j < .@c; .@j++)
				if (.@ea[.@j] == .@aid[.@k] && .@ec[.@j] == .@cid[.@k] && .@et[.@j] == .@team[.@k])
					.@hit = 1;
			.@same = .@hit;
		}
		.@done = 0;
		if (.@same) {
			if (query_sql("SELECT COUNT(*) FROM need_bg2_match_done WHERE match_pk = " + .@dup, .@done) < 0)
				return -1;
		}
		if (!.@same || !.@done) {
			$@NBG3P3_PART_WHY$ = (!.@same ? (.@c < 6 ? "missing_players" : "players_mismatch") : "no_completion_mark") + ":match_pk=" + .@dup + ":count=" + .@c;
			callfunc "F_NEED_BG3P3_Log", "P3_PARTIAL match_key=" + .@key$ + " " + $@NBG3P3_PART_WHY$;
			return -3;
		}
		callfunc "F_NEED_BG3P3_Log", "P3_DUPLICATE key=" + .@key$ + " match_pk=" + .@dup + " players=6 done ok";
		return 0;
	}
	if (query_sql("INSERT INTO need_bg2_match (match_key, ext_match_id, `mode`, `source`, started_at, ended_at, week_start, result, end_reason, engaged, hostile_total, rounds, active_ms, win_b, win_r, valid, account_key) VALUES ('"
		+ escape_sql(.@key$) + "', " + .@mid + ", " + $@NBG3P3_MODE + ", '" + escape_sql(.@src$) + "', " + .@t0 + ", " + .@te + ", " + .@ws + ", " + .@kind + ", '" + escape_sql(.@why$) + "', "
		+ .@engaged + ", " + .@hostile + ", 1, " + .@act + ", " + .@sb + ", " + .@sr + ", " + .@valid + ", '" + .@akey$ + "')") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_match");
	.@pk = 0;
	query_sql "SELECT LAST_INSERT_ID()", .@pk;
	if (.@pk <= 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_pk");
	// 검토 신호(점수가 생기는 경기만, 기준값은 2:2 와 같은 설정): 적은 교전 / 같은 6계정 반복 / 상대 팀과 같은 IP / 몰수 승.
	.@sig$ = "";
	.@nsig = 0;
	if (.@valid) {
		if (.@hostile < $@NBG2P3_RV_MINHOST) {
			.@sig$ += "low_hostile:" + .@hostile + " ";
			.@nsig++;
		}
		.@rep = 0;
		if (query_sql("SELECT COUNT(*) FROM need_bg2_match WHERE account_key = '" + .@akey$ + "' AND `source` = 'queue' AND ended_at > " + (.@te - $@NBG2P3_RV_WINDOW) + " AND ended_at <= " + .@te, .@rep) < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_repeat");
		if (.@rep >= $@NBG2P3_RV_REPEAT) {
			.@sig$ += "repeat_group:" + .@rep + " ";
			.@nsig++;
		}
		if (.@sigip) {
			.@sig$ += "shared_ip ";
			.@nsig++;
		}
		if ((.@why$ == "forfeit_blue" || .@why$ == "forfeit_red") && .@engaged) {
			.@sig$ += "desert_gift ";
			.@nsig++;
		}
	}
	.@review = (.@valid && .@nsig >= $@NBG2P3_RV_NEED) ? 1 : 0;
	if (.@sig$ != "") {
		if (query_sql("UPDATE need_bg2_match SET review = " + .@review + ", signals = '" + escape_sql(.@sig$) + "' WHERE match_pk = " + .@pk) < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_signals");
	}
	if (.@review) {
		if (query_sql("INSERT INTO need_bg2_review (match_pk, signals, state, created_at) VALUES (" + .@pk + ", '" + escape_sql(.@sig$) + "', 'HOLD', " + .@te + ")") < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_review");
	}
	.@vals$ = "";
	for (.@k = 0; .@k < 6; .@k++)
		.@vals$ += (.@k ? ", " : "") + "(" + .@pk + ", " + .@aid[.@k] + ", " + .@cid[.@k] + ", " + .@team[.@k] + ", " + .@cls[.@k] + ", " + .@out[.@k] + ", " + .@des[.@k]
			+ ", " + .@pts[.@k] + ", " + .@val[.@k] + ", " + .@win[.@k] + ", " + .@hs[.@k] + ", " + .@ht[.@k] + ", " + .@sp[.@k] + ")";
	if (query_sql("INSERT INTO need_bg2_match_player (match_pk, account_id, char_id, team, class, outcome, deserter, points, valid, win, hostile, hit, support) VALUES " + .@vals$) < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_players");
	if (.@desert) {
		for (.@k = 0; .@k < 6; .@k++) {
			if (!.@des[.@k])
				continue;
			if (query_sql("INSERT IGNORE INTO need_bg2_desertion (account_id, char_id, match_pk, `at`, kind) VALUES (" + .@aid[.@k] + ", " + .@cid[.@k] + ", " + .@pk + ", " + .@te + ", '" + .@dkind$ + "')") < 0)
				return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_desertion");
		}
	}
	// 계정 단계 보상권(검토 보류 경기는 승인 뒤). 이번 경기 점수를 더해 계산하고 상태 6 으로 넣는다.
	.@tflip$ = "";
	if (.@valid && !.@review) {
		for (.@k = 0; .@k < 6; .@k++) {
			.@acc_new[.@k] = -1;
			if (!.@val[.@k])
				continue;
			.@acc_new[.@k] = callfunc("F_NEED_BG3P3_TierSync", .@aid[.@k], .@ws, .@pts[.@k], 6);
			if (.@acc_new[.@k] < 0)
				return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_tier");
			.@acc_pts[.@k] = $@NBG3P3_LASTPTS;
			if ($@NBG3P3_TKEYS$ != "")
				.@tflip$ += (.@tflip$ == "" ? "" : " OR ") + "(account_id = " + .@aid[.@k] + " AND reward_key IN (" + $@NBG3P3_TKEYS$ + "))";
		}
	}
	if (.@valid && (.@flags & 3)) {
		for (.@k = 0; .@k < 6; .@k++) {
			$@NBG3P3_MR_AID[.@k] = .@aid[.@k];
			$@NBG3P3_MR_CID[.@k] = .@cid[.@k];
			$@NBG3P3_MR_OUT[.@k] = .@out[.@k];
			$@NBG3P3_MR_VAL[.@k] = .@val[.@k];
			$@NBG3P3_MR_IP$[.@k] = .@ip$[.@k];
		}
		if (callfunc("F_NEED_BG3P3_MatchRewards", .@pk, .@v3, .@v2, .@flags, .@ws, .@te, .@review) < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_match_rewards");
	}
	// 보상권 공개: 상태 6 → 0(검토 경기의 승패 보상은 7). 이 두 문장과 완료 표식이 트랜잭션의 끝이다.
	if (.@valid && (.@flags & 3)) {
		if (query_sql("UPDATE need_bg2_reward SET status = " + (.@review ? 7 : 0) + " WHERE `mode` = " + $@NBG3P3_MODE + " AND status = 6 AND reward_kind IN ('MCASH', 'MITEM') AND reward_key = '" + .@pk + "'") < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_open_match");
	}
	if (.@tflip$ != "") {
		if (query_sql("UPDATE need_bg2_reward SET status = 0 WHERE `mode` = " + $@NBG3P3_MODE + " AND status = 6 AND reward_kind = 'TIERI' AND week_start = " + .@ws + " AND (" + .@tflip$ + ")") < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_open_tier");
	}
	if (query_sql("INSERT INTO need_bg2_match_done (match_pk, done_at) VALUES (" + .@pk + ", " + gettimetick(2) + ")") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_done");
	if (callfunc("F_NEED_BG2P3_Exec", "COMMIT", "bg3_apply_commit") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_apply_commit");
	callfunc "F_NEED_BG3P3_Log", "P3_LEDGER match=" + .@mid + " match_pk=" + .@pk + " src=" + .@src$ + " kind=" + .@kind + " why=" + .@why$ + " week=" + .@ws
		+ " score=" + .@sb + ":" + .@sr + " engaged=" + .@engaged + " hostile=" + .@hostile + " valid=" + .@valid + " review=" + .@review + " signals=" + .@sig$
		+ " pts=" + .@pts[0] + "/" + .@pts[1] + "/" + .@pts[2] + "/" + .@pts[3] + "/" + .@pts[4] + "/" + .@pts[5]
		+ " deserters=" + .@des[0] + .@des[1] + .@des[2] + .@des[3] + .@des[4] + .@des[5] + " desert_rec=" + .@desert + " flags=" + .@flags;
	if (!.@notify)
		return 1;
	// 승패 직접 캐시: 접속 중인 계정은 공용 지급 함수로 바로(수령 스위치는 2:2 와 공용). 접속하지 않았으면 다음 로그인 / 보상 NPC.
	if (.@valid && (.@flags & 1) && !.@review && $NEED_BG2_P3_MCASH)
		callfunc "F_NEED_BG2P3_PayMatchCash", .@pk;
	for (.@k = 0; .@k < 6; .@k++) {
		if (!.@queue)
			break;
		if (.@des[.@k]) {
			if (.@desert)
				callfunc "F_NEED_BG3P3_TellSanction", .@aid[.@k], .@cid[.@k], .@te;
			continue;
		}
		if (!.@valid) {
			if (.@kind <= 2 && !.@engaged)
				callfunc "F_NEED_BG3_Tell", .@aid[.@k], .@cid[.@k], "[3:3 룬 거점전] 실제 교전이 없어 유효 경기로 인정되지 않았습니다(0점).",
					"[3v3 Rune Control] No real combat took place, so this is not a valid match (0 points).";
			continue;
		}
		if (.@review) {
			callfunc "F_NEED_BG3_Tell", .@aid[.@k], .@cid[.@k], "[3:3 룬 거점전] 이 경기는 운영 검토 대기 중입니다. 점수 +" + .@pts[.@k] + "점은 검토 후 반영됩니다.",
				"[3v3 Rune Control] This match is waiting for a GM review. Your +" + .@pts[.@k] + " points count after the review.";
			continue;
		}
		callfunc "F_NEED_BG3_Tell", .@aid[.@k], .@cid[.@k], "[3:3 룬 거점전] 주간 3:3 점수 +" + .@pts[.@k] + "점 (이번 주 계정 합계 " + .@acc_pts[.@k] + "점)"
			+ (.@acc_new[.@k] > 0 ? " - 새 보상 " + .@acc_new[.@k] + "개! 파요 보상관에서 받으세요." : ""),
			"[3v3 Rune Control] Weekly 3v3 points +" + .@pts[.@k] + " (account total this week " + .@acc_pts[.@k] + ")"
			+ (.@acc_new[.@k] > 0 ? " - " + .@acc_new[.@k] + " new reward(s)! Claim them from the reward NPC in Payon." : "");
	}
	return 1;
}

// 탈주자에게(접속 중이면) 제재 안내(2:2·3:3 합산 횟수).
function	script	F_NEED_BG3P3_TellSanction	{
	.@left = callfunc("F_NEED_BG2P3_Sanction", getarg(0), getarg(2));
	.@cnt = $@NBG2P3_SC_CNT;
	if (.@left > 0)
		callfunc "F_NEED_BG3_Tell", getarg(0), getarg(1), "[3:3 룬 거점전] 경기 이탈이 확정되었습니다(최근 24시간 2:2·3:3 합산 " + .@cnt + "회). 전장 신청 제한: " + callfunc("F_NEED_BG2P3_Left$", .@left),
			"[3v3 Rune Control] Your departure counts as leaving the match (" + .@cnt + " in the last 24 hours, 2v2 and 3v3 together). Sign-ups are blocked for " + callfunc("F_NEED_BG2P3_Left$", .@left);
	else
		callfunc "F_NEED_BG3_Tell", getarg(0), getarg(1), "[3:3 룬 거점전] 경기 이탈이 확정되었습니다(최근 24시간 2:2·3:3 합산 " + .@cnt + "회, 경고). 2회부터 신청이 제한됩니다.",
			"[3v3 Rune Control] Your departure counts as leaving the match (" + .@cnt + " in the last 24 hours, 2v2 and 3v3 together; warning). From the 2nd time, sign-ups are restricted.";
	return;
}

//------------------------------------------------------------
// 주간 랭킹 (3:3 전용: mode 3, 완료 표식, 8계정·20경기·계정당 1자리)
//------------------------------------------------------------

function	script	F_NEED_BG3P3_RankRows	{
	deletearray $@NBG3P3_R_CID; deletearray $@NBG3P3_R_AID; deletearray $@NBG3P3_R_PTS; deletearray $@NBG3P3_R_W; deletearray $@NBG3P3_R_G; deletearray $@NBG3P3_R_NAME$;
	.@n = query_sql("SELECT x.char_id, x.account_id, x.pts, x.w, x.g, COALESCE(c.name, '?') FROM (SELECT p.char_id, p.account_id, SUM(p.points) AS pts, SUM(p.win) AS w, SUM(p.valid) AS g,"
		+ " MAX(m.ended_at) AS reach, MAX(m.match_pk) AS reachpk FROM need_bg2_match_player p JOIN need_bg2_match m ON m.match_pk = p.match_pk" + callfunc("F_NEED_BG3P3_Valid$", getarg(0))
		+ " AND m.review IN (0, 2" + (getarg(1) ? ", 1" : "") + ") GROUP BY p.char_id, p.account_id HAVING g >= " + getarg(2, $@NBG3P3_MINGAMES) + ") x LEFT JOIN `char` c ON c.char_id = x.char_id"
		+ " ORDER BY x.pts DESC, x.w DESC, (x.w * 1.000000000000 / x.g) DESC, x.reach ASC, x.reachpk ASC, x.char_id ASC LIMIT " + getarg(3, 200),
		$@NBG3P3_R_CID, $@NBG3P3_R_AID, $@NBG3P3_R_PTS, $@NBG3P3_R_W, $@NBG3P3_R_G, $@NBG3P3_R_NAME$);
	return .@n;
}

// 순위 보상 대상(계정당 1자리). 결과 $@NBG3P3_AW_CID/AID[1..3], $@NBG3P3_AW_N, $@NBG3P3_AW_ACC. "NA:<계정수>" = 8계정 미만, "ERR" = DB 실패.
function	script	F_NEED_BG3P3_Award$	{
	.@ws = getarg(0);
	$@NBG3P3_AW_N = 0;
	cleararray $@NBG3P3_AW_CID[0], 0, 4;
	cleararray $@NBG3P3_AW_AID[0], 0, 4;
	.@acc = 0;
	if (query_sql("SELECT COUNT(DISTINCT p.account_id) FROM need_bg2_match_player p JOIN need_bg2_match m ON m.match_pk = p.match_pk" + callfunc("F_NEED_BG3P3_Valid$", .@ws)
		+ " AND m.review IN (0, 2)", .@acc) < 0)
		return "ERR";
	$@NBG3P3_AW_ACC = .@acc;
	if (.@acc < $@NBG3P3_MINACC)
		return "NA:" + .@acc;
	.@n = callfunc("F_NEED_BG3P3_RankRows", .@ws, 0);
	if (.@n < 0)
		return "ERR";
	.@sig$ = "AW";
	for (.@i = 0; .@i < .@n && $@NBG3P3_AW_N < 3; .@i++) {
		.@dup = 0;
		for (.@j = 1; .@j <= $@NBG3P3_AW_N; .@j++)
			if ($@NBG3P3_AW_AID[.@j] == $@NBG3P3_R_AID[.@i]) .@dup = 1;
		if (.@dup)
			continue;
		$@NBG3P3_AW_N++;
		$@NBG3P3_AW_AID[$@NBG3P3_AW_N] = $@NBG3P3_R_AID[.@i];
		$@NBG3P3_AW_CID[$@NBG3P3_AW_N] = $@NBG3P3_R_CID[.@i];
		.@sig$ += " " + $@NBG3P3_AW_N + ":" + $@NBG3P3_R_AID[.@i] + ":" + $@NBG3P3_R_CID[.@i];
	}
	return .@sig$;
}

// 주차 확정. commit 0 = 드라이런(첨부된 GM 에게 출력). 반환 0 확정, 1 순위 보상 없음, 2 보류(HOLD), 3 이미 확정, 4 마감 전, -1 DB 실패.
// 그 주차에 미처리 HOLD(3:3 경기, review = 1)가 하나라도 있으면 확정하지 않는다. 2:2 HOLD 는 보지 않는다.
function	script	F_NEED_BG3P3_Finalize	{
	.@ws = getarg(0);
	.@commit = getarg(1);
	.@actor = getarg(2, 0);
	.@reason$ = getarg(3, "scheduled");
	.@now = gettimetick(2);
	.@out = (.@commit == 0);
	if (.@ws != callfunc("F_NEED_BG2P3_WeekStart", .@ws) || .@now < .@ws + $@NBG3P3_WEEK) {
		if (.@out) dispbottom "[3:3 정산] 주차 " + callfunc("F_NEED_BG2P3_Date$", .@ws) + " 는 올바른 주차 시작이 아니거나 아직 마감 전입니다.";
		return 4;
	}
	.@st$ = "";
	if (query_sql("SELECT state FROM need_bg2_week WHERE `mode` = " + $@NBG3P3_MODE + " AND week_start = " + .@ws, .@st$) < 0)
		return -1;
	if (.@st$ == "FINALIZED" || .@st$ == "NO_AWARD") {
		if (.@out) dispbottom "[3:3 정산] 주차 " + callfunc("F_NEED_BG2P3_Date$", .@ws) + " 는 이미 " + .@st$ + " 입니다.";
		return 3;
	}
	.@sigA$ = callfunc("F_NEED_BG3P3_Award$", .@ws);
	if (.@sigA$ == "ERR")
		return -1;
	.@held = -1;
	if (query_sql("SELECT COUNT(*) FROM need_bg2_match WHERE `mode` = " + $@NBG3P3_MODE + " AND week_start = " + .@ws + " AND review = 1", .@held) < 0 || .@held < 0)
		return -1;
	if (.@held > 0) {
		.@detail$ = "held=" + .@held + " (unresolved review hold: ranking waits) excl[" + .@sigA$ + "]";
		if (.@out) {
			dispbottom "[3:3 정산] 보류(HOLD): 해당 주 3:3 HOLD 미처리 " + .@held + "건, 확정 보류.";
			return 2;
		}
		if (.@st$ != "HOLD") {
			query_sql "INSERT INTO need_bg2_week (`mode`, week_start, state, accounts, detail, attempts, decided_at) VALUES (" + $@NBG3P3_MODE + ", " + .@ws + ", 'HOLD', " + $@NBG3P3_AW_ACC
				+ ", '" + escape_sql(.@detail$) + "', 1, 0) ON DUPLICATE KEY UPDATE state = 'HOLD', detail = VALUES(detail), attempts = attempts + 1";
			callfunc "F_NEED_BG2P3_Audit", .@actor, "BG3_WEEK_HOLD", "week=" + .@ws, .@detail$;
			callfunc "F_NEED_BG3P3_Log", "P3_WEEK_HOLD week=" + .@ws + " " + .@detail$;
		}
		return 2;
	}
	.@award = (compare(.@sigA$, "NA:") == 0 && $@NBG3P3_AW_N > 0);
	.@state$ = .@award ? "FINALIZED" : "NO_AWARD";
	.@detail$ = "accounts=" + $@NBG3P3_AW_ACC + " " + .@sigA$ + (.@reason$ != "" ? " reason=" + .@reason$ : "");
	if (.@out) {
		dispbottom "[3:3 정산] 드라이런 주차 " + callfunc("F_NEED_BG2P3_Date$", .@ws) + " → " + .@state$ + " (참가 계정 " + $@NBG3P3_AW_ACC + ")";
		for (.@s = 1; .@s <= $@NBG3P3_AW_N; .@s++)
			dispbottom "[3:3 정산]   " + .@s + "위 aid=" + $@NBG3P3_AW_AID[.@s] + " cid=" + $@NBG3P3_AW_CID[.@s];
		return .@award ? 0 : 1;
	}
	if (!$@NBG3P3_RULES_OK)
		return -1;
	if (callfunc("F_NEED_BG2P3_Begin", "bg3_final_begin") < 0)
		return -1;
	.@st2$ = "";
	if (query_sql("SELECT state FROM need_bg2_week WHERE `mode` = " + $@NBG3P3_MODE + " AND week_start = " + .@ws + " FOR UPDATE", .@st2$) < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_final_lock");
	if (.@st2$ == "FINALIZED" || .@st2$ == "NO_AWARD") {
		query_sql "ROLLBACK";
		return 3;
	}
	.@rv = callfunc("F_NEED_BG3P3_RuleVer", .@ws);
	for (.@s = 1; .@s <= $@NBG3P3_AW_N; .@s++) {
		if (callfunc("F_NEED_BG3P3_ItemTicket", .@ws, $@NBG3P3_AW_AID[.@s], "RANKI", "" + .@s, .@s, $@NBG3P3_AW_CID[.@s], 0, $@NBG3R_F$[.@rv * 9 + 5 + .@s]) < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "bg3_final_ticket");
	}
	if (query_sql("INSERT INTO need_bg2_week (`mode`, week_start, state, accounts, detail, attempts, decided_at) VALUES (" + $@NBG3P3_MODE + ", " + .@ws + ", '" + .@state$ + "', "
		+ $@NBG3P3_AW_ACC + ", '" + escape_sql(.@detail$) + "', 1, " + .@now + ") ON DUPLICATE KEY UPDATE state = VALUES(state), accounts = VALUES(accounts), detail = VALUES(detail), attempts = attempts + 1, decided_at = VALUES(decided_at)") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_final_week");
	callfunc "F_NEED_BG2P3_Audit", .@actor, "BG3_WEEK_" + .@state$, "week=" + .@ws, .@detail$;
	if (callfunc("F_NEED_BG2P3_Exec", "COMMIT", "bg3_final_commit") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_final_commit");
	callfunc "F_NEED_BG3P3_Log", "P3_WEEK_" + .@state$ + " week=" + .@ws + " " + .@detail$;
	return .@award ? 0 : 1;
}

function	script	F_NEED_BG3P3_FinalizeDue	{
	.@now = getarg(0, gettimetick(2));
	deletearray .@w;
	.@n = query_sql("SELECT DISTINCT m.week_start FROM need_bg2_match m LEFT JOIN need_bg2_week w ON w.`mode` = m.`mode` AND w.week_start = m.week_start WHERE m.`mode` = "
		+ $@NBG3P3_MODE + " AND m.valid = 1 AND m.week_start <= " + (.@now - $@NBG3P3_FINAL_AT) + " AND (w.state IS NULL OR w.state NOT IN ('FINALIZED', 'NO_AWARD')) ORDER BY m.week_start LIMIT 8", .@w);
	if (.@n < 0)
		return -1;
	for (.@i = 0; .@i < .@n; .@i++)
		callfunc "F_NEED_BG3P3_Finalize", .@w[.@i], 1, 0, "scheduled";
	return .@n;
}

//------------------------------------------------------------
// GM 검토 결정 / 보상권 회수 (3:3 경기·보상권만)
//------------------------------------------------------------

// act: "approve" / "release" / "void". 성공 1, 거절 0, DB 실패 -1. mode 3 경기가 아니면 아무것도 쓰지 않고 거절한다.
function	script	F_NEED_BG3P3_Decide	{
	.@pk = getarg(0);
	.@act$ = getarg(1);
	.@gm = getcharid(3);
	.@reason$ = getarg(2);
	.@now = gettimetick(2);
	if (callfunc("F_NEED_BG2P3_Begin", "bg3_decide_begin") < 0)
		return -1;
	.@rv = -1; .@ws = 0; .@valid = 0; .@md = 0;
	if (query_sql("SELECT review, week_start, valid, `mode` FROM need_bg2_match WHERE match_pk = " + .@pk + " FOR UPDATE", .@rv, .@ws, .@valid, .@md) < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_decide_lock");
	.@ok = 0;
	if (.@rv < 0)
		.@msg$ = "경기를 찾지 못했습니다.";
	else if (.@md != $@NBG3P3_MODE)
		.@msg$ = "3:3 경기가 아닙니다 (mode=" + .@md + "). 2:2 경기는 @duo2v2p3 로 처리하세요.";
	else if ((.@act$ == "approve" || .@act$ == "release") && .@rv != 1)
		.@msg$ = "검토 보류(HOLD) 상태가 아닙니다 (review=" + .@rv + ").";
	else if (.@act$ == "void" && .@rv == 3)
		.@msg$ = "이미 무효 처리된 경기입니다.";
	else
		.@ok = 1;
	if (!.@ok) {
		query_sql "ROLLBACK";
		dispbottom "[3:3 정산] " + .@msg$;
		return 0;
	}
	.@newrv = (.@act$ == "approve") ? 2 : ((.@act$ == "release") ? 0 : 3);
	.@state$ = (.@act$ == "approve") ? "APPROVED" : ((.@act$ == "release") ? "RELEASED" : "VOIDED");
	if (query_sql("UPDATE need_bg2_match SET review = " + .@newrv + " WHERE match_pk = " + .@pk) < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_decide_match");
	if (query_sql("INSERT INTO need_bg2_review (match_pk, signals, state, gm_aid, reason, created_at, decided_at) VALUES (" + .@pk + ", 'gm', '" + .@state$ + "', " + .@gm + ", '"
		+ escape_sql(.@reason$) + "', " + .@now + ", " + .@now + ") ON DUPLICATE KEY UPDATE state = VALUES(state), gm_aid = VALUES(gm_aid), reason = VALUES(reason), decided_at = VALUES(decided_at)") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_decide_review");
	deletearray .@acc;
	.@n = query_sql("SELECT DISTINCT account_id FROM need_bg2_match_player WHERE match_pk = " + .@pk + " AND valid = 1", .@acc);
	if (.@n < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_decide_accounts");
	.@chg = 0;
	for (.@i = 0; .@i < .@n; .@i++) {
		if (.@newrv == 3)
			.@r = callfunc("F_NEED_BG3P3_TierRevoke", .@acc[.@i], .@ws);
		else
			.@r = callfunc("F_NEED_BG3P3_TierSync", .@acc[.@i], .@ws, 0, 0);
		if (.@r < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "bg3_decide_tier");
		.@chg += .@r;
	}
	// 승패 보상권(보류 7 → 0 / 무효 → 회수 8 + IP 반납)은 match_pk 기준 공용 함수(모드와 상관없는 동작).
	.@r = callfunc("F_NEED_BG2P3_MatchRewardDecide", .@pk, .@newrv);
	if (.@r < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_decide_match_rewards");
	.@chg += .@r;
	.@wst$ = "";
	query_sql "SELECT state FROM need_bg2_week WHERE `mode` = " + $@NBG3P3_MODE + " AND week_start = " + .@ws, .@wst$;
	.@detail$ = "review " + .@rv + "->" + .@newrv + " tickets_changed=" + .@chg + " week_state=" + .@wst$ + " reason=" + .@reason$;
	callfunc "F_NEED_BG2P3_Audit", .@gm, "BG3_REVIEW_" + .@state$, "match_pk=" + .@pk, .@detail$;
	if (callfunc("F_NEED_BG2P3_Exec", "COMMIT", "bg3_decide_commit") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "bg3_decide_commit");
	callfunc "F_NEED_BG3P3_Log", "P3_REVIEW_" + .@state$ + " match_pk=" + .@pk + " by_aid=" + .@gm + " " + .@detail$;
	dispbottom "[3:3 정산] 경기 " + .@pk + " → " + .@state$ + " (보상권 변경 " + .@chg + "건)";
	if (.@wst$ == "FINALIZED" || .@wst$ == "NO_AWARD")
		dispbottom "[3:3 정산] 주의: 이 주차 랭킹은 이미 " + .@wst$ + " 입니다. 순위 보상은 자동으로 다시 쓰지 않습니다.";
	return 1;
}

// GM 회수: 3:3 보상권(mode 3)만. 모드를 먼저 확인하고, 아니면 상태 변경·IP 반납 없이 거절한다.
function	script	F_NEED_BG3P3_Revoke	{
	.@id = getarg(0);
	.@why$ = getarg(1);
	.@gm = getcharid(3);
	.@rk$ = ""; .@rkey$ = ""; .@raid = 0; .@md = -1;
	if (query_sql("SELECT reward_kind, reward_key, account_id, `mode` FROM need_bg2_reward WHERE id = " + .@id, .@rk$, .@rkey$, .@raid, .@md) < 1) {
		dispbottom "[3:3 정산] 보상권 " + .@id + " 을(를) 찾지 못했습니다.";
		return 0;
	}
	if (.@md != $@NBG3P3_MODE) {
		dispbottom "[3:3 정산] 3:3 보상권이 아닙니다 (mode=" + .@md + "). 2:2 보상권은 @duo2v2p3 로 처리하세요.";
		return 0;
	}
	query_sql "UPDATE need_bg2_reward SET status = 8, note = '" + escape_sql("revoked by " + .@gm + ": " + .@why$) + "' WHERE id = " + .@id + " AND `mode` = " + $@NBG3P3_MODE + " AND status IN (0, 7)";
	.@rc = 0;
	query_sql "SELECT ROW_COUNT()", .@rc;
	if (.@rc)
		callfunc "F_NEED_BG2P3_Audit", .@gm, "BG3_TICKET_REVOKE", "id=" + .@id, .@why$;
	if (.@rc && .@rk$ == "MCASH") {
		.@ip$ = ""; .@dk = 0; .@al = 0;
		if (query_sql("SELECT ip, day_key, allocated FROM need_bg2_ipcash_alloc WHERE match_pk = " + atoi(.@rkey$) + " AND account_id = " + .@raid + " AND state IN (0, 1)", .@ip$, .@dk, .@al) > 0) {
			query_sql "UPDATE need_bg2_ipcash_day SET used = IF(used >= " + .@al + ", used - " + .@al + ", 0) WHERE ip = '" + escape_sql(.@ip$) + "' AND day_key = " + .@dk;
			query_sql "UPDATE need_bg2_ipcash_alloc SET state = 3 WHERE match_pk = " + atoi(.@rkey$) + " AND account_id = " + .@raid;
			callfunc "F_NEED_BG3P3_Log", "P3_MCASH_RELEASED match_pk=" + .@rkey$ + " aid=" + .@raid + " day=" + .@dk + " amount=" + .@al + " (GM revoke)";
		}
	}
	dispbottom "[3:3 정산] 보상권 " + .@id + (.@rc ? " 회수" : " : 미수령(0)·검토 대기(7) 상태가 아니라 회수하지 않음");
	return .@rc;
}
