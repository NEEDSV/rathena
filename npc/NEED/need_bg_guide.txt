//=        주간·랭킹 보상 첫 주차 = $@NBG2P3_WREW_FROM (그 전은 테스트 주간: 주간·랭킹 보상권 생성·확정·수령 없음).
//= 정리 : 지난 추석 이벤트 재료(7613 작은 찹쌀반죽 / 1001827 솔잎)를 본인이 확인한 뒤 인벤토리에서만 회수한다(보상 없음).

// 기존 전장(@전장대기) 대기열의 유효 대기자 수: 접속 중 + 같은 캐릭터 + 계정 중복 제외.
function	script	F_NEED_BGUX_RmQueueValid	{
	.@n = 0;
	for (.@i = 0; .@i < $@BG_RM_QUEUE_COUNT; .@i++) {
		.@aid = $@BG_RM_QUEUE_AID[.@i];
		.@cid = $@BG_RM_QUEUE_CID[.@i];
		if (.@aid <= 0 || .@cid <= 0)
			continue;
		if (!isloggedin(.@aid, .@cid))
			continue;
		for (.@j = 0; .@j < .@n; .@j++)
			if (.@seen[.@j] == .@aid) break;
		if (.@j < .@n)
			continue;
		.@seen[.@n] = .@aid;
		.@n++;
	}
	return .@n;
}

// 2:2 / 3:3 대기열의 유효 대기자. getarg(0) 2 = 2:2, 3 = 3:3.
// 반환 = 참가 확인 대상이 아닌 대기자 수, $@NBGUX_RC = 참가 확인 중(접속 중) 인원.
function	script	F_NEED_BGUX_ModeQueue	{
	.@mode = getarg(0);
	.@n = 0;
	.@rc = 0;
	.@qn = (.@mode == 3) ? $@NBG3_Q_N : $@NBG2_Q_N;
	for (.@i = 0; .@i < .@qn; .@i++) {
		.@aid = (.@mode == 3) ? $@NBG3_Q_AID[.@i] : $@NBG2_Q_AID[.@i];
		.@cid = (.@mode == 3) ? $@NBG3_Q_CID[.@i] : $@NBG2_Q_CID[.@i];
		if (.@aid <= 0 || .@cid <= 0)
			continue;
		if (!isloggedin(.@aid, .@cid))
			continue;
		for (.@j = 0; .@j < .@cnt; .@j++)
			if (.@seen[.@j] == .@aid) break;
		if (.@j < .@cnt)
			continue;
		.@seen[.@cnt] = .@aid;
		.@cnt++;
		.@in_rc = 0;
		if (.@mode == 3) {
			if ($@NBG3_RC) {
				for (.@k = 0; .@k < 6; .@k++)
					if ($@NBG3_RC_AID[.@k] == .@aid) .@in_rc = 1;
			}
		}
		else if ($@NBG2_RC) {
			for (.@k = 0; .@k < 4; .@k++)
				if ($@NBG2_RC_AID[.@k] == .@aid) .@in_rc = 1;
		}
		if (.@in_rc)
			.@rc++;
		else
			.@n++;
	}
	$@NBGUX_RC = .@rc;
	return .@n;
}

// 주간·랭킹 보상 정책 한 줄(현재 주차 기준). 2:2 정산 파일이 없으면 빈 문자열.
function	script	F_NEED_BGUX_WeekPolicy$	{
	if (!$@NBG2P3_LOADED)
		return "";
	.@ws = callfunc("F_NEED_BG2P3_WeekStart", gettimetick(2));
	if (.@ws < $@NBG2P3_WREW_FROM)
		return needtr("^FF0000테스트 주간^000000 · 주간 달성·순위 보상 없음. 시작: ",
			"^FF0000Test week^000000 · no weekly tier or rank rewards. Starts: ")
			+ callfunc("F_NEED_BG2P3_Date$", $@NBG2P3_WREW_FROM) + " KST";
	return needtr("이번 주부터 주간 달성·순위 보상을 집계합니다.",
		"Weekly tier and ranking rewards count from this week.");
}

// 메뉴의 "바로 신청": 원래 명령어 이벤트를 예약하고 대화창을 닫는다.
// 대화 중 doevent 는 엔진이 대기열에 넣었다가 이 스크립트가 끝난 뒤(0.1초) 실행한다.
function	script	F_NEED_BGUX_Apply	{
	.@ev$ = getarg(0);
	.@cmd$ = getarg(1);
	mes needtr("대화창을 닫으면 ^0055FF" + .@cmd$ + "^000000 명령어와 같은 절차로 신청합니다.",
		"When this window closes you apply exactly as with ^0055FF" + .@cmd$ + "^000000.");
	mes needtr("결과는 채팅창에 표시됩니다.", "The result is shown in the chat window.");
	close2;
	debugmes "[NEED_BGUX] APPLY aid=" + getcharid(3) + " cid=" + getcharid(0) + " event=" + .@ev$;
	doevent .@ev$;
	end;
}

// 기존 전장 bat_room 이동(마롤 전투 모집관과 같은 처리).
function	script	F_NEED_BGUX_GoBatRoom	{
	getmapxy(.@m$, .@x, .@y, BL_PC);
	if (.@m$ == "prontera") bat_return = 1;
	else if (.@m$ == "moc_ruins") bat_return = 2;
	else if (.@m$ == "aldebaran") bat_return = 3;
	else if (.@m$ == "geffen") bat_return = 4;
	else if (.@m$ == "payon") bat_return = 5;
	else if (.@m$ == "lighthalzen") bat_return = 6;
	else if (.@m$ == "rachel") bat_return = 7;
	else if (.@m$ == "morocc") bat_return = 8;
	else bat_return = 1;
	close2;
	warp "bat_room", 154, 150;
	end;
}

function	script	F_NEED_BGUX_Queues	{
	.@n$ = getarg(0);
	mes .@n$;
	mes needtr("^0055FF대기 현황^000000", "^0055FFQueue status^000000");

	// 기존 전장
	// 2026-11-12 통합 모집: 추가 모집·마감 중에는 모집 명단 인원(감시 NPC 가 1초마다 이탈자를 뺀 값), 그 밖에는 대기열 유효 인원
	if ($@BG_RM_STATE == 1 || $@BG_RM_STATE == 2) {
		.@q = $@BG_RM_COUNT;
		mes needtr("기존 전장: ^0055FF" + .@q + "명^000000", "Classic BG: ^0055FF" + .@q + "^000000")
			+ needtr(" (최대 " + (($@BG_RM_MAX_PLAYERS > 0) ? $@BG_RM_MAX_PLAYERS : 48) + "명)", " (max " + (($@BG_RM_MAX_PLAYERS > 0) ? $@BG_RM_MAX_PLAYERS : 48) + ")");
	}
	else {
		.@q = callfunc("F_NEED_BGUX_RmQueueValid");
		mes needtr("기존 전장: ^0055FF" + .@q + "명^000000", "Classic BG: ^0055FF" + .@q + "^000000")
			+ needtr(" (시작 " + (($@BG_RM_MIN_PLAYERS > 0) ? $@BG_RM_MIN_PLAYERS : 8) + "명)", " (starts at " + (($@BG_RM_MIN_PLAYERS > 0) ? $@BG_RM_MIN_PLAYERS : 8) + ")");
	}
	mes "  ^777777" + callfunc("F_NEED_BGUX_RmStateLine$") + "^000000";

	// 3:3
	if ($@NBG3_LOADED) {
		.@q = callfunc("F_NEED_BGUX_ModeQueue", 3);
		.@rc = $@NBGUX_RC;
		.@st$ = "";
		if (!$NEED_BG3_QUEUE_OPEN)
			.@st$ = needtr(" / 대기열 닫힘", " / queue closed");
		if ($@NBG3_STATE >= 2 && $@NBG3_STATE <= 6)
			.@st$ += needtr(" / 경기 진행 중", " / match running");
		if (.@rc)
			.@st$ += needtr(" / 참가 확인 중 " + .@rc + "명", " / " + .@rc + " in ready check");
		mes needtr("3:3 룬: ^0055FF" + .@q + "명^000000 (시작 6명)", "3v3 Rune: ^0055FF" + .@q + "^000000 (starts at 6)") + .@st$;
	}
	else
		mes needtr("3:3 룬: 운영하지 않음", "3v3 Rune: not running");

	// 2:2
	if ($@NBG2_MAP$ != "") {
		.@q = callfunc("F_NEED_BGUX_ModeQueue", 2);
		.@rc = $@NBGUX_RC;
		.@st$ = "";
		if (!$NEED_BG2_QUEUE_OPEN)
			.@st$ = needtr(" / 대기열 닫힘", " / queue closed");
		if ($@NBG2_STATE >= 2 && $@NBG2_STATE <= 6)
			.@st$ += needtr(" / 경기 진행 중", " / match running");
		if (.@rc)
			.@st$ += needtr(" / 참가 확인 중 " + .@rc + "명", " / " + .@rc + " in ready check");
		mes needtr("2:2 듀오: ^0055FF" + .@q + "명^000000 (시작 4명)", "2v2 Duo: ^0055FF" + .@q + "^000000 (starts at 4)") + .@st$;
	}
	else
		mes needtr("2:2 듀오: 운영하지 않음", "2v2 Duo: not running");

	mes needtr("^777777접속 중인 대기자만 표시합니다. 입장·참가 확정 인원은 제외합니다.^000000", "^777777Online queued players only. Confirmed and in-match players are excluded.^000000");
	next;
	clear;
	return;
}

function	script	F_NEED_BGUX_Rewards	{
	.@n$ = getarg(0);
	while (1) {
		mes .@n$;
		mes needtr("^0055FF점수·보상^000000", "^0055FFPoints & rewards^000000");
		mes needtr("보상 수령은 아래 '보상 받기'에서 진행합니다.", "Use 'Claim rewards' to collect available rewards.");
		.@pol$ = callfunc("F_NEED_BGUX_WeekPolicy$");
		if (.@pol$ != "") mes .@pol$;
		.@m = select(needtr("내 점수 확인:보상 받기:주간 순위:기존 전장 보상:이벤트 재료 정리:뒤로", "My points:Claim rewards:Weekly rankings:Classic BG rewards:Clear event materials:Back"));
		clear;
		switch (.@m) {
		case 1:
			.@s = select(needtr("2대2 점수:3대3 점수:뒤로", "2v2 points:3v3 points:Back"));
			if (.@s == 3) break;
			if ((.@s == 1 && !$@NBG2P3_LOADED) || (.@s == 2 && !$@NBG3P3_LOADED)) {
				mes needtr("현재 이 종목의 점수를 조회할 수 없습니다.", "Points for this mode are currently unavailable.");
				next;
				clear;
				break;
			}
			mes .@n$;
			mes needtr("창을 닫으면 채팅창에 점수를 표시합니다.", "Your points will appear in chat when this window closes.");
			close2;
			if (.@s == 1) doevent "NEED_BG2P3::OnCmdPoints";
			else doevent "NEED_BG3P3::OnCmdPoints";
			end;
		case 2:
			if (!$@NBG2P3_LOADED) {
				mes needtr("현재 보상관을 이용할 수 없습니다.", "The reward officer is currently unavailable.");
				next;
				clear;
				break;
			}
			callfunc "F_NEED_BG2P3_Npc";
			end;
		case 3:
			.@s = select(needtr("2대2 순위:3대3 순위:뒤로", "2v2 ranking:3v3 ranking:Back"));
			if (.@s == 1) callfunc "F_NEED_BGUX_Ranking", .@n$, 2;
			else if (.@s == 2) callfunc "F_NEED_BGUX_Ranking", .@n$, 3;
			break;
		case 4:
			mes .@n$;
			mes needtr("^0055FF기존 전장 보상^000000", "^0055FFClassic BG rewards^000000");
			mes needtr("경기 종료 후 휘장·금화·대전 승리의 증표를 받습니다.", "Receive badges, gold coins and Tokens of Victory War after the match.");
			mes needtr("보유 전장 포인트: ", "Your BG points: ") + #BATTLEGROUND_POINT;
			mes needtr("교환 상점은 전장 대기실에 있습니다. 기존 전장은 주간 순위 보상이 없습니다.", "Exchange shops are in the waiting room. Classic BG has no weekly ranking rewards.");
			next;
			.@s = select(needtr("보상표 보기:대기실로 이동:뒤로", "Reward table:Go to waiting room:Back"));
			clear;
			if (.@s == 1) callfunc "F_BG_Reward";
			if (.@s == 2) callfunc "F_NEED_BGUX_GoBatRoom";
			break;
		case 5:
			callfunc "F_NEED_BGUX_EventMaterial", .@n$;
			break;
		default:
			return;
		}
	}
}

//------------------------------------------------------------
// 3:3 주간 순위 조회 (읽기 전용)
// 2:2 보상관의 주간 순위 화면(F_NEED_BG2P3_Npc case 6)과 같은 형식. 순위 계산은 3:3 원본 F_NEED_BG3P3_RankRows
// (mode 3, 완료 표식 need_bg2_match_done 이 있는 유효 경기만, 검토 보류 제외, 점수 > 승 > 승률 > 먼저 도달 > 경기 PK > 캐릭터 ID)
// 를 그대로 부른다. 지난 주 확정 상태·순위 보상권은 need_bg2_week / need_bg2_reward 를 SELECT 만 한다.
// 정산·확정·보상권 생성은 하지 않는다. "조회 순위"와 "보상 대상 순위"를 문구로 나눈다.
//------------------------------------------------------------
// 이전 호출과의 호환. 순위 계산/보상 확정은 변경하지 않는다.
function	script	F_NEED_BGUX_Rank3	{
	callfunc "F_NEED_BGUX_Ranking", getarg(0), 3;
	return;
}

function	script	F_NEED_BGUX_Ranking	{
	.@n$ = getarg(0);
	.@mode = getarg(1);
	mes .@n$;
	if (!$@NBG2P3_LOADED || (.@mode == 3 && !$@NBG3P3_LOADED)) {
		mes needtr("현재 순위를 조회할 수 없습니다.", "Rankings are currently unavailable.");
		next;
		clear;
		return;
	}
	.@ws = callfunc("F_NEED_BG2P3_WeekStart", gettimetick(2));
	if (.@mode == 3) {
		.@r = callfunc("F_NEED_BG3P3_RankRows", .@ws, 0, 1, 10);
		.@week = $@NBG3P3_WEEK;
		.@min = $@NBG3P3_MINGAMES;
		.@minacc = $@NBG3P3_MINACC;
	}
	else {
		.@r = callfunc("F_NEED_BG2P3_RankRows", .@ws, 0, 1, 10);
		.@week = $@NBG2P3_WEEK;
		.@min = $@NBG2P3_MINGAMES;
		.@minacc = $@NBG2P3_MINACC;
	}
	mes needtr("^0055FF" + .@mode + "대" + .@mode + " 이번 주 순위^000000", "^0055FF" + .@mode + "v" + .@mode + " weekly ranking^000000");
	mes needtr("현재 점수 기준 상위 10명입니다. 최종 보상 순위와 다를 수 있습니다.", "Top 10 by current points. Final reward ranks may differ.");
	if (.@r <= 0) mes needtr("아직 기록이 없습니다.", "No records yet.");
	for (.@i = 0; .@i < .@r; .@i++) {
		.@name$ = (.@mode == 3) ? $@NBG3P3_R_NAME$[.@i] : $@NBG2P3_R_NAME$[.@i];
		.@pts = (.@mode == 3) ? $@NBG3P3_R_PTS[.@i] : $@NBG2P3_R_PTS[.@i];
		.@games = (.@mode == 3) ? $@NBG3P3_R_G[.@i] : $@NBG2P3_R_G[.@i];
		mes (.@i + 1) + ". " + .@name$ + "  " + .@pts + needtr("점", " pts") + (.@games < .@min ? needtr(" (경기 수 미달)", " (not enough matches)") : "");
	}
	if (.@ws < $@NBG2P3_WREW_FROM)
		mes needtr("^FF0000테스트 주간에는 순위 보상이 없습니다.^000000", "^FF0000No ranking rewards during test weeks.^000000");
	else
		mes needtr("보상 조건: 유효 " + .@min + "경기, 주간 참가 계정 " + .@minacc + "개 이상. 다음 주 수요일 오전 4시 확정 후 1~3위가 대상입니다.", "Reward eligibility: " + .@min + " valid games and " + .@minacc + " participating accounts. Ranks 1-3 are settled next Wednesday at 04:00 KST.");
	next;
	mes .@n$;
	.@pw = .@ws - .@week;
	.@pst$ = "";
	query_sql "SELECT state FROM need_bg2_week WHERE `mode` = " + .@mode + " AND week_start = " + .@pw, .@pst$;
	mes needtr("^0055FF지난주 확정 보상^000000", "^0055FFLast week's settled rewards^000000");
	mes callfunc("F_NEED_BG2P3_Date$", .@pw) + " KST";
	if (.@pw < $@NBG2P3_WREW_FROM) mes needtr("테스트 주간 · 순위 보상 없음", "Test week · no ranking rewards");
	else mes (.@pst$ == "" ? needtr("아직 확정되지 않았습니다.", "Not settled yet.") : .@pst$);
	.@c = query_sql("SELECT r.rank_slot, COALESCE(c.name, '?'), r.amount, r.id, r.reward_kind FROM need_bg2_reward r LEFT JOIN `char` c ON c.char_id = r.char_id WHERE r.`mode` = " + .@mode
		+ " AND r.week_start = " + .@pw + " AND r.reward_kind IN ('RANK', 'RANKI') ORDER BY r.rank_slot", .@rs, .@rn$, .@ra, .@ri, .@rk$);
	for (.@i = 0; .@i < .@c; .@i++)
		mes .@rs[.@i] + needtr("위 ", ". ") + .@rn$[.@i] + " - " + callfunc("F_NEED_BG2P3_TicketText$", .@ri[.@i], .@rk$[.@i], .@ra[.@i]);
	mes needtr("받을 보상이 있으면 점수·보상의 '보상 받기'를 이용하세요.", "Use 'Claim rewards' under Points & rewards to collect available rewards.");
	next;
	clear;
	return;
}

//------------------------------------------------------------
// 지난 추석 이벤트 재료 정리 (7613 작은 찹쌀반죽 / 1001827 솔잎)
// 본인이 수량을 확인하고 고른 경우에만, 인벤토리에 있는 것만 회수한다. 창고·카트·길드 창고는 건드리지 않는다.
// 추석 제작/교환 NPC 가 내려가 쓸 곳이 없는 재료다. 보상은 없다. 두 아이템은 원래 버리기·판매도 막혀 있지 않다.
//------------------------------------------------------------
function	script	F_NEED_BGUX_EventMaterial	{
	.@n$ = getarg(0);
	setarray .@id[0], 7613, 1001827;
	mes .@n$;
	mes needtr("^0055FF지난 추석 이벤트 재료 정리^000000", "^0055FFClear old Chuseok event materials^000000");
	mes needtr("추석 송편 이벤트가 끝나 더 이상 쓸 곳이 없는 재료입니다.", "The Chuseok rice cake event is over, so these materials have no use any more.");
	.@total = 0;
	for (.@i = 0; .@i < 2; .@i++) {
		.@c[.@i] = countitem(.@id[.@i]);
		.@total += .@c[.@i];
		mes "- " + getitemname(.@id[.@i]) + " : " + .@c[.@i] + needtr("개", "");
	}
	mes needtr("^777777인벤토리만 확인합니다. 창고·카트에 있는 것은 인벤토리로 옮긴 뒤 다시 오세요.^000000", "^777777Only your inventory is checked. Move any in storage or your cart to the inventory first.^000000");
	if (.@total <= 0) {
		mes needtr("정리할 재료가 없습니다.", "You have nothing to clear.");
		next;
		clear;
		return;
	}
	mes needtr("^FF0000회수한 재료는 되돌릴 수 없고, 보상도 없습니다.^000000", "^FF0000Cleared materials cannot be restored and give no reward.^000000");
	if (select(needtr("모두 회수", "Clear them all") + ":" + needtr("취소", "Cancel")) != 1) {
		clear;
		return;
	}
	clear;
	mes .@n$;
	for (.@i = 0; .@i < 2; .@i++) {
		.@now = countitem(.@id[.@i]);	// 대화 중 수량이 바뀌었을 수 있으니 지금 수량으로 회수한다
		if (.@now > 0)
			delitem .@id[.@i], .@now;
		.@done[.@i] = .@now;
	}
	debugmes "[NEED_BGUX] EVENT_MAT_CLEAR aid=" + getcharid(3) + " cid=" + getcharid(0) + " 7613=" + .@done[0] + " 1001827=" + .@done[1];
	mes needtr("정리했습니다.", "Done.") + " " + getitemname(7613) + " " + .@done[0] + ", " + getitemname(1001827) + " " + .@done[1];
	next;
	clear;
	return;
}
