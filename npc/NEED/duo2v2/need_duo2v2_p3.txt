//============================================================
//= 이 파일은 need_duo2v2.txt (Phase 2) 뒤에 로드한다. Phase 2 는 $@NBG2P3_LOADED 일 때만 이 파일의 함수를 부른다
//= (이 파일을 빼도 Phase 2 경기는 그대로 돈다).
//= 스위치 (서버 영구, 기본 모두 0 = 꺼짐. GM 60+ 가 @duo2v2p3 set ... 으로 바꾼다):
//=   $NEED_BG2_P3_ON    원장/점수/단계 보상권 생성/2:2 탈주 제재. 경기 시작(Launch) 때 켜져 있던 경기만 정산한다.
//=   $NEED_BG2_P3_RANK  수요일 04:00 KST 지난 주 랭킹 자동 확정(+ GM 수동 확정).
//=   $NEED_BG2_P3_CLAIM 보상 NPC 캐시 수령(주간 단계·랭킹 캐시 보상권, 규칙 버전 0 방식).
//=   $NEED_BG2_P3_MCASH 승패 직접 캐시: 경기 시작 때 켜져 있던 경기만 할당, 지급(접속 중 즉시 / 다음 로그인 / NPC)도 이 스위치.
//=   $NEED_BG2_P3_MITEM 승패 아이템 보상권: 경기 시작 때 켜져 있던 경기만 생성, NPC 수령도 이 스위치.
//=   $NEED_BG2_P3_WITEM 주간 단계·랭킹 아이템 보상권 NPC 수령(생성은 그 주차 규칙 버전이 아이템 방식일 때).
//= 규칙  : need_duo2v2_p3_rules.txt (점수·승패 캐시/아이템·IP 하루 한도·주간 아이템). 버전 = 적용 주차, 다음 주부터만 추가.
//= 원장  : DB 테이블 need_bg2_* (NEED_Duo_Survival_Phase3_DB_Migration.sql). 스크립트 변수로 보상 원장을 관리하지 않는다.
//= 시간  : 유닉스 초(UTC). 주차 = 월요일 04:00 KST 시작 [start, start+7일). KST = UTC+9 를 상수로 쓴다(서버 시간대 무관).
//=         경기는 종료 시각의 주차에 들어간다. 랭킹 확정 = 주차 시작 + 9일(다음 주 수요일 04:00 KST).
//= 정산  : F_NEED_BG2_Finish (서버 권위, 경기당 한 번) → F_NEED_BG2P3_Settle → F_NEED_BG2P3_Apply (트랜잭션, match_key 유니크).
//=         DB 실패 시 경기 결과를 영구 서버 변수 $NEED_BG2_PEND_H$/P$ 에 남기고(PENDING_RECONCILIATION) 60초마다 재시도한다.
//=         재시도는 한 번에 최대 $@NBG2P3_PEND_BATCH 건(스크립트 실행 한도 안). 이미 있는 경기 키는 참가자 4행이 payload 와
//=         같을 때만 정상 중복으로 보고, 반쪽·불일치·형식 오류는 검토 보관함 $NEED_BG2_PART_H$/P$/R$ 로 옮긴다(삭제하지 않음).
//= 캐시  : #CASHPOINTS (pc_setparam SP_CASHPOINTS → cashlog). 지급 영수증 #NEED_BG2_PAIDSEQ 를 같은 스크립트에서 함께 바꾼다.
//=         두 계정 변수는 스크립트 종료 시 같은 저장 패킷으로 char-server 에 간다. 보상권은 0 PENDING → 1 RESERVED → 2 CLAIMED,
//=         다음 로그인 때 #NEED_BG2_PAIDSEQ 로 RESERVED 를 확정(2)하거나 미지급(0)으로 되돌린다.
//= 변수  : $@NBG2P3_*  임시(리로드 시 다시 만든다).  $@NBG2_P3ARMED / $@NBG2_CB_* / $@NBG2_IP$ / $@NBG2_CLS / $@NBG2_T0 /
//=         $@NBG2_TICK0 / $@NBG2_ACT_MS / $@NBG2_RACT  경기 중 기록(Phase 2 경기 변수 옆).
//=         @NEED_BG2_CB_HOST / HIT / SUP  세션. C++ bg_need_combat_note 가 올린다(맵플래그 need_combatlog).
//=         NEED_BG2_P3 / NEED_BG2_PREDES  캐릭터. 경기 전부터 있던 엔진 탈주자 상태(다른 전장)의 만료 시각.
//=         #NEED_BG2_PAIDSEQ  계정. 마지막으로 지급된 보상권 순번(영수증).
//=         $NEED_BG2_RV_*  검토 신호 기준(운영 결정 전 임시값, 비어 있으면 기본값).
//============================================================

function	script	F_NEED_BG2P3_Init	{
	$@NBG2P3_MODE = 2;
	setarray $@NBG2P3_TIER_PT[0], 10, 30, 60;
	setarray $@NBG2P3_TIER_CASH[0], 10000, 20000, 40000;
	setarray $@NBG2P3_RANK_CASH[1], 50000, 30000, 20000;
	$@NBG2P3_MINGAMES = 20;		// 랭킹 후보: 그 주 유효 경기 20 이상
	$@NBG2P3_MINACC = 6;		// 순위 보상 개설: 그 주 유효 경기 참가 계정 6 이상
	$@NBG2P3_WEEK = 604800;
	$@NBG2P3_FINAL_AT = 777600;	// 주차 시작 + 9일 = 다음 주 수요일 04:00 KST
	// 검토 신호(독립 신호 2개 이상이면 잠정 HOLD). 수치는 운영 확정값이 아니다(보고서 결정 항목).
	$@NBG2P3_RV_MINHOST = $NEED_BG2_RV_MINHOST ? $NEED_BG2_RV_MINHOST : 3;		// 경기 전체 적대 상호작용 수가 이보다 적음
	$@NBG2P3_RV_SHORTMS = $NEED_BG2_RV_SHORTMS ? $NEED_BG2_RV_SHORTMS : 15000;	// 라운드 평균 진행 시간이 이보다 짧음
	$@NBG2P3_RV_REPEAT = $NEED_BG2_RV_REPEAT ? $NEED_BG2_RV_REPEAT : 4;			// 같은 4계정 경기가 창 안에 이 수 이상
	$@NBG2P3_RV_WINDOW = $NEED_BG2_RV_WINDOW ? $NEED_BG2_RV_WINDOW : 3600;
	$@NBG2P3_RV_NEED = $NEED_BG2_RV_NEED ? $NEED_BG2_RV_NEED : 2;
	$@NBG2P3_PEND_MAX = 64;
	$@NBG2P3_PEND_BATCH = 4;	// 재시도 한 번의 단위(일반 정산 1 = 분기 약 240회, 승패 보상 정산 2). 엔진 한도 check_gotocount 2048
	$@NBG2P3_LOADED = 1;
	callfunc "F_NEED_BG2P3_RulesLoad";
	return;
}

// 주차 시작(월요일 04:00 KST)의 유닉스 초. t + 9시간 - 4시간 = t + 18000 을 하루 단위로 나누면 1970-01-01(목) 기준 일수.
function	script	F_NEED_BG2P3_WeekStart	{
	.@d = (getarg(0) + 18000) / 86400;
	.@wd = (.@d + 3) % 7;	// 월 0 .. 일 6
	return (.@d - .@wd) * 86400 - 18000;
}

// 유닉스 초 → "YYYY-MM-DD HH:MM" (KST).
function	script	F_NEED_BG2P3_Date$	{
	.@t = getarg(0) + 32400;
	.@days = .@t / 86400;
	.@sec = .@t % 86400;
	.@z = .@days + 719468;
	.@era = .@z / 146097;
	.@doe = .@z - .@era * 146097;
	.@yoe = (.@doe - .@doe / 1460 + .@doe / 36524 - .@doe / 146096) / 365;
	.@y = .@yoe + .@era * 400;
	.@doy = .@doe - (365 * .@yoe + .@yoe / 4 - .@yoe / 100);
	.@mp = (5 * .@doy + 2) / 153;
	.@dd = .@doy - (153 * .@mp + 2) / 5 + 1;
	.@m = (.@mp < 10) ? .@mp + 3 : .@mp - 9;
	if (.@m <= 2)
		.@y++;
	.@hh = .@sec / 3600;
	.@mi = (.@sec % 3600) / 60;
	return "" + .@y + "-" + (.@m < 10 ? "0" : "") + .@m + "-" + (.@dd < 10 ? "0" : "") + .@dd + " " + (.@hh < 10 ? "0" : "") + .@hh + ":" + (.@mi < 10 ? "0" : "") + .@mi;
}

// 결과 없는 SQL. 실패하면 -1 과 로그.
function	script	F_NEED_BG2P3_Exec	{
	.@r = query_sql(getarg(0));
	if (.@r < 0)
		callfunc "F_NEED_BG2_Log", "P3_SQL_ERROR at=" + getarg(1, "?");
	return .@r;
}

// 지급·정산 기록용 로그: $NEED_BG2_LOG_OFF 와 상관없이 항상 남긴다(수동 재처리·대조 근거).
// 트랜잭션 시작. query_sql 은 모든 스크립트가 같은 DB 연결을 쓴다. 이전 스크립트가 트랜잭션 도중 중단되면(엔진 실행 한도 등)
// 그 트랜잭션이 열린 채 남고, 그 뒤 다른 스크립트의 쓰기도 그 안에 쌓인다. 여기서 ROLLBACK 하면 남의 쓰기까지 지우므로
// 명시적으로 COMMIT 하고 크게 기록한다(START TRANSACTION 의 암묵 커밋과 결과는 같지만 흔적이 남는다).
// 그때 커밋된 2:2 반쪽 원장은 재시도 때 F_NEED_BG2P3_Apply 의 참가자 4행 검사(P3_PARTIAL)가 잡는다.
function	script	F_NEED_BG2P3_Begin	{
	.@open = 0;
	if (query_sql("SELECT @@in_transaction", .@open) < 0) {
		callfunc "F_NEED_BG2_Log", "P3_SQL_ERROR at=" + getarg(0) + "_txncheck";
		return -1;
	}
	if (.@open) {
		callfunc "F_NEED_BG2P3_Log", "P3_TXN_LEFTOVER at=" + getarg(0) + " (a transaction was left open on the query_sql connection; committed explicitly to keep other scripts' writes; partial 2v2 rows are caught by P3_PARTIAL)";
		if (query_sql("COMMIT") < 0)
			return -1;
	}
	return callfunc("F_NEED_BG2P3_Exec", "START TRANSACTION", getarg(0));
}

function	script	F_NEED_BG2P3_Log	{
	debugmes "[NEED_BG2] " + getarg(0);
	return;
}

function	script	F_NEED_BG2P3_Rollback	{
	query_sql "ROLLBACK";
	callfunc "F_NEED_BG2_Log", "P3_ROLLBACK at=" + getarg(0);
	return -1;
}

function	script	F_NEED_BG2P3_Audit	{
	query_sql "INSERT INTO need_bg2_audit (`at`, actor_aid, action, target, detail) VALUES (" + gettimetick(2) + ", " + getarg(0) + ", '"
		+ escape_sql(getarg(1)) + "', '" + escape_sql(getarg(2)) + "', '" + escape_sql(getarg(3, "")) + "')";
	return;
}

	$@NBG2R_PD[.@v] = getarg(4);

	$@NBG2R_CL[.@v] = getarg(6);
	$@NBG2R_CD[.@v] = getarg(7);
	$@NBG2R_IW$[.@v] = getarg(8);
	$@NBG2R_IL$[.@v] = getarg(9);
	$@NBG2R_ID$[.@v] = getarg(10);
	$@NBG2R_CAP[.@v] = getarg(11);
	$@NBG2R_WM[.@v] = getarg(12);
	$@NBG2R_T10$[.@v] = getarg(13);
	$@NBG2R_T30$[.@v] = getarg(14);
	$@NBG2R_T60$[.@v] = getarg(15);
	$@NBG2R_R1$[.@v] = getarg(16);
	$@NBG2R_R2$[.@v] = getarg(17);
	$@NBG2R_R3$[.@v] = getarg(18);

	return;
}

// 한 버전의 지문(바뀌었는지 비교). 첫 칸은 적용 주차.
function	script	F_NEED_BG2P3_RuleSig$	{
	.@v = getarg(0);
	return $@NBG2R_EFF[.@v] + "|" + $@NBG2R_PW[.@v] + "/" + $@NBG2R_PL[.@v] + "/" + $@NBG2R_PD[.@v] + "|" + $@NBG2R_CW[.@v] + "/" + $@NBG2R_CL[.@v] + "/" + $@NBG2R_CD[.@v]
		+ "|" + $@NBG2R_IW$[.@v] + "/" + $@NBG2R_IL$[.@v] + "/" + $@NBG2R_ID$[.@v] + "|" + $@NBG2R_CAP[.@v] + "|" + $@NBG2R_WM[.@v]
		+ "|" + $@NBG2R_T10$[.@v] + "/" + $@NBG2R_T30$[.@v] + "/" + $@NBG2R_T60$[.@v] + "|" + $@NBG2R_R1$[.@v] + "/" + $@NBG2R_R2$[.@v] + "/" + $@NBG2R_R3$[.@v];
}

// 아이템 목록 "id:수량,id:수량" → $@NBG2P3_IT_ID[] / $@NBG2P3_IT_N[]. 반환 종류 수(빈 문자열 0), 잘못되면 -1($@NBG2P3_IT_ERR$).
function	script	F_NEED_BG2P3_ParseItems	{
	deletearray $@NBG2P3_IT_ID;
	deletearray $@NBG2P3_IT_N;
	$@NBG2P3_IT_ERR$ = "";
	if (getarg(0) == "")
		return 0;
	explode(.@e$, getarg(0), ",");
	.@n = getarraysize(.@e$);
	if (.@n > 8) {
		$@NBG2P3_IT_ERR$ = "more than 8 items";
		return -1;
	}
	for (.@i = 0; .@i < .@n; .@i++) {
		deletearray .@f$;
		explode(.@f$, .@e$[.@i], ":");
		.@id = atoi(.@f$[0]);
		.@amt = (getarraysize(.@f$) == 2) ? atoi(.@f$[1]) : 0;
		if (getarraysize(.@f$) != 2 || .@id <= 0 || "" + .@id != .@f$[0] || "" + .@amt != .@f$[1]) {
			$@NBG2P3_IT_ERR$ = "bad entry '" + .@e$[.@i] + "'";
			return -1;
		}
		if (getitemname(.@id) == "null") {
			$@NBG2P3_IT_ERR$ = "item " + .@id + " not in the item DB";
			return -1;
		}
		if (.@amt < 1 || .@amt > 30000) {
			$@NBG2P3_IT_ERR$ = "item " + .@id + " amount " + .@amt + " not 1..30000";
			return -1;
		}
		$@NBG2P3_IT_ID[.@i] = .@id;
		$@NBG2P3_IT_N[.@i] = .@amt;
	}
	return .@n;
}

// 규칙을 읽고 검사한다(Init / 리로드). 실패하면 $@NBG2P3_RULES_OK = 0: 새 정산·보상권 생성을 멈춘다(경기 결과는 보관 정산).
// 이미 시작된 버전이 바뀌거나 사라졌는지, 새 버전이 다음 주 월요일 이후 시작인지를 영구 변수 $NEED_BG2_RVSIG$[] 와 비교한다.
function	script	F_NEED_BG2P3_RulesLoad	{
	deletearray $@NBG2R_EFF; deletearray $@NBG2R_PW; deletearray $@NBG2R_PL; deletearray $@NBG2R_PD;
	deletearray $@NBG2R_CW; deletearray $@NBG2R_CL; deletearray $@NBG2R_CD;
	deletearray $@NBG2R_IW$; deletearray $@NBG2R_IL$; deletearray $@NBG2R_ID$;
	deletearray $@NBG2R_CAP; deletearray $@NBG2R_WM;
	deletearray $@NBG2R_T10$; deletearray $@NBG2R_T30$; deletearray $@NBG2R_T60$;
	deletearray $@NBG2R_R1$; deletearray $@NBG2R_R2$; deletearray $@NBG2R_R3$;
	deletearray $@NBG2R_F$;
	$@NBG2R_N = 0;
	$@NBG2R_ERR$ = "";
	callfunc "F_NEED_BG2P3_RulesDef";
	.@now = gettimetick(2);
	.@next = callfunc("F_NEED_BG2P3_WeekStart", .@now) + $@NBG2P3_WEEK;
	if ($@NBG2R_N < 1 || $@NBG2R_EFF[0] != 0)
		$@NBG2R_ERR$ += "version 0 (effective 0) missing; ";
	for (.@v = 0; .@v < $@NBG2R_N; .@v++) {
		if (.@v > 0) {	// rAthena 의 && 는 뒤도 계산하므로 [.@v - 1] 은 블록 안에서만 읽는다
			if ($@NBG2R_EFF[.@v] != callfunc("F_NEED_BG2P3_WeekStart", $@NBG2R_EFF[.@v]) || $@NBG2R_EFF[.@v] <= $@NBG2R_EFF[.@v - 1])
				$@NBG2R_ERR$ += "v" + .@v + " start is not a later Monday 04:00 KST; ";
		}
		if ($@NBG2R_PW[.@v] < 0 || $@NBG2R_PW[.@v] > 10 || $@NBG2R_PL[.@v] < 0 || $@NBG2R_PL[.@v] > 10 || $@NBG2R_PD[.@v] < 0 || $@NBG2R_PD[.@v] > 10)
			$@NBG2R_ERR$ += "v" + .@v + " points not 0..10; ";
		if ($@NBG2R_CW[.@v] < 0 || $@NBG2R_CW[.@v] > 1000000 || $@NBG2R_CL[.@v] < 0 || $@NBG2R_CL[.@v] > 1000000 || $@NBG2R_CD[.@v] < 0 || $@NBG2R_CD[.@v] > 1000000)
			$@NBG2R_ERR$ += "v" + .@v + " match cash not 0..1000000; ";
		if ($@NBG2R_CAP[.@v] < 0 || $@NBG2R_CAP[.@v] > 100000000)
			$@NBG2R_ERR$ += "v" + .@v + " IP daily cap not 0..100000000; ";
		if ($@NBG2R_WM[.@v] != 0 && $@NBG2R_WM[.@v] != 1)
			$@NBG2R_ERR$ += "v" + .@v + " weekly mode not 0/1; ";
		setarray .@l$[0], $@NBG2R_IW$[.@v], $@NBG2R_IL$[.@v], $@NBG2R_ID$[.@v], $@NBG2R_T10$[.@v], $@NBG2R_T30$[.@v], $@NBG2R_T60$[.@v], $@NBG2R_R1$[.@v], $@NBG2R_R2$[.@v], $@NBG2R_R3$[.@v];
		setarray .@ln$[0], "win", "lose", "draw", "tier10", "tier30", "tier60", "rank1", "rank2", "rank3";
		for (.@j = 0; .@j < 9; .@j++) {
			.@c = callfunc("F_NEED_BG2P3_ParseItems", .@l$[.@j]);
			if (.@c < 0) {
				$@NBG2R_ERR$ += "v" + .@v + " " + .@ln$[.@j] + ": " + $@NBG2P3_IT_ERR$ + "; ";
				continue;
			}
			// 정산 때는 목록을 다시 해석하지 않는다(분기 한도): "(%R%, 순번, 아이템, 수량), ..." 를 미리 만들어 둔다.
			.@f$ = "";
			for (.@i = 0; .@i < .@c; .@i++)
				.@f$ += (.@i ? ", " : "") + "(%R%, " + .@i + ", " + $@NBG2P3_IT_ID[.@i] + ", " + $@NBG2P3_IT_N[.@i] + ")";
			$@NBG2R_F$[.@v * 9 + .@j] = .@f$;
		}
		.@sig$ = callfunc("F_NEED_BG2P3_RuleSig$", .@v);
		.@old$ = $NEED_BG2_RVSIG$[.@v];
		if (.@old$ != "" && .@old$ != .@sig$ && $@NBG2R_EFF[.@v] <= .@now)
			$@NBG2R_ERR$ += "v" + .@v + " already started and was changed; ";
		if (.@old$ == "" && .@v > 0 && $@NBG2R_EFF[.@v] < .@next)
			$@NBG2R_ERR$ += "v" + .@v + " is new but starts before next Monday 04:00 KST; ";
	}
	// 지워진 버전: 이미 시작된 것이면 거부.
	.@n = getarraysize($NEED_BG2_RVSIG$);
	for (.@v = $@NBG2R_N; .@v < .@n; .@v++) {
		if ($NEED_BG2_RVSIG$[.@v] == "")
			continue;
		explode(.@sf$, $NEED_BG2_RVSIG$[.@v], "|");
		if (atoi(.@sf$[0]) <= .@now)
			$@NBG2R_ERR$ += "v" + .@v + " already started and was removed; ";
	}
	if ($@NBG2R_ERR$ != "") {
		$@NBG2P3_RULES_OK = 0;
		callfunc "F_NEED_BG2P3_Log", "P3_RULES_INVALID " + $@NBG2R_ERR$ + "(new settlements and reward tickets are held until the rules file is fixed and reloaded)";
		return 0;
	}
	for (.@v = 0; .@v < $@NBG2R_N; .@v++)
		if ($NEED_BG2_RVSIG$[.@v] != callfunc("F_NEED_BG2P3_RuleSig$", .@v))
			$NEED_BG2_RVSIG$[.@v] = callfunc("F_NEED_BG2P3_RuleSig$", .@v);
	for (.@v = $@NBG2R_N; .@v < .@n; .@v++)
		$NEED_BG2_RVSIG$[.@v] = "";
	$@NBG2P3_RULES_OK = 1;
	callfunc "F_NEED_BG2P3_Log", "P3_RULES_OK versions=" + $@NBG2R_N;
	return 1;
}

// 시각 t 가 속한 주차의 규칙 버전.
function	script	F_NEED_BG2P3_RuleVer	{
	.@ws = callfunc("F_NEED_BG2P3_WeekStart", getarg(0));
	for (.@v = $@NBG2R_N - 1; .@v > 0; .@v--)
		if ($@NBG2R_EFF[.@v] <= .@ws)
			return .@v;
	return 0;
}

// 보상권 하나에 아이템 목록을 스냅샷으로 붙인다(트랜잭션 안). frag$ = 규칙 로드 때 만든 VALUES 조각. 반환 1 / 0(없음), 실패 -1.

		return 0;
	if (query_sql("INSERT INTO need_bg2_reward_item (reward_id, seq, item_id, amount) VALUES " + replacestr(getarg(1), "%R%", "" + getarg(0))) < 0)
		return -1;
	return 1;
}

// 아이템 보상권 하나(트랜잭션 안): INSERT IGNORE + 스냅샷. 새로 만들면 1, 이미 있으면 0, 실패 -1.
function	script	F_NEED_BG2P3_ItemTicket	{
	// getarg: ws, aid, kind$, key$, rank_slot(0=없음), cid, status, list$(로그용), frag$(규칙 로드 때 만든 VALUES 조각)

// 승패 보상(트랜잭션 안, Apply 가 부른다). 경기 시작 때 고정한 버전 ver / 스위치 flags(1 직접 캐시, 2 아이템).
// 참가자: $@NBG2P3_MR_AID/CID/OUT/VAL[], $@NBG2P3_MR_IP$[] (OUT 1 승 2 패 3 무, VAL 1 유효).
// 직접 캐시: 경기 시작 때 IP 별로 하루(04:00 KST, 경기 종료 시각 기준) 한도 안에서 할당한다. 같은 경기에서 같은 IP 가 한도를 넘으면
// 요청액 비율로 나누고(내림) 남는 금액은 계정 ID 오름차순으로 1씩. 검토 보류(review) 경기도 할당은 잡아 두고(보류 7) 승인 때 연다.
// 반환 0, 실패 -1.
function	script	F_NEED_BG2P3_MatchRewards	{
	.@pk = getarg(0);
	.@ver = getarg(1);
	.@flags = getarg(2);
	.@ws = getarg(3);
	.@te = getarg(4);
	.@st = getarg(5) ? 7 : 0;
	if (!(.@flags & 3))
		return 0;
	// 계정 ID 오름차순 처리 순서
	setarray .@o[0], 0, 1, 2, 3;
	for (.@a = 0; .@a < 3; .@a++)
		for (.@b = 0; .@b < 3 - .@a; .@b++)
			if ($@NBG2P3_MR_AID[.@o[.@b]] > $@NBG2P3_MR_AID[.@o[.@b + 1]]) {
				.@t = .@o[.@b];
				.@o[.@b] = .@o[.@b + 1];
				.@o[.@b + 1] = .@t;
			}
	if (.@flags & 2) {
		for (.@k = 0; .@k < 4; .@k++) {
			if (!$@NBG2P3_MR_VAL[.@k])
				continue;
			.@out = $@NBG2P3_MR_OUT[.@k];
			.@l$ = (.@out == 1) ? $@NBG2R_IW$[.@ver] : ((.@out == 2) ? $@NBG2R_IL$[.@ver] : $@NBG2R_ID$[.@ver]);
			.@j = (.@out == 1) ? 0 : ((.@out == 2) ? 1 : 2);
			if (callfunc("F_NEED_BG2P3_ItemTicket", .@ws, $@NBG2P3_MR_AID[.@k], "MITEM", "" + .@pk, 0, $@NBG2P3_MR_CID[.@k], .@st, .@l$, $@NBG2R_F$[.@ver * 9 + .@j]) < 0)
				return -1;
		}
	}
	.@cap = $@NBG2R_CAP[.@ver];
	if (!(.@flags & 1) || .@cap <= 0)
		return 0;
	.@day = (.@te + 18000) / 86400;
	for (.@k = 0; .@k < 4; .@k++) {
		.@out = $@NBG2P3_MR_OUT[.@k];
		.@req[.@k] = !$@NBG2P3_MR_VAL[.@k] ? 0 : ((.@out == 1) ? $@NBG2R_CW[.@ver] : ((.@out == 2) ? $@NBG2R_CL[.@ver] : $@NBG2R_CD[.@ver]));
		.@alloc[.@k] = 0;
		.@done[.@k] = 0;
	}
	for (.@x = 0; .@x < 4; .@x++) {
		.@k = .@o[.@x];
		if (.@done[.@k] || .@req[.@k] <= 0)
			continue;
		.@ip$ = $@NBG2P3_MR_IP$[.@k];
		if (.@ip$ == "") {
			// 시작 IP 를 모르면 캐시만 보류(지급 없음). 점수·아이템은 그대로.
			.@done[.@k] = 1;
			if (query_sql("INSERT IGNORE INTO need_bg2_ipcash_alloc (match_pk, account_id, char_id, ip, day_key, requested, allocated, state, created_at) VALUES ("
				+ .@pk + ", " + $@NBG2P3_MR_AID[.@k] + ", " + $@NBG2P3_MR_CID[.@k] + ", '', " + .@day + ", " + .@req[.@k] + ", 0, 4, " + gettimetick(2) + ")") < 0)
				return -1;
			callfunc "F_NEED_BG2P3_Log", "P3_MCASH_NOIP match_pk=" + .@pk + " aid=" + $@NBG2P3_MR_AID[.@k] + " requested=" + .@req[.@k] + " (cash withheld, points/items unchanged)";
			continue;
		}
		// 같은 IP 무리(계정 ID 오름차순)
		deletearray .@g;
		.@gn = 0;
		.@tot = 0;
		for (.@y = .@x; .@y < 4; .@y++) {
			.@j = .@o[.@y];
			if (.@done[.@j] || .@req[.@j] <= 0 || $@NBG2P3_MR_IP$[.@j] != .@ip$)
				continue;
			.@g[.@gn] = .@j;
			.@gn++;
			.@tot += .@req[.@j];
			.@done[.@j] = 1;
		}
		.@used = 0;
		if (query_sql("SELECT used FROM need_bg2_ipcash_day WHERE ip = '" + escape_sql(.@ip$) + "' AND day_key = " + .@day + " FOR UPDATE", .@used) < 0)
			return -1;
		.@rem = .@cap - .@used;
		if (.@rem < 0)
			.@rem = 0;
		.@sum = 0;
		for (.@y = 0; .@y < .@gn; .@y++) {
			.@j = .@g[.@y];
			.@alloc[.@j] = (.@tot <= .@rem) ? .@req[.@j] : .@req[.@j] * .@rem / .@tot;
			.@sum += .@alloc[.@j];
		}
		.@left = ((.@tot <= .@rem) ? .@tot : .@rem) - .@sum;
		for (.@y = 0; .@y < .@gn && .@left > 0; .@y++) {
			.@j = .@g[.@y];
			if (.@alloc[.@j] < .@req[.@j]) {
				.@alloc[.@j]++;
				.@sum++;
				.@left--;
			}
		}
		if (.@sum > 0) {
			if (query_sql("INSERT INTO need_bg2_ipcash_day (ip, day_key, used, updated_at) VALUES ('" + escape_sql(.@ip$) + "', " + .@day + ", " + .@sum + ", " + gettimetick(2)
				+ ") ON DUPLICATE KEY UPDATE used = used + " + .@sum + ", updated_at = VALUES(updated_at)") < 0)
				return -1;
		}
		for (.@y = 0; .@y < .@gn; .@y++) {
			.@j = .@g[.@y];
			if (query_sql("INSERT INTO need_bg2_ipcash_alloc (match_pk, account_id, char_id, ip, day_key, requested, allocated, state, created_at) VALUES ("
				+ .@pk + ", " + $@NBG2P3_MR_AID[.@j] + ", " + $@NBG2P3_MR_CID[.@j] + ", '" + escape_sql(.@ip$) + "', " + .@day + ", " + .@req[.@j] + ", " + .@alloc[.@j] + ", "
				+ (.@st ? 1 : 0) + ", " + gettimetick(2) + ")") < 0)
				return -1;
			if (.@alloc[.@j] > 0) {
				if (query_sql("INSERT INTO need_bg2_reward (`mode`, week_start, account_id, reward_kind, reward_key, char_id, amount, status, created_at) VALUES ("
					+ $@NBG2P3_MODE + ", " + .@ws + ", " + $@NBG2P3_MR_AID[.@j] + ", 'MCASH', '" + .@pk + "', " + $@NBG2P3_MR_CID[.@j] + ", " + .@alloc[.@j] + ", " + .@st + ", " + gettimetick(2) + ")") < 0)
					return -1;
			}
		}
		callfunc "F_NEED_BG2P3_Log", "P3_MCASH_ALLOC match_pk=" + .@pk + " day=" + .@day + " ip_used_before=" + .@used + " cap=" + .@cap + " requested=" + .@tot + " granted=" + .@sum
			+ " accounts=" + .@gn + (.@st ? " held" : "");
	}
	return 0;
}

// 승패 보상권의 검토 결정(트랜잭션 안, Decide 가 부른다). newrv 2/0 = 승인/해제(보류 7 → 0), 3 = 무효(미수령·보류 → 회수 8 + IP 한도 반납,
// 이미 지급·예약 → 수동 정산 9, 한도 그대로). 반환 바뀐 보상권 수, 실패 -1.
function	script	F_NEED_BG2P3_MatchRewardDecide	{
	.@pk = getarg(0);
	.@nr = getarg(1);
	.@w$ = " WHERE reward_kind IN ('MCASH', 'MITEM') AND reward_key = '" + .@pk + "'";
	.@chg = 0;
	if (.@nr != 3) {
		if (query_sql("UPDATE need_bg2_reward SET status = 0" + .@w$ + " AND status = 7") < 0)
			return -1;
		query_sql "SELECT ROW_COUNT()", .@chg;
		if (query_sql("UPDATE need_bg2_ipcash_alloc SET state = 0 WHERE match_pk = " + .@pk + " AND state = 1") < 0)
			return -1;
		return .@chg;
	}
	if (query_sql("UPDATE need_bg2_reward SET status = 8, note = 'revoked: match voided, never paid'" + .@w$ + " AND status IN (0, 7)") < 0)
		return -1;
	.@rc = 0;
	query_sql "SELECT ROW_COUNT()", .@rc;
	.@chg += .@rc;
	if (query_sql("UPDATE need_bg2_reward SET note = CONCAT('MANUAL: paid or reserved (status ', status, '), match voided'), status = 9" + .@w$ + " AND status IN (1, 2)") < 0)
		return -1;
	.@rc = 0;
	query_sql "SELECT ROW_COUNT()", .@rc;
	.@chg += .@rc;
	deletearray .@aa; deletearray .@ip$; deletearray .@dk; deletearray .@al;
	.@n = query_sql("SELECT a.account_id, a.ip, a.day_key, a.allocated FROM need_bg2_ipcash_alloc a WHERE a.match_pk = " + .@pk + " AND a.state IN (0, 1)"
		+ " AND NOT EXISTS (SELECT 1 FROM need_bg2_reward r WHERE r.reward_kind = 'MCASH' AND r.reward_key = '" + .@pk + "' AND r.account_id = a.account_id AND r.status = 9)", .@aa, .@ip$, .@dk, .@al);
	if (.@n < 0)
		return -1;
	for (.@i = 0; .@i < .@n; .@i++) {
		if (.@al[.@i] > 0 && query_sql("UPDATE need_bg2_ipcash_day SET used = IF(used >= " + .@al[.@i] + ", used - " + .@al[.@i] + ", 0) WHERE ip = '" + escape_sql(.@ip$[.@i]) + "' AND day_key = " + .@dk[.@i]) < 0)
			return -1;
		if (query_sql("UPDATE need_bg2_ipcash_alloc SET state = 3 WHERE match_pk = " + .@pk + " AND account_id = " + .@aa[.@i]) < 0)
			return -1;
		callfunc "F_NEED_BG2_Log", "P3_MCASH_RELEASED match_pk=" + .@pk + " aid=" + .@aa[.@i] + " day=" + .@dk[.@i] + " amount=" + .@al[.@i];
	}
	return .@chg;
}

// 아이템 보상권 수령(첨부된 플레이어). 1 지급, 0 거절. 무게·칸을 먼저 전부 검사하고, 모자라면 보상권을 그대로 둔다.
// 순서: 0 → 1 (조건부 UPDATE, 순번) → getitem 전부 → 캐릭터 영수증 NEED_BG2_IPAIDSEQ = 순번. 그 캐릭터의 다음 접속 때 2 / 0 으로 확정.
function	script	F_NEED_BG2P3_ClaimItems	{
	.@id = getarg(0);
	.@aid = getcharid(3);
	.@cid = getcharid(0);
	.@k$ = "";
	.@st = -1;
	if (query_sql("SELECT reward_kind, status FROM need_bg2_reward WHERE id = " + .@id + " AND account_id = " + .@aid, .@k$, .@st) < 1 || .@st != 0) {
		dispbottom needtr("[2:2 듀오] 받을 수 있는 보상이 아닙니다(이미 받았거나 없음).", "[2v2 Duo] This reward cannot be claimed (already claimed or not found)."), 0xFF7777;
		return 0;
	}
	if ((.@k$ == "MITEM" && !$NEED_BG2_P3_MITEM) || (.@k$ != "MITEM" && !$NEED_BG2_P3_WITEM)) {
		dispbottom needtr("[2:2 듀오] 지금은 이 보상의 수령을 받지 않습니다.", "[2v2 Duo] This reward cannot be claimed now."), 0xFF7777;
		return 0;
	}
	deletearray .@it;
	deletearray .@am;
	.@n = query_sql("SELECT item_id, amount FROM need_bg2_reward_item WHERE reward_id = " + .@id + " ORDER BY seq", .@it, .@am);
	if (.@n < 1) {
		callfunc "F_NEED_BG2P3_Log", "P3_ITEM_EMPTY id=" + .@id + " (ticket without an item list: GM check)";
		dispbottom needtr("[2:2 듀오] 보상 내용을 확인할 수 없습니다. 운영자에게 문의하세요.", "[2v2 Duo] The reward contents are missing. Please contact a GM."), 0xFF7777;
		return 0;
	}
	for (.@i = 0; .@i < .@n; .@i++) {
		if (getitemname(.@it[.@i]) == "null") {
			callfunc "F_NEED_BG2P3_Log", "P3_ITEM_INVALID id=" + .@id + " item=" + .@it[.@i] + " (not in the item DB now: ticket kept)";
			dispbottom needtr("[2:2 듀오] 지급할 수 없는 아이템이 있어 보류합니다. 운영자에게 문의하세요.", "[2v2 Duo] An item of this reward cannot be given now. Please contact a GM."), 0xFF7777;
			return 0;
		}
	}
	if (!checkweight2(.@it, .@am)) {
		dispbottom needtr("[2:2 듀오] 무게나 소지품 칸이 모자랍니다. 정리한 뒤 다시 받으세요(보상은 그대로 남아 있습니다).",
			"[2v2 Duo] Not enough weight or inventory space. Make room and try again (the reward is kept)."), 0xFF7777;
		return 0;
	}
	.@mx = 0;
	if (query_sql("SELECT COALESCE(MAX(claim_seq), 0) FROM need_bg2_reward WHERE account_id = " + .@aid, .@mx) < 0)
		return 0;
	.@seq = .@mx;
	if (#NEED_BG2_PAIDSEQ > .@seq) .@seq = #NEED_BG2_PAIDSEQ;
	if (NEED_BG2_IPAIDSEQ > .@seq) .@seq = NEED_BG2_IPAIDSEQ;
	.@seq++;
	if (query_sql("UPDATE need_bg2_reward SET status = 1, claim_seq = " + .@seq + ", claim_char = " + .@cid + ", claimed_at = " + gettimetick(2) + " WHERE id = " + .@id + " AND account_id = " + .@aid + " AND status = 0") < 0)
		return 0;
	.@rc = 0;
	query_sql "SELECT ROW_COUNT()", .@rc;
	if (.@rc != 1) {
		callfunc "F_NEED_BG2P3_Log", "P3_CLAIM_RACE id=" + .@id + " aid=" + .@aid + " cid=" + .@cid;
		return 0;
	}
	$NEED_BG2_P3_CLAIMUSED = 1;
	.@l$ = "";
	for (.@i = 0; .@i < .@n; .@i++) {
		getitem .@it[.@i], .@am[.@i];
		.@l$ += (.@i ? "," : "") + .@it[.@i] + ":" + .@am[.@i];
	}
	NEED_BG2_IPAIDSEQ = .@seq;
	callfunc "F_NEED_BG2P3_Audit", .@aid, "ITEM_PAID", "id=" + .@id, "cid=" + .@cid + " kind=" + .@k$ + " items=" + .@l$ + " seq=" + .@seq;
	callfunc "F_NEED_BG2P3_Log", "P3_ITEM_CLAIM id=" + .@id + " aid=" + .@aid + " cid=" + .@cid + " kind=" + .@k$ + " items=" + .@l$ + " seq=" + .@seq;
	dispbottom needtr("[2:2 듀오] 보상 아이템을 받았습니다.", "[2v2 Duo] You received the reward items."), 0x66FF66;
	return 1;
}

// 보상권 내용 글(아이템이면 이름 x 수량, 캐시면 금액).
function	script	F_NEED_BG2P3_TicketText$	{
	if (getarg(1) != "MITEM" && getarg(1) != "TIERI" && getarg(1) != "RANKI")
		return getarg(2) + needtr(" 캐시", " cash");
	deletearray .@it;
	deletearray .@am;
	.@n = query_sql("SELECT item_id, amount FROM need_bg2_reward_item WHERE reward_id = " + getarg(0) + " ORDER BY seq", .@it, .@am);
	.@t$ = "";
	for (.@i = 0; .@i < .@n; .@i++)
		.@t$ += (.@i ? ", " : "") + getitemname(.@it[.@i]) + " x" + .@am[.@i];
	return .@t$;
}

// 아이템 목록 글(규칙 미리보기).

	return .@t$;
}

}

// 보상권 종류의 수령 스위치가 켜져 있나.

}

// Phase 2 경기 흐름 연결 (Phase 2 가 $@NBG2P3_LOADED 일 때만 부른다)

// Launch: 경기 기록 초기화. P3 가 켜져 있을 때 시작한 경기만 정산한다(경기 중 스위치를 켜도 그 경기는 정산하지 않는다).
function	script	F_NEED_BG2P3_Arm	{
	$@NBG2_P3ARMED = $NEED_BG2_P3_ON ? 1 : 0;
	$@NBG2_P3FLAGS = ($NEED_BG2_P3_MCASH ? 1 : 0) | ($NEED_BG2_P3_MITEM ? 2 : 0);	// 승패 직접 캐시 / 아이템: 시작 때 값으로 고정
	$@NBG2_P3VER = callfunc("F_NEED_BG2P3_RuleVer", gettimetick(2));
	$@NBG2_P3SRC$ = (getarg(0) == 2) ? "queue" : "gm";
	$@NBG2_T0 = gettimetick(2);
	$@NBG2_TICK0 = gettimetick(0);
	$@NBG2_ACT_MS = 0;
	$@NBG2_RACT = 0;
	cleararray $@NBG2_CB_H[0], 0, 4;
	cleararray $@NBG2_CB_HT[0], 0, 4;
	cleararray $@NBG2_CB_S[0], 0, 4;
	cleararray $@NBG2_CLS[0], 0, 4;
	cleararray $@NBG2_IP$[0], "", 4;
	return;
}

// Launch: 첨부된 참가자 k. 교전 카운터 초기화, 직업/IP(경기 중 메모리에만) 기록, 기존 엔진 탈주자 상태 보존.
function	script	F_NEED_BG2P3_LaunchSelf	{
	.@k = getarg(0);
	@NEED_BG2_CB_HOST = 0;
	@NEED_BG2_CB_HIT = 0;
	@NEED_BG2_CB_SUP = 0;
	$@NBG2_CLS[.@k] = Class;
	$@NBG2_IP$[.@k] = getcharip();
	if (!$@NBG2_P3ARMED)
		return;
	NEED_BG2_P3 = 1;
	NEED_BG2_PREDES = 0;
	if (getstatus(SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT))
		NEED_BG2_PREDES = gettimetick(2) + getstatus(SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT, 5) / 1000;
	return;
}

// 첨부된 참가자 i 의 세션 교전 카운터를 경기 기록으로 옮긴다(감시 0.5초마다, 접속 종료 직전, 정산 직전).
function	script	F_NEED_BG2P3_SyncSelf	{
	.@i = getarg(0);
	if (getcharid(0) != $@NBG2_CID[.@i])
		return;
	$@NBG2_CB_H[.@i] = @NEED_BG2_CB_HOST;
	$@NBG2_CB_HT[.@i] = @NEED_BG2_CB_HIT;
	$@NBG2_CB_S[.@i] = @NEED_BG2_CB_SUP;
	return;
}

// 재접속 유예 안 복귀: 새 세션에 경기 카운터를 되돌린다.
function	script	F_NEED_BG2P3_RejoinSelf	{
	.@i = getarg(0);
	@NEED_BG2_CB_HOST = $@NBG2_CB_H[.@i];
	@NEED_BG2_CB_HIT = $@NBG2_CB_HT[.@i];
	@NEED_BG2_CB_SUP = $@NBG2_CB_S[.@i];
	return;
}

// 라운드 진행 시간 누적(라운드 종료/매치 종료에서). 마감 시각 뒤는 세지 않는다.
function	script	F_NEED_BG2P3_CloseActive	{
	if (!$@NBG2_RACT)
		return;
	$@NBG2_RACT = 0;
	.@end = gettimetick(0);
	if ($@NBG2_DEADLINE && .@end > $@NBG2_DEADLINE)
		.@end = $@NBG2_DEADLINE;
	if (.@end > $@NBG2_ROUND_AT)
		$@NBG2_ACT_MS += .@end - $@NBG2_ROUND_AT;
	return;
}

// RestoreSelf(첨부): 정산된 경기에서는 2:2 탈주를 2:2 제재(원장)로만 처리한다. 엔진이 건 탈주자 상태
// (다른 전장 대기열을 막는다)는 지우고, 경기 전부터 있던 다른 전장 탈주자 상태는 남은 시간만큼 되돌린다.
function	script	F_NEED_BG2P3_EngineDeserter	{
	.@now = gettimetick(2);
	.@had = getstatus(SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT);
	if (.@had)
		sc_end SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT;
	.@back = 0;
	if (NEED_BG2_PREDES > .@now) {
		.@back = NEED_BG2_PREDES - .@now;
		sc_start SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT, .@back * 1000, 1, 10000, SCSTART_NOAVOID | SCSTART_NOTICKDEF;
	}
	if (.@had || .@back)
		callfunc "F_NEED_BG2_Log", "P3_ENGINE_DESERTER cid=" + getcharid(0) + " cleared=" + .@had + " restored_sec=" + .@back;
	NEED_BG2_P3 = 0;
	NEED_BG2_PREDES = 0;
	return;
}

// 정산 (P3-A / P3-B)
//------------------------------------------------------------

// Finish 직후(상태 6, 경기 변수가 아직 있음). 경기 결과를 고정해 원장에 넣는다. 경기당 한 번.
function	script	F_NEED_BG2P3_Settle	{
	.@kind = getarg(0);
	.@why$ = getarg(1);
	if (!$@NBG2_P3ARMED)
		return 0;
	$@NBG2_P3ARMED = 0;	// 같은 경기를 두 번 정산하지 않는다(원장 match_key 유니크가 2차 방어)
	.@caller = playerattached();
	for (.@i = 0; .@i < $@NBG2_N; .@i++) {
		if ($@NBG2_GONE[.@i] || !isloggedin($@NBG2_AID[.@i], $@NBG2_CID[.@i]) || !attachrid($@NBG2_AID[.@i]))
			continue;
		callfunc "F_NEED_BG2P3_SyncSelf", .@i;
		detachrid;
	}
	callfunc "F_NEED_BG2_Reattach", .@caller;
	// 다른 팀끼리 같은 접속 IP(경기 시작 때 기록, 원장에는 0/1 신호만 남긴다).
	.@sigip = 0;
	for (.@b = 0; .@b < 2; .@b++)
		for (.@r = 2; .@r < 4; .@r++)
			if ($@NBG2_IP$[.@b] != "" && $@NBG2_IP$[.@b] == $@NBG2_IP$[.@r])
				.@sigip = 1;
	.@key$ = $@NBG2_MATCH + "-" + $@NBG2_TICK0;
	.@h$ = .@key$ + "|" + $@NBG2_MATCH + "|" + $@NBG2_P3SRC$ + "|" + $@NBG2_T0 + "|" + gettimetick(2) + "|" + .@kind + "|" + .@why$
		+ "|" + $@NBG2_ROUND + "|" + $@NBG2_ACT_MS + "|" + $@NBG2_WIN_B + "|" + $@NBG2_WIN_R + "|" + .@sigip + "|" + $@NBG2_P3FLAGS + "|" + $@NBG2_P3VER;
	.@p$ = "";
	for (.@i = 0; .@i < 4; .@i++)
		.@p$ += (.@i ? ";" : "") + $@NBG2_AID[.@i] + "," + $@NBG2_CID[.@i] + "," + $@NBG2_TEAM[.@i] + "," + $@NBG2_CLS[.@i] + "," + $@NBG2_GONE[.@i]
			+ "," + $@NBG2_CB_H[.@i] + "," + $@NBG2_CB_HT[.@i] + "," + $@NBG2_CB_S[.@i] + "," + $@NBG2_IP$[.@i];
	.@r = callfunc("F_NEED_BG2P3_Apply", .@h$, .@p$, 1);
	callfunc "F_NEED_BG2P3_Log", "P3_SETTLE match=" + $@NBG2_MATCH + " key=" + .@key$ + " result=" + .@r + " head=" + .@h$ + " players=" + .@p$;
	if (.@r == -1)
		callfunc "F_NEED_BG2P3_PendPush", .@h$, .@p$;
	else if (.@r < 0)
		callfunc "F_NEED_BG2P3_PartPush", .@h$, .@p$, .@r;
	return .@r;
}

// DB 실패한 정산 결과를 영구 서버 변수에 보관한다(재시도 대상). 로그에도 같은 내용이 있다.
function	script	F_NEED_BG2P3_PendPush	{
	.@n = getarraysize($NEED_BG2_PEND_H$);
	if (.@n >= $@NBG2P3_PEND_MAX) {
		callfunc "F_NEED_BG2P3_Log", "P3_PENDING_OVERFLOW head=" + getarg(0) + " players=" + getarg(1) + " (manual replay from this log line)";
		return 0;
	}
	$NEED_BG2_PEND_H$[.@n] = getarg(0);
	$NEED_BG2_PEND_P$[.@n] = getarg(1);
	callfunc "F_NEED_BG2P3_Log", "P3_PENDING_RECONCILIATION slot=" + .@n + " head=" + getarg(0) + " players=" + getarg(1);
	return 1;
}

// 보관된 정산을 다시 넣는다. 성공/중복/형식 오류는 목록에서 빼고, DB 실패면 다음 번으로 미룬다.
// 보관 정산을 앞에서부터 최대 $@NBG2P3_PEND_BATCH 건 다시 적용한다. 한 번의 스크립트 실행 안에서 끝나야 하므로(엔진 분기 한도)
// 나머지는 다음 1분 타이머나 GM retry 가 이어서 한다. DB 오류(-1)면 그 건에서 멈추고 순서를 유지한다.
// 반쪽·불일치(-3)·형식 오류(-2)는 검토 보관함으로 옮겨 다른 건을 막지 않는다(삭제·자동 정정 없음).
// 반환: 이번에 보관 목록에서 뺀 건수. 결과는 $@NBG2P3_RT_* 에도 둔다(GM 표시).
function	script	F_NEED_BG2P3_PendRetry	{
	.@ok = 0;
	.@dup = 0;
	.@part = 0;
	.@err = 0;
	.@units = 0;
	for (.@i = 0; .@i < $@NBG2P3_PEND_BATCH; .@i++) {
		if (getarraysize($NEED_BG2_PEND_H$) < 1)
			break;
		.@h$ = $NEED_BG2_PEND_H$[0];
		.@p$ = $NEED_BG2_PEND_P$[0];
		// 승패 보상이 있는 정산은 분기를 약 두 배 쓴다: 2단위로 세어 한 번의 실행 안에 묶음 크기 단위만큼만.
		deletearray .@hh$;
		explode(.@hh$, .@h$, "|");
		.@w = (getarraysize(.@hh$) == 14 && (atoi(.@hh$[12]) & 3)) ? 2 : 1;
		if (.@units + .@w > $@NBG2P3_PEND_BATCH)
			break;
		.@units += .@w;
		.@r = callfunc("F_NEED_BG2P3_Apply", .@h$, .@p$, 1);
		if (.@r == -1) {
			.@err = 1;
			break;
		}
		if (.@r < 0) {
			callfunc "F_NEED_BG2P3_PartPush", .@h$, .@p$, .@r;
			.@part++;
		} else if (.@r == 0)
			.@dup++;
		else
			.@ok++;
		deletearray $NEED_BG2_PEND_H$[0], 1;
		deletearray $NEED_BG2_PEND_P$[0], 1;
		callfunc "F_NEED_BG2P3_Log", "P3_PENDING_DONE result=" + .@r + " head=" + .@h$;
	}
	$@NBG2P3_RT_OK = .@ok;
	$@NBG2P3_RT_DUP = .@dup;
	$@NBG2P3_RT_PART = .@part;
	$@NBG2P3_RT_ERR = .@err;
	callfunc "F_NEED_BG2P3_Log", "P3_PENDING_BATCH applied=" + .@ok + " duplicate_ok=" + .@dup + " partial=" + .@part + " db_error=" + .@err
		+ " left=" + getarraysize($NEED_BG2_PEND_H$) + " review_box=" + callfunc("F_NEED_BG2P3_PartCount");
	return .@ok + .@dup + .@part;
}

// 검토 보관함: 반쪽·불일치 원장이나 형식 오류 payload. 자동 재시도·삭제하지 않고 GM 이 대조한 뒤 requeue 한다.
function	script	F_NEED_BG2P3_PartPush	{
	.@n = getarraysize($NEED_BG2_PART_H$);
	if (.@n >= $@NBG2P3_PEND_MAX) {
		callfunc "F_NEED_BG2P3_Log", "P3_PARTIAL_OVERFLOW code=" + getarg(2) + " head=" + getarg(0) + " players=" + getarg(1) + " (manual replay from this log line)";
		return 0;
	}
	$NEED_BG2_PART_H$[.@n] = getarg(0);
	$NEED_BG2_PART_P$[.@n] = getarg(1);
	$NEED_BG2_PART_R$[.@n] = (getarg(2) == -2 ? "bad_payload" : $@NBG2P3_PART_WHY$);
	callfunc "F_NEED_BG2P3_Log", "P3_PARTIAL_KEPT slot=" + .@n + " why=" + $NEED_BG2_PART_R$[.@n] + " head=" + getarg(0) + " players=" + getarg(1);
	return 1;
}

// 검토 보관함 건수. 처리한 칸은 당기지 않고 비워 둔다(가운데를 deletearray 로 당기면 mapreg 가 덮어쓴 칸을 300초 뒤에야
// DB 에 저장해, 그 사이 서버가 죽으면 뒤쪽 항목이 사라진다). 빈칸은 건너뛴다.
function	script	F_NEED_BG2P3_PartCount	{
	.@c = 0;
	.@n = getarraysize($NEED_BG2_PART_H$);
	for (.@i = 0; .@i < .@n; .@i++)
		if ($NEED_BG2_PART_H$[.@i] != "")
			.@c++;
	return .@c;
}

// 경기 결과 → 원장. 1 기록, 0 이미 기록됨(중복), -1 DB 실패(롤백), -2 형식 오류.
// head = key|ext_mid|source|started|ended|kind|why|rounds|active_ms|win_b|win_r|sig_ip
// players = aid,cid,team,class,gone,hostile,hit,support ; x4 (팀 1 블루 2 레드, gone 1 = 유예 만료 이탈)
function	script	F_NEED_BG2P3_Apply	{
	.@notify = getarg(2, 0);
	explode(.@h$, getarg(0), "|");
	if (getarraysize(.@h$) != 12 && getarraysize(.@h$) != 14)
		return -2;
	.@key$ = .@h$[0];
	.@mid = atoi(.@h$[1]);
	.@src$ = .@h$[2];
	.@t0 = atoi(.@h$[3]);
	.@te = atoi(.@h$[4]);
	.@kind = atoi(.@h$[5]);
	.@why$ = .@h$[6];
	.@rounds = atoi(.@h$[7]);
	.@act = atoi(.@h$[8]);
	.@wb = atoi(.@h$[9]);
	.@wr = atoi(.@h$[10]);
	.@sigip = atoi(.@h$[11]);
	// 13·14번째: 승패 보상 스위치(1 직접 캐시, 2 아이템, 시작 때 고정)와 시작 때 본 규칙 버전. 예전 12칸 payload 는 승패 보상 없음.
	.@flags = (getarraysize(.@h$) == 14) ? atoi(.@h$[12]) : 0;
	if (!$@NBG2P3_RULES_OK) {
		callfunc "F_NEED_BG2P3_Log", "P3_RULES_BLOCKED key=" + .@h$[0] + " (rules file invalid: kept for retry)";
		return -1;
	}
	// 규칙 버전 = 경기 시작 시각 주차의 버전. 이미 시작된 버전은 바꿀 수 없으므로 시작 때와 같다. 시작 때 규칙이 잘못 로드돼
	// 있었다면 payload 의 버전이 지금은 없을 수 있다: 그때도 시작 주차의 (지금 올바른) 버전을 쓰고 기록만 남긴다(보관 정산이 막히지 않게).
	.@ver = callfunc("F_NEED_BG2P3_RuleVer", .@t0);
	if (getarraysize(.@h$) == 14 && atoi(.@h$[13]) != .@ver)
		callfunc "F_NEED_BG2P3_Log", "P3_RULES_VER_NOTE key=" + .@h$[0] + " launch_ver=" + .@h$[13] + " used_ver=" + .@ver + " (version in force for the launch week)";
	explode(.@pl$, getarg(1), ";");
	if (getarraysize(.@pl$) != 4 || .@key$ == "" || .@te <= 0)
		return -2;
	.@hostile = 0;
	for (.@k = 0; .@k < 4; .@k++) {
		deletearray .@f$;
		explode(.@f$, .@pl$[.@k], ",");
		if (getarraysize(.@f$) != 8 && getarraysize(.@f$) != 9)
			return -2;
		.@aid[.@k] = atoi(.@f$[0]);
		.@cid[.@k] = atoi(.@f$[1]);
		.@team[.@k] = atoi(.@f$[2]);
		.@cls[.@k] = atoi(.@f$[3]);
		.@gone[.@k] = atoi(.@f$[4]);
		.@hs[.@k] = atoi(.@f$[5]);
		.@ht[.@k] = atoi(.@f$[6]);
		.@sp[.@k] = atoi(.@f$[7]);
		.@ip$[.@k] = (getarraysize(.@f$) == 9) ? .@f$[8] : "";	// 경기 시작 때 접속 IP(승패 직접 캐시 하루 한도)
		.@hostile += .@hs[.@k];
	}
	.@ws = callfunc("F_NEED_BG2P3_WeekStart", .@te);
	.@queue = (.@src$ == "queue");
	.@engaged = (.@hostile > 0);
	// 개인 정산. 공개 대기열 + 실교전 + 정상 결과(무/블루/레드)만 점수가 있다. 탈주자(gone 1)는 0점·불인정.
	.@valid = 0;
	for (.@k = 0; .@k < 4; .@k++) {
		.@out[.@k] = 0;
		if (.@kind == 0)
			.@out[.@k] = 3;
		else if (.@kind == 1 || .@kind == 2)
			.@out[.@k] = (.@team[.@k] == .@kind) ? 1 : 2;
		.@des[.@k] = (.@gone[.@k] == 1) ? 1 : 0;
		.@pts[.@k] = 0;
		.@val[.@k] = 0;
		.@win[.@k] = 0;
		if (!.@queue || !.@engaged || .@kind > 2 || .@des[.@k])
			continue;
		.@val[.@k] = 1;
		.@valid = 1;
		if (.@out[.@k] == 1) {
			.@pts[.@k] = $@NBG2R_PW[.@ver];
			.@win[.@k] = 1;
		} else
			.@pts[.@k] = (.@out[.@k] == 2) ? $@NBG2R_PL[.@ver] : $@NBG2R_PD[.@ver];
	}
	// 탈주 기록(제재 대상): 공개 대기열 경기의 유예 만료 이탈만. 준비 취소/GM 중단/강제 이동/리로드/시작 실패는 면제.
	.@desert = (.@queue && (.@why$ == "forfeit_blue" || .@why$ == "forfeit_red" || .@why$ == "both_teams_departed"));
	if (.@why$ == "both_teams_departed")
		.@dkind$ = "both_left";
	else
		.@dkind$ = .@engaged ? "forfeit" : "no_combat";
	// 4계정 정렬 키(반복 매칭 신호).
	copyarray .@sa[0], .@aid[0], 4;
	for (.@a = 0; .@a < 3; .@a++)
		for (.@b = 0; .@b < 3 - .@a; .@b++)
			if (.@sa[.@b] > .@sa[.@b + 1]) {
				.@tmp = .@sa[.@b];
				.@sa[.@b] = .@sa[.@b + 1];
				.@sa[.@b + 1] = .@tmp;
			}
	.@akey$ = .@sa[0] + "-" + .@sa[1] + "-" + .@sa[2] + "-" + .@sa[3];

	if (callfunc("F_NEED_BG2P3_Begin", "apply_begin") < 0)
		return -1;
	deletearray .@dup;
	.@n = query_sql("SELECT match_pk FROM need_bg2_match WHERE match_key = '" + escape_sql(.@key$) + "'", .@dup);
	if (.@n < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "apply_dupcheck");
	if (.@n > 0) {
		query_sql "ROLLBACK";
		// 경기 키가 있다고 끝난 것이 아니다: 참가자 정확히 4행 + payload 의 aid/cid/team 과 같아야 정상 중복(멱등 성공).
		deletearray .@ea;
		deletearray .@ec;
		deletearray .@et;
		.@c = query_sql("SELECT account_id, char_id, team FROM need_bg2_match_player WHERE match_pk = " + .@dup, .@ea, .@ec, .@et);
		if (.@c < 0)
			return -1;
		.@same = (.@c == 4);
		for (.@k = 0; .@k < 4; .@k++) {
			if (!.@same)
				break;
			.@hit = 0;
			for (.@j = 0; .@j < .@c; .@j++)
				if (.@ea[.@j] == .@aid[.@k] && .@ec[.@j] == .@cid[.@k] && .@et[.@j] == .@team[.@k])
					.@hit = 1;
			.@same = .@hit;
		}
		if (!.@same) {
			$@NBG2P3_PART_WHY$ = (.@c < 4 ? "missing_players" : "players_mismatch") + ":match_pk=" + .@dup + ":count=" + .@c;
			callfunc "F_NEED_BG2P3_Log", "P3_PARTIAL match_key=" + .@key$ + " match_pk=" + .@dup + " count=" + .@c + " why=" + (.@c < 4 ? "missing_players" : "players_mismatch");
			return -3;
		}
		// 승패 보상 경기: 참가자 4행이 있어도 보상권이 다 들어갔는지는 완료 표식으로만 안다(트랜잭션 도중 중단 + 암묵 커밋 대비).
		if (.@flags & 3) {
			.@done = 0;
			if (query_sql("SELECT COUNT(*) FROM need_bg2_match_done WHERE match_pk = " + .@dup, .@done) < 0)
				return -1;
			if (!.@done) {
				$@NBG2P3_PART_WHY$ = "rewards_incomplete:match_pk=" + .@dup + ":count=4";
				callfunc "F_NEED_BG2P3_Log", "P3_PARTIAL match_key=" + .@key$ + " match_pk=" + .@dup + " count=4 why=rewards_incomplete (no completion mark)";
				return -3;
			}
		}
		callfunc "F_NEED_BG2_Log", "P3_DUPLICATE key=" + .@key$ + " match_pk=" + .@dup + " players=4 ok";
		return 0;
	}
	if (query_sql("INSERT INTO need_bg2_match (match_key, ext_match_id, `mode`, `source`, started_at, ended_at, week_start, result, end_reason, engaged, hostile_total, rounds, active_ms, win_b, win_r, valid, account_key) VALUES ('"
		+ escape_sql(.@key$) + "', " + .@mid + ", " + $@NBG2P3_MODE + ", '" + escape_sql(.@src$) + "', " + .@t0 + ", " + .@te + ", " + .@ws + ", " + .@kind + ", '" + escape_sql(.@why$) + "', "
		+ .@engaged + ", " + .@hostile + ", " + .@rounds + ", " + .@act + ", " + .@wb + ", " + .@wr + ", " + .@valid + ", '" + .@akey$ + "')") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "apply_match");
	.@pk = 0;
	query_sql "SELECT LAST_INSERT_ID()", .@pk;
	if (.@pk <= 0)
		return callfunc("F_NEED_BG2P3_Rollback", "apply_pk");
	// 검토 신호: 점수가 생기는 경기만. 독립 신호 RV_NEED 개 이상이면 잠정 HOLD(점수는 기록하되 보상·랭킹에서 빠짐).
	.@sig$ = "";
	.@nsig = 0;
	if (.@valid) {
		if (.@hostile < $@NBG2P3_RV_MINHOST) {
			.@sig$ += "low_hostile:" + .@hostile + " ";
			.@nsig++;
		}
		if (.@rounds > 0 && .@act / .@rounds < $@NBG2P3_RV_SHORTMS) {
			.@sig$ += "short_rounds:" + (.@act / .@rounds) + " ";
			.@nsig++;
		}
		.@rep = 0;
		if (query_sql("SELECT COUNT(*) FROM need_bg2_match WHERE account_key = '" + .@akey$ + "' AND `source` = 'queue' AND ended_at > " + (.@te - $@NBG2P3_RV_WINDOW) + " AND ended_at <= " + .@te, .@rep) < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "apply_repeat");
		if (.@rep >= $@NBG2P3_RV_REPEAT) {
			.@sig$ += "repeat_group:" + .@rep + " ";
			.@nsig++;
		}
		if (.@sigip) {
			.@sig$ += "shared_ip ";
			.@nsig++;
		}
		if (.@desert && .@engaged && .@kind <= 2) {
			.@sig$ += "desert_gift ";
			.@nsig++;
		}
	}
	.@review = (.@valid && .@nsig >= $@NBG2P3_RV_NEED) ? 1 : 0;
	if (.@sig$ != "") {
		if (query_sql("UPDATE need_bg2_match SET review = " + .@review + ", signals = '" + escape_sql(.@sig$) + "' WHERE match_pk = " + .@pk) < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "apply_signals");
	}
	if (.@review) {
		if (query_sql("INSERT INTO need_bg2_review (match_pk, signals, state, created_at) VALUES (" + .@pk + ", '" + escape_sql(.@sig$) + "', 'HOLD', " + .@te + ")") < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "apply_review");
	}
	.@vals$ = "";
	for (.@k = 0; .@k < 4; .@k++)
		.@vals$ += (.@k ? ", " : "") + "(" + .@pk + ", " + .@aid[.@k] + ", " + .@cid[.@k] + ", " + .@team[.@k] + ", " + .@cls[.@k] + ", " + .@out[.@k] + ", " + .@des[.@k]
			+ ", " + .@pts[.@k] + ", " + .@val[.@k] + ", " + .@win[.@k] + ", " + .@hs[.@k] + ", " + .@ht[.@k] + ", " + .@sp[.@k] + ")";
	if (query_sql("INSERT INTO need_bg2_match_player (match_pk, account_id, char_id, team, class, outcome, deserter, points, valid, win, hostile, hit, support) VALUES " + .@vals$) < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "apply_players");
	if (.@desert) {
		for (.@k = 0; .@k < 4; .@k++) {
			if (!.@des[.@k])
				continue;
			if (query_sql("INSERT IGNORE INTO need_bg2_desertion (account_id, char_id, match_pk, `at`, kind) VALUES (" + .@aid[.@k] + ", " + .@cid[.@k] + ", " + .@pk + ", " + .@te + ", '" + .@dkind$ + "')") < 0)
				return callfunc("F_NEED_BG2P3_Rollback", "apply_desertion");
		}
	}
	// 계정 단계 보상권(검토 보류 경기는 승인 뒤 소급).
	if (.@valid && !.@review) {
		for (.@k = 0; .@k < 4; .@k++) {
			.@acc_new[.@k] = -1;
			if (!.@val[.@k])
				continue;
			.@acc_new[.@k] = callfunc("F_NEED_BG2P3_TierSync", .@aid[.@k], .@ws);
			if (.@acc_new[.@k] < 0)
				return callfunc("F_NEED_BG2P3_Rollback", "apply_tier");
			.@acc_pts[.@k] = $@NBG2P3_LASTPTS;
		}
	}
	// 승패 직접 캐시 / 아이템 보상권(경기 시작 때 켠 경기만). 검토 보류 경기는 보류(7)로 만들어 두고 승인 때 연다.
	if (.@valid && (.@flags & 3)) {
		for (.@k = 0; .@k < 4; .@k++) {
			$@NBG2P3_MR_AID[.@k] = .@aid[.@k];
			$@NBG2P3_MR_CID[.@k] = .@cid[.@k];
			$@NBG2P3_MR_OUT[.@k] = .@out[.@k];
			$@NBG2P3_MR_VAL[.@k] = .@val[.@k];
			$@NBG2P3_MR_IP$[.@k] = .@ip$[.@k];
		}
		if (callfunc("F_NEED_BG2P3_MatchRewards", .@pk, .@ver, .@flags, .@ws, .@te, .@review) < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "apply_match_rewards");
	}
	// 완료 표식(트랜잭션의 마지막 문장). 승패 보상 경기의 재시도 중복 판정은 이 표식까지 있어야 정상으로 본다.
	if (query_sql("INSERT INTO need_bg2_match_done (match_pk, done_at) VALUES (" + .@pk + ", " + gettimetick(2) + ")") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "apply_done");
	if (callfunc("F_NEED_BG2P3_Exec", "COMMIT", "apply_commit") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "apply_commit");
	callfunc "F_NEED_BG2_Log", "P3_LEDGER match=" + .@mid + " match_pk=" + .@pk + " src=" + .@src$ + " kind=" + .@kind + " why=" + .@why$ + " week=" + .@ws
		+ " engaged=" + .@engaged + " hostile=" + .@hostile + " valid=" + .@valid + " review=" + .@review + " signals=" + .@sig$
		+ " pts=" + .@pts[0] + "/" + .@pts[1] + "/" + .@pts[2] + "/" + .@pts[3] + " deserters=" + .@des[0] + .@des[1] + .@des[2] + .@des[3] + " desert_rec=" + .@desert;
	if (!.@notify)
		return 1;
	// 승패 직접 캐시: 접속 중인 계정은 바로 지급(보상권 0 → 1 → 영수증). 접속하지 않았으면 다음 로그인 / 보상 NPC.
	if (.@valid && (.@flags & 1) && !.@review && $NEED_BG2_P3_MCASH)
		callfunc "F_NEED_BG2P3_PayMatchCash", .@pk;
	// 안내(접속 중인 사람만).
	for (.@k = 0; .@k < 4; .@k++) {
		if (!.@queue)
			break;
		if (.@des[.@k]) {
			if (.@desert)
				callfunc "F_NEED_BG2P3_TellSanction", .@aid[.@k], .@cid[.@k], .@te;
			continue;
		}
		if (!.@valid) {
			if (.@kind <= 2 && !.@engaged)
				callfunc "F_NEED_BG2_Tell", .@aid[.@k], .@cid[.@k], "[2:2 듀오] 실제 교전이 없어 유효 경기로 인정되지 않았습니다(0점).",
					"[2v2 Duo] No real combat took place, so this is not a valid match (0 points).";
			continue;
		}
		if (.@review) {
			callfunc "F_NEED_BG2_Tell", .@aid[.@k], .@cid[.@k], "[2:2 듀오] 이 경기는 운영 검토 대기 중입니다. 점수 +" + .@pts[.@k] + "점은 검토 후 반영됩니다.",
				"[2v2 Duo] This match is waiting for a GM review. Your +" + .@pts[.@k] + " points count after the review.";
			continue;
		}
		callfunc "F_NEED_BG2_Tell", .@aid[.@k], .@cid[.@k], "[2:2 듀오] 주간 점수 +" + .@pts[.@k] + "점 (이번 주 계정 합계 " + .@acc_pts[.@k] + "점)"
			+ (.@acc_new[.@k] > 0 ? " - 새 보상 " + .@acc_new[.@k] + "개! 파요 2:2 듀오 보상관에서 받으세요." : ""),
			"[2v2 Duo] Weekly points +" + .@pts[.@k] + " (account total this week " + .@acc_pts[.@k] + ")"
			+ (.@acc_new[.@k] > 0 ? " - " + .@acc_new[.@k] + " new reward(s)! Claim them from the 2v2 Duo reward NPC in Payon." : "");
	}
	return 1;
}

// 계정의 주간 점수(보류 경기 제외)로 단계 보상권을 만든다(이미 있으면 그대로). 트랜잭션 안에서 부른다.
// 새로 만든 보상권 수, DB 실패 -1. $@NBG2P3_LASTPTS = 계정 주간 점수.
function	script	F_NEED_BG2P3_TierSync	{
	.@aid = getarg(0);
	.@ws = getarg(1);
	.@pts = 0;
	if (query_sql("SELECT COALESCE(SUM(p.points), 0) FROM need_bg2_match_player p JOIN need_bg2_match m ON m.match_pk = p.match_pk WHERE p.account_id = " + .@aid
		+ " AND m.`mode` = " + $@NBG2P3_MODE + " AND m.week_start = " + .@ws + " AND p.valid = 1 AND m.review IN (0, 2)", .@pts) < 0)
		return -1;
	$@NBG2P3_LASTPTS = .@pts;
	.@new = 0;
	if (!$@NBG2P3_RULES_OK)
		return -1;
	.@v = callfunc("F_NEED_BG2P3_RuleVer", .@ws);
	if ($@NBG2R_WM[.@v]) {
		// 아이템 방식 주차: 구간별 아이템 목록을 스냅샷으로(빈 목록 = 그 구간 보상 없음). 계정·주차·구간 1회(uk_ticket).
		setarray .@l$[0], $@NBG2R_T10$[.@v], $@NBG2R_T30$[.@v], $@NBG2R_T60$[.@v];
		for (.@t = 0; .@t < 3; .@t++) {
			if (.@pts < $@NBG2P3_TIER_PT[.@t])
				break;
			.@r = callfunc("F_NEED_BG2P3_ItemTicket", .@ws, .@aid, "TIERI", "" + $@NBG2P3_TIER_PT[.@t], 0, 0, 0, .@l$[.@t], $@NBG2R_F$[.@v * 9 + 3 + .@t]);
			if (.@r < 0)
				return -1;
			.@new += .@r;
		}
		return .@new;
	}
	for (.@t = 0; .@t < 3; .@t++) {
		if (.@pts < $@NBG2P3_TIER_PT[.@t])
			break;
		if (query_sql("INSERT IGNORE INTO need_bg2_reward (`mode`, week_start, account_id, reward_kind, reward_key, amount, status, created_at) VALUES ("
			+ $@NBG2P3_MODE + ", " + .@ws + ", " + .@aid + ", 'TIER', '" + $@NBG2P3_TIER_PT[.@t] + "', " + $@NBG2P3_TIER_CASH[.@t] + ", 0, " + gettimetick(2) + ")") < 0)
			return -1;
		.@rc = 0;
		query_sql "SELECT ROW_COUNT()", .@rc;
		if (.@rc > 0) {
			.@new++;
			callfunc "F_NEED_BG2_Log", "P3_TICKET kind=TIER key=" + $@NBG2P3_TIER_PT[.@t] + " aid=" + .@aid + " week=" + .@ws + " amount=" + $@NBG2P3_TIER_CASH[.@t] + " pts=" + .@pts;
		}
	}
	return .@new;
}

// 경기 무효 뒤: 계정 점수가 단계 아래로 내려가면 미수령 보상권은 회수(8), 지급된 것은 수동 정산(9)으로 표시만 한다.
function	script	F_NEED_BG2P3_TierRevoke	{
	.@aid = getarg(0);
	.@ws = getarg(1);
	.@pts = 0;
	if (query_sql("SELECT COALESCE(SUM(p.points), 0) FROM need_bg2_match_player p JOIN need_bg2_match m ON m.match_pk = p.match_pk WHERE p.account_id = " + .@aid
		+ " AND m.`mode` = " + $@NBG2P3_MODE + " AND m.week_start = " + .@ws + " AND p.valid = 1 AND m.review IN (0, 2)", .@pts) < 0)
		return -1;
	.@cnt = 0;
	for (.@t = 0; .@t < 3; .@t++) {
		if (.@pts >= $@NBG2P3_TIER_PT[.@t])
			continue;
		.@base$ = " WHERE `mode` = " + $@NBG2P3_MODE + " AND week_start = " + .@ws + " AND account_id = " + .@aid + " AND reward_kind IN ('TIER', 'TIERI') AND reward_key = '" + $@NBG2P3_TIER_PT[.@t] + "'";
		if (query_sql("UPDATE need_bg2_reward SET status = 8, note = 'revoked: match voided, never paid'" + .@base$ + " AND status = 0") < 0)
			return -1;
		.@rc = 0;
		query_sql "SELECT ROW_COUNT()", .@rc;
		.@cnt += .@rc;
		if (query_sql("UPDATE need_bg2_reward SET note = CONCAT('MANUAL: paid or reserved (status ', status, '), match voided'), status = 9" + .@base$ + " AND status IN (1, 2)") < 0)
			return -1;
		.@rc = 0;
		query_sql "SELECT ROW_COUNT()", .@rc;
		.@cnt += .@rc;
		if (.@rc)
			callfunc "F_NEED_BG2_Log", "P3_TICKET_MANUAL aid=" + .@aid + " week=" + .@ws + " tier=" + $@NBG2P3_TIER_PT[.@t] + " (paid cash is never charged back automatically)";
	}
	return .@cnt;
}

// 계정 주간 점수. held = 1 이면 검토 보류 중인 점수만.
function	script	F_NEED_BG2P3_AccPoints	{
	.@pts = 0;
	query_sql "SELECT COALESCE(SUM(p.points), 0) FROM need_bg2_match_player p JOIN need_bg2_match m ON m.match_pk = p.match_pk WHERE p.account_id = " + getarg(0)
		+ " AND m.`mode` = " + $@NBG2P3_MODE + " AND m.week_start = " + getarg(1) + " AND p.valid = 1 AND m.review " + (getarg(2, 0) ? "= 1" : "IN (0, 2)"), .@pts;
	return .@pts;
}

//------------------------------------------------------------
// 탈주 제재 (P3-D)
//------------------------------------------------------------

// 최근 24시간 확정 탈주(면제 제외). 가장 최근 탈주 시점까지 24시간 안의 횟수로 단계를 정하고, 그 시점부터 제한한다.
// 1회 경고, 2회 10분, 3회 30분, 4회 이상 2시간. 남은 제한(초)을 돌려주고 $@NBG2P3_SC_CNT 에 횟수를 둔다.
function	script	F_NEED_BG2P3_Sanction	{
	.@aid = getarg(0);
	.@now = getarg(1);
	$@NBG2P3_SC_CNT = 0;
	.@last = 0;
	if (query_sql("SELECT COALESCE(MAX(`at`), 0) FROM need_bg2_desertion WHERE account_id = " + .@aid + " AND exempt = 0 AND `at` > " + (.@now - 86400) + " AND `at` <= " + .@now, .@last) < 0) {
		callfunc "F_NEED_BG2_Log", "P3_SANCTION_CHECK_FAILED aid=" + .@aid + " (sign-up allowed)";
		return 0;
	}
	if (!.@last)
		return 0;
	.@cnt = 0;
	query_sql "SELECT COUNT(*) FROM need_bg2_desertion WHERE account_id = " + .@aid + " AND exempt = 0 AND `at` > " + (.@last - 86400) + " AND `at` <= " + .@last, .@cnt;
	$@NBG2P3_SC_CNT = .@cnt;
	.@dur = 0;
	if (.@cnt >= 4) .@dur = 7200;
	else if (.@cnt == 3) .@dur = 1800;
	else if (.@cnt == 2) .@dur = 600;
	.@left = .@last + .@dur - .@now;
	return (.@left > 0) ? .@left : 0;
}

function	script	F_NEED_BG2P3_Left$	{
	.@s = getarg(0);
	if (.@s >= 3600)
		return needtr((.@s / 3600) + "시간 " + ((.@s % 3600) / 60) + "분", (.@s / 3600) + " h " + ((.@s % 3600) / 60) + " min");
	if (.@s >= 60)
		return needtr((.@s / 60) + "분 " + (.@s % 60) + "초", (.@s / 60) + " min " + (.@s % 60) + " s");
	return needtr(.@s + "초", .@s + " s");
}

// 주차 ws 의 캐릭터 순위. incl_hold = 1 이면 검토 보류 경기도 포함. min_games 미만은 뺀다.
// 결과: $@NBG2P3_R_CID/AID/PTS/W/G/REACH[], $@NBG2P3_R_NAME$[], 반환 행 수(-1 DB 실패).
// 정렬: 점수↓ 승리↓ 승률↓ 최종 점수 도달 시각↑ (동시각이면 그 경기 PK↑, 캐릭터 ID↑: 공개 우선순위를 바꾸지 않는 결정론 정렬).
function	script	F_NEED_BG2P3_RankRows	{
	deletearray $@NBG2P3_R_CID;
	deletearray $@NBG2P3_R_AID;
	deletearray $@NBG2P3_R_PTS;
	deletearray $@NBG2P3_R_W;
	deletearray $@NBG2P3_R_G;
	deletearray $@NBG2P3_R_REACH;
	deletearray $@NBG2P3_R_NAME$;
	.@n = query_sql("SELECT x.char_id, x.account_id, x.pts, x.w, x.g, x.reach, COALESCE(c.name, '?') FROM (SELECT p.char_id, p.account_id, SUM(p.points) AS pts, SUM(p.win) AS w, SUM(p.valid) AS g,"
		+ " MAX(m.ended_at) AS reach, MAX(m.match_pk) AS reachpk FROM need_bg2_match_player p JOIN need_bg2_match m ON m.match_pk = p.match_pk"
		+ " WHERE m.`mode` = " + $@NBG2P3_MODE + " AND m.week_start = " + getarg(0) + " AND p.valid = 1 AND m.review IN (0, 2" + (getarg(1) ? ", 1" : "") + ")"
		+ " GROUP BY p.char_id, p.account_id HAVING g >= " + getarg(2, $@NBG2P3_MINGAMES) + ") x LEFT JOIN `char` c ON c.char_id = x.char_id"
		+ " ORDER BY x.pts DESC, x.w DESC, (x.w * 1.000000000000 / x.g) DESC, x.reach ASC, x.reachpk ASC, x.char_id ASC LIMIT " + getarg(3, 200),
		$@NBG2P3_R_CID, $@NBG2P3_R_AID, $@NBG2P3_R_PTS, $@NBG2P3_R_W, $@NBG2P3_R_G, $@NBG2P3_R_REACH, $@NBG2P3_R_NAME$);
	return .@n;
}

// 주차 ws 의 순위 보상 대상(계정당 1자리, 빈자리 채우지 않음). 결과 $@NBG2P3_AW_CID/AID[1..3], $@NBG2P3_AW_N, $@NBG2P3_AW_ACC.
// 반환: 비교용 서명 문자열. "NA:<계정수>" = 6계정 미만(순위 보상 없음), "ERR" = DB 실패.
function	script	F_NEED_BG2P3_Award$	{
	.@ws = getarg(0);
	.@incl = getarg(1);
	$@NBG2P3_AW_N = 0;
	cleararray $@NBG2P3_AW_CID[0], 0, 4;
	cleararray $@NBG2P3_AW_AID[0], 0, 4;
	.@acc = 0;
	if (query_sql("SELECT COUNT(DISTINCT p.account_id) FROM need_bg2_match_player p JOIN need_bg2_match m ON m.match_pk = p.match_pk WHERE m.`mode` = " + $@NBG2P3_MODE
		+ " AND m.week_start = " + .@ws + " AND p.valid = 1 AND m.review IN (0, 2" + (.@incl ? ", 1" : "") + ")", .@acc) < 0)
		return "ERR";
	$@NBG2P3_AW_ACC = .@acc;
	if (.@acc < $@NBG2P3_MINACC)
		return "NA:" + .@acc;
	.@n = callfunc("F_NEED_BG2P3_RankRows", .@ws, .@incl);
	if (.@n < 0)
		return "ERR";
	.@sig$ = "AW";
	for (.@i = 0; .@i < .@n && $@NBG2P3_AW_N < 3; .@i++) {
		.@dup = 0;
		for (.@j = 1; .@j <= $@NBG2P3_AW_N; .@j++)
			if ($@NBG2P3_AW_AID[.@j] == $@NBG2P3_R_AID[.@i]) .@dup = 1;
		if (.@dup)
			continue;
		$@NBG2P3_AW_N++;
		$@NBG2P3_AW_AID[$@NBG2P3_AW_N] = $@NBG2P3_R_AID[.@i];
		$@NBG2P3_AW_CID[$@NBG2P3_AW_N] = $@NBG2P3_R_CID[.@i];
		.@sig$ += " " + $@NBG2P3_AW_N + ":" + $@NBG2P3_R_AID[.@i] + ":" + $@NBG2P3_R_CID[.@i];
	}
	return .@sig$;
}

// 주차 ws 확정. commit 0 = 드라이런(첨부된 GM 에게 출력), 1 = 실제 기록.
// 반환 0 확정(보상권 생성), 1 순위 보상 없음(6계정 미만/후보 없음), 2 보류(HOLD), 3 이미 확정됨, 4 아직 마감 전, -1 DB 실패.
function	script	F_NEED_BG2P3_Finalize	{
	.@ws = getarg(0);
	.@commit = getarg(1);
	.@actor = getarg(2, 0);
	.@reason$ = getarg(3, "scheduled");
	.@now = gettimetick(2);
	.@out = (.@commit == 0);
	if (.@ws != callfunc("F_NEED_BG2P3_WeekStart", .@ws) || .@now < .@ws + $@NBG2P3_WEEK) {
		if (.@out) dispbottom "[2:2 정산] 주차 " + callfunc("F_NEED_BG2P3_Date$", .@ws) + " 는 올바른 주차 시작이 아니거나 아직 마감 전입니다.";
		return 4;
	}
	.@st$ = "";
	if (query_sql("SELECT state FROM need_bg2_week WHERE `mode` = " + $@NBG2P3_MODE + " AND week_start = " + .@ws, .@st$) < 0)
		return -1;
	if (.@st$ == "FINALIZED" || .@st$ == "NO_AWARD") {
		if (.@out) dispbottom "[2:2 정산] 주차 " + callfunc("F_NEED_BG2P3_Date$", .@ws) + " 는 이미 " + .@st$ + " 입니다. 다시 확정하지 않습니다.";
		return 3;
	}
	.@sigA$ = callfunc("F_NEED_BG2P3_Award$", .@ws, 0);	// 보류 제외 순위($@NBG2P3_AW_*)
	if (.@sigA$ == "ERR")
		return -1;
	// ISSUE-02: 그 주차에 미처리 검토 보류(review = 1) 경기가 1건이라도 있으면 순위를 확정하지 않는다(영향도 계산 없음).
	// GM 이 모두 approve / release / void 해 0건이 되면 다음 자동 확정(FinalizeDue) 또는 GM due / finalize 에서 확정된다.
	.@held = -1;
	if (query_sql("SELECT COUNT(*) FROM need_bg2_match WHERE `mode` = " + $@NBG2P3_MODE + " AND week_start = " + .@ws + " AND review = 1", .@held) < 0 || .@held < 0)
		return -1;
	if (.@held > 0) {
		.@detail$ = "held=" + .@held + " (unresolved review hold: ranking waits) excl[" + .@sigA$ + "]";
		if (.@out) {
			dispbottom "[2:2 정산] 보류(HOLD): 해당 주 HOLD 미처리 " + .@held + "건, 확정 보류. 모두 approve / release / void 하면 다음 확정 때 순위가 정해집니다.";
			dispbottom "[2:2 정산] " + .@detail$;
			return 2;
		}
		if (.@st$ != "HOLD") {
			query_sql "INSERT INTO need_bg2_week (`mode`, week_start, state, accounts, detail, attempts, decided_at) VALUES (" + $@NBG2P3_MODE + ", " + .@ws + ", 'HOLD', " + $@NBG2P3_AW_ACC
				+ ", '" + escape_sql(.@detail$) + "', 1, 0) ON DUPLICATE KEY UPDATE state = 'HOLD', detail = VALUES(detail), attempts = attempts + 1";
			callfunc "F_NEED_BG2P3_Audit", .@actor, "WEEK_HOLD", "week=" + .@ws, .@detail$;
			callfunc "F_NEED_BG2_Log", "P3_WEEK_HOLD week=" + .@ws + " " + .@detail$;
		}
		return 2;
	}
	.@award = (compare(.@sigA$, "NA:") == 0 && $@NBG2P3_AW_N > 0);
	.@state$ = .@award ? "FINALIZED" : "NO_AWARD";
	.@detail$ = "accounts=" + $@NBG2P3_AW_ACC + " held=" + .@held + " " + .@sigA$ + (.@reason$ != "" ? " reason=" + .@reason$ : "");
	if (.@out) {
		dispbottom "[2:2 정산] 드라이런 주차 " + callfunc("F_NEED_BG2P3_Date$", .@ws) + " → " + .@state$ + " (참가 계정 " + $@NBG2P3_AW_ACC + ", 보류 경기 " + .@held + ")";
		for (.@s = 1; .@s <= $@NBG2P3_AW_N; .@s++)
			dispbottom "[2:2 정산]   " + .@s + "위 aid=" + $@NBG2P3_AW_AID[.@s] + " cid=" + $@NBG2P3_AW_CID[.@s] + " " + $@NBG2P3_RANK_CASH[.@s] + " 캐시";
		return .@award ? 0 : 1;
	}
	if (!$@NBG2P3_RULES_OK) {
		callfunc "F_NEED_BG2P3_Log", "P3_RULES_BLOCKED finalize week=" + .@ws + " (retried after the rules are fixed)";
		return -1;
	}
	if (callfunc("F_NEED_BG2P3_Begin", "final_begin") < 0)
		return -1;
	.@st2$ = "";
	if (query_sql("SELECT state FROM need_bg2_week WHERE `mode` = " + $@NBG2P3_MODE + " AND week_start = " + .@ws + " FOR UPDATE", .@st2$) < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "final_lock");
	if (.@st2$ == "FINALIZED" || .@st2$ == "NO_AWARD") {
		query_sql "ROLLBACK";
		return 3;
	}
	.@rv = callfunc("F_NEED_BG2P3_RuleVer", .@ws);
	setarray .@rl$[1], $@NBG2R_R1$[.@rv], $@NBG2R_R2$[.@rv], $@NBG2R_R3$[.@rv];
	for (.@s = 1; .@s <= $@NBG2P3_AW_N; .@s++) {
		if ($@NBG2R_WM[.@rv]) {
			// 아이템 방식 주차: 순위별 아이템 목록 스냅샷(빈 목록 = 그 순위 보상 없음). 순위·계정 1자리 규칙은 그대로.
			if (callfunc("F_NEED_BG2P3_ItemTicket", .@ws, $@NBG2P3_AW_AID[.@s], "RANKI", "" + .@s, .@s, $@NBG2P3_AW_CID[.@s], 0, .@rl$[.@s], $@NBG2R_F$[.@rv * 9 + 5 + .@s]) < 0)
				return callfunc("F_NEED_BG2P3_Rollback", "final_ticket");
			continue;
		}
		if (query_sql("INSERT IGNORE INTO need_bg2_reward (`mode`, week_start, account_id, reward_kind, reward_key, rank_slot, char_id, amount, status, created_at) VALUES ("
			+ $@NBG2P3_MODE + ", " + .@ws + ", " + $@NBG2P3_AW_AID[.@s] + ", 'RANK', '" + .@s + "', " + .@s + ", " + $@NBG2P3_AW_CID[.@s] + ", " + $@NBG2P3_RANK_CASH[.@s] + ", 0, " + .@now + ")") < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "final_ticket");
	}
	if (query_sql("INSERT INTO need_bg2_week (`mode`, week_start, state, accounts, detail, attempts, decided_at) VALUES (" + $@NBG2P3_MODE + ", " + .@ws + ", '" + .@state$ + "', "
		+ $@NBG2P3_AW_ACC + ", '" + escape_sql(.@detail$) + "', 1, " + .@now + ") ON DUPLICATE KEY UPDATE state = VALUES(state), accounts = VALUES(accounts), detail = VALUES(detail), attempts = attempts + 1, decided_at = VALUES(decided_at)") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "final_week");
	callfunc "F_NEED_BG2P3_Audit", .@actor, "WEEK_" + .@state$, "week=" + .@ws, .@detail$;
	if (callfunc("F_NEED_BG2P3_Exec", "COMMIT", "final_commit") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "final_commit");
	callfunc "F_NEED_BG2_Log", "P3_WEEK_" + .@state$ + " week=" + .@ws + " " + .@detail$;
	return .@award ? 0 : 1;
}

// 확정 시각(주차 시작 + 9일)이 지난, 아직 확정되지 않은 주차를 오래된 것부터 처리한다(누락 주차 복원).
function	script	F_NEED_BG2P3_FinalizeDue	{
	.@now = getarg(0, gettimetick(2));
	deletearray .@w;
	.@n = query_sql("SELECT DISTINCT m.week_start FROM need_bg2_match m LEFT JOIN need_bg2_week w ON w.`mode` = m.`mode` AND w.week_start = m.week_start WHERE m.`mode` = "
		+ $@NBG2P3_MODE + " AND m.valid = 1 AND m.week_start <= " + (.@now - $@NBG2P3_FINAL_AT) + " AND (w.state IS NULL OR w.state NOT IN ('FINALIZED', 'NO_AWARD')) ORDER BY m.week_start LIMIT 8", .@w);
	if (.@n < 0)
		return -1;
	for (.@i = 0; .@i < .@n; .@i++)
		callfunc "F_NEED_BG2P3_Finalize", .@w[.@i], 1, 0, "scheduled";
	return .@n;
}

//------------------------------------------------------------
// GM 검토 결정 / 탈주 면제 / 보상권 회수
//------------------------------------------------------------

// act: "approve" / "release" / "void". 성공 1, 거절 0, DB 실패 -1. 결과는 첨부된 GM 에게 출력.
function	script	F_NEED_BG2P3_Decide	{
	.@pk = getarg(0);
	.@act$ = getarg(1);
	.@gm = getcharid(3);
	.@reason$ = getarg(2);
	.@now = gettimetick(2);
	if (callfunc("F_NEED_BG2P3_Begin", "decide_begin") < 0)
		return -1;
	.@rv = -1;
	.@ws = 0;
	.@valid = 0;
	if (query_sql("SELECT review, week_start, valid FROM need_bg2_match WHERE match_pk = " + .@pk + " FOR UPDATE", .@rv, .@ws, .@valid) < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "decide_lock");
	.@ok = 0;
	if (.@rv < 0)
		.@msg$ = "경기를 찾지 못했습니다.";
	else if ((.@act$ == "approve" || .@act$ == "release") && .@rv != 1)
		.@msg$ = "검토 보류(HOLD) 상태가 아닙니다 (review=" + .@rv + ").";
	else if (.@act$ == "void" && .@rv == 3)
		.@msg$ = "이미 무효 처리된 경기입니다.";
	else
		.@ok = 1;
	if (!.@ok) {
		query_sql "ROLLBACK";
		dispbottom "[2:2 정산] " + .@msg$;
		return 0;
	}
	.@newrv = (.@act$ == "approve") ? 2 : ((.@act$ == "release") ? 0 : 3);
	.@state$ = (.@act$ == "approve") ? "APPROVED" : ((.@act$ == "release") ? "RELEASED" : "VOIDED");
	if (query_sql("UPDATE need_bg2_match SET review = " + .@newrv + " WHERE match_pk = " + .@pk) < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "decide_match");
	if (query_sql("INSERT INTO need_bg2_review (match_pk, signals, state, gm_aid, reason, created_at, decided_at) VALUES (" + .@pk + ", 'gm', '" + .@state$ + "', " + .@gm + ", '"
		+ escape_sql(.@reason$) + "', " + .@now + ", " + .@now + ") ON DUPLICATE KEY UPDATE state = VALUES(state), gm_aid = VALUES(gm_aid), reason = VALUES(reason), decided_at = VALUES(decided_at)") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "decide_review");
	deletearray .@acc;
	.@n = query_sql("SELECT DISTINCT account_id FROM need_bg2_match_player WHERE match_pk = " + .@pk + " AND valid = 1", .@acc);
	if (.@n < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "decide_accounts");
	.@chg = 0;
	for (.@i = 0; .@i < .@n; .@i++) {
		if (.@newrv == 3)
			.@r = callfunc("F_NEED_BG2P3_TierRevoke", .@acc[.@i], .@ws);
		else
			.@r = callfunc("F_NEED_BG2P3_TierSync", .@acc[.@i], .@ws);
		if (.@r < 0)
			return callfunc("F_NEED_BG2P3_Rollback", "decide_tier");
		.@chg += .@r;
	}
	.@r = callfunc("F_NEED_BG2P3_MatchRewardDecide", .@pk, .@newrv);
	if (.@r < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "decide_match_rewards");
	.@chg += .@r;
	.@wst$ = "";
	query_sql "SELECT state FROM need_bg2_week WHERE `mode` = " + $@NBG2P3_MODE + " AND week_start = " + .@ws, .@wst$;
	.@detail$ = "review " + .@rv + "->" + .@newrv + " tickets_changed=" + .@chg + " week_state=" + .@wst$ + " reason=" + .@reason$;
	callfunc "F_NEED_BG2P3_Audit", .@gm, "REVIEW_" + .@state$, "match_pk=" + .@pk, .@detail$;
	if (callfunc("F_NEED_BG2P3_Exec", "COMMIT", "decide_commit") < 0)
		return callfunc("F_NEED_BG2P3_Rollback", "decide_commit");
	callfunc "F_NEED_BG2_Log", "P3_REVIEW_" + .@state$ + " match_pk=" + .@pk + " by_aid=" + .@gm + " " + .@detail$;
	dispbottom "[2:2 정산] 경기 " + .@pk + " → " + .@state$ + " (보상권 변경 " + .@chg + "건)";
	if (.@wst$ == "FINALIZED" || .@wst$ == "NO_AWARD")
		dispbottom "[2:2 정산] 주의: 이 주차 랭킹은 이미 " + .@wst$ + " 입니다. 순위 보상은 자동으로 다시 쓰지 않습니다(감사 로그 확인 후 수동 정산).";
	return 1;
}

//------------------------------------------------------------
// 보상 수령 (P3-B)
//------------------------------------------------------------

// 첨부된 플레이어가 보상권 id 를 받는다. 1 지급, 0 거절.
// 순서: 보상권 0 → 1 (조건부 UPDATE + ROW_COUNT, 순번 claim_seq) → #CASHPOINTS 증가 + 영수증 #NEED_BG2_PAIDSEQ = claim_seq.
// 두 계정 변수는 이 스크립트가 끝날 때 함께 저장된다. 다음 로그인 때 F_NEED_BG2P3_Reconcile 이 1 을 2(지급) 또는 0(미지급)으로 정리한다.
function	script	F_NEED_BG2P3_Claim	{
	.@id = getarg(0);
	.@aid = getcharid(3);
	.@cid = getcharid(0);
	.@now = gettimetick(2);
	.@amt = 0;
	.@st = -1;
	.@k$ = "";
	if (query_sql("SELECT amount, status, reward_kind FROM need_bg2_reward WHERE id = " + .@id + " AND account_id = " + .@aid, .@amt, .@st, .@k$) < 1 || .@st != 0
		|| (.@k$ != "TIER" && .@k$ != "RANK" && .@k$ != "MCASH")) {
		dispbottom needtr("[2:2 듀오] 받을 수 있는 보상이 아닙니다(이미 받았거나 없음).", "[2v2 Duo] This reward cannot be claimed (already claimed or not found)."), 0xFF7777;
		return 0;
	}
	if (!callfunc("F_NEED_BG2P3_KindOpen", .@k$)) {
		dispbottom needtr("[2:2 듀오] 지금은 보상 수령을 받지 않습니다.", "[2v2 Duo] Reward claiming is not open now."), 0xFF7777;
		return 0;
	}
	if (#CASHPOINTS > 2147483647 - .@amt) {
		dispbottom needtr("[2:2 듀오] 캐시 보유 한도를 넘어 지급할 수 없습니다.", "[2v2 Duo] Your cash points would exceed the limit."), 0xFF7777;
		return 0;
	}
	.@mx = 0;
	if (query_sql("SELECT COALESCE(MAX(claim_seq), 0) FROM need_bg2_reward WHERE account_id = " + .@aid, .@mx) < 0) {
		dispbottom needtr("[2:2 듀오] 지금은 처리할 수 없습니다. 잠시 후 다시 시도하세요.", "[2v2 Duo] Cannot process now. Please try again later."), 0xFF7777;
		return 0;
	}
	.@seq = ((.@mx > #NEED_BG2_PAIDSEQ) ? .@mx : #NEED_BG2_PAIDSEQ) + 1;
	if (query_sql("UPDATE need_bg2_reward SET status = 1, claim_seq = " + .@seq + ", claim_char = " + .@cid + ", claimed_at = " + .@now + " WHERE id = " + .@id + " AND account_id = " + .@aid + " AND status = 0") < 0) {
		dispbottom needtr("[2:2 듀오] 지금은 처리할 수 없습니다. 잠시 후 다시 시도하세요.", "[2v2 Duo] Cannot process now. Please try again later."), 0xFF7777;
		return 0;
	}
	.@rc = 0;
	query_sql "SELECT ROW_COUNT()", .@rc;
	if (.@rc != 1) {
		callfunc "F_NEED_BG2P3_Log", "P3_CLAIM_RACE id=" + .@id + " aid=" + .@aid + " cid=" + .@cid;
		dispbottom needtr("[2:2 듀오] 이미 처리된 보상입니다.", "[2v2 Duo] This reward was already processed."), 0xFF7777;
		return 0;
	}
	$NEED_BG2_P3_CLAIMUSED = 1;
	#CASHPOINTS += .@amt;
	#NEED_BG2_PAIDSEQ = .@seq;
	callfunc "F_NEED_BG2P3_Audit", .@aid, "CLAIM_PAID", "id=" + .@id, "cid=" + .@cid + " amount=" + .@amt + " seq=" + .@seq + " cash_after=" + #CASHPOINTS;
	callfunc "F_NEED_BG2P3_Log", "P3_CLAIM id=" + .@id + " aid=" + .@aid + " cid=" + .@cid + " amount=" + .@amt + " seq=" + .@seq + " cash_after=" + #CASHPOINTS;
	dispbottom needtr("[2:2 듀오] " + .@amt + " 캐시를 받았습니다. (보유 " + #CASHPOINTS + ")", "[2v2 Duo] You received " + .@amt + " cash points. (Now " + #CASHPOINTS + ")"), 0x66FF66;
	return 1;
}

// 로그인(첨부): RESERVED 보상권을 영수증으로 확정/복원한다.
function	script	F_NEED_BG2P3_Reconcile	{
	.@aid = getcharid(3);
	deletearray .@id;
	deletearray .@seq;
	.@n = query_sql("SELECT id, claim_seq FROM need_bg2_reward WHERE account_id = " + .@aid + " AND status = 1 AND reward_kind IN ('TIER', 'RANK', 'MCASH')", .@id, .@seq);
	for (.@i = 0; .@i < .@n; .@i++) {
		if (.@seq[.@i] <= #NEED_BG2_PAIDSEQ) {
			query_sql "UPDATE need_bg2_reward SET status = 2 WHERE id = " + .@id[.@i] + " AND status = 1";
			.@what$ = "CLAIMED";
		} else {
			query_sql "UPDATE need_bg2_reward SET status = 0, claim_seq = 0, claim_char = 0, claimed_at = 0, note = 'reserved but not paid: restored' WHERE id = " + .@id[.@i] + " AND status = 1";
			.@what$ = "RESTORED";
		}
		callfunc "F_NEED_BG2P3_Audit", 0, "CLAIM_" + .@what$, "id=" + .@id[.@i], "aid=" + .@aid + " seq=" + .@seq[.@i] + " paidseq=" + #NEED_BG2_PAIDSEQ;
		callfunc "F_NEED_BG2P3_Log", "P3_RECONCILE id=" + .@id[.@i] + " aid=" + .@aid + " seq=" + .@seq[.@i] + " paidseq=" + #NEED_BG2_PAIDSEQ + " -> " + .@what$;
	}
	// 아이템 보상권: 받은 캐릭터의 영수증(NEED_BG2_IPAIDSEQ)으로만 확정한다(다른 캐릭터 접속으로는 판단하지 않음).
	deletearray .@id;
	deletearray .@seq;
	.@m = query_sql("SELECT id, claim_seq FROM need_bg2_reward WHERE account_id = " + .@aid + " AND status = 1 AND reward_kind IN ('MITEM', 'TIERI', 'RANKI') AND claim_char = " + getcharid(0), .@id, .@seq);
	for (.@i = 0; .@i < .@m; .@i++) {
		if (.@seq[.@i] <= NEED_BG2_IPAIDSEQ) {
			query_sql "UPDATE need_bg2_reward SET status = 2 WHERE id = " + .@id[.@i] + " AND status = 1";
			.@what$ = "CLAIMED";
		} else {
			query_sql "UPDATE need_bg2_reward SET status = 0, claim_seq = 0, claim_char = 0, claimed_at = 0, note = 'reserved but not given: restored' WHERE id = " + .@id[.@i] + " AND status = 1";
			.@what$ = "RESTORED";
		}
		callfunc "F_NEED_BG2P3_Audit", 0, "ITEM_" + .@what$, "id=" + .@id[.@i], "cid=" + getcharid(0) + " seq=" + .@seq[.@i] + " ipaidseq=" + NEED_BG2_IPAIDSEQ;
		callfunc "F_NEED_BG2P3_Log", "P3_RECONCILE_ITEM id=" + .@id[.@i] + " cid=" + getcharid(0) + " seq=" + .@seq[.@i] + " ipaidseq=" + NEED_BG2_IPAIDSEQ + " -> " + .@what$;
	}
	return .@n + .@m;
}

// 승패 직접 캐시 지급: 경기 match_pk 의 미수령(0) MCASH 보상권을 접속 중인 계정에 지급한다(첨부를 바꿔 가며).
function	script	F_NEED_BG2P3_PayMatchCash	{
	deletearray .@id;
	deletearray .@aid;
	.@n = query_sql("SELECT id, account_id FROM need_bg2_reward WHERE reward_kind = 'MCASH' AND reward_key = '" + getarg(0) + "' AND status = 0", .@id, .@aid);
	.@caller = playerattached();
	for (.@i = 0; .@i < .@n; .@i++) {
		if (!isloggedin(.@aid[.@i]) || !attachrid(.@aid[.@i]))
			continue;
		callfunc "F_NEED_BG2P3_Claim", .@id[.@i];
		detachrid;
	}
	callfunc "F_NEED_BG2_Reattach", .@caller;
	return;
}

function	script	F_NEED_BG2P3_TicketName$	{
	if (getarg(0) == "RANK" || getarg(0) == "RANKI")
		return needtr("주간 랭킹 " + getarg(1) + "위", "Weekly rank " + getarg(1));
	if (getarg(0) == "MCASH")
		return needtr("경기 보상(캐시)", "Match reward (cash)");
	if (getarg(0) == "MITEM")
		return needtr("경기 보상(아이템)", "Match reward (items)");
	return needtr(getarg(1) + "점 단계", getarg(1) + "-point tier");
}

			next;
			break;
		case 7:
			.@left = callfunc("F_NEED_BG2P3_Sanction", .@aid, .@now);
			mes needtr("최근 24시간 경기 이탈: " + $@NBG2P3_SC_CNT + "회", "Matches left in the last 24 hours: " + $@NBG2P3_SC_CNT);

			mes needtr("1회 경고, 2회 10분, 3회 30분, 4회 이상 2시간. 재접속 15초 유예 뒤 이탈만 셉니다.", "1 = warning, 2 = 10 min, 3 = 30 min, 4+ = 2 h. Only departures after the 15-second reconnect grace count.");
			next;
			break;
		default:
			close;
		}
		mes .@n$;
	}
	close;
}

			callfunc "F_NEED_BG2P3_Audit", .@gm, "TICKET_REVOKE", "id=" + atoi(.@a1$), .@why$;
		if (.@rc && .@rk$ == "MCASH") {
			// 승패 직접 캐시 회수: 그 계정의 IP 하루 할당을 돌려준다.
			.@ip$ = ""; .@dk = 0; .@al = 0;
			if (query_sql("SELECT ip, day_key, allocated FROM need_bg2_ipcash_alloc WHERE match_pk = " + atoi(.@rkey$) + " AND account_id = " + .@raid + " AND state IN (0, 1)", .@ip$, .@dk, .@al) > 0) {
				query_sql "UPDATE need_bg2_ipcash_day SET used = IF(used >= " + .@al + ", used - " + .@al + ", 0) WHERE ip = '" + escape_sql(.@ip$) + "' AND day_key = " + .@dk;
				query_sql "UPDATE need_bg2_ipcash_alloc SET state = 3 WHERE match_pk = " + atoi(.@rkey$) + " AND account_id = " + .@raid;
				callfunc "F_NEED_BG2P3_Log", "P3_MCASH_RELEASED match_pk=" + .@rkey$ + " aid=" + .@raid + " day=" + .@dk + " amount=" + .@al + " (GM revoke)";
			}
		}
		dispbottom "[2:2 정산] 보상권 " + atoi(.@a1$) + (.@rc ? " 회수" : " : 미수령(0)·검토 대기(7) 상태가 아니라 회수하지 않음");
		end;
	}
	// 보상 규칙 버전 목록과 검사 결과.
	if (.@sub$ == "rules") {
		dispbottom "[2:2 정산] 보상 규칙 " + ($@NBG2P3_RULES_OK ? "OK" : "오류: " + $@NBG2R_ERR$) + " / 버전 " + $@NBG2R_N + "개 / 이번 주 v" + callfunc("F_NEED_BG2P3_RuleVer", .@now)
			+ " / 다음 주(" + callfunc("F_NEED_BG2P3_Date$", .@ws + $@NBG2P3_WEEK) + ") v" + callfunc("F_NEED_BG2P3_RuleVer", .@ws + $@NBG2P3_WEEK);
		for (.@v = 0; .@v < $@NBG2R_N; .@v++) {
			dispbottom "[2:2 정산] v" + .@v + " 시작 " + ($@NBG2R_EFF[.@v] ? callfunc("F_NEED_BG2P3_Date$", $@NBG2R_EFF[.@v]) : "처음") + " 점수 " + $@NBG2R_PW[.@v] + "/" + $@NBG2R_PL[.@v] + "/" + $@NBG2R_PD[.@v]
				+ " 캐시 " + $@NBG2R_CW[.@v] + "/" + $@NBG2R_CL[.@v] + "/" + $@NBG2R_CD[.@v] + " IP하루 " + $@NBG2R_CAP[.@v] + " 주간 " + ($@NBG2R_WM[.@v] ? "아이템" : "캐시(기존)");
			dispbottom "[2:2 정산]   아이템 승[" + $@NBG2R_IW$[.@v] + "] 패[" + $@NBG2R_IL$[.@v] + "] 무[" + $@NBG2R_ID$[.@v] + "] 10/30/60[" + $@NBG2R_T10$[.@v] + "|" + $@NBG2R_T30$[.@v] + "|" + $@NBG2R_T60$[.@v]
				+ "] 1/2/3위[" + $@NBG2R_R1$[.@v] + "|" + $@NBG2R_R2$[.@v] + "|" + $@NBG2R_R3$[.@v] + "]";
		}
		end;
	}
	// IP 하루 승패 캐시 사용량: ipcash <IP|aid|접속중 이름> [유닉스초(그날)]
	if (.@sub$ == "ipcash") {
		.@t = (.@a2$ != "") ? atoi(.@a2$) : .@now;
		.@day = (.@t + 18000) / 86400;
		.@cap = $@NBG2R_CAP[callfunc("F_NEED_BG2P3_RuleVer", .@t)];
		if (.@a1$ == "") {
			dispbottom "[2:2 정산] 사용법: @duo2v2p3 ipcash <IP|aid|이름> [유닉스초]";
			end;
		}
		deletearray .@ip$;
		if (compare(.@a1$, ".") || compare(.@a1$, ":"))
			.@ip$[0] = .@a1$;
		else {
			.@aid = callfunc("F_NEED_BG2P3_GmAid", .@a1$);
			query_sql "SELECT DISTINCT ip FROM need_bg2_ipcash_alloc WHERE account_id = " + .@aid + " AND day_key = " + .@day + " AND ip <> ''", .@ip$;
		}
		dispbottom "[2:2 정산] 하루 " + callfunc("F_NEED_BG2P3_Date$", .@day * 86400 - 18000) + " ~ +24h KST, 한도 " + .@cap + " (0 = 직접 캐시 꺼짐), IP " + getarraysize(.@ip$) + "개";
		for (.@i = 0; .@i < getarraysize(.@ip$); .@i++) {
			.@u = 0;
			query_sql "SELECT used FROM need_bg2_ipcash_day WHERE ip = '" + escape_sql(.@ip$[.@i]) + "' AND day_key = " + .@day, .@u;
			dispbottom "[2:2 정산]  " + .@ip$[.@i] + " 사용 " + .@u;
			deletearray .@pa; deletearray .@pm; deletearray .@pr; deletearray .@pg; deletearray .@ps;
			.@c = query_sql("SELECT account_id, match_pk, requested, allocated, state FROM need_bg2_ipcash_alloc WHERE ip = '" + escape_sql(.@ip$[.@i]) + "' AND day_key = " + .@day + " ORDER BY match_pk, account_id LIMIT 12",
				.@pa, .@pm, .@pr, .@pg, .@ps);
			for (.@j = 0; .@j < .@c; .@j++)
				dispbottom "[2:2 정산]   aid=" + .@pa[.@j] + " match_pk=" + .@pm[.@j] + " 요청 " + .@pr[.@j] + " 할당 " + .@pg[.@j] + " state=" + .@ps[.@j] + " (0 할당 1 보류 3 반납 4 IP없음)";
		}
		end;
	}
	if (.@sub$ == "pending") {
		.@n = getarraysize($NEED_BG2_PEND_H$);
		dispbottom "[2:2 정산] 보관 정산 " + .@n + "건 (검토 보관함 " + callfunc("F_NEED_BG2P3_PartCount") + "건: @duo2v2p3 partial)";
		for (.@i = 0; .@i < .@n && .@i < 10; .@i++)
			dispbottom "[2:2 정산]  " + .@i + ": " + $NEED_BG2_PEND_H$[.@i];
		end;
	}
	if (.@sub$ == "retry") {
		.@d = callfunc("F_NEED_BG2P3_PendRetry");
		dispbottom "[2:2 정산] 재시도 " + .@d + "건 처리 (적용 " + $@NBG2P3_RT_OK + ", 정상 중복 " + $@NBG2P3_RT_DUP + ", 검토 보관 " + $@NBG2P3_RT_PART + ", DB 오류 " + $@NBG2P3_RT_ERR
			+ "), 남음 " + getarraysize($NEED_BG2_PEND_H$) + "건 / 한 번에 최대 " + $@NBG2P3_PEND_BATCH + "건";
		end;
	}
	// 검토 보관함: 목록, 또는 GM 이 원장을 대조·정리한 뒤 보관 정산 끝으로 되돌리기(다시 검사되며, 여전히 반쪽이면 보관함으로 돌아온다).
	if (.@sub$ == "partial") {
		.@n = getarraysize($NEED_BG2_PART_H$);
		if (.@a1$ == "requeue") {
			.@x = atoi(.@a2$);
			if (.@a2$ == "" || .@x < 0 || .@x >= .@n || .@why3$ == "" || $NEED_BG2_PART_H$[.@x] == "") {
				dispbottom "[2:2 정산] 사용법: @duo2v2p3 partial requeue <번호> <사유>";
				end;
			}
			if (!callfunc("F_NEED_BG2P3_PendPush", $NEED_BG2_PART_H$[.@x], $NEED_BG2_PART_P$[.@x])) {
				dispbottom "[2:2 정산] 보관 정산이 가득 차 되돌리지 못했습니다.";
				end;
			}
			explode(.@hk$, $NEED_BG2_PART_H$[.@x], "|");
			.@det$ = "why=" + $NEED_BG2_PART_R$[.@x] + " reason=" + .@why3$;
			if (getstrlen(.@det$) > 250)
				.@det$ = substr(.@det$, 0, 249);
			callfunc "F_NEED_BG2P3_Audit", .@gm, "PARTIAL_REQUEUE", "key=" + .@hk$[0], .@det$;	// target 64자, detail 255바이트 열
			callfunc "F_NEED_BG2P3_Log", "P3_PARTIAL_REQUEUE by_aid=" + .@gm + " head=" + $NEED_BG2_PART_H$[.@x];
			$NEED_BG2_PART_H$[.@x] = "";	// 당기지 않고 비운다(즉시 DB 삭제)
			$NEED_BG2_PART_P$[.@x] = "";
			$NEED_BG2_PART_R$[.@x] = "";
			dispbottom "[2:2 정산] 검토 보관 " + .@x + "번을 보관 정산 끝으로 되돌렸습니다.";
			end;
		}
		dispbottom "[2:2 정산] 검토 보관함 " + callfunc("F_NEED_BG2P3_PartCount") + "건 (반쪽·불일치 원장 / 형식 오류. 자동 재시도·삭제 없음)";
		.@shown = 0;
		for (.@i = 0; .@i < .@n && .@shown < 10; .@i++) {
			if ($NEED_BG2_PART_H$[.@i] == "")
				continue;
			dispbottom "[2:2 정산]  " + .@i + ": " + $NEED_BG2_PART_R$[.@i] + " | " + $NEED_BG2_PART_H$[.@i] + " | " + $NEED_BG2_PART_P$[.@i];
			.@shown++;
		}
		end;
	}
	dispbottom "[2:2 정산] @duo2v2p3 status | set <ledger|rank|claim|mcash|mitem|witem> <on|off> <사유> | rules | ipcash <IP|aid|이름> [유닉스초] | week [유닉스초] | points <aid|이름> [주차] | tickets <aid|이름>";
	dispbottom "[2:2 정산] rank [cur|주차] | finalize <주차> dry|commit <사유> | due | reviews | match <pk> | approve|release|void <pk> <사유>";
	dispbottom "[2:2 정산] deserts <aid|이름> | exempt <id> <사유> | revoke <보상권 id> <사유> | pending | retry | partial [requeue <번호> <사유>]";
	end;

S_Tickets:
	deletearray .@id; deletearray .@w; deletearray .@k$; deletearray .@key$; deletearray .@amt; deletearray .@st; deletearray .@sq; deletearray .@nt$;
	.@c = query_sql("SELECT id, week_start, reward_kind, reward_key, amount, status, claim_seq, note FROM need_bg2_reward WHERE account_id = " + getarg(0) + " ORDER BY id DESC LIMIT 12",
		.@id, .@w, .@k$, .@key$, .@amt, .@st, .@sq, .@nt$);
	dispbottom "[2:2 정산] aid " + getarg(0) + " 보상권 최근 " + .@c + "건 (status 0 미수령 1 지급확인대기 2 지급 7 검토대기 8 회수 9 수동정산)";
	for (.@i = 0; .@i < .@c; .@i++)
		dispbottom "[2:2 정산]  id=" + .@id[.@i] + " week=" + .@w[.@i] + " " + .@k$[.@i] + ":" + .@key$[.@i] + " " + callfunc("F_NEED_BG2P3_TicketText$", .@id[.@i], .@k$[.@i], .@amt[.@i]) + " status=" + .@st[.@i] + " seq=" + .@sq[.@i] + " " + .@nt$[.@i];
	return;
}
