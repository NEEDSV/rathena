//============================================================
//= 맵    : need_bg2 (pvp_2vs2 지형 복제, npc/NEED/duo2v2/need_bg2_mapflag.txt).
//=         내부 미로만 쓰고, 중앙 6x6 밀폐 박스(37..42)는 탈락자 관전석이다.
//=         중앙 거점 = 관전석을 둘러싼 2칸 폭 고리(x/y 34..45 에서 36..43 제외, 80칸). 양 팀 시작점에서 최단 40칸.
//= 팀    : rAthena 엔진 BG 팀 2개(블루/레드). 아군/적군은 bg_id 로만 판정한다(battle.cpp battle_check_target).
//=         파티/길드는 팀을 정하지 않고, 파티를 만들거나 해산하지도 않는다. 같은 파티원이 서로 다른 팀이 되어도
//=         맵플래그 need_bgpartysplit 이 파티 스킬/인원 계산/합주 상대에서 다른 BG 팀을 뺀다(party.cpp 등).
//= 신청  : 공개 개인 FIFO 대기열($NEED_BG2_QUEUE_OPEN, GM 이 @duo2v2 queue on/off). 계정당 1건, 다른 NEED 전장
//=         신청/대기와 양방향 중복 금지(@NEED_BG2_BUSY). 경기장이 비고 4명이 모이면 맨 앞 4명에게 30초 참가 확인.
//=         GM 수동 4인 테스트(@duo2v2 open/start)는 그대로 있고, 경기장/참가 확인과 겹치지 않는다.
//= 편성  : 확정된 4명 안에서만. 지원 점수(아크비숍 2, 소서러/민스트럴/원더러 1) 합 차이 최소 → 같은 직업 집중 최소
//=         → 무작위. 팀 색 무작위.
//= 라운드: 준비 45초(첫 라운드)/15초, 90초 라운드, 2선승(최대 5라운드). 60초부터 중앙 거점, 단독 점유 10초 = 승리.
//=         90초 종료 시 생존 인원 → 단독 점유 시간 → 무승부.
//= 보상  : 없음. 이 파일은 보상/전적/일일퀘스트/마블/이벤트 함수를 하나도 호출하지 않는다.
//= 명령  : @duo2v2 (@듀오관리)  GM 60+ : 대기열 열기/닫기, GM 테스트 모집/시작, 상태, 강제 중단, 고립자 복구, 설정
//=         @duojoin (@듀오참가) / @duocancel (@듀오취소) / @duostatus (@듀오현황)
//=         @duoaccept (@듀오수락) / @duodecline (@듀오거절) : 참가 확인 응답
//= 상태  : 0 IDLE  1 RECRUITING(GM 테스트)  2 PREPARING  3 ROUND_READY  4 ROUND_ACTIVE  5 ROUND_RESULT
//=         6 MATCH_FINISHED  7 CLEANUP.  ABORTED 는 결과 VOID/CANCELLED 로 6을 거쳐 7로 간다.
//=         참가 확인($@NBG2_RC)은 경기장이 IDLE 일 때만 열린다.
//= 판정  : 사망 = BG 팀 die 이벤트(status_damage), 접속 종료 = BG 팀 quit 이벤트(map_quit),
//=         맵/팀 이탈 = 0.5초 감시. 전이 함수는 sleep 없이 상태를 먼저 바꾼 뒤 처리하므로
//=         같은 결과가 두 번 확정되지 않는다.
//= 보완  : (2026-10-26 확정 정책, 2단계에서도 유지)
//=         - 자동 부활 금지: 맵플래그 need_noautorevive(엔진) + 입장/라운드마다 자동 부활 상태 해제.
//=         - 준비시간: 자기 팀 준비구역 안 이동 허용, 경계 셀 차단(setcell), 벗어나면 시작점 복귀,
//=           공격 불가/피해 면역, 지면 설치·소환 스킬은 맵플래그 need_preplock 으로 엔진이 거부.
//=         - 라운드 사이: 전투 스킬 버프·디버프 정리(need_duo2v2_sclist.txt), 음식/포션/캐시/직업 상태/쿨타임 유지,
//=           동반 유닛 HP/SP 회복, 임시 소환물·설치물 제거.
//=         - 종료/취소/무효/복구: 항상 저장 지점. 접속 중 경기장 이탈(시스템/GM 이동)은 경기 무효.
//=         - 접속 종료: 경기당 개인 누적 15초 재접속 유예(돌아오면 원래 팀으로 재합류).
//= 로그  : debugmes "[NEED_BG2] ..." -> log/map-msg_log.log. 계정/캐릭터 ID만 남긴다.
//=         $NEED_BG2_LOG_OFF = 1 이면 끈다.
//= 변수  : $@NBG2_*                        임시. 리로드/재시작 시 사라지므로 그 경기는 무효, 대기열은 재신청.
//=         NEED_BG2_MID   캐릭터. 재접속 시 재합류/정리 판단용(복귀 지점은 항상 저장 지점).
//=         (NEED_BG2_RMAP$ / RX / RY 는 이전 버전 값 정리용으로 비우기만 한다)
//=         @NEED_BG2_BGID / @NEED_BG2_PB   세션. 리로드 후 정리용.
//=         @NEED_BG2_BUSY  세션. 1 대기/GM 명단, 2 참가 확인, 3 경기. 기존 전장 신청 검사가 읽는다.
//=         $NEED_BG2_SEQ (참고 순번), $NEED_BG2_LVBYPASS (GM 테스트 옵션, 공개 대기열에는 적용 안 함), $NEED_BG2_LOG_OFF,
//=         $NEED_BG2_QUEUE_OPEN (공개 대기열 스위치, 기본 0), $NEED_BG2_LASTMID (마지막 매치 ID - 같은 초 충돌 방지),
//=         $NEED_BG2_NP[32] / $NEED_BG2_NP_POS (최근 무효·취소 경기 번호 - 재접속 시 탈주 제재 해제용)
//============================================================

	setarray $@NBG2_SPX[0], 15, 16, 64, 63;
	setarray $@NBG2_SPY[0], 39, 40, 40, 39;

	setarray $@NBG2_BOXX[1], 38, 41;

	$@NBG2_C_PREP = 45000;
	$@NBG2_C_READY = 15000;
	$@NBG2_C_ROUND = 90000;
	$@NBG2_C_SHOW = 5000;
	$@NBG2_C_SIMUL = 500;		// 첫 전멸 후 이 시간 안의 사망은 동시 사망으로 본다
	$@NBG2_C_GRACE = 1000;		// 첫 이탈 후 이 시간 안의 이탈은 동시 이탈로 본다
	$@NBG2_C_WINS = 2;
	$@NBG2_C_MAXROUND = 5;		// 기본 3라운드 + 연장 최대 2라운드
	$@NBG2_C_MINLV = 150;		// NEED 기본 전장 참가 레벨
	$@NBG2_C_RECONNECT = 15000;	// 경기당 개인별 누적 재접속 유예

	$@NBG2_C_CAP_ACTIVE = 60000;	// 라운드 시작 후 중앙 거점이 열리는 시점
	$@NBG2_C_CAP_WIN = 10000;	// 단독 점유 게이지가 이만큼 차면 그 팀 라운드 승리

	$@NBG2_C_CAP_ANN = 2000;	// 점령 상태 공지의 최소 간격(도배 방지)
	// 준비시간: 자기 팀 준비구역 안에서는 이동 가능, 공격 불가, 피해 면역. 스킬/아이템은 허용하되

// 준비시간 경계: on = 1 이면 경계 셀을 막고 지면 설치·소환 스킬을 잠근다. 0 이면 둘 다 푼다. 여러 번 불려도 같다.
function	script	F_NEED_BG2_PrepLock	{
	.@on = getarg(0);
	for (.@k = 0; .@k < getarraysize($@NBG2_WX); .@k++)
		setcell $@NBG2_MAP$, $@NBG2_WX[.@k], $@NBG2_WY[.@k], $@NBG2_WX[.@k], $@NBG2_WY[.@k], cell_walkable, !.@on;
	if (.@on)
		setmapflag $@NBG2_MAP$, mf_need_preplock;
	else
		removemapflag $@NBG2_MAP$, mf_need_preplock;
	$@NBG2_PREPLOCK = .@on;
	return;
}

function	script	F_NEED_BG2_InZone	{
	.@t = getarg(0);
	return (getarg(1) >= $@NBG2_ZX1[.@t] && getarg(1) <= $@NBG2_ZX2[.@t] && getarg(2) >= $@NBG2_ZY1[.@t] && getarg(2) <= $@NBG2_ZY2[.@t]);
}

// 첨부된 플레이어의 동반 유닛: 살아 있으면 HP/SP 최대치, 공격 대상 해제. 죽은/소환 해제된 유닛은 건드리지 않는다.
// 맵 유닛 검색이 아니라 주인에게서 GID 를 얻는다(같은 맵 워프 직후 동반 유닛은 잠시 맵에서 빠져 있어 검색에 안 잡힌다).
// 위치는 주인의 warp 가 함께 옮긴다(pc_setpos).
function	script	F_NEED_BG2_Companions	{
	setarray .@g[0], gethominfo(7), getmercinfo(8), geteleminfo(ELEMINFO_GAMEID);
	for (.@t = 0; .@t < 3; .@t++) {
		if (.@g[.@t] <= 0 || !unitexists(.@g[.@t]))
			continue;
		deletearray .@d;
		getunitdata .@g[.@t], .@d;
		if (.@t == 0 && .@d[UHOM_HP] > 0) {
			setunitdata .@g[.@t], UHOM_HP, .@d[UHOM_MAXHP];
			setunitdata .@g[.@t], UHOM_SP, .@d[UHOM_MAXSP];
		} else if (.@t == 1 && .@d[UMER_HP] > 0) {
			setunitdata .@g[.@t], UMER_HP, .@d[UMER_MAXHP];
		} else if (.@t == 2 && .@d[UELE_HP] > 0) {
			setunitdata .@g[.@t], UELE_HP, .@d[UELE_MAXHP];
			setunitdata .@g[.@t], UELE_SP, .@d[UELE_MAXSP];
		} else
			continue;
		unitstopattack .@g[.@t];
		.@cnt++;
	}
	return .@cnt;
}

// 참가자가 소환한 임시 몬스터(플로라, 마린스피어, 실버스나이퍼, 분신 등)를 모두 없앤다.
function	script	F_NEED_BG2_KillSummons	{
	.@n = getmapunits(BL_MOB, $@NBG2_MAP$, .@g);
	for (.@k = 0; .@k < .@n; .@k++) {
		deletearray .@d;
		getunitdata .@g[.@k], .@d;
		for (.@i = 0; .@i < $@NBG2_N; .@i++) {
			if (.@d[UMOB_MASTERAID] != $@NBG2_AID[.@i])
				continue;
			unitkill .@g[.@k];
			.@cnt++;
			break;
		}
	}
	if (.@cnt)
		callfunc "F_NEED_BG2_Log", "SUMMONS_REMOVED match=" + $@NBG2_MATCH + " n=" + .@cnt;
	return .@cnt;
}

function	script	F_NEED_BG2_Log	{
	if (!$NEED_BG2_LOG_OFF)
		debugmes "[NEED_BG2] " + getarg(0);
	return;
}

function	script	F_NEED_BG2_StateName$	{
	switch (getarg(0)) {
	case 0: return "IDLE";
	case 1: return "RECRUITING";
	case 2: return "PREPARING";
	case 3: return "ROUND_READY";
	case 4: return "ROUND_ACTIVE";
	case 5: return "ROUND_RESULT";
	case 6: return "MATCH_FINISHED";
	case 7: return "CLEANUP";
	}
	return "UNKNOWN";
}

function	script	F_NEED_BG2_TeamName$	{
	if (getarg(0) == 1) return needtr("블루", "Blue");
	if (getarg(0) == 2) return needtr("레드", "Red");
	return "-";
}

// 첨부된 플레이어에게 거절 사유를 보여 줄 문장.
function	script	F_NEED_BG2_Reason$	{
	switch (getarg(0)) {
	case 1: return needtr("접속 중이 아닙니다.", "not online.");
	case 2: return needtr("사망 상태입니다.", "dead.");
	case 3: return needtr("이미 전장 팀에 소속되어 있습니다.", "already in a battleground team.");
	case 4: return needtr("전장/PvP/GvG/인스턴스/이동 금지 맵에서는 신청할 수 없습니다.", "cannot sign up from a battleground, PvP, GvG, instance or no-warp map.");
	case 5: return needtr("BaseLevel " + $@NBG2_C_MINLV + " 이상만 참가할 수 있습니다.", "BaseLevel " + $@NBG2_C_MINLV + " or higher is required.");
	case 6: return needtr("노점/자동 판매 중에는 참가할 수 없습니다.", "cannot join while vending.");
	case 8: return needtr("다른 전장(랜덤 매칭/전장 대기열/파티 입장 등록/전장 신청)에 신청 중입니다. 그쪽을 먼저 취소하세요.", "you are signed up for another battleground (random match, battleground queue, party entry or battleground application). Cancel it first.");
	case 9: return needtr("이 계정은 이미 2:2 대기열/참가 확인/경기에 있습니다.", "this account is already in the 2v2 queue, a ready check or a match.");
	case 10: return needtr("지금은 참가 신청을 받지 않습니다.", "sign-ups are not open now.");
	case 11: return needtr("정원 4명이 모두 찼습니다.", "all 4 places are taken.");
	case 12: return needtr("거래 중에는 참가할 수 없습니다.", "cannot join while trading.");
	}
	return "?";
}

// 복귀 지점이 안전한지: 존재하는 일반 맵, 전장/PvP/GvG/인스턴스가 아니고 이동 가능한 칸.
function	script	F_NEED_BG2_ReturnOk	{
	.@m$ = getarg(0);
	if (.@m$ == "" || .@m$ == $@NBG2_MAP$)
		return 0;
	if (getmapusers(.@m$) < 0 || compare(.@m$, "@"))
		return 0;
	if (getmapflag(.@m$, mf_battleground) || getmapflag(.@m$, mf_pvp) || getmapflag(.@m$, mf_gvg)
		|| getmapflag(.@m$, mf_gvg_castle) || getmapflag(.@m$, mf_gvg_te) || getmapflag(.@m$, mf_gvg_te_castle))
		return 0;
	if (!checkcell(.@m$, getarg(1), getarg(2), cell_chkpass))
		return 0;
	return 1;
}

// 첨부된 플레이어가 기존 NEED 전장에 신청/등록되어 있는지(계정 기준). 읽기만 하고 아무것도 바꾸지 않는다.
// 랜덤 매칭 모집·사이클 팀, 전장 대기열(@전장대기), 전장 관리인 파티 입장 등록(파티 단위),
// 엔진 전장 대기열(패킷 신청), 전장 팀 소속.
function	script	F_NEED_BG2_InRandomMatch	{
	.@aid = getarg(0);
	for (.@i = 0; .@i < $@BG_RM_COUNT; .@i++)
		if ($@BG_RM_AID[.@i] == .@aid) return 1;
	for (.@i = 0; .@i < $@BG_RM_G_COUNT; .@i++)
		if ($@BG_RM_G_AID[.@i] == .@aid) return 1;
	for (.@i = 0; .@i < $@BG_RM_C_COUNT; .@i++)
		if ($@BG_RM_C_AID[.@i] == .@aid) return 1;
	for (.@i = 0; .@i < $@BG_RM_QUEUE_COUNT; .@i++)
		if ($@BG_RM_QUEUE_AID[.@i] == .@aid) return 1;
	.@pid = getcharid(1);
	if (.@pid && ((.@pid == $BG_GUILLAUME_PARTY_ID) || (.@pid == $BG_CROIX_PARTY_ID)))
		return 1;
	if (need_bg2_pcstate(0))
		return 1;
	return 0;
}

// 첨부된 플레이어의 참가 조건 검사. 0 = 통과, 그 밖에는 F_NEED_BG2_Reason$ 코드.
// getarg(0) = 1 이면 공개 대기열: GM 테스트 옵션(레벨 우회)을 적용하지 않는다.
// 사냥 중 이동/스킬 사용은 거절 사유가 아니다. 신청 때와 입장 직전에 같은 검사를 다시 한다.
function	script	F_NEED_BG2_CheckSelf	{
	if (Hp < 1)
		return 2;
	if (getcharid(4))
		return 3;
	if (getmapxy(.@m$, .@x, .@y) != 0)
		return 1;
	if (!callfunc("F_NEED_BG2_ReturnOk", .@m$, .@x, .@y) || getmapflag(.@m$, mf_nowarp))
		return 4;
	if ((getarg(0, 0) || !$NEED_BG2_LVBYPASS) && BaseLevel < $@NBG2_C_MINLV)
		return 5;
	if (checkvending())
		return 6;
	if (need_bg2_pcstate(1))
		return 12;
	if (callfunc("F_NEED_BG2_InRandomMatch", getcharid(3)))
		return 8;
	return 0;
}

function	script	F_NEED_BG2_IndexOf	{
	for (.@i = 0; .@i < $@NBG2_N; .@i++)
		if ($@NBG2_CID[.@i] == getarg(0)) return .@i;
	return -1;
}

function	script	F_NEED_BG2_AliveCount	{
	.@c = 0;
	for (.@i = 0; .@i < $@NBG2_N; .@i++)
		if ($@NBG2_TEAM[.@i] == getarg(0) && $@NBG2_ALIVE[.@i] && !$@NBG2_GONE[.@i]) .@c++;
	return .@c;
}

function	script	F_NEED_BG2_InBox	{
	return (getarg(0) >= 37 && getarg(0) <= 42 && getarg(1) >= 37 && getarg(1) <= 42);
}

function	script	F_NEED_BG2_Reattach	{
	if (getarg(0))
		attachrid getarg(0);
	else
		detachrid;
	return;
}

// 첨부된 플레이어의 행동 제한을 mode 로 맞춘다. 이 시스템이 걸지 않은 기존 제한 비트
// (@NEED_BG2_PB: 매크로 탐지 등)는 건드리지 않는다.
function	script	F_NEED_BG2_SetMode	{
	.@mine = $@NBG2_C_OURS & ~@NEED_BG2_PB;
	setpcblock .@mine, 0;
	if (getarg(0) & .@mine)
		setpcblock getarg(0) & .@mine, 1;
	return;
}

// 경기장에 있는 참가자 전원의 행동 제한을 mode 로 맞춘다.
function	script	F_NEED_BG2_ModeAll	{
	.@caller = playerattached();
	for (.@i = 0; .@i < $@NBG2_N; .@i++) {
		if ($@NBG2_GONE[.@i] || !isloggedin($@NBG2_AID[.@i], $@NBG2_CID[.@i]))
			continue;
		if (!attachrid($@NBG2_AID[.@i]))
			continue;
		if (getcharid(0) == $@NBG2_CID[.@i] && strcharinfo(3) == $@NBG2_MAP$)
			callfunc "F_NEED_BG2_SetMode", getarg(0);
		detachrid;
	}
	callfunc "F_NEED_BG2_Reattach", .@caller;
	return;
}

// 첨부된 참가자를 라운드 시작 상태로: 부활/HP·SP 100%/전투 버프·디버프 정리/자동 부활 효과 제거/
// 설치물 제거/동반 유닛 회복/시작점/준비 제한. 음식·포션·캐시·직업 상태·쿨타임은 유지한다.
function	script	F_NEED_BG2_ResetSelf	{
	.@cid = getcharid(0);
	.@p = (getarg(0) - 1) * 2 + getarg(1);
	recovery 0, .@cid, 1;
	freeloop(1);
	for (.@k = 0; .@k < getarraysize($@NBG2_SC_CLEAR); .@k++)
		if (getstatus($@NBG2_SC_CLEAR[.@k]))
			sc_end $@NBG2_SC_CLEAR[.@k];
	freeloop(0);
	callfunc "F_NEED_BG2_EndNoRevive";
	need_bg2_clearunits .@cid;
	callfunc "F_NEED_BG2_Companions";
	if (getarg(2, 1))
		warp $@NBG2_MAP$, $@NBG2_SPX[.@p], $@NBG2_SPY[.@p];
	callfunc "F_NEED_BG2_SetMode", $@NBG2_C_FREEZE;
	return;
}

// 첨부된 플레이어의 자동 부활 상태를 끝낸다(엔진도 경기장에서는 새로 걸리지 않게 막는다).
function	script	F_NEED_BG2_EndNoRevive	{
	for (.@k = 0; .@k < getarraysize($@NBG2_SC_NOREV); .@k++)
		if (getstatus($@NBG2_SC_NOREV[.@k]))
			sc_end $@NBG2_SC_NOREV[.@k];
	return;
}

// 첨부된 (전) 참가자를 정리하고 저장 지점으로 보낸다. 여러 번 불려도 결과가 같다.
// getarg(1) = 1 이면 탈주 제재 없음(무효/취소/시스템 이동): 엔진이 건 탈주자 상태를 지운다.
function	script	F_NEED_BG2_RestoreSelf	{
	.@cid = getcharid(0);
	.@on = (strcharinfo(3) == $@NBG2_MAP$);
	if (.@on || Hp < 1)
		recovery 0, .@cid, 1;
	.@mine = $@NBG2_C_OURS & ~@NEED_BG2_PB;
	setpcblock .@mine, 0;
	.@bg = getcharid(4);
	if (.@bg && (.@bg == @NEED_BG2_BGID || .@bg == $@NBG2_BG[1] || .@bg == $@NBG2_BG[2]))
		bg_leave;	// 엔진(feature.bgqueue)이 이미 저장 지점으로 보내면 아래 이동은 생략된다.
	.@to$ = "stay";
	if (strcharinfo(3) == $@NBG2_MAP$) {
		.@to$ = "SavePoint";
		warp "SavePoint", 0, 0;
	} else if (.@on)
		.@to$ = "SavePoint(bg_leave)";
	else if (getarg(2, 0) && NEED_BG2_MID) {
		// 경기장 밖으로 밀려나 있던 참가자(시스템/GM 이동)도 정책대로 저장 지점으로.
		.@to$ = "SavePoint(off_arena)";
		warp "SavePoint", 0, 0;
	}
	if (getarg(1, 0) && getstatus(SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT)) {
		sc_end SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT;
		.@to$ += " deserter_cleared";
	}
	callfunc "F_NEED_BG2_Log", "RESTORE match=" + NEED_BG2_MID + " aid=" + getcharid(3) + " cid=" + .@cid + " why=" + getarg(0) + " to=" + .@to$;
	NEED_BG2_MID = 0;
	NEED_BG2_RMAP$ = "";
	NEED_BG2_RX = 0;
	NEED_BG2_RY = 0;
	@NEED_BG2_BGID = 0;
	@NEED_BG2_PB = 0;
	@NEED_BG2_BUSY = 0;
	viewpoint 2, 39, 39, 1, 0xFFCC00;
	return;
}

// GM 이 첨부된 상태에서 호출(GM 수동 4인 테스트). 성공 1, 거절 0. 거절 시 모집 상태와 명단은 그대로다.
function	script	F_NEED_BG2_Start	{
	.@gm = playerattached();
	if ($@NBG2_STATE != 1) {
		dispbottom "[2:2 듀오] 모집 중이 아니어서 시작할 수 없습니다. (상태 " + callfunc("F_NEED_BG2_StateName$", $@NBG2_STATE) + ")";
		return 0;
	}
	if ($@NBG2_REG_N != 4) {
		dispbottom "[2:2 듀오] 정확히 4명이 필요합니다. 현재 " + $@NBG2_REG_N + "명";
		callfunc "F_NEED_BG2_Log", "START_REFUSED reason=count n=" + $@NBG2_REG_N;
		return 0;
	}
	.@bad = 0;
	for (.@i = 0; .@i < 4; .@i++) {
		.@code = 0;
		if (!isloggedin($@NBG2_REG_AID[.@i], $@NBG2_REG_CID[.@i]) || !attachrid($@NBG2_REG_AID[.@i]))
			.@code = 1;
		else if (getcharid(0) != $@NBG2_REG_CID[.@i])
			.@code = 1;
		else {
			.@code = callfunc("F_NEED_BG2_CheckSelf", 0);
			$@NBG2_CAND_CLS[.@i] = Class;
		}
		detachrid;
		if (.@code) {
			.@bad = .@i + 1;
			.@badcode = .@code;
			break;
		}
	}
	callfunc "F_NEED_BG2_Reattach", .@gm;
	if (.@bad) {
		dispbottom "[2:2 듀오] 시작 거절: 신청 #" + .@bad + " (cid " + $@NBG2_REG_CID[.@bad - 1] + ") - " + callfunc("F_NEED_BG2_Reason$", .@badcode);
		callfunc "F_NEED_BG2_Log", "START_REFUSED reason=" + .@badcode + " slot=" + .@bad + " cid=" + $@NBG2_REG_CID[.@bad - 1];
		return 0;
	}
	for (.@i = 0; .@i < 4; .@i++) {
		$@NBG2_CAND_AID[.@i] = $@NBG2_REG_AID[.@i];
		$@NBG2_CAND_CID[.@i] = $@NBG2_REG_CID[.@i];
		$@NBG2_CAND_SEQ[.@i] = 0;
	}
	.@ok = callfunc("F_NEED_BG2_Launch", 1);
	callfunc "F_NEED_BG2_Reattach", .@gm;
	if (.@ok)
		dispbottom "[2:2 듀오] 매치 " + $@NBG2_MATCH + " 시작 (준비 45초).";
	else
		dispbottom "[2:2 듀오] 시작 실패: 이동/팀 합류 실패. 경기장에 들어간 사람은 저장 지점으로 보냈고 모집은 유지됩니다.";
	return .@ok;
}

// 직업 지원 점수: 아크비숍 계열 2, 소서러/민스트럴/원더러 계열 1, 그 외 0.
// eaclass & EAJ_THIRDMASK 로 보므로 전생·베이비·4차(카디널/엘리멘탈 마스터/트루바두르/트루베르)도 같은 계열이다.
function	script	F_NEED_BG2_Support	{
	.@m = eaclass(getarg(0)) & EAJ_THIRDMASK;
	if (.@m == EAJ_ARCH_BISHOP)
		return 2;
	if (.@m == EAJ_SORCERER || .@m == EAJ_MINSTRELWANDERER)
		return 1;
	return 0;
}

// 확정된 4명($@NBG2_CAND_*) 안에서만 2:2 를 나눈다. 3가지 분할 중
// (1) 지원 점수 합 차이 최소 (2) 같은 직업 2명이 한 팀에 몰리는 수 최소 (3) 그래도 같으면 무작위.
// 팀 색도 무작위로 정해 한쪽으로 치우치지 않게 한다. 결과: $@NBG2_ORD[0..1] 블루, [2..3] 레드 (CAND 인덱스).
// MMR/승률/레벨/장비/길드/IP/파티는 보지 않는다.
function	script	F_NEED_BG2_Balance	{
	for (.@i = 0; .@i < 4; .@i++) {
		.@s[.@i] = callfunc("F_NEED_BG2_Support", $@NBG2_CAND_CLS[.@i]);
		.@j[.@i] = eaclass($@NBG2_CAND_CLS[.@i]) & EAJ_FOURTHMASK;	// 같은 직업 판정(전생/베이비 구분 없음)
	}
	setarray .@a1[0], 0, 0, 0;
	setarray .@a2[0], 1, 2, 3;
	setarray .@b1[0], 2, 1, 1;
	setarray .@b2[0], 3, 3, 2;
	.@best = -1;
	for (.@p = 0; .@p < 3; .@p++) {
		.@d = (.@s[.@a1[.@p]] + .@s[.@a2[.@p]]) - (.@s[.@b1[.@p]] + .@s[.@b2[.@p]]);
		if (.@d < 0)
			.@d = -.@d;
		.@c = 0;
		if (.@j[.@a1[.@p]] == .@j[.@a2[.@p]])
			.@c++;
		if (.@j[.@b1[.@p]] == .@j[.@b2[.@p]])
			.@c++;
		.@key[.@p] = .@d * 10 + .@c;
		if (.@best < 0 || .@key[.@p] < .@best)
			.@best = .@key[.@p];
	}
	.@n = 0;
	for (.@p = 0; .@p < 3; .@p++) {
		if (.@key[.@p] != .@best)
			continue;
		.@pick[.@n] = .@p;
		.@n++;
	}
	.@p = .@pick[0];
	if (.@n > 1)	// rand(1) 은 오류다(범위 0). 후보가 하나면 그대로 쓴다.
		.@p = .@pick[rand(.@n)];
	.@swap = rand(2);
	if (.@swap) {
		setarray .@t1[0], .@b1[.@p], .@b2[.@p];
		setarray .@t2[0], .@a1[.@p], .@a2[.@p];
	} else {
		setarray .@t1[0], .@a1[.@p], .@a2[.@p];
		setarray .@t2[0], .@b1[.@p], .@b2[.@p];
	}
	.@r = rand(2);
	$@NBG2_ORD[0] = .@t1[.@r];
	$@NBG2_ORD[1] = .@t1[1 - .@r];
	.@r = rand(2);
	$@NBG2_ORD[2] = .@t2[.@r];
	$@NBG2_ORD[3] = .@t2[1 - .@r];
	callfunc "F_NEED_BG2_Log", "TEAM_ASSIGN classes=" + $@NBG2_CAND_CLS[0] + "/" + $@NBG2_CAND_CLS[1] + "/" + $@NBG2_CAND_CLS[2] + "/" + $@NBG2_CAND_CLS[3]
		+ " support=" + .@s[0] + .@s[1] + .@s[2] + .@s[3] + " keys=" + .@key[0] + "," + .@key[1] + "," + .@key[2]
		+ " tied=" + .@n + " split=" + .@p + " swap=" + .@swap
		+ " blue_sum=" + (.@s[$@NBG2_ORD[0]] + .@s[$@NBG2_ORD[1]]) + " red_sum=" + (.@s[$@NBG2_ORD[2]] + .@s[$@NBG2_ORD[3]]);
	return;
}

// 확정된 4명($@NBG2_CAND_AID/CID/CLS)으로 경기를 연다. origin: 1 GM 수동 / 2 공개 대기열.
// 실패하면 경기장에 들어간 사람을 저장 지점으로 돌리고 0 (GM: 모집 명단 유지, 대기열: 호출한 쪽이 순번 복원).
function	script	F_NEED_BG2_Launch	{
	.@caller = playerattached();
	.@origin = getarg(0);
	callfunc "F_NEED_BG2_Balance";
	callfunc "F_NEED_BG2_ClearMatch";
	$@NBG2_ORIGIN = .@origin;
	// 매치 ID = 시작 시각(유닉스 초). 같은 초에 다음 경기가 열려도(빠른 연속 경기) 겹치지 않게 직전 값보다 크게 한다.
	// $NEED_BG2_SEQ 는 주기 저장이라 비정상 종료 뒤 같은 값이 다시 나올 수 있어 참고용 순번으로만 쓴다.
	$NEED_BG2_SEQ++;
	.@mid = gettimetick(2);
	if (.@mid <= $@NBG2_LASTMID)
		.@mid = $@NBG2_LASTMID + 1;
	if (.@mid <= $NEED_BG2_LASTMID)
		.@mid = $NEED_BG2_LASTMID + 1;
	$@NBG2_LASTMID = .@mid;
	$NEED_BG2_LASTMID = .@mid;
	$@NBG2_MATCH = .@mid;
	$@NBG2_N = 4;
	.@reglog$ = "";
	for (.@i = 0; .@i < 4; .@i++)
		.@reglog$ += (.@i ? "," : "") + $@NBG2_CAND_AID[.@i] + "/" + $@NBG2_CAND_CID[.@i];
	for (.@k = 0; .@k < 4; .@k++) {
		.@r = $@NBG2_ORD[.@k];
		$@NBG2_AID[.@k] = $@NBG2_CAND_AID[.@r];
		$@NBG2_CID[.@k] = $@NBG2_CAND_CID[.@r];
		$@NBG2_TEAM[.@k] = (.@k < 2) ? 1 : 2;
		$@NBG2_SLOT[.@k] = .@k % 2;
	}
	callfunc "F_NEED_BG2_Log", "MATCH_START match=" + $@NBG2_MATCH + " seq=" + $NEED_BG2_SEQ + " origin=" + (.@origin == 2 ? "queue" : "gm") + " reg=" + .@reglog$
		+ " order=" + $@NBG2_ORD[0] + $@NBG2_ORD[1] + $@NBG2_ORD[2] + $@NBG2_ORD[3]
		+ " blue=" + $@NBG2_AID[0] + "/" + $@NBG2_CID[0] + "," + $@NBG2_AID[1] + "/" + $@NBG2_CID[1]
		+ " red=" + $@NBG2_AID[2] + "/" + $@NBG2_CID[2] + "," + $@NBG2_AID[3] + "/" + $@NBG2_CID[3]
		+ " lvbypass=" + (.@origin == 1 ? $NEED_BG2_LVBYPASS : 0);

	$@NBG2_BG[1] = bg_create($@NBG2_MAP$, $@NBG2_BOXX[1], $@NBG2_BOXY[1], "NEED_BG2::OnQuitB", "NEED_BG2::OnDieB");
	$@NBG2_BG[2] = bg_create($@NBG2_MAP$, $@NBG2_BOXX[2], $@NBG2_BOXY[2], "NEED_BG2::OnQuitR", "NEED_BG2::OnDieR");
	$@NBG2_STATE = 2;
	$@NBG2_DEADLINE = gettimetick(0) + $@NBG2_C_PREP;
	.@fail = 0;
	if (!$@NBG2_BG[1] || !$@NBG2_BG[2])
		.@fail = -1;
	for (.@k = 0; .@k < 4 && !.@fail; .@k++) {
		.@t = $@NBG2_TEAM[.@k];
		.@p = (.@t - 1) * 2 + $@NBG2_SLOT[.@k];
		if (!attachrid($@NBG2_AID[.@k])) {
			.@fail = .@k + 1;
			break;
		}
		NEED_BG2_MID = $@NBG2_MATCH;	// 재접속 시 복귀/재합류 판단용. 복귀 지점은 항상 저장 지점이다.
		@NEED_BG2_PB = getpcblock() & $@NBG2_C_OURS;
		@NEED_BG2_BGID = $@NBG2_BG[.@t];
		@NEED_BG2_BUSY = 3;
		detachrid;
		if (!bg_join($@NBG2_BG[.@t], $@NBG2_MAP$, $@NBG2_SPX[.@p], $@NBG2_SPY[.@p], $@NBG2_CID[.@k]))
			.@fail = .@k + 1;
	}
	if (.@fail) {
		callfunc "F_NEED_BG2_Log", "START_FAILED match=" + $@NBG2_MATCH + " at=" + .@fail + " origin=" + .@origin;
		$@NBG2_STATE = 6;
		$@NBG2_RESULT = 4;
		callfunc "F_NEED_BG2_Cleanup", "start_failed", (.@origin == 1), 0;
		callfunc "F_NEED_BG2_Reattach", .@caller;
		return 0;
	}
	if (.@origin == 1)
		callfunc "F_NEED_BG2_ClearRecruit";
	for (.@k = 0; .@k < 4; .@k++) {
		if (!attachrid($@NBG2_AID[.@k]))
			continue;
		@NEED_BG2_BUSY = 3;
		callfunc "F_NEED_BG2_ResetSelf", $@NBG2_TEAM[.@k], $@NBG2_SLOT[.@k], 0;
		dispbottom needtr("[2:2 듀오] 당신은 " + callfunc("F_NEED_BG2_TeamName$", $@NBG2_TEAM[.@k]) + " 팀입니다. 45초 준비 후 1라운드가 시작됩니다. (테스트 경기, 보상 없음)",
			"[2v2 Duo] You are on the " + callfunc("F_NEED_BG2_TeamName$", $@NBG2_TEAM[.@k]) + " team. Round 1 starts after 45 seconds of preparation. (Test match, no rewards)"), 0x66CCFF;
		detachrid;
	}
	callfunc "F_NEED_BG2_PrepLock", 1;
	bg_updatescore $@NBG2_MAP$, 0, 0;
	mapannouncelang $@NBG2_MAP$, "[2:2 듀오] 매치 " + $@NBG2_MATCH + " 준비 45초. 자기 팀 준비구역 안에서만 움직일 수 있고, 공격과 지면 설치·소환 스킬은 막히며 피해를 받지 않습니다. 라운드 시작 60초 뒤 중앙 거점이 열립니다.",
		"[2v2 Duo] Match " + $@NBG2_MATCH + ": 45 seconds of preparation. Move inside your team's zone only; attacks and ground/summon skills are blocked and you take no damage. The centre point opens 60 seconds into each round.", bc_map, "0x66CCFF";
	callfunc "F_NEED_BG2_Reattach", .@caller;
	return 1;
}

function	script	F_NEED_BG2_AnyAway	{
	for (.@i = 0; .@i < $@NBG2_N; .@i++)
		if ($@NBG2_AWAY[.@i]) return 1;
	return 0;
}

function	script	F_NEED_BG2_BeginRound	{
	if ($@NBG2_STATE != 2 && $@NBG2_STATE != 3)
		return 0;
	// 재접속 유예 중인 참가자가 있으면 시작만 보류한다(유예가 끝나면 이탈로 처리).
	if (callfunc("F_NEED_BG2_AnyAway")) {
		if (!$@NBG2_HOLD_LOGGED) {
			$@NBG2_HOLD_LOGGED = 1;
			callfunc "F_NEED_BG2_Log", "START_HELD match=" + $@NBG2_MATCH + " next_round=" + ($@NBG2_ROUND + 1) + " reason=reconnect_grace";
			mapannouncelang $@NBG2_MAP$, "[2:2 듀오] 접속이 끊긴 참가자를 기다리는 중입니다. 라운드 시작을 보류합니다.",
				"[2v2 Duo] Waiting for a disconnected player. The round start is on hold.", bc_map, "0xFF7777";
		}
		return 0;
	}
	$@NBG2_HOLD_LOGGED = 0;
	callfunc "F_NEED_BG2_PrepLock", 0;
	$@NBG2_ROUND++;
	$@NBG2_STATE = 4;
	$@NBG2_JUDGE_AT = 0;
	$@NBG2_ROUND_AT = gettimetick(0);
	$@NBG2_DEADLINE = $@NBG2_ROUND_AT + $@NBG2_C_ROUND;
	callfunc "F_NEED_BG2_CapReset";
	for (.@i = 0; .@i < $@NBG2_N; .@i++) {
		$@NBG2_ALIVE[.@i] = !$@NBG2_GONE[.@i];
		if ($@NBG2_GONE[.@i] || !attachrid($@NBG2_AID[.@i]))
			continue;
		if (getcharid(0) == $@NBG2_CID[.@i] && strcharinfo(3) == $@NBG2_MAP$) {
			// 준비 시간에 쓰러졌거나 관전석에 있으면 시작점으로 되돌린다.
			getmapxy(.@m$, .@x, .@y);
			if (Hp < 1 || callfunc("F_NEED_BG2_InBox", .@x, .@y))
				callfunc "F_NEED_BG2_ResetSelf", $@NBG2_TEAM[.@i], $@NBG2_SLOT[.@i], 1;
			callfunc "F_NEED_BG2_SetMode", 0;
		}
		detachrid;
	}
	callfunc "F_NEED_BG2_Log", "ROUND_START match=" + $@NBG2_MATCH + " round=" + $@NBG2_ROUND + " score=" + $@NBG2_WIN_B + ":" + $@NBG2_WIN_R;
	mapannouncelang $@NBG2_MAP$, "[2:2 듀오] " + $@NBG2_ROUND + "라운드 시작! (90초, 60초 뒤 중앙 거점 개방, 블루 " + $@NBG2_WIN_B + " : " + $@NBG2_WIN_R + " 레드)",
		"[2v2 Duo] Round " + $@NBG2_ROUND + " - fight! (90 seconds, centre point opens at 60 s, Blue " + $@NBG2_WIN_B + " : " + $@NBG2_WIN_R + " Red)", bc_map, "0xFFCC33";
	return 1;
}

// 라운드 결과 확정. res: 0 무승부 / 1 블루 / 2 레드. 한 라운드에 한 번만 통과한다.
function	script	F_NEED_BG2_EndRound	{
	.@res = getarg(0);
	if ($@NBG2_STATE != 4) {
		callfunc "F_NEED_BG2_Log", "ROUND_END_IGNORED match=" + $@NBG2_MATCH + " state=" + $@NBG2_STATE + " res=" + .@res + " why=" + getarg(1);
		return 0;
	}
	$@NBG2_STATE = 5;
	$@NBG2_JUDGE_AT = 0;
	if (.@res == 1) $@NBG2_WIN_B++;
	else if (.@res == 2) $@NBG2_WIN_R++;
	else $@NBG2_DRAWS++;
	callfunc "F_NEED_BG2_Log", "ROUND_END match=" + $@NBG2_MATCH + " round=" + $@NBG2_ROUND + " res=" + .@res + " why=" + getarg(1)
		+ " alive=" + callfunc("F_NEED_BG2_AliveCount", 1) + ":" + callfunc("F_NEED_BG2_AliveCount", 2) + " score=" + $@NBG2_WIN_B + ":" + $@NBG2_WIN_R + " draws=" + $@NBG2_DRAWS;
	bg_updatescore $@NBG2_MAP$, $@NBG2_WIN_B, $@NBG2_WIN_R;
	if (.@res == 1) {
		.@kr$ = "블루 팀 승리";
		.@en$ = "Blue team wins";
	} else if (.@res == 2) {
		.@kr$ = "레드 팀 승리";
		.@en$ = "Red team wins";
	} else {
		.@kr$ = (getarg(1) == "timeout") ? "시간 종료 무승부" : "동시 전멸 무승부";
		.@en$ = (getarg(1) == "timeout") ? "draw (time up)" : "draw (both teams fell)";
	}
	if (getarg(1) == "capture") {
		.@kr$ += " (중앙 거점 점령)";
		.@en$ += " (centre point captured)";
	} else if (getarg(1) == "timeout_alive") {
		.@kr$ += " (시간 종료 - 생존 인원 우세)";
		.@en$ += " (time up - more players alive)";
	} else if (getarg(1) == "timeout_capture") {
		.@kr$ += " (시간 종료 - 중앙 거점 점유 시간 우세)";
		.@en$ += " (time up - held the centre point longer)";
	}
	if ($@NBG2_CAP_ON)
		callfunc "F_NEED_BG2_CapMarker", 0;
	mapannouncelang $@NBG2_MAP$, "[2:2 듀오] " + $@NBG2_ROUND + "라운드 " + .@kr$ + ". 점수 블루 " + $@NBG2_WIN_B + " : " + $@NBG2_WIN_R + " 레드",
		"[2v2 Duo] Round " + $@NBG2_ROUND + ": " + .@en$ + ". Score Blue " + $@NBG2_WIN_B + " : " + $@NBG2_WIN_R + " Red", bc_map, "0xFFCC33";

	if ($@NBG2_WIN_B >= $@NBG2_C_WINS)
		callfunc "F_NEED_BG2_Finish", 1, "wins";
	else if ($@NBG2_WIN_R >= $@NBG2_C_WINS)
		callfunc "F_NEED_BG2_Finish", 2, "wins";
	else if ($@NBG2_ROUND >= $@NBG2_C_MAXROUND)
		callfunc "F_NEED_BG2_Finish", 0, "max_rounds";
	else
		callfunc "F_NEED_BG2_ResetRound";
	return 1;
}

function	script	F_NEED_BG2_ResetRound	{
	if ($@NBG2_STATE != 5)
		return 0;
	$@NBG2_STATE = 3;
	$@NBG2_DEADLINE = gettimetick(0) + $@NBG2_C_READY;
	.@caller = playerattached();
	callfunc "F_NEED_BG2_KillSummons";
	for (.@i = 0; .@i < $@NBG2_N; .@i++) {
		if ($@NBG2_GONE[.@i] || $@NBG2_AWAY[.@i] || !isloggedin($@NBG2_AID[.@i], $@NBG2_CID[.@i]) || !attachrid($@NBG2_AID[.@i]))
			continue;
		if (getcharid(0) == $@NBG2_CID[.@i] && strcharinfo(3) == $@NBG2_MAP$)
			callfunc "F_NEED_BG2_ResetSelf", $@NBG2_TEAM[.@i], $@NBG2_SLOT[.@i], 1;
		detachrid;
	}
	callfunc "F_NEED_BG2_PrepLock", 1;
	callfunc "F_NEED_BG2_Reattach", .@caller;
	.@next = $@NBG2_ROUND + 1;
	.@extra$ = (.@next > 3) ? " (연장)" : "";
	.@extra_en$ = (.@next > 3) ? " (extra round)" : "";
	callfunc "F_NEED_BG2_Log", "ROUND_READY match=" + $@NBG2_MATCH + " next=" + .@next;
	mapannouncelang $@NBG2_MAP$, "[2:2 듀오] 15초 후 " + .@next + "라운드" + .@extra$ + " 시작. 전원 부활/회복 후 시작점으로 이동했습니다.",
		"[2v2 Duo] Round " + .@next + .@extra_en$ + " starts in 15 seconds. Everyone was revived, healed and moved to the start point.", bc_map, "0x66CCFF";
	return 1;
}

// 매치 결과 확정. kind: 0 무승부 / 1 블루 승 / 2 레드 승 / 3 무효(VOID) / 4 취소(CANCELLED).
function	script	F_NEED_BG2_Finish	{
	.@kind = getarg(0);
	if ($@NBG2_STATE < 2 || $@NBG2_STATE > 5) {
		callfunc "F_NEED_BG2_Log", "FINISH_IGNORED match=" + $@NBG2_MATCH + " state=" + $@NBG2_STATE + " kind=" + .@kind + " why=" + getarg(1);
		return 0;
	}
	$@NBG2_STATE = 6;
	$@NBG2_JUDGE_AT = 0;
	$@NBG2_DEPART_AT = 0;
	$@NBG2_RESULT = .@kind;
	$@NBG2_DEADLINE = gettimetick(0) + $@NBG2_C_SHOW;
	if (.@kind == 3 || .@kind == 4)
		callfunc "F_NEED_BG2_RememberNoPenalty";	// 무효/취소: 전원 0점, 유효 경기 아님, 탈주 제재 없음
	setarray .@name$[0], "DRAW", "BLUE_WIN", "RED_WIN", "VOID", "CANCELLED";
	setarray .@kr$[0], "무승부", "블루 팀 승리", "레드 팀 승리", "무효 경기", "경기 취소";
	setarray .@en$[0], "draw", "Blue team wins", "Red team wins", "void match", "match cancelled";
	.@gone$ = "";
	for (.@i = 0; .@i < $@NBG2_N; .@i++)
		if ($@NBG2_GONE[.@i])
			.@gone$ += (.@gone$ == "" ? "" : ",") + $@NBG2_CID[.@i] + ":" + $@NBG2_GONE[.@i];
	$@NBG2_LAST$ = "match " + $@NBG2_MATCH + " " + .@name$[.@kind] + " (" + getarg(1) + ") B" + $@NBG2_WIN_B + ":R" + $@NBG2_WIN_R
		+ " draws " + $@NBG2_DRAWS + " rounds " + $@NBG2_ROUND + (.@gone$ == "" ? "" : " gone " + .@gone$);
	callfunc "F_NEED_BG2_Log", "MATCH_END match=" + $@NBG2_MATCH + " result=" + .@name$[.@kind] + " why=" + getarg(1)
		+ " score=" + $@NBG2_WIN_B + ":" + $@NBG2_WIN_R + " draws=" + $@NBG2_DRAWS + " rounds=" + $@NBG2_ROUND + " gone=" + .@gone$ + " reward=none";
	mapannouncelang $@NBG2_MAP$, "[2:2 듀오] 매치 종료: " + .@kr$[.@kind] + " (블루 " + $@NBG2_WIN_B + " : " + $@NBG2_WIN_R + " 레드). 테스트 경기라 보상은 없습니다. 5초 후 저장 지점으로 돌아갑니다.",
		"[2v2 Duo] Match over: " + .@en$[.@kind] + " (Blue " + $@NBG2_WIN_B + " : " + $@NBG2_WIN_R + " Red). Test match, no rewards. You return to your save point in 5 seconds.", bc_map, "0x99FF99";
	callfunc "F_NEED_BG2_ModeAll", $@NBG2_C_LOCK;
	return 1;
}

// 매치 정리. keep_recruit = 1 이면 모집 명단을 남기고 RECRUITING 으로 돌아간다(시작 실패).
// force_save = 1 이면 경기장 밖으로 밀려난(시스템/GM 이동) 접속 중 참가자도 저장 지점으로 보낸다.
function	script	F_NEED_BG2_Cleanup	{
	.@caller = playerattached();
	if ($@NBG2_STATE != 6) {
		callfunc "F_NEED_BG2_Log", "CLEANUP_IGNORED state=" + $@NBG2_STATE + " why=" + getarg(0);
		return 0;
	}
	$@NBG2_STATE = 7;
	callfunc "F_NEED_BG2_PrepLock", 0;
	callfunc "F_NEED_BG2_KillSummons";
	.@nopen = ($@NBG2_RESULT == 3 || $@NBG2_RESULT == 4);
	for (.@i = 0; .@i < $@NBG2_N; .@i++) {
		.@aid = $@NBG2_AID[.@i];
		.@cid = $@NBG2_CID[.@i];
		if ($@NBG2_GONE[.@i] == 1 || !isloggedin(.@aid, .@cid) || !attachrid(.@aid)) {
			callfunc "F_NEED_BG2_Log", "RESTORE_DEFERRED match=" + $@NBG2_MATCH + " cid=" + .@cid + " (restored at next login)";
			continue;
		}
		if (getcharid(0) == .@cid)
			callfunc "F_NEED_BG2_RestoreSelf", getarg(0), .@nopen, getarg(2, 1);
		detachrid;
	}
	if ($@NBG2_BG[1]) bg_destroy $@NBG2_BG[1];
	if ($@NBG2_BG[2]) bg_destroy $@NBG2_BG[2];
	bg_updatescore $@NBG2_MAP$, 0, 0;
	callfunc "F_NEED_BG2_Log", "CLEANUP_DONE match=" + $@NBG2_MATCH + " why=" + getarg(0);
	callfunc "F_NEED_BG2_ClearMatch";
	if (getarg(1, 0)) {
		$@NBG2_STATE = 1;
		for (.@i = 0; .@i < $@NBG2_REG_N; .@i++)
			callfunc "F_NEED_BG2_SetBusy", $@NBG2_REG_AID[.@i], $@NBG2_REG_CID[.@i], 1;
	} else {
		callfunc "F_NEED_BG2_ClearRecruit";
		$@NBG2_STATE = 0;
		if (!$@NBG2_Q_N)
			stopnpctimer "NEED_BG2";
	}
	callfunc "F_NEED_BG2_Reattach", .@caller;
	return 1;
}

// 참가자 이탈. code: 1 접속 종료/캐릭터 선택, 2 맵 이탈, 3 BG 팀 이탈.
// - 접속 종료(1): 경기당 개인 누적 15초 재접속 유예. 유예가 끝나야 이탈로 확정한다.
// - 접속 중인데 경기장/팀을 벗어남(2, 3): 참가자가 스스로 나갈 경로가 없는(텔레포트/귀환/워프 금지)
//   경기장이라 GM·시스템 이동으로 보고 경기 전체를 무효로 한다(전원 0점, 탈주 제재 없음).
function	script	F_NEED_BG2_Depart	{
	.@i = getarg(0);
	.@code = getarg(1);
	.@s = $@NBG2_STATE;
	if ($@NBG2_GONE[.@i] || .@s < 2 || .@s > 6)
		return 0;
	.@now = gettimetick(0);
	if (.@code == 1 && .@s <= 5 && !$@NBG2_AWAY[.@i] && $@NBG2_GRACE_USED[.@i] < $@NBG2_C_RECONNECT) {
		$@NBG2_AWAY[.@i] = 1;
		$@NBG2_AWAY_AT[.@i] = .@now;
		.@left = ($@NBG2_C_RECONNECT - $@NBG2_GRACE_USED[.@i]) / 1000;
		callfunc "F_NEED_BG2_Log", "AWAY match=" + $@NBG2_MATCH + " cid=" + $@NBG2_CID[.@i] + " team=" + $@NBG2_TEAM[.@i]
			+ " state=" + .@s + " round=" + $@NBG2_ROUND + " grace_used=" + $@NBG2_GRACE_USED[.@i] + " at=" + .@now;
		mapannouncelang $@NBG2_MAP$, "[2:2 듀오] 참가자 접속이 끊겼습니다. " + .@left + "초 안에 돌아오면 경기에 복귀합니다.",
			"[2v2 Duo] A player disconnected. If they return within " + .@left + " seconds, they rejoin the match.", bc_map, "0xFF7777";
		if (.@s == 4 && $@NBG2_ALIVE[.@i]) {
			// 라운드 중에는 그 라운드에서 탈락으로 본다(돌아오면 관전석, 다음 라운드부터 정상 참가).
			$@NBG2_ALIVE[.@i] = 0;
			callfunc "F_NEED_BG2_Log", "DEATH match=" + $@NBG2_MATCH + " round=" + $@NBG2_ROUND + " cid=" + $@NBG2_CID[.@i] + " team=" + $@NBG2_TEAM[.@i]
				+ " via=disconnect alive=" + callfunc("F_NEED_BG2_AliveCount", 1) + ":" + callfunc("F_NEED_BG2_AliveCount", 2);
			if (!$@NBG2_JUDGE_AT && (!callfunc("F_NEED_BG2_AliveCount", 1) || !callfunc("F_NEED_BG2_AliveCount", 2)))
				$@NBG2_JUDGE_AT = .@now + $@NBG2_C_SIMUL;
		}
		return 1;
	}
	$@NBG2_GONE[.@i] = .@code;
	$@NBG2_AWAY[.@i] = 0;
	callfunc "F_NEED_BG2_Log", "DEPART match=" + $@NBG2_MATCH + " aid=" + $@NBG2_AID[.@i] + " cid=" + $@NBG2_CID[.@i]
		+ " team=" + $@NBG2_TEAM[.@i] + " code=" + .@code + " state=" + .@s + " round=" + $@NBG2_ROUND + " grace_used=" + $@NBG2_GRACE_USED[.@i];
	if (.@s == 6)
		return 1;
	if (.@code != 1) {
		callfunc "F_NEED_BG2_Log", "ERROR forced_move match=" + $@NBG2_MATCH + " cid=" + $@NBG2_CID[.@i] + " code=" + .@code + " state=" + .@s
			+ " (left the arena while online: GM/system move, not a desertion) -> VOID";
		mapannouncelang $@NBG2_MAP$, "[2:2 듀오] 참가자가 시스템/운영자 이동으로 경기장을 벗어나 경기를 무효로 합니다.",
			"[2v2 Duo] A player was moved out of the arena by the system or a GM, so the match is void.", bc_map, "0xFF7777";
		callfunc "F_NEED_BG2_Finish", 3, "forced_move";
		return 1;
	}
	if (.@s == 2) {
		// 최초 준비시간에 유예가 끝남 = 경기 미개시 취소. 전원 0점, 탈주 제재 없음.
		callfunc "F_NEED_BG2_Finish", 4, "depart_in_preparation";
		callfunc "F_NEED_BG2_Cleanup", "depart_in_preparation", 0;
		return 1;
	}
	if (!$@NBG2_DEPART_AT) {
		$@NBG2_DEPART_AT = gettimetick(0) + $@NBG2_C_GRACE;
		$@NBG2_STATE = 5;
		$@NBG2_JUDGE_AT = 0;
		callfunc "F_NEED_BG2_ModeAll", $@NBG2_C_LOCK;
		mapannouncelang $@NBG2_MAP$, "[2:2 듀오] 참가자가 경기를 이탈했습니다. 판정 중입니다.",
			"[2v2 Duo] A player left the match. Deciding the result.", bc_map, "0xFF7777";
	}
	return 1;
}

function	script	F_NEED_BG2_DepartVerdict	{
	.@gb = 0;
	.@gr = 0;
	for (.@i = 0; .@i < $@NBG2_N; .@i++) {
		if (!$@NBG2_GONE[.@i]) continue;
		if ($@NBG2_TEAM[.@i] == 1) .@gb++;
		else .@gr++;
	}
	if (.@gb && .@gr)
		callfunc "F_NEED_BG2_Finish", 3, "both_teams_departed";
	else if (.@gb)
		callfunc "F_NEED_BG2_Finish", 2, "forfeit_blue";
	else if (.@gr)
		callfunc "F_NEED_BG2_Finish", 1, "forfeit_red";
	else
		callfunc "F_NEED_BG2_Finish", 3, "depart_unknown";
	return;
}

// 사망 이벤트 (첨부 = 사망자). 탈락은 한 라운드에 한 번만 센다.
function	script	F_NEED_BG2_OnDeath	{
	.@cid = getcharid(0);
	.@i = callfunc("F_NEED_BG2_IndexOf", .@cid);
	if (.@i < 0) {
		callfunc "F_NEED_BG2_Log", "DEATH_UNKNOWN cid=" + .@cid;
		return;
	}
	if ($@NBG2_STATE != 4 || $@NBG2_GONE[.@i]) {
		callfunc "F_NEED_BG2_Log", "DEATH_IGNORED match=" + $@NBG2_MATCH + " cid=" + .@cid + " state=" + $@NBG2_STATE;
		return;
	}
	if (!$@NBG2_ALIVE[.@i]) {
		callfunc "F_NEED_BG2_Log", "DEATH_DUP match=" + $@NBG2_MATCH + " round=" + $@NBG2_ROUND + " cid=" + .@cid;
		return;
	}
	// 엔진은 사망자가 NPC 대화 중이면 die 이벤트를 큐에 넣고 대화가 끝난 뒤 실행한다.
	// 그 사이 다음 라운드가 시작됐다면 살아 있는 사람이므로 지난 사망으로 탈락시키지 않는다.
	// (정상 경로는 사망 직후 HP 0 에서, 감시 경로는 HP 0 이거나 관전석에서 불린다.)
	getmapxy(.@m$, .@x, .@y);
	if (Hp > 0 && !callfunc("F_NEED_BG2_InBox", .@x, .@y)) {
		callfunc "F_NEED_BG2_Log", "DEATH_STALE match=" + $@NBG2_MATCH + " round=" + $@NBG2_ROUND + " cid=" + .@cid + " pos=" + .@x + "," + .@y;
		return;
	}
	$@NBG2_ALIVE[.@i] = 0;
	callfunc "F_NEED_BG2_SetMode", $@NBG2_C_SPEC;
	callfunc "F_NEED_BG2_Log", "DEATH match=" + $@NBG2_MATCH + " round=" + $@NBG2_ROUND + " cid=" + .@cid + " team=" + $@NBG2_TEAM[.@i]
		+ " killer=" + killerrid + " tick=" + gettimetick(0) + " via=" + getarg(0, "event") + " alive=" + callfunc("F_NEED_BG2_AliveCount", 1) + ":" + callfunc("F_NEED_BG2_AliveCount", 2);
	dispbottom needtr("[2:2 듀오] 이번 라운드에서 탈락했습니다. 다음 라운드에 다시 참여합니다.", "[2v2 Duo] You are out for this round. You come back next round."), 0xFF7777;
	if (!$@NBG2_JUDGE_AT && (!callfunc("F_NEED_BG2_AliveCount", 1) || !callfunc("F_NEED_BG2_AliveCount", 2))) {
		$@NBG2_JUDGE_AT = gettimetick(0) + $@NBG2_C_SIMUL;
		callfunc "F_NEED_BG2_Log", "WIPE match=" + $@NBG2_MATCH + " round=" + $@NBG2_ROUND + " judge_in=" + $@NBG2_C_SIMUL;
	}
	return;
}

function	script	F_NEED_BG2_Judge	{
	.@b = callfunc("F_NEED_BG2_AliveCount", 1);
	.@r = callfunc("F_NEED_BG2_AliveCount", 2);
	if (!.@b && !.@r) .@res = 0;
	else if (!.@r) .@res = 1;
	else if (!.@b) .@res = 2;
	else {
		$@NBG2_JUDGE_AT = 0;
		return;
	}
	callfunc "F_NEED_BG2_EndRound", .@res, "wipe";
	return;
}

// 감시: 접속/맵/팀 이탈, 탈락자 관전석 유지. 탈락은 감시가 아니라 사망 이벤트로만 정한다.
function	script	F_NEED_BG2_Watch	{
	for (.@i = 0; .@i < $@NBG2_N; .@i++) {
		if ($@NBG2_STATE < 2 || $@NBG2_STATE > 5)
			break;
		if ($@NBG2_GONE[.@i] || $@NBG2_AWAY[.@i])
			continue;
		.@aid = $@NBG2_AID[.@i];
		.@cid = $@NBG2_CID[.@i];
		if (!isloggedin(.@aid, .@cid) || !attachrid(.@aid)) {
			callfunc "F_NEED_BG2_Depart", .@i, 1;
			continue;
		}
		.@code = 0;
		if (getcharid(0) != .@cid)
			.@code = 1;
		else if (strcharinfo(3) != $@NBG2_MAP$)
			.@code = 2;
		else if (getcharid(4) != $@NBG2_BG[$@NBG2_TEAM[.@i]])
			.@code = 3;
		else if (($@NBG2_STATE == 2 || $@NBG2_STATE == 3) && Hp > 0) {
			// 준비구역을 벗어났으면(경계를 넘는 이동 스킬 등) 시작 지점으로 되돌린다.
			getmapxy(.@m$, .@x, .@y);
			if (!callfunc("F_NEED_BG2_InZone", $@NBG2_TEAM[.@i], .@x, .@y)) {
				.@p = ($@NBG2_TEAM[.@i] - 1) * 2 + $@NBG2_SLOT[.@i];
				warp $@NBG2_MAP$, $@NBG2_SPX[.@p], $@NBG2_SPY[.@p];
				callfunc "F_NEED_BG2_Log", "ZONE_RESET match=" + $@NBG2_MATCH + " cid=" + .@cid + " from=" + .@x + "," + .@y;
			}
		}
		else if ($@NBG2_STATE == 4 && $@NBG2_ALIVE[.@i]) {
			// 안전망: 사망 이벤트가 큐에 묶여 아직 안 왔지만 실제로 쓰러졌거나(HP 0)
			// 엔진이 이미 관전석에서 부활시킨 참가자는 여기서 탈락 처리한다.
			getmapxy(.@m$, .@x, .@y);
			if (Hp < 1 || callfunc("F_NEED_BG2_InBox", .@x, .@y))
				callfunc "F_NEED_BG2_OnDeath", "watch";
		}
		else if ($@NBG2_STATE == 4 && !$@NBG2_ALIVE[.@i] && Hp > 0) {
			getmapxy(.@m$, .@x, .@y);
			if (!callfunc("F_NEED_BG2_InBox", .@x, .@y)) {
				// 같은 라운드에 부활 스킬/아이템으로 일어난 탈락자: 관전석으로 보낸다.
				callfunc "F_NEED_BG2_SetMode", $@NBG2_C_SPEC;
				warp $@NBG2_MAP$, $@NBG2_BOXX[$@NBG2_TEAM[.@i]], $@NBG2_BOXY[$@NBG2_TEAM[.@i]];
				callfunc "F_NEED_BG2_Log", "SPECTATOR_MOVED match=" + $@NBG2_MATCH + " round=" + $@NBG2_ROUND + " cid=" + .@cid + " from=" + .@x + "," + .@y;
			}
		}
		detachrid;
		if (.@code)
			callfunc "F_NEED_BG2_Depart", .@i, .@code;
	}
	// (Phase 1 의 "경기 중 같은 파티가 되면 무효" 는 없앴다. need_bg2 는 맵플래그 need_bgpartysplit 으로
	//  다른 BG 팀의 파티원을 파티 스킬/인원 계산/합주 상대에서 제외하므로, 파티 상태와 상관없이 경기를 이어간다.)
	// 탈락자의 호문클루스/용병도 관전석으로 (주인 사망 시 엔진이 지우지 않는 경우).
	if ($@NBG2_STATE == 4) {
		setarray .@types[0], BL_HOM, BL_MER;
		setarray .@mx[0], UHOM_MASTERCID, UMER_MASTERCID;
		setarray .@ux[0], UHOM_X, UMER_X;
		setarray .@uy[0], UHOM_Y, UMER_Y;
		for (.@t = 0; .@t < 2; .@t++) {
			.@n = getmapunits(.@types[.@t], $@NBG2_MAP$, .@g);
			for (.@k = 0; .@k < .@n; .@k++) {
				deletearray .@d;
				getunitdata .@g[.@k], .@d;
				.@j = callfunc("F_NEED_BG2_IndexOf", .@d[.@mx[.@t]]);
				if (.@j < 0)	// || 도 단락 평가가 아니므로 음수 인덱스는 따로 거른다
					continue;
				if ($@NBG2_ALIVE[.@j] || callfunc("F_NEED_BG2_InBox", .@d[.@ux[.@t]], .@d[.@uy[.@t]]))
					continue;
				unitwarp .@g[.@k], $@NBG2_MAP$, $@NBG2_BOXX[$@NBG2_TEAM[.@j]], $@NBG2_BOXY[$@NBG2_TEAM[.@j]];
				callfunc "F_NEED_BG2_Log", "SUMMON_MOVED match=" + $@NBG2_MATCH + " owner=" + .@d[.@mx[.@t]] + " gid=" + .@g[.@k];
			}
		}
	}
	return;
}

// 모집 중 접속 종료/조건 이탈자 자동 제외.
function	script	F_NEED_BG2_RecruitWatch	{
	for (.@i = $@NBG2_REG_N - 1; .@i >= 0; .@i--) {
		if (isloggedin($@NBG2_REG_AID[.@i], $@NBG2_REG_CID[.@i]))
			continue;
		callfunc "F_NEED_BG2_Log", "UNREGISTER cid=" + $@NBG2_REG_CID[.@i] + " why=offline";
		callfunc "F_NEED_BG2_RemoveReg", .@i;
	}
	return;
}

function	script	F_NEED_BG2_Tick	{
	.@s = $@NBG2_STATE;
	if (.@s == 0) {
		callfunc "F_NEED_BG2_QueueTick";
		return;
	}
	if (.@s == 1) {
		callfunc "F_NEED_BG2_RecruitWatch";
		return;
	}
	if (.@s < 2 || .@s > 6)
		return;
	.@now = gettimetick(0);
	if (.@s == 6) {
		if (.@now >= $@NBG2_DEADLINE)
			callfunc "F_NEED_BG2_Cleanup", "finished", 0;
		return;
	}
	// 재접속 유예 만료 = 이탈 확정.
	for (.@i = 0; .@i < $@NBG2_N; .@i++) {
		if (!$@NBG2_AWAY[.@i] || .@now - $@NBG2_AWAY_AT[.@i] + $@NBG2_GRACE_USED[.@i] < $@NBG2_C_RECONNECT)
			continue;
		$@NBG2_AWAY[.@i] = 0;
		callfunc "F_NEED_BG2_Log", "AWAY_EXPIRED match=" + $@NBG2_MATCH + " cid=" + $@NBG2_CID[.@i] + " away_at=" + $@NBG2_AWAY_AT[.@i] + " at=" + .@now
			+ " used_before=" + $@NBG2_GRACE_USED[.@i];
		$@NBG2_GRACE_USED[.@i] = $@NBG2_C_RECONNECT;
		callfunc "F_NEED_BG2_Depart", .@i, 1;
		if ($@NBG2_STATE < 2 || $@NBG2_STATE > 5)
			return;
	}
	callfunc "F_NEED_BG2_Watch";
	.@s = $@NBG2_STATE;
	if (.@s < 2 || .@s > 5)
		return;
	if ($@NBG2_DEPART_AT) {
		if (.@now >= $@NBG2_DEPART_AT)
			callfunc "F_NEED_BG2_DepartVerdict";
		return;
	}
	if (.@s == 4 && $@NBG2_JUDGE_AT) {
		// 전멸이 먼저 일어났으면 시간 종료/점령보다 전멸 판정이 우선한다(판정 대기 중에는 점령을 세지 않는다).
		if (.@now >= $@NBG2_JUDGE_AT)
			callfunc "F_NEED_BG2_Judge";
		return;
	}
	// 중앙 거점: 마감 시각까지의 시간만 센다. 점령 승리로 라운드가 끝나면 시간 종료 판정은 하지 않는다.
	if (.@s == 4 && callfunc("F_NEED_BG2_Capture", .@now))
		return;
	if (.@now < $@NBG2_DEADLINE)
		return;
	if (.@s == 2 || .@s == 3)
		callfunc "F_NEED_BG2_BeginRound";
	else if (.@s == 4)
		callfunc "F_NEED_BG2_TimeoutDecide";
	return;
}

// 재접속 유예 안에 돌아온 참가자(첨부)를 원래 팀으로 다시 넣는다.
// 라운드 중이면 그 라운드는 관전석(탈락), 준비/라운드 사이면 시작점에서 정상 준비.
function	script	F_NEED_BG2_Rejoin	{
	.@i = getarg(0);
	.@cid = getcharid(0);
	.@t = $@NBG2_TEAM[.@i];
	.@p = (.@t - 1) * 2 + $@NBG2_SLOT[.@i];
	.@now = gettimetick(0);
	$@NBG2_GRACE_USED[.@i] += .@now - $@NBG2_AWAY_AT[.@i];
	$@NBG2_AWAY[.@i] = 0;
	if (getstatus(SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT))
		sc_end SC_ENTRY_QUEUE_NOTIFY_ADMISSION_TIME_OUT;	// 유예 안 복귀는 탈주가 아니다
	if (Hp < 1)
		recovery 0, .@cid, 4;
	@NEED_BG2_PB = getpcblock() & $@NBG2_C_OURS;
	@NEED_BG2_BGID = $@NBG2_BG[.@t];
	.@inround = ($@NBG2_STATE == 4);
	.@x = .@inround ? $@NBG2_BOXX[.@t] : $@NBG2_SPX[.@p];
	.@y = .@inround ? $@NBG2_BOXY[.@t] : $@NBG2_SPY[.@p];
	if (!bg_join($@NBG2_BG[.@t], $@NBG2_MAP$, .@x, .@y, .@cid)) {
		callfunc "F_NEED_BG2_Log", "REJOIN_FAILED match=" + $@NBG2_MATCH + " cid=" + .@cid;
		$@NBG2_GRACE_USED[.@i] = $@NBG2_C_RECONNECT;
		callfunc "F_NEED_BG2_Depart", .@i, 1;
		return 0;
	}
	if (.@inround) {
		callfunc "F_NEED_BG2_EndNoRevive";
		callfunc "F_NEED_BG2_SetMode", $@NBG2_C_SPEC;
	} else if ($@NBG2_STATE == 5)
		callfunc "F_NEED_BG2_SetMode", $@NBG2_C_LOCK;
	else
		callfunc "F_NEED_BG2_ResetSelf", .@t, $@NBG2_SLOT[.@i], 0;
	callfunc "F_NEED_BG2_Log", "REJOIN match=" + $@NBG2_MATCH + " cid=" + .@cid + " state=" + $@NBG2_STATE + " grace_used=" + $@NBG2_GRACE_USED[.@i] + " at=" + .@now;
	@NEED_BG2_BUSY = 3;
	dispbottom needtr("[2:2 듀오] 경기에 복귀했습니다." + (.@inround ? " 이번 라운드는 관전석에서 기다립니다." : ""),
		"[2v2 Duo] You are back in the match." + (.@inround ? " You wait in the spectator box for this round." : "")), 0x66CCFF;
	return 1;
}

//============================================================
// Phase 2 - 공개 개인 FIFO 대기열 + 30초 참가 확인
//============================================================
// 대기열 : $@NBG2_Q_AID[] / $@NBG2_Q_CID[] / $@NBG2_Q_SEQ[] (서버가 주는 증가 순번, 배열은 항상 순번 오름차순), $@NBG2_Q_N.
//          계정당 1건. 접속 종료(캐릭터 선택 포함)하면 빠진다(기존 @전장대기와 같은 방식).
// 참가 확인: 경기장이 비고($@NBG2_STATE == 0) 대기자 4명 이상이면 맨 앞 4명만 예약한다($@NBG2_RC_*).
//          4명 모두 30초 안에 수락해야 입장 직전 자격 재검사 → 팀 편성 → 입장. 거절/미응답자만 빠지고
//          나머지는 원래 순번을 지킨 채 다음 확인에서 처음부터 다시 수락한다(수락 이월 없음). 불이익 없음.
// 세대 번호: $@NBG2_RC_GEN (확인 회차), 매치 ID, 라운드 번호. 늦은 응답은 현재 회차/마감 시각으로 거른다.
// 다른 전장 연동: 세션 변수 @NEED_BG2_BUSY (1 대기/GM 명단, 2 참가 확인, 3 경기) 를 기존 전장 신청 검사와
//          엔진 전장 대기열(battleground.cpp) 이 읽는다. 이 스크립트가 없어도 변수가 0 이라 기존 동작 그대로다.

// 첨부된 플레이어 또는 접속 중인 다른 플레이어에게 안내.
function	script	F_NEED_BG2_Tell	{
	.@caller = playerattached();
	if (isloggedin(getarg(0), getarg(1)) && attachrid(getarg(0))) {
		if (getcharid(0) == getarg(1)) {
			if (getarg(5, 0))
				announce needtr(getarg(2), getarg(3)), bc_self, getarg(4, "0x66CCFF");
			dispbottom needtr(getarg(2), getarg(3)), 0x66CCFF;
		}
		detachrid;
	}
	callfunc "F_NEED_BG2_Reattach", .@caller;
	return;
}

function	script	F_NEED_BG2_SetBusy	{
	.@caller = playerattached();
	if (isloggedin(getarg(0), getarg(1)) && attachrid(getarg(0))) {
		if (getcharid(0) == getarg(1))
			@NEED_BG2_BUSY = getarg(2);
		detachrid;
	}
	callfunc "F_NEED_BG2_Reattach", .@caller;
	return;
}

function	script	F_NEED_BG2_QIndex	{
	for (.@i = 0; .@i < $@NBG2_Q_N; .@i++)
		if ($@NBG2_Q_AID[.@i] == getarg(0)) return .@i;
	return -1;
}

function	script	F_NEED_BG2_RcIndex	{
	if (!$@NBG2_RC)
		return -1;
	for (.@k = 0; .@k < 4; .@k++)
		if ($@NBG2_RC_AID[.@k] == getarg(0)) return .@k;
	return -1;
}

// 이 계정이 2:2 대기열/참가 확인/진행 중 경기/GM 모집 명단 어디에든 있으면 1.
function	script	F_NEED_BG2_AidBusy	{
	.@aid = getarg(0);
	if (callfunc("F_NEED_BG2_QIndex", .@aid) >= 0 || callfunc("F_NEED_BG2_RcIndex", .@aid) >= 0)
		return 1;
	for (.@i = 0; .@i < $@NBG2_REG_N; .@i++)
		if ($@NBG2_REG_AID[.@i] == .@aid) return 1;
	if ($@NBG2_STATE >= 2 && $@NBG2_STATE <= 6) {
		for (.@i = 0; .@i < $@NBG2_N; .@i++)
			if ($@NBG2_AID[.@i] == .@aid && !$@NBG2_GONE[.@i]) return 1;
	}
	return 0;
}

// 대기열에서 i 번째를 뺀다(뒤 사람의 순서는 그대로). getarg(1) = 로그 태그와 사유.
function	script	F_NEED_BG2_QRemove	{
	.@i = getarg(0);
	if (.@i < 0 || .@i >= $@NBG2_Q_N)
		return 0;
	.@aid = $@NBG2_Q_AID[.@i];
	.@cid = $@NBG2_Q_CID[.@i];
	.@seq = $@NBG2_Q_SEQ[.@i];
	for (.@k = .@i; .@k < $@NBG2_Q_N - 1; .@k++) {
		$@NBG2_Q_AID[.@k] = $@NBG2_Q_AID[.@k + 1];
		$@NBG2_Q_CID[.@k] = $@NBG2_Q_CID[.@k + 1];
		$@NBG2_Q_SEQ[.@k] = $@NBG2_Q_SEQ[.@k + 1];
	}
	$@NBG2_Q_N--;
	$@NBG2_Q_AID[$@NBG2_Q_N] = 0;
	$@NBG2_Q_CID[$@NBG2_Q_N] = 0;
	$@NBG2_Q_SEQ[$@NBG2_Q_N] = 0;
	callfunc "F_NEED_BG2_Log", getarg(1) + " aid=" + .@aid + " cid=" + .@cid + " seq=" + .@seq + " queue=" + $@NBG2_Q_N;
	if (getarg(2, 1))
		callfunc "F_NEED_BG2_SetBusy", .@aid, .@cid, 0;
	return 1;
}

// 원래 순번 그대로 대기열에 되돌린다(입장 실패 복원용). 이미 있으면 아무것도 하지 않는다.
function	script	F_NEED_BG2_QInsert	{
	.@aid = getarg(0);
	.@cid = getarg(1);
	.@seq = getarg(2);
	if (callfunc("F_NEED_BG2_QIndex", .@aid) >= 0)
		return 0;
	for (.@at = 0; .@at < $@NBG2_Q_N; .@at++)
		if ($@NBG2_Q_SEQ[.@at] > .@seq) break;
	for (.@k = $@NBG2_Q_N; .@k > .@at; .@k--) {
		$@NBG2_Q_AID[.@k] = $@NBG2_Q_AID[.@k - 1];
		$@NBG2_Q_CID[.@k] = $@NBG2_Q_CID[.@k - 1];
		$@NBG2_Q_SEQ[.@k] = $@NBG2_Q_SEQ[.@k - 1];
	}
	$@NBG2_Q_AID[.@at] = .@aid;
	$@NBG2_Q_CID[.@at] = .@cid;
	$@NBG2_Q_SEQ[.@at] = .@seq;
	$@NBG2_Q_N++;
	callfunc "F_NEED_BG2_SetBusy", .@aid, .@cid, 1;
	return 1;
}

function	script	F_NEED_BG2_TimerOn	{
	if (!getnpctimer(1, "NEED_BG2"))
		initnpctimer "NEED_BG2";
	return;
}

// 첨부된 플레이어의 공개 대기열 신청. 0 = 등록, 그 밖에는 거절 사유 코드.
function	script	F_NEED_BG2_QJoin	{
	.@aid = getcharid(3);
	.@cid = getcharid(0);
	.@code = 0;
	if (!$NEED_BG2_QUEUE_OPEN)
		.@code = 10;
	else if (callfunc("F_NEED_BG2_AidBusy", .@aid))
		.@code = 9;
	else
		.@code = callfunc("F_NEED_BG2_CheckSelf", 1);
	if (.@code) {
		callfunc "F_NEED_BG2_Log", "QUEUE_REFUSED aid=" + .@aid + " cid=" + .@cid + " code=" + .@code;
		return .@code;
	}
	$@NBG2_QSEQ++;
	$@NBG2_Q_AID[$@NBG2_Q_N] = .@aid;
	$@NBG2_Q_CID[$@NBG2_Q_N] = .@cid;
	$@NBG2_Q_SEQ[$@NBG2_Q_N] = $@NBG2_QSEQ;
	$@NBG2_Q_N++;
	@NEED_BG2_BUSY = 1;
	callfunc "F_NEED_BG2_Log", "QUEUE_JOIN aid=" + .@aid + " cid=" + .@cid + " seq=" + $@NBG2_QSEQ + " pos=" + $@NBG2_Q_N + " class=" + Class;
	callfunc "F_NEED_BG2_TimerOn";
	return 0;
}

// 참가 확인을 연다(경기장이 비고 대기자 4명 이상일 때). 맨 앞 4명만, 순번 그대로.
function	script	F_NEED_BG2_RcOpen	{
	for (.@i = $@NBG2_Q_N - 1; .@i >= 0; .@i--)
		if (!isloggedin($@NBG2_Q_AID[.@i], $@NBG2_Q_CID[.@i]))
			callfunc "F_NEED_BG2_QRemove", .@i, "QUEUE_CANCEL why=offline", 0;
	if ($@NBG2_Q_N < 4)
		return 0;
	$@NBG2_RC_GEN++;
	$@NBG2_RC = 1;
	$@NBG2_RC_END = gettimetick(0) + $@NBG2_C_RC;
	$@NBG2_RC_NOTE = $@NBG2_C_RC / 1000;
	.@who$ = "";
	for (.@k = 0; .@k < 4; .@k++) {
		$@NBG2_RC_AID[.@k] = $@NBG2_Q_AID[.@k];
		$@NBG2_RC_CID[.@k] = $@NBG2_Q_CID[.@k];
		$@NBG2_RC_SEQ[.@k] = $@NBG2_Q_SEQ[.@k];
		$@NBG2_RC_ACC[.@k] = 0;
		.@who$ += (.@k ? "," : "") + $@NBG2_RC_AID[.@k] + "/" + $@NBG2_RC_CID[.@k] + "#" + $@NBG2_RC_SEQ[.@k];
		callfunc "F_NEED_BG2_SetBusy", $@NBG2_RC_AID[.@k], $@NBG2_RC_CID[.@k], 2;
		callfunc "F_NEED_BG2_Tell", $@NBG2_RC_AID[.@k], $@NBG2_RC_CID[.@k],
			"[2:2 듀오] 경기 준비 완료! 30초 안에 @듀오수락 (거절: @듀오거절). 4명 모두 수락하면 입장합니다.",
			"[2v2 Duo] A match is ready! Type @duoaccept within 30 seconds (decline: @duodecline). You enter when all 4 accept.", "0xFFFF66", 1;
	}
	callfunc "F_NEED_BG2_Log", (($@NBG2_RC_RESTART) ? "READY_RESTART" : "READY_OPEN") + " gen=" + $@NBG2_RC_GEN + " members=" + .@who$ + " queue=" + $@NBG2_Q_N;
	$@NBG2_RC_RESTART = 0;
	return 1;
}

// 참가 확인을 닫는다. 대기열에 남은 예약자는 대기(1)로 돌아가고 다음 확인에서 다시 수락해야 한다.
function	script	F_NEED_BG2_RcClose	{
	if (!$@NBG2_RC)
		return 0;
	$@NBG2_RC = 0;
	for (.@k = 0; .@k < 4; .@k++) {
		if (callfunc("F_NEED_BG2_QIndex", $@NBG2_RC_AID[.@k]) < 0)
			continue;
		callfunc "F_NEED_BG2_SetBusy", $@NBG2_RC_AID[.@k], $@NBG2_RC_CID[.@k], 1;
		if (getarg(0, "") != "")
			callfunc "F_NEED_BG2_Tell", $@NBG2_RC_AID[.@k], $@NBG2_RC_CID[.@k], getarg(0), getarg(1);
	}
	cleararray $@NBG2_RC_AID[0], 0, 4;
	cleararray $@NBG2_RC_CID[0], 0, 4;
	cleararray $@NBG2_RC_SEQ[0], 0, 4;
	cleararray $@NBG2_RC_ACC[0], 0, 4;
	$@NBG2_RC_RESTART = getarg(2, 1);
	return 1;
}

// 예약자 k 를 대기열에서 뺀다(거절/미응답/입장 조건 불충족). 불이익 없음.
function	script	F_NEED_BG2_RcDrop	{
	.@k = getarg(0);
	.@q = callfunc("F_NEED_BG2_QIndex", $@NBG2_RC_AID[.@k]);
	if (.@q >= 0)
		callfunc "F_NEED_BG2_QRemove", .@q, getarg(1) + " gen=" + $@NBG2_RC_GEN;
	if (getarg(2, "") != "")
		callfunc "F_NEED_BG2_Tell", $@NBG2_RC_AID[.@k], $@NBG2_RC_CID[.@k], getarg(2), getarg(3), "0xFF7777", 1;
	$@NBG2_RC_AID[.@k] = 0;
	return 1;
}

// 4명 전원 수락: 입장 직전 자격 재검사. 실패한 사람만 빼고 나머지는 순번 그대로 다시 확인한다.
function	script	F_NEED_BG2_RcConfirm	{
	.@caller = playerattached();
	for (.@k = 0; .@k < 4; .@k++) {
		.@code[.@k] = 1;
		if (!isloggedin($@NBG2_RC_AID[.@k], $@NBG2_RC_CID[.@k]) || !attachrid($@NBG2_RC_AID[.@k]))
			continue;
		if (getcharid(0) == $@NBG2_RC_CID[.@k]) {
			.@code[.@k] = callfunc("F_NEED_BG2_CheckSelf", 1);
			$@NBG2_CAND_CLS[.@k] = Class;
		}
		detachrid;
	}
	callfunc "F_NEED_BG2_Reattach", .@caller;
	.@bad = 0;
	for (.@k = 0; .@k < 4; .@k++) {
		if (!.@code[.@k])
			continue;
		.@bad++;
		callfunc "F_NEED_BG2_Log", "ADMISSION_FAIL gen=" + $@NBG2_RC_GEN + " aid=" + $@NBG2_RC_AID[.@k] + " cid=" + $@NBG2_RC_CID[.@k] + " code=" + .@code[.@k];
		.@r$ = callfunc("F_NEED_BG2_Reason$", .@code[.@k]);
		callfunc "F_NEED_BG2_RcDrop", .@k, "QUEUE_CANCEL why=admission_fail",
			"[2:2 듀오] 입장 조건이 맞지 않아 대기열에서 빠졌습니다(불이익 없음): " + .@r$,
			"[2v2 Duo] You were removed from the queue because you cannot enter now (no penalty): " + .@r$;
	}
	if (.@bad) {
		callfunc "F_NEED_BG2_RcClose", "[2:2 듀오] 다른 참가자가 입장할 수 없어 다시 참가 확인을 진행합니다. 순번은 그대로입니다.",
			"[2v2 Duo] Another player cannot enter, so the ready check starts again. Your place in the queue is kept.";
		return 0;
	}
	for (.@k = 0; .@k < 4; .@k++) {
		$@NBG2_CAND_AID[.@k] = $@NBG2_RC_AID[.@k];
		$@NBG2_CAND_CID[.@k] = $@NBG2_RC_CID[.@k];
		$@NBG2_CAND_SEQ[.@k] = $@NBG2_RC_SEQ[.@k];
		callfunc "F_NEED_BG2_QRemove", callfunc("F_NEED_BG2_QIndex", $@NBG2_RC_AID[.@k]), "QUEUE_MATCHED gen=" + $@NBG2_RC_GEN, 0;
	}
	callfunc "F_NEED_BG2_Log", "READY_CONFIRMED gen=" + $@NBG2_RC_GEN;
	$@NBG2_RC = 0;
	cleararray $@NBG2_RC_AID[0], 0, 4;
	cleararray $@NBG2_RC_CID[0], 0, 4;
	cleararray $@NBG2_RC_ACC[0], 0, 4;
	if (callfunc("F_NEED_BG2_Launch", 2)) {
		callfunc "F_NEED_BG2_Reattach", .@caller;
		return 1;
	}
	// 입장 실패: 경기장에 들어간 사람은 Cleanup 이 저장 지점으로 보냈다. 원래 순번으로 되돌리고 다시 확인한다.
	for (.@k = 0; .@k < 4; .@k++) {
		if (!isloggedin($@NBG2_CAND_AID[.@k], $@NBG2_CAND_CID[.@k]))
			continue;
		callfunc "F_NEED_BG2_QInsert", $@NBG2_CAND_AID[.@k], $@NBG2_CAND_CID[.@k], $@NBG2_CAND_SEQ[.@k];
		callfunc "F_NEED_BG2_Tell", $@NBG2_CAND_AID[.@k], $@NBG2_CAND_CID[.@k],
			"[2:2 듀오] 입장에 실패해 저장 지점으로 돌아왔습니다. 대기 순번은 그대로이며 곧 다시 참가 확인을 합니다.",
			"[2v2 Duo] Entering the arena failed and you are back at your save point. Your place in the queue is kept and a new ready check follows.";
	}
	callfunc "F_NEED_BG2_Log", "READY_RESTORE after=start_failed queue=" + $@NBG2_Q_N;
	$@NBG2_RC_RESTART = 1;
	callfunc "F_NEED_BG2_Reattach", .@caller;
	return 0;
}

// 0.5초마다 (경기장 IDLE 일 때): 참가 확인 진행/마감, 또는 새 참가 확인 열기.
function	script	F_NEED_BG2_QueueTick	{
	if (!$@NBG2_RC) {
		if ($NEED_BG2_QUEUE_OPEN && $@NBG2_Q_N >= 4)
			callfunc "F_NEED_BG2_RcOpen";
		return;
	}
	.@acc = 0;
	for (.@k = 0; .@k < 4; .@k++)
		if ($@NBG2_RC_ACC[.@k]) .@acc++;
	if (.@acc == 4) {
		callfunc "F_NEED_BG2_RcConfirm";
		return;
	}
	.@left = $@NBG2_RC_END - gettimetick(0);
	if (.@left <= 0) {
		for (.@k = 0; .@k < 4; .@k++) {
			if ($@NBG2_RC_ACC[.@k])
				continue;
			callfunc "F_NEED_BG2_Log", "READY_TIMEOUT gen=" + $@NBG2_RC_GEN + " aid=" + $@NBG2_RC_AID[.@k] + " cid=" + $@NBG2_RC_CID[.@k];
			callfunc "F_NEED_BG2_RcDrop", .@k, "QUEUE_CANCEL why=ready_timeout",
				"[2:2 듀오] 30초 안에 수락하지 않아 대기열에서 빠졌습니다(불이익 없음). 다시 참가하려면 @듀오참가.",
				"[2v2 Duo] You did not accept within 30 seconds and left the queue (no penalty). To queue again: @duojoin.";
		}
		callfunc "F_NEED_BG2_RcClose", "[2:2 듀오] 수락하지 않은 참가자가 있어 다시 참가 확인을 진행합니다. 순번은 그대로입니다.",
			"[2v2 Duo] Someone did not accept, so the ready check starts again. Your place in the queue is kept.";
		return;
	}
	// 남은 20초/10초에 아직 수락하지 않은 사람에게만 다시 알린다.
	.@sec = (.@left + 999) / 1000;
	if (.@sec <= $@NBG2_RC_NOTE - 10) {
		$@NBG2_RC_NOTE -= 10;
		for (.@k = 0; .@k < 4; .@k++) {
			if ($@NBG2_RC_ACC[.@k])
				continue;
			callfunc "F_NEED_BG2_Tell", $@NBG2_RC_AID[.@k], $@NBG2_RC_CID[.@k],
				"[2:2 듀오] 참가 확인 " + $@NBG2_RC_NOTE + "초 남음: @듀오수락 / @듀오거절",
				"[2v2 Duo] Ready check: " + $@NBG2_RC_NOTE + " seconds left - @duoaccept / @duodecline", "0xFFFF66", 1;
		}
	}
	return;
}

// 공개 대기열 닫기: 참가 확인을 취소하고 대기자 전원을 뺀다(불이익 없음).
function	script	F_NEED_BG2_QueueClose	{
	if ($@NBG2_RC)
		callfunc "F_NEED_BG2_RcClose", "", "", 0;
	for (.@i = $@NBG2_Q_N - 1; .@i >= 0; .@i--) {
		callfunc "F_NEED_BG2_Tell", $@NBG2_Q_AID[.@i], $@NBG2_Q_CID[.@i],
			"[2:2 듀오] 운영자가 2:2 대기열을 닫아 신청이 취소되었습니다.", "[2v2 Duo] The GM closed the 2v2 queue and your entry was cancelled.";
		callfunc "F_NEED_BG2_QRemove", .@i, "QUEUE_CANCEL why=" + getarg(0);
	}
	return;
}

//============================================================
// Phase 2 - 중앙 거점 점령 (관전석을 둘러싼 2칸 폭 고리, x/y 34..45 에서 36..43 제외 = 80칸)
//============================================================
// 라운드 시작 60초 뒤 활성화. 한 팀만(살아 있고 은신하지 않은 참가자) 고리 안에 있으면 게이지가 실제 시간만큼 오른다.
// 같은 팀 2명도 1명과 같은 속도. 양 팀이 함께 있으면 경합(정지). 아무도 없이 2초를 넘기면 실제 1초당 2초분씩 준다.
// 상대 팀이 혼자 점유하면 먼저 그 팀 게이지를 같은 속도로 깎고(상쇄), 0 이 되면 소유가 넘어간다. 10초에 그 팀 라운드 승리.
// 90초 시간 종료 판정용으로 팀별 단독 점유 누적 시간($@NBG2_CAP_T[1..2])을 따로 센다(활성화 이후, 게이지와 별개).
// 0.5초 틱마다 실제 경과 시간(dt)으로 계산하고, 라운드 마감 시각 뒤의 시간은 세지 않는다.

function	script	F_NEED_BG2_InRing	{
	.@x = getarg(0);
	.@y = getarg(1);
	return (.@x >= 34 && .@x <= 45 && .@y >= 34 && .@y <= 45 && !(.@x >= 36 && .@x <= 43 && .@y >= 36 && .@y <= 43));
}

function	script	F_NEED_BG2_CapReset	{
	$@NBG2_CAP_ON = 0;
	$@NBG2_CAP_OWNER = 0;
	$@NBG2_CAP_G = 0;
	$@NBG2_CAP_EMPTY = 0;
	$@NBG2_CAP_T[1] = 0;
	$@NBG2_CAP_T[2] = 0;
	$@NBG2_CAP_LAST = 0;
	$@NBG2_CAP_STATE = 0;
	$@NBG2_CAP_MILE = 0;
	$@NBG2_CAP_ANN_AT = 0;
	return;
}

// 첨부된 참가자가 점령 인원으로 세어지는지(점유 위치/생존/은신 아님).
function	script	F_NEED_BG2_CapCounts	{
	if (Hp < 1 || strcharinfo(3) != $@NBG2_MAP$)
		return 0;
	getmapxy(.@m$, .@x, .@y);
	if (!callfunc("F_NEED_BG2_InRing", .@x, .@y))
		return 0;
	if (checkoption(OPTION_HIDE | OPTION_CLOAK | OPTION_CHASEWALK | OPTION_INVISIBLE))
		return 0;
	if (getstatus(SC_CAMOUFLAGE) || getstatus(SC__INVISIBILITY) || getstatus(SC_STEALTHFIELD) || getstatus(SC_SUHIDE) || getstatus(SC_NEWMOON))
		return 0;
	return 1;
}

// 미니맵 표시(참가자 전원). on = 1 표시, 0 지우기.
function	script	F_NEED_BG2_CapMarker	{
	.@caller = playerattached();
	for (.@i = 0; .@i < $@NBG2_N; .@i++) {
		if ($@NBG2_GONE[.@i] || !isloggedin($@NBG2_AID[.@i], $@NBG2_CID[.@i]) || !attachrid($@NBG2_AID[.@i]))
			continue;
		if (getcharid(0) == $@NBG2_CID[.@i] && strcharinfo(3) == $@NBG2_MAP$)
			viewpoint getarg(0) ? 1 : 2, 39, 39, 1, 0xFFCC00;
		detachrid;
	}
	callfunc "F_NEED_BG2_Reattach", .@caller;
	return;
}

function	script	F_NEED_BG2_CapAnnounce	{
	.@now = gettimetick(0);
	if (!getarg(2, 0) && .@now < $@NBG2_CAP_ANN_AT)
		return;
	$@NBG2_CAP_ANN_AT = .@now + $@NBG2_C_CAP_ANN;
	mapannouncelang $@NBG2_MAP$, getarg(0), getarg(1), bc_map, "0xFFCC33";
	return;
}

// 라운드 중 0.5초마다. 점령 승리로 라운드가 끝나면 1.
function	script	F_NEED_BG2_Capture	{
	if ($@NBG2_STATE != 4)
		return 0;
	.@now = getarg(0);
	if (.@now > $@NBG2_DEADLINE)
		.@now = $@NBG2_DEADLINE;
	if (!$@NBG2_CAP_ON) {
		if (.@now - $@NBG2_ROUND_AT < $@NBG2_C_CAP_ACTIVE)
			return 0;
		$@NBG2_CAP_ON = 1;
		$@NBG2_CAP_LAST = $@NBG2_ROUND_AT + $@NBG2_C_CAP_ACTIVE;
		callfunc "F_NEED_BG2_Log", "CAPTURE_ACTIVE match=" + $@NBG2_MATCH + " round=" + $@NBG2_ROUND + " at=" + $@NBG2_CAP_LAST;
		callfunc "F_NEED_BG2_CapMarker", 1;
		callfunc "F_NEED_BG2_CapAnnounce", "[2:2 듀오] 중앙 거점 활성화! 가운데 관전석을 둘러싼 고리에 혼자 10초 머문 팀이 라운드 승리.",
			"[2v2 Duo] Centre point is open! The team that alone holds the ring around the central box for 10 seconds wins the round.", 1;
	}
	.@dt = .@now - $@NBG2_CAP_LAST;
	if (.@dt <= 0)
		return 0;
	$@NBG2_CAP_LAST = .@now;
	.@caller = playerattached();
	.@n[1] = 0;
	.@n[2] = 0;
	for (.@i = 0; .@i < $@NBG2_N; .@i++) {
		if (!$@NBG2_ALIVE[.@i] || $@NBG2_GONE[.@i] || $@NBG2_AWAY[.@i])
			continue;
		if (!isloggedin($@NBG2_AID[.@i], $@NBG2_CID[.@i]) || !attachrid($@NBG2_AID[.@i]))
			continue;
		if (getcharid(0) == $@NBG2_CID[.@i] && callfunc("F_NEED_BG2_CapCounts"))
			.@n[$@NBG2_TEAM[.@i]] += 1;
		detachrid;
	}
	callfunc "F_NEED_BG2_Reattach", .@caller;
	.@prev = $@NBG2_CAP_STATE;
	if (.@n[1] && .@n[2]) {
		$@NBG2_CAP_STATE = 3;
		$@NBG2_CAP_EMPTY = 0;
	} else if (.@n[1] || .@n[2]) {
		.@t = .@n[1] ? 1 : 2;
		$@NBG2_CAP_STATE = .@t;
		$@NBG2_CAP_EMPTY = 0;
		$@NBG2_CAP_T[.@t] += .@dt;
		if (!$@NBG2_CAP_OWNER || $@NBG2_CAP_OWNER == .@t) {
			if ($@NBG2_CAP_OWNER != .@t)
				$@NBG2_CAP_MILE = 0;
			$@NBG2_CAP_OWNER = .@t;
			$@NBG2_CAP_G += .@dt;
		} else {
			// 상대 팀 진행량을 먼저 상쇄한다. 넘친 시간은 새 소유 팀 게이지가 된다.
			$@NBG2_CAP_G -= .@dt;
			if ($@NBG2_CAP_G <= 0) {
				$@NBG2_CAP_G = -$@NBG2_CAP_G;
				$@NBG2_CAP_OWNER = ($@NBG2_CAP_G > 0) ? .@t : 0;
				$@NBG2_CAP_MILE = 0;
			}
		}
	} else {
		$@NBG2_CAP_EMPTY += .@dt;
		$@NBG2_CAP_STATE = 0;
		if ($@NBG2_CAP_EMPTY > $@NBG2_C_CAP_IDLE && $@NBG2_CAP_G > 0) {
			.@over = $@NBG2_CAP_EMPTY - $@NBG2_C_CAP_IDLE;
			if (.@over > .@dt)
				.@over = .@dt;
			$@NBG2_CAP_G -= .@over * 2;
			if ($@NBG2_CAP_G <= 0) {
				$@NBG2_CAP_G = 0;
				$@NBG2_CAP_OWNER = 0;
				$@NBG2_CAP_MILE = 0;
			}
		}
	}
	if ($@NBG2_CAP_STATE != .@prev) {
		callfunc "F_NEED_BG2_Log", "CAPTURE_" + ($@NBG2_CAP_STATE == 3 ? "CONTEST" : "PROGRESS") + " match=" + $@NBG2_MATCH + " round=" + $@NBG2_ROUND
			+ " state=" + $@NBG2_CAP_STATE + " owner=" + $@NBG2_CAP_OWNER + " gauge=" + $@NBG2_CAP_G + " held=" + $@NBG2_CAP_T[1] + ":" + $@NBG2_CAP_T[2];
		if ($@NBG2_CAP_STATE == 3)
			callfunc "F_NEED_BG2_CapAnnounce", "[2:2 듀오] 중앙 거점 경합 중 - 게이지 정지", "[2v2 Duo] Centre point contested - gauge paused";
		else if ($@NBG2_CAP_STATE)
			callfunc "F_NEED_BG2_CapAnnounce", "[2:2 듀오] " + callfunc("F_NEED_BG2_TeamName$", $@NBG2_CAP_STATE) + " 팀 중앙 거점 점유 중",
				"[2v2 Duo] " + callfunc("F_NEED_BG2_TeamName$", $@NBG2_CAP_STATE) + " team is holding the centre point";
	}
	if ($@NBG2_CAP_OWNER && $@NBG2_CAP_G >= $@NBG2_C_CAP_WIN) {
		callfunc "F_NEED_BG2_Log", "CAPTURE_WIN match=" + $@NBG2_MATCH + " round=" + $@NBG2_ROUND + " team=" + $@NBG2_CAP_OWNER + " gauge=" + $@NBG2_CAP_G
			+ " held=" + $@NBG2_CAP_T[1] + ":" + $@NBG2_CAP_T[2];
		callfunc "F_NEED_BG2_EndRound", $@NBG2_CAP_OWNER, "capture";
		return 1;
	}
	// 진행 안내: 소유 팀 게이지 5초, 8초에 한 번씩.
	if ($@NBG2_CAP_OWNER && $@NBG2_CAP_STATE == $@NBG2_CAP_OWNER) {
		.@m = ($@NBG2_CAP_G >= 8000) ? 2 : (($@NBG2_CAP_G >= 5000) ? 1 : 0);
		if (.@m > $@NBG2_CAP_MILE) {
			$@NBG2_CAP_MILE = .@m;
			callfunc "F_NEED_BG2_CapAnnounce", "[2:2 듀오] " + callfunc("F_NEED_BG2_TeamName$", $@NBG2_CAP_OWNER) + " 팀 점령 " + ($@NBG2_CAP_G / 1000) + "/10초",
				"[2v2 Duo] " + callfunc("F_NEED_BG2_TeamName$", $@NBG2_CAP_OWNER) + " team capture " + ($@NBG2_CAP_G / 1000) + "/10 s", 1;
		}
	}
	return 0;
}

// 90초 시간 종료: 생존 인원 많은 팀 → 단독 점유 누적 시간 긴 팀 → 무승부.
function	script	F_NEED_BG2_TimeoutDecide	{
	.@b = callfunc("F_NEED_BG2_AliveCount", 1);
	.@r = callfunc("F_NEED_BG2_AliveCount", 2);
	.@why$ = "timeout";
	.@res = 0;
	if (.@b != .@r) {
		.@res = (.@b > .@r) ? 1 : 2;
		.@why$ = "timeout_alive";
	} else if ($@NBG2_CAP_T[1] != $@NBG2_CAP_T[2]) {
		.@res = ($@NBG2_CAP_T[1] > $@NBG2_CAP_T[2]) ? 1 : 2;
		.@why$ = "timeout_capture";
	}
	callfunc "F_NEED_BG2_Log", "ROUND_TIMEOUT_DECISION match=" + $@NBG2_MATCH + " round=" + $@NBG2_ROUND + " alive=" + .@b + ":" + .@r
		+ " held=" + $@NBG2_CAP_T[1] + ":" + $@NBG2_CAP_T[2] + " res=" + .@res + " why=" + .@why$;
	callfunc "F_NEED_BG2_EndRound", .@res, .@why$;
	return;
}

// GM 강제 중단(무효). 어떤 상태에서 불러도 한 번만 정리된다.
function	script	F_NEED_BG2_GmStop	{
	.@s = $@NBG2_STATE;
	if (.@s == 0 && $@NBG2_RC) {
		callfunc "F_NEED_BG2_Log", "READY_CANCEL gen=" + $@NBG2_RC_GEN + " by=gm";
		callfunc "F_NEED_BG2_RcClose", "[2:2 듀오] 운영자가 참가 확인을 취소했습니다. 대기 순번은 그대로입니다.",
			"[2v2 Duo] The GM cancelled the ready check. Your place in the queue is kept.";
		dispbottom "[2:2 듀오] 참가 확인을 취소했습니다(대기열은 유지, 대기열이 열려 있으면 곧 다시 확인합니다).";
		return 1;
	}
	if (.@s == 0 || .@s == 7) {
		dispbottom "[2:2 듀오] 진행 중인 모집/경기가 없습니다.";
		return 0;
	}
	if (.@s == 1) {
		callfunc "F_NEED_BG2_TellRegs", "[2:2 듀오] 운영자가 모집을 닫았습니다.", "[2v2 Duo] The GM closed sign-ups.";
		callfunc "F_NEED_BG2_ClearRecruit";
		$@NBG2_STATE = 0;
		if (!$@NBG2_Q_N)
			stopnpctimer "NEED_BG2";
		callfunc "F_NEED_BG2_Log", "RECRUIT_CLOSED by=gm";
		dispbottom "[2:2 듀오] 모집을 닫았습니다.";
		return 1;
	}
	if (.@s >= 2 && .@s <= 5)
		callfunc "F_NEED_BG2_Finish", 3, "gm_stop";
	callfunc "F_NEED_BG2_Cleanup", "gm_stop", 0;
	dispbottom "[2:2 듀오] 강제 중단(무효) 처리했습니다.";
	return 1;
}

// GM 고립자 복구: 이름으로 찾은 접속 중 캐릭터의 2:2 상태를 정리한다.
function	script	F_NEED_BG2_GmRecover	{
	.@gm = playerattached();
	.@name$ = getarg(0);
	.@aid = getcharid(3, .@name$);
	if (!.@aid || !attachrid(.@aid)) {
		callfunc "F_NEED_BG2_Reattach", .@gm;
		dispbottom "[2:2 듀오] 접속 중인 캐릭터를 찾지 못했습니다: " + .@name$ + " (접속 종료 상태면 다음 접속 때 자동 복구됩니다)";
		return 0;
	}
	.@i = callfunc("F_NEED_BG2_IndexOf", getcharid(0));
	if (.@i >= 0 && $@NBG2_STATE >= 2 && $@NBG2_STATE <= 6)
		.@live = !$@NBG2_GONE[.@i];	// && 는 단락 평가가 아니므로 .@i < 0 일 때 배열을 읽지 않는다
	if (.@live) {
		callfunc "F_NEED_BG2_Reattach", .@gm;
		dispbottom "[2:2 듀오] 진행 중인 경기 참가자입니다. 강제 중단을 먼저 사용하세요.";
		return 0;
	}
	.@had = (NEED_BG2_MID || @NEED_BG2_BGID || strcharinfo(3) == $@NBG2_MAP$);
	if (.@had)
		callfunc "F_NEED_BG2_RestoreSelf", "gm_recover", 1, 1;	// 운영자 복구는 탈주가 아니다
	if (strcharinfo(3) == $@NBG2_MAP$)
		warp "SavePoint", 0, 0;
	callfunc "F_NEED_BG2_Reattach", .@gm;
	dispbottom "[2:2 듀오] " + .@name$ + (.@had ? " 복구 완료." : ": 정리할 2:2 상태가 없습니다.");
	return 1;
}

	.@aid = getcharid(3);
	if ($@NBG2_STATE == 1) {
		// GM 수동 4인 테스트 모집 중: 1단계 방식 그대로(공개 대기열과 겹칠 수 없다).
		.@code = 0;
		if (callfunc("F_NEED_BG2_AidBusy", .@aid))
			.@code = 9;
		if (!.@code && $@NBG2_REG_N >= 4)
			.@code = 11;
		if (!.@code)
			.@code = callfunc("F_NEED_BG2_CheckSelf", 0);
		if (.@code) {
			callfunc "F_NEED_BG2_Log", "JOIN_REFUSED aid=" + .@aid + " cid=" + getcharid(0) + " code=" + .@code + " regs=" + $@NBG2_REG_N;
			dispbottom needtr("[2:2 듀오] 신청할 수 없습니다: ", "[2v2 Duo] You cannot sign up: ") + callfunc("F_NEED_BG2_Reason$", .@code), 0xFF7777;
			end;
		}
		$@NBG2_REG_AID[$@NBG2_REG_N] = .@aid;
		$@NBG2_REG_CID[$@NBG2_REG_N] = getcharid(0);
		$@NBG2_REG_N++;
		@NEED_BG2_BUSY = 1;
		callfunc "F_NEED_BG2_Log", "JOIN aid=" + .@aid + " cid=" + getcharid(0) + " regs=" + $@NBG2_REG_N;
		dispbottom needtr("[2:2 듀오] GM 테스트 신청 완료 (" + $@NBG2_REG_N + "/4). 팀은 시작할 때 정해집니다. 취소: @듀오취소",
			"[2v2 Duo] GM test sign-up done (" + $@NBG2_REG_N + "/4). Teams are set at the start. To cancel: @duocancel"), 0x66CCFF;
		end;
	}
	.@code = callfunc("F_NEED_BG2_QJoin");
	if (.@code) {
		dispbottom needtr("[2:2 듀오] 대기열에 신청할 수 없습니다: ", "[2v2 Duo] You cannot join the queue: ") + callfunc("F_NEED_BG2_Reason$", .@code), 0xFF7777;
		end;
	}
	.@pos = callfunc("F_NEED_BG2_QIndex", .@aid) + 1;
	dispbottom needtr("[2:2 듀오] 개인 대기열 신청 완료. 순번 " + .@pos + " / 대기 " + $@NBG2_Q_N + "명. 다른 맵에서 사냥해도 대기가 유지됩니다.",
		"[2v2 Duo] You joined the solo queue. Position " + .@pos + " / " + $@NBG2_Q_N + " waiting. You stay queued while playing on other maps."), 0x66CCFF;
	dispbottom needtr("[2:2 듀오] 경기장이 비고 4명이 모이면 30초 참가 확인(@듀오수락)을 요청합니다. 취소: @듀오취소 / 현황: @듀오현황",
		"[2v2 Duo] When the arena is free and 4 players are waiting you get a 30-second ready check (@duoaccept). Cancel: @duocancel / status: @duostatus"), 0x66CCFF;
	end;

OnCmdCancel:
	.@aid = getcharid(3);
	.@cid = getcharid(0);
	.@k = callfunc("F_NEED_BG2_RcIndex", .@aid);
	if (.@k >= 0) {
		callfunc "F_NEED_BG2_Log", "READY_DECLINE gen=" + $@NBG2_RC_GEN + " aid=" + .@aid + " cid=" + .@cid + " why=cancel";
		callfunc "F_NEED_BG2_RcDrop", .@k, "QUEUE_CANCEL why=cancel";
		callfunc "F_NEED_BG2_RcClose", "[2:2 듀오] 참가자 한 명이 취소해 다시 참가 확인을 진행합니다. 순번은 그대로입니다.",
			"[2v2 Duo] A player cancelled, so the ready check starts again. Your place in the queue is kept.";
		dispbottom needtr("[2:2 듀오] 대기열 신청을 취소했습니다.", "[2v2 Duo] You left the queue."), 0x66CCFF;
		end;
	}
	.@q = callfunc("F_NEED_BG2_QIndex", .@aid);
	if (.@q >= 0) {
		callfunc "F_NEED_BG2_QRemove", .@q, "QUEUE_CANCEL why=cancel";
		dispbottom needtr("[2:2 듀오] 대기열 신청을 취소했습니다.", "[2v2 Duo] You left the queue."), 0x66CCFF;
		end;
	}
	if ($@NBG2_STATE == 1) {
		for (.@i = 0; .@i < $@NBG2_REG_N; .@i++) {
			if ($@NBG2_REG_AID[.@i] != .@aid) continue;
			callfunc "F_NEED_BG2_RemoveReg", .@i;
			callfunc "F_NEED_BG2_Log", "UNREGISTER cid=" + .@cid + " why=cancel regs=" + $@NBG2_REG_N;
			dispbottom needtr("[2:2 듀오] 신청을 취소했습니다.", "[2v2 Duo] Your sign-up was cancelled."), 0x66CCFF;
			end;
		}
	}
	if (callfunc("F_NEED_BG2_IndexOf", .@cid) >= 0 && $@NBG2_STATE >= 2 && $@NBG2_STATE <= 5) {
		dispbottom needtr("[2:2 듀오] 경기 중에는 취소할 수 없습니다. 경기장을 떠나면 소속 팀의 몰수패가 됩니다.",
			"[2v2 Duo] You cannot cancel during a match. Leaving the arena forfeits the match for your team."), 0xFF7777;
		end;
	}
	dispbottom needtr("[2:2 듀오] 취소할 신청이 없습니다.", "[2v2 Duo] You have no sign-up to cancel."), 0xFF7777;
	end;

OnCmdAccept:
	.@aid = getcharid(3);
	.@k = callfunc("F_NEED_BG2_RcIndex", .@aid);
	if (.@k >= 0) {	// || 는 단락 평가가 아니므로 음수 인덱스로 배열을 읽지 않게 나눈다
		if ($@NBG2_RC_CID[.@k] != getcharid(0))
			.@k = -1;
	}
	if (.@k < 0) {
		dispbottom needtr("[2:2 듀오] 지금 수락할 참가 확인이 없습니다.", "[2v2 Duo] There is no ready check for you right now."), 0xFF7777;
		end;
	}
	if (gettimetick(0) >= $@NBG2_RC_END) {
		callfunc "F_NEED_BG2_Log", "READY_LATE gen=" + $@NBG2_RC_GEN + " aid=" + .@aid + " cid=" + getcharid(0);
		dispbottom needtr("[2:2 듀오] 참가 확인 시간이 지났습니다.", "[2v2 Duo] The ready check time is over."), 0xFF7777;
		end;
	}
	if ($@NBG2_RC_ACC[.@k]) {
		dispbottom needtr("[2:2 듀오] 이미 수락했습니다. 다른 참가자를 기다립니다.", "[2v2 Duo] You already accepted. Waiting for the others."), 0x66CCFF;
		end;
	}
	$@NBG2_RC_ACC[.@k] = 1;
	.@acc = 0;
	for (.@j = 0; .@j < 4; .@j++)
		if ($@NBG2_RC_ACC[.@j]) .@acc++;
	callfunc "F_NEED_BG2_Log", "READY_ACCEPT gen=" + $@NBG2_RC_GEN + " aid=" + .@aid + " cid=" + getcharid(0) + " accepted=" + .@acc + "/4";
	dispbottom needtr("[2:2 듀오] 수락했습니다 (" + .@acc + "/4). 4명이 모두 수락하면 바로 입장합니다.",
		"[2v2 Duo] Accepted (" + .@acc + "/4). You enter as soon as all 4 accept."), 0x66CCFF;
	end;

OnCmdDecline:
	.@aid = getcharid(3);
	.@k = callfunc("F_NEED_BG2_RcIndex", .@aid);
	if (.@k >= 0) {
		if ($@NBG2_RC_CID[.@k] != getcharid(0))
			.@k = -1;
	}
	if (.@k < 0) {
		dispbottom needtr("[2:2 듀오] 지금 거절할 참가 확인이 없습니다. 대기 취소는 @듀오취소.", "[2v2 Duo] There is no ready check for you right now. To leave the queue: @duocancel."), 0xFF7777;
		end;
	}
	callfunc "F_NEED_BG2_Log", "READY_DECLINE gen=" + $@NBG2_RC_GEN + " aid=" + .@aid + " cid=" + getcharid(0) + " why=decline";
	callfunc "F_NEED_BG2_RcDrop", .@k, "QUEUE_CANCEL why=decline";
	callfunc "F_NEED_BG2_RcClose", "[2:2 듀오] 참가자 한 명이 거절해 다시 참가 확인을 진행합니다. 순번은 그대로입니다.",
		"[2v2 Duo] A player declined, so the ready check starts again. Your place in the queue is kept.";
	dispbottom needtr("[2:2 듀오] 거절했습니다. 대기열에서 빠졌습니다(불이익 없음).", "[2v2 Duo] Declined. You left the queue (no penalty)."), 0x66CCFF;
	end;

OnCmdStatus:
	.@left = 0;
	if ($@NBG2_DEADLINE)
		.@left = ($@NBG2_DEADLINE - gettimetick(0)) / 1000;
	if (.@left < 0) .@left = 0;
	.@i = callfunc("F_NEED_BG2_IndexOf", getcharid(0));
	.@mine$ = (.@i >= 0) ? needtr(" / 내 팀 ", " / my team ") + callfunc("F_NEED_BG2_TeamName$", $@NBG2_TEAM[.@i]) : "";
	.@q = callfunc("F_NEED_BG2_QIndex", getcharid(3));
	.@k = callfunc("F_NEED_BG2_RcIndex", getcharid(3));
	dispbottom needtr("[2:2 듀오] 대기열 ", "[2v2 Duo] queue ") + ($NEED_BG2_QUEUE_OPEN ? needtr("열림", "open") : needtr("닫힘", "closed"))
		+ needtr(" / 대기 ", " / waiting ") + $@NBG2_Q_N
		+ ((.@q >= 0) ? needtr(" / 내 순번 ", " / my position ") + (.@q + 1) : "")
		+ ((.@k >= 0) ? needtr(" / 참가 확인 중 ", " / ready check ") + (($@NBG2_RC_END - gettimetick(0)) / 1000) + needtr("초 남음 (@듀오수락)", "s left (@duoaccept)") : ""), 0x66CCFF;
	dispbottom needtr("[2:2 듀오] 상태 ", "[2v2 Duo] state ") + callfunc("F_NEED_BG2_StateName$", $@NBG2_STATE)
		+ needtr(" / 신청 ", " / sign-ups ") + $@NBG2_REG_N + "/4"
		+ needtr(" / 라운드 ", " / round ") + $@NBG2_ROUND
		+ needtr(" / 블루 ", " / Blue ") + $@NBG2_WIN_B + " : " + $@NBG2_WIN_R + needtr(" 레드", " Red")
		+ needtr(" / 남은 시간 ", " / time left ") + .@left + needtr("초", "s") + .@mine$, 0x66CCFF;
	end;

OnCmdGM:
	if (getgmlevel() < 60)
		end;
	.@sub$ = (.@atcmd_numparameters > 0) ? .@atcmd_parameters$[0] : "";
	if (.@sub$ == "open") callsub S_Open;
	else if (.@sub$ == "start") callfunc "F_NEED_BG2_Start";
	else if (.@sub$ == "status") callfunc "F_NEED_BG2_StatusLines";
	else if (.@sub$ == "stop") callfunc "F_NEED_BG2_GmStop";
	else if (.@sub$ == "recover") callfunc "F_NEED_BG2_GmRecover", .@atcmd_parameters$[1];
	else if (.@sub$ == "lvbypass") callsub S_LvBypass, atoi(.@atcmd_parameters$[1]);
	else if (.@sub$ == "log") callsub S_Log, atoi(.@atcmd_parameters$[1]);
	else if (.@sub$ == "queue") callsub S_Queue, .@atcmd_parameters$[1];
	else if (.@sub$ != "") dispbottom "[2:2 듀오] 사용법: @duo2v2 [open|start|status|stop|recover <이름>|lvbypass 0/1|log 0/1|queue on/off]";
	else callsub S_Menu;
	end;

S_Menu:
	mes "[NEED 2:2 듀오 서바이벌 - GM 테스트]";
	mes "상태: " + callfunc("F_NEED_BG2_StateName$", $@NBG2_STATE) + " / 신청 " + $@NBG2_REG_N + "/4";
	if ($@NBG2_N)
		mes "매치 " + $@NBG2_MATCH + " / " + $@NBG2_ROUND + "라운드 / 블루 " + $@NBG2_WIN_B + " : " + $@NBG2_WIN_R + " 레드";
	mes "레벨 제한: " + ($NEED_BG2_LVBYPASS ? "^FF0000우회 중(GM 테스트)^000000" : $@NBG2_C_MINLV + " 이상");
	mes "공개 대기열: " + ($NEED_BG2_QUEUE_OPEN ? "열림" : "닫힘") + " / 대기 " + $@NBG2_Q_N + "명";
	switch (select("GM 테스트 모집 열기", "신청 현황 / 상태 / 점수", "4인 편성 후 시작 (직업 균형)", "강제 중단 (무효)", "고립 참가자 복구", "설정", "닫기")) {
	case 1: close2; callsub S_Open; end;
	case 2: close2; callfunc "F_NEED_BG2_StatusLines"; end;
	case 3: close2; callfunc "F_NEED_BG2_Start"; end;
	case 4:
		mes "진행 중인 모집/경기를 무효로 끝내고 전원을 저장 지점으로 돌려보냅니다. 참가 확인 중이면 그 확인만 취소합니다.";
		if (select("중단", "취소") == 1) {
			close2;
			callfunc "F_NEED_BG2_GmStop";
		} else close2;
		end;
	case 5:
		mes "복구할 캐릭터 이름을 입력하세요.";
		input .@n$;
		close2;
		callfunc "F_NEED_BG2_GmRecover", .@n$;
		end;
	case 6:
		mes "레벨 제한 우회(GM 테스트 옵션): " + ($NEED_BG2_LVBYPASS ? "ON" : "OFF");
		mes "디버그 로그: " + ($NEED_BG2_LOG_OFF ? "OFF" : "ON");
		mes "공개 대기열: " + ($NEED_BG2_QUEUE_OPEN ? "열림" : "닫힘");
		switch (select("레벨 제한 우회 전환", "디버그 로그 전환", "공개 대기열 열기/닫기", "닫기")) {
		case 1: close2; callsub S_LvBypass, !$NEED_BG2_LVBYPASS; end;
		case 2: close2; callsub S_Log, $NEED_BG2_LOG_OFF; end;
		case 3: close2; callsub S_Queue, ($NEED_BG2_QUEUE_OPEN ? "off" : "on"); end;
		}
		close;
	}
	close;

S_Queue:
	if (getarg(0) == "on") {
		$NEED_BG2_QUEUE_OPEN = 1;
		callfunc "F_NEED_BG2_TimerOn";
		callfunc "F_NEED_BG2_Log", "QUEUE_OPEN by_aid=" + getcharid(3);
		dispbottom "[2:2 듀오] 공개 개인 대기열을 열었습니다. 일반 유저가 @듀오참가 로 신청합니다.";
	} else if (getarg(0) == "off") {
		$NEED_BG2_QUEUE_OPEN = 0;
		callfunc "F_NEED_BG2_QueueClose", "queue_closed";
		callfunc "F_NEED_BG2_Log", "QUEUE_CLOSE by_aid=" + getcharid(3);
		dispbottom "[2:2 듀오] 공개 대기열을 닫았습니다(대기자 신청 취소, 진행 중인 경기는 그대로).";
	} else
		dispbottom "[2:2 듀오] 공개 대기열: " + ($NEED_BG2_QUEUE_OPEN ? "열림" : "닫힘") + " / 대기 " + $@NBG2_Q_N + "명 / 참가 확인 " + ($@NBG2_RC ? "진행 중" : "없음") + " (@duo2v2 queue on|off)";
	return;

S_Open:
	if ($@NBG2_STATE != 0 || $@NBG2_RC) {
		dispbottom "[2:2 듀오] 경기장이 사용 중이거나 공개 대기열 참가 확인이 진행 중입니다. (상태 " + callfunc("F_NEED_BG2_StateName$", $@NBG2_STATE) + ", 참가 확인 " + $@NBG2_RC + ")";
		return;
	}
	callfunc "F_NEED_BG2_ClearRecruit";
	callfunc "F_NEED_BG2_ClearMatch";
	$@NBG2_STATE = 1;
	initnpctimer "NEED_BG2";
	callfunc "F_NEED_BG2_Log", "RECRUIT_OPEN by_aid=" + getcharid(3) + " lvbypass=" + $NEED_BG2_LVBYPASS;
	dispbottom "[2:2 듀오] 모집을 열었습니다. 참가자는 @duojoin (@듀오참가) 로 신청합니다.";
	return;

S_LvBypass:
	if ($@NBG2_STATE >= 2 && $@NBG2_STATE <= 6) {
		dispbottom "[2:2 듀오] 경기 중에는 바꿀 수 없습니다.";
		return;
	}
	$NEED_BG2_LVBYPASS = getarg(0) ? 1 : 0;
	callfunc "F_NEED_BG2_Log", "LVBYPASS=" + $NEED_BG2_LVBYPASS + " by_aid=" + getcharid(3);
	dispbottom "[2:2 듀오] 레벨 제한 우회: " + ($NEED_BG2_LVBYPASS ? "ON (GM 테스트 옵션)" : "OFF (BaseLevel " + $@NBG2_C_MINLV + " 이상)");
	return;

S_Log:
	$NEED_BG2_LOG_OFF = getarg(0) ? 0 : 1;
	dispbottom "[2:2 듀오] 디버그 로그: " + ($NEED_BG2_LOG_OFF ? "OFF" : "ON");
	return;
}
