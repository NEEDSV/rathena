// Shop active when a GM test override is set OR the current logical date
// (04:00 boundary) is within the event/exchange window 2026-08-15 ~ 2026-09-18.
function	script	F_NeedSummerShopActive	{
	if ( $NS_SHOP_TEST ) return 1;
	.@ld = atoi( gettimestr("%Y%m%d", 9, gettimetick(2) - 14400) );
	if ( .@ld >= 20260815 && .@ld <= 20260918 ) return 1;
	return 0;
}
