//============================================================
// NEED - 과거 행운의 알 상점
//
// 1단계 (랜덤 한정 판매)
// - 매일 00:00 / 12:00 갱신
// - 활성화된 과거 행운의 알을 무작위 셔플 후 3개 그룹으로 나누고,
//   회차마다 그 중 한 그룹(활성 풀의 약 1/3)을 통째로 판매한다.
//   예) 활성 36종 -> 12 / 12 / 12
//       활성 35종 -> 12 / 12 / 11
//       활성 37종 -> 13 / 12 / 12
//   3그룹을 모두 소진하면 다시 전체를 셔플한다.
//   => 한 사이클(12시간 x 3회) 동안 활성 행알이 정확히 한 번씩 등장한다.
// - 상품별 기본가 1,000 Cash
// - 상품별 랜덤 판매가: 110 / 120 / 130 / 140 / 150%
// - 상품별 서버 재고: 300~500개
// - 계정당 회차 구매 제한: 상품별 10개
//
// 2단계
// - 추억의 행운 조각(900106) 30개
// - 11,000 Cash
// - 원하는 활성화 과거 행운의 알 10개 선택 구매
//
// 활성화 정책
// - 12855 교황 스크롤까지 활성화
// - 이후 품목은 모두 비활성 상태로 미리 등록
// - 시즌 개방 시 .EggEnabled[]의 해당 항목만 0 -> 1 변경
//
// 갱신 10분 전 전체 공지
// - 11:50 / 23:50
//
//------------------------------------------------------------
// [로테이션 영속화]
// 같은 12시간 슬롯 안에서는 map-server 재시작 / @reloadscript 가
// 발생해도 판매 목록·가격·재고가 그대로 유지되어야 한다.
// 그래서 회차 상태를 전부 $ (mapreg) 전역 변수에 저장한다.
//
//   $NeedPastEggRotation$   현재 12시간 슬롯 키 "YYYY-M-D-슬롯"
//   $NeedPastEggSaleKey$    회차 식별 키 (슬롯 키 + "#" + 갱신 일련번호)
//   $NeedPastEggSerial      갱신 일련번호. 갱신될 때마다 +1
//   $NeedPastEggCycle       셔플 사이클 번호. 전체 재셔플마다 +1
//   $NeedPastEggOrder[]     현재 사이클의 셔플 결과
//                           (활성 행알의 .EggId[] 인덱스 전체)
//   $NeedPastEggListSig     셔플 당시의 .EggId[] 서명. 목록 자체가 바뀌면 재셔플
//   $NeedPastEggGroup       현재 판매 중인 그룹 번호 (0 ~ 2)
//
//   $NeedPastEggSaleIdx[]   이번 회차 판매 상품의 .EggId[] 인덱스
//   $NeedPastEggSaleId[]    이번 회차 판매 상품 ItemID
//   $NeedPastEggSaleRate[]  상품별 가격 배율 %
//   $NeedPastEggSalePrice[] 상품별 개당 Cash 가격
//   $NeedPastEggSaleStock[] 상품별 남은 서버 재고
//   $NeedPastEggSaleInit[]  상품별 최초 재고 (90% 공지 판정용)
//   $NeedPastEggSaleN90[]   상품별 90% 공지 완료 플래그
//   $NeedPastEggSaleOut[]   상품별 품절 공지 완료 플래그
//
//   #NeedPastEggRotation$   계정이 마지막으로 본 회차 키
//   #NeedPastEggBought[]    계정의 상품별 이번 회차 구매량
//
// OnInit 은 S_EnsureRotation 만 부른다. 슬롯 키가 같으면 아무것도
// 다시 뽑지 않으므로 재시작/리로드로 목록이 바뀌지 않는다.
// 그룹 전진은 OnClock0000 / OnClock1200 (그리고 @eggadmin 강제 갱신)
// 에서만 일어난다.
//
// [중요] conf/script_athena.conf 의 check_gotocount 가 2048 이라
// 한 번에 도는 루프 총량을 작게 유지해야 한다. 그래서 셔플 결과를
// ItemID 가 아니라 .EggId[] 인덱스로 저장한다.
// 덕분에 "저장된 순서가 현재 활성 풀과 같은가" 검사를
// .EggEnabled[인덱스] 조회 한 번(O(1))으로 끝낼 수 있다.
// ItemID 로 저장하면 매 회차 O(n^2) 역조회가 필요해
// infinity loop 오류로 갱신이 통째로 중단된다.
//============================================================

lasagna,169,195,4	script	추억의 행운의 알 상인#need_past_egg	4_F_KAFRA6,{

	callsub S_EnsureRotation;
	callsub S_SyncAccountRotation;

	while (1) {
		mes "[추억의 행운의 알 상인]";
		mes "지나간 시즌의 행운의 알을";
		mes "다시 만날 수 있는 특별 판매소입니다.";
		mes " ";

		.@list_n = getarraysize($NeedPastEggSaleId);

		if (.@list_n > 0) {
			mes "이번 회차 한정 판매: ^0055FF" + .@list_n + "종^000000";
			mes "(전체 활성 행알의 약 1/3)";
			mes "갱신: ^0055FF00:00 / 12:00^000000";
			mes "계정 구매 제한: 상품별 " + .BuyLimit + "개";
		}
		else {
			mes "^FF0000현재 판매 가능한 행운의 알이 없습니다.^000000";
		}
		setdialogsize(300,300);
		next;

		switch (select("랜덤 한정 판매 이용:과거 행운의 알 선택 구매:시스템 설명:닫기")) {
		case 1:
			callsub S_RandomShop;
			break;

		case 2:
			callsub S_SelectShop;
			break;

		case 3:
			mes "[추억의 행운의 알 상인]";
			mes "랜덤 한정 판매는 매일";
			mes "^0055FF00:00 / 12:00^000000에 갱신됩니다.";
			mes " ";
			mes "활성화된 과거 행운의 알 전체를 무작위로 섞어";
			mes "^0055FF3개의 그룹^000000으로 나눈 뒤,";
			mes "회차마다 한 그룹씩 판매합니다.";
			next;

			mes "[추억의 행운의 알 상인]";
			mes "3개의 그룹을 모두 판매하고 나면";
			mes "다시 전체를 섞어 새로운 순서를 만듭니다.";
			mes " ";
			mes "덕분에 같은 알이 계속 반복되거나";
			mes "특정 알이 오랫동안 나오지 않는 일이 없습니다.";
			next;

			mes "[추억의 행운의 알 상인]";
			mes "판매 가격과 서버 재고는";
			mes "상품마다 따로 결정됩니다.";
			mes " ";
			mes "한정 판매 상품은 계정당";
			mes "^0055FF상품별 최대 " + .BuyLimit + "개^000000까지 구매할 수 있습니다.";
			next;

			mes "[추억의 행운의 알 상인]";
			mes "원하는 과거 행운의 알을 직접 구매할 경우";
			mes "^FF0000" + getitemname(.MemoryFragment) + " " + .SelectFragmentCost + "개^000000";
			mes "+ ^FF8800" + .SelectCashPrice + " Cash^000000가 필요합니다.";
			mes "선택한 행운의 알 " + .SelectEggAmount + "개를 받을 수 있습니다.";
			next;
			break;

		default:
			close;
		}
	}


//============================================================
// 1단계 - 랜덤 한정 판매 (이번 회차 그룹 전체 목록)
//============================================================
S_RandomShop:
	callsub S_EnsureRotation;
	callsub S_SyncAccountRotation;

	while (1) {
		deletearray .@slot[0];
		.@menu$ = "";
		.@cnt = 0;
		.@size = getarraysize($NeedPastEggSaleId);

		for (.@i = 0; .@i < .@size; .@i++) {
			.@id = $NeedPastEggSaleId[.@i];
			if (.@id <= 0 || $NeedPastEggSalePrice[.@i] <= 0)
				continue;

			// 회차 도중 운영자가 비활성으로 바꾼 행알은 노출하지 않는다.
			.@x = $NeedPastEggSaleIdx[.@i];
			if (.EggId[.@x] != .@id || !.EggEnabled[.@x])
				continue;

			.@slot[.@cnt] = .@i;

			if (.@cnt > 0)
				.@menu$ += ":";

			.@menu$ += getitemname(.@id)
				+ " " + $NeedPastEggSalePrice[.@i] + "C"
				+ " (재고 " + $NeedPastEggSaleStock[.@i]
				+ " / 내구매 " + #NeedPastEggBought[.@i] + ")";

			.@cnt++;
		}

		if (.@cnt <= 0) {
			mes "[추억의 행운의 알 상인]";
			mes "현재 판매 가능한 상품이 없습니다.";
			next;
			return;
		}

		mes "[추억의 행운의 알 한정 판매]";
		mes "이번 회차 상품 ^0055FF" + .@cnt + "종^000000입니다.";
		mes " ";
		mes "구매할 상품을 선택해주세요.";
		mes "계정당 ^0055FF상품별 " + .BuyLimit + "개^000000까지 구매할 수 있습니다.";
		setdialogsize(420,340);
		next;

		.@sel = select(.@menu$ + ":돌아가기") - 1;
		if (.@sel < 0 || .@sel >= .@cnt)
			return;

		callsub S_BuyOne, .@slot[.@sel];
	}
	return;


//============================================================
// 한정 판매 상품 1종 구매
//
// getarg(0) : $NeedPastEggSale* 배열의 index
//============================================================
S_BuyOne:
	.@b_idx = getarg(0);
	.@b_key$ = $NeedPastEggSaleKey$;
	.@b_item = $NeedPastEggSaleId[.@b_idx];
	.@b_price = $NeedPastEggSalePrice[.@b_idx];

	if (.@b_item <= 0 || .@b_price <= 0) {
		mes "[추억의 행운의 알 상인]";
		mes "현재 판매 가능한 상품이 없습니다.";
		next;
		return;
	}

	if ($NeedPastEggSaleStock[.@b_idx] <= 0) {
		mes "[추억의 행운의 알 상인]";
		mes "^FF0000" + getitemname(.@b_item) + "^000000";
		mes "^FF0000은(는) 이번 회차 수량이 모두 판매되었습니다.^000000";
		mes "다음 갱신을 기다려주세요.";
		next;
		return;
	}

	.@b_remain = .BuyLimit - #NeedPastEggBought[.@b_idx];
	if (.@b_remain <= 0) {
		mes "[추억의 행운의 알 상인]";
		mes "^0055FF" + getitemname(.@b_item) + "^000000";
		mes "의 이번 회차 계정 구매 가능 수량을";
		mes "모두 사용했습니다.";
		next;
		return;
	}

	.@b_max = $NeedPastEggSaleStock[.@b_idx];
	if (.@b_remain < .@b_max)
		.@b_max = .@b_remain;

	mes "[추억의 행운의 알 상인]";
	mes "상품: ^0055FF" + getitemname(.@b_item) + "^000000";
	mes "개당 가격: ^FF8800" + .@b_price + " Cash^000000";
	mes "가격 배율: " + $NeedPastEggSaleRate[.@b_idx] + "%";
	mes "남은 서버 재고: " + $NeedPastEggSaleStock[.@b_idx] + "개";
	mes "내 구매 가능: " + .@b_remain + "개";
	mes " ";
	mes "구매할 수량을 입력해주세요.";
	mes "(1 ~ " + .@b_max + ")";
	next;

	input .@b_amount, 1, .@b_max;

	.@b_total = .@b_price * .@b_amount;

	mes "[추억의 행운의 알 상인]";
	mes "상품: ^0055FF" + getitemname(.@b_item) + "^000000";
	mes "수량: " + .@b_amount + "개";
	mes "필요 Cash: ^FF8800" + .@b_total + "^000000";
	mes " ";
	mes "구매하시겠습니까?";
	next;

	if (select("구매한다:취소") == 2)
		return;

	callsub S_EnsureRotation;
	callsub S_SyncAccountRotation;

	if ($NeedPastEggSaleKey$ != .@b_key$ ||
		$NeedPastEggSaleId[.@b_idx] != .@b_item ||
		$NeedPastEggSalePrice[.@b_idx] != .@b_price) {
		mes "[추억의 행운의 알 상인]";
		mes "^FF0000대화 중 판매 상품이 갱신되었습니다.^000000";
		mes "현재 상품을 다시 확인해주세요.";
		next;
		return;
	}

	.@b_remain = .BuyLimit - #NeedPastEggBought[.@b_idx];
	if (.@b_remain <= 0 || .@b_amount > .@b_remain) {
		mes "[추억의 행운의 알 상인]";
		mes "계정 구매 가능 수량이 부족합니다.";
		next;
		return;
	}

	if ($NeedPastEggSaleStock[.@b_idx] < .@b_amount) {
		mes "[추억의 행운의 알 상인]";
		mes "^FF0000다른 모험가의 구매로 재고가 부족해졌습니다.^000000";
		mes "현재 남은 재고: " + $NeedPastEggSaleStock[.@b_idx] + "개";
		next;
		return;
	}

	.@b_total = .@b_price * .@b_amount;

	if (#CASHPOINTS < .@b_total) {
		mes "[추억의 행운의 알 상인]";
		mes "Cash가 부족합니다.";
		mes "필요 Cash: " + .@b_total;
		mes "보유 Cash: " + #CASHPOINTS;
		next;
		return;
	}

	if (!checkweight(.@b_item, .@b_amount)) {
		mes "[추억의 행운의 알 상인]";
		mes "소지 무게 또는 인벤토리 공간을 확인해주세요.";
		next;
		return;
	}

	#CASHPOINTS -= .@b_total;
	$NeedPastEggSaleStock[.@b_idx] = $NeedPastEggSaleStock[.@b_idx] - .@b_amount;
	callsub S_CheckStockNotice, .@b_idx;
	#NeedPastEggBought[.@b_idx] = #NeedPastEggBought[.@b_idx] + .@b_amount;
	getitem .@b_item, .@b_amount;
	getitem .MemoryFragment, .@b_amount;

	mes "[추억의 행운의 알 상인]";
	mes "^0055FF" + getitemname(.@b_item) + "^000000 " + .@b_amount + "개를 구매했습니다.";
	mes "사용 Cash: ^FF8800" + .@b_total + "^000000";
	mes "남은 Cash: " + #CASHPOINTS;
	mes "남은 서버 재고: " + $NeedPastEggSaleStock[.@b_idx] + "개";
	mes "이번 회차 내 구매량: " + #NeedPastEggBought[.@b_idx] + " / " + .BuyLimit;
	next;
	return;


//============================================================
// 2단계 - 원하는 과거 행운의 알 10개 선택 구매
//============================================================
S_SelectShop:
	.@menu$ = "";
	.@count = 0;

	for (.@i = 0; .@i < getarraysize(.EggId); .@i++) {
		if (!.EggEnabled[.@i])
			continue;

		.@eligible_index[.@count] = .@i;

		if (.@count > 0)
			.@menu$ += ":";

		.@menu$ += getitemname(.EggId[.@i]) + " " + .SelectEggAmount + "개 (" + .SelectCashPrice + " Cash)";
		.@count++;
	}

	if (.@count <= 0) {
		mes "[추억의 행운의 알 상인]";
		mes "선택 구매 가능한 과거 행운의 알이 없습니다.";
		next;
		return;
	}

	mes "[과거 행운의 알 선택 구매]";
	mes "원하는 과거 행운의 알을 선택해주세요.";
	mes " ";
	mes "교환 비용:";
	mes "^FF0000" + getitemname(.MemoryFragment) + " " + .SelectFragmentCost + "개^000000";
	mes "+ ^FF8800" + .SelectCashPrice + " Cash^000000";
	mes " ";
	mes "지급 수량: ^0055FF" + .SelectEggAmount + "개^000000";
	next;

	.@selected = select(.@menu$) - 1;
	.@index = .@eligible_index[.@selected];
	.@item = .EggId[.@index];

	mes "[과거 행운의 알 선택 구매]";
	mes "선택 상품:";
	mes "^0055FF" + getitemname(.@item) + " " + .SelectEggAmount + "개^000000";
	mes " ";
	mes "필요 재료:";
	mes "- " + getitemname(.MemoryFragment) + " ^FF0000" + .SelectFragmentCost + "개^000000";
	mes "- Cash ^FF8800" + .SelectCashPrice + "^000000";
	mes " ";
	mes "현재 보유:";
	mes "- " + getitemname(.MemoryFragment) + " " + countitem(.MemoryFragment) + "개";
	mes "- Cash " + #CASHPOINTS;
	mes " ";
	mes "교환하시겠습니까?";
	next;

	if (select("교환한다:취소") == 2)
		return;

	if (!.EggEnabled[.@index] || .EggId[.@index] != .@item) {
		mes "[추억의 행운의 알 상인]";
		mes "상품 설정이 변경되었습니다.";
		mes "다시 시도해주세요.";
		next;
		return;
	}

	if (countitem(.MemoryFragment) < .SelectFragmentCost) {
		mes "[추억의 행운의 알 상인]";
		mes getitemname(.MemoryFragment) + "이(가) 부족합니다.";
		mes "필요 수량: " + .SelectFragmentCost + "개";
		next;
		return;
	}

	if (#CASHPOINTS < .SelectCashPrice) {
		mes "[추억의 행운의 알 상인]";
		mes "Cash가 부족합니다.";
		mes "필요 Cash: " + .SelectCashPrice;
		mes "보유 Cash: " + #CASHPOINTS;
		next;
		return;
	}

	if (!checkweight(.@item, .SelectEggAmount)) {
		mes "[추억의 행운의 알 상인]";
		mes "소지 무게 또는 인벤토리 공간을 확인해주세요.";
		next;
		return;
	}

	delitem .MemoryFragment, .SelectFragmentCost;
	#CASHPOINTS -= .SelectCashPrice;
	getitem .@item, .SelectEggAmount;

	mes "[추억의 행운의 알 상인]";
	mes "^0055FF" + getitemname(.@item) + "^000000 " + .SelectEggAmount + "개를 교환했습니다.";
	mes " ";
	mes "사용 재료:";
	mes "- " + getitemname(.MemoryFragment) + " " + .SelectFragmentCost + "개";
	mes "- Cash " + .SelectCashPrice;
	mes "남은 Cash: " + #CASHPOINTS;
	next;
	return;


//============================================================
// 현재 12시간 회차 보정
//
// 슬롯이 바뀌었을 때만 다음 그룹으로 전진한다.
// map-server 재시작 / @reloadscript 로 OnInit 이 다시 돌아도
// 슬롯 키가 같으면 판매 목록·가격·재고가 그대로 유지된다.
//
// [주의] callsub 은 .@ 스코프를 공유하지 않는다(script.cpp:5578 에서
// 새 scope 를 새로 할당한다). 슬롯 키를 별도 서브루틴에서 만들어
// .@ 로 돌려받으면 항상 빈 문자열이 되어 매 OnInit 마다 그룹이
// 전진해 버린다. 그래서 키 계산은 여기서 직접 한다.
// sprintf 는 서식 사이의 구분자가 사라지는 문제가 있어 직접 조립한다.
//============================================================
S_EnsureRotation:
	.@ens_slot = 0;
	if (gettime(3) >= 12)
		.@ens_slot = 1;

	.@ens_key$ = "" + gettime(7) + "-" + gettime(6) + "-" + gettime(5)
		+ "-" + .@ens_slot;

	if ($NeedPastEggRotation$ != .@ens_key$) {
		callsub S_Rotate, 1;
		return;
	}

	// 슬롯은 같은데 판매 목록이 비어 있는 경우(최초 도입 등)만 재구성.
	// 그룹은 전진시키지 않는다.
	if (getarraysize($NeedPastEggSaleId) <= 0)
		callsub S_Rotate, 0;

	return;


//============================================================
// 회차 갱신
//
// getarg(0) : 1 = 다음 그룹으로 전진 / 0 = 현재 그룹 그대로 재구성
//
// 활성 풀이 바뀌었거나(수 변화 / 구성 변화) 3그룹을 모두 소진했으면
// 전체를 다시 셔플하고 0번 그룹부터 시작한다.
//============================================================
S_Rotate:
	.@rot_adv = getarg(0);

	.@rot_slot = 0;
	if (gettime(3) >= 12)
		.@rot_slot = 1;

	.@rot_key$ = "" + gettime(7) + "-" + gettime(6) + "-" + gettime(5)
		+ "-" + .@rot_slot;

	// --- 현재 활성 행알 인덱스 수집 ---
	.@rot_total = getarraysize(.EggId);
	.@rot_n = 0;
	deletearray .@rot_act[0];
	for (.@rot_i = 0; .@rot_i < .@rot_total; .@rot_i++) {
		if (!.EggEnabled[.@rot_i])
			continue;
		.@rot_act[.@rot_n] = .@rot_i;
		.@rot_n++;
	}

	deletearray $NeedPastEggSaleIdx[0];
	deletearray $NeedPastEggSaleId[0];
	deletearray $NeedPastEggSaleRate[0];
	deletearray $NeedPastEggSalePrice[0];
	deletearray $NeedPastEggSaleStock[0];
	deletearray $NeedPastEggSaleInit[0];
	deletearray $NeedPastEggSaleN90[0];
	deletearray $NeedPastEggSaleOut[0];

	$NeedPastEggRotation$ = .@rot_key$;
	$NeedPastEggSerial = $NeedPastEggSerial + 1;
	$NeedPastEggSaleKey$ = .@rot_key$ + "#" + $NeedPastEggSerial;

	if (.@rot_n <= 0) {
		deletearray $NeedPastEggOrder[0];
		$NeedPastEggGroup = 0;
		return;
	}

	// --- 저장된 셔플이 현재 활성 풀과 같은지 검사 ---
	//
	// 저장된 값은 서로 다른 인덱스들이므로,
	//   (개수가 같다) + (전부 현재 활성이다)  =>  같은 집합
	// 이 성립한다. 인덱스 조회라서 O(n) 으로 끝난다.
	.@rot_new = 0;

	if (getarraysize($NeedPastEggOrder) != .@rot_n)
		.@rot_new = 1;

	if ($NeedPastEggListSig != .ListSig)
		.@rot_new = 1;

	if ($NeedPastEggGroup < 0 || $NeedPastEggGroup >= .GroupCount)
		.@rot_new = 1;

	if (!.@rot_new) {
		for (.@rot_i = 0; .@rot_i < .@rot_n; .@rot_i++) {
			.@rot_x = $NeedPastEggOrder[.@rot_i];
			if (.@rot_x < 0 || .@rot_x >= .@rot_total || !.EggEnabled[.@rot_x]) {
				.@rot_new = 1;
				break;
			}
		}
	}

	// --- 다음 그룹 결정 ---
	if (.@rot_new) {
		.@rot_next = 0;
	}
	else if (.@rot_adv) {
		.@rot_next = $NeedPastEggGroup + 1;
		if (.@rot_next >= .GroupCount) {
			.@rot_new = 1;
			.@rot_next = 0;
		}
	}
	else {
		.@rot_next = $NeedPastEggGroup;
	}

	// --- 전체 재셔플 (Fisher-Yates) ---
	if (.@rot_new) {
		deletearray $NeedPastEggOrder[0];

		for (.@rot_i = 0; .@rot_i < .@rot_n; .@rot_i++)
			$NeedPastEggOrder[.@rot_i] = .@rot_act[.@rot_i];

		for (.@rot_i = .@rot_n - 1; .@rot_i > 0; .@rot_i--) {
			.@rot_j = rand(.@rot_i + 1);
			.@rot_tmp = $NeedPastEggOrder[.@rot_i];
			$NeedPastEggOrder[.@rot_i] = $NeedPastEggOrder[.@rot_j];
			$NeedPastEggOrder[.@rot_j] = .@rot_tmp;
		}

		$NeedPastEggListSig = .ListSig;
		$NeedPastEggCycle = $NeedPastEggCycle + 1;
	}

	$NeedPastEggGroup = .@rot_next;

	// --- 그룹 구간 계산 (최대한 균등 분배, 앞 그룹부터 나머지 배분) ---
	.@rot_base = .@rot_n / .GroupCount;
	.@rot_rem = .@rot_n % .GroupCount;
	.@rot_g = $NeedPastEggGroup;

	.@rot_start = .@rot_g * .@rot_base + ((.@rot_g < .@rot_rem) ? .@rot_g : .@rot_rem);
	.@rot_size = .@rot_base + ((.@rot_g < .@rot_rem) ? 1 : 0);

	// --- 이번 회차 판매 목록 생성 ---
	.@rot_k = 0;
	.@rot_end = .@rot_start + .@rot_size;
	for (.@rot_i = .@rot_start; .@rot_i < .@rot_end; .@rot_i++) {
		.@rot_x = $NeedPastEggOrder[.@rot_i];
		.@rot_id = .EggId[.@rot_x];
		if (.@rot_id <= 0)
			continue;

		$NeedPastEggSaleIdx[.@rot_k] = .@rot_x;
		$NeedPastEggSaleId[.@rot_k] = .@rot_id;
		$NeedPastEggSaleRate[.@rot_k] = 110 + (rand(5) * 10);
		$NeedPastEggSalePrice[.@rot_k] = (.BaseCashPrice * $NeedPastEggSaleRate[.@rot_k]) / 100;
		$NeedPastEggSaleStock[.@rot_k] = rand(.StockMin, .StockMax);
		$NeedPastEggSaleInit[.@rot_k] = $NeedPastEggSaleStock[.@rot_k];
		$NeedPastEggSaleN90[.@rot_k] = 0;
		$NeedPastEggSaleOut[.@rot_k] = 0;

		.@rot_k++;
	}

	return;


//============================================================

S_SyncAccountRotation:
	if (#NeedPastEggRotation$ != $NeedPastEggSaleKey$) {
		#NeedPastEggRotation$ = $NeedPastEggSaleKey$;
		deletearray #NeedPastEggBought[0];
	}
	return;

S_CheckStockNotice:
	.@n_idx = getarg(0);

	// 품절
	if ($NeedPastEggSaleStock[.@n_idx] <= 0) {
		$NeedPastEggSaleStock[.@n_idx] = 0;

		if (!$NeedPastEggSaleOut[.@n_idx]) {
			$NeedPastEggSaleOut[.@n_idx] = 1;

			announce "[행운의 알] "
				+ getitemname($NeedPastEggSaleId[.@n_idx])
				+ "의 준비된 수량이 모두 판매되었습니다!", bc_all;
		}

		return;
	}

	// 약 90% 판매 완료 = 최초 재고의 10% 이하 남음
	if (!$NeedPastEggSaleN90[.@n_idx] &&
		$NeedPastEggSaleInit[.@n_idx] > 0 &&
		$NeedPastEggSaleStock[.@n_idx] * 10 <= $NeedPastEggSaleInit[.@n_idx]) {

		$NeedPastEggSaleN90[.@n_idx] = 1;

		announce "[행운의 알] "
			+ getitemname($NeedPastEggSaleId[.@n_idx])
			+ "의 준비된 수량이 얼마 남지 않았습니다! 남은 수량: "
			+ $NeedPastEggSaleStock[.@n_idx] + "개", bc_all;
	}

	return;
