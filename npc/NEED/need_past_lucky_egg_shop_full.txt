//============================================================
// NEED - 과거 행운의 알 상점
//
// 1단계
// - 매일 00:00 / 12:00 갱신
// - 활성화된 과거 행운의 알 중 1종 랜덤 판매
// - 기본가 1,000 Cash
// - 랜덤 판매가: 110 / 120 / 130 / 140 / 150%
// - 서버 재고: 300~500개
// - 계정당 회차 구매 제한: 10개
//
// 2단계
// - 추억의 행운 조각(900106) 30개
// - 11,000 Cash
// - 원하는 활성화 과거 행운의 알 10개 선택 구매
//
// 활성화 정책
// - 12855 교황 스크롤까지 활성화
// - 이후 품목은 모두 비활성 상태로 미리 등록
// - 시즌 개방 시 .EggEnabled[]의 해당 항목만 0 -> 1 변경
//
// 갱신 10분 전 전체 공지
// - 11:50 / 23:50
//============================================================

lasagna,169,195,4	script	추억의 행운의 알 상인#need_past_egg	4_F_KAFRA6,{

	callsub S_EnsureRotation;
	callsub S_SyncAccountRotation;

	while (1) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes needtr("지나간 시즌의 행운의 알을", "This is a special shop where you can meet");
		mes needtr("다시 만날 수 있는 특별 판매소입니다.", "Lucky Eggs from past seasons again.");
		mes " ";

		if ($NeedPastEggMarketItem > 0) {
			mes needtr("현재 한정 판매:", "Current limited sale:");
			mes "^0055FF" + getitemname($NeedPastEggMarketItem) + "^000000";
			mes needtr("판매가: ^FF8800", "Price: ^FF8800") + $NeedPastEggMarketUnitPrice + " Cash^000000";
			mes needtr("가격 배율: ", "Price rate: ") + $NeedPastEggMarketRate + "%";
			mes needtr("남은 서버 재고: ", "Server stock left: ") + $NeedPastEggMarketStock + needtr("개", "");
			mes needtr("내 구매량: ", "My purchases: ") + #NeedPastEggBought + " / " + .BuyLimit + needtr("개", "");
		}
		else {
			mes needtr("^FF0000현재 판매 가능한 행운의 알이 없습니다.^000000", "^FF0000No Lucky Egg is on sale right now.^000000");
		}
		setdialogsize(300,300);
		next;

		switch (select(needtr("랜덤 한정 판매 이용:과거 행운의 알 선택 구매:시스템 설명:닫기", "Random Limited Sale:Buy a Specific Past Egg:How It Works:Close"))) {
		case 1:
			callsub S_RandomShop;
			break;

		case 2:
			callsub S_SelectShop;
			break;

		case 3:
			mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
			mes needtr("랜덤 한정 판매는 매일", "The random limited sale refreshes every day");
			mes needtr("^0055FF00:00 / 12:00^000000에 갱신됩니다.", "at ^0055FF00:00 / 12:00^000000.");
			mes " ";
			mes needtr("상품, 판매 가격, 서버 재고가", "The product, the price and the server stock");
			mes needtr("갱신 시마다 새롭게 결정됩니다.", "are all rerolled at every refresh.");
			next;

			mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
			mes needtr("한정 판매 상품은 계정당", "The limited sale is limited per account");
			mes needtr("회차별 최대 ", "to at most ") + .BuyLimit + needtr("개까지 구매할 수 있습니다.", " per round.");
			mes " ";
			mes needtr("원하는 과거 행운의 알을 직접 구매할 경우", "To buy a specific past Lucky Egg you need");
			mes "^FF0000" + getitemname(.MemoryFragment) + " " + .SelectFragmentCost + needtr("개^000000", "^000000");
			mes "+ ^FF8800" + .SelectCashPrice + needtr(" Cash^000000가 필요합니다.", " Cash^000000.");
			mes needtr("선택한 행운의 알 ", "You then receive ") + .SelectEggAmount + needtr("개를 받을 수 있습니다.", " of the egg you chose.");
			next;
			break;

		default:
			close;
		}
	}


//============================================================
// 1단계 - 랜덤 한정 판매
//============================================================
S_RandomShop:
	callsub S_EnsureRotation;
	callsub S_SyncAccountRotation;

	if ($NeedPastEggMarketItem <= 0 || $NeedPastEggMarketUnitPrice <= 0) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes needtr("현재 판매 가능한 상품이 없습니다.", "There is no product on sale right now.");
		next;
		return;
	}

	if ($NeedPastEggMarketStock <= 0) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes needtr("^FF0000현재 회차의 상품이 모두 판매되었습니다.^000000", "^FF0000This round's stock is sold out.^000000");
		mes needtr("다음 갱신을 기다려주세요.", "Please wait for the next refresh.");
		next;
		return;
	}

	.@remaining = .BuyLimit - #NeedPastEggBought;
	if (.@remaining <= 0) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes needtr("이번 회차의 계정 구매 가능 수량을", "You have used up your account's purchases");
		mes needtr("모두 사용했습니다.", "for this round.");
		next;
		return;
	}

	.@max = $NeedPastEggMarketStock;
	if (.@remaining < .@max)
		.@max = .@remaining;

	mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
	mes needtr("상품: ^0055FF", "Product: ^0055FF") + getitemname($NeedPastEggMarketItem) + "^000000";
	mes needtr("개당 가격: ^FF8800", "Price each: ^FF8800") + $NeedPastEggMarketUnitPrice + " Cash^000000";
	mes needtr("남은 서버 재고: ", "Server stock left: ") + $NeedPastEggMarketStock + needtr("개", "");
	mes needtr("내 구매 가능: ", "I can still buy: ") + .@remaining + needtr("개", "");
	mes " ";
	mes needtr("구매할 수량을 입력해주세요.", "Please enter how many you want.");
	mes "(1 ~ " + .@max + ")";
	next;

	input .@amount, 1, .@max;

	.@rotation$ = $NeedPastEggRotation$;
	.@item = $NeedPastEggMarketItem;
	.@unit_price = $NeedPastEggMarketUnitPrice;
	.@total = .@unit_price * .@amount;

	mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
	mes needtr("상품: ^0055FF", "Product: ^0055FF") + getitemname(.@item) + "^000000";
	mes needtr("수량: ", "Amount: ") + .@amount + needtr("개", "");
	mes needtr("필요 Cash: ^FF8800", "Cash needed: ^FF8800") + .@total + "^000000";
	mes " ";
	mes needtr("구매하시겠습니까?", "Would you like to purchase");
	next;

	if (select(needtr("구매한다:취소", "Buy:Cancel")) == 2)
		return;

	callsub S_EnsureRotation;
	callsub S_SyncAccountRotation;

	if ($NeedPastEggRotation$ != .@rotation$ ||
		$NeedPastEggMarketItem != .@item ||
		$NeedPastEggMarketUnitPrice != .@unit_price) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes needtr("^FF0000대화 중 판매 상품이 갱신되었습니다.^000000", "^FF0000The sale refreshed while we were talking.^000000");
		mes needtr("현재 상품을 다시 확인해주세요.", "Please check the current product again.");
		next;
		return;
	}

	.@remaining = .BuyLimit - #NeedPastEggBought;
	if (.@remaining <= 0 || .@amount > .@remaining) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes needtr("계정 구매 가능 수량이 부족합니다.", "Your account does not have enough purchases left.");
		next;
		return;
	}

	if ($NeedPastEggMarketStock < .@amount) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes needtr("^FF0000다른 모험가의 구매로 재고가 부족해졌습니다.^000000", "^FF0000Another adventurer bought some and the stock ran short.^000000");
		mes needtr("현재 남은 재고: ", "Stock left now: ") + $NeedPastEggMarketStock + needtr("개", "");
		next;
		return;
	}

	.@total = .@unit_price * .@amount;

	if (#CASHPOINTS < .@total) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes needtr("Cash가 부족합니다.", "You do not have enough Cash.");
		mes needtr("필요 Cash: ", "Cash needed: ") + .@total;
		mes needtr("보유 Cash: ", "Cash held: ") + #CASHPOINTS;
		next;
		return;
	}

	if (!checkweight(.@item, .@amount)) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes needtr("소지 무게 또는 인벤토리 공간을 확인해주세요.", "Please check your weight and inventory space.");
		next;
		return;
	}

	#CASHPOINTS -= .@total;
	$NeedPastEggMarketStock -= .@amount;
	callsub S_CheckStockNotice;
	#NeedPastEggBought += .@amount;
	getitem .@item, .@amount;
	getitem .MemoryFragment, .@amount;

	mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
	mes "^0055FF" + getitemname(.@item) + "^000000 " + .@amount + needtr("개를 구매했습니다.", " purchased.");
	mes needtr("사용 Cash: ^FF8800", "Cash spent: ^FF8800") + .@total + "^000000";
	mes needtr("남은 Cash: ", "Cash left: ") + #CASHPOINTS;
	mes needtr("남은 서버 재고: ", "Server stock left: ") + $NeedPastEggMarketStock + needtr("개", "");
	mes needtr("이번 회차 내 구매량: ", "My purchases this round: ") + #NeedPastEggBought + " / " + .BuyLimit;
	next;
	return;


//============================================================
// 2단계 - 원하는 과거 행운의 알 10개 선택 구매
//============================================================
S_SelectShop:
	.@menu$ = "";
	.@count = 0;

	for (.@i = 0; .@i < getarraysize(.EggId); .@i++) {
		if (!.EggEnabled[.@i])
			continue;

		.@eligible_index[.@count] = .@i;

		if (.@count > 0)
			.@menu$ += ":";

		.@menu$ += getitemname(.EggId[.@i]) + " " + .SelectEggAmount + needtr("개 (", " (") + .SelectCashPrice + " Cash)";
		.@count++;
	}

	if (.@count <= 0) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes needtr("선택 구매 가능한 과거 행운의 알이 없습니다.", "There is no past Lucky Egg available for direct purchase.");
		next;
		return;
	}

	mes needtr("[과거 행운의 알 선택 구매]", "[Buy a Specific Past Egg]");
	mes needtr("원하는 과거 행운의 알을 선택해주세요.", "Please choose the past Lucky Egg you want.");
	mes " ";
	mes needtr("교환 비용:", "Exchange cost:");
	mes "^FF0000" + getitemname(.MemoryFragment) + " " + .SelectFragmentCost + needtr("개^000000", "^000000");
	mes "+ ^FF8800" + .SelectCashPrice + " Cash^000000";
	mes " ";
	mes needtr("지급 수량: ^0055FF", "You receive: ^0055FF") + .SelectEggAmount + needtr("개^000000", "^000000");
	next;

	.@selected = select(.@menu$) - 1;
	.@index = .@eligible_index[.@selected];
	.@item = .EggId[.@index];

	mes needtr("[과거 행운의 알 선택 구매]", "[Buy a Specific Past Egg]");
	mes needtr("선택 상품:", "Chosen product:");
	mes "^0055FF" + getitemname(.@item) + " " + .SelectEggAmount + needtr("개^000000", "^000000");
	mes " ";
	mes needtr("필요 재료:", "Materials needed:");
	mes "- " + getitemname(.MemoryFragment) + " ^FF0000" + .SelectFragmentCost + needtr("개^000000", "^000000");
	mes "- Cash ^FF8800" + .SelectCashPrice + "^000000";
	mes " ";
	mes needtr("현재 보유:", "You have:");
	mes "- " + getitemname(.MemoryFragment) + " " + countitem(.MemoryFragment) + needtr("개", "");
	mes "- Cash " + #CASHPOINTS;
	mes " ";
	mes needtr("교환하시겠습니까?", "Do you want to exchange?");
	next;

	if (select(needtr("교환한다:취소", "Exchange:Cancel")) == 2)
		return;

	if (!.EggEnabled[.@index] || .EggId[.@index] != .@item) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes needtr("상품 설정이 변경되었습니다.", "The product setup has changed.");
		mes needtr("다시 시도해주세요.", "Please try again.");
		next;
		return;
	}

	if (countitem(.MemoryFragment) < .SelectFragmentCost) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes getitemname(.MemoryFragment) + needtr("이(가) 부족합니다.", " is missing.");
		mes needtr("필요 수량: ", "Amount needed: ") + .SelectFragmentCost + needtr("개", "");
		next;
		return;
	}

	if (#CASHPOINTS < .SelectCashPrice) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes needtr("Cash가 부족합니다.", "You do not have enough Cash.");
		mes needtr("필요 Cash: ", "Cash needed: ") + .SelectCashPrice;
		mes needtr("보유 Cash: ", "Cash held: ") + #CASHPOINTS;
		next;
		return;
	}

	if (!checkweight(.@item, .SelectEggAmount)) {
		mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
		mes needtr("소지 무게 또는 인벤토리 공간을 확인해주세요.", "Please check your weight and inventory space.");
		next;
		return;
	}

	delitem .MemoryFragment, .SelectFragmentCost;
	#CASHPOINTS -= .SelectCashPrice;
	getitem .@item, .SelectEggAmount;

	mes needtr("[추억의 행운의 알 상인]", "[Memory Lucky Egg Seller]");
	mes "^0055FF" + getitemname(.@item) + "^000000 " + .SelectEggAmount + needtr("개를 교환했습니다.", " exchanged.");
	mes " ";
	mes needtr("사용 재료:", "Materials used:");
	mes "- " + getitemname(.MemoryFragment) + " " + .SelectFragmentCost + needtr("개", "");
	mes "- Cash " + .SelectCashPrice;
	mes needtr("남은 Cash: ", "Cash left: ") + #CASHPOINTS;
	next;
	return;


//============================================================
// 현재 12시간 회차 보정
//============================================================
S_EnsureRotation:
	.@slot = 0;
	if (gettime(3) >= 12)
		.@slot = 1;

	.@key$ = sprintf("%04d%02d%02d-%d",
		gettime(7), gettime(6), gettime(5), .@slot);

	if ($NeedPastEggRotation$ != .@key$ || $NeedPastEggMarketItem <= 0)
		callsub S_Rotate;

	return;


//============================================================
// 랜덤 회차 생성
//============================================================
S_Rotate:
	.@slot = 0;
	if (gettime(3) >= 12)
		.@slot = 1;

	.@key$ = sprintf("%04d%02d%02d-%d",
		gettime(7), gettime(6), gettime(5), .@slot);

	.@count = 0;

	for (.@i = 0; .@i < getarraysize(.EggId); .@i++) {
		if (!.EggEnabled[.@i])
			continue;

		.@eligible_index[.@count] = .@i;
		.@count++;
	}

	if (.@count <= 0) {
		$NeedPastEggRotation$ = .@key$;
		$NeedPastEggMarketItem = 0;
		$NeedPastEggMarketRate = 0;
		$NeedPastEggMarketUnitPrice = 0;
		$NeedPastEggMarketStock = 0;
		return;
	}

	.@pick = .@eligible_index[rand(.@count)];

	$NeedPastEggRotation$ = .@key$;
	$NeedPastEggMarketItem = .EggId[.@pick];

	$NeedPastEggMarketRate = 110 + (rand(5) * 10);
	$NeedPastEggMarketUnitPrice = (.BaseCashPrice * $NeedPastEggMarketRate) / 100;

	$NeedPastEggMarketStock = rand(.StockMin, .StockMax);
	
	// 판매율 계산용 최초 재고 저장
	$NeedPastEggMarketInitialStock = $NeedPastEggMarketStock;
	
	// 회차별 알림 플래그 초기화
	$NeedPastEggStock90Notice = 0;
	$NeedPastEggSoldOutNotice = 0;

	return;


//============================================================
// 계정 회차 구매량 동기화
//============================================================
S_SyncAccountRotation:
	if (#NeedPastEggRotation$ != $NeedPastEggRotation$) {
		#NeedPastEggRotation$ = $NeedPastEggRotation$;
		#NeedPastEggBought = 0;
	}
	return;


//============================================================
// 갱신 10분 전 전체 공지
//============================================================
OnClock1150:
	announce "[행운의 알] 10분 후 과거 행운의 알이 새롭게 입고됩니다! 구매처 : 마을 중앙 12시", bc_all;
	end;

OnClock2350:
	announce "[행운의 알] 10분 후 과거 행운의 알이 새롭게 입고됩니다! 구매처 : 마을 중앙 12시", bc_all;
	end;


//============================================================
// 자동 갱신
//============================================================
OnClock0000:
	announce "[행운의 알] 과거 행운의 알 한정 판매가 시작되었습니다! 추억의 행운의 알 상인을 확인해주세요. 구매처 : 마을 중앙 12시", bc_all;
	callsub S_Rotate;
	end;

OnClock1200:
	announce "[행운의 알] 과거 행운의 알 한정 판매가 시작되었습니다! 추억의 행운의 알 상인을 확인해주세요. 구매처 : 마을 중앙 12시", bc_all;
	callsub S_Rotate;
	end;

//============================================================
// 판매 재고 알림
//============================================================
S_CheckStockNotice:

	// 품절
	if ($NeedPastEggMarketStock <= 0) {
		$NeedPastEggMarketStock = 0;

		if (!$NeedPastEggSoldOutNotice) {
			$NeedPastEggSoldOutNotice = 1;

			announce "[행운의 알] "
				+ getitemname($NeedPastEggMarketItem)
				+ "의 준비된 수량이 모두 판매되었습니다!", bc_all;
		}

		return;
	}

	// 약 90% 판매 완료 = 최초 재고의 10% 이하 남음
	if (!$NeedPastEggStock90Notice &&
		$NeedPastEggMarketInitialStock > 0 &&
		$NeedPastEggMarketStock * 10 <= $NeedPastEggMarketInitialStock) {

		$NeedPastEggStock90Notice = 1;

		announce "[행운의 알] "
			+ getitemname($NeedPastEggMarketItem)
			+ "의 준비된 수량이 얼마 남지 않았습니다! 남은 수량: "
			+ $NeedPastEggMarketStock + "개", bc_all;
	}

	return;

//============================================================
// 설정
//============================================================
OnInit:
	.MemoryFragment = 900106;		// 추억의 행운 조각
	.BaseCashPrice = 1000;			// 행운의 알 1개 기존 기본가
	.StockMin = 300;				// 회차 최소 서버 재고
	.StockMax = 500;				// 회차 최대 서버 재고
	.BuyLimit = 10;					// 계정당 회차 구매 제한

	.SelectFragmentCost = 30;		// 선택 교환 필요 조각
	.SelectEggAmount = 10;			// 선택 교환 지급 수량
	.SelectCashRate = 110;			// 10개 기본가의 110%
	.SelectCashPrice = (.BaseCashPrice * .SelectEggAmount * .SelectCashRate) / 100; // 11,000 Cash

	// ---------------------------------------------------------
	// 과거 행운의 알 전체 목록
	// EggEnabled: 1=판매/선택 가능, 0=비활성
	// 12855 교황 스크롤까지 활성 / 이후 전부 비활성
	// ---------------------------------------------------------
	setarray .EggId[0],
		12654,	// 행운의 알Ⅸ
		12673,	// 행운의 알 X
		12685,	// 그리폰의 알
