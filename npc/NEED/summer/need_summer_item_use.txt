//------------------------------------------------------------
// 1억 제니 주머니
//------------------------------------------------------------
function	script	F_NEED_Summer_ZenyPouch	{
	.@item_id = 399929;
	.@amount = 100000000;

	if (countitem(.@item_id) < 1)
		return 0;

	if (Zeny > MAX_ZENY - .@amount) {
		dispbottom needtr("[여름 이벤트] 제니 보유 한도를 초과하여 사용할 수 없습니다.", "[Summer Event] Cannot be used - it would exceed your Zeny holding limit.");
		return 0;
	}

	delitem .@item_id,1;
	.@before = Zeny;
	Zeny += .@amount;
	.@granted = Zeny - .@before;
	if ( .@granted > 0 ) callfunc "F_NEED_Summer_RewardLog", .@item_id, 1, .@granted;

	dispbottom needtr("[여름 이벤트] 제니 1억을 획득했습니다.", "[Summer Event] You obtained 100,000,000 Zeny.");
	return 1;
}


//------------------------------------------------------------
// 캐시 10만 포인트 교환권
//------------------------------------------------------------
function	script	F_NEED_Summer_CashTicket	{
	.@item_id = 399930;
	.@amount = 100000;
	.@max_cash = 2147483647;

	if (countitem(.@item_id) < 1)
		return 0;

	if (#CASHPOINTS > .@max_cash - .@amount) {
		dispbottom needtr("[여름 이벤트] 캐시 포인트 보유 한도를 초과하여 사용할 수 없습니다.", "[Summer Event] Cannot be used - it would exceed your Cash Point holding limit.");
		return 0;
	}

	delitem .@item_id,1;
	.@before = #CASHPOINTS;
	#CASHPOINTS += .@amount;
	.@granted = #CASHPOINTS - .@before;
	if ( .@granted > 0 ) callfunc "F_NEED_Summer_RewardLog", .@item_id, 2, .@granted;

	dispbottom needtr("[여름 이벤트] 캐시 포인트 10만을 획득했습니다.", "[Summer Event] You obtained 100,000 Cash Points.");
	return 1;
}
