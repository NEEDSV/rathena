lasagna,175,131,4	script	환생 관리자	811,{
	mes "[환생 관리자]";
	mes "환생을 통해 특별한 보상을 획득할 수 있습니다.";
	mes "환생은 장기 성장과 수집을 위한 콘텐츠입니다.";
	setdialogsize(300,200);

	// 업데이트 이전 환생 횟수에 대한 포인트를 최초 1회 자동 소급 지급
	callsub S_ApplyRetroactivePoints;

	// 주간 상점 구매 횟수 갱신
	callsub S_ResetRebirthShopWeek;
	next;

	while (true) {
		callsub S_ResetRebirthShopWeek;

		.@count = getd(.CountVar$);
		.@point = getd(.PointVar$);

		mes "[환생 관리자]";
		mes "현재 환생 횟수: ^0055FF" + .@count + "회^000000";
		mes "보유 환생 포인트: ^00AA00" + .@point + "점^000000";
		mes "환생 조건: Base Lv." + .NeedBaseLv + " / Job Lv." + .NeedJobLv;
		mes "소모 아이템: " + callsub(S_ItemName, .TicketItemID) + " " + .TicketAmount + "개";
		next;
		
		.@menu = select(
			"환생하기",
			"환생 상점",
			"환생 보상 보기",
			"미수령 구간 보상 확인",
			"의상 미리보기",
			"닫기"
		);

		switch (.@menu) {
			case 1:
				callsub S_DoRebirth;
				close;

			case 2:
				callsub S_OpenRebirthShop;
				break;

			case 3:
				callsub S_ShowRewardInfo;
				next;
				break;

			case 4:
				callsub S_GiveUnclaimedMiles;
				close;

			case 5:
				mes "[환생 관리자]";
				mes "환생 의상 미리보기는 우측 NPC를 확인해주세요.";
				close;

			case 6:
				close;
		}
	}

S_DoRebirth:
	.@count = getd(.CountVar$);

	if (BaseLevel < .NeedBaseLv) {
		mes "[환생 관리자]";
		mes "Base Lv." + .NeedBaseLv + " 이상부터 환생할 수 있습니다.";
		mes "현재 Base Lv.: " + BaseLevel;
		return;
	}

	if (JobLevel < .NeedJobLv) {
		mes "[환생 관리자]";
		mes "Job Lv." + .NeedJobLv + " 이상부터 환생할 수 있습니다.";
		mes "현재 Job Lv.: " + JobLevel;
		return;
	}

	if (.TicketItemID <= 0) {
		mes "[환생 관리자]";
		mes "^FF0000환생 티켓 아이템 ID가 설정되지 않았습니다.^000000";
		mes "NPC OnInit의 .TicketItemID 값을 설정해주세요.";
		return;
	}

	if (countitem(.TicketItemID) < .TicketAmount) {
		mes "[환생 관리자]";
		mes "환생에는 " + callsub(S_ItemName, .TicketItemID) + " " + .TicketAmount + "개가 필요합니다.";
		return;
	}

	mes "[환생 관리자]";
	mes "환생을 진행하시겠습니까?";
	mes "환생 시 환생 티켓이 소모되고, 환생 횟수가 1회 증가합니다.";
	mes " ";
	mes "현재 환생 횟수: ^0055FF" + .@count + "회^000000";
	mes "환생 후 횟수: ^0055FF" + (.@count + 1) + "회^000000";
	next;

	if (select("환생한다", "취소") == 2) {
		mes "[환생 관리자]";
		mes "환생을 취소했습니다.";
		return;
	}

	delitem .TicketItemID, .TicketAmount;

	.@new_count = .@count + 1;
	setd .CountVar$, .@new_count;

	// 환생 1회당 캐릭터 단위 환생 포인트 지급
	if (.PointPerRebirth > 0) {
		setd .PointVar$, getd(.PointVar$) + .PointPerRebirth;
	}

	mes "[환생 관리자]";
	mes "환생이 완료되었습니다.";
	mes "현재 환생 횟수: ^0055FF" + .@new_count + "회^000000";

	if (.PointPerRebirth > 0) {
		mes "환생 포인트: ^00AA00+" + .PointPerRebirth + "점^000000";
		mes "보유 포인트: ^00AA00" + getd(.PointVar$) + "점^000000";
	}
		if (.@new_count == .Cnt_EffectStone ||
		.@new_count == .Cnt_UpperCostume ||
		.@new_count == .Cnt_FootprintStone ||
		.@new_count == .Cnt_MiddleCostume ||
		.@new_count == .Cnt_EffectStoneTicket ||
		.@new_count == .Cnt_LowerCostume ||
		.@new_count == .Cnt_FootprintTicket ||
		.@new_count == .Cnt_GarmentCostume ||
		.@new_count == .Cnt_Aura ||
		.@new_count == .Cnt_Title ||
		.@new_count == .Cnt_FinalEffect) {

		announce .AnnouncePrefix$ + " 축하합니다! " + strcharinfo(0) + "님이 " + .@new_count + "회 환생 구간 보상을 달성했습니다!", bc_all;
	} else {
		announce .AnnouncePrefix$ + " " + strcharinfo(0) + "님이 " + .@new_count + "회 환생을 달성했습니다!", bc_all;
	}
	next;

	callsub S_GiveEveryReward, .@new_count;
	callsub S_GiveEveryNReward, .@new_count;
	callsub S_GiveMilestoneByCount, .@new_count;

	if (.UseLevelReset) {
		callfunc "F_NEED_Rebirth_ResetCharacter";
	}

	return;

S_ApplyRetroactivePoints:
	// 캐릭터별 최초 1회만 기존 환생 횟수를 기준으로 소급 지급
	if (PointRetroactiveDoneVar)
		return;

	.@count = getd(.CountVar$);
	.@retroactive_point = .@count * .RetroactivePointPerRebirth;

	if (.@retroactive_point > 0) {
		setd .PointVar$, getd(.PointVar$) + .@retroactive_point;
		next;
		mes " ";
		mes "^00AA00[환생 포인트 소급 지급]^000000";
		mes "기존 환생 " + .@count + "회를 기준으로";
		mes "환생 포인트 " + .@retroactive_point + "점이 지급되었습니다.";
	}

	// 환생 횟수가 0이어도 최초 확인 완료 처리하여 이후 중복 소급 방지
	PointRetroactiveDoneVar = 1;
	return;

S_ResetRebirthShopWeek:
	// 서버 시간 기준 매주 월요일 00:00에 캐릭터별 구매 횟수 초기화
	// gettime(4): 요일(0=일요일), gettime(3/2/1): 시/분/초
	.@day_of_week = gettime(4);
	.@days_from_monday = (.@day_of_week + 6) % 7;
	.@week_start = gettimetick(2)
		- (.@days_from_monday * 86400)
		- (gettime(3) * 3600)
		- (gettime(2) * 60)
		- gettime(1);

	if (getd(.ShopWeekVar$) != .@week_start) {
		setd .ShopWeekVar$, .@week_start;
		setd .ShopBlessingCountVar$, 0;
	}

	return;

S_OpenRebirthShop:
	while (true) {
		callsub S_ResetRebirthShopWeek;

		.@point = getd(.PointVar$);
		.@used = getd(.ShopBlessingCountVar$);
		.@remaining = .Shop_Blessing_WeeklyLimit - .@used;

		if (.@remaining < 0)
			.@remaining = 0;

		mes "[환생 상점]";
		mes "보유 환생 포인트: ^00AA00" + .@point + "점^000000";
		mes "구매할 상품을 선택해주세요.";
		next;

		.@shop_menu = select(
			"대장장이의 축복 (주 " + .@remaining + "회 남음)",
			"뒤로"
		);

		switch (.@shop_menu) {
			case 1:
				callsub S_BuyRebirthShopBlessing;
				break;

			case 2:
				return;
		}
	}

S_BuyRebirthShopBlessing:
	callsub S_ResetRebirthShopWeek;

	.@point = getd(.PointVar$);
	.@used = getd(.ShopBlessingCountVar$);
	.@remaining = .Shop_Blessing_WeeklyLimit - .@used;

	if (.@remaining < 0)
		.@remaining = 0;

	mes "[환생 상점]";
	mes "판매 상품: " + callsub(S_ItemName, .Shop_Blessing_ItemID);
	mes " ";
	mes "필요 환생 포인트: ^00AA00" + .Shop_Blessing_PointCost + "점^000000";
	mes "필요 제니: ^FF8800" + callfunc("F_InsertComma", .Shop_Blessing_ZenyCost) + " Zeny^000000";
	mes " ";
	mes "보유 환생 포인트: " + .@point + "점";
	mes "이번 주 구매 가능: " + .@remaining + "회";
	next;

	if (select("구매한다", "상품 목록으로") == 2)
		return;

	if (.Shop_Blessing_ItemID <= 0) {
		mes "[환생 상점]";
		mes "^FF0000대장장이의 축복 아이템 ID가 설정되지 않았습니다.^000000";
		next;
		return;
	}

	if (.@remaining <= 0) {
		mes "[환생 상점]";
		mes "이번 주 구매 가능 횟수를 모두 사용했습니다.";
		next;
		return;
	}

	if (.Shop_Blessing_PointCost < 0 || .Shop_Blessing_ZenyCost < 0) {
		mes "[환생 상점]";
		mes "^FF0000상점 가격 설정이 올바르지 않습니다.^000000";
		next;
		return;
	}

	if (getd(.PointVar$) < .Shop_Blessing_PointCost) {
		mes "[환생 상점]";
		mes "환생 포인트가 부족합니다.";
		mes "필요 포인트: " + .Shop_Blessing_PointCost + "점";
		mes "보유 포인트: " + getd(.PointVar$) + "점";
		next;
		return;
	}

	if (Zeny < .Shop_Blessing_ZenyCost) {
		mes "[환생 상점]";
		mes "제니가 부족합니다.";
		mes "필요 제니: " + .Shop_Blessing_ZenyCost + " Zeny";
		next;
		return;
	}

	if (!checkweight(.Shop_Blessing_ItemID, 1)) {
		mes "[환생 상점]";
		mes "소지 무게 또는 인벤토리 공간을 확인해주세요.";
		next;
		return;
	}

	// 조건 확인 후 재화 차감 및 상품 지급
	setd .PointVar$, getd(.PointVar$) - .Shop_Blessing_PointCost;
	Zeny -= .Shop_Blessing_ZenyCost;
	getitem .Shop_Blessing_ItemID, 1;
	setd .ShopBlessingCountVar$, getd(.ShopBlessingCountVar$) + 1;

	.@remaining = .Shop_Blessing_WeeklyLimit - getd(.ShopBlessingCountVar$);

	if (.@remaining < 0)
		.@remaining = 0;

	mes "[환생 상점]";
	mes callsub(S_ItemName, .Shop_Blessing_ItemID) + " 1개를 구매했습니다.";
	mes " ";
	mes "남은 환생 포인트: ^00AA00" + getd(.PointVar$) + "점^000000";
	mes "이번 주 남은 구매 횟수: ^0055FF" + .@remaining + "회^000000";
	next;
	return;

S_GiveEveryReward:
	.@count = getarg(0);

	mes "[환생 보상]";
	mes "환생 " + .@count + "회 기본 보상을 지급합니다.";

	if (.Every_CostumeBox_ID > 0 && .Every_CostumeBox_Amount > 0) {
		callsub S_GiveItem, .Every_CostumeBox_ID, .Every_CostumeBox_Amount;
		mes "- " + callsub(S_ItemName, .Every_CostumeBox_ID) + " " + .Every_CostumeBox_Amount + "개";
	}

	if (.Every_RebirthCoin_ID > 0 && .Every_RebirthCoin_Amount > 0) {
		callsub S_GiveItem, .Every_RebirthCoin_ID, .Every_RebirthCoin_Amount;
		mes "- " + callsub(S_ItemName, .Every_RebirthCoin_ID) + " " + .Every_RebirthCoin_Amount + "개";
	}

	if (.Every_ExpManual_ID > 0 && .Every_ExpManual_Amount > 0) {
		callsub S_GiveItem, .Every_ExpManual_ID, .Every_ExpManual_Amount;
		mes "- " + callsub(S_ItemName, .Every_ExpManual_ID) + " " + .Every_ExpManual_Amount + "개";
	}

	next;
	return;

S_GiveEveryNReward:
	.@count = getarg(0);

	if (.EveryN_Count <= 0)
		return;

	if (.@count % .EveryN_Count != 0)
		return;

	if (.EveryN_ItemID <= 0 || .EveryN_Amount <= 0)
		return;

	mes "[환생 추가 보상]";
	mes .EveryN_Count + "회마다 지급되는 추가 보상입니다.";
	mes "- " + callsub(S_ItemName, .EveryN_ItemID) + " " + .EveryN_Amount + "개";
	callsub S_GiveItem, .EveryN_ItemID, .EveryN_Amount;
	next;

	return;

S_GiveUnclaimedMiles:
	.@count = getd(.CountVar$);
	.@given = 0;

	mes "[환생 관리자]";
	mes "현재 환생 횟수 기준으로 미수령 구간 보상을 확인합니다.";
	next;

	for (.@i = 0; .@i < getarraysize(.Milestone_Count); .@i++) {
		.@milestone = .Milestone_Count[.@i];

		if (.@count < .@milestone)
			continue;

		if (getd(.ClaimPrefix$ + .@milestone))
			continue;

		callsub S_GiveMilestoneByCount, .@milestone;
		.@given++;
	}

	if (.@given == 0) {
		mes "[환생 관리자]";
		mes "수령 가능한 미수령 구간 보상이 없습니다.";
	}

	return;

S_GiveMilestoneByCount:
	.@count = getarg(0);

	if (getd(.ClaimPrefix$ + .@count))
		return;

	if (.@count == .Cnt_EffectStone) {
		callsub S_GiveMilestoneFixed, .@count, .Item_EffectStone, .Amount_EffectStone, "환생 전용 이펙트 스톤";
		return;
	}

	if (.@count == .Cnt_UpperCostume) {
		callsub S_GiveCostumeChoice, .@count, 1;
		return;
	}

	if (.@count == .Cnt_FootprintStone) {
		callsub S_GiveMilestoneFixed, .@count, .Item_FootprintStone, .Amount_FootprintStone, "환생 전용 발자국 이펙트 스톤";
		return;
	}

	if (.@count == .Cnt_MiddleCostume) {
		callsub S_GiveCostumeChoice, .@count, 2;
		return;
	}

	if (.@count == .Cnt_EffectStoneTicket) {
		callsub S_GiveMilestoneFixed, .@count, .Item_EffectStoneTicket, .Amount_EffectStoneTicket, "이펙트 스톤 선택권";
		return;
	}

	if (.@count == .Cnt_LowerCostume) {
		callsub S_GiveCostumeChoice, .@count, 3;
		return;
	}

	if (.@count == .Cnt_FootprintTicket) {
		callsub S_GiveMilestoneFixed, .@count, .Item_FootprintTicket, .Amount_FootprintTicket, "발자국 이펙트 스톤 선택권";
		return;
	}

	if (.@count == .Cnt_GarmentCostume) {
		callsub S_GiveCostumeChoice, .@count, 4;
		return;
	}

	if (.@count == .Cnt_Aura) {
		callsub S_GiveMilestoneFixed, .@count, .Item_Aura, .Amount_Aura, "환생 전용 오라";
		return;
	}

	if (.@count == .Cnt_Title) {
		callsub S_GiveMilestoneFixed, .@count, .Item_Title, .Amount_Title, "환생 전용 칭호";
		return;
	}

	if (.@count == .Cnt_FinalEffect) {
		callsub S_GiveMilestoneFixed, .@count, .Item_FinalEffect, .Amount_FinalEffect, "환생 전용 이펙트";
		return;
	}

	return;

S_GiveMilestoneFixed:
	.@count = getarg(0);
	.@item_id = getarg(1);
	.@amount = getarg(2);
	.@reward_name$ = getarg(3);

	mes "[환생 구간 보상]";
	mes "환생 " + .@count + "회 보상을 지급합니다.";
	mes "보상: " + .@reward_name$;

	if (.@item_id <= 0) {
		mes "^FF0000아이템 ID가 설정되지 않았습니다.^000000";
		mes "OnInit 설정을 확인해주세요.";
		next;
		return;
	}

	callsub S_GiveItem, .@item_id, .@amount;
	setd .ClaimPrefix$ + .@count, 1;

	mes "- " + callsub(S_ItemName, .@item_id) + " " + .@amount + "개";
	next;
	return;

S_GiveCostumeChoice:
	.@count = getarg(0);
	.@type = getarg(1);

	callsub S_LoadCostumeChoice, .@type;

	mes "[환생 구간 보상]";
	mes "환생 " + .@count + "회 보상입니다.";
	mes .ChoiceTitle$ + " 3종 중 1개를 선택할 수 있습니다.";
	next;

	while (true) {
		mes "[의상 선택]";
		mes .ChoiceTitle$;
		mes "원하는 보상을 선택해주세요.";
		mes " ";

		for (.@i = 0; .@i < 3; .@i++) {
			mes (.@i + 1) + ". " + .ChoiceName$[.@i] + " - " + callsub(S_ItemName, .ChoiceItem[.@i]);
		}

		next;

		if (.UsePreviewCutin) {
			.@menu = select(
				"1번 미리보기",
				"2번 미리보기",
				"3번 미리보기",
				"1번 선택",
				"2번 선택",
				"3번 선택",
				"취소"
			);

			if (.@menu >= 1 && .@menu <= 3) {
				.@idx = .@menu - 1;

				if (.ChoicePreview$[.@idx] != "") {
					cutin .ChoicePreview$[.@idx], 2;
				}

				mes "[의상 미리보기]";
				mes .ChoiceName$[.@idx];
				mes "미리보기 이미지를 확인했습니다.";
				next;
				continue;
			}

			if (.@menu == 7) {
				cutin "", 255;
				mes "[환생 관리자]";
				mes "구간 보상은 수령 전까지 다시 받을 수 있습니다.";
				return;
			}

			.@idx = .@menu - 4;
		} else {
			.@menu = select(
				"1번 선택",
				"2번 선택",
				"3번 선택",
				"취소"
			);

			if (.@menu == 4) {
				mes "[환생 관리자]";
				mes "구간 보상은 수령 전까지 다시 받을 수 있습니다.";
				return;
			}

			.@idx = .@menu - 1;
		}

		cutin "", 255;

		mes "[의상 선택]";
		mes "선택한 보상:";
		mes "^0055FF" + .ChoiceName$[.@idx] + "^000000";
		mes callsub(S_ItemName, .ChoiceItem[.@idx]);
		next;

		if (select("이 보상으로 확정", "다시 선택") == 2)
			continue;

		if (.ChoiceItem[.@idx] <= 0) {
			mes "[환생 관리자]";
			mes "^FF0000선택 보상 아이템 ID가 설정되지 않았습니다.^000000";
			mes "OnInit 설정을 확인해주세요.";
			return;
		}

		callsub S_GiveItem, .ChoiceItem[.@idx], 1;
		setd .ClaimPrefix$ + .@count, 1;

		mes "[환생 관리자]";
		mes "보상이 지급되었습니다.";
		mes "- " + .ChoiceName$[.@idx];
		mes "- " + callsub(S_ItemName, .ChoiceItem[.@idx]);
		next;
		return;
	}

S_LoadCostumeChoice:

function	script	F_NEED_Rebirth_ResetCharacter	{
	// 환생 전 직업 및 세컨드 코스튬 상태 저장
	.@old_class = Class;
	.@old_body_style = getlook(LOOK_BODY2);

	// 기존 직업 계열에 맞는 1차 직업 확인
	.@first_job = callfunc("F_NEED_Rebirth_GetFirstJob", .@old_class);

	if (.@first_job == 0) {
		mes "[환생]";
		mes "현재 직업은 환생을 진행할 수 없는 직업입니다.";
		close;
	}

	// 기존 스탯/스킬 초기화
	resetstatus;
	resetskill;

	// 기존 계열 1차 직업으로 변경
	jobchange .@first_job;

	// 레벨 초기화
	resetlvl 1;
	set JobLevel, 1;
	set JobExp, 0;

	// 포인트 정리
	set StatusPoint, 48;
	set SkillPoint, 0;

	// 기본 기능 스킬 지급
	skill "NV_BASIC",9,0;

	// 환생 전 세컨드 코스튬 상태였다면 외형 초기화 후 티켓 반환
	if (.@old_body_style != .@old_class) {
		getitem 6959, 1;
		dispbottom "[환생] 세컨드 코스튬 티켓 1개가 지급되었습니다.";
	}

	percentheal 100,100;
	specialeffect2 EF_ANGEL2;
	return;
}
