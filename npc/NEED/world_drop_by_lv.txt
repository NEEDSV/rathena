//===== rAthena Script =======================================
//= Global World Drop by Monster Level (All maps / Non-boss)
//= Lv 1~99   : 673 (2%)
//= Lv 100~109: 673 (3%)
//= Lv 110~119: 673 (4%)
//= Lv 120~129: 673 (5%)
//= Lv 130~139: 673 (6%)
//= Lv 140~169: 677 (3%)
//= Lv 170+   : 674 (2%)
//= + Lv 50+ : 6705 (0.05%)  [Warrior's Proof]
//============================================================

-	script	NEED_KillEventDispatcher	-1,{
OnInit:
	// 캡차 발동 대상 맵
	setarray .check_maps$[0],
		"gef_fild10",
		"moc_fild17",
		"pay_dun03";

	// 확률: 10000 기준
	// 5 = 0.05%, 10 = 0.1%, 50 = 0.5%, 100 = 1%
	.check_rate = 10;

	// 개인별 재발동 쿨타임, 초 단위
	// 3600 = 1시간
	.player_delay = 3600;

	// GM 레벨 몇 이상 제외할지
	.gm_exclude_level = 1;
	end;

OnNPCKillEvent:
	// 죽인 몬스터/플레이어 RID가 붙어있는 상태에서 실행됨

	// 월드 드랍
	callsub S_WorldDrop;

	// 매크로 체크
	callsub S_MacroCheck;

	end;

// ==================================================
// 월드 드랍 처리
// ==================================================
S_WorldDrop:
	.@mobid = killedrid;
	if (.@mobid <= 0)
		return;
		
	.@map$ = strcharinfo(3);
	
	if (getmonsterinfo(.@mobid, MOB_MVPEXP) > 0)
	{
			if (.@map$ == "bossnia_01" || .@map$ == "bossnia_02" || .@map$ == "bossnia_03")
				return;
				
		if (rand(10000) < 5000)
		{
			getitem 677, 1;
			return;
		}
	}

	// 보스/MVP 제외: MOB_MODE의 0x20 (대부분 공통)
	if (getmonsterinfo(.@mobid, MOB_MODE) & 0x20)
		return;

	// 몬스터 레벨
	.@lv = getmonsterinfo(.@mobid, MOB_LV);
	if (.@lv <= 0)
		return;

	// -------------------------------------------------
	// [기존 월드드랍] 코인 3종
	// -------------------------------------------------
	.@item = 0;
	
	if (.@lv > 99)
	{
		if (rand(10000) < 100)
			getitem 9530, 1;
	}

	// 99~154 금화 3%
	if (.@lv > 29 && .@lv <= 98) {
		.@rate = 50 + .@lv;
		.@rate = callfunc("F_NEED_GoldEvent_HuntRate", .@rate);

		if (rand(10000) < .@rate)
			.@item = 399990;
	}
	else if (.@lv > 99 && .@lv <= 154) {
		.@rate = 100 + .@lv;
		.@rate = callfunc("F_NEED_GoldEvent_HuntRate", .@rate);

		if (rand(10000) < .@rate)
			.@item = 399990;
	}
	// 155~169 : 미스릴화 3%
	else if (.@lv > 155 && .@lv <= 169) {
		if (rand(10000) < 100 + .@lv) .@item = 674;
	}
	// 170+ : 미스릴화 2%
	else if (.@lv > 169) {
		if (rand(10000) < 200 + .@lv) .@item = 674;
	}

	if (.@item <= 0) return;
	
	if (.@item == 399990 && getgmlevel() >= 60) {
		dispbottom "[금화 이벤트 디버그] Lv " + .@lv + " / 이벤트 " + (($NEED_EVENT_HUNT_GOLD_ON) ? "ON" : "OFF") + " / 금화 지급";
	}

	// 캐시주화는 자동지급
	getitem .@item, 1;

	return;
	
// ==================================================
// 매크로 체크 처리
// ==================================================
S_MacroCheck:
	// GM 제외
//	if (getgmlevel() >= .gm_exclude_level)
//		return;

	// 대상 맵이 아니면 제외
//	if (inarray(.check_maps$, strcharinfo(3)) == -1)
//		return;

	// 개인별 쿨타임
	.@now = gettimetick(2);
	if (NEED_MACRO_CHECK_NEXT > .@now)
		return;

	// 확률 체크
	if (rand(10000) >= .check_rate)
		return;

	// 다음 체크 가능 시간 저장
	NEED_MACRO_CHECK_NEXT = .@now + .player_delay;

	// 현재 캐릭터에게 캡차 발동
	macro_detector(strcharinfo(0));
	return;
}
