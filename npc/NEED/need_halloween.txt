//------------------------------------------------------------
// 진행 여부

function	script	F_NEED_Halloween_Active	{
	return ($NEED_HALLOWEEN_EVENT_ON != 0);
}

//------------------------------------------------------------
// 스위치를 실제 상태에 반영한다 (OnInit / GM 변경 시).
//
// - 사냥 battle flag 는 conf 값 대신 GM 이 저장한 값으로 맞춘다 (재시작 후에도 유지).
// - 할로윈 NPC 는 이벤트가 켜져 있을 때만 보인다.
// - 잭 NPC 두 개는 추석 NPC 와 같은 칸을 쓴다. 할로윈이 켜져 있고 추석이 끝났으면
//   추석 NPC 를 숨기고, 할로윈이 꺼지면 다시 보이게 한다.
//------------------------------------------------------------
function	script	F_NEED_Halloween_Sync	{
	setbattleflag("need_halloween_hunt_enable", ($NEED_HALLOWEEN_EVENT_ON && $NEED_HALLOWEEN_HUNT_ON) ? 1 : 0);

	if ($NEED_HALLOWEEN_EVENT_ON) {
		enablenpc "꼬마 호박 잭#hw26j";
		enablenpc "할로윈 상인 잭#hw26s";
		if (!$NEED_CHUSEOK_EVENT_ON) {
			disablenpc "송편 빚는 아낙#chuseok2026";
			disablenpc "한가위 교환상 달이#chuseok2026";
		}
	}
	else {
		disablenpc "꼬마 호박 잭#hw26j";
		disablenpc "할로윈 상인 잭#hw26s";
		enablenpc "송편 빚는 아낙#chuseok2026";
		enablenpc "한가위 교환상 달이#chuseok2026";
	}
	return;
}

//------------------------------------------------------------
// 전장 할로윈 코인
//
// getarg(0) 1=티에라 / 2=플라비우스 / 3=KVM
//
// 추석 F_NEED_Chuseok_BG_Reward 와 같은 자리(bg_common.txt F_BG_Badge,
// kvm01.txt F_KvM_GrantReward 끝)에서 호출되므로, 기존 전장 보상 자격과
// 랜덤 매칭 중복 수령 방지(F_BG_RM_TryClaimReward)를 그대로 따른다.
// 승패 분기 바깥이라 승리/패배/무승부 모두 같은 수량이다.
//------------------------------------------------------------
function	script	F_NEED_Halloween_BG_Reward	{
	.@stage = getarg(0);

	if (.@stage < 1 || .@stage > 3)
		return;
	if (!$NEED_HALLOWEEN_EVENT_ON || !$NEED_HALLOWEEN_BG_ON)
		return;

	// 티에라 2 / 플라비우스 2 / KVM 3 = 한 사이클 7개
	setarray .@coin[1], 2, 2, 3;

	getitem 6658, .@coin[.@stage];
	dispbottom needtr("[할로윈 이벤트] 전장 보상으로 할로윈 코인 ", "[Halloween Event] Battleground reward: Halloween Coin x") + .@coin[.@stage] + needtr("개를 받았습니다.", ".");
	return;
}

// 잭 잡기
//
// 개인 인스턴스 '할로윈 잭 잡기' (db/import/instance_db.yml Id 1031)
//   기반 맵 pvp_n_1-2 (80x80, 통행 셀 전체가 한 영역). NoNpc 라 공용 맵 NPC 는 복사되지 않는다.
//   맵플래그는 원본에서 복사된 뒤, 입장 NPC 가 PvP/GvG/전장 플래그를 인스턴스 맵에서만 끈다.
//   (NoMapFlag 는 쓰지 않는다: 이 rAthena 에서는 처음 쓰는 인스턴스 슬롯의 플래그 배열이
//    비어 있어 그 맵에 setmapflag 를 하면 맵서버가 죽는다. 2026-10-22 격리 서버에서 확인.)
// 잭은 떠 있는 템플릿 NPC 를 인스턴스 맵에 duplicate 로 1개 만들고,
// 잡힐 때마다 movenpc + setnpcdisplay 로 새 위치/외형(일반/황금)으로 바꾼다.
//
// 상태 (캐릭터 임시 변수, 로그아웃 시 자동 소멸)
//   @HW_JACK_STATE  0 없음 / 1 진행 중 / 2 성공 처리 대기
//   @HW_JACK_IID    인스턴스 ID     @HW_JACK_MAP$  인스턴스 맵 이름
//   @HW_JACK_NPC$   잭 NPC 고유 이름 @HW_JACK_GOLD  현재 잭이 황금 잭이면 1
//   @HW_JACK_SCORE  점수            @HW_JACK_CATCH  잡은 수
//   @HW_JACK_START / @HW_JACK_END   시작 / 종료 시각 (gettimetick(2))
//------------------------------------------------------------
function	script	F_NEED_Halloween_JackSpawn	{
	.@map$ = @HW_JACK_MAP$;
	.@npc$ = @HW_JACK_NPC$;
	if (.@map$ == "" || .@npc$ == "")
		return 0;

	getmapxy(.@pmap$, .@px, .@py, BL_PC);

	// 플레이어에게서 6~28칸 떨어진 통행 가능 칸. 맵 안쪽 통행 영역은 x,y 61~138.
	.@found = 0;
	for (.@i = 0; .@i < 80 && !.@found; .@i++) {
		.@x = rand(61, 138);
		.@y = rand(61, 138);
		if (!checkcell(.@map$, .@x, .@y, cell_chkpass))
			continue;
		.@dx = (.@x > .@px) ? .@x - .@px : .@px - .@x;
		.@dy = (.@y > .@py) ? .@y - .@py : .@py - .@y;
		.@d = (.@dx > .@dy) ? .@dx : .@dy;
		if (.@d >= 6 && .@d <= 28)
			.@found = 1;
	}
	// 거리 조건을 못 맞추면 아무 통행 가능 칸.
	for (.@i = 0; .@i < 200 && !.@found; .@i++) {
		.@x = rand(61, 138);
		.@y = rand(61, 138);
		if (checkcell(.@map$, .@x, .@y, cell_chkpass))
			.@found = 1;
	}
	if (!.@found)
		return 0;

	// 잭이 나타날 때마다 10% 확률로 황금 잭.
	@HW_JACK_GOLD = (rand(100) < 10) ? 1 : 0;
	if (@HW_JACK_GOLD) {
		setnpcdisplay .@npc$, "황금 잭", 1130;
		setnpcnameen "Golden Jack", .@npc$;
	}
	else {
		setnpcdisplay .@npc$, "잭", 1130;
		setnpcnameen "Jack", .@npc$;
	}
	movenpc .@npc$, .@x, .@y;
	if (@HW_JACK_GOLD) {
		specialeffect EF_LIGHTSPHERE, AREA, .@npc$;
		dispbottom needtr("[잭 잡기] 반짝! 황금 잭이 나타났다! (+3점)", "[Jack Catch] Sparkle! A Golden Jack appeared! (+3 pts)"), 0xFFD700;
	}
	return 1;
}

// 남은 시간 "m:ss"
function	script	F_NEED_Halloween_JackLeft$	{
	.@left = @HW_JACK_END - gettimetick(2);
	if (.@left < 0)
		.@left = 0;
	return (.@left / 60) + ":" + ((.@left % 60 < 10) ? "0" : "") + (.@left % 60);
}

// 타이머/인스턴스/임시 변수를 모두 정리하고 맵 밖으로 내보낸다.
// 인스턴스 안 NPC 의 스크립트 실행 중에는 부르지 않는다 (OnFinish / OnTimeUp 에서만).
function	script	F_NEED_Halloween_JackCleanup	{
	deltimer "HalloweenJack#hw26ctl::OnWarn60";
	deltimer "HalloweenJack#hw26ctl::OnWarn30";
	deltimer "HalloweenJack#hw26ctl::OnWarn10";
	deltimer "HalloweenJack#hw26ctl::OnTimeUp";
	deltimer "HalloweenJack#hw26ctl::OnFinish";

	.@iid = @HW_JACK_IID;
	.@map$ = @HW_JACK_MAP$;

	@HW_JACK_STATE = 0;
	@HW_JACK_IID = 0;
	@HW_JACK_MAP$ = "";
	@HW_JACK_NPC$ = "";
	@HW_JACK_GOLD = 0;
	@HW_JACK_SCORE = 0;
	@HW_JACK_CATCH = 0;
	@HW_JACK_START = 0;
	@HW_JACK_END = 0;

	// 먼저 내보내고 나서 인스턴스를 없앤다 (없애면 남은 사람은 세이브 포인트로 간다).
	if (.@map$ != "" && strcharinfo(3) == .@map$)
		warp "payo2", 113, 98;
	if (.@iid > 0)
		instance_destroy .@iid;
	return;
}

// 잭 템플릿. 직접 배치하지 않고 인스턴스 맵에 duplicate 로만 만든다.
-	script	HalloweenJack#hw26tpl	-1,{
	if (@HW_JACK_STATE != 1 || @HW_JACK_NPC$ != strnpcinfo(3) || strcharinfo(3) != strnpcinfo(4) || instance_id(IM_CHAR) != @HW_JACK_IID)
		end;
	if (gettimetick(2) > @HW_JACK_END) {
		dispbottom needtr("[잭 잡기] 시간이 다 되었습니다.", "[Jack Catch] Time is up."), 0xFF6600;
		end;
	}

	.@pt = (@HW_JACK_GOLD) ? 3 : 1;
	@HW_JACK_SCORE += .@pt;
	@HW_JACK_CATCH++;

	if (@HW_JACK_SCORE >= 15) {
		// 성공. 보상과 정리는 이 NPC 밖(컨트롤러)에서 한다: 이 NPC 는 인스턴스와 함께 사라진다.
		@HW_JACK_STATE = 2;
		specialeffect EF_LIGHTSPHERE, AREA, strnpcinfo(3);
		dispbottom needtr("[잭 잡기] ", "[Jack Catch] ") + ((@HW_JACK_GOLD) ? needtr("황금 잭 포획! +3점", "Golden Jack caught! +3 pts") : needtr("잭 포획! +1점", "Jack caught! +1 pt")) + " (" + @HW_JACK_SCORE + " / 15)", 0xFFA500;
		deltimer "HalloweenJack#hw26ctl::OnFinish";
		addtimer 100, "HalloweenJack#hw26ctl::OnFinish";
		end;
	}

	dispbottom needtr("[잭 잡기] ", "[Jack Catch] ") + ((@HW_JACK_GOLD) ? needtr("황금 잭 포획! +3점", "Golden Jack caught! +3 pts") : needtr("잭 포획! +1점", "Jack caught! +1 pt")) + " (" + @HW_JACK_SCORE + " / 15" + needtr(", 남은 시간 ", ", time left ") + callfunc("F_NEED_Halloween_JackLeft$") + ")", 0xFFA500;
	callfunc "F_NEED_Halloween_JackSpawn";
	end;
}

// 잭 잡기 컨트롤러 (캐릭터 타이머 이벤트 전용, 맵에 없음)
-	script	HalloweenJack#hw26ctl	-1,{
	end;

OnWarn60:
	if (@HW_JACK_STATE == 1)
		dispbottom needtr("[잭 잡기] 남은 시간 1분! 현재 ", "[Jack Catch] 1 minute left! Score ") + @HW_JACK_SCORE + " / 15", 0xFF6600;
	end;

OnWarn30:
	if (@HW_JACK_STATE == 1)
		dispbottom needtr("[잭 잡기] 남은 시간 30초! 현재 ", "[Jack Catch] 30 seconds left! Score ") + @HW_JACK_SCORE + " / 15", 0xFF6600;
	end;

OnWarn10:
	if (@HW_JACK_STATE == 1)
		dispbottom needtr("[잭 잡기] 남은 시간 10초! 현재 ", "[Jack Catch] 10 seconds left! Score ") + @HW_JACK_SCORE + " / 15", 0xFF6600;
	end;

// 시간 초과: 기록을 남기지 않으므로 다시 도전할 수 있다.
OnTimeUp:
	if (@HW_JACK_STATE != 1)
		end;
	.@score = @HW_JACK_SCORE;
	callfunc "F_NEED_Halloween_JackCleanup";
	dispbottom needtr("[잭 잡기] 시간 초과! ", "[Jack Catch] Time over! ") + .@score + needtr(" / 15점. 보상 기록은 남지 않았으니 다시 도전할 수 있어요.", " / 15 pts. Nothing was recorded, so you can try again."), 0xFF6600;
	end;

// 성공: 맵 밖으로 내보낸 뒤 일일 원장(계정 + IP)을 한 트랜잭션으로 확인하고 지급한다.
OnFinish:
	if (@HW_JACK_STATE != 2)
		end;
	.@score = @HW_JACK_SCORE;
	.@catch = @HW_JACK_CATCH;
	.@sec = gettimetick(2) - @HW_JACK_START;
	callfunc "F_NEED_Halloween_JackCleanup";

	dispbottom needtr("[잭 잡기] 성공! 잭 ", "[Jack Catch] Clear! ") + .@catch + needtr("마리 / ", " jacks / ") + .@score + needtr("점 / ", " pts / ") + .@sec + needtr("초", " sec"), 0x00CC66;

	if (!$NEED_HALLOWEEN_EVENT_ON || !$NEED_HALLOWEEN_JACK_REWARD_ON) {
		dispbottom needtr("[잭 잡기] 지금은 보상 지급이 중지되어 있어 보상이 없습니다. (기록도 남지 않음)", "[Jack Catch] Rewards are paused right now, so nothing was given. (Nothing was recorded.)"), 0xFF6600;
		end;
	}

	// 1 지급 / 0 계정 / -1 IP / -2 인벤토리 / -3 DB 오류 / -4 기능 중지
	.@r = needhalloweenjack(1);
	if (.@r == 1)
		dispbottom needtr("[잭 잡기] 보상으로 할로윈 코인 10개를 받았습니다! 내일 04:00 이후 다시 도전하세요.", "[Jack Catch] You received 10 Halloween Coins! Come back after 04:00 tomorrow."), 0x00CC66;
	else if (.@r == 0)
		dispbottom needtr("[잭 잡기] 이 계정은 오늘 이미 보상을 받았습니다.", "[Jack Catch] This account has already been rewarded today."), 0xFF6600;
	else if (.@r == -1)
		dispbottom needtr("[잭 잡기] 현재 접속 IP에서 오늘 이미 보상을 받았습니다.", "[Jack Catch] A reward was already given to your current IP today."), 0xFF6600;
	else if (.@r == -2)
		dispbottom needtr("[잭 잡기] 인벤토리 공간이 없어 보상을 받지 못했습니다. 정리 후 다시 도전하세요. (기록은 남지 않음)", "[Jack Catch] No inventory space for the reward. Make room and try again. (Nothing was recorded.)"), 0xFF6600;
	else
		dispbottom needtr("[잭 잡기] 보상 처리 중 오류가 발생했습니다. 운영자에게 문의해 주세요.", "[Jack Catch] An error occurred while giving the reward. Please contact an administrator."), 0xFF0000;
	end;
}

// 할로윈 교환 한도 (계정 단독)
//
// 추석 원장 함수 F_NEED_Chuseok_LimitUsed / LimitReserve 에 event_id 202610 을 넘겨
// need_chuseok_exchange_limit 을 그대로 쓴다. scope 는 ACCOUNT 만 사용한다.
//   SHADOW_HAMMER  period_key = 주간 키 (토요일 22:00, F_NEED_Chuseok_WeekKey)
//   ARIA           period_key = 0 (이벤트 기간)
//   REFINE9        period_key = 0, 무기/방어구 공용 키 -> 둘 중 하나만 총 1회
//
// getarg(0) product / getarg(1) period_key / getarg(2) 한도
// getarg(3) 지급 아이템 / getarg(4) 개당 코인 / getarg(5) 수량
//
// 반환  1 성공 / 0 계정 한도 초과 / -1 코인 부족 / -2 무게·공간 부족 / -4 기록 조회 실패
//
// 검사와 차감·지급 사이에 next / input / select 가 없다. 같은 계정의 다른 캐릭터가
// 동시에 와도 한도 예약(INSERT ... ON DUPLICATE KEY UPDATE)이 먼저 된 쪽만 통과한다.
//------------------------------------------------------------
function	script	F_NEED_Halloween_LimitedExchange	{
	.@product$ = getarg(0);
	.@period = getarg(1);
	.@cap = getarg(2);
	.@item = getarg(3);
	.@cost = getarg(4);
	.@amount = getarg(5);

	if (.@amount < 1 || .@amount > .@cap)
		return 0;

	.@acc$ = "" + getcharid(3);
	.@used = callfunc("F_NEED_Chuseok_LimitUsed", "ACCOUNT", .@acc$, .@product$, .@period, 202610);
	if (.@used < 0)
		return -4;
	if (.@used + .@amount > .@cap)
		return 0;

	.@need = .@cost * .@amount;
	if (countitem(6658) < .@need)
		return -1;
	if (!checkweight(.@item, .@amount))
		return -2;

	.@after = callfunc("F_NEED_Chuseok_LimitReserve", "ACCOUNT", .@acc$, .@product$, .@period, .@cap, .@amount, 202610);
	if (.@after < 0)
		return 0;

	delitem 6658, .@need;
	getitem .@item, .@amount;

	query_sql("INSERT INTO `need_chuseok_exchange_log` (`event_id`,`account_id`,`char_id`,`char_name`,`ip`,`product`,`period_key`,`family_group_id`,`amount`,`songpyun_spent`,`used_after`,`ip_used_after`,`created_at`) VALUES (202610," + getcharid(3) + "," + getcharid(0) + ",'" + escape_sql(strcharinfo(0)) + "','" + escape_sql(getcharip()) + "','" + escape_sql(.@product$) + "'," + .@period + ",0," + .@amount + "," + .@need + "," + .@after + ",0,NOW())");

	return 1;
}

// 한도 교환 실패 안내. getarg(0) 반환값, getarg(1) 1=주간 한도
function	script	F_NEED_Halloween_LimitFailMes	{
	.@r = getarg(0);

	if (.@r == -1)
		mes needtr("^FF0000할로윈 코인이 모자라.^000000", "^FF0000You don't have enough Halloween Coins.^000000");
	else if (.@r == -2)
		mes needtr("^FF0000짐이 너무 많아서 줄 수가 없어.^000000", "^FF0000You're carrying too much to receive it.^000000");
	else if (.@r == -4)
		mes needtr("^FF0000교환 기록을 확인할 수 없어. 잠시 후 다시 해 줘.^000000", "^FF0000The exchange record could not be checked. Please try again shortly.^000000");
	else
		mes needtr("^FF0000이 계정은 ", "^FF0000This account has used up its ") + ((getarg(1)) ? needtr("이번 주 ", "weekly ") : "") + needtr("교환 한도를 모두 썼어.^000000", "exchange limit.^000000");
	return;
}

		mes needtr("^FF0000교환 기록을 확인할 수 없어. 잠시 후 다시 와 줘.^000000", "^FF0000The exchange record cannot be checked right now. Please come back shortly.^000000");
		close;
	}
	if (.@used > .@cap)
		.@used = .@cap;

	mes needtr("^FFCC00+9 무기 · 방어구 보장권 (택1)^000000", "^FFCC00+9 Weapon / Armor Certificate (pick one)^000000");
	mes needtr("이벤트 기간 교환 : ^0055FF", "Exchanged this event : ^0055FF") + .@used + " / " + .@cap + "^000000";
	mes needtr("필요 코인 : ", "Coins needed : ") + .@cost + needtr("개", "");
	mes needtr("가진 할로윈 코인 : ^0055FF", "Halloween Coins held : ^0055FF") + countitem(6658) + needtr("^000000 개", "^000000");

	if (.@used >= .@cap) {
		mes " ";
		mes needtr("^FF0000이미 바꿨어. 무기·방어구 합쳐 계정당 1개야.^000000", "^FF0000You already have one. The limit is 1 per account, weapon and armor combined.^000000");
		close;
	}
	if (countitem(6658) < .@cost) {
		mes " ";
		mes needtr("^FF0000할로윈 코인이 ", "^FF0000You need ") + (.@cost - countitem(6658)) + needtr("개 더 필요해.^000000", " more Halloween Coins.^000000");
		close;
	}

	next;
	mes .@n$;
	mes needtr("어떤 보장권으로 할래?", "Which certificate would you like?");
	mes needtr("^FF0000하나를 고르면 다른 하나는 이번 이벤트에서 더 받을 수 없어.^000000", "^FF0000Once you pick one, you can't get the other during this event.^000000");
	next;

	.@sel = select(needtr("+9 무기 보장권", "+9 Weapon Certificate"), needtr("+9 방어구 보장권", "+9 Armor Certificate"), needtr("그만둔다", "Cancel"));
	if (.@sel == 3) {
		mes .@n$;
		mes needtr("천천히 골라.", "Take your time.");
		close;
	}
	.@item = (.@sel == 1) ? 6228 : 6232;

	mes .@n$;
	mes "^0055FF" + getitemname(.@item) + needtr("^000000 (으)로 바꿀래?", "^000000 - trade for this?");
	mes needtr("할로윈 코인 ", "It costs ") + .@cost + needtr("개가 들어.", " Halloween Coins.");
	next;
	if (select(needtr("바꾼다", "Trade"), needtr("그만둔다", "Cancel")) == 2) {
		mes .@n$;
		mes needtr("천천히 골라.", "Take your time.");
		close;
	}

	mes .@n$;
	if (!callfunc("F_NEED_Halloween_Active") || !$NEED_HALLOWEEN_SHOP_ON) {
		mes needtr("앗, 방금 가게 문을 닫았어.", "Oops, the shop just closed.");
		close;
	}
	.@result = callfunc("F_NEED_Halloween_LimitedExchange", "REFINE9", 0, .@cap, .@item, .@cost, 1);
	if (.@result == 1) {
		mes "^0055FF" + getitemname(.@item) + needtr("^000000 여기 있어.", "^000000 - here you go.");
		mes needtr("할로윈 코인 ", "I took ") + .@cost + needtr("개를 받았어.", " Halloween Coins.");
	}
	else
		callfunc "F_NEED_Halloween_LimitFailMes", .@result, 0;
	close;

OnInit:
	setnpcnameen "Halloween Merchant Jack";
	end;
}
