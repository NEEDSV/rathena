//------------------------------------------------------------
// 전장 종료 보상 - 송편 재료 지급 + 풀사이클 꿀송편 판정
//
// getarg(0) 1=티에라 / 2=플라비우스 / 3=KVM
//
// 기존 전장 정상 보상 지급 대상자 판정을 그대로 사용한다.
// (bg_common.txt F_BG_Badge, kvm01.txt F_KvM_GrantReward 에서 호출)
// 승리팀 / 패배팀 지급량 차이는 두지 않는다.
//------------------------------------------------------------
function	script	F_NEED_Chuseok_BG_Reward	{
	.@stage = getarg(0);

	if (.@stage < 1 || .@stage > 3)
		return;
	if (!$NEED_CHUSEOK_EVENT_ON)
		return;

	// 송편 2개분 / 2개분 / 3개분 = 한 사이클 송편 7개분
	setarray .@dough[1], 8, 8, 12;
	setarray .@pine[1], 2, 2, 3;

	if ($NEED_CHUSEOK_BG_MATERIAL_ON) {
		getitem 7613, .@dough[.@stage];
		getitem 1001827, .@pine[.@stage];
		dispbottom "[추석 이벤트] 송편 재료를 받았습니다. (작은 찹쌀반죽 " + .@dough[.@stage] + "개 / 솔잎 " + .@pine[.@stage] + "개)";
	}

	// 풀사이클 추적.
	// 랜덤 매칭이면 사이클 토큰으로, 수동 진행이면 티에라 참여를 기준으로 초기화한다.
	.@token = ($@BG_RM_CYCLE_ACTIVE) ? $@BG_RM_TOKEN : 0;
	if (@NEED_CHUSEOK_BG_TOKEN != .@token || .@stage == 1) {
		@NEED_CHUSEOK_BG_TOKEN = .@token;
		@NEED_CHUSEOK_BG_MASK = 0;
	}
	@NEED_CHUSEOK_BG_MASK = @NEED_CHUSEOK_BG_MASK | (1 << (.@stage - 1));

	// 꿀송편 판정은 KVM 까지 정상 완료한 캐릭터에 한해 사이클당 1회.
	if (.@stage != 3)
		return;

	.@full = ((@NEED_CHUSEOK_BG_MASK & 7) == 7);
	@NEED_CHUSEOK_BG_MASK = 0;

	if (!.@full)
		return;
	if (!$NEED_CHUSEOK_BG_HONEY_ON)
		return;
	if (rand(10000) >= 500)		// 5%
		return;

	// 사냥 꿀송편과 계정 + IP 일일 제한을 공유한다.
	// 1=지급 / 0=계정 제한 / -1=IP 제한 / -2=인벤토리 / -3=DB 오류 / -4=기능 정지
	.@result = needchuseokhoney("BG");
	if (.@result == 1)
		dispbottom "[추석 이벤트] 전장 풀사이클 보상으로 꿀송편 1개를 획득했습니다.";

	return;
}

//------------------------------------------------------------
// 계정 + IP 단위 교환 한도
//
// 주간 주기 기준은 행운의 알 천장(need_lucky_ceiling.txt S_WeekKey)과 동일하게
// 매주 토요일 22:00 이다.
//
// 꿀송편 일일 제한과 같은 방식으로 계정과 접속 IP 를 모두 본다. 둘 중 하나라도
// 한도에 걸리면 교환되지 않으므로, 다중 계정으로 우회할 수 없다.
//
//   scope      : 'ACCOUNT' 또는 'IP'
//   subject    : ACCOUNT 는 account_id, IP 는 접속 IP 문자열
//   product    : 'DAECHUK' 대축(6635) / 'ARIA' 아리아 의상 선택 상자(399935)
//   period_key : 주간 상품은 직전 토요일 22:00 의 YYYYMMDD, 시즌 상품은 0
//------------------------------------------------------------

// 매주 토요일 22:00 기준 주기 키 (YYYYMMDD)
function	script	F_NEED_Chuseok_WeekKey	{
	.@dow = gettime(DT_DAYOFWEEK);		// 0 = 일 ... 6 = 토
	.@back = (.@dow + 1) % 7;			// 직전 토요일까지 며칠 전인가

	if (.@back == 0 && gettime(DT_HOUR) < 22)
		.@back = 7;						// 토요일 21:59 까지는 지난 주기

	return atoi(gettimestr("%Y%m%d", 11, gettimetick(2) - .@back * 86400));
}

// 다음 주간 초기화 시점 안내 문자열
function	script	F_NEED_Chuseok_WeekResetStr$	{
	.@dow = gettime(DT_DAYOFWEEK);
	.@fwd = (6 - .@dow + 7) % 7;

	if (.@fwd == 0 && gettime(DT_HOUR) >= 22)
		.@fwd = 7;

	return gettimestr("%Y-%m-%d", 11, gettimetick(2) + .@fwd * 86400) + " (토) 22:00";
}

// 현재 주기에서 이 대상(scope+subject)이 사용한 횟수. 조회 실패 시 -1.
// getarg(0) scope$ / getarg(1) subject$ / getarg(2) product$ / getarg(3) period_key
function	script	F_NEED_Chuseok_LimitUsed	{
	.@rows = query_sql("SELECT `used` FROM `need_chuseok_exchange_limit` WHERE `event_id`=202609 AND `scope`='" + escape_sql(getarg(0)) + "' AND `subject`='" + escape_sql(getarg(1)) + "' AND `product`='" + escape_sql(getarg(2)) + "' AND `period_key`=" + getarg(3), .@used);

	if (.@rows < 0)
		return -1;
	if (.@rows < 1)
		return 0;

	return .@used;
}

// 한 쪽(scope) 한도를 예약한다. 성공하면 갱신된 사용량, 한도 초과면 -1.
// getarg(0) scope$ / getarg(1) subject$ / getarg(2) product$
// getarg(3) period_key / getarg(4) 한도 / getarg(5) 수량
function	script	F_NEED_Chuseok_LimitReserve	{
	.@scope$ = getarg(0);
	.@subject$ = getarg(1);
	.@product$ = getarg(2);
	.@period = getarg(3);
	.@cap = getarg(4);
	.@amount = getarg(5);

	.@before = callfunc("F_NEED_Chuseok_LimitUsed", .@scope$, .@subject$, .@product$, .@period);
	if (.@before < 0 || .@before + .@amount > .@cap)
		return -1;

	// 한도를 넘기는 갱신은 SQL 쪽에서도 한 번 더 막는다.
	query_sql("INSERT INTO `need_chuseok_exchange_limit` (`event_id`,`scope`,`subject`,`product`,`period_key`,`used`) VALUES (202609,'" + escape_sql(.@scope$) + "','" + escape_sql(.@subject$) + "','" + escape_sql(.@product$) + "'," + .@period + "," + .@amount + ") ON DUPLICATE KEY UPDATE `used` = IF(`used` + " + .@amount + " <= " + .@cap + ", `used` + " + .@amount + ", `used`)");

	.@after = callfunc("F_NEED_Chuseok_LimitUsed", .@scope$, .@subject$, .@product$, .@period);
	if (.@after != .@before + .@amount)
		return -1;			// 예약되지 않았다.

	return .@after;
}

// 예약을 되돌린다. 한쪽만 예약된 채로 끝나는 일이 없도록 쓴다.
function	script	F_NEED_Chuseok_LimitRelease	{
	query_sql("UPDATE `need_chuseok_exchange_limit` SET `used` = IF(`used` >= " + getarg(4) + ", `used` - " + getarg(4) + ", 0) WHERE `event_id`=202609 AND `scope`='" + escape_sql(getarg(0)) + "' AND `subject`='" + escape_sql(getarg(1)) + "' AND `product`='" + escape_sql(getarg(2)) + "' AND `period_key`=" + getarg(3));
	return;
}

// 한도가 걸린 교환 처리.
//
// getarg(0) product 문자열 / getarg(1) period_key / getarg(2) 한도
// getarg(3) 지급 아이템 ID / getarg(4) 개당 송편 소모량 / getarg(5) 교환 수량
//
// 반환  1 = 성공
//       0 = 계정 한도 초과
//      -1 = 송편 부족
//      -2 = 무게·슬롯 부족
//      -3 = 접속 IP 확인 불가
//      -4 = 원장 조회 실패
//      -5 = IP 한도 초과
//
// 검사부터 차감까지 사이에 next / input 이 없다. 스크립트는 그 사이에 다른
// 스크립트로 양보하지 않으므로 같은 계정·IP 의 다른 캐릭터가 끼어들 수 없다.
function	script	F_NEED_Chuseok_LimitedExchange	{
	.@product$ = getarg(0);
	.@period = getarg(1);
	.@cap = getarg(2);
	.@item = getarg(3);
	.@cost = getarg(4);
	.@amount = getarg(5);

	if (.@amount < 1 || .@amount > .@cap)
		return 0;

	.@acc$ = "" + getcharid(3);
	.@ip$ = getcharip();

	// IP 를 못 읽으면 IP 한도를 강제할 수 없으므로 지급하지 않는다.
	if (.@ip$ == "")
		return -3;

	.@used_acc = callfunc("F_NEED_Chuseok_LimitUsed", "ACCOUNT", .@acc$, .@product$, .@period);
	.@used_ip = callfunc("F_NEED_Chuseok_LimitUsed", "IP", .@ip$, .@product$, .@period);
	if (.@used_acc < 0 || .@used_ip < 0)
		return -4;
	if (.@used_acc + .@amount > .@cap)
		return 0;
	if (.@used_ip + .@amount > .@cap)
		return -5;

	.@need = .@cost * .@amount;
	if (countitem(663) < .@need)
		return -1;
	if (!checkweight(.@item, .@amount))
		return -2;

	.@after_acc = callfunc("F_NEED_Chuseok_LimitReserve", "ACCOUNT", .@acc$, .@product$, .@period, .@cap, .@amount);
	if (.@after_acc < 0)
		return 0;

	.@after_ip = callfunc("F_NEED_Chuseok_LimitReserve", "IP", .@ip$, .@product$, .@period, .@cap, .@amount);
	if (.@after_ip < 0) {
		// 계정 쪽만 잡힌 상태로 남지 않도록 되돌린다. 아직 아무것도 차감하지 않았다.
		callfunc "F_NEED_Chuseok_LimitRelease", "ACCOUNT", .@acc$, .@product$, .@period, .@amount;
		return -5;
	}

	delitem 663, .@need;
	getitem .@item, .@amount;

	query_sql("INSERT INTO `need_chuseok_exchange_log` (`event_id`,`account_id`,`char_id`,`char_name`,`ip`,`product`,`period_key`,`amount`,`songpyun_spent`,`used_after`,`ip_used_after`,`created_at`) VALUES (202609," + getcharid(3) + "," + getcharid(0) + ",'" + escape_sql(strcharinfo(0)) + "','" + escape_sql(.@ip$) + "','" + escape_sql(.@product$) + "'," + .@period + "," + .@amount + "," + .@need + "," + .@after_acc + "," + .@after_ip + ",NOW())");

	return 1;
}

// 한도 교환 실패 사유 안내. getarg(0) 반환값, getarg(1) 주간 여부(1=주간)
function	script	F_NEED_Chuseok_LimitFailMes	{
	.@r = getarg(0);
	.@weekly = getarg(1);

	if (.@r == -1)
		mes "^FF0000송편이 모자랍니다.^000000";
	else if (.@r == -2)
		mes "^FF0000짐이 너무 많아 받으실 수 없습니다.^000000";
	else if (.@r == -3)
		mes "^FF0000접속 정보를 확인할 수 없어 교환할 수 없습니다.^000000";
	else if (.@r == -4)
		mes "^FF0000교환 기록을 확인할 수 없습니다. 잠시 후 다시 시도해 주세요.^000000";
	else if (.@r == -5)
		mes "^FF0000현재 접속 IP의 " + ((.@weekly) ? "이번 주 " : "") + "교환 한도를 모두 사용했습니다.^000000";
	else
		mes "^FF0000이 계정의 " + ((.@weekly) ? "이번 주 " : "") + "교환 한도를 모두 사용했습니다.^000000";
	return;
}
