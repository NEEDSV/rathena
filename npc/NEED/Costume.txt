L_Register:
	deletearray .@item_list[0], getarraysize(.@item_list);
	deletearray .@cid_list[0], getarraysize(.@cid_list);
	deletearray .@amount_list[0], getarraysize(.@amount_list);

	getinventorylist;
	set .@bound_total, countbound(.ClosetBoundType);
	set .@count, 0;
	set .@total_amount, 0;

	freeloop(1);

	for (set .@i, 0; .@i < @inventorylist_count; set .@i, .@i + 1) {
		set .@item_id, @inventorylist_id[.@i];

		// 현재 장착 중인 의상은 일괄 등록 대상에서 제외
		// 동일 아이템을 여러 개 보유해도 하나라도 장착 중이면 해당 종류 전체를 보호
		if (callsub(L_IsCostumeEquipped, .@item_id))
			continue;

		set .@bound_amount_check, 0;

		for (set .@b, 0; .@b < .@bound_total; set .@b, .@b + 1) {
			if (@bound_items[.@b] == .@item_id) {
				set .@bound_amount_check, @bound_amount[.@b];
				break;
			}
		}

		// 귀속이 하나라도 있으면 일반 아이템 보호를 위해 제외
		if (.@bound_amount_check > 0)
			continue;

		set .@cid, getcostumecollectionid(.@item_id);

		if (.@cid <= 0)
			continue;

		set .@dup, 0;
		for (set .@j, 0; .@j < .@count; set .@j, .@j + 1) {
			if (.@item_list[.@j] == .@item_id) {
				set .@dup, 1;
				break;
			}
		}

		if (.@dup)
			continue;

		set .@amount, countitem(.@item_id);

		if (.@amount <= 0)
			continue;

		set .@item_list[.@count], .@item_id;
		set .@cid_list[.@count], .@cid;
		set .@amount_list[.@count], .@amount;
		set .@total_amount, .@total_amount + .@amount;
		set .@count, .@count + 1;
	}

	freeloop(0);

	if (.@count <= 0) {
		mes needtr("[의상 등록]", "[Register Costumes]");
		mes needtr("등록 가능한 의상이 없습니다.", "You have no costume that can be registered.");
		close;
	}

	mes needtr("[의상 등록 / 일괄 코인화]", "[Register / Bulk Convert]");
	mes needtr("등록 대상: ^FF8000", "Selected: ^FF8000") + .@count + needtr("^000000종 / ^FF8000", "^000000 kinds / ^FF8000") + .@total_amount + needtr("^000000개", "^000000");
	mes " ";
	mes needtr("아래 의상은 승인 시", "When you confirm, every one of these costumes");
	mes needtr("^FF0000보유 수량 전체가 삭제^000000되고", "is ^FF0000deleted in full^000000 and");
	mes needtr("삭제 수량만큼 의상 코인이 지급됩니다.", "you get one Costume Coin per deleted item.");
	mes " ";
	mes needtr("^FF0000※ 이미 등록된 의상도 코인화 대상입니다.^000000", "^FF0000* Already registered costumes are converted too.^000000");
	mes needtr("^FF0000※ 같은 의상의 귀속 아이템을 보유 중이면 제외됩니다.^000000", "^FF0000* Excluded if you hold a bound copy of the same costume.^000000");
	mes needtr("^0055FF※ 현재 장착 중인 의상은 일괄 등록 대상에서 제외됩니다.^000000", "^0055FF* A costume you are wearing is left out of the bulk run.^000000");
	mes needtr("^0055FF※ 같은 아이템을 여러 개 보유해도 하나라도 장착 중이면 해당 종류 전체가 제외됩니다.^000000", "^0055FF* If even one copy is worn the whole kind is left out.^000000");
	mes " ";

	for (set .@i, 0; .@i < .@count; set .@i, .@i + 1) {
		set .@cid, .@cid_list[.@i];
		set .@item_id, .@item_list[.@i];
		set .@name$, getitemname(.@item_id);

		if (iscostumeregistered(.@cid))
			mes (.@i + 1) + ". [" + .@cid + "] " + .@name$ + " x" + .@amount_list[.@i] + needtr(" ^ff8c00(등록됨)^000000", " ^ff8c00(registered)^000000");
		else
			mes (.@i + 1) + ". [" + .@cid + "] " + .@name$ + " x" + .@amount_list[.@i];
	}

	next;

	mes needtr("[의상 등록 / 일괄 코인화]", "[Register / Bulk Convert]");
	mes needtr("위 목록의 의상을 모두 등록/코인화합니다.", "All costumes above will be registered and converted.");
	mes " ";
	mes needtr("획득 예정 의상 코인:", "Costume Coins you will receive:");
	mes "^FF8000" + .@total_amount + needtr("^000000개", "^000000");
	mes " ";
	mes needtr("^FF0000진행 후 의상 아이템은 복구할 수 없습니다.^000000", "^FF0000The costume items cannot be recovered afterwards.^000000");
	next;

	if (select(needtr("일괄 등록한다:취소", "Register All:Cancel")) == 2)
		goto L_CollectionMenu;

	set .@new_count, 0;
	set .@already_count, 0;
	set .@coin_count, 0;
	set .@skip_count, 0;
	set .@fail_count, 0;

	freeloop(1);

	for (set .@i, 0; .@i < .@count; set .@i, .@i + 1) {
		set .@item_id, .@item_list[.@i];
		set .@cid, .@cid_list[.@i];
		set .@amount, .@amount_list[.@i];

		// 목록 확인 후 새로 장착했을 수 있으므로 진행 직전 재검사
		if (callsub(L_IsCostumeEquipped, .@item_id)) {
			set .@skip_count, .@skip_count + 1;
			continue;
		}

		// 진행 직전 귀속 보유 여부 재검사
		set .@bound_total, countbound(.ClosetBoundType);
		set .@bound_amount_check, 0;

		for (set .@b, 0; .@b < .@bound_total; set .@b, .@b + 1) {
			if (@bound_items[.@b] == .@item_id) {
				set .@bound_amount_check, @bound_amount[.@b];
				break;
			}
		}

		if (.@bound_amount_check > 0) {
			set .@skip_count, .@skip_count + 1;
			continue;
		}

		if (countitem(.@item_id) <= 0) {
			set .@skip_count, .@skip_count + 1;
			continue;
		}

		if (countitem(.@item_id) < .@amount)
			set .@amount, countitem(.@item_id);

		if (.@amount <= 0) {
			set .@skip_count, .@skip_count + 1;
			continue;
		}

		set .@ret, registercostume(.@item_id);

		if (.@ret == 1 || .@ret == 2) {
			delitem .@item_id, .@amount;
			getitem .CostumeCoinID, .@amount;

			if (.@ret == 1)
				set .@new_count, .@new_count + 1;
			else
				set .@already_count, .@already_count + 1;

			set .@coin_count, .@coin_count + .@amount;
			continue;
		}

		set .@fail_count, .@fail_count + 1;
	}

	freeloop(0);

	mes needtr("[의상 등록 / 일괄 코인화]", "[Register / Bulk Convert]");
	mes needtr("처리가 완료되었습니다.", "Done.");
	mes " ";
	mes needtr("신규 도감 등록: ^00AAFF", "Newly registered: ^00AAFF") + .@new_count + needtr("^000000종", "^000000 kinds");
	mes needtr("이미 등록된 의상: ^FF8000", "Already registered: ^FF8000") + .@already_count + needtr("^000000종", "^000000 kinds");
	mes needtr("획득 의상 코인: ^FF8000", "Costume Coins gained: ^FF8000") + .@coin_count + needtr("^000000개", "^000000");

	if (.@skip_count > 0)
		mes needtr("스킵: ^777777", "Skipped: ^777777") + .@skip_count + needtr("^000000종", "^000000 kinds");

	if (.@fail_count > 0)
		mes needtr("실패: ^FF0000", "Failed: ^FF0000") + .@fail_count + needtr("^000000종", "^000000 kinds");

	close;

L_ClosetWear:
	deletearray .@slot_list[0], getarraysize(.@slot_list);
	deletearray .@cid_list[0], getarraysize(.@cid_list);

	set .@slot_count, callsub(L_ClosetGetSlotCount);
	set .@count, 0;

	// 옷장 슬롯 전체 검사 중 중첩 callsub로 infinity 보호 한도에 걸리지 않도록 허용
	freeloop(1);

	for (set .@slot_idx, 0; .@slot_idx < .@slot_count; set .@slot_idx, .@slot_idx + 1) {
		set .@slot_cid, callsub(L_ClosetGetSlotValue, .@slot_idx);

		if (.@slot_cid <= 0)
			continue;

		set .@item_id, getcostumeitemid(.@slot_cid);

		if (.@item_id <= 0)
			continue;

		set .@item_part, callsub(L_ClosetPartCheck, .@item_id);

		if (.@item_part != .@closet_part)
			continue;

		set .@slot_list[.@count], .@slot_idx;
		set .@cid_list[.@count], .@slot_cid;
		set .@count, .@count + 1;
	}

	freeloop(0);

	if (.@count <= 0) {
		mes needtr("[옷장 입기]", "[Wear from Wardrobe]");
		mes needtr("착용할 수 있는 옷장 의상이 없습니다.", "You have no wardrobe costume to wear.");
		close;
	}

	mes needtr("[옷장 입기 - ", "[Wear from Wardrobe - ") + .@closet_part_name$ + "]";
	mes needtr("입을 의상을 선택해주세요.", "Please choose the costume to wear.");
	mes " ";

	set .@i, 0;
	while (.@i < .@count) {
		set .@cid, .@cid_list[.@i];
		mes (.@i + 1) + ". [" + .@cid + "] " + getcostumename(.@cid);
		set .@i, .@i + 1;
	}

	next;

	mes needtr("[옷장 입기 - ", "[Wear from Wardrobe - ") + .@closet_part_name$ + "]";
	mes needtr("입을 번호를 입력해주세요.", "Please enter the number to wear.");
	mes "1 ~ " + .@count + needtr("번까지 입력 가능합니다.", " may be entered.");
	next;

	input .@input_no;

	if (.@input_no < 1 || .@input_no > .@count) {
		mes needtr("[옷장 입기]", "[Wear from Wardrobe]");
		mes needtr("잘못된 번호입니다.", "That number is not valid.");
		close;
	}

	set .@idx, .@input_no - 1;
	set .@cid, .@cid_list[.@idx];
	set .@item_id, getcostumeitemid(.@cid);

	if (.@item_id <= 0) {
		mes needtr("[옷장 입기]", "[Wear from Wardrobe]");
		mes needtr("아이템 정보를 찾을 수 없습니다.", "The item data could not be found.");
		close;
	}

	goto L_PutCloset;
		
L_PutCloset:
	if (countitem(.@item_id) > 0) {
		mes needtr("[의상 옷장]", "[Costume Wardrobe]");
		mes needtr("동일한 의상을 이미 보유 중입니다.", "You already hold the same costume.");
		mes needtr("인벤토리 또는 장비창의 동일 의상을 정리한 후 다시 시도해주세요.", "Clear the duplicate from your inventory or gear and try again.");
		close;
	}
	set .@loc, getiteminfo(.@item_id, ITEMINFO_LOCATIONS);

	if (.@loc & EQP_COSTUME_HEAD_TOP) {
		set .@part, EQP_COSTUME_HEAD_TOP;
		set .@old_cid, COSTUME_RENT_TOP;
	} else if (.@loc & EQP_COSTUME_HEAD_MID) {
		set .@part, EQP_COSTUME_HEAD_MID;
		set .@old_cid, COSTUME_RENT_MID;
	} else if (.@loc & EQP_COSTUME_HEAD_LOW) {
		set .@part, EQP_COSTUME_HEAD_LOW;
		set .@old_cid, COSTUME_RENT_LOW;
	} else if (.@loc & EQP_COSTUME_GARMENT) {
		set .@part, EQP_COSTUME_GARMENT;
		set .@old_cid, COSTUME_RENT_GARMENT;
	} else {
		mes needtr("[의상 옷장]", "[Costume Wardrobe]");
		mes needtr("옷장에 넣을 수 없는 의상입니다.", "This costume cannot go into the wardrobe.");
		close;
	}

	if (.@old_cid > 0) {
		set .@old_item_id, getcostumeitemid(.@old_cid);

		if (.@old_item_id > 0) {
			set .@bound_total, countbound(.ClosetBoundType);
			set .@old_bound_amount, 0;

			for (set .@i, 0; .@i < .@bound_total; set .@i, .@i + 1) {
				if (@bound_items[.@i] == .@old_item_id) {
					set .@old_bound_amount, @bound_amount[.@i];
					break;
				}
			}

			if (.@old_bound_amount > 0 && .@old_bound_amount == countitem(.@old_item_id)) {
				delitem .@old_item_id, 1;
			}
		}
	}

	set .@stone_card_id, 0;
	set .@effect_card_id, 0;

	if (.@part == EQP_COSTUME_HEAD_TOP)
	{
		set .@stone_card_id, COSTUME_STONE_TOP;
		set .@effect_card_id, COSTUME_EFFECT_TOP;
	    COSTUME_RENT_TOP = .@cid;
	}
	else if (.@part == EQP_COSTUME_HEAD_MID)
	{
		set .@stone_card_id, COSTUME_STONE_MID;
		set .@effect_card_id, COSTUME_EFFECT_MID;
	    COSTUME_RENT_MID = .@cid;
	}
	else if (.@part == EQP_COSTUME_HEAD_LOW)
	{
	    COSTUME_RENT_LOW = .@cid;
		set .@stone_card_id, COSTUME_STONE_LOW;
		set .@effect_card_id, COSTUME_EFFECT_LOW;
	}
	else if (.@part == EQP_COSTUME_GARMENT)
	{
		set .@stone_card_id, COSTUME_STONE_GARMENT;
		set .@effect_card_id, COSTUME_EFFECT_GARMENT;
		COSTUME_RENT_GARMENT = .@cid;
	}

	if (.@part == EQP_COSTUME_GARMENT)
	{
		// 걸칠것: 일반 card[0] + 듀얼 card[1] + 이펙트 card[3]
		getitembound2 .@item_id,1,1,0,0,COSTUME_STONE_GARMENT,COSTUME_STONE_DUAL,0,COSTUME_EFFECT_GARMENT,.ClosetBoundType;
	}
	else
	{
		// 일반 스톤 = card[0], 이펙트 스톤 = card[3]
		getitembound2 .@item_id, 1, 1, 0, 0, .@stone_card_id, 0, 0, .@effect_card_id, .ClosetBoundType;
	}
		
	mes needtr("[의상 옷장]", "[Costume Wardrobe]");
	mes needtr("옷장 의상이 지급되었습니다.", "The wardrobe costume has been delivered.");
	mes needtr("지급된 의상을 직접 착용해주세요.", "Please equip it yourself.");
	close;

L_ViewBookPage:
	set .@start, .@page * .PageSize + 1;
	set .@end, .@start + .PageSize - 1;
	if (.@end > .@last)
		set .@end, .@last;

	mes needtr("[의상 도감]", "[Costume Album]");
	mes needtr("진행도: ", "Progress: ") + getcostumeregistercount() + " / " + getcostumeactivecount();
	mes needtr("페이지: ", "Page: ") + (.@page + 1) + " / " + (.@max_page + 1);
	mes " ";
	for (set .@cid, .@start; .@cid <= .@end; set .@cid, .@cid + 1) {
		set .@item_id, getcostumeitemid(.@cid);

		if (.@item_id <= 0)
			continue;

		if (iscostumeregistered(.@cid))
			mes .@cid + ". " + getcostumename(.@cid);
		else
			mes .@cid + ". ???";
	}

	next;

	switch(select(needtr("다음 페이지:이전 페이지:페이지 입력:뒤로", "Next Page:Prev Page:Go to Page:Back"))) {
	case 1:
		if (.@page < .@max_page)
			set .@page, .@page + 1;
		goto L_ViewBookPage;
	case 2:
		if (.@page > 0)
			set .@page, .@page - 1;
		goto L_ViewBookPage;
	case 3:
		mes needtr("[의상 도감]", "[Costume Album]");
		mes needtr("이동할 페이지 번호를 입력해주세요.", "Please enter the page number to jump to.");
		input .@input_page;

		if (.@input_page < 1)
			set .@input_page, 1;
		if (.@input_page > (.@max_page + 1))
			set .@input_page, .@max_page + 1;

		set .@page, .@input_page - 1;
		goto L_ViewBookPage;
	default:
		goto L_Main;
	}

	mes needtr("의상 코인 ", "You received Costume Coin x") + .RewardCoinAmount + needtr("개를 획득했습니다.", ".");

		set .@reward_item_id, .CollectionCompleteRewardPurple;
		set .@reward_name$, needtr("냐르의 보랏빛 발자국(걸칠것)", "Nyar's Purple Footprint(Garment)");

	mes needtr("1개를 선택했습니다.", "You chose 1.");

	mes needtr("이 보상을 수령하시겠습니까?", "Do you want to claim this reward?");

		goto L_RewardComplete;

	close;

	if (@NEED_CLOSET_PREVIEW_ACTIVE)
		callsub L_ClosetPreviewRestore;

	mes needtr("[의상 검색 / 미리보기]", "[Search / Preview]");
	mes needtr("도감에 등록한 의상을 검색합니다.", "Searches the costumes registered in your album.");
	mes " ";
	mes needtr("검색 가능한 항목:", "You can search by:");
	mes needtr("- 의상 이름 일부", "- part of the costume name");
	mes needtr("- 아이템 번호", "- item number");
	mes needtr("- 도감 번호", "- album number");
	mes " ";
	mes needtr("예) 대천사 / 20765 / 120", "e.g. Archangel / 20765 / 120");
	next;

	input .@search_word$;

	if (.@search_word$ == "") {
		mes needtr("[의상 검색]", "[Costume Search]");
		mes needtr("검색어를 입력해주세요.", "Please enter a search term.");
		close;
	}

	deletearray .@search_cid_list[0], getarraysize(.@search_cid_list);
	deletearray .@search_item_list[0], getarraysize(.@search_item_list);

	set .@search_count, 0;
	set .@search_truncated, 0;
	set .@search_number, atoi(.@search_word$);
	set .@search_lower$, strtolower(.@search_word$);
	set .@last_cid, getcostumelastcollectionid();

	freeloop(1);

	for (set .@search_cid, 1; .@search_cid <= .@last_cid; set .@search_cid, .@search_cid + 1) {
		if (!iscostumeregistered(.@search_cid))
			continue;

		set .@search_item_id, getcostumeitemid(.@search_cid);

		if (.@search_item_id <= 0)
			continue;

		// 옷장에서 사용할 수 있는 의상 부위만 검색
		if (callsub(L_ClosetPartCheck, .@search_item_id) <= 0)
			continue;

		set .@search_name$, getcostumename(.@search_cid);
		set .@search_match, 0;

		if (strpos(strtolower(.@search_name$), .@search_lower$) >= 0)
			set .@search_match, 1;

		if (.@search_number > 0) {
			if (.@search_cid == .@search_number || .@search_item_id == .@search_number)
				set .@search_match, 1;
		}

		if (!.@search_match)
			continue;

		set .@search_cid_list[.@search_count], .@search_cid;
		set .@search_item_list[.@search_count], .@search_item_id;
		set .@search_count, .@search_count + 1;

		if (.@search_count >= .SearchMaxResults) {
			set .@search_truncated, 1;
			break;
		}
	}

	freeloop(0);

	if (.@search_count <= 0) {
		mes needtr("[의상 검색]", "[Costume Search]");
		mes needtr("검색 결과가 없습니다.", "No result has been found.");
		mes " ";
		mes needtr("도감에 등록한 의상만 검색됩니다.", "Only costumes registered in your album are searched.");
		next;

		if (select(needtr("다시 검색:뒤로", "Search Again:Back")) == 1)
			goto L_ClosetSearch;

		goto L_Closet;
	}

	set .@search_page, 0;
	goto L_ClosetSearchPage;

L_ClosetSearchPage:
	set .@search_max_page, (.@search_count - 1) / .PageSize;
	set .@search_start, .@search_page * .PageSize;
	set .@search_end, .@search_start + .PageSize;

	if (.@search_end > .@search_count)
		set .@search_end, .@search_count;

	set .@search_page_count, .@search_end - .@search_start;

	mes needtr("[의상 검색 결과]", "[Search Results]");
	mes needtr("검색어: ^0055FF", "Search term: ^0055FF") + .@search_word$ + "^000000";
	mes needtr("검색 결과: ^FF8000", "Results: ^FF8000") + .@search_count + needtr("^000000개", "^000000");
	mes needtr("페이지: ", "Page: ") + (.@search_page + 1) + " / " + (.@search_max_page + 1);

	if (.@search_truncated)
		mes needtr("^FF0000※ 결과가 많아 ", "^FF0000* Too many results - showing only the first ") + .SearchMaxResults + needtr("개까지만 표시합니다.^000000", ".^000000");

	mes " ";

	// 페이지 출력 중 각 항목마다 옷장 슬롯을 재검사하므로 infinity 보호 한도 완화
	freeloop(1);

	for (set .@search_i, .@search_start; .@search_i < .@search_end; set .@search_i, .@search_i + 1) {
		set .@search_line_no, .@search_i - .@search_start + 1;
		set .@search_cid, .@search_cid_list[.@search_i];
		set .@search_item_id, .@search_item_list[.@search_i];
		set .@search_closet_mark$, "";

		if (callsub(L_ClosetIsRegistered, .@search_cid))
			set .@search_closet_mark$, needtr(" ^00AA00[옷장 등록]^000000", " ^00AA00[in wardrobe]^000000");

		mes .@search_line_no + ". [" + .@search_cid + "/" + .@search_item_id + "] " + getcostumename(.@search_cid) + .@search_closet_mark$;
	}

	freeloop(0);

	next;

	switch(select(needtr("번호 선택:다음 페이지:이전 페이지:다시 검색:뒤로", "Pick a Number:Next Page:Prev Page:Search Again:Back"))) {
	case 1:
		mes needtr("[의상 검색 결과]", "[Search Results]");
		mes needtr("확인할 번호를 입력해주세요.", "Please enter the number to inspect.");
		mes "1 ~ " + .@search_page_count + needtr("번까지 입력 가능합니다.", " may be entered.");
		next;

		input .@search_input_no;

		if (.@search_input_no < 1 || .@search_input_no > .@search_page_count) {
			mes needtr("[의상 검색]", "[Costume Search]");
			mes needtr("잘못된 번호입니다.", "That number is not valid.");
			close;
		}

		set .@search_selected_idx, .@search_start + .@search_input_no - 1;
		set .@cid, .@search_cid_list[.@search_selected_idx];
		set .@item_id, .@search_item_list[.@search_selected_idx];
		goto L_ClosetSearchResult;

	case 2:
		if (.@search_page < .@search_max_page)
			set .@search_page, .@search_page + 1;
		goto L_ClosetSearchPage;

	case 3:
		if (.@search_page > 0)
			set .@search_page, .@search_page - 1;
		goto L_ClosetSearchPage;

	case 4:
		goto L_ClosetSearch;

	default:
		goto L_Closet;
	}

L_ClosetSearchResult:
	set .@item_id, getcostumeitemid(.@cid);

	if (.@item_id <= 0 || !iscostumeregistered(.@cid)) {
		if (@NEED_CLOSET_PREVIEW_ACTIVE)
			callsub L_ClosetPreviewRestore;

		mes needtr("[의상 검색]", "[Costume Search]");
		mes needtr("의상 정보를 다시 확인할 수 없습니다.", "The costume data could not be read again.");
		close;
	}

	set .@closet_part, callsub(L_ClosetPartCheck, .@item_id);

	if (.@closet_part == EQP_COSTUME_HEAD_TOP)
		set .@closet_part_name$, needtr("상단", "Upper");
	else if (.@closet_part == EQP_COSTUME_HEAD_MID)
		set .@closet_part_name$, needtr("중단", "Middle");
	else if (.@closet_part == EQP_COSTUME_HEAD_LOW)
		set .@closet_part_name$, needtr("하단", "Lower");
	else if (.@closet_part == EQP_COSTUME_GARMENT)
		set .@closet_part_name$, needtr("걸칠것", "Garment");
	else
		set .@closet_part_name$, needtr("확인 불가", "Unknown");

	set .@is_in_closet, callsub(L_ClosetIsRegistered, .@cid);

	mes needtr("[의상 검색 결과]", "[Search Results]");
	mes "^FF8000[" + .@cid + "] " + getcostumename(.@cid) + "^000000";
	mes needtr("아이템 번호: ", "Item number: ") + .@item_id;
	mes needtr("부위: ", "Slot: ") + .@closet_part_name$;
	mes needtr("옷장 등록: ", "In wardrobe: ") + (.@is_in_closet ? needtr("^00AA00등록됨^000000", "^00AA00yes^000000") : needtr("^777777미등록^000000", "^777777no^000000"));

	if (@NEED_CLOSET_PREVIEW_ACTIVE)
		mes needtr("^0055FF현재 이 의상을 미리보기 중입니다.^000000", "^0055FFYou are previewing this costume.^000000");

	next;

	set .@search_action_menu$, needtr("미리보기:", "Preview:");

	if (.@is_in_closet)
		set .@search_action_menu$, .@search_action_menu$ + needtr("옷 입기:", "Wear:");
	else
		set .@search_action_menu$, .@search_action_menu$ + needtr("옷장 등록:", "Store:");

	set .@search_action_menu$, .@search_action_menu$ + needtr("검색 결과로:다시 검색:뒤로", "Back to Results:Search Again:Back");
	set .@search_action, select(.@search_action_menu$);

	switch(.@search_action) {
	case 1:
		set .@preview_result, callsub(L_ClosetPreviewApply, .@cid);

		mes needtr("[의상 미리보기]", "[Costume Preview]");

		if (.@preview_result <= 0) {
			mes needtr("이 의상은 미리보기를 적용할 수 없습니다.", "This costume cannot be previewed.");
			mes needtr("아이템 View 또는 착용 부위 설정을 확인해주세요.", "Check the item's View id and equip slot settings.");
			next;
			goto L_ClosetSearchResult;
		}

		mes getcostumename(.@cid) + needtr(" 미리보기를 적용했습니다.", " preview applied.");
		mes " ";
		mes "^0055FF" + (.PreviewDuration / 1000) + needtr("초 후 자동으로 원래 모습으로 돌아갑니다.^000000", " seconds until your look reverts automatically.^000000");
		mes needtr("다른 메뉴를 선택하면 즉시 복원됩니다.", "Choosing another menu restores it immediately.");
		next;
		goto L_ClosetSearchResult;

	case 2:
		if (@NEED_CLOSET_PREVIEW_ACTIVE)
			callsub L_ClosetPreviewRestore;

		if (.@is_in_closet)
			goto L_PutCloset;

		set .@closet_register_from_search, 1;
		set .@closet_register_from_part, 0;
		goto L_ClosetRegisterSelected;

	case 3:
		if (@NEED_CLOSET_PREVIEW_ACTIVE)
			callsub L_ClosetPreviewRestore;
		goto L_ClosetSearchPage;

	case 4:
		if (@NEED_CLOSET_PREVIEW_ACTIVE)
			callsub L_ClosetPreviewRestore;
		goto L_ClosetSearch;

	default:
		if (@NEED_CLOSET_PREVIEW_ACTIVE)
			callsub L_ClosetPreviewRestore;
		goto L_Closet;
	}

L_ClosetPreviewApply:
	set .@pv_cid, getarg(0);
	set .@pv_item_id, getcostumeitemid(.@pv_cid);

	if (.@pv_item_id <= 0)
		return 0;

	set .@pv_view, getiteminfo(.@pv_item_id, ITEMINFO_VIEW);
	set .@pv_loc, getiteminfo(.@pv_item_id, ITEMINFO_LOCATIONS);
	set .@pv_look_type, -1;

	if (.@pv_view <= 0)
		return 0;

	// rAthena 기본 미리보기 방식과 동일한 우선순위
	if (.@pv_loc & EQP_COSTUME_HEAD_LOW)
		set .@pv_look_type, LOOK_HEAD_BOTTOM;
	else if (.@pv_loc & EQP_COSTUME_HEAD_TOP)
		set .@pv_look_type, LOOK_HEAD_TOP;
	else if (.@pv_loc & EQP_COSTUME_HEAD_MID)
		set .@pv_look_type, LOOK_HEAD_MID;
	else if (.@pv_loc & EQP_COSTUME_GARMENT)
		set .@pv_look_type, LOOK_ROBE;

	if (.@pv_look_type < 0)
		return 0;

	// 기존 미리보기가 있으면 먼저 원상복구
	if (@NEED_CLOSET_PREVIEW_ACTIVE)
		callsub L_ClosetPreviewRestore;

	set @NEED_CLOSET_PREVIEW_ACTIVE, 1;
	set @NEED_CLOSET_PREVIEW_LOOK_TYPE, .@pv_look_type;
	set @NEED_CLOSET_PREVIEW_OLD_VIEW, getlook(.@pv_look_type);
	set @NEED_CLOSET_PREVIEW_NEW_VIEW, .@pv_view;

	changelook .@pv_look_type, .@pv_view;

	deltimer strnpcinfo(3) + "::OnCostumePreviewEnd";
	addtimer .PreviewDuration, strnpcinfo(3) + "::OnCostumePreviewEnd";

	return 1;

L_ClosetPreviewRestore:
	if (!@NEED_CLOSET_PREVIEW_ACTIVE)
		return;

	deltimer strnpcinfo(3) + "::OnCostumePreviewEnd";

	changelook @NEED_CLOSET_PREVIEW_LOOK_TYPE, @NEED_CLOSET_PREVIEW_OLD_VIEW;

	set @NEED_CLOSET_PREVIEW_ACTIVE, 0;
	set @NEED_CLOSET_PREVIEW_LOOK_TYPE, 0;
	set @NEED_CLOSET_PREVIEW_OLD_VIEW, 0;
	set @NEED_CLOSET_PREVIEW_NEW_VIEW, 0;
	return;

L_IsCostumeEquipped:
	set .@check_item_id, getarg(0);

	if (.@check_item_id <= 0)
		return 0;

	if (getequipid(EQI_COSTUME_HEAD_TOP) == .@check_item_id)
		return 1;

	if (getequipid(EQI_COSTUME_HEAD_MID) == .@check_item_id)
		return 1;

	if (getequipid(EQI_COSTUME_HEAD_LOW) == .@check_item_id)
		return 1;

	if (getequipid(EQI_COSTUME_GARMENT) == .@check_item_id)
		return 1;

	return 0;

L_ClosetGetSlotValue:
	return #NEED_CLOSET[getarg(0)];

L_ClosetSetSlotValue:
	set #NEED_CLOSET[getarg(0)], getarg(1);
	return;

L_ClosetIsRegistered:
	set .@check_cid, getarg(0);
	set .@slot_count, callsub(L_ClosetGetSlotCount);

	for (set .@slot_idx, 0; .@slot_idx < .@slot_count; set .@slot_idx, .@slot_idx + 1) {
		if (callsub(L_ClosetGetSlotValue, .@slot_idx) == .@check_cid)
			return 1;
	}

	return 0;

L_ClosetFindEmptySlot:
	set .@slot_count, callsub(L_ClosetGetSlotCount);

	for (set .@i, 0; .@i < .@slot_count; set .@i, .@i + 1) {
		if (callsub(L_ClosetGetSlotValue, .@i) <= 0)
			return .@i;
	}

	return -1;

L_ClosetPartCheck:
	set .@check_item_id, getarg(0);

	if (.@check_item_id <= 0)
		return 0;

	set .@loc, getiteminfo(.@check_item_id, ITEMINFO_LOCATIONS);

	if (.@loc <= 0)
		return 0;

	if (.@loc & EQP_COSTUME_HEAD_TOP)
		return EQP_COSTUME_HEAD_TOP;
	else if (.@loc & EQP_COSTUME_HEAD_MID)
		return EQP_COSTUME_HEAD_MID;
	else if (.@loc & EQP_COSTUME_HEAD_LOW)
		return EQP_COSTUME_HEAD_LOW;
	else if (.@loc & EQP_COSTUME_GARMENT)
		return EQP_COSTUME_GARMENT;

	return 0;

L_ClosetRegister:
	// 검색 또는 이전 선택에서 남은 미리보기가 있으면 먼저 복원
	if (@NEED_CLOSET_PREVIEW_ACTIVE)
		callsub L_ClosetPreviewRestore;

	set .@reg_count, getregisteredcostumelist(0, .@closet_part);
	set .@page, 0;

	if (.@reg_count <= 0) {
		mes needtr("[옷장 등록]", "[Store in Wardrobe]");
		mes needtr("등록된 해당 부위 의상이 없습니다.", "You have no registered costume for that slot.");
		close;
	}

	goto L_ClosetRegisterPage;

L_ClosetRegisterPage:
	// 미리보기 화면에서 목록으로 돌아온 경우 원래 모습으로 복원
	if (@NEED_CLOSET_PREVIEW_ACTIVE)
		callsub L_ClosetPreviewRestore;

	set .@max_page, (.@reg_count - 1) / .PageSize;
	set .@start, .@page * .PageSize;
	set .@end, .@start + .PageSize;

	if (.@end > .@reg_count)
		set .@end, .@reg_count;

	set .@page_count, .@end - .@start;

	mes needtr("[옷장 등록 - ", "[Store in Wardrobe - ") + .@closet_part_name$ + "]";
	mes needtr("페이지: ", "Page: ") + (.@page + 1) + " / " + (.@max_page + 1);
	mes needtr("미리보기 또는 등록할 번호를 선택해주세요.", "Choose a number to preview or store.");
	mes " ";

	// 한 페이지의 각 의상마다 L_ClosetIsRegistered가 전체 슬롯을 순회하므로
	// 실제 종료되는 반복이라도 rAthena infinity 보호 한도를 넘을 수 있어 허용
	freeloop(1);

	set .@i, .@start;
	while (.@i < .@end) {
		set .@cid, @costume_collection_ids[.@i];
		set .@line_no, .@i - .@start + 1;
		set .@closet_mark$, "";

		if (callsub(L_ClosetIsRegistered, .@cid))
			set .@closet_mark$, needtr(" ^00AA00[옷장 등록]^000000", " ^00AA00[in wardrobe]^000000");

		mes .@line_no + ". [" + .@cid + "] " + getcostumename(.@cid) + .@closet_mark$;

		set .@i, .@i + 1;
	}

	freeloop(0);

	next;

	switch(select(needtr("번호 선택:다음 페이지:이전 페이지:취소", "Pick a Number:Next Page:Prev Page:Cancel"))) {
	case 1:
		mes needtr("[옷장 등록 - ", "[Store in Wardrobe - ") + .@closet_part_name$ + "]";
		mes needtr("확인할 번호를 입력해주세요.", "Please enter the number to inspect.");
		mes "1 ~ " + .@page_count + needtr("번까지 입력 가능합니다.", " may be entered.");
		next;

		input .@input_no;

		if (.@input_no < 1 || .@input_no > .@page_count) {
			mes needtr("[옷장 등록]", "[Store in Wardrobe]");
			mes needtr("잘못된 번호입니다.", "That number is not valid.");
			next;
			goto L_ClosetRegisterPage;
		}

		set .@idx, .@start + .@input_no - 1;
		break;

	case 2:
		if (.@page < .@max_page)
			set .@page, .@page + 1;
		goto L_ClosetRegisterPage;

	case 3:
		if (.@page > 0)
			set .@page, .@page - 1;
		goto L_ClosetRegisterPage;

	default:
		goto L_ClosetPartMenu;
	}

	set .@cid, @costume_collection_ids[.@idx];
	set .@item_id, getcostumeitemid(.@cid);

	if (.@item_id <= 0) {
		mes needtr("[옷장 등록]", "[Store in Wardrobe]");
		mes needtr("아이템 정보를 찾을 수 없습니다.", "The item data could not be found.");
		close;
	}

	goto L_ClosetRegisterResult;

L_ClosetRegisterResult:
	set .@item_id, getcostumeitemid(.@cid);

	if (.@item_id <= 0 || !iscostumeregistered(.@cid)) {
		if (@NEED_CLOSET_PREVIEW_ACTIVE)
			callsub L_ClosetPreviewRestore;

		mes needtr("[옷장 등록]", "[Store in Wardrobe]");
		mes needtr("도감에 등록된 의상 정보를 찾을 수 없습니다.", "No album entry was found for that costume.");
		close;
	}

	set .@is_in_closet, callsub(L_ClosetIsRegistered, .@cid);

	mes needtr("[옷장 등록 - ", "[Store in Wardrobe - ") + .@closet_part_name$ + "]";
	mes "^FF8000[" + .@cid + "] " + getcostumename(.@cid) + "^000000";
	mes needtr("아이템 번호: ", "Item number: ") + .@item_id;
	mes needtr("옷장 등록: ", "In wardrobe: ") + (.@is_in_closet ? needtr("^00AA00등록됨^000000", "^00AA00yes^000000") : needtr("^777777미등록^000000", "^777777no^000000"));

	if (@NEED_CLOSET_PREVIEW_ACTIVE)
		mes needtr("^0055FF현재 이 의상을 미리보기 중입니다.^000000", "^0055FFYou are previewing this costume.^000000");

	next;

	set .@register_action_menu$, needtr("미리보기:", "Preview:");

	if (.@is_in_closet)
		set .@register_action_menu$, .@register_action_menu$ + needtr("옷 입기:", "Wear:");
	else
		set .@register_action_menu$, .@register_action_menu$ + needtr("옷장 등록:", "Store:");

	set .@register_action_menu$, .@register_action_menu$ + needtr("목록으로:부위 메뉴로", "Back to List:Slot Menu");
	set .@register_action, select(.@register_action_menu$);

	switch(.@register_action) {
	case 1:
		set .@preview_result, callsub(L_ClosetPreviewApply, .@cid);

		mes needtr("[의상 미리보기]", "[Costume Preview]");

		if (.@preview_result <= 0) {
			mes needtr("이 의상은 미리보기를 적용할 수 없습니다.", "This costume cannot be previewed.");
			mes needtr("아이템 View 또는 착용 부위 설정을 확인해주세요.", "Check the item's View id and equip slot settings.");
			next;
			goto L_ClosetRegisterResult;
		}

		mes getcostumename(.@cid) + needtr(" 미리보기를 적용했습니다.", " preview applied.");
		mes " ";
		mes "^0055FF" + (.PreviewDuration / 1000) + needtr("초 후 자동으로 원래 모습으로 돌아갑니다.^000000", " seconds until your look reverts automatically.^000000");
		mes needtr("목록이나 다른 메뉴로 이동하면 즉시 복원됩니다.", "Going back to the list or another menu restores it at once.");
		next;
		goto L_ClosetRegisterResult;

	case 2:
		if (@NEED_CLOSET_PREVIEW_ACTIVE)
			callsub L_ClosetPreviewRestore;

		if (.@is_in_closet)
			goto L_PutCloset;

		set .@closet_register_from_search, 0;
		set .@closet_register_from_part, 1;
		goto L_ClosetRegisterSelected;

	case 3:
		if (@NEED_CLOSET_PREVIEW_ACTIVE)
			callsub L_ClosetPreviewRestore;
		goto L_ClosetRegisterPage;

	default:
		if (@NEED_CLOSET_PREVIEW_ACTIVE)
			callsub L_ClosetPreviewRestore;
		goto L_ClosetPartMenu;
	}

L_ClosetRegisterSelected:
	set .@item_id, getcostumeitemid(.@cid);

	if (.@item_id <= 0 || !iscostumeregistered(.@cid)) {
		mes needtr("[옷장 등록]", "[Store in Wardrobe]");
		mes needtr("도감에 등록된 의상 정보를 찾을 수 없습니다.", "No album entry was found for that costume.");
		close;
	}

	set .@empty_slot, callsub(L_ClosetFindEmptySlot);

	if (.@empty_slot < 0) {
		set .@slot_count, callsub(L_ClosetGetSlotCount);

		mes needtr("[옷장 등록]", "[Store in Wardrobe]");
		mes needtr("옷장 슬롯이 가득 찼습니다.", "The wardrobe is full.");
		mes needtr("현재 슬롯: ", "Slots: ") + .@slot_count + " / " + .@slot_count;
		mes needtr("슬롯 확장이 필요합니다.", "You need to expand your slots.");
		close;
	}

	mes needtr("[옷장 등록]", "[Store in Wardrobe]");
	mes "[" + .@cid + "] " + getcostumename(.@cid);
	mes needtr("이 의상을 옷장에 등록하시겠습니까?", "Store this costume in the wardrobe?");
	mes needtr("등록 비용: ", "Storage cost: ") + callfunc("F_InsertComma", .ClosetRegisterZeny) + " Zeny";
	next;

	if (select(needtr("등록한다:취소", "Store:Cancel")) == 2) {
		if (.@closet_register_from_search)
			goto L_ClosetSearchResult;

		if (.@closet_register_from_part)
			goto L_ClosetRegisterResult;

		goto L_ClosetPartMenu;
	}

	if (Zeny < .ClosetRegisterZeny) {
		mes needtr("[옷장 등록]", "[Store in Wardrobe]");
		mes needtr("Zeny가 부족합니다.", "You do not have enough Zeny.");
		close;
	}

	if (callsub(L_ClosetIsRegistered, .@cid)) {
		mes needtr("[옷장 등록]", "[Store in Wardrobe]");
		mes needtr("이미 옷장에 등록된 의상입니다.", "This costume is already in the wardrobe.");
		close;
	}

	set .@empty_slot, callsub(L_ClosetFindEmptySlot);

	if (.@empty_slot < 0) {
		mes needtr("[옷장 등록]", "[Store in Wardrobe]");
		mes needtr("옷장 슬롯이 가득 찼습니다.", "The wardrobe is full.");
		close;
	}

	set Zeny, Zeny - .ClosetRegisterZeny;
	callsub L_ClosetSetSlotValue, .@empty_slot, .@cid;

	mes needtr("[옷장 등록]", "[Store in Wardrobe]");
	mes needtr("옷장에 등록되었습니다.", "Stored in the wardrobe.");
	close;

L_ClosetView:
	mes needtr("[옷장 보기 - ", "[Wardrobe - ") + .@closet_part_name$ + "]";

	set .@used_total, 0;
	set .@used_part, 0;
	set .@slot_count, callsub(L_ClosetGetSlotCount);

	for (set .@slot_idx, 0; .@slot_idx < .@slot_count; set .@slot_idx, .@slot_idx + 1) {
		set .@slot_cid, callsub(L_ClosetGetSlotValue, .@slot_idx);

		if (.@slot_cid <= 0)
			continue;

		set .@used_total, .@used_total + 1;
		set .@item_id, getcostumeitemid(.@slot_cid);

		if (.@item_id <= 0)
			continue;

		set .@item_part, callsub(L_ClosetPartCheck, .@item_id);

		if (.@item_part != .@closet_part)
			continue;

		set .@used_part, .@used_part + 1;
		mes (.@slot_idx + 1) + ". [" + .@slot_cid + "] " + getcostumename(.@slot_cid);
	}

	if (.@used_part <= 0)
		mes needtr("해당 부위에 등록된 옷장 의상이 없습니다.", "No wardrobe costume is stored for that slot.");

	set .@slot_count, callsub(L_ClosetGetSlotCount);

	mes " ";
	mes needtr("현재 부위 등록 수: ", "Stored in this slot: ") + .@used_part + needtr("개", "");
	mes needtr("전체 옷장 사용: ", "Wardrobe used: ") + .@used_total + " / " + .@slot_count;
	close;

L_ClosetRemove:
	deletearray .@slot_list[0], getarraysize(.@slot_list);
	deletearray .@cid_list[0], getarraysize(.@cid_list);

	set .@slot_count, callsub(L_ClosetGetSlotCount);
	set .@count, 0;

	for (set .@slot_idx, 0; .@slot_idx < .@slot_count; set .@slot_idx, .@slot_idx + 1) {
		set .@slot_cid, callsub(L_ClosetGetSlotValue, .@slot_idx);

		if (.@slot_cid <= 0)
			continue;

		set .@item_id, getcostumeitemid(.@slot_cid);

		if (.@item_id <= 0)
			continue;

		set .@item_part, callsub(L_ClosetPartCheck, .@item_id);

		if (.@item_part != .@closet_part)
			continue;

		set .@slot_list[.@count], .@slot_idx;
		set .@cid_list[.@count], .@slot_cid;
		set .@count, .@count + 1;
	}

	if (.@count <= 0) {
		mes needtr("[옷장 빼기]", "[Remove from Wardrobe]");
		mes needtr("제거할 옷장 의상이 없습니다.", "There is no wardrobe costume to remove.");
		close;
	}

	mes needtr("[옷장 빼기 - ", "[Remove from Wardrobe - ") + .@closet_part_name$ + "]";
	mes needtr("옷장에서 제거할 의상을 선택해주세요.", "Choose the costume to remove from the wardrobe.");
	mes " ";

	set .@menu$, "";

	for (set .@i, 0; .@i < .@count; set .@i, .@i + 1) {
		set .@cid, .@cid_list[.@i];
		set .@menu$, .@menu$ + "[" + .@cid + "] " + getcostumename(.@cid) + ":";
		mes (.@i + 1) + ". [" + .@cid + "] " + getcostumename(.@cid);
	}

	set .@menu$, .@menu$ + needtr("취소", "Cancel");
	next;

	set .@sel, select(.@menu$);

	if (.@sel > .@count)
		goto L_ClosetPartMenu;

	set .@idx, .@sel - 1;
	set .@slot_idx, .@slot_list[.@idx];
	set .@cid, .@cid_list[.@idx];

	mes needtr("[옷장 빼기]", "[Remove from Wardrobe]");
	mes "[" + .@cid + "] " + getcostumename(.@cid);
	mes needtr("옷장에서 제거하시겠습니까?", "Remove it from the wardrobe?");
	next;

	if (select(needtr("제거한다:취소", "Remove:Cancel")) == 2)
		goto L_ClosetPartMenu;

	set .@set_cid, 0;
	callsub L_ClosetSetSlotValue, .@slot_idx, 0;

	mes needtr("[옷장 빼기]", "[Remove from Wardrobe]");
	mes needtr("옷장에서 제거되었습니다.", "Removed from the wardrobe.");
	close;

L_ClosetGetUsedCount:
	set .@slot_count, callsub(L_ClosetGetSlotCount);
	set .@used, 0;

	for (set .@i, 0; .@i < .@slot_count; set .@i, .@i + 1) {
		if (callsub(L_ClosetGetSlotValue, .@i) > 0)
			set .@used, .@used + 1;
	}

	return .@used;

L_ClosetGetSlotCount:
	set .@expand, #NEED_CLOSET_SLOT;

	if (.@expand < 0)
		set .@expand, 0;

	set .@slot_count, .ClosetDefaultSlot + .@expand;

	if (.@slot_count > .ClosetMaxSlot)
		set .@slot_count, .ClosetMaxSlot;

	if (.@slot_count <= 0)
		set .@slot_count, .ClosetDefaultSlot;

	return .@slot_count;
	
L_ClosetGetExpandCost:
	if (.ClosetExpandBaseZeny <= 0)
		set .ClosetExpandBaseZeny, 1000000;

	if (.ClosetExpandTierSlot <= 0)
		set .ClosetExpandTierSlot, 5;

	set .@expand_count, #NEED_CLOSET_SLOT;

	if (.@expand_count < 0)
		set .@expand_count, 0;

	set .@tier, .@expand_count / .ClosetExpandTierSlot;
	set .@expand_cost, .ClosetExpandBaseZeny;

	for (set .@i, 0; .@i < .@tier; set .@i, .@i + 1)
		set .@expand_cost, .@expand_cost * 2;

	return .@expand_cost;

L_ClosetExpand:
	set .@slot_count, callsub(L_ClosetGetSlotCount);
	set .@used_total, callsub(L_ClosetGetUsedCount);
	set .@after_slot_count, .@slot_count + .ClosetExpandTierSlot;

	if (.@after_slot_count > .ClosetMaxSlot)
		set .@after_slot_count, .ClosetMaxSlot;

	if (.@slot_count >= .ClosetMaxSlot) {
		mes needtr("[옷장 슬롯 확장]", "[Expand Wardrobe Slots]");
		mes needtr("이미 최대 슬롯까지 확장했습니다.", "You already expanded to the maximum.");
		mes needtr("현재 슬롯: ", "Slots: ") + .@used_total + " / " + .@slot_count;
		close;
	}

	set .@expand_cost, callsub(L_ClosetGetExpandCost);

	mes needtr("[옷장 슬롯 확장]", "[Expand Wardrobe Slots]");
	mes needtr("현재 슬롯: ^FF8000", "Slots: ^FF8000") + .@used_total + " / " + .@slot_count + "^000000";
	mes needtr("확장 후 현재 슬롯: ^FF8000", "After expanding: ^FF8000") + .@used_total + " / " + .@after_slot_count + "^000000";
	mes " ";
	mes needtr("확장 비용: ^FF8000", "Expansion cost: ^FF8000") + callfunc("F_InsertComma", .@expand_cost) + "^000000 Zeny";
	mes needtr("확장 비용은 ^FF8000", "The cost doubles every ^FF8000") + .ClosetExpandTierSlot + needtr("칸^000000마다 ^FF80002배 증가^000000합니다.", " slots^000000 (^FF8000x2^000000).");
	next;

	if (select(needtr("확장한다:취소", "Expand:Cancel")) == 2)
		goto L_Closet;

	if (Zeny < .@expand_cost) {
		mes needtr("[옷장 슬롯 확장]", "[Expand Wardrobe Slots]");
		mes needtr("Zeny가 부족합니다.", "You do not have enough Zeny.");
		close;
	}

	set Zeny, Zeny - .@expand_cost;
	set .@add_slot_count, .@after_slot_count - .@slot_count;
	set #NEED_CLOSET_SLOT, #NEED_CLOSET_SLOT + .@add_slot_count;

	set .@slot_count, callsub(L_ClosetGetSlotCount);
	
	mes needtr("[옷장 슬롯 확장]", "[Expand Wardrobe Slots]");
	mes needtr("옷장 슬롯이 ", "The wardrobe gained ") + .@add_slot_count + needtr("칸 확장되었습니다.", " slots.");
	mes needtr("현재 슬롯: ", "Slots: ") + .@used_total + " / " + .@slot_count;
	close;

L_StoneMenu:

	else
		mes needtr("중단 : ^777777없음^000000", "Middle : ^777777none^000000");

	mes needtr("^0055FF이펙트 스톤 (100% 성공)^000000", "^0055FFEffect Stone (always succeeds)^000000");
	if (COSTUME_EFFECT_TOP)

	if (COSTUME_EFFECT_LOW)
		mes needtr("하단 이펙트 : ^FF8000", "Lower effect : ^FF8000") + getitemname(COSTUME_EFFECT_LOW) + "^000000";
	else
		mes needtr("하단 이펙트 : ^777777없음^000000", "Lower effect : ^777777none^000000");
	if (COSTUME_EFFECT_GARMENT)
		mes needtr("걸칠것 이펙트 : ^FF8000", "Garment effect : ^FF8000") + getitemname(COSTUME_EFFECT_GARMENT) + "^000000";
	else
		mes needtr("걸칠것 이펙트 : ^777777없음^000000", "Garment effect : ^777777none^000000");

	next;

	switch(select(
		needtr("상단 스톤 부여", "Apply Upper Stone"),
		needtr("중단 스톤 부여", "Apply Middle Stone"),
		needtr("하단 스톤 부여", "Apply Lower Stone"),
		needtr("걸칠것 스톤 부여", "Apply Garment Stone"),
		needtr("듀얼 스톤 부여", "Apply Dual Stone"),
		needtr("이펙트 스톤 부여 (100%)", "Apply Effect Stone (100%)"),
		needtr("종료", "Exit")
	)) {
	case 1:

		set .@stone_part, EQP_COSTUME_HEAD_TOP;

		set .@stone_part, EQP_COSTUME_HEAD_MID;

		set .@stone_rent_cid, COSTUME_RENT_MID;

		set .@stone_part, EQP_COSTUME_HEAD_LOW;
		set .@stone_part_name$, needtr("하단", "Lower");
		set .@stone_rent_cid, COSTUME_RENT_LOW;
		goto L_StoneEnchant;
	case 4:

		set .@stone_part, EQP_COSTUME_GARMENT;
		set .@stone_part_name$, needtr("걸칠것", "Garment");
		set .@stone_rent_cid, COSTUME_RENT_GARMENT;
		goto L_StoneEnchant;
	case 5:
		set .@stone_equip_index, EQI_COSTUME_GARMENT;
		set .@stone_part_name$, needtr("듀얼", "Dual");
		set .@stone_part, 5;
		goto L_StoneEnchantDual;
	case 6:
		goto L_EffectStoneMenu;

	default:
		goto L_Main;

L_StoneEnchant:
	if (.@stone_rent_cid <= 0) {
		mes needtr("[인챈트 스톤]", "[Enchant Stone]");
		mes needtr("현재 ", "You are not wearing a ") + .@stone_part_name$ + needtr(" 옷장 의상을 착용 중이 아닙니다.", " wardrobe costume right now.");
		close;
	}

	set .@equip_item_id, getequipid(.@stone_equip_index);
	set .@rent_item_id, getcostumeitemid(.@stone_rent_cid);

	if (.@equip_item_id <= 0 || .@equip_item_id != .@rent_item_id) {
		mes needtr("[인챈트 스톤]", "[Enchant Stone]");
		mes needtr("현재 착용 중인 ", "The ") + .@stone_part_name$ + needtr(" 의상이", " costume you are wearing");
		mes needtr("옷장 대여 의상과 일치하지 않습니다.", "does not match the wardrobe copy.");
		close;
	}

	deletearray .@stone_idx_list[0], getarraysize(.@stone_idx_list);

	set .@data_size, getarraysize(.StoneData);
	set .@row_count, .@data_size / 3;
	set .@count, 0;

	// StoneData 전체 순회는 데이터 증가 시 infinity 보호 한도에 걸릴 수 있으므로 허용
	freeloop(1);

	for (set .@i, 0; .@i < .@row_count; set .@i, .@i + 1) {
		set .@base, .@i * 3;
		set .@part, .StoneData[.@base];
		set .@card_id, .StoneData[.@base + 1];
		set .@etc_id, .StoneData[.@base + 2];

		if (.@part != .@stone_part)
			continue;

		if (.@card_id <= 0 || .@etc_id <= 0)
			continue;

		if (countitem(.@etc_id) <= 0)
			continue;

		set .@stone_idx_list[.@count], .@i;
		set .@count, .@count + 1;
	}

	freeloop(0);

	if (.@count <= 0) {
		mes needtr("[인챈트 스톤]", "[Enchant Stone]");
		mes needtr("보유 중인 ", "You have no ") + .@stone_part_name$ + needtr(" 인챈트 스톤이 없습니다.", " enchant stone.");
		close;
	}

	mes needtr("[인챈트 스톤 - ", "[Enchant Stone - ") + .@stone_part_name$ + "]";

	set .@etc_id, .StoneData[.@base + 2];

		mes needtr("[인챈트 스톤]", "[Enchant Stone]");
		mes needtr("스톤 매칭 데이터가 올바르지 않습니다.", "The stone matching data is invalid.");
		close;
	}

	set .@old_card_id, getequipcardid(.@stone_equip_index, .StoneCardSlot);

		mes needtr("비용: ", "Cost: ") + callfunc("F_InsertComma", .StoneEnchantCostZeny) + " Zeny";

		mes " ";
		mes needtr("^FF0000이미 인챈트 스톤이 장착되어 있습니다.^000000", "^FF0000An enchant stone is already mounted.^000000");

		close;

	if (.@stone_part == EQP_COSTUME_HEAD_TOP)
		set COSTUME_STONE_TOP, .@card_id;
	else if (.@stone_part == EQP_COSTUME_HEAD_MID)
		set COSTUME_STONE_MID, .@card_id;
	else if (.@stone_part == EQP_COSTUME_HEAD_LOW)
		set COSTUME_STONE_LOW, .@card_id;
	else if (.@stone_part == EQP_COSTUME_GARMENT)
		set COSTUME_STONE_GARMENT, .@card_id;

	specialeffect2 EF_LEVELUP;
	mes needtr("[인챈트 스톤]", "[Enchant Stone]");
	mes needtr("^00AA00스톤 부여에 성공했습니다.^000000", "^00AA00The stone was applied.^000000");
	mes "^FF8000"+getitemname(.@card_id) + needtr("^000000 세팅이 저장되었습니다.", "^000000 setting saved.");
	mes needtr("이후 해당 부위 옷장 의상을 꺼낼 때", "From now on it is applied automatically");
	mes needtr("자동으로 이 스톤이 적용됩니다.", "whenever you take that slot's costume out.");
	next;
	if (.@bound_amount > 0) {
		delitem .@equip_item_id, 1;
		if (.@stone_part == EQP_COSTUME_GARMENT)
		{
			// 일반 걸칠것 스톤을 갱신해도 듀얼과 이펙트는 유지
			getitembound2 .@equip_item_id,1,1,0,0,COSTUME_STONE_GARMENT,COSTUME_STONE_DUAL,0,COSTUME_EFFECT_GARMENT,.ClosetBoundType;
		}
		else
		{
			set .@effect_card_id, 0;

			if (.@stone_part == EQP_COSTUME_HEAD_TOP)
				set .@effect_card_id, COSTUME_EFFECT_TOP;
			else if (.@stone_part == EQP_COSTUME_HEAD_MID)
				set .@effect_card_id, COSTUME_EFFECT_MID;
			else if (.@stone_part == EQP_COSTUME_HEAD_LOW)
				set .@effect_card_id, COSTUME_EFFECT_LOW;

			// 일반 스톤을 갱신해도 기존 이펙트(card[3])는 유지

L_EffectStoneEnchant:

	set .@effect_rent_item_id, getcostumeitemid(.@effect_rent_cid);

		mes needtr("[이펙트 스톤]", "[Effect Stone]");
		mes needtr("현재 착용 중인 ", "The ") + .@effect_part_name$ + needtr(" 의상이", " costume you are wearing");

	set .@effect_data_size, getarraysize(.EffectStoneData);
	set .@effect_row_count, .@effect_data_size / 3;
	set .@effect_count, 0;

	freeloop(1);

	for (set .@i, 0; .@i < .@effect_row_count; set .@i, .@i + 1) {
		set .@base, .@i * 3;
		set .@part, .EffectStoneData[.@base];
		set .@card_id, .EffectStoneData[.@base + 1];
		set .@etc_id, .EffectStoneData[.@base + 2];

		if (.@part != .@effect_part)
			continue;

		if (.@card_id <= 0 || .@etc_id <= 0)
			continue;

		if (countitem(.@etc_id) <= 0)
			continue;

		set .@effect_idx_list[.@effect_count], .@i;
		set .@effect_count, .@effect_count + 1;
	}

	freeloop(0);

	if (.@effect_count <= 0) {
		mes needtr("[이펙트 스톤]", "[Effect Stone]");
		mes needtr("보유 중인 ", "You have no ") + .@effect_part_name$ + needtr(" 이펙트 스톤이 없습니다.", " effect stone.");
		close;
	}

	mes needtr("[이펙트 스톤 - ", "[Effect Stone - ") + .@effect_part_name$ + "]";
	mes needtr("보유 중인 이펙트 스톤", "Effect stones you own");
	mes " ";

	set .@i, 0;
	while (.@i < .@effect_count) {
		set .@row, .@effect_idx_list[.@i];
		set .@base, .@row * 3;
		set .@etc_id, .EffectStoneData[.@base + 2];

		mes (.@i + 1) + ". " + getitemname(.@etc_id) + " x" + countitem(.@etc_id);
		set .@i, .@i + 1;
	}

	next;

	mes needtr("[이펙트 스톤 - ", "[Effect Stone - ") + .@effect_part_name$ + "]";
	mes needtr("부여할 스톤 번호를 입력해주세요.", "Please enter the stone number to apply.");
	mes "1 ~ " + .@effect_count + needtr("번까지 입력 가능합니다.", " may be entered.");
	next;

	input .@input_no;

	if (.@input_no < 1 || .@input_no > .@effect_count) {
		mes needtr("[이펙트 스톤]", "[Effect Stone]");
		mes needtr("잘못된 번호입니다.", "That number is not valid.");
		close;
	}

	set .@row, .@effect_idx_list[.@input_no - 1];
	set .@base, .@row * 3;
	set .@effect_card_id, .EffectStoneData[.@base + 1];
	set .@effect_etc_id, .EffectStoneData[.@base + 2];

	if (.@effect_card_id <= 0 || .@effect_etc_id <= 0) {
		mes needtr("[이펙트 스톤]", "[Effect Stone]");
		mes needtr("이펙트 스톤 매칭 데이터가 올바르지 않습니다.", "The effect stone matching data is invalid.");
		close;
	}

	mes needtr("[이펙트 스톤]", "[Effect Stone]");
	mes needtr("대상 의상:", "Target costume:");
	mes getitemname(.@effect_equip_item_id);
	mes " ";
	mes needtr("사용 스톤:", "Stone used:");
	mes getitemname(.@effect_etc_id);
	mes " ";
	mes needtr("부여 이펙트:", "Effect granted:");
	mes getitemname(.@effect_card_id);
	mes " ";
	mes needtr("성공 확률: ^00AAFF100.00%^000000", "Success rate: ^00AAFF100.00%^000000");

	if (.@effect_old_card_id > 0) {
		mes " ";
		mes needtr("^FF8000현재 이펙트: ", "^FF8000Current effect: ") + getitemname(.@effect_old_card_id) + "^000000";
		mes needtr("^FF0000진행 시 기존 이펙트는 새 이펙트로 교체됩니다.^000000", "^FF0000The old effect will be replaced by the new one.^000000");
	}

	next;

	if (select(needtr("부여한다:취소", "Apply:Cancel")) == 2)
		goto L_EffectStoneMenu;

	// 진행 직전 재검사
	if (countitem(.@effect_etc_id) <= 0) {
		mes needtr("[이펙트 스톤]", "[Effect Stone]");
		mes needtr("선택한 이펙트 스톤을 보유하고 있지 않습니다.", "You do not own the effect stone you chose.");
		close;
	}

	set .@effect_equip_item_id, getequipid(.@effect_equip_index);

	if (.@effect_equip_item_id <= 0 || .@effect_equip_item_id != .@effect_rent_item_id) {
		mes needtr("[이펙트 스톤]", "[Effect Stone]");
		mes needtr("착용 상태가 변경되어 진행할 수 없습니다.", "Your equipment changed so this cannot continue.");
		close;
	}

	// 옷장 귀속 의상인지 먼저 확인하고 난 뒤 스톤을 소비
	unequip .@effect_equip_index;

	set .@bound_total, countbound(.ClosetBoundType);
	set .@bound_amount, 0;

	for (set .@i, 0; .@i < .@bound_total; set .@i, .@i + 1) {
		if (@bound_items[.@i] == .@effect_equip_item_id) {
			set .@bound_amount, @bound_amount[.@i];
			break;
		}
	}

	if (.@bound_amount <= 0) {
		mes needtr("[이펙트 스톤]", "[Effect Stone]");
		mes needtr("옷장 의상을 회수할 수 없습니다.", "The wardrobe costume could not be taken back.");
		mes needtr("일반 의상 보호를 위해 진행을 중단합니다.", "Stopping here to protect your normal costume.");
		close;
	}

	// 100% 확정 성공: 실패 확률 판정 없음
	delitem .@effect_etc_id, 1;

	if (.@effect_part == EQP_COSTUME_HEAD_TOP) {
		set COSTUME_EFFECT_TOP, .@effect_card_id;
		set .@normal_card_id, COSTUME_STONE_TOP;
	}
	else if (.@effect_part == EQP_COSTUME_HEAD_MID) {
		set COSTUME_EFFECT_MID, .@effect_card_id;
		set .@normal_card_id, COSTUME_STONE_MID;
	}
	else if (.@effect_part == EQP_COSTUME_HEAD_LOW) {
		set COSTUME_EFFECT_LOW, .@effect_card_id;
		set .@normal_card_id, COSTUME_STONE_LOW;
	}
	else {
		set COSTUME_EFFECT_GARMENT, .@effect_card_id;
		set .@normal_card_id, COSTUME_STONE_GARMENT;
	}

	delitem .@effect_equip_item_id, 1;

	if (.@effect_part == EQP_COSTUME_GARMENT) {
		// 걸칠것: 일반 card[0], 듀얼 card[1]을 보존하고 이펙트만 card[3] 갱신
		getitembound2 .@effect_equip_item_id, 1, 1, 0, 0, .@normal_card_id, COSTUME_STONE_DUAL, 0, .@effect_card_id, .ClosetBoundType;
	}
	else {
		// 상/중/하단: 기존 일반 스톤(card[0])은 유지하고 이펙트만 card[3] 갱신
		getitembound2 .@effect_equip_item_id, 1, 1, 0, 0, .@normal_card_id, 0, 0, .@effect_card_id, .ClosetBoundType;
	}

	specialeffect2 EF_LEVELUP;
	mes needtr("[이펙트 스톤]", "[Effect Stone]");
	mes needtr("^00AA00이펙트 스톤 부여에 성공했습니다.^000000", "^00AA00The effect stone was applied.^000000");
	mes "^FF8000" + getitemname(.@effect_card_id) + needtr("^000000 세팅이 저장되었습니다.", "^000000 setting saved.");
	mes needtr("현재 의상도 새 이펙트로 갱신되었습니다.", "Your current costume was updated with the new effect.");
	mes needtr("인벤토리에서 다시 착용해주세요.", "Please equip it again from your inventory.");
	close;

L_StoneEnchantDual:
	deletearray .@stone_idx_list[0], getarraysize(.@stone_idx_list);

	set .@data_size, getarraysize(.StoneData);
	set .@row_count, .@data_size / 3;
	set .@count, 0;

	// 듀얼 스톤도 StoneData 전체를 순회하므로 동일하게 보호 한도 완화
	freeloop(1);

	for (set .@i, 0; .@i < .@row_count; set .@i, .@i + 1) {
		set .@base, .@i * 3;

		// StoneData 구조: Part, CardID, EtcID
		// 듀얼은 Part 값으로 5 사용 추천
		set .@part, .StoneData[.@base];
		set .@card_id, .StoneData[.@base + 1];
		set .@etc_id, .StoneData[.@base + 2];

		if (.@part != 5)
			continue;

		if (.@card_id <= 0 || .@etc_id <= 0)
			continue;

		if (countitem(.@etc_id) <= 0)
			continue;

		set .@stone_idx_list[.@count], .@i;
		set .@count, .@count + 1;
	}

	freeloop(0);

	if (.@count <= 0) {
		mes needtr("[듀얼 스톤]", "[Dual Stone]");
		mes needtr("보유 중인 듀얼 스톤이 없습니다.", "You have no dual stone.");
		close;
	}

	mes needtr("[듀얼 스톤]", "[Dual Stone]");
	mes needtr("보유 중인 듀얼 스톤", "Dual stones you own");
	mes " ";

	set .@i, 0;
	while (.@i < .@count) {
		set .@row, .@stone_idx_list[.@i];
		set .@base, .@row * 3;
		set .@etc_id, .StoneData[.@base + 2];

		mes (.@i + 1) + ". " + getitemname(.@etc_id) + " x" + countitem(.@etc_id);

		set .@i, .@i + 1;
	}

	next;

	mes needtr("[듀얼 스톤]", "[Dual Stone]");
	mes needtr("부여할 듀얼 스톤 번호를 입력해주세요.", "Please enter the dual stone number to apply.");
	mes "1 ~ " + .@count + needtr("번까지 입력 가능합니다.", " may be entered.");
	next;

	input .@input_no;

	if (.@input_no < 1 || .@input_no > .@count) {
		mes needtr("[듀얼 스톤]", "[Dual Stone]");
		mes needtr("잘못된 번호입니다.", "That number is not valid.");
		close;
	}

	set .@row, .@stone_idx_list[.@input_no - 1];
	set .@base, .@row * 3;
	set .@card_id, .StoneData[.@base + 1];
	set .@etc_id, .StoneData[.@base + 2];

	if (.@card_id <= 0 || .@etc_id <= 0) {
		mes needtr("[듀얼 스톤]", "[Dual Stone]");
		mes needtr("스톤 매칭 데이터가 올바르지 않습니다.", "The stone matching data is invalid.");
		close;
	}

	mes needtr("[듀얼 스톤]", "[Dual Stone]");
	mes needtr("사용 스톤:", "Stone used:");
	mes getitemname(.@etc_id);
	mes " ";
	mes needtr("성공 시 부여 스톤:", "Granted on success:");
	mes getitemname(.@card_id);
	mes " ";
	mes needtr("성공 확률: ", "Success rate: ") + (.StoneEnchantRate / 100) + "." + sprintf("%02d", .StoneEnchantRate % 100) + "%";

	if (.StoneEnchantCostZeny > 0)
		mes needtr("비용: ", "Cost: ") + callfunc("F_InsertComma", .StoneEnchantCostZeny) + " Zeny";

	if (COSTUME_STONE_DUAL > 0) {
		mes " ";
		mes needtr("^FF8000이미 듀얼 스톤이 장착되어 있습니다.^000000", "^FF8000A dual stone is already mounted.^000000");
		mes needtr("현재 듀얼 스톤: ", "Current dual stone: ") + getitemname(COSTUME_STONE_DUAL);
		mes needtr("성공 시 기존 듀얼 스톤은 새 스톤으로 교체됩니다.", "On success the old dual stone is replaced.");
		mes needtr("실패 시 기존 듀얼 스톤은 유지됩니다.", "On failure the old dual stone stays.");
	}

	next;

	if (select(needtr("부여한다:취소", "Apply:Cancel")) == 2)
		goto L_StoneMenu;

	if (countitem(.@etc_id) <= 0) {
		mes needtr("[듀얼 스톤]", "[Dual Stone]");
		mes needtr("선택한 스톤을 보유하고 있지 않습니다.", "You do not own the stone you chose.");
		close;
	}

	if (.StoneEnchantCostZeny > 0 && Zeny < .StoneEnchantCostZeny) {
		mes needtr("[듀얼 스톤]", "[Dual Stone]");
		mes needtr("Zeny가 부족합니다.", "You do not have enough Zeny.");
		close;
	}

	delitem .@etc_id, 1;

	if (.StoneEnchantCostZeny > 0)
		set Zeny, Zeny - .StoneEnchantCostZeny;

	if (rand(10000) >= .StoneEnchantRate) {
		specialeffect2 EF_SUI_EXPLOSION;
		mes needtr("[듀얼 스톤]", "[Dual Stone]");
		mes needtr("^FF0000스톤 부여에 실패했습니다.^000000", "^FF0000The stone could not be applied.^000000");
		mes needtr("기존 듀얼 스톤은 유지됩니다.", "The old dual stone stays.");
		close;
	}

	set COSTUME_STONE_DUAL, .@card_id;

	specialeffect2 EF_LEVELUP;
	mes needtr("[듀얼 스톤]", "[Dual Stone]");
	mes needtr("듀얼 스톤 부여에 성공했습니다.", "The dual stone was applied.");
	mes getitemname(.@card_id) + needtr(" 세팅이 저장되었습니다.", " setting saved.");
	mes needtr("^FF8000현재 착용 중인 의상에는 즉시 반영되지 않습니다.^000000", "^FF8000It does not apply to the costume you are wearing right now.^000000");
	mes needtr("^FF8000옷장에서 다시 꺼내면 적용됩니다.^000000", "^FF8000It applies when you take it out of the wardrobe again.^000000");
	close;
}
