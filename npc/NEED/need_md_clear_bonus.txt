//============================================================
// NEED - 메모리얼 던전 클리어 보너스 (행운 금화 주머니)
//
// 던전 최종 보스를 처치하면, 그 순간 보스 맵에 있는 캐릭터 각자에게
// 행운 금화 주머니를 지급한다.
//
// [횟수 제한]
// 던전마다 IP당 하루 .IpLimit 회 (04:00 초기화).
// 기존 보스 드랍 IP 제한(instance_db IpDailyRewardLimit)과는 별개로 센다.
// 기존 제한은 막타 친 캐릭터의 IP만 차감하므로, 파티원 각자에게 주는
// 보상에는 쓸 수 없다. 그래서 이 보너스는 캐릭터마다 자기 IP로 따로 센다.
//
// [같은 판 중복 방지]
// 한 캐릭터가 같은 인스턴스에서 두 번 받지 않도록 @ 변수로 막는다.
//
// [저장소]
// need_md_clear_bonus_ip 테이블. 없으면 OnInit 에서 만든다.
// (sql-files/need_md_clear_bonus.sql 과 같은 정의)
// DB 오류가 나면 지급하지 않는다 (fail-closed).
//
// [사용법] 인스턴스 스크립트의 최종 보스 처치 지점에서
//   callfunc "F_NEED_MD_ClearBonus", "<던전 키>", <보스 맵 이름>;
//
// [지급 개수] 맨 아래 OnInit 의 .Keys$ / .Counts 만 고치면 된다.
// 2026-10-08 : 10곳 모두 최소 보상 5개. 이후 클리어 시간별 차등 예정.
//============================================================

function	script	F_NEED_MD_ClearBonus	{
	.@key$ = getarg(0);
	.@map$ = getarg(1);
	.@cfg$ = "NEED_MD_ClearBonus";

	if (!getvariableofnpc(.Enabled, .@cfg$))
		return 0;

	if (!getvariableofnpc(.SqlReady, .@cfg$)) {
		debugmes "[MDBonus] storage not ready - bonus skipped key=" + .@key$;
		return 0;
	}

	// 던전 키 -> 지급 개수
	.@nkeys = getarraysize(getvariableofnpc(.Keys$, .@cfg$));
	.@count = 0;
	for (.@k = 0; .@k < .@nkeys; .@k++) {
		if (getvariableofnpc(.Keys$[.@k], .@cfg$) == .@key$) {
			.@count = getvariableofnpc(.Counts[.@k], .@cfg$);
			break;
		}
	}
	if (.@count <= 0)
		return 0;

	.@item = getvariableofnpc(.ItemId, .@cfg$);
	.@limit = getvariableofnpc(.IpLimit, .@cfg$);
	.@inst = instance_id();
	.@day = atoi(gettimestr("%Y%m%d", 9, gettimetick(2) - 14400));	// 04:00 기준 날짜
	.@ekey$ = escape_sql(.@key$);

	.@orig = playerattached();
	.@n = getmapunits(BL_PC, .@map$, .@aids);
	.@given = 0;

	for (.@i = 0; .@i < .@n; .@i++) {
		if (!attachrid(.@aids[.@i]))
			continue;

		// 같은 판에서 이미 처리한 캐릭터
		if (@NEED_MDB_INST == .@inst && @NEED_MDB_KEY$ == .@key$)
			continue;

		.@ip$ = getcharip();
		if (.@ip$ == "") {
			dispbottom needtr("[클리어 보너스] 접속 정보를 확인할 수 없어 지급하지 못했습니다.", "[Clear Bonus] Your connection could not be checked, so no bonus was given.");
			continue;
		}
		.@eip$ = escape_sql(.@ip$);

		.@used = 0;
		.@q = query_sql("SELECT `used` FROM `need_md_clear_bonus_ip` WHERE `day`=" + .@day
			+ " AND `dkey`='" + .@ekey$ + "' AND `ip`='" + .@eip$ + "'", .@used);
		if (.@q < 0) {
			dispbottom needtr("[클리어 보너스] 기록 확인 중 오류가 발생해 지급하지 못했습니다.", "[Clear Bonus] An error occurred while checking the record, so no bonus was given.");
			continue;
		}

		// 이 판은 처리 완료로 표시 (한도 초과/지급 모두)
		@NEED_MDB_INST = .@inst;
		@NEED_MDB_KEY$ = .@key$;

		if (.@used >= .@limit) {
			dispbottom needtr("[클리어 보너스] 오늘 이 던전의 보너스(IP당 하루 ", "[Clear Bonus] You have already received today's bonus for this dungeon (")
				+ .@limit + needtr("회)를 모두 받았습니다.", " per IP per day).");
			continue;
		}

		if (!checkweight(.@item, .@count)) {
			@NEED_MDB_INST = 0;
			dispbottom needtr("[클리어 보너스] 인벤토리 공간이 부족해 지급하지 못했습니다.", "[Clear Bonus] Not enough inventory space, so no bonus was given.");
			continue;
		}

		.@r = query_sql("INSERT INTO `need_md_clear_bonus_ip` (`day`,`dkey`,`ip`,`used`,`last_account_id`,`last_char_id`,`updated_at`) VALUES ("
			+ .@day + ",'" + .@ekey$ + "','" + .@eip$ + "',1," + getcharid(3) + "," + getcharid(0) + ",NOW()) "
			+ "ON DUPLICATE KEY UPDATE `used`=`used`+1,`last_account_id`=VALUES(`last_account_id`),"
			+ "`last_char_id`=VALUES(`last_char_id`),`updated_at`=NOW()");
		if (.@r < 0) {
			@NEED_MDB_INST = 0;
			dispbottom needtr("[클리어 보너스] 기록 저장 중 오류가 발생해 지급하지 못했습니다.", "[Clear Bonus] An error occurred while saving the record, so no bonus was given.");
			continue;
		}

		getitem .@item, .@count;
		.@given++;
		dispbottom needtr("[클리어 보너스] ", "[Clear Bonus] ") + getitemname(.@item) + " " + .@count
			+ needtr("개를 받았습니다. (오늘 ", " received. (Today ") + (.@used + 1) + " / " + .@limit + ")";
		logmes "[MDBonus] key=" + .@key$ + " inst=" + .@inst + " ip=" + .@ip$ + " used=" + (.@used + 1) + " item=" + .@item + "x" + .@count;
	}

	detachrid;
	if (.@orig)
		attachrid .@orig;
	return .@given;
}

-	script	NEED_MD_ClearBonus	-1,{
	end;

OnInit:
	.Enabled = 1;
	.ItemId = 399906;	// 행운 금화 주머니
	.IpLimit = 3;		// 던전마다 IP당 하루 횟수

	// 던전 키 / 지급 개수 (같은 위치끼리 짝)
	setarray .Keys$[0],
		"GEFFEN",		// 게펜 마법 대회
		"CULVERT",		// 지하 배수로 (일반/상급 합산)
		"JITTERBUG",	// 악몽의 지터벅
		"AIRSHIP",		// 비공정 습격
		"CHARLESTON",	// 위기의 찰스턴
		"DEVILTOWER",	// 마신의 탑
		"CENTRALLAB",	// 중앙 실험실
		"HORROR",		// 호러 장난감 공장
		"SARAH",		// 펜릴과 사라
		"LASTROOM";		// 마지막 방
	setarray .Counts[0],
		5,	// GEFFEN
		5,	// CULVERT
		5,	// JITTERBUG
		5,	// AIRSHIP
		5,	// CHARLESTON
		5,	// DEVILTOWER
		5,	// CENTRALLAB
		5,	// HORROR
		5,	// SARAH
		5;	// LASTROOM

	.SqlReady = 0;
	query_sql("CREATE TABLE IF NOT EXISTS `need_md_clear_bonus_ip` ("
		+ "`day` INT UNSIGNED NOT NULL,"
		+ "`dkey` VARCHAR(32) NOT NULL,"
		+ "`ip` VARCHAR(45) NOT NULL,"
		+ "`used` SMALLINT UNSIGNED NOT NULL DEFAULT 0,"
		+ "`last_account_id` INT UNSIGNED NOT NULL DEFAULT 0,"
		+ "`last_char_id` INT UNSIGNED NOT NULL DEFAULT 0,"
		+ "`updated_at` DATETIME NULL,"
		+ "PRIMARY KEY (`day`,`dkey`,`ip`)"
		+ ")");
	if (query_sql("SELECT `day` FROM `need_md_clear_bonus_ip` LIMIT 0") < 0) {
		debugmes "[MDBonus] need_md_clear_bonus_ip 테이블을 사용할 수 없습니다. sql-files/need_md_clear_bonus.sql 을 적용하세요. (보너스 지급 중지)";
		end;
	}
	.SqlReady = 1;
	end;
}
