// - 입장레벨: BaseLv 150 이상
// - 입장료: 200,000 Zeny
// - 재입장 대기시간: 3분
// - NPC 이펙트: 408
//===============================================================

lasagna,159, 171,6	script	전사자의 무덤	615,{
	// -----------------------------
	// 설정값
	// -----------------------------
	.@need_lv = 150;
	.@need_zeny = 200000;
	.@cooltime = 180; // 3분 = 180초

	mes needtr("[전사자의 무덤]","[Tomb of the Fallen]");
	mes "……";
	mes needtr("네 발소리가 들리는 순간부터","From the moment your footsteps were heard,");
	mes needtr("이 무덤은 이미 너를 기억하고 있었다.","this tomb already remembered you.");
	next;
	mes needtr("[전사자의 무덤]","[Tomb of the Fallen]");
	mes needtr("이 앞은 패배한 자의 쉼터가 아니다.","What lies ahead is no resting place for the defeated.");
	mes needtr("수많은 강자들이 이름도 남기지 못한 채","Countless mighty ones, leaving not even a name,");
	mes needtr("차갑게 가라앉은, 죽음의 성역이다.","sank cold into this sanctuary of death.");
	next;
	mes needtr("[전사자의 무덤]","[Tomb of the Fallen]");
	mes needtr("살아서 돌아온 자는 드물고,","Few have returned alive,");
	mes needtr("정신을 온전히 간직한 자는 더욱 드물다.","and fewer still with their minds intact.");
	mes needtr("그래도 들어가겠다면, 자격을 증명해라.","If you would still enter, prove you are worthy.");
	next;
	mes needtr("[전사자의 무덤]","[Tomb of the Fallen]");
	mes needtr("입장 조건은 다음과 같다.","The conditions for entry are as follows.");
	mes needtr("기본 레벨 ^0000FF", "Base Level ^0000FF") + .@need_lv + needtr("^000000 이상", "^000000 or higher");
	mes needtr("입장료 ^FF0000", "Entry fee ^FF0000") + F_InsertComma(.@need_zeny) + " Zeny^000000";
	mes needtr("재입장 대기시간 ^7777773분^000000","Re-entry cooldown ^7777773 minutes^000000");
	next;

	// 레벨 체크
	if (BaseLevel < .@need_lv) {
		mes needtr("[전사자의 무덤]","[Tomb of the Fallen]");
		mes needtr("……네 영혼은 아직 너무 가볍다.","......your soul is still far too light.");
		mes needtr("이곳은 최소 ^0000FFBaseLv ", "This place requires at least ^0000FFBase Lv ") + .@need_lv + needtr("^000000 이상의", "^000000");
		mes needtr("강자만이 발을 들일 수 있다.","Only the mighty may set foot here.");
		close;
	}

	// 재입장 대기시간 체크
	if (#lhz_dun_n_cd > gettimetick(2)) {
		.@remain = #lhz_dun_n_cd - gettimetick(2);
		.@min = .@remain / 60;
		.@sec = .@remain % 60;

		mes needtr("[전사자의 무덤]","[Tomb of the Fallen]");
		mes needtr("무덤이 아직 너를 다시 받아들일 준비가 되지 않았다.","The tomb is not yet ready to take you again.");
		mes needtr("죽음의 문이 다시 열리기까지 기다려라.","Wait until the gate of death opens once more.");
		if (.@min > 0)
			mes needtr("남은 시간 : ^FF0000", "Time remaining: ^FF0000") + .@min + needtr("분 ", "minutes ") + .@sec + needtr("초^000000", "s^000000");
		else
			mes needtr("남은 시간 : ^FF0000", "Time remaining: ^FF0000") + .@sec + needtr("초^000000", "s^000000");
		close;
	}

	// 제니 체크
	if (Zeny < .@need_zeny) {
		mes needtr("[전사자의 무덤]","[Tomb of the Fallen]");
		mes needtr("대가도 치르지 못한 자에게","For one who cannot even pay the price,");
		mes needtr("죽음의 문은 열리지 않는다.","the gate of death does not open.");
		mes "^FF0000" + F_InsertComma(.@need_zeny) + needtr(" Zeny^000000 가 필요하다.", " Zeny^000000 is required.");
		close;
	}

	switch(select(needtr("죽음의 성역으로 들어간다:아직은 물러난다","Enter the sanctuary of death:Withdraw for now"))) {
	case 1:
		Zeny -= .@need_zeny;
		set #lhz_dun_n_cd, gettimetick(2) + .@cooltime;

		mes needtr("[전사자의 무덤]","[Tomb of the Fallen]");
		mes needtr("좋다…","So be it...");
		mes needtr("네가 선택한 것은 도전이 아니라","What you have chosen is not a challenge,");
		mes needtr("죽음과의 계약이다.","but a contract with death.");
		next;
		mes needtr("[전사자의 무덤]","[Tomb of the Fallen]");
		mes needtr("들어가라.","Enter.");
		mes needtr("그리고 증명해라.","And prove it.");
		mes needtr("네가 무덤에 새겨질 이름이 될지,","Whether you become a name carved into the tomb,");
		mes needtr("무덤을 지배할 이름이 될지를.","or the name that rules it.");
		close2;
		warp "lhz_dun_n",96,203;
		end;

	case 2:
		mes needtr("[전사자의 무덤]","[Tomb of the Fallen]");
		mes needtr("현명한 선택일 수도 있다.","That may be the wise choice.");
		mes needtr("준비가 끝났다면 다시 와라.","Come again when you are ready.");
		mes needtr("이 문은 언제나 강자의 최후를 기다리고 있다.","This gate always awaits the end of the mighty.");
		close;
	}

OnInit:
	initnpctimer;
	end;

OnTimer800:
	specialeffect 773;
	initnpctimer;
	end;
}
