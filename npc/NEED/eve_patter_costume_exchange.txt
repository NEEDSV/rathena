//===== rAthena Script =======================================
//= Costume Exchange Coupon NPC - Eve Patter
//===== Description: =========================================
//= 지정된 일반 투구 1개와 의상 교환쿠폰(6697) 1개를
//= 대응하는 의상 장비 1개로 교환합니다.
//============================================================

itemmall,29,74,4	script	이브 패터#costex	4_F_01,{
	mes "[이브 패터]";
	mes "안녕하세요.";
	mes "^3131FF의상 교환쿠폰^000000과 지정된 투구를 가져오시면";
	mes "해당 투구를 의상 장비로 교환해 드립니다.";
	mes "";
	mes "교환 시 기존 장비는 사라지며,";
	mes "^FF0000제련도, 카드, 인챈트 및 랜덤 옵션은 이전되지 않습니다.^000000";
	mes "";
	mes "추가로 교환이 필요한 아이템이 있다면";
	mes "^3B48CC디스코드 건의게시판^000000에 남겨 주세요.";
	setdialogsize(400,200);
	next;

	switch (select("의상 장비로 교환한다:교환 가능 목록을 확인한다:그만둔다")) {
	case 1:
		if (countitem(.CouponItem) < 1) {
			mes "[이브 패터]";
			mes "교환에는 ^3131FF의상 교환쿠폰^000000 1개가 필요합니다.";
			close;
		}

		mes "[이브 패터]";
		mes "어떤 장비를 의상으로 교환하시겠어요?";
		next;

		.@menu$ = "";
		for (.@i = 0; .@i < getarraysize(.NormalItem); ++.@i)
			.@menu$ += .NormalName$[.@i] + ":";
		.@menu$ += "취소";

		.@type = select(.@menu$) - 1;
		if (.@type >= getarraysize(.NormalItem))
			close;

		.@normal_id = .NormalItem[.@type];
		.@costume_id = .CostumeItem[.@type];

		// 동일 장비를 여러 개 가지고 있을 수 있으므로 실제 인벤토리 개체를 선택합니다.
		getinventorylist();
		.@found = 0;
		.@equipped = 0;
		.@rental = 0;
		.@item_menu$ = "";

		for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
			if (@inventorylist_id[.@i] != .@normal_id)
				continue;

			if (@inventorylist_equip[.@i]) {
				.@equipped++;
				continue;
			}

			// 대여 장비는 교환 대상에서 제외합니다.
			if (@inventorylist_expire[.@i]) {
				.@rental++;
				continue;
			}

			.@list_index[.@found] = .@i;
			.@inventory_index[.@found] = @inventorylist_idx[.@i];

			if (@inventorylist_refine[.@i] > 0)
				.@item_menu$ += "+" + @inventorylist_refine[.@i] + " " + .NormalName$[.@type] + ":";
			else
				.@item_menu$ += .NormalName$[.@type] + ":";

			.@found++;
		}

		if (!.@found) {
			mes "[이브 패터]";
			mes "교환 가능한 ^3131FF" + .NormalName$[.@type] + "^000000을 찾지 못했습니다.";
			if (.@equipped)
				mes "장착 중인 장비는 해제한 뒤 다시 시도해 주세요.";
			if (.@rental)
				mes "대여 장비는 교환할 수 없습니다.";
			close;
		}

		.@item_menu$ += "취소";
		mes "[이브 패터]";
		mes "교환할 장비를 선택해 주세요.";
		next;

		.@pick = select(.@item_menu$) - 1;
		if (.@pick >= .@found)
			close;

		.@list = .@list_index[.@pick];
		.@inv_index = .@inventory_index[.@pick];
		.@bound = @inventorylist_bound[.@list];

		mes "[이브 패터]";
		mes "다음과 같이 교환합니다.";
		mes "";
		mes "^3131FF[소모]^000000";
		mes "ㆍ" + mesitemlink(.@normal_id) + " 1개";
		mes "ㆍ" + mesitemlink(.CouponItem) + " 1개";
		mes "";
		mes "^008000[획득]^000000";
		mes "ㆍ" + mesitemlink(.@costume_id) + " 1개";
		mes "";

		if (@inventorylist_refine[.@list] > 0)
			mes "선택한 장비의 제련도: ^FF0000+" + @inventorylist_refine[.@list] + "^000000";

		if (@inventorylist_card1[.@list] || @inventorylist_card2[.@list] || @inventorylist_card3[.@list] || @inventorylist_card4[.@list])
			mes "^FF0000선택한 장비에 카드 또는 특수 정보가 들어 있습니다.^000000";

		if (@inventorylist_option_id1[.@list] || @inventorylist_option_id2[.@list] || @inventorylist_option_id3[.@list] || @inventorylist_option_id4[.@list] || @inventorylist_option_id5[.@list])
			mes "^FF0000선택한 장비에 랜덤 옵션이 들어 있습니다.^000000";

		if (@inventorylist_enchantgrade[.@list])
			mes "^FF0000선택한 장비에 등급 강화가 적용되어 있습니다.^000000";

		if (.@bound)
			mes "선택한 장비의 귀속 상태는 결과 의상에 유지됩니다.";

		mes "^FF0000기존 장비의 모든 정보는 사라지고 의상에 이전되지 않습니다.^000000";
		next;

		if (select("교환한다:취소") == 2)
			close;

		// 확인 창 이후 상태가 바뀌었을 가능성에 대비해 재검사합니다.
		if (countitem(.CouponItem) < 1) {
			mes "[이브 패터]";
			mes "의상 교환쿠폰이 부족합니다.";
			close;
		}

		// 선택한 인벤토리 개체만 정확히 삭제합니다.
		if (!delitemidx(.@inv_index, 1)) {
			mes "[이브 패터]";
			mes "선택한 장비를 확인할 수 없습니다.";
			mes "장비를 다시 확인한 뒤 재시도해 주세요.";
			close;
		}

		delitem .CouponItem, 1;

		// 귀속 장비를 교환한 경우 결과 의상에도 동일한 귀속 상태를 유지합니다.
		if (.@bound)
			getitembound .@costume_id, 1, .@bound;
		else
			getitem .@costume_id, 1;

		specialeffect2 EF_COIN;
		mes "[이브 패터]";
		mes "교환이 완료되었습니다.";
		mes "";
		mes "획득: " + mesitemlink(.@costume_id) + " 1개";
		close;

	case 2:
		mes "[이브 패터]";
		mes "현재 교환 가능한 아이템은 다음과 같습니다.";
		mes "";
		for (.@i = 0; .@i < getarraysize(.NormalItem); ++.@i)
			mes "ㆍ^3131FF" + .NormalName$[.@i] + "^000000 → " + mesitemlink(.CostumeItem[.@i]);
		mes "";
		mes "각 교환에는 ^3131FF의상 교환쿠폰 1개^000000가 필요합니다.";
		mes "추가 품목이 필요하다면 디스코드 건의게시판에 남겨 주세요.";
		close;

	case 3:
		close;
	}

OnInit:
	waitingroom "의상 교환쿠폰 사용처",0;
	.CouponItem = 6697;

	// 일반 장비 ID
	setarray .NormalItem[0],
		2235, // 크라운
		5074, // 천사의 날개 귀
		2254, // 천사의 머리띠
		2255, // 악마의 머리띠
		2256, // 마제스틱 고우트
		5019, // 커세어
		5143; // 가부키 가면

	// 대응 의상 장비 ID
	setarray .CostumeItem[0],
		19630, // 의상 크라운
		19781, // 의상 천사의 날개 귀
		19529, // 의상 천사의 머리띠
		19731, // 의상 악마의 머리띠
		19549, // 의상 마제스틱 고우트
		19737, // 의상 커세어
		19556; // 의상 가부키가면

	setarray .NormalName$[0],
		"크라운",
		"천사의 날개 귀",
		"천사의 머리띠",
		"악마의 머리띠",
		"마제스틱 고우트",
		"커세어",
		"가부키가면";
	end;
}
