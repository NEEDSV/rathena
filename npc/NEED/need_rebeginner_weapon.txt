// NEED - 초심자 무기 지급 NPC
// 캐릭터당 1회 지급
//============================================================

lasagna,164,153,6	script	초심자 무기 지급원	871,{
	mes "[초심자 무기 지급원]";

	if (NEED_REBEGINNER_WEAPON) {
		mes "이미 초심자 무기를 지급받았습니다.";
		close;
	}

	.@item_id = callfunc("F_NEED_RebeginnerWeapon", Class);

	if (.@item_id <= 0) {
		mes "현재 직업으로 지급 가능한 초심자 무기가 없습니다.";
		mes "전직 후 다시 이용해주세요.";
		close;
	}

	mes "현재 직업에 맞는 무기는";
	mes "^0055FF" + getitemname(.@item_id) + "^000000 입니다.";
	next;

	if (select("지급받는다:취소") == 2) {
		mes "[초심자 무기 지급원]";
		mes "필요할 때 다시 찾아와주세요.";
		close;
	}

	if (NEED_REBEGINNER_WEAPON) {
		mes "[초심자 무기 지급원]";
		mes "이미 초심자 무기를 지급받았습니다.";
		close;
	}

	if (!checkweight(.@item_id, 1)) {
		mes "[초심자 무기 지급원]";
		mes "인벤토리 무게나 소지 한도를 확인해주세요.";
		close;
	}

	getitem .@item_id, 1;
	NEED_REBEGINNER_WEAPON = 1;

	mes "[초심자 무기 지급원]";
	mes "^0055FF" + getitemname(.@item_id) + "^000000 1개를 지급했습니다.";
	close;
OnInit:
		waitingroom "초심자 무기 지급",0;
		end;
}


//============================================================
// 직업별 초심자 무기 매칭
//============================================================
function	script	F_NEED_RebeginnerWeapon	{
	.@job = getarg(0);

	switch (.@job) {

	// 룬나이트 계열
	case Job_Knight:
	case Job_Knight2:
	case Job_Lord_Knight:
	case Job_Lord_Knight2:
	case Job_Rune_Knight:
	case Job_Rune_Knight_T:
		return 21037; // Rebeginer_RK_T_Sword

	// 로얄가드 계열
	case Job_Crusader:
	case Job_Crusader2:
	case Job_Paladin:
	case Job_Paladin2:
	case Job_Royal_Guard:
	case Job_Royal_Guard_T:
		return 26015; // Rebeginer_LG_Lance

	// 워록 계열
	case Job_Wizard:
	case Job_High_Wizard:
	case Job_Warlock:
	case Job_Warlock_T:
		return 2046; // Rebeginer_WL_Staff

	// 소서러 계열
	case Job_Sage:
	case Job_Professor:
	case Job_Sorcerer:
	case Job_Sorcerer_T:
		return 26119; // Rebeginer_SO_Rod

	// 레인저 계열
	case Job_Hunter:
	case Job_Sniper:
	case Job_Ranger:
	case Job_Ranger_T:
		return 18165; // Rebeginer_RN_Bow

	// 민스트럴 / 원더러 계열
	case Job_Bard:
	case Job_Dancer:
	case Job_Clown:
	case Job_Gypsy:
	case Job_Minstrel:
	case Job_Minstrel_T:
	case Job_Wanderer:
	case Job_Wanderer_T:
		return 18166; // Rebeginer_WM_Bow

	// 아크비숍 계열
	case Job_Priest:
	case Job_High_Priest:
	case Job_Arch_Bishop:
	case Job_Arch_Bishop_T:
		return 28616; // Rebeginer_AB_Bible

	// 수라 계열
	case Job_Monk:
	case Job_Champion:
	case Job_Sura:
	case Job_Sura_T:
		return 1848; // Rebeginer_SR_Knuckle

	// 미케닉 계열
	case Job_Blacksmith:
	case Job_Whitesmith:
	case Job_Mechanic:
	case Job_Mechanic_T:
		return 16075; // Rebeginer_NC_Mace

	// 제네릭 계열
	case Job_Alchemist:
	case Job_Creator:
	case Job_Genetic:
	case Job_Genetic_T:
		return 13483; // Rebeginer_GN_Sword

	// 길로틴크로스 계열
	case Job_Assassin:
	case Job_Assassin_Cross:
	case Job_Guillotine_Cross:
	case Job_Guillotine_Cross_T:
		return 28027; // Rebeginer_GC_Katar

	// 쉐도우체이서 계열
	case Job_Rogue:
	case Job_Stalker:
	case Job_Shadow_Chaser:
	case Job_Shadow_Chaser_T:
		return 28740; // Rebeginer_SC_Dagger

	// 슈퍼노비스 계열
	case Job_Super_Novice:
	case Job_Super_Novice_E:
		return 16076; // Rebeginer_N2_Mace

	// 닌자 / 카게로우 / 오보로 계열
	case Job_Ninja:
	case Job_Kagerou:
	case Job_Oboro:
		return 13341; // Rebeginer_KO_Humma

	// 건슬링거 / 리벨리온 계열
	case Job_Gunslinger:
	case Job_Rebellion:
		return 13144; // Rebeginer_RL_Gun

	// 도람 / 소환사
	case Job_Summoner:
		return 26120; // Rebeginer_DO_Rod
	}

	// 1차 직업은 계열 선택
	switch (.@job) {
	case Job_Swordman:
	case Job_Swordman_High:
		mes "검사 계열은 받을 무기를 선택해주세요.";
		next;
		if (select("룬나이트 계열 무기:로얄가드 계열 무기") == 1)
			return 21037;
		return 26015;

	case Job_Mage:
	case Job_Mage_High:
		mes "마법사 계열은 받을 무기를 선택해주세요.";
		next;
		if (select("워록 계열 무기:소서러 계열 무기") == 1)
			return 2046;
		return 26119;

	case Job_Archer:
	case Job_Archer_High:
		mes "궁수 계열은 받을 무기를 선택해주세요.";
		next;
		if (select("레인저 계열 무기:민스트럴/원더러 계열 무기") == 1)
			return 18165;
		return 18166;

	case Job_Acolyte:
	case Job_Acolyte_High:
		mes "복사 계열은 받을 무기를 선택해주세요.";
		next;
		if (select("아크비숍 계열 무기:수라 계열 무기") == 1)
			return 28616;
		return 1848;

	case Job_Merchant:
	case Job_Merchant_High:
		mes "상인 계열은 받을 무기를 선택해주세요.";
		next;
		if (select("미케닉 계열 무기:제네릭 계열 무기") == 1)
			return 16075;
		return 13483;

	case Job_Thief:
	case Job_Thief_High:
		mes "도둑 계열은 받을 무기를 선택해주세요.";
		next;
		if (select("길로틴크로스 계열 무기:쉐도우체이서 계열 무기") == 1)
			return 28027;
		return 28740;
	}

	return 0;
	
}
