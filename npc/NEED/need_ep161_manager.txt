// NEED: 오늘의 반복 일일퀘스트 진행 상황을 계산합니다.
//   반환값            : 오늘 아직 받지 못한 명예의 증표 수 (0~66)
//   $@ep161_done      : 오늘 완료한 반복 일일퀘스트 수 (0~15)
//   $@ep161_remain    : 남은 증표 수 (반환값과 동일)
//   $@ep161_pend[]    : 아직 완료하지 않은 쿨타임 퀘스트 ID 목록
//   $@ep161_pend_amt[]: 각 항목의 증표 수
// 판정 기준은 각 반복 일일퀘스트의 쿨타임 퀘스트 checkquest(id,PLAYTIME) 입니다.
function	script	F_NEED_EP161_DailyStatus	{
	// 쿨타임 퀘스트 15종 / 각 증표 수 (10 + 4*14 = 66)
	setarray .@cd[0],
		 7706,	// 의식의 방
		11425,	// 왕궁 주방 - 그리폰 구이
		11422,	// 왕궁 주방 - 소스 만들기
		11427,	// 왕궁 주방 - 접시 수거
		14509,	// 아그네스 뢰겐부르그
		 5414,	// 리하르트 전사 수련
		14492,	// 유르겐 위그너
		 5452,	// 침공당한 프론테라
		 5426, 5429, 5433, 5437, 5440, 5443, 5446;	// 지하감옥 A B C E F G H
	setarray .@amt[0],
		10,
		 4, 4, 4,
		 4, 4, 4, 4,
		 4, 4, 4, 4, 4, 4, 4;

	.@size = getarraysize(.@cd);
	deletearray $@ep161_pend[0];
	deletearray $@ep161_pend_amt[0];
	$@ep161_done = 0;
	$@ep161_remain = 0;
	.@n = 0;

	for (.@i = 0; .@i < .@size; ++.@i) {
		.@state = checkquest(.@cd[.@i],PLAYTIME);
		if (.@state == 0 || .@state == 1) {	// 오늘 이미 완료
			$@ep161_done++;
			continue;
		}
		$@ep161_pend[.@n] = .@cd[.@i];
		$@ep161_pend_amt[.@n] = .@amt[.@i];
		$@ep161_remain += .@amt[.@i];
		.@n++;
	}
	return $@ep161_remain;
}

// NEED: 즉시 완료를 차단해야 하는 상태인지 판정합니다.
//   0 : 사용 가능
//   1 : 의식의 방 메인 스토리 진행 중 (7700)
//   2 : 아그네스 최초 장편 의뢰 진행 중 (14508)
//
// 실제 코드에서 확인된 충돌만 차단합니다.
//   - 7706 쿨타임이 있으면 "흔들리는 공간#evt_gate02" 가 입장을 막으므로
//     메인 스토리(7700)로 의식의 방에 들어가야 하는 캐릭터는 진행이 막힌다.
//   - 14509 쿨타임이 있으면 아그네스의 최초 의뢰(14508) 보고 분기에 도달하지 못한다.
// 그 밖의 계열은 최초/스토리 분기가 쿨타임 판정보다 앞에 있어 충돌하지 않는다.
//   (지하감옥 최초 5425 등은 isbegin_quest 분기, 주방 최초는 ep16_cookbs 분기,
//    리하르트는 v5 수정으로 스토리 우선, 위그너는 v7 수정으로 완전 분리)
function	script	F_NEED_EP161_InstantBlock	{
	if (isbegin_quest(7700) == 1)
		return 1;
	if (isbegin_quest(14508) == 1)
		return 2;
	return 0;
}

// NEED: $@ep161_pend[] 에 남아 있는 반복 일일퀘스트를 오늘 완료 상태로 만듭니다.
// 진행 중인 "반복" 퀘스트만 정리하고 쿨타임을 설정하며,
// 메인/최초/구역 퀘스트와 ep16_fix 는 절대 건드리지 않습니다.
function	script	F_NEED_EP161_InstantApply	{
	.@size = getarraysize($@ep161_pend);
	for (.@i = 0; .@i < .@size; ++.@i) {
		.@cd = $@ep161_pend[.@i];

		switch (.@cd) {
		case 7706:	// 의식의 방 - 7705 만 정리 (7700~7703 메인은 제외)
			if (isbegin_quest(7705) > 0)
				erasequest 7705;
			break;
		case 11422:	// 소스 만들기 - 11421 과 재료 11428~11431
			if (isbegin_quest(11421) > 0)
				erasequest 11421;
			for (.@q = 11428; .@q <= 11431; ++.@q) {
				if (isbegin_quest(.@q) > 0)
					erasequest .@q;
			}
			// 소스 일일퀘 상태 머신(25~33)만 대기값 23 으로 되돌립니다.
			// 사이클 중인 값만 정확히 지정해 복구하므로,
			// 스토리 진행도나 향후 추가될 다른 상태값이 롤백되지 않습니다.
			//   23    : 대기(일일퀘 수령 가능)
			//   25~33 : 재료 4종 수집 사이클 진행 중
			if (ep16_cookbs >= 25 && ep16_cookbs <= 33)
				ep16_cookbs = 23;
			if (ep16_cookdaily >= 25 && ep16_cookdaily <= 33)
				ep16_cookdaily = 23;
			break;
		case 11425:	// 그리폰 구이
			if (isbegin_quest(11423) > 0)
				erasequest 11423;
			if (isbegin_quest(11424) > 0)
				erasequest 11424;
			break;
		case 11427:	// 접시 수거
			if (isbegin_quest(11426) > 0)
				erasequest 11426;
			break;
		case 14509:	// 아그네스 - 반복 의뢰 14510~14515 만 (최초 14508 제외)
			for (.@q = 14510; .@q <= 14515; ++.@q) {
				if (isbegin_quest(.@q) > 0)
					erasequest .@q;
			}
			break;
		case 5414:	// 리하르트 전사 수련 - 5404~5413 (5402/5403 메인 제외)
			for (.@q = 5404; .@q <= 5413; ++.@q) {
				if (isbegin_quest(.@q) > 0)
					erasequest .@q;
			}
			break;
		case 14492:	// 유르겐 - 14493/14494 만 (14486~14491 스토리 제외)
			if (isbegin_quest(14493) > 0)
				erasequest 14493;
			if (isbegin_quest(14494) > 0)
				erasequest 14494;
			break;
		case 5452:	// 침공당한 프론테라 - 5448~5468 (쿨 5452 자신 제외)
			for (.@q = 5448; .@q <= 5468; ++.@q) {
				if (.@q == 5452)
					continue;
				if (isbegin_quest(.@q) > 0)
					erasequest .@q;
			}
			break;
		case 5426:	// 지하감옥 A - 반복 5427 만
			if (isbegin_quest(5427) > 0)
				erasequest 5427;
			break;
		case 5429:	// 지하감옥 B
			if (isbegin_quest(5430) > 0)
				erasequest 5430;
			break;
		case 5433:	// 지하감옥 C - 반복 5434/5435 만 (최초 5431/5432 제외)
			if (isbegin_quest(5434) > 0)
				erasequest 5434;
			if (isbegin_quest(5435) > 0)
				erasequest 5435;
			break;
		case 5437:	// 지하감옥 E
			if (isbegin_quest(5438) > 0)
				erasequest 5438;
			break;
		case 5440:	// 지하감옥 F
			if (isbegin_quest(5441) > 0)
				erasequest 5441;
			break;
		case 5443:	// 지하감옥 G
			if (isbegin_quest(5444) > 0)
				erasequest 5444;
			break;
		case 5446:	// 지하감옥 H
			if (isbegin_quest(5447) > 0)
				erasequest 5447;
			break;
		}

		// 만료되어 남아 있는 쿨타임 기록을 정리한 뒤 오늘 쿨타임을 새로 설정합니다.
		if (isbegin_quest(.@cd) > 0)
			erasequest .@cd;
		setquest .@cd;
	}
	return;
}
