// NEED 장비 세팅 공유소 NPC
// 위치와 표시 이름은 아래 NPC 선언부에서 변경할 수 있습니다.

lasagna,161,162,6	script	장비 세팅 공유소	729,{
	while (true) {
		mes needtr("[장비 세팅 공유소]", "[Equip Build Sharing]");
		mes needtr("원하는 메뉴를 선택하세요.", "Please choose a menu.");
		.@menu$ = needtr("장비 세팅 등록:공개 세팅 열람:내 등록 내역:보상 수령:이용 안내", "Register a Set:Browse Public Sets:My Registrations:Claim Reward:Guide");
		if (need_equipment_build_is_admin())
			.@menu$ += needtr(":운영자 관리", ":GM Admin");
		switch (select(.@menu$)) {
		case 1:
			callsub S_Register;
			break;
		case 2:
			callsub S_PublicBrowse;
			break;
		case 3:
			callsub S_MyBuilds;
			break;
		case 4:
			callsub S_Rewards;
			break;
		case 5:
			clear;
			mes needtr("[이용 안내 - 등록 및 공개]", "[Guide - Register & Publish]");
			mes needtr("현재 착용 중인 장비와 기본 스테이터스를 세팅으로 등록합니다.", "Registers your currently equipped gear and base stats as one set.");
			mes needtr("운영자의 검수를 거쳐 승인된 세팅만 공개됩니다.", "Only sets approved by a GM review are published.");
			mes needtr("공개된 세팅에는 등록자의 계정 및 캐릭터 정보가 표시되지 않습니다.", "Published sets never show the owner's account or character.");
			next;
			clear;
			mes needtr("[이용 안내 - 취소 및 좋아요]", "[Guide - Cancel & Likes]");
			mes needtr("검수 대기 상태에서는 등록자가 직접 취소할 수 있습니다.", "While a set is pending review you can cancel it yourself.");
			mes needtr("승인된 세팅은 등록자가 직접 취소하거나 삭제할 수 없습니다.", "Once approved you can no longer cancel or delete it yourself.");
			mes needtr("도움이 된 공개 세팅에는 좋아요를 표시할 수 있습니다.", "You can like any published set you found useful.");
			next;
			clear;
			mes needtr("[이용 안내 - 등록 보상]", "[Guide - Registration Reward]");
			mes needtr("승인된 모든 세팅이 보상 대상으로 선정되는 것은 아닙니다.", "Not every approved set is chosen for a reward.");
			mes needtr("운영자가 검수 과정에서 보상 지급 여부와 수량을 결정합니다.", "The GM decides during review whether a reward is given and how much.");
			mes needtr("보상은 해당 세팅을 등록한 계정과 캐릭터로만 수령할 수 있습니다.", "A reward can only be claimed by the account and character that registered the set.");
			next;
			break;
		case 6:
			if (need_equipment_build_is_admin())
				callsub S_Admin;
			break;
		}
	}

S_Register:
	clear;
	mes needtr("[장비 세팅 등록]", "[Register Equipment Set]");
	mes needtr("현재 착용 중인 장비와 기본 스테이터스를 하나의 세팅으로 등록합니다.", "Your equipped gear and base stats will be registered as one set.");
	mes needtr("이 세팅의 사용 목적을 선택해 주세요.", "Please choose what this set is for.");
	.@category = select(needtr("일반 사냥:보스 사냥:재료 / 제니 노가다:인스턴스 던전:전장 / 공성:기타", "General Hunting:Boss Hunting:Farming / Zeny:Instance Dungeon:Battleground / WoE:Other"));
	next;
	clear;
	mes needtr("[장비 세팅 등록]", "[Register Equipment Set]");
	mes needtr("세팅을 쉽게 구분할 수 있는 제목을 입력해 주세요.", "Please enter a title that makes this set easy to recognise.");
	mes needtr("1~30자 이내로 입력해 주세요.", "Use between 1 and 30 characters.");
	mes needtr("줄바꿈, 콜론(:), 색상 코드(^)는 사용할 수 없습니다.", "Line breaks, colons (:) and colour codes (^) are not allowed.");
	input .@title$;
	// ':'는 select() 메뉴 구분자로 사용되므로 제목에서 차단합니다.
	if (compare(.@title$, ":")) {
		mes needtr("제목에는 콜론(:)을 사용할 수 없습니다.", "The title cannot contain a colon (:).");
		next;
		return;
	}
	if (compare(.@title$, "^")) {
		mes needtr("제목에는 색상 코드(^)를 사용할 수 없습니다.", "The title cannot contain a colour code (^).");
		next;
		return;
	}
	next;
	clear;
	mes needtr("[장비 세팅 등록]", "[Register Equipment Set]");
	mes needtr("세팅의 특징이나 활용 방법을 입력해 주세요.", "Please describe what makes this set work or how to use it.");
	mes needtr("최대 200자까지 입력할 수 있습니다.", "Up to 200 characters.");
	mes needtr("설명을 생략하려면 공백 한 칸만 입력해 주세요.", "To skip the description enter a single space.");
	input .@description$;
	if (.@description$ == " ")
		.@description$ = "";
	next;
	clear;
	mes needtr("[등록 내용 확인]", "[Confirm Registration]");
	mes needtr("사용 목적: ", "Purpose: ") + need_equipment_build_category_name(.@category);
	mes needtr("제목: ", "Title: ") + .@title$;
	mes needtr("설명: ", "Description: ") + (.@description$ == "" ? needtr("없음", "None") : .@description$);
	mes needtr("등록 후 검수 대기 상태에서는 직접 취소할 수 있습니다.", "After registering you can cancel it yourself while it is pending review.");
	mes needtr("승인 이후에는 운영자에게 요청해야 수정 또는 삭제할 수 있습니다.", "After approval only a GM can change or delete it.");
	if (select(needtr("등록하기:취소", "Register:Cancel")) == 2)
		return;
	.@result = need_register_equipment_build(.@category, .@title$, .@description$);
	clear;
	mes needtr("[장비 세팅 등록]", "[Register Equipment Set]");
	if (.@result > 0) {
		mes needtr("장비 세팅이 등록되었습니다.", "Your equipment set has been registered.");
		mes needtr("세팅 번호: #", "Set No.: #") + .@result;
	} else {
		mes needtr("장비 세팅 등록에 실패했습니다.", "Registration failed.");
		switch (.@result) {
		case -2: mes needtr("사용 목적, 제목 또는 설명을 다시 확인해 주세요.", "Please check the purpose, title and description again."); break;
		case -3: mes needtr("캐릭터 또는 계정의 등록 가능 개수를 초과했습니다.", "You have reached the registration limit for this character or account."); break;
		case -4: mes needtr("동일한 장비 구성의 세팅이 이미 등록되어 있습니다.", "A set with the same equipment is already registered."); break;
		case -5: mes needtr("지원되는 장비를 한 부위 이상 착용한 뒤 다시 시도해 주세요.", "Please equip at least one supported item and try again."); break;
		case -16: mes needtr("잠시 후 다시 시도해 주세요.", "Please try again in a moment."); break;
		default: mes needtr("잠시 후 다시 시도하거나 운영자에게 문의해 주세요.", "Please try again later or contact a GM."); break;
		}
	}
	next;
	return;

S_PublicBrowse:
	clear;
	mes needtr("[공개 장비 세팅]", "[Public Equipment Sets]");
	mes needtr("확인할 직업 계열을 선택해 주세요.", "Please choose a job class to browse.");
	.@family = select(needtr("노비스 계열:검사 계열:마법사 계열:궁수 계열:복사 계열:상인 계열:도둑 계열:확장 직업", "Novice Class:Swordsman Class:Mage Class:Archer Class:Acolyte Class:Merchant Class:Thief Class:Expanded Classes"));
	.@count = need_get_public_equipment_build_jobs(.@family);
	if (.@count < 0) {
		mes needtr("직업 목록을 불러오지 못했습니다.", "Could not load the job list.");
		next;
		return;
	}
	if (.@count == 0) {
		mes needtr("선택한 직업 계열에는 공개된 세팅이 없습니다.", "There are no published sets for that job class.");
		next;
		return;
	}

	.@menu$ = "";
	for (.@i = 0; .@i < .@count; ++.@i)
		.@menu$ += replacestr(@need_build_job_names$[.@i], ":", " - ") + ":";

	.@choice = select(.@menu$) - 1;
	.@job_id = @need_build_job_ids[.@choice];
	.@job_name$ = @need_build_job_names$[.@choice];

	// 공개 목록은 사용 목적 전체를 함께 조회합니다.
	// category 0은 C++ 조회 명령에서 '전체 목적'으로 처리되어야 합니다.
	callsub S_PublicPage, .@job_id, 1, .@job_name$;
	return;

S_PublicPage:
	.@job_id = getarg(0);
	.@sort_type = getarg(1);
	.@job_name$ = getarg(2);
	.@page = 0;

	while (true) {
		clear;

		// category 0 = 전체 사용 목적
		.@count = need_get_public_equipment_build_list(.@job_id, 0, .@sort_type, .@page);

		mes needtr("[공개 장비 세팅]", "[Public Equipment Sets]");
		mes .@job_name$ + " / " + (.@sort_type == 1 ? needtr("좋아요순", "Most Liked") : needtr("최신순", "Newest"));
		mes needtr((.@page + 1) + "페이지", "Page " + (.@page + 1));

		if (.@count < 0) {
			mes needtr("목록을 불러오지 못했습니다.", "Could not load the list.");
			next;
			return;
		}

		if (.@count == 0)
			mes needtr("이 직업에 공개된 세팅이 없습니다.", "There are no published sets for this job.");
		else
			mes needtr("다른 세팅을 보려면 열려 있는 장비창을 먼저 닫아 주세요.", "Close the open equipment window before viewing another set.");

		.@menu$ = "";
		for (.@i = 0; .@i < .@count; ++.@i) {
			.@category_name$ = need_equipment_build_category_name(@need_build_categories[.@i]);
			.@menu$ += "[#" + @need_build_ids[.@i] + "] [" + .@category_name$ + "] "
				+ replacestr(@need_build_titles$[.@i], ":", " - ")
				+ needtr(" / 좋아요 ", " / likes ") + @need_build_like_counts[.@i] + ":";
		}

		if (.@page > 0)
			.@menu$ += needtr("이전 페이지:", "Prev Page:");
		if (@need_build_has_more)
			.@menu$ += needtr("다음 페이지:", "Next Page:");

		.@menu$ += (.@sort_type == 1 ? needtr("최신순으로 보기:", "Sort by Newest:") : needtr("좋아요순으로 보기:", "Sort by Likes:"));
		.@menu$ += needtr("직업 다시 선택:창 닫기", "Pick Job Again:Close");

		.@choice = select(.@menu$);

		if (.@choice <= .@count) {
			.@build_id = @need_build_ids[.@choice - 1];
			.@result = need_show_public_equipment_build(.@build_id);

			clear;
			mes needtr("[공개 장비 세팅]", "[Public Equipment Sets]");

			if (.@result <= 0) {
				mes needtr("현재 공개 상태가 아닌 세팅입니다.", "This set is not published right now.");
				next;
				continue;
			}

			mes needtr("세팅 요약은 채팅창에서 확인할 수 있습니다.", "The set summary is shown in the chat window.");
			callsub S_RequestView, .@build_id, 1;

			while (true) {
				mes needtr("좋아요: ", "Likes: ") + @need_build_detail_like_count;

				if (@need_build_detail_is_owner)
					.@detail_choice = select(needtr("장비 다시 보기:목록으로 돌아가기:창 닫기", "View Equipment:Back to List:Close"));
				else
					.@detail_choice = select(needtr("장비 다시 보기:", "View Equipment:") + (@need_build_detail_liked ? needtr("좋아요 취소", "Unlike") : needtr("좋아요", "Like")) + needtr(":목록으로 돌아가기:창 닫기", ":Back to List:Close"));

				if (.@detail_choice == 1) {
					clear;
					mes needtr("[공개 장비 세팅]", "[Public Equipment Sets]");
					callsub S_RequestView, .@build_id, 1;
					continue;
				}

				if (@need_build_detail_is_owner) {
					if (.@detail_choice == 2)
						break;
					close;
				}

				if (.@detail_choice == 2) {
					.@like_result = need_toggle_equipment_build_like(.@build_id);
					clear;
					mes needtr("[공개 장비 세팅]", "[Public Equipment Sets]");

					if (.@like_result == 1) {
						mes needtr("좋아요를 표시했습니다.", "You liked this set.");
						@need_build_detail_liked = 1;
						++@need_build_detail_like_count;
					} else if (.@like_result == 2) {
						mes needtr("좋아요를 취소했습니다.", "You removed your like.");
						@need_build_detail_liked = 0;
						if (@need_build_detail_like_count > 0)
							--@need_build_detail_like_count;
					} else if (.@like_result == -2)
						mes needtr("현재 공개 중인 세팅에만 좋아요를 표시할 수 있습니다.", "You can only like a set that is currently published.");
					else if (.@like_result == -3)
						mes needtr("본인 계정으로 등록한 세팅에는 좋아요를 표시할 수 없습니다.", "You cannot like a set registered on your own account.");
					else if (.@like_result == -1)
						mes needtr("해당 세팅이 존재하지 않습니다.", "That set does not exist.");
					else if (.@like_result == -5)
						mes needtr("잠시 후 다시 시도해 주세요.", "Please try again in a moment.");
					else
						mes needtr("좋아요 요청을 처리하지 못했습니다. 잠시 후 다시 시도해 주세요.", "Could not process the like. Please try again in a moment.");

					continue;
				}

				if (.@detail_choice == 3)
					break;

				close;
			}

			continue;
		}

		.@nav = .@count + 1;

		if (.@page > 0) {
			if (.@choice == .@nav) {
				--.@page;
				continue;
			}
			++.@nav;
		}

		if (@need_build_has_more) {
			if (.@choice == .@nav) {
				++.@page;
				continue;
			}
			++.@nav;
		}

		// 정렬 방식 전환
		if (.@choice == .@nav) {
			.@sort_type = .@sort_type == 1 ? 2 : 1;
			.@page = 0;
			continue;
		}
		++.@nav;

		// 직업 다시 선택
		if (.@choice == .@nav)
			return;

		close;
	}

S_MyBuilds:
	.@page = 0;
	while (true) {
		clear;
		.@count = need_get_my_equipment_build_list(.@page);
		mes needtr("[내 등록 내역]", "[My Registrations]");
		mes needtr("현재 캐릭터로 등록한 세팅입니다. ", "Sets registered on this character - page ") + (.@page + 1) + needtr("페이지", ".");
		if (.@count < 0) {
			mes needtr("목록을 불러오지 못했습니다.", "Could not load the list.");
			next;
			return;
		}
		if (.@count == 0)
			mes needtr("등록한 장비 세팅이 없습니다.", "You have not registered any equipment sets.");
		else
			mes needtr("다른 세팅을 보려면 열려 있는 장비창을 먼저 닫아 주세요.", "Close the open equipment window before viewing another set.");

		.@menu$ = "";
		for (.@i = 0; .@i < .@count; ++.@i)
			.@menu$ += "[#" + @need_build_ids[.@i] + "] " + need_equipment_build_status_name(@need_build_statuses[.@i]) + " / " + replacestr(@need_build_titles$[.@i], ":", " - ") + ":";
		if (.@page > 0) .@menu$ += needtr("이전 페이지:", "Prev Page:");
		if (@need_build_has_more) .@menu$ += needtr("다음 페이지:", "Next Page:");
		.@menu$ += needtr("돌아가기:닫기", "Back:Close");
		.@choice = select(.@menu$);

		if (.@choice <= .@count) {
			.@index = .@choice - 1;
			.@build_id = @need_build_ids[.@index];
			.@status = @need_build_statuses[.@index];
			.@result = need_show_my_equipment_build(.@build_id);
			clear;
			mes needtr("[내 등록 내역]", "[My Registrations]");
			if (.@result <= 0) {
				mes needtr("해당 세팅 정보를 불러올 수 없습니다.", "Could not load that set.");
				next;
				continue;
			}

			mes needtr("등록 정보는 채팅창에서 확인할 수 있습니다.", "The registration details are shown in the chat window.");
			callsub S_RequestView, .@build_id, 2;

			while (true) {
				if (.@status == 0)
					.@detail_choice = select(needtr("장비 다시 보기:등록 취소:목록으로 돌아가기:창 닫기", "View Equipment:Cancel Registration:Back to List:Close"));
				else
					.@detail_choice = select(needtr("장비 다시 보기:목록으로 돌아가기:창 닫기", "View Equipment:Back to List:Close"));

				if (.@detail_choice == 1) {
					clear;
					mes needtr("[내 등록 내역]", "[My Registrations]");
					callsub S_RequestView, .@build_id, 2;
					continue;
				}

				if (.@status != 0) {
					if (.@detail_choice == 2)
						break;
					close;
				}

				if (.@detail_choice == 2) {
					clear;
					mes needtr("[장비 세팅 등록 취소]", "[Cancel Registration]");
					mes needtr("등록을 취소하면 검수 대상에서 제외되며 다시 복구할 수 없습니다.", "Cancelling removes the set from review and cannot be undone.");
					mes needtr("정말 이 세팅의 등록을 취소하시겠습니까?", "Do you really want to cancel this registration?");
					if (select(needtr("돌아가기:등록 취소", "Back:Cancel Registration")) == 1) {
						clear;
						mes needtr("[내 등록 내역]", "[My Registrations]");
						continue;
					}

					.@cancel = need_cancel_equipment_build(.@build_id);
					clear;
					mes needtr("[장비 세팅 등록 취소]", "[Cancel Registration]");
					if (.@cancel > 0) {
						mes needtr("장비 세팅 등록을 취소했습니다.", "The registration has been cancelled.");
						next;
						break;
					} else if (.@cancel == -13)
						mes needtr("이미 처리된 세팅이므로 취소할 수 없습니다.", "This set has already been processed and cannot be cancelled.");
					else if (.@cancel == -15)
						mes needtr("현재 캐릭터로 등록한 세팅이 아닙니다.", "This set was not registered on your current character.");
					else
						mes needtr("등록 취소를 처리하지 못했습니다.", "Could not cancel the registration.");
					next;
					break;
				}

				if (.@detail_choice == 3)
					break;
				close;
			}
			continue;
		}

		.@nav = .@count + 1;
		if (.@page > 0) {
			if (.@choice == .@nav) { --.@page; continue; }
			++.@nav;
		}
		if (@need_build_has_more) {
			if (.@choice == .@nav) { ++.@page; continue; }
			++.@nav;
		}
		if (.@choice == .@nav) return;
		close;
	}

S_Rewards:
	// 수령 시에는 현재 NPC 설정이 아니라 승인 당시 SQL에 저장된
	// reward_item_id와 reward_amount를 C++ 처리 계층에서 검증합니다.
	.@page = 0;
	while (true) {
		clear;
		.@count = need_get_my_equipment_build_reward_list(.@page);
		mes needtr("[보상 수령]", "[Claim Reward]");
		mes needtr("현재 캐릭터로 등록한 세팅 중 수령 가능한 보상입니다.", "Rewards you can claim for sets registered on this character.");
		mes needtr((.@page + 1) + "페이지", "Page " + (.@page + 1));
		if (.@count < 0) {
			mes needtr("보상 목록을 불러오지 못했습니다.", "Could not load the reward list.");
			next;
			return;
		}
		if (.@count == 0)
			mes needtr("수령 가능한 보상이 없습니다.", "You have no rewards to claim.");
		.@menu$ = "";
		for (.@i = 0; .@i < .@count; ++.@i)
			.@menu$ += "[#" + @need_reward_build_ids[.@i] + "] " + replacestr(@need_reward_titles$[.@i], ":", " - ") + ":";
		if (.@page > 0) .@menu$ += needtr("이전 페이지:", "Prev Page:");
		if (@need_reward_has_more) .@menu$ += needtr("다음 페이지:", "Next Page:");
		.@menu$ += needtr("돌아가기:닫기", "Back:Close");
		.@choice = select(.@menu$);
		if (.@choice <= .@count) {
			.@build_id = @need_reward_build_ids[.@choice - 1];
			.@detail = need_show_my_equipment_build_reward(.@build_id);
			clear;
			mes needtr("[보상 수령]", "[Claim Reward]");
			if (.@detail <= 0) {
				mes needtr("현재는 수령할 수 없는 보상입니다.", "This reward cannot be claimed right now.");
				next;
				continue;
			}
			mes needtr("세팅 번호: #", "Set No.: #") + .@build_id;
			mes needtr("세팅 제목: ", "Set title: ") + @need_reward_detail_title$;
			mes needtr("세팅 상태: 승인 및 공개", "Set status: Approved and published");
			mes needtr("보상 아이템: ", "Reward item: ") + getitemname(@need_reward_detail_item_id);
			mes needtr("보상 수량: ", "Reward amount: ") + @need_reward_detail_amount;
			mes needtr("이 보상을 수령하시겠습니까?", "Do you want to claim this reward?");
			if (select(needtr("보상 수령:목록으로 돌아가기", "Claim Reward:Back to List")) == 1) {
				.@claim = need_claim_equipment_build_reward(.@build_id);
				if (.@claim == 1) mes needtr("보상이 인벤토리에 지급되었습니다.", "The reward has been added to your inventory.");
				else if (.@claim == -2) mes needtr("해당 세팅을 등록한 계정과 캐릭터만 보상을 수령할 수 있습니다.", "Only the account and character that registered the set can claim the reward.");
				else if (.@claim == -3) mes needtr("현재 보상 수령 대상이 아닌 세팅입니다.", "This set is not eligible for a reward right now.");
				else if (.@claim == -4) mes needtr("이미 수령했거나 현재 지급 처리 중인 보상입니다.", "This reward was already claimed or is being processed.");
				else if (.@claim == -6) {
					mes needtr("인벤토리 공간 또는 소지 무게가 부족합니다.", "You do not have enough inventory space or weight capacity.");
					mes needtr("인벤토리를 정리한 후 다시 수령해 주세요.", "Please make room and claim it again.");
				} else if (.@claim == -5) mes needtr("보상 정보를 확인할 수 없습니다. 운영자에게 문의해 주세요.", "Could not read the reward data. Please contact a GM.");
				else if (.@claim == -9) mes needtr("잠시 후 다시 시도해 주세요.", "Please try again in a moment.");
				else mes needtr("보상 지급을 완료하지 못했습니다. 반복해서 시도하지 말고 운영자에게 문의해 주세요.", "The reward could not be delivered. Do not retry - please contact a GM.");
			}
			next;
			continue;
		}
		.@nav = .@count + 1;
		if (.@page > 0) {
			if (.@choice == .@nav) { --.@page; continue; }
			++.@nav;
		}
		if (@need_reward_has_more) {
			if (.@choice == .@nav) { ++.@page; continue; }
			++.@nav;
		}
		if (.@choice == .@nav) return;
		close;
	}

S_Admin:
	clear;
	if (!need_equipment_build_is_admin()) {
		mes needtr("운영자 권한이 없습니다.", "You do not have GM permission.");
		next;
		return;
	}
	mes needtr("[장비 세팅 관리]", "[Equipment Set Admin]");
	.@choice = select(needtr("검수 대기:공개 세팅:반려 세팅:삭제 세팅:세팅 번호로 검색:캐릭터명으로 검색:계정 ID로 검색:돌아가기", "Pending Review:Public Sets:Rejected Sets:Deleted Sets:Search by Set No.:Search by Character:Search by Account ID:Back"));
	if (.@choice <= 4) {
		callsub S_AdminPage, .@choice - 1, 0, "";
		return;
	}
	if (.@choice == 8)
		return;
	if (.@choice == 5) {
		mes needtr("세팅 번호를 입력해 주세요.", "Please enter the set number.");
		input .@value;
		.@value$ = "" + .@value;
		.@type = 1;
	} else if (.@choice == 6) {
		mes needtr("정확한 캐릭터명을 입력해 주세요.", "Please enter the exact character name.");
		input .@value$;
		.@type = 2;
	} else {
		mes needtr("계정 ID를 입력해 주세요.", "Please enter the account ID.");
		input .@value;
		.@value$ = "" + .@value;
		.@type = 3;
	}
	callsub S_AdminPage, 0, .@type, .@value$;
	return;

S_AdminPage:
	.@status_filter = getarg(0);
	.@search_type = getarg(1);
	.@search_value$ = getarg(2);
	.@page = 0;
	while (true) {
		clear;
		if (!need_equipment_build_is_admin()) {
			mes needtr("운영자 권한이 확인되지 않습니다.", "GM permission could not be confirmed.");
			next;
			return;
		}
		if (.@search_type)
			.@count = need_search_admin_equipment_builds(.@search_type, .@search_value$, .@page);
		else
			.@count = need_get_admin_equipment_build_list(.@status_filter, .@page);

		mes needtr("[장비 세팅 관리]", "[Equipment Set Admin]");
		mes needtr("페이지 ", "Page ") + (.@page + 1) + ".";
		if (.@count < 0) {
			mes .@count == -11 ? needtr("운영자 권한이 없습니다.", "You do not have GM permission.") : needtr("목록을 불러오지 못했습니다.", "Could not load the list.");
			next;
			return;
		}
		if (.@count == 0)
			mes needtr("조건에 맞는 장비 세팅이 없습니다.", "No equipment set matches those conditions.");
		else
			mes needtr("다른 세팅을 보려면 열려 있는 장비창을 먼저 닫아 주세요.", "Close the open equipment window before viewing another set.");

		.@menu$ = "";
		for (.@i = 0; .@i < .@count; ++.@i)
			.@menu$ += "[#" + @need_build_ids[.@i] + "] " + replacestr(@need_build_owners$[.@i], ":", " - ") + " / " + replacestr(@need_build_titles$[.@i], ":", " - ") + ":";
		if (.@page > 0) .@menu$ += needtr("이전 페이지:", "Prev Page:");
		if (@need_build_has_more) .@menu$ += needtr("다음 페이지:", "Next Page:");
		.@menu$ += needtr("관리 메뉴로 돌아가기:창 닫기", "Back to Admin Menu:Close");
		.@choice = select(.@menu$);

		if (.@choice <= .@count) {
			.@index = .@choice - 1;
			.@build_id = @need_build_ids[.@index];
			.@status = @need_build_statuses[.@index];
			.@detail = need_show_admin_equipment_build(.@build_id);
			clear;
			mes needtr("[장비 세팅 관리]", "[Equipment Set Admin]");
			if (.@detail <= 0) {
				mes needtr("해당 세팅 정보를 불러올 수 없습니다.", "Could not load that set.");
				next;
				continue;
			}

			mes needtr("검수 정보와 처리 내역은 채팅창에서 확인할 수 있습니다.", "The review data and action history are shown in the chat window.");
			callsub S_RequestView, .@build_id, 3;

			while (true) {
				if (.@status == 3)
					.@detail_choice = select(needtr("장비 다시 보기:목록으로 돌아가기:창 닫기", "View Equipment:Back to List:Close"));
				else
					.@detail_choice = select(needtr("장비 다시 보기:검수 및 상태 처리:목록으로 돌아가기:창 닫기", "View Equipment:Review & Status:Back to List:Close"));

				if (.@detail_choice == 1) {
					clear;
					mes needtr("[장비 세팅 관리]", "[Equipment Set Admin]");
					callsub S_RequestView, .@build_id, 3;
					continue;
				}

				if (.@status == 3) {
					if (.@detail_choice == 2)
						break;
					close;
				}

				if (.@detail_choice == 2) {
					callsub S_AdminAction, .@build_id, .@status;
					break;
				}

				if (.@detail_choice == 3)
					break;
				close;
			}
			continue;
		}

		.@nav = .@count + 1;
		if (.@page > 0) {
			if (.@choice == .@nav) { --.@page; continue; }
			++.@nav;
		}
		if (@need_build_has_more) {
			if (.@choice == .@nav) { ++.@page; continue; }
			++.@nav;
		}
		if (.@choice == .@nav) return;
		close;
	}

S_AdminAction:
	.@build_id = getarg(0);
	.@status = getarg(1);
	clear;
	mes needtr("[장비 세팅 관리]", "[Equipment Set Admin]");
	mes needtr("세팅 번호: #", "Set No.: #") + .@build_id;
	if (.@status == 0) {
		.@choice = select(needtr("승인 및 보상 지급:승인(보상 없음):반려:삭제:돌아가기", "Approve with Reward:Approve (No Reward):Reject:Delete:Back"));
		if (.@choice == 5)
			return;
		if (.@choice == 1 || .@choice == 2) {
			clear;
			mes needtr("[장비 세팅 승인]", "[Approve Equipment Set]");
			.@reward_item_id = .@choice == 1 ? .RewardItemId : 0;
			.@reward_amount = 0;
			if (.@choice == 1) {
				if (.RewardItemId <= 0 || getiteminfo(.RewardItemId, ITEMINFO_ID) != .RewardItemId) {
					mes needtr("보상 아이템이 설정되지 않아 보상 지급 승인을 진행할 수 없습니다.", "No reward item is configured so approval with a reward is not possible.");
					return;
				}
				mes needtr("보상 아이템: ", "Reward item: ") + getitemname(.RewardItemId);
				mes needtr("보상 수량을 입력해 주세요. (", "Enter the reward amount. (") + .RewardAmountMin + "-" + .RewardAmountMax + ").";
				input .@reward_amount;
				if (.@reward_amount < .RewardAmountMin || .@reward_amount > .RewardAmountMax) {
					mes needtr("보상 수량이 설정 범위를 벗어났습니다.", "The reward amount is outside the configured range.");
					return;
				}
				mes needtr("세팅 번호: #", "Set No.: #") + .@build_id;
				mes needtr("처리 내용: 승인 및 보상 지급", "Action: Approve with reward");
				mes needtr("보상 아이템: ", "Reward item: ") + getitemname(.RewardItemId);
				mes needtr("보상 수량: ", "Reward amount: ") + .@reward_amount;
			} else {
				mes needtr("세팅 번호: #", "Set No.: #") + .@build_id;
				mes needtr("처리 내용: 승인(보상 없음)", "Action: Approve (no reward)");
			}
			mes needtr("이 장비 세팅을 승인하시겠습니까?", "Do you want to approve this equipment set?");
			if (select(needtr("승인하기:돌아가기", "Approve:Back")) == 2)
				return;
			.@result = need_approve_equipment_build(.@build_id, .@reward_item_id, .@reward_amount);
			if (.@result > 0) mes needtr("장비 세팅을 승인하고 보상 설정을 저장했습니다.", "The set has been approved and the reward setting saved.");
			else if (.@result == -11) mes needtr("운영자 권한이 없습니다.", "You do not have GM permission.");
			else if (.@result == -13) mes needtr("이미 처리되어 현재 상태에서는 승인할 수 없습니다.", "It was already processed and cannot be approved in its current state.");
			else if (.@result == -2) mes needtr("보상 설정이 올바르지 않습니다. 아이템과 수량을 확인해 주세요.", "The reward setting is invalid. Please check the item and amount.");
			else mes needtr("승인 처리에 실패했습니다. 변경 사항은 저장되지 않았습니다.", "Approval failed. Nothing was saved.");
			return;
		}
		.@action = .@choice == 3 ? 2 : 3;
	} else if (.@status != 3) {
		if (select(needtr("삭제하기:돌아가기", "Delete:Back")) == 2)
			return;
		.@action = 3;
	} else {
		return;
	}
	clear;
	mes .@action == 2 ? needtr("[장비 세팅 반려]", "[Reject Equipment Set]") : needtr("[장비 세팅 삭제]", "[Delete Equipment Set]");
	mes needtr("세팅 번호: #", "Set No.: #") + .@build_id;
	.@reason$ = "";
	mes .@action == 2 ? needtr("반려 사유를 입력해 주세요. (필수, 최대 200자)", "Enter the reason for rejection. (required, up to 200 characters)") : needtr("삭제 사유를 입력해 주세요. (필수, 최대 200자)", "Enter the reason for deletion. (required, up to 200 characters)");
	input .@reason$;
	mes needtr("세팅 #", "Apply this action to set #") + .@build_id + needtr("에 해당 처리를 적용하시겠습니까?", "?");
	if (select(needtr("처리하기:돌아가기", "Apply:Back")) == 2)
		return;
	.@result = need_review_equipment_build(.@build_id, .@action, .@reason$);
	if (.@result > 0)
		mes needtr("처리를 완료했으며 관리 기록에 저장했습니다.", "Done - the action was saved to the admin log.");
	else if (.@result == -11)
		mes needtr("운영자 권한이 없습니다.", "You do not have GM permission.");
	else if (.@result == -12)
		mes needtr("해당 세팅이 존재하지 않습니다.", "That set does not exist.");
	else if (.@result == -13)
		mes needtr("이미 처리된 장비 세팅입니다.", "This equipment set was already processed.");
	else if (.@result == -14)
		mes needtr("사유는 1~200자로 입력해야 하며 제어 문자를 사용할 수 없습니다.", "The reason must be 1-200 characters and cannot contain control characters.");
	else
		mes needtr("처리에 실패했습니다. 변경 사항은 저장되지 않았습니다.", "The action failed. Nothing was saved.");
	return;

S_RequestView:
	.@build_id = getarg(0);
	.@view_mode = getarg(1);
	.@view_result = need_open_equipment_build_window(.@build_id, .@view_mode);
	if (.@view_result > 0) {
		mes needtr("장비 보기 요청을 전송했습니다.", "The equipment view request has been sent.");
		mes needtr("창이 열리지 않았다면 잠시 후 [장비 다시 보기]를 선택해 주세요.", "If the window did not open choose [View Equipment] again in a moment.");
	} else {
		mes needtr("장비 보기 요청을 처리하지 못했습니다.", "Could not process the equipment view request.");
		mes needtr("잠시 후 다시 시도해 주세요.", "Please try again in a moment.");
	}
	return;

OnInit:
	.RewardItemId = 7720; // 장비 세팅 승인 보상 아이템 ID
	.RewardAmountMin = 1;
	.RewardAmountMax = 100;
	end;
}
