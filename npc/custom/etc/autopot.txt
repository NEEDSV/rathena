//============================================================
// NEED 오토큐어 설정 UI
// - 실제 자동 물약 처리는 C++ @autopot 사용
// - 이 스크립트는 UI/명령어 호출만 담당
//============================================================

-	script	NEED_오토큐어_UI	-1,{

OnInit:
	// 소스의 @오토큐어와 충돌하지 않도록 별도 진입 명령 사용
	bindatcmd "오토큐어설정", strnpcinfo(3) + "::OnCommand", 0, 99;
	bindatcmd "autopotui", strnpcinfo(3) + "::OnCommand", 0, 99;
	end;

OnCommand:
	callsub S_Main;
	end;


//============================================================
// 메인 메뉴
//============================================================
S_Main:
	while (1) {
		mes "[오토큐어]";
		mes "자동 물약 설정을 관리합니다.";
		mes "원하는 항목을 선택해주세요.";
		next;

		switch(select("HP 자동 물약 설정:SP 자동 물약 설정:HP / SP 활성화 관리:사용 속도 설정:프리셋 관리:현재 설정 확인:전체 비활성화:종료")) {
		case 1:
			callsub S_SetPotion, "hp";
			break;

		case 2:
			callsub S_SetPotion, "sp";
			break;

		case 3:
			callsub S_Toggle;
			break;

		case 4:
			callsub S_Delay;
			break;

		case 5:
			callsub S_Preset;
			break;

		case 6:
			close2;
			atcommand "@autopot info";
			end;

		case 7:
			mes "[오토큐어]";
			mes "HP와 SP 자동 물약을 모두 비활성화하시겠습니까?";
			next;

			if (select("비활성화:취소") == 2)
				break;

			close2;
			atcommand "@autopot off";
			end;

		default:
			close;
		}
	}

	return;


//============================================================
// HP / SP 물약 설정
//============================================================
S_SetPotion:
	.@type$ = getarg(0);

	.@item = callsub(S_SelectHealingItem);

	if (.@item <= 0)
		return;

	mes "[오토큐어]";
	mes "선택한 아이템:";
	mes "^0055FF" + getitemname(.@item) + "^000000";
	mes "보유 수량: " + countitem(.@item) + "개";
	next;

	mes "[오토큐어]";
	if (.@type$ == "hp")
		mes "HP가 몇 % 미만일 때 사용할까요?";
	else
		mes "SP가 몇 % 미만일 때 사용할까요?";

	mes "1 ~ 100 사이의 숫자를 입력해주세요.";
	input .@percent, 1, 100;
	next;

	mes "[오토큐어]";
	mes "자동 물약 반응 속도를 선택해주세요.";
	mes "현재 서버 허용 범위: 300 ~ 2000ms";
	next;

	switch(select("빠름 - 300ms:보통 - 500ms:여유 - 1000ms:직접 입력:취소")) {
	case 1:
		.@delay = 300;
		break;

	case 2:
		.@delay = 500;
		break;

	case 3:
		.@delay = 1000;
		break;

	case 4:
		mes "[오토큐어]";
		mes "300 ~ 2000 사이의 값을 입력해주세요.";
		input .@delay, 300, 2000;
		break;

	default:
		return;
	}

	mes "[오토큐어]";
	mes "다음과 같이 설정합니다.";
	mes " ";

	if (.@type$ == "hp")
		mes "구분: ^FF0000HP^000000";
	else
		mes "구분: ^0055FFSP^000000";

	mes "물약: " + getitemname(.@item);
	mes "사용 기준: " + .@percent + "% 미만";
	mes "반응 속도: " + .@delay + "ms";
	next;

	if (select("적용:취소") == 2)
		return;

	.@command$ =
		"@autopot " + .@type$ + " " +
		.@item + " " +
		.@percent + " " +
		.@delay;

	close2;
	atcommand .@command$;
	end;


//============================================================
// 인벤토리 회복 아이템 선택
//============================================================
S_SelectHealingItem:
	getinventorylist;

	.@count = 0;
	.@menu$ = "";

	for (.@i = 0; .@i < @inventorylist_count; .@i++) {

		.@item = @inventorylist_id[.@i];

		if (getiteminfo(.@item, ITEMINFO_TYPE) != IT_HEALING)
			continue;

		// UI가 너무 길어지는 것을 방지
		if (.@count >= 20)
			break;

		.@name$ = getitemname(.@item);

		if (.@name$ == "" || .@name$ == "null")
			continue;

		// select 구분자 충돌 방지
		.@name$ = replacestr(.@name$, ":", "：");

		.@items[.@count] = .@item;

		.@menu$ +=
			.@name$ +
			" (" + countitem(.@item) + "개):";

		.@count++;
	}

	if (.@count <= 0) {
		mes "[오토큐어]";
		mes "인벤토리에 설정 가능한";
		mes "회복 아이템이 없습니다.";
		next;
		return 0;
	}

	mes "[오토큐어]";
	mes "사용할 회복 아이템을 선택해주세요.";

	if (.@count >= 20)
		mes "^777777※ 최대 20종까지만 표시됩니다.^000000";

	next;

	.@selected = select(.@menu$ + "취소") - 1;

	if (.@selected >= .@count)
		return 0;

	return .@items[.@selected];


//============================================================
// HP / SP ON/OFF
//============================================================
S_Toggle:
	mes "[오토큐어]";
	mes "활성화 상태를 변경합니다.";
	next;

	switch(select("HP 활성화:HP 비활성화:SP 활성화:SP 비활성화:HP / SP 모두 비활성화:뒤로가기")) {
	case 1:
		close2;
		atcommand "@autopot hp on";
		end;

	case 2:
		close2;
		atcommand "@autopot hp off";
		end;

	case 3:
		close2;
		atcommand "@autopot sp on";
		end;

	case 4:
		close2;
		atcommand "@autopot sp off";
		end;

	case 5:
		close2;
		atcommand "@autopot off";
		end;

	default:
		return;
	}


//============================================================
// 전체 딜레이 변경
//============================================================
S_Delay:
	mes "[오토큐어]";
	mes "자동 물약 검사 속도를 설정합니다.";
	mes "현재 허용 범위는 300 ~ 2000ms입니다.";
	next;

	switch(select("빠름 - 300ms:보통 - 500ms:여유 - 1000ms:직접 입력:뒤로가기")) {
	case 1:
		.@delay = 300;
		break;

	case 2:
		.@delay = 500;
		break;

	case 3:
		.@delay = 1000;
		break;

	case 4:
		mes "[오토큐어]";
		mes "300 ~ 2000 사이의 값을 입력해주세요.";
		input .@delay, 300, 2000;
		break;

	default:
		return;
	}

	close2;
	atcommand "@autopot delay " + .@delay;
	end;


//============================================================
// 프리셋
//============================================================
S_Preset:
	while (1) {
		mes "[오토큐어 프리셋]";
		mes "현재 설정을 최대 5개까지";
		mes "계정 단위로 저장할 수 있습니다.";
		next;

		switch(select("프리셋 목록 확인:프리셋 저장:프리셋 불러오기:프리셋 삭제:뒤로가기")) {
		case 1:
			close2;
			atcommand "@autopot preset";
			end;

		case 2:
			callsub S_PresetSave;
			break;

		case 3:
			callsub S_PresetLoad;
			break;

		case 4:
			callsub S_PresetDelete;
			break;

		default:
			return;
		}
	}

	return;


//============================================================
// 프리셋 저장
//============================================================
S_PresetSave:
	mes "[오토큐어 프리셋]";
	mes "저장할 슬롯을 선택해주세요.";
	next;

	.@slot = select("1번:2번:3번:4번:5번:취소");
	if (.@slot > 5)
		return;

	mes "[오토큐어 프리셋]";
	mes "프리셋 이름을 입력해주세요.";
	mes "^7777771 ~ 24바이트^000000";
	next;

	input .@name$;

	if (.@name$ == "") {
		mes "[오토큐어 프리셋]";
		mes "이름을 입력해주세요.";
		next;
		return;
	}

	close2;
	atcommand "@autopot save " + .@slot + " " + .@name$;
	end;


//============================================================
// 프리셋 불러오기
//============================================================
S_PresetLoad:
	mes "[오토큐어 프리셋]";
	mes "불러올 슬롯을 선택해주세요.";
	next;

	.@slot = select("1번:2번:3번:4번:5번:취소");

	if (.@slot > 5)
		return;

	close2;
	atcommand "@autopot load " + .@slot;
	end;


//============================================================
// 프리셋 삭제
//============================================================
S_PresetDelete:
	mes "[오토큐어 프리셋]";
	mes "삭제할 슬롯을 선택해주세요.";
	next;
	.@slot = select("1번:2번:3번:4번:5번:취소");
	if (.@slot > 5)
		return;

	mes "[오토큐어 프리셋]";
	mes .@slot + "번 프리셋을 삭제하시겠습니까?";
	next;

	if (select("삭제:취소") == 2)
		return;

	close2;
	atcommand "@autopot delete " + .@slot;
	end;
}
