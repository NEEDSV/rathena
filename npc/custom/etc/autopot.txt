//============================================================
// NEED 오토큐어 설정 UI
// - 실제 자동 물약 처리는 C++ @autopot 사용
// - 이 스크립트는 UI/명령어 호출만 담당
//============================================================

-	script	NEED_오토큐어_UI	-1,{

OnInit:
	// 소스의 @오토큐어와 충돌하지 않도록 별도 진입 명령 사용
	bindatcmd "오토큐어설정", strnpcinfo(3) + "::OnCommand", 0, 99;
	bindatcmd "autopotui", strnpcinfo(3) + "::OnCommand", 0, 99;
	end;

OnCommand:
	callsub S_Main;
	end;


//============================================================
// 메인 메뉴
//============================================================
S_Main:
	while (1) {
		mes needtr("[오토큐어]", "[Auto Cure]");
		mes needtr("자동 물약 설정을 관리합니다.", "Manage your auto-potion settings.");
		mes needtr("원하는 항목을 선택해주세요.", "Please choose what you want to do.");
		next;

		switch(select(needtr("HP 자동 물약 설정:SP 자동 물약 설정:HP / SP 활성화 관리:사용 속도 설정:프리셋 관리:현재 설정 확인:전체 비활성화:종료", "HP auto-potion setup:SP auto-potion setup:Manage HP / SP activation:Response speed:Manage presets:Check current settings:Disable everything:Exit"))) {
		case 1:
			callsub S_SetPotion, "hp";
			break;

		case 2:
			callsub S_SetPotion, "sp";
			break;

		case 3:
			callsub S_Toggle;
			break;

		case 4:
			callsub S_Delay;
			break;

		case 5:
			callsub S_Preset;
			break;

		case 6:
			close2;
			atcommand "@autopot info";
			end;

		case 7:
			mes needtr("[오토큐어]", "[Auto Cure]");
			mes needtr("HP와 SP 자동 물약을 모두 비활성화하시겠습니까?", "Disable both the HP and SP auto-potion?");
			next;

			if (select(needtr("비활성화:취소", "Disable:Cancel")) == 2)
				break;

			close2;
			atcommand "@autopot off";
			end;

		default:
			close;
		}
	}

	return;


//============================================================
// HP / SP 물약 설정
//============================================================
S_SetPotion:
	.@type$ = getarg(0);

	.@item = callsub(S_SelectHealingItem);

	if (.@item <= 0)
		return;

	mes needtr("[오토큐어]", "[Auto Cure]");
	mes needtr("선택한 아이템:", "Selected item:");
	mes "^0055FF" + getitemname(.@item) + "^000000";
	mes needtr("보유 수량: ", "You have ") + countitem(.@item) + needtr("개", ".");
	next;

	mes needtr("[오토큐어]", "[Auto Cure]");
	if (.@type$ == "hp")
		mes needtr("HP가 몇 % 미만일 때 사용할까요?", "Use it when HP falls below what %?");
	else
		mes needtr("SP가 몇 % 미만일 때 사용할까요?", "Use it when SP falls below what %?");

	mes needtr("1 ~ 100 사이의 숫자를 입력해주세요.", "Please enter a number between 1 and 100.");
	input .@percent, 1, 100;
	next;

	mes needtr("[오토큐어]", "[Auto Cure]");
	mes needtr("자동 물약 반응 속도를 선택해주세요.", "Please choose the auto-potion response speed.");
	mes needtr("현재 서버 허용 범위: 100 ~ 2000ms", "Server allows: 100 ~ 2000ms");
	next;

	switch(select(needtr("매우 빠름 - 100ms:빠름 - 300ms:보통 - 500ms:여유 - 1000ms:직접 입력:취소", "Very fast - 100ms:Fast - 300ms:Normal - 500ms:Relaxed - 1000ms:Enter a value:Cancel"))) {
	case 1:
		.@delay = 100;
		break;

	case 2:
		.@delay = 300;
		break;

	case 3:
		.@delay = 500;
		break;

	case 4:
		.@delay = 1000;
		break;

	case 5:
		mes needtr("[오토큐어]", "[Auto Cure]");
		mes needtr("100 ~ 2000 사이의 값을 입력해주세요.", "Please enter a value between 100 and 2000.");
		input .@delay, 100, 2000;
		break;

	default:
		return;
	}

	mes needtr("[오토큐어]", "[Auto Cure]");
	mes needtr("다음과 같이 설정합니다.", "These are the settings to apply.");
	mes " ";

	if (.@type$ == "hp")
		mes needtr("구분: ^FF0000HP^000000", "Type: ^FF0000HP^000000");
	else
		mes needtr("구분: ^0055FFSP^000000", "Type: ^0055FFSP^000000");

	mes needtr("물약: ", "Potion: ") + getitemname(.@item);
	mes needtr("사용 기준: ", "Trigger: below ") + .@percent + needtr("% 미만", "%");
	mes needtr("반응 속도: ", "Response speed: ") + .@delay + "ms";
	next;

	if (select(needtr("적용:취소", "Apply:Cancel")) == 2)
		return;

	.@command$ =
		"@autopot " + .@type$ + " " +
		.@item + " " +
		.@percent + " " +
		.@delay;

	close2;
	atcommand .@command$;
	end;


//============================================================
// 인벤토리 회복 아이템 선택
//============================================================
S_SelectHealingItem:
	getinventorylist;

	.@count = 0;
	.@menu$ = "";

	for (.@i = 0; .@i < @inventorylist_count; .@i++) {

		.@item = @inventorylist_id[.@i];

		if (getiteminfo(.@item, ITEMINFO_TYPE) != IT_HEALING)
			continue;

		// UI가 너무 길어지는 것을 방지
		if (.@count >= 20)
			break;

		.@name$ = getitemname(.@item);

		if (.@name$ == "" || .@name$ == "null")
			continue;

		// select 구분자 충돌 방지
		.@name$ = replacestr(.@name$, ":", "：");

		.@items[.@count] = .@item;

		.@menu$ +=
			.@name$ +
			" (" + countitem(.@item) + needtr("개):", "):");

		.@count++;
	}

	if (.@count <= 0) {
		mes needtr("[오토큐어]", "[Auto Cure]");
		mes needtr("인벤토리에 설정 가능한", "There are no recovery items");
		mes needtr("회복 아이템이 없습니다.", "in your inventory to set up.");
		next;
		return 0;
	}

	mes needtr("[오토큐어]", "[Auto Cure]");
	mes needtr("사용할 회복 아이템을 선택해주세요.", "Please choose a recovery item to use.");

	if (.@count >= 20)
		mes needtr("^777777※ 최대 20종까지만 표시됩니다.^000000", "^777777※ Only the first 20 kinds are shown.^000000");

	next;

	.@selected = select(.@menu$ + needtr("취소", "Cancel")) - 1;

	if (.@selected >= .@count)
		return 0;

	return .@items[.@selected];


//============================================================
// HP / SP ON/OFF
//============================================================
S_Toggle:
	mes needtr("[오토큐어]", "[Auto Cure]");
	mes needtr("활성화 상태를 변경합니다.", "Change the activation state.");
	next;

	switch(select(needtr("HP 활성화:HP 비활성화:SP 활성화:SP 비활성화:HP / SP 모두 비활성화:뒤로가기", "Enable HP:Disable HP:Enable SP:Disable SP:Disable both HP / SP:Back"))) {
	case 1:
		close2;
		atcommand "@autopot hp on";
		end;

	case 2:
		close2;
		atcommand "@autopot hp off";
		end;

	case 3:
		close2;
		atcommand "@autopot sp on";
		end;

	case 4:
		close2;
		atcommand "@autopot sp off";
		end;

	case 5:
		close2;
		atcommand "@autopot off";
		end;

	default:
		return;
	}


//============================================================
// 전체 딜레이 변경
//============================================================
S_Delay:
	mes needtr("[오토큐어]", "[Auto Cure]");
	mes needtr("자동 물약 검사 속도를 설정합니다.", "Set the auto-potion check interval.");
	mes needtr("현재 허용 범위는 100 ~ 2000ms입니다.", "The allowed range is 100 ~ 2000ms.");
	next;

	switch(select(needtr("매우 빠름 - 100ms:빠름 - 300ms:보통 - 500ms:여유 - 1000ms:직접 입력:뒤로가기", "Very fast - 100ms:Fast - 300ms:Normal - 500ms:Relaxed - 1000ms:Enter a value:Back"))) {
	case 1:
		.@delay = 100;
		break;

	case 2:
		.@delay = 300;
		break;

	case 3:
		.@delay = 500;
		break;

	case 4:
		.@delay = 1000;
		break;

	case 5:
		mes needtr("[오토큐어]", "[Auto Cure]");
		mes needtr("100 ~ 2000 사이의 값을 입력해주세요.", "Please enter a value between 100 and 2000.");
		input .@delay, 100, 2000;
		break;

	default:
		return;
	}

	close2;
	atcommand "@autopot delay " + .@delay;
	end;


//============================================================
// 프리셋
//============================================================
S_Preset:
	while (1) {
		mes needtr("[오토큐어 프리셋]", "[Auto Cure Preset]");
		mes needtr("현재 설정을 최대 5개까지", "You can save up to five of your");
		mes needtr("계정 단위로 저장할 수 있습니다.", "current settings per account.");
		next;

		switch(select(needtr("프리셋 목록 확인:프리셋 저장:프리셋 불러오기:프리셋 삭제:뒤로가기", "List presets:Save preset:Load preset:Delete preset:Back"))) {
		case 1:
			close2;
			atcommand "@autopot preset";
			end;

		case 2:
			callsub S_PresetSave;
			break;

		case 3:
			callsub S_PresetLoad;
			break;

		case 4:
			callsub S_PresetDelete;
			break;

		default:
			return;
		}
	}

	return;


//============================================================
// 프리셋 저장
//============================================================
S_PresetSave:
	mes needtr("[오토큐어 프리셋]", "[Auto Cure Preset]");
	mes needtr("저장할 슬롯을 선택해주세요.", "Please choose a slot to save to.");
	next;

	.@slot = select(needtr("1번:2번:3번:4번:5번:취소", "Slot 1:Slot 2:Slot 3:Slot 4:Slot 5:Cancel"));
	if (.@slot > 5)
		return;

	mes needtr("[오토큐어 프리셋]", "[Auto Cure Preset]");
	mes needtr("프리셋 이름을 입력해주세요.", "Please enter a name for the preset.");
	mes needtr("^7777771 ~ 24바이트^000000", "^7777771 ~ 24 bytes^000000");
	next;

	input .@name$;

	if (.@name$ == "") {
		mes needtr("[오토큐어 프리셋]", "[Auto Cure Preset]");
		mes needtr("이름을 입력해주세요.", "Please enter a name.");
		next;
		return;
	}

	close2;
	atcommand "@autopot save " + .@slot + " " + .@name$;
	end;


//============================================================
// 프리셋 불러오기
//============================================================
S_PresetLoad:
	mes needtr("[오토큐어 프리셋]", "[Auto Cure Preset]");
	mes needtr("불러올 슬롯을 선택해주세요.", "Please choose a slot to load.");
	next;

	.@slot = select(needtr("1번:2번:3번:4번:5번:취소", "Slot 1:Slot 2:Slot 3:Slot 4:Slot 5:Cancel"));

	if (.@slot > 5)
		return;

	close2;
	atcommand "@autopot load " + .@slot;
	end;


//============================================================
// 프리셋 삭제
//============================================================
S_PresetDelete:
	mes needtr("[오토큐어 프리셋]", "[Auto Cure Preset]");
	mes needtr("삭제할 슬롯을 선택해주세요.", "Please choose a slot to delete.");
	next;
	.@slot = select(needtr("1번:2번:3번:4번:5번:취소", "Slot 1:Slot 2:Slot 3:Slot 4:Slot 5:Cancel"));
	if (.@slot > 5)
		return;

	mes needtr("[오토큐어 프리셋]", "[Auto Cure Preset]");
	mes .@slot + needtr("번 프리셋을 삭제하시겠습니까?", ". Delete this preset?");
	next;

	if (select(needtr("삭제:취소", "Delete:Cancel")) == 2)
		return;

	close2;
	atcommand "@autopot delete " + .@slot;
	end;
}
