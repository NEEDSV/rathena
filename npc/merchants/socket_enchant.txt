			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[세야블렘]";
				mes "좋습니다. 그럼 지금 바로 작업을 시작하겠습니다!";
				mes "좋은 결과가 나오길 간절히 기도하는 게 좋을 겁니다.";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY; // only S Class items have Lord of Vermillion effect
					mes "[세야블렘]";
					mes "성공이군요! 아주 잘 되었습니다.";
					mes "작업이 아주 깔끔하게 마무리됐네요. 축하드립니다!";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[세야블렘]";
					mes "다음에 또 뵙지요, 친구!";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[세야블렘]";
					mes "아차! ...유감이군요. 실패하고 말았습니다.";
					mes "하지만 제 실수가 아니라 운이 없었을 뿐이니 절 탓하진 마십시오.";
					mes "이건 자네의 운명이며, 신의 뜻이었을 겁니다. 아시겠습니까?";
					mes "너무 상심하지 마시고, 다음에 다시 도전해 보십시오.";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[세야블렘]";
					mes "다음번엔 반드시 행운이 따르길 빌겠습니다!";
					close;
				}
			}
			else
			{
				mes "[세야블렘]";
				mes "지금 바로 작업을 시작하고 싶지만, 몇 가지 빠진 것들이 있군요.";
				mes "장비와 필수 재료, 그리고 수수료를 모두 챙겼는지 다시 한번 확인해 주십시오.";
				close;
			}
		case 2:
			mes "[세야블렘]";
			mes "생각할 시간이 좀 필요하신 모양이군요?";
			mes "이해합니다. 충분히 고민해 보십시오.";
			mes "하지만 인생은 위험을 감수하지 않으면 재미가 없는 법이라네~";
			close;
	};
