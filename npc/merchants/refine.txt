lasa_in01,161,67,6	script	홀그렌	4_M_03,{
	mes needtr("[홀그렌]", "[Hollgrehenn]");
	mes needtr("저는 무기와 장비를 제련할 수 있는 대장장이입니다.", "I am a blacksmith who can refine weapons and equipment.");
	mes needtr("어떤 제련 서비스를 이용하시겠습니까?", "Which refine service would you like to use?");
	next;

	switch(select(needtr("일반 제련", "Normal Refine"),needtr("안전 제련", "Safe Refine"),needtr("제련권 사용", "Use a Refine Ticket"),needtr("취소", "Cancel")))
	{
		case 1:
			if (getbattleflag("feature.refineui")) {
				close2;
				refineui();
				end;
			}

			callfunc "refinemain","Hollgrehenn",0;
			end;

		case 2:
			callfunc "F_NeedSafeRefine";
			end;

		case 3:
			callfunc "F_RefineTime";
			end;

		default:
			close;
	}

OnInit:
	waitingroom "제련",0;
	end;
}

// 안전 제련
// - 방어구: +4까지, 에르늄 1개 / 1회, 20,000 제니 / 1회
// - 무기 Lv.1: +7까지, 프라콘 1개 / 1회, 50 제니 / 1회
// - 무기 Lv.2: +6까지, 엠베르타콘 1개 / 1회, 200 제니 / 1회
// - 무기 Lv.3: +5까지, 오리데오콘 1개 / 1회, 5,000 제니 / 1회
// - 무기 Lv.4: +4까지, 오리데오콘 1개 / 1회, 20,000 제니 / 1회
// - 쉐도우 장비: +4까지, 서버 refine DB의 Normal 재료/제니 비용 사용
//============================================================
function	script	F_NeedSafeRefine	{
	disable_items;

	setarray .@position$[0],
		"상단",
		"갑옷",
		"왼손",
		"오른손",
		"걸칠것",
		"신발",
		"악세사리(L)",
		"악세사리(R)",
		"중단",
		"하단",
		"쉐도우 아머",
		"쉐도우 웨폰",
		"쉐도우 쉴드",
		"쉐도우 슈즈",
		"쉐도우 악세사리(L)",
		"쉐도우 악세사리(R)";

	setarray .@indices[0],
		EQI_HEAD_TOP,
		EQI_ARMOR,
		EQI_HAND_L,
		EQI_HAND_R,
		EQI_GARMENT,
		EQI_SHOES,
		EQI_ACC_L,
		EQI_ACC_R,
		EQI_HEAD_MID,
		EQI_HEAD_LOW,
		EQI_SHADOW_ARMOR,
		EQI_SHADOW_WEAPON,
		EQI_SHADOW_SHIELD,
		EQI_SHADOW_SHOES,
		EQI_SHADOW_ACC_L,
		EQI_SHADOW_ACC_R;

	.@count = 0;
	.@menu$ = "";

	for (.@i = 0; .@i < getarraysize(.@indices); ++.@i) {
		.@part = .@indices[.@i];

		if (!getequipisequiped(.@part))
			continue;

		if (!getequipisenableref(.@part))
			continue;

		.@item_id = getequipid(.@part);
		.@item_type = getiteminfo(.@item_id, ITEMINFO_TYPE);
		.@current_refine = getequiprefinerycnt(.@part);
		.@safe_refine = 0;

		if (.@item_type == IT_WEAPON) {
			switch (getequipweaponlv(.@part)) {
				case 1: .@safe_refine = 7; break;
				case 2: .@safe_refine = 6; break;
				case 3: .@safe_refine = 5; break;
				case 4: .@safe_refine = 4; break;
				default: continue;
			}
		}
		else if (.@item_type == IT_ARMOR || .@item_type == IT_SHADOWGEAR) {
			// 방어구 및 쉐도우 장비의 안전 제련 한계는 +4
			.@safe_refine = 4;
		}
		else {
			continue;
		}

		// 이미 안전 제련 한계에 도달한 장비는 목록에서 제외
		if (.@current_refine >= .@safe_refine)
			continue;

		.@part_list[.@count] = .@part;
		.@target_list[.@count] = .@safe_refine;
		.@menu$ += "+" + .@current_refine + " " + getequipname(.@part)
			+ "  →  +" + .@safe_refine + ":";
		++.@count;
	}

	if (.@count <= 0) {
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes needtr("현재 장착 중인 장비 중", "None of the equipment you are wearing");
		mes needtr("안전 제련이 가능한 장비가 없습니다.", "can be safe refined.");
		mes " ";
		mes needtr("제련 가능한 장비를 장착하거나,", "Please equip a refinable item or check");
		mes needtr("이미 안전 제련 한계에 도달했는지 확인해주세요.", "whether it has already reached the safe refine limit.");
		close;
	}

	.@menu$ += needtr("취소", "Cancel");

	mes needtr("[홀그렌]", "[Hollgrehenn]");
	mes needtr("안전 제련할 장비를 선택해주세요.", "Please choose the equipment to safe refine.");
	mes " ";
	mes needtr("현재 제련도에서 안전 제련 한계까지", "All the materials and zeny needed to go from the current");
	mes needtr("필요한 재료와 제니를 한 번에 사용합니다.", "refine level up to the safe limit are spent at once.");
	next;

	.@select = select(.@menu$) - 1;

	if (.@select < 0 || .@select >= .@count)
		close;

	.@part = .@part_list[.@select];
	.@safe_refine = .@target_list[.@select];

	if (!getequipisequiped(.@part) || !getequipisenableref(.@part)) {
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes needtr("선택한 장비의 착용 상태가 변경되었습니다.", "The equipped state of the selected item has changed.");
		mes needtr("다시 시도해주세요.", "Please try again.");
		close;
	}

	.@item_id = getequipid(.@part);
	.@current_refine = getequiprefinerycnt(.@part);
	.@item_type = getiteminfo(.@item_id, ITEMINFO_TYPE);
	.@material_id = 0;
	.@zeny_per_refine = 0;

	if (.@item_type == IT_WEAPON) {
		switch (getequipweaponlv(.@part)) {
			case 1:
				.@safe_refine = 7;
				.@material_id = 1010; // Phracon
				.@zeny_per_refine = 50;
				break;

			case 2:
				.@safe_refine = 6;
				.@material_id = 1011; // Emveretarcon
				.@zeny_per_refine = 200;
				break;

			case 3:
				.@safe_refine = 5;
				.@material_id = 984; // Oridecon
				.@zeny_per_refine = 5000;
				break;

			case 4:
				.@safe_refine = 4;
				.@material_id = 984; // Oridecon
				.@zeny_per_refine = 20000;
				break;

			default:
				mes needtr("[홀그렌]", "[Hollgrehenn]");
				mes needtr("지원하지 않는 무기 레벨입니다.", "That weapon level is not supported.");
				close;
		}
	}
	else if (.@item_type == IT_ARMOR) {
		.@safe_refine = 4;
		.@material_id = 985; // Elunium
		.@zeny_per_refine = 20000;
	}
	else if (.@item_type == IT_SHADOWGEAR) {
		// 쉐도우 장비는 별도 아이템 타입/제련 그룹을 사용한다.
		// 서버 refine DB의 Normal 재료/비용을 그대로 사용해
		// 쉐도우 웨폰/기타 쉐도우 장비의 차이를 자동 반영한다.
		.@safe_refine = 4;
		.@material_id = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
		.@zeny_per_refine = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);

		if (.@material_id <= 0) {
			mes needtr("[홀그렌]", "[Hollgrehenn]");
			mes needtr("이 쉐도우 장비의 제련 재료 정보를 확인할 수 없습니다.", "I cannot read the refine material data for this Shadow gear.");
			mes needtr("refine DB 설정을 확인해주세요.", "Please check the refine DB settings.");
			close;
		}
	}
	else {
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes needtr("안전 제련을 지원하지 않는 장비입니다.", "This equipment does not support safe refining.");
		close;
	}

	if (.@current_refine >= .@safe_refine) {
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes needtr("이 장비는 이미 안전 제련 한계에 도달했습니다.", "This equipment has already reached the safe refine limit.");
		mes needtr("현재 제련도: ^0055FF+", "Current refine: ^0055FF+") + .@current_refine + "^000000";
		close;
	}

	.@refine_count = .@safe_refine - .@current_refine;
	.@total_zeny = .@zeny_per_refine * .@refine_count;

	// 진행 전 장비 상태 저장
	.@saved_item_id = .@item_id;
	.@saved_refine = .@current_refine;
	setarray .@saved_card[0],
		getequipcardid(.@part, 0),
		getequipcardid(.@part, 1),
		getequipcardid(.@part, 2),
		getequipcardid(.@part, 3);

	mes needtr("[홀그렌]", "[Hollgrehenn]");
	mes needtr("선택 장비: ^0055FF", "Selected: ^0055FF") + getequipname(.@part) + "^000000";
	mes needtr("현재 제련도: ^FF8000+", "Current refine: ^FF8000+") + .@current_refine + "^000000";
	mes needtr("목표 제련도: ^00AAFF+", "Target refine: ^00AAFF+") + .@safe_refine + "^000000";
	mes " ";
	mes needtr("필요 제련 횟수: ^FF8000", "Refine attempts needed: ^FF8000") + .@refine_count + needtr("회^000000", "x^000000");
	mes needtr("필요 재료: ^FF8000", "Materials needed: ^FF8000") + getitemname(.@material_id) + " " + .@refine_count + needtr("개^000000", " ea^000000");
	mes needtr("필요 제니: ^FF8000", "Zeny needed: ^FF8000") + callfunc("F_InsertComma", .@total_zeny) + needtr(" 제니^000000", " zeny^000000");
	mes " ";
	mes needtr("^00AA00실패 없이 안전 한계까지 한 번에 제련됩니다.^000000", "^00AA00Refines up to the safe limit at once with no chance of failure.^000000");
	next;

	if (select(needtr("안전 제련한다:취소", "Safe refine it:Cancel")) == 2)
		close;

	// 결제 및 제련 직전 재검사
	if (!getequipisequiped(.@part) || !getequipisenableref(.@part)) {
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes needtr("장비의 착용 상태가 변경되었습니다.", "The equipped state of the item has changed.");
		close;
	}

	if (callfunc("F_IsEquipIDHack", .@part, .@saved_item_id) ||
		callfunc("F_IsEquipRefineHack", .@part, .@saved_refine) ||
		callfunc("F_IsEquipCardHack", .@part,
			.@saved_card[0], .@saved_card[1], .@saved_card[2], .@saved_card[3])) {
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		emotion ET_FRET;
		mes needtr("잠시만요...", "Wait a second...");
		mes needtr("제련을 진행하는 사이 장비가 변경되었습니다.", "The equipment changed while the refining was in progress.");
		mes needtr("처음부터 다시 시도해주세요.", "Please start over.");
		close;
	}

	if (countitem(.@material_id) < .@refine_count) {
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes needtr("제련 재료가 부족합니다.", "You do not have enough refine materials.");
		mes needtr("필요: ^FF0000", "Needed: ^FF0000") + getitemname(.@material_id) + " " + .@refine_count + needtr("개^000000", " ea^000000");
		mes needtr("보유: ^0055FF", "You have: ^0055FF") + countitem(.@material_id) + needtr("개^000000", " ea^000000");
		close;
	}

	if (Zeny < .@total_zeny) {
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes needtr("제련 비용이 부족합니다.", "You do not have enough zeny for the refine fee.");
		mes needtr("필요: ^FF0000", "Needed: ^FF0000") + callfunc("F_InsertComma", .@total_zeny) + needtr(" 제니^000000", " zeny^000000");
		mes needtr("보유: ^0055FF", "You have: ^0055FF") + callfunc("F_InsertComma", Zeny) + needtr(" 제니^000000", " zeny^000000");
		close;
	}

	delitem .@material_id, .@refine_count;
	Zeny = Zeny - .@total_zeny;

	specialeffect EF_HOLYHIT;
	successrefitem .@part, .@refine_count;

	next;
	emotion ET_DELIGHT;
	mes needtr("[홀그렌]", "[Hollgrehenn]");
	mes needtr("안전 제련이 완료되었습니다!", "Safe refining complete!");
	mes "^0055FF" + getequipname(.@part) + "^000000";
	mes needtr("제련 결과: ^00AAFF+", "Result: ^00AAFF+") + getequiprefinerycnt(.@part) + "^000000";
	close;
}

// 제련권 사용 시 목표 제련 수치별 제니 비용 설정
// 예: +9 제련권을 사용하면 F_NeedRefineTicketFee(9)의 값을 비용으로 받습니다.
// 비용을 받지 않을 제련 수치는 0으로 두면 됩니다.
function	script	F_NeedRefineTicketFee	{
	.@refine_lv = getarg(0);

	setarray .@fee[0],
		0, // +0
		0, // +1
		0, // +2
		0, // +3
		0, // +4
		1000000, // +5
		2000000, // +6
		4000000, // +7
		8000000, // +8
		12000000, // +9
		24000000, // +10
		48000000, // +11
		96000000, // +12
		200000000, // +13
		400000000, // +14
		800000000, // +15
		0, // +16
		0, // +17
		0, // +18
		0, // +19
		0; // +20

	if (.@refine_lv < 0 || .@refine_lv >= getarraysize(.@fee))
		return 0;

	return .@fee[.@refine_lv];
}

		6229,	// Guarantee_Weapon_8Up
		6230,	// Guarantee_Weapon_7Up
		6231,	// Guarantee_Weapon_6Up
		6456;	// Guarantee_Weapon_5Up
	setarray .@cert_armor[0],
		6865,	// Guarantee_Armor_19Up
		6881,	// Guarantee_Armor_18Up
		6880,	// Guarantee_Armor_17Up
		6879,	// Guarantee_Armor_16Up
		6878,	// Guarantee_Armor_15Up
		6877,	// Guarantee_Armor_14Up
		6876,	// Guarantee_Armor_13Up
		6585,	// Guarantee_Armor_12Up
		6239,	// Guarantee_Armor_11Up
		6232,	// Guarantee_Armor_9Up
		6233,	// Guarantee_Armor_8Up
		6234,	// Guarantee_Armor_7Up
		6235,	// Guarantee_Armor_6Up
		6457;	// Guarantee_Armor_5Up
	setarray .@cert_level[0],19,18,17,16,15,14,13,12,11,9,8,7,6,5;
	.@size_cert = getarraysize(.@cert_weapon);

	// 초심자 무기 7제련 이용권(25428) 전용 allowlist
	// 399908 Select_Rebeginer_Box 의 구성 16종 (db/re/item_packages.yml)
	// 이 목록에 있는 무기에만 25428 을 사용할 수 있다.
	setarray .@rebeginer_w[0],
		1848,	// 초심자 수라의 너클
		2046,	// 초심자 워록의 스태프
		13144,	// 초심자 리벨리온의 리볼버
		13341,	// 초심자 카게로우&오보로의 풍마수리검
		13483,	// 초심자 제네릭의 소드
		16075,	// 초심자 미케닉의 메이스
		16076,	// 초심자 슈퍼노비스의 메이스
		18165,	// 초심자 레인저의 활
		18166,	// 초심자 원더러&민스트럴의 활

		28027,	// 초심자 길로틴크로스의 카타르
		28616,	// 초심자 아크비숍의 성서
		28740;	// 초심자 쉐도우체이서의 대거
	.@rebeginer_ticket = 25428;

	for ( .@i = 0; .@i < .@size_cert; ++.@i ) {
		if (countitem(.@cert_weapon[.@i]) > 0 || countitem(.@cert_armor[.@i]) > 0) {
			.@check = 1;
			break;
		}
	}

	// 25428 만 가진 경우에도 제련권 사용 메뉴에 들어올 수 있어야 한다.
	if (countitem(.@rebeginer_ticket) > 0)
		.@check = 1;

	if (.@check == 0) {
			mes needtr("[홀그렌]", "[Hollgrehenn]");
			mes needtr("사실, 제가 제련을 해 드리는 경우가 있기는 합니다만, 제련 보장권티켓을 가진 분들이 ^006400장착할 수 있는 장비^000000만을 제련해 드리죠.", "Well, I do refine things now and then, but only ^006400equipment you can actually wear^000000, and only for people holding a refine ticket.");
			mes needtr("그럼 안녕히~!", "So long~!");
			close;
	}
	emotion ET_SURPRISE;
	mes needtr("[홀그렌]", "[Hollgrehenn]");
	mes needtr("^006400당신의 티켓과 같은 레벨로^000000 아이템을 제련 할수 있습니다.", "I can refine an item ^006400up to the level printed on your ticket^000000.");
	mes needtr("걱정하지 마세요!. 아이템이 파괴 될 가능성은 없습니다.", "Don't worry! There is no chance of the item breaking.");
	next;
	if(select(needtr("다시오겠습니다.:티켓으로 제련을 한다.", "I'll come back later.:Refine with a ticket.")) == 1) {
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes needtr("오케이.", "Okay.");
		mes needtr("다음에 다시 오세요.", "Come again next time.");
		close;
	}
	mes needtr("[홀그렌]", "[Hollgrehenn]");
	mes needtr("제련하고 싶은 장비는 무엇인가요?", "Which equipment would you like to refine?");
	next;
	setarray .@position$[1],needtr("상단", "Upper"),needtr("갑옷", "Armor"),needtr("왼손", "Left Hand"),needtr("오른손", "Right Hand"),needtr("걸칠것", "Garment"),needtr("신발", "Shoes"),needtr("악세사리(L)", "Accessory (L)"),needtr("악세사리(R)", "Accessory (R)"),needtr("중단", "Middle"),needtr("하단", "Lower");
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ += (getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : .@position$[.@i] + needtr("- [없음]", " - [None]")) + ":";

	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes needtr("제련할려는 장비는 장착을해야 합니다..", "The equipment you want to refine has to be equipped.");
		close;
	}
	if (!getequipisenableref(.@part)) {
		emotion ET_OTL;
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes needtr("죄송합니다..", "refine this item at all...");
		mes needtr("이아이템은 제련을 할 수 없습니다.", "This item cannot be refined.");
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	if( .@itemtype == IT_WEAPON ){
		switch( getequipweaponlv( .@part ) ){
			case 1:
			case 2:
			case 3:
			case 4:
				copyarray .@tickets[0], .@cert_weapon[0], .@size_cert;
				copyarray .@tlevel[0], .@cert_level[0], .@size_cert;
				.@size_list = .@size_cert;
				// 초심자 무기 allowlist 에 있는 무기에만 25428 을 목록에 넣는다.
				if (inarray(.@rebeginer_w[0], .@refineitemid) >= 0 &&
					countitem(.@rebeginer_ticket) > 0) {
					.@tickets[.@size_list] = .@rebeginer_ticket;
					.@tlevel[.@size_list] = 7;
					.@size_list += 1;
				}
				.@type$ = "Weapon";
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_ARMOR ){
		switch( getequiparmorlv( .@part ) ){
			case 1:
				copyarray .@tickets[0], .@cert_armor[0], .@size_cert;
				copyarray .@tlevel[0], .@cert_level[0], .@size_cert;
				// 방어구에는 25428 을 절대 넣지 않는다.
				.@size_list = .@size_cert;
				.@type$ = "Armor";
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}
	.@check = 0;
	for ( .@i = 0; .@i < .@size_list; ++.@i ) {
		if (countitem(.@tickets[.@i]) > 0) {
			.@check = 1;
			break;
		}
	}
	if (.@check == 0) {
		// 25428 만 가지고 일반 장비를 고른 경우를 따로 안내한다.
		if (countitem(.@rebeginer_ticket) > 0) {
			emotion ET_THINK;
			mes needtr("[홀그렌]", "[Hollgrehenn]");
			mes needtr("보유 중인 ^006400초심자 무기 제련권^000000은", "The ^006400Beginner Weapon Refine Ticket^000000 you are holding");
			mes needtr("이 장비에 사용할 수 없습니다.", "cannot be used on this equipment.");
			mes "";
			mes needtr("초심자 무기를 장착한 뒤", "Please equip a beginner weapon");
			mes needtr("다시 이용해주세요.", "and come back.");
			close;
		}
		emotion ET_THINK;
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes "If you want to refine this ^006400"+.@type$+"^000000, please come along with ^006400"+.@type$+" Refine Ticket^000000.";
		mes "See you later!";
		close;
	}
	mes needtr("[홀그렌]", "[Hollgrehenn]");
	mes needtr("이 ^006400", "To refine this ^006400")+.@type$+needtr("^000000을 제련하려면 ^006400", "^000000 you need a ^006400")+.@type$+needtr(" 제련권^000000이 필요합니다.", " Refine Ticket^000000.");
	next;
	.@menu$ = "";
	for ( .@i = 0; .@i < .@size_list; ++.@i )
		.@menu$ += getitemname(.@tickets[.@i]) + ":";
	.@select = select(.@menu$)-1;
	.@ticket_lv = .@tlevel[.@select];
	.@ticket_id = .@tickets[.@select];
	// 25428 은 성장지원 전용 제련권이다. 아이템 DB / 기존 스크립트 어디에도
	// 사용 비용 근거가 없으므로 0 제니로 두고, 일반 +7 제련권 비용을 쓰지 않는다.
	if (.@ticket_id == .@rebeginer_ticket)
		.@ticket_fee = 0;
	else
		.@ticket_fee = callfunc("F_NeedRefineTicketFee", .@ticket_lv);

	if (countitem(.@ticket_id) == 0) {
		emotion ET_QUESTION;
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes getitemname(.@ticket_id)+needtr(" 이 인벤토리에 없습니다. 창고에 있는거 아닌가요?", " is not in your inventory. Is it in your storage?");
		mes needtr("다시 한번 확인해주세요.", "Please check again.");
		mes needtr("다음에 다시 방문해주세요!", "Please visit me again!");
		close;
	}
	if (getequiprefinerycnt(.@part) >= .@ticket_lv) {
		emotion ET_PROFUSELY_SWEAT;
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		if (.@ticket_id == .@rebeginer_ticket) {
			mes needtr("이 초심자 무기는 이미 ^006400+7 이상^000000입니다.", "This beginner weapon is already ^006400+7 or higher^000000.");
			mes needtr("해당 제련권을 사용할 수 없습니다.", "That refine ticket cannot be used.");
		}
		else {
			mes needtr("^8B4513이 장비는 제련보장권 만큼 이미 제련이 되었습니다.^000000", "^8B4513This equipment is already refined as high as the ticket allows.^000000");
			mes needtr("제련보장권 이하의 제련된 장비를 가져오세요.", "Please bring equipment refined below the ticket's level.");
		}
		close;
	}
	mes needtr("[홀그렌]", "[Hollgrehenn]");
	mes "^006400"+getequipname(.@part)+needtr("^8B4513을(를) ^006400", "^8B4513 will be refined with ^006400")+getitemname(.@ticket_id)+needtr("^000000으로 제련합니다.", "^000000.");
	mes needtr("목표 제련 수치: ^006400+", "Target refine level: ^006400+")+.@ticket_lv+"^000000";
	mes needtr("필요 비용: ^FF0000", "Cost: ^FF0000")+callfunc("F_InsertComma", .@ticket_fee)+needtr(" 제니^000000", " zeny^000000");
	mes needtr("진행해도 될까요??", "Shall I go ahead?");
	next;
	if(select(needtr("아뇨.:예.", "No.:Yes.")) == 1) {
		emotion ET_THINK;
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes needtr("아, 마음을 바꾸었군요.", "Ah, you changed your mind.");
		mes needtr("오케이.", "Okay.");
		mes needtr("다음에 다시 오세요.", "Come again next time.");
		close;
	}
	if (Zeny < .@ticket_fee) {
		emotion ET_MONEY;
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes needtr("제련 비용이 부족합니다.", "You do not have enough zeny for the refine fee.");
		mes needtr("필요 비용: ^FF0000", "Cost: ^FF0000")+callfunc("F_InsertComma",.@ticket_fee)+needtr(" 제니^000000", " zeny^000000");
		close;
	}
	mes needtr("[홀그렌]", "[Hollgrehenn]");
	mes needtr("좋습니다.", "Very well.");
	mes "As you wish!";
	mes needtr("나만의 특별한 제련 방법이 있습니다.", "I have my own special way of refining.");
	specialeffect EF_SUI_EXPLOSION;
	if (countitem(.@ticket_id) < 1) {
		next;
		mes needtr("에러!", "Error!");
		mes needtr("이것을 보고헤주세요.", "Take a look at this.");
		close;
	}
	if (Zeny < .@ticket_fee) {
		next;
		mes needtr("에러!", "Error!");
		mes needtr("제련 비용이 부족합니다.", "You do not have enough zeny for the refine fee.");
		close;
	}
	// 25428 전용 경로는 티켓 / 제니를 소비하기 전에 모든 조건을 다시 본다.
	// (장비 교체 / 카드 변경 / 제련도 변경 / 장비 해제 / 티켓 소실)
	if (.@ticket_id == .@rebeginer_ticket) {
		if (!getequipisequiped(.@part) || !getequipisenableref(.@part) ||
			inarray(.@rebeginer_w[0], getequipid(.@part)) < 0 ||
			getequiprefinerycnt(.@part) >= 7 ||
			countitem(.@rebeginer_ticket) < 1 ||
			callfunc("F_IsEquipIDHack", .@part, .@refineitemid) ||
			callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) ||
			callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
			emotion ET_FRET;
			next;
			mes needtr("[홀그렌]", "[Hollgrehenn]");
			mes needtr("장비 상태가 처음과 달라졌습니다.", "The equipment is not in the state it started in.");
			mes needtr("제련권을 사용하지 않고 중단합니다.", "Stopping without using the refine ticket.");
			close;
		}
	}

	delitem .@ticket_id,1;
	Zeny = Zeny - .@ticket_fee;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		emotion ET_FRET;
		mes needtr("잠시만요...", "Wait a second...");
		mes needtr("내가 멍청하다고 생각하나요?!", "Do you think I'm a fool?!");
		mes needtr("내가 보지 않는사이에 아이템을 바꾸다니! 여기서 나가!", "Swapping the item while I wasn't looking! Get out of here!");
		close;
	}
	successrefitem .@part, .@ticket_lv - getequiprefinerycnt(.@part);
	next;
	emotion ET_DELIGHT;
	if (.@ticket_id == .@rebeginer_ticket) {
		mes needtr("[홀그렌]", "[Hollgrehenn]");
		mes needtr("초심자 무기 제련이 완료되었습니다.", "Beginner weapon refining complete.");
		mes "";
		mes needtr("대상 무기 : ^006400", "Target weapon: ^006400") + getequipname(.@part) + "^000000";
		mes needtr("제련 결과 : ^006400+7^000000", "Result: ^006400+7^000000");
		close;
	}
	mes needtr("[홀그렌]", "[Hollgrehenn]");
	mes needtr("좋습니다, 여기 있습니다~", "Alright, here you go~");
	mes "^0000FF"+strcharinfo(0)+needtr("^000000님!", "^000000!");
	mes needtr("축하합니다.", "Congratulations, your ")+.@type$+needtr("가 되었습니다.", " is refined.");
	mes needtr("멋지게 생겼군요!", "Cry Hammer!! Cry!!!");
	mes needtr("안녕히 가세요~!", "Goodbye~!");
	close;
}
